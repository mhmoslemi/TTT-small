def coordinate_descent(a0, deadline, n_rounds=500):
    """Coordinate descent: optimize one coordinate at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    rng = np.random.default_rng(42)

        if time.monotonic() > deadline:
            break
        
        improved = False
        # Random order
        order = rng.permutation(n)

            if time.monotonic() > deadline:
                break
            
            a_old = a[j]
            # Try to optimize a[j] while keeping others fixed
            # J as function of a[j]: 2n * max_k (b_k(a)) / (sum(a))^2
            # b_k depends on a[j] quadratically for some k
            # Simple approach: try a few values of a[j]
            
            # The contribution of a[j] to b_k:
            # b_k includes terms a[j]*a[k-j] and a[k-j]*a[j] (if k-j != j)
            # or a[j]^2 (if k-j == j, i.e., k = 2j)
            # So b_k(a) = b_k_base + a[j] * (a[k-j] + a[j-(k-j)] ... ) + a[j]^2 * [k=2j]
            
            # Simpler: just try a few values
            candidates = [0.0, a_old * 0.5, a_old * 0.8, a_old * 1.0, a_old * 1.2, a_old * 1.5,
                         a_old + 0.01, a_old - 0.01]
            candidates = [max(0.0, c) for c in candidates]
            
            best_val = a_old
            best_J_c = best_J
            
            for cv in candidates:
                a[j] = cv
                s = a.sum()
                if s < 0.01:
                    continue
                J_c = compute_J(a)
                if J_c < best_J_c:
                    best_J_c = J_c
                    best_val = cv

            if J_new < best_J - 1e-12:
                best_J = J_new
                improved = True

            break

Given a[0..n-1] with a[j] = t (variable), all other a[i] fixed:
- sum(a) = S + t where S = sum(a) - a[j]
- b_k = b_k_base + t * c_k + t^2 * d_k
  where c_k = a[(k-j) mod ...] terms and d_k = 1 if k = 2j
  
More precisely:
b_k = sum_i a_i * a_{k-i}
= sum_{i≠j} a_i * a_{k-i} + a_j * a_{k-j} + a_{k-j} * a_j - [if k-j=j: a_j * a_j counted twice]

Let me define:
- For i ≠ j and k-i ≠ j: a_i * a_{k-i} (constant)
- For i = j, k-i ≠ j: a_j * a_{k-j} = t * a_{k-j}
- For i ≠ j, k-i = j: a_i * a_{k-i} = a_{k-j} * t (same as above, just different i)
- For i = j, k-i = j (i.e., k = 2j): a_j * a_j = t^2

So: b_k = B_k + t * C_k + t^2 * D_k

- B_k = sum_{i≠j, k-i≠j} a_i * a_{k-i}

  Actually: C_k = a_{k-j} (contribution from i=j) + a_{k-j} (contribution from k-i=j, i=k-j)
  = 2*a_{k-j} if k-j ≠ j, else a_{k-j} = a_j (when k=2j)
  
  Hmm wait, when k=2j: the term a_j * a_{k-j} = a_j * a_j = t^2, not t * a_j.
  So C_k should only count the "linear" terms.
  
  Let me redo:
  b_k = sum_i a_i * a_{k-i}
  
  Terms involving t = a[j]:
  1. i=j: a[j] * a[k-j] = t * a[k-j] (valid if 0≤k-j<n)
  2. k-i=j, i=k-j: a[k-j] * a[j] = a[k-j] * t (valid if 0≤k-j<n)
  
  If k-j ≠ j (k ≠ 2j): both terms are present and distinct, giving 2*t*a[k-j]
  If k-j = j (k = 2j): both terms are the same (i=j=k-j), giving t*t = t^2

          = B_k + t^2 (if k=2j)
          = B_k (if k-j not in [0,n))

Now, J(t) = 2n * max_k (B_k + 2*t*a[k-j] or B_k + t^2) / (S+t)^2

To minimize J(t), I need to find t ≥ 0 that minimizes this. This is a 1D optimization problem! I can use golden section search or just evaluate at a few key points.

The key points are where the max changes (i.e., where two b_k values cross). But that's hard to find analytically. Instead, I can:
1. Evaluate J(t) at t=0, t=a_old, and a few other points
2. Use ternary search on [0, t_max] for a more precise answer

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast analytic coordinate descent (per-coordinate 1D golden-section search)
2. Soft-max + L-BFGS-B gradient refinement with aggressive annealing
3. Subgradient method with momentum
4. Multi-length search (256-1024) with resampling
5. Diverse initialization from parent + structured sequences

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def soft_max_b(b, T):
    """Soft max value of b with temperature T."""
    Tb = T * b
    m = Tb.max()
    exp_Tb = np.exp(Tb - m)
    Z = exp_Tb.sum()
    return (m + np.log(Z)) / T


def soft_max_weights(b, T):
    """Soft max weights (normalized softmax of T*b)."""
    Tb = T * b
    m = Tb.max()
    w = np.exp(Tb - m)
    return w / w.sum()


def weighted_subgrad(a, w):
    """
    Compute g_j = 2 * sum_k w[k]*a[k-j] for valid k-j.
    w has length 2n-1, a has length n. Result has length n.
    Uses FFT-based convolution with reversed w.
    """
    n = len(a)
    size = 1
    while size < 3 * n:
        size <<= 1
    w_rev = w[::-1]
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)
    # c(j) = conv(2n-2-j) for j=0,...,n-1
    c = conv[n - 1:2 * n - 1][::-1]
    return 2.0 * c


def smooth_obj_and_grad(x, T, n):
    """Objective and gradient for L-BFGS-B."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    b = fast_conv(a)
    val = soft_max_b(b, T)
    w = soft_max_weights(b, T)
    obj = 2.0 * n * val / (s * s)
    g_sub = weighted_subgrad(a, w)
    grad = (2.0 * n / (s * s)) * g_sub - (4.0 * n * val / (s * s * s)) * np.ones(n)
    return obj, grad


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = 1.0 - u * np.arange(1, n + 1) > cssv
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coordinate_descent(a0, deadline, max_rounds=200):
    """
    Fast analytic coordinate descent. For each coordinate j,
    solve a 1D optimization: minimize max_k(B_k + C_k*t + D_k*t^2)/(S+t)^2.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    rng = np.random.default_rng(7)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        improved = False

        for j in order:
            if time.monotonic() > deadline:
                break

            a_j = a[j]
            if a_j < 1e-15:
                # Try setting to small positive
                a[j] = a.mean() * 0.1
                J_try = compute_J(a)
                if J_try < best_J - 1e-12:
                    best_J = J_try
                    improved = True
                else:
                    a[j] = 0.0
                continue

            S = a.sum() - a_j
            b_full = fast_conv(a)

            # Compute B, C, D arrays
            B = b_full.copy()
            C = np.zeros(2 * n - 1)
            D = np.zeros(2 * n - 1)

            k_min = max(0, j)
            k_max = min(2 * n - 2, j + n - 1)
            for k in range(k_min, k_max + 1):
                m = k - j
                if m < 0 or m >= n:
                    continue
                if k == 2 * j:
                    D[k] = 1.0
                    B[k] = b_full[k] - a_j * a_j
                else:
                    C[k] = 2.0 * a[m]
                    B[k] = b_full[k] - 2.0 * a_j * a[m]

            # Minimize f(t) = max_k(B_k + C_k*t + D_k*t^2) / (S+t)^2 for t >= 0
            # Use golden section search on [0, t_hi]

            def f_of_t(t):
                vals = B + C * t + D * t * t
                return vals.max() / (S + t) ** 2

            # Find upper bound
            t_hi = max(a_j * 10.0, 0.1, 1.0 / n)
            while f_of_t(t_hi) < f_of_t(t_hi * 2) and t_hi < 100.0:
                t_hi *= 2

            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            t1 = t_hi * (1.0 - gr)
            t2 = t_hi * gr
            f1 = f_of_t(t1)
            f2 = f_of_t(t2)

            for _ in range(40):
                if f1 < f2:
                    t_hi = t2
                    t2 = t1
                    f2 = f1
                    t1 = t_hi * (1.0 - gr)
                    f1 = f_of_t(t1)
                else:
                    t_lo = t1
                    t1 = t2
                    f1 = f2
                    t2 = t_lo + gr * (t_hi - t_lo)
                    f2 = f_of_t(t2)

            t_opt = (t1 + t2) / 2.0
            # Also check t=0
            f0 = f_of_t(0.0)
            f_opt = f_of_t(t_opt)
            if f0 < f_opt:
                t_opt = 0.0
                f_opt = f0

            a[j] = t_opt
            improved = True

        # Renormalize
        s = a.sum()
        if s > 0.01:
            a = a / s
            J_new = compute_J(a)
            if J_new < best_J - 1e-14:
                best_J = J_new

    return a, best_J

def subgrad_method(a0, deadline, max_iters=100000):
    """Subgradient method with momentum on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    x = a.copy()
    v = np.zeros_like(a)
    best_max = float(fast_conv(x).max())
    best_a = x.copy()

    step0 = 1.0 / np.sqrt(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, it / 50000.0)
        T = 80.0 * (1.0 / 20.0) ** frac
        momentum = 0.5 * (1.0 - np.exp(-it / 2000.0))
        step = step0 / (1.0 + 0.005 * it)

        b = fast_conv(x)
        M = float(b.max())
        w = soft_max_weights(b, T)
        g = weighted_subgrad(x, w)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        v = momentum * v + (1.0 - momentum) * g
        x_new = x - step * v
        x_new = project_simplex(x_new)
        x = x_new

        if it % 100 == 0:
            M_check = float(fast_conv(x).max())
            if M_check < best_max - 1e-12:
                best_max = M_check
                best_a = x.copy()
                if it % 1000 == 0:
                    print(f"      subgrad iter={it}: J={2*n*best_max:.8f}", flush=True)

    return best_a, float(2 * n * best_max)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_zadoff_chu(n, rng):
    """Smoothed Zadoff-Chu inspired sequence."""
    t = np.linspace(0, 1, n, endpoint=False)
    # Complex Zadoff-Chu: exp(-i*pi*n^2/N)
    # Take real part, shift to [0,1]
    phase = np.pi * (t * n) ** 2 / n
    seq = 0.5 + 0.5 * np.cos(phase + rng.uniform(0, 2 * np.pi))
    return np.maximum(seq, 0.0)


def make_m_sequence_smoothed(n, rng):
    """Smoothed m-sequence inspired."""
    # Generate a pseudo-random binary sequence using LFSR
    L = 8  # LFSR length
    state = rng.integers(1, 2 ** L - 1)
    poly = 0x49  # primitive polynomial for L=8
    bits = np.zeros(n, dtype=np.float64)
    for i in range(n):
        bits[i] = float(state & 1) * 2.0 - 1.0
        # LFSR step
        lsb = (state ^ (state >> 1) ^ (state >> 4) ^ (state >> 6) ^ (state >> 7)) & 1
        state = (state >> 1) | (lsb << (L - 1))
    # Smooth with Gaussian
    kernel_size = max(3, n // 64)
    kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1) / (kernel_size // 3)) ** 2)
    kernel /= kernel.sum()
    smoothed = np.convolve(bits, kernel, mode='same')
    return np.maximum((smoothed - smoothed.min()) / (smoothed.max() - smoothed.min() + 1e-10), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))

    # Zadoff-Chu variants
    for _ in range(3):
        inits.append(make_zadoff_chu(n, rng))

    # M-sequence
    inits.append(make_m_sequence_smoothed(n, rng))
    inits.append(make_m_sequence_smoothed(n, rng))

    # Sinusoidal
    for freq in [2, 3, 5, 8, 13]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase))

    # Random
    inits.append(rng.uniform(0.1, 1.0, n))

    # Gaussian-shaped
    inits.append(0.3 + 0.7 * np.exp(-4.0 * (t - 0.5) ** 2))

    # Two-peak
    inits.append(0.2 + 0.6 * np.exp(-8 * (t - 0.3) ** 2) + 0.6 * np.exp(-8 * (t - 0.7) ** 2))

    # Sparse + smooth
    sp = np.zeros(n)
    n_sp = max(3, n // 16)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sp, kernel, 'same') + 0.05)

    # Sine with noise
    inits.append(0.5 + 0.3 * np.sin(6 * np.pi * t) + 0.1 * rng.standard_normal(n))

    # Power law
    inits.append(0.3 + 0.7 * (t + 0.1) ** rng.uniform(-0.5, 0.5))

    return inits

            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic() - self.t0:.0f}s", flush=True)


def refine_sequence(a, deadline):
    """
    Full refinement pipeline for a single sequence:
    coordinate descent -> smooth refine -> subgrad -> coordinate -> smooth
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    # Phase A: Coordinate descent (fast, many rounds)
    cd_deadline = min(deadline, time.monotonic() + 60.0)
    a1, J1 = coordinate_descent(best_a, cd_deadline, max_rounds=150)
    if J1 < best_J:
        best_J = J1
        best_a = a1.copy()
        print(f"    coord: J={best_J:.8f}", flush=True)

    # Phase B: Smooth refine
    sr_deadline = min(deadline, time.monotonic() + 120.0)
    a2, J2 = smooth_refine(best_a, sr_deadline)
    if J2 < best_J:
        best_J = J2
        best_a = a2.copy()
        print(f"    smooth: J={best_J:.8f}", flush=True)

    # Phase C: More coordinate descent
    cd2_deadline = min(deadline, time.monotonic() + 40.0)
    a3, J3 = coordinate_descent(best_a, cd2_deadline, max_rounds=100)
    if J3 < best_J:
        best_J = J3
        best_a = a3.copy()
        print(f"    coord2: J={best_J:.8f}", flush=True)

    # Phase D: Final smooth with lower temps
    if time.monotonic() < deadline - 10:
        sr_deadline2 = min(deadline, time.monotonic() + 80.0)
        a4, J4 = smooth_refine(best_a, sr_deadline2,
                               T_list=[15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 2.0, 1.5, 1.2, 1.0, 0.8, 0.6, 0.5])
        if J4 < best_J:
            best_J = J4
            best_a = a4.copy()
            print(f"    smooth2: J={best_J:.8f}", flush=True)

    # Phase E: Final coordinate polish
    if time.monotonic() < deadline - 5:
        cd3_deadline = min(deadline, time.monotonic() + 20.0)
        a5, J5 = coordinate_descent(best_a, cd3_deadline, max_rounds=50)
        if J5 < best_J:
            best_J = J5
            best_a = a5.copy()
            print(f"    coord3: J={best_J:.8f}", flush=True)

    return best_a, best_J


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: multi-length coordinate descent + smooth refine + subgradient.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 20.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, TypeError):
        pass

    # Multi-length search
    # Allocate time based on length
    lengths = [384, 512, 640, 768]
    # Give more time to lengths closer to parent
    if parent is not None:
        pn = parent.size
        lengths = sorted(lengths, key=lambda x: abs(x - pn))
        # Put parent length first, then nearby
        if pn not in lengths:
            lengths = [pn] + lengths

    # Remove duplicates, keep unique
    seen = set()
    unique_lengths = []
    for ln in lengths:
        if ln not in seen and 64 <= ln <= 1200:
            seen.add(ln)
            unique_lengths.append(ln)
    lengths = unique_lengths[:5]  # At most 5 lengths

    print(f"Search lengths: {lengths}", flush=True)

    # Allocate time budget
    n_lengths = len(lengths)
    base_time = (total_budget - 20.0) / n_lengths

    for li, n in enumerate(lengths):
        if time.monotonic() > deadline - 30:
            break

        t0 = time.monotonic()
        print(f"\n{'=' * 60}", flush=True)
        print(f"Length n={n} (budget ~{base_time:.0f}s)", flush=True)
        print(f"{'=' * 60}", flush=True)

        # Time allocation for this length
        length_deadline = min(deadline, t0 + base_time * 1.2)

        # Generate inits (limit to 8 to save time)
        inits = make_inits(n, rng, parent)
        inits = inits[:8]

        # For each init, run refinement
        for i, a0 in enumerate(inits):
            if time.monotonic() > length_deadline - 20:
                break

            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 = a0 / a0.sum()

            init_J = compute_J(a0)
            print(f"  init[{i}] n={n}: J={init_J:.8f}", flush=True)

            # Allocate remaining time for this init
            init_deadline = min(length_deadline - 5, time.monotonic() + base_time * 1.0 / len(inits) * 1.5)

            a_best, J_best = refine_sequence(a0, init_deadline)
            tracker.update(a_best)

        # After all inits, do one more round on the best found so far at this length
        if tracker.best is not None and tracker.best.size == n and time.monotonic() < length_deadline - 10:
            print(f"  Extra refinement on best n={n}...", flush=True)
            a_ex, J_ex = refine_sequence(tracker.best, min(deadline, time.monotonic() + 60.0))
            tracker.update(a_ex)

        elapsed = time.monotonic() - t0
        tracker.report(f"n={n} done ({elapsed:.0f}s)")

    # Final global refinement on the overall best
    if tracker.best is not None and time.monotonic() < deadline - 10:
        print(f"\n{'=' * 60}", flush=True)
        print(f"Final global refinement (n={tracker.best.size})", flush=True)
        print(f"{'=' * 60}", flush=True)
        a_final, J_final = refine_sequence(tracker.best, deadline)
        tracker.update(a_final)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128) / 128.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t0:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
