This means that at the optimum, the "effective pressure" on each coordinate is either zero (if g*_i > 0) or positive (if g*_i = 0).

This is useful for understanding the structure of the solution.

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Anderson acceleration on the LP fixed-point map a -> LP_response(a),
with multi-length exploration, diverse restarts, and smoothed GD polish.
Key idea: The optimal sequence is a fixed point of the LP best-response map.
Anderson acceleration (DIIS) dramatically speeds up convergence to this fixed point.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv_fft(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_pair(a, b):
    """Linear convolution of a and b via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fft(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=verbose)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_response(a):
    """Compute LP best response T(a). Returns normalized g or a itself if LP fails."""
    g, M = solve_lp(a)
    if g is not None:
        return g, M
    return a.copy(), np.inf


def anderson_fixed_point(a_init, T_func, memory=6, max_iter=50, 
                         callback=None, time_limit=600.0, verbose=True):
    """
    Anderson acceleration for the fixed point problem a = T(a).
    T_func(a) returns (T(a), extra_info).
    
    Uses DIIS (direct inversion in the iterative subspace) formulation.
    """
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    
    # History
    X_hist = []  # a values
    G_hist = []  # residuals a - T(a)
    T_hist = []  # T(a) values
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iter):
        if time.monotonic() - start > time_limit:
            break
        
        T_a, extra = T_func(a)
        residual = a - T_a
        res_norm = np.linalg.norm(residual)
        
        if verbose:
            obj = compute_obj(a)
            print(f"    Anderson it{it}: res={res_norm:.6f}, obj={obj:.8f}, t={time.monotonic()-start:.1f}s")
        
        X_hist.append(a.copy())
        G_hist.append(residual.copy())
        T_hist.append(T_a.copy())
        
        # Keep only last (memory+1) entries
        if len(X_hist) > memory + 1:
            X_hist = X_hist[-(memory+1):]
            G_hist = G_hist[-(memory+1):]
            T_hist = T_hist[-(memory+1):]
        
        # Compute Anderson coefficients
        m = len(X_hist) - 1  # number of residual vectors to use (at least 1)
        
        if m >= 2:
            # Form the m x m matrix H where H[i,j] = G_hist[-m+i] @ G_hist[-m+j]
            G_mat = np.array(G_hist[-m:])  # m x n
            H = G_mat @ G_mat.T  # m x m
            
            # Solve H @ beta = ones(m) (with regularization)
            H_reg = H + 1e-12 * np.eye(m)
            try:
                beta = np.linalg.solve(H_reg, np.ones(m))
            except np.linalg.LinAlgError:
                beta = np.ones(m) / m
            
            # Ensure beta sums to 1 (it should by construction, but clamp)
            beta_sum = beta.sum()
            if abs(beta_sum) > 1e-10:
                beta = beta / beta_sum
            else:
                beta = np.ones(m) / m
            
            # New point: weighted average of T(a_i) for last m iterations
            T_mat = np.array(T_hist[-m:])  # m x n
            a_new = T_mat.T @ beta  # n
        else:
            a_new = T_hist[-1].copy()
        
        # Ensure valid (non-negative, sum=1)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Accept/reject: only accept if objective doesn't worsen too much
        obj_new = compute_obj(a_new)
        obj_curr = compute_obj(a)
        
        if obj_new <= obj_curr * 1.01 + 1e-10:  # allow slight worsening
            a = a_new
        else:
            # Damped step
            for damp in [0.5, 0.3, 0.1]:
                a_damp = (1 - damp) * a + damp * a_new
                a_damp = np.maximum(a_damp, 0)
                a_damp = a_damp / a_damp.sum()
                obj_damp = compute_obj(a_damp)
                if obj_damp <= obj_curr:
                    a = a_damp
                    break
            else:
                a = a_new  # accept anyway to make progress
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()
        
        if callback:
            callback(a, it)
    
    return best_a


def smoothed_gradient_fft(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a).
    Returns (gradient, convolution values).
    """
    n = len(a)
    conv = compute_conv_fft(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def nesterov_descent(a, time_budget=100.0, tau_init=20.0, tau_end=500.0, seed=0):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.8
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init + (tau_end - tau_init) * (1.0 - np.cos(np.pi * progress)) / 2.0

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9995)

        # Reset if stuck
        if it % 50 == 0:
            obj_curr = compute_obj(a)
            if obj_curr > best_obj * 1.05:
                a = best_a.copy()
                v = np.zeros(n)
                lr = 2.0 / np.sqrt(n)

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussians
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        # Cosine taper
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
        a = a / a.sum()
    elif mode == 5:
        # Random smooth
        a = rng.uniform(0, 1, n)
        k = max(2, n // 30)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 6:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1, 1)
        a = a / a.sum()
    elif mode == 7:
        # Tapered (Hann-like)
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2
        a = a / a.sum()
    elif mode == 8:
        # Concentrated center
        x = np.arange(n)
        c = n // 2
        a = np.exp(-((x - c) ** 2) / (2 * (n // 4) ** 2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Anderson acceleration (DIIS) on the LP fixed-point map a -> LP_response(a) as the
primary optimizer, with multi-length exploration, diverse restarts, and smoothed GD polish.
Key insight: the optimal sequence is a fixed point of the LP best-response map.
Anderson acceleration dramatically speeds convergence versus simple iteration.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv_fft(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fft(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=verbose)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_response(a):
    """Compute LP best response T(a). Returns (g, M) or (a, inf) if LP fails."""
    g, M = solve_lp(a)
    if g is not None:
        return g, M
    return a.copy(), np.inf


def anderson_fixed_point(a_init, T_func, memory=6, max_iter=50,
                         time_limit=600.0, verbose=True, tol=1e-8):
    """
    Anderson acceleration (DIIS) for the fixed point problem a = T(a).
    T_func(a) returns (T_a, extra_info).
    Maintains a history of residuals and extrapolates to accelerate convergence.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    X_hist = []   # a values
    G_hist = []   # residuals a - T(a)
    T_hist = []   # T(a) values

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() - start > time_limit:
            break

        T_a, extra = T_func(a)
        T_a = np.maximum(T_a, 0)
        s_t = float(np.sum(T_a))
        if s_t < 1e-10:
            T_a = a.copy()
        else:
            T_a = T_a / s_t

        residual = a - T_a
        res_norm = np.linalg.norm(residual)

        if verbose:
            obj = compute_obj(a)
            lp_val = extra if extra != np.inf else float('nan')
            print(f"    Anderson it{it}: res={res_norm:.8f}, obj={obj:.8f}, LP_M={lp_val:.8f}, t={time.monotonic()-start:.1f}s")

        X_hist.append(a.copy())
        G_hist.append(residual.copy())
        T_hist.append(T_a.copy())

        # Keep only last (memory+1) entries
        max_hist = memory + 1
        if len(X_hist) > max_hist:
            X_hist = X_hist[-max_hist:]
            G_hist = G_hist[-max_hist:]
            T_hist = T_hist[-max_hist:]

        # Compute Anderson coefficients (DIIS)
        m = len(X_hist) - 1  # number of residual vectors

        a_new = T_hist[-1].copy()

        if m >= 2:
            G_mat = np.array(G_hist[-m:])  # m x n
            H = G_mat @ G_mat.T  # m x m Gram matrix

            # Add small regularization for numerical stability
            H_reg = H + 1e-14 * np.trace(H) / m * np.eye(m)

            try:
                beta = np.linalg.solve(H_reg, np.ones(m))
            except np.linalg.LinAlgError:
                beta = np.ones(m) / m

            # Normalize to sum to 1
            bsum = beta.sum()
            if abs(bsum) > 1e-15:
                beta = beta / bsum
            else:
                beta = np.ones(m) / m

            # New point: weighted average of T(a_i)
            T_mat = np.array(T_hist[-m:])  # m x n
            a_new = T_mat.T @ beta  # shape (n,)

        # Ensure valid (non-negative, sum=1)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Accept/reject with damping
        obj_new = compute_obj(a_new)
        obj_curr = compute_obj(a)

        if obj_new <= obj_curr + 1e-12:
            a = a_new
        else:
            # Try damped steps
            accepted = False
            for damp in [0.5, 0.3, 0.1]:
                a_damp = (1.0 - damp) * a + damp * a_new
                a_damp = np.maximum(a_damp, 0)
                s_d = float(np.sum(a_damp))
                if s_d < 1e-10:
                    continue
                a_damp = a_damp / s_d
                obj_d = compute_obj(a_damp)
                if obj_d <= obj_curr + 1e-12:
                    a = a_damp
                    accepted = True
                    break
            if not accepted:
                a = a_new  # accept to make progress

        # Track best
        obj_a = compute_obj(a)
        if obj_a < best_obj:
            best_obj = obj_a
            best_a = a.copy()

        # Convergence check
        if res_norm < tol:
            if verbose:
                print(f"    Anderson converged at it{it} (res={res_norm:.2e})")
            break

    return best_a


def smoothed_gradient_fft(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT.
    Returns (gradient, convolution values).
    """
    n = len(a)
    conv = compute_conv_fft(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def nesterov_descent(a, time_budget=100.0, tau_init=20.0, tau_end=800.0, seed=0):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.8
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init + (tau_end - tau_init) * (1.0 - np.cos(np.pi * progress)) / 2.0

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)

        # BB step size adjustment
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * np.clip(bb_lr, 1e-10, 10.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9997)

        # Reset if stuck for too long
        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)
            best_it = it

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
        a = a / a.sum()
    elif mode == 5:
        a = rng.uniform(0, 1, n)
        k = max(2, n // 30)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1, 1)
        a = a / a.sum()
    elif mode == 7:
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2
        a = a / a.sum()
    elif mode == 8:
        x = np.arange(n)
        c = n // 2
        a = np.exp(-((x - c) ** 2) / (2 * (n // 4) ** 2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
