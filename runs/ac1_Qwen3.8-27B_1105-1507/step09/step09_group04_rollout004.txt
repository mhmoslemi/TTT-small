from scipy.fft import rfft, irfft
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, M_init=None, solver_list=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    constraints = [cp.sum(g) == 1, A @ g <= M]
    
    if M_init is not None:
        constraints.append(M <= M_init)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    # Recompute M for accuracy
                    conv_check = A @ (g_val / s)
                    M_val = float(np.max(conv_check))
                    return g_val / s, M_val
        except Exception:
            continue
    
    return None, None

        best_a_local = best_a.copy()
        
        for t in candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        iter_count += 1
        time_per_iter = time.monotonic() - start
        
        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            no_improve = 0
            if verbose and iter_count % 3 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t_val:.2f}, time={time_per_iter:.0f}s")
        else:
            no_improve += 1
            if no_improve >= 5:
                break
            # Perturb
            a = best_a * (1 + 0.01 * np.random.randn(n))
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a


def coord_descent_sweep(a, time_budget=None, start_time=0, rng=None, verbose=False):
    """
    Coordinate descent on the normalized self-convolution peak.
    For each coordinate k, find optimal value via golden section search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        return a
    a = a / S
    conv = compute_conv(a)
    
    best_obj = float(np.max(conv))
    best_a = a.copy()
    gr = (np.sqrt(5) - 1) / 2.0
    stagnant = 0
    
    while True:
        if time_budget is not None and time.monotonic() - start_time > time_budget:
            break
        
        obj_before = float(np.max(conv))
        improved = False
        
        for k in range(n):
            # Direction: d[m] = a[m-k] for m in [k, k+n-1] (with a[x]=0 outside [0,n))
            # Actually, conv(a + delta*e_k, a + delta*e_k)[m] 
            #   = conv(a,a)[m] + 2*delta*a[m-k] + delta^2*(delta_m,2k)
            # where a[m-k] is valid when 0 <= m-k < n
            
            d = np.zeros(2 * n - 1)
            # a[m-k] is nonzero when 0 <= m-k < n, i.e., k <= m <= k+n-1
            d[k:k + n] = a
            
            lo_delta = -a[k]
            # Upper bound: ensure conv doesn't blow up
            max_conv_cur = float(np.max(conv))
            hi_delta = a[k] + 2.0 * max_conv_cur / (2.0 * a[k] + 1e-10) if a[k] > 1e-10 else 2.0 / n
            hi_delta = max(hi_delta, lo_delta + 1.0 / (2 * n))
            
            # Golden section search (fewer iterations for speed)
            a_gs = lo_delta
            b_gs = hi_delta
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)
            
            def obj_at_delta(delta):
                val = conv + 2.0 * delta * d
                val[2 * k] += delta * delta
                return float(np.max(val))
            
            fc = obj_at_delta(c_gs)
            fd = obj_at_delta(d_gs)
            
            for _ in range(40):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = obj_at_delta(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = obj_at_delta(d_gs)
            
            delta_opt = (a_gs + b_gs) / 2.0
            new_obj_est = obj_at_delta(delta_opt)
            
            if new_obj_est < obj_before - 1e-14:
                a[k] += delta_opt
                a[k] = max(0.0, a[k])
                conv += 2.0 * delta_opt * d
                conv[2 * k] += delta_opt * delta_opt
                S_new = float(np.sum(a))
                if S_new < 1e-10:
                    a[:] = 1.0 / n
                    conv = compute_conv(a)
                else:
                    a /= S_new
                    conv /= (S_new * S_new)
                improved = True
        
        obj_after = float(np.max(conv))
        
        if obj_after < best_obj - 1e-12:
            best_obj = obj_after
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant >= 3:
                break
        
        if verbose:
            elapsed = time.monotonic() - start_time
            print(f"    CD sweep: obj={2.0*n*obj_after:.8f}, improved={improved}, time={elapsed:.0f}s")
    
    return best_a


def smoothed_gradient(a, tau):
    """
    Compute gradient of (1/tau)*log(sum(exp(tau*conv))) with respect to a.
    Returns (gradient, smoothed_value).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    smoothed_val = max_conv + np.log(float(np.sum(exp_vals))) / tau
    
    # Gradient: grad[k] = 2 * sum_m p[m] * a[m-k]
    # = 2 * (p_reversed * a)[k] (convolution of p with reversed a, evaluated at k)
    # Actually: sum_m p[m] * a[m-k] = (p * a_reversed)[k] ... let me think
    # sum_m p[m] * a[m-k]: let j = m-k, then = sum_j p[j+k] * a[j]
    # This is the cross-correlation of p and a.
    # = (p * flip(a))[k + n - 1] ... hmm let me just use FFT
    
    # Using FFT: sum_m p[m] * a[m-k] for m in [0, 2n-2], a[j] for j in [0, n-1]
    # = sum_j p[j+k] * a[j] (where p is zero-padded to size 2n-1)
    # This is the cross-correlation.
    
    # Let's use the direct approach with FFT:
    # grad[k] = 2 * sum_{j=0}^{n-1} a[j] * p[j+k]
    # Let q[m] = p[m] for m in [0, 2n-2]
    # Then grad[k] = 2 * sum_j a[j] * q[j+k] = 2 * (a * q_reversed)[k + (2n-2) - (n-1)] ... 
    
    # Simpler: use FFT to compute the convolution
    N = 1
    while N < 3 * n:
        N <<= 1
    A = rfft(a, N)
    P = rfft(p, N)
    # conv(a, p) has length n + (2n-1) - 1 = 3n-2
    # We want grad[k] = 2 * sum_j a[j] * p[j+k]
    # = 2 * (p * flip(a))[k+n-1] ... 
    # Actually (a * p)[k+j] = sum_i a[i] * p[k+j-i], that's not quite right
    # Let me just use: grad[k] = 2 * sum_j a[j] * p[j+k]
    # Let r = p flipped: r[m] = p[2n-2-m]
    # Then sum_j a[j] * p[j+k] = sum_j a[j] * r[2n-2-j-k] = (a * r)[k + ... ]
    # This is getting confusing. Let me just do it directly for the gradient.
    
    # Actually the simplest: use ifft(fft(p) * conj(fft(a)))
    from scipy.fft import fft, ifft
    Pf = fft(p, N)
    Af = fft(a, N)
    # cross-correlation: conj(A) * P gives sum_j a[j] * p[j+k] (circular)
    # but we need linear cross-correlation
    xcorr = ifft(np.conj(Af) * Pf).real
    # xcorr[k] = sum_j a[j] * p[j+k] (for k in [0, ...])
    # Actually ifft(conj(A)*P)[k] = sum_j a[j] * p[(j+k) mod N]
    # For our case, since p has support [0, 2n-2] and a has support [0, n-1],
    # the linear cross-correlation at lag k is sum_j a[j] * p[j+k] for k >= 0
    # This matches xcorr[k] for k in [0, 2n-2] (as long as N is large enough to avoid wraparound)
    
    grad = 2.0 * xcorr[:n]
    return grad, smoothed_val, conv


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_smoothed_descent(a, time_budget=200.0, tau_init=20.0, tau_end=400.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    lr = 4.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    no_improve = 0
    
    from scipy.fft import fft, ifft
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)
        
        y = a + beta * v
        s_y = float(np.sum(y))
        if s_y < 1e-10 or np.any(y < -1e-6):
            y = np.maximum(y, 0)
            s_y = float(np.sum(y))
            if s_y < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
        y = y / s_y
        
        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient: grad[k] = 2 * sum_j y[j] * p[j+k]
        N = 1
        while N < 3 * n:
            N <<= 1
        Pf = fft(p, N)
        Yf = fft(y, N)
        xcorr = ifft(np.conj(Yf) * Pf).real
        g = 2.0 * xcorr[:n]
        
        # Project gradient onto simplex tangent
        g = g - np.sum(g) / n  # Project onto sum=0 subspace
        
        # Barzilai-Borwein step size (every 2 iterations)
        if it >= 2 and it % 2 == 0:
            # Use previous iterate info
            pass  # Skip BB for simplicity, use fixed lr
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)
        
        if it - best_it > max(10000, 50000 / max(tau, 1)):
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n)
            best_it = it
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.4f}, time={elapsed:.0f}s")
    
    return best_a


def spectral_init(n, rng):
    """Initialize with a sequence that has approximately flat magnitude spectrum."""
    # Random phases with constant magnitude
    N = 1
    while N < 2 * n:
        N <<= 1
    phases = rng.uniform(0, 2 * np.pi, N)
    spectrum = np.exp(1j * phases)
    # Make it real and positive by taking real part of IFFT
    # Better: use a window function
    x = np.linspace(0, 1, n)
    a = 1.0
    # Add some sinusoidal components for diversity
    for i in range(rng.integers(2, 5)):
        f = rng.integers(1, 10)
        p = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.5)
        a = a + amp * np.sin(2 * np.pi * f * x + p)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        x = np.linspace(0, np.pi, n)
        a = np.sin(x)
        a = np.maximum(a, 0.001)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.001)
    elif mode == 3:
        centers = [n // 3, 2 * n // 3]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 0.001)
    elif mode == 4:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 0.1))
    elif mode == 5:
        a = rng.uniform(0, 1, n)
    else:
        x = np.linspace(0, 1, n)
        a = 1.0 / (1 + 5 * x * (1 - x))
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Multi-phase optimization combining LP, coordinate descent,
    and smoothed gradient with Nesterov acceleration.
    """

    best_seq = None
    best_score = np.inf

    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")
    
    # === Phase 1: LP Polishing (primary method) ===
    print(f"\n=== Phase 1: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 450.0)
    if p1_budget > 50:
        a_lp = lp_polish(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 LP: obj={obj_lp:.8f}")
    
    # === Phase 2: Coordinate Descent ===
    print(f"\n=== Phase 2: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 300.0)
    if p2_budget > 30:
        a_cd = coord_descent_sweep(best_seq, time_budget=p2_budget,
                                   start_time=time.monotonic(), rng=rng, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase2 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 CD: obj={obj_cd:.8f}")
    
    # === Phase 3: More LP (finer) ===
    print(f"\n=== Phase 3: LP Refinement ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 250.0)
    if p3_budget > 30:
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            alloc = min(p3_budget / (3 - rnd), remaining * 0.6)
            if alloc < 20:
                break
            a_lp2 = lp_polish(best_seq, time_budget=alloc, verbose=True)
            obj_lp2 = compute_obj(a_lp2)
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  Phase3 LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  Phase3 LP round {rnd}: obj={obj_lp2:.8f}")
                if rnd < 2:
                    scale = 0.015 * (rnd + 1)
                    best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                    best_seq = np.maximum(best_seq, 0)
                    best_seq = best_seq / best_seq.sum()
    
    # === Phase 4: Coordinate Descent (fine) ===
    print(f"\n=== Phase 4: Fine Coordinate Descent ===")

    if p4_budget > 20:

    # === Phase 5: Multi-length exploration ===

            n_options.append(n_fixed)
    
    n_options = sorted(set(n_options))[:5]

        remaining = deadline - time.monotonic()
        if remaining < 120:
            break
        
        a_try = resample_sequence(best_seq, int(n_try))

                                        start_time=time.monotonic(), rng=rng)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = int(n_try)

        mode = (r + 1) % 6
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.3 + 0.15 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()
        
        # Coordinate descent
        alloc1 = min(remaining * 0.3, 60.0)
        if alloc1 > 15:
            a_r = coord_descent_sweep(a_r, time_budget=alloc1,
                                      start_time=time.monotonic(), rng=rng)
        
        # LP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 40.0)
        if alloc2 > 15:
            a_r
