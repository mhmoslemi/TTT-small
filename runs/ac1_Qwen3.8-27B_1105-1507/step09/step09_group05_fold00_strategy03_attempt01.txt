def evaluate_sequence(seq):
    """
    Helper function that directly evaluates a sequence using the provided function's definition.
    Includes defensive measures against invalid inputs.
    """
    if not isinstance(seq, list):
        return np.inf
    n_val = len(seq)
    if n_val == 0:
        return np.inf

    # Check each element
    for x in seq:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert to float and clip
    seq = [float(x) for x in seq]
    for i in range(len(seq)):
        if seq[i] < 0:
            seq[i] = abs(seq[i])  # Ensure non-negative
        if seq[i] > 1000:
            seq[i] = 1000.0  # Cap at 1000

    if n_val == 0:
        return np.inf

    n = n_val
    s = float(np.sum(seq))
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq, mode='full')
    max_val = np.max(conv)
    return float(2.0 * n * max_val) / (s**2)
