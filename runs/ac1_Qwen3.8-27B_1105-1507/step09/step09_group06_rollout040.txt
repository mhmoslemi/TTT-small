"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Hedge/MWU (multiplicative weights) for the zero-sum game formulation,
KKT Newton refinement for precision, LP-based polishing, and multi-length exploration.
Key innovation: replaces LP-based inner loop with fast Hedge iterations (O(n log n) each),
enabling far more iterations within the time budget.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def _next_pow2(m):
    size = 1
    while size < m:
        size <<= 1
    return size


def hedge_game_inner(a, n_iters=20000, eta_init=0.5, eta_end=0.01,
                     time_budget=120.0, verbose=False, seed=0):
    """
    Solve the zero-sum game via Hedge/MWU for the current 'a'.
    Game: Player 1 (min) picks g from simplex, Player 2 (max) picks mu from simplex.
    Payoff P(g, mu) = sum_k mu[k] * (a*g)[k].
    Returns (g_closest, mu_closest, value).
    """
    n = len(a)
    n_lags = 2 * n - 1

    # FFT size must accommodate convolutions of length up to 3n-2
    fft_size = _next_pow2(3 * n)

    # Precompute FFT of a
    a_full = np.zeros(fft_size)
    a_full[:n] = a
    A_fft = fft(a_full)

    # Precompute a_flipped for the dual payoff
    a_flip = np.zeros(fft_size)
    a_flip[:n] = a[::-1]
    AF_fft = fft(a_flip)

    # Initialize
    g = np.ones(n) / n
    mu = np.ones(n_lags) / n_lags

    best_value = np.inf
    best_g = g.copy()
    best_mu = mu.copy()

    start = time.monotonic()
    eta = eta_init

    for t in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        # Anneal learning rate
        progress = t / max(1, n_iters)
        eta = eta_init + (eta_end - eta_init) * progress

        # --- Player 2 update (mu): maximize payoff ---
        # Payoff for each lag k: (a*g)[k]
        g_full = np.zeros(fft_size)
        g_full[:n] = g
        G_fft = fft(g_full)
        ag = np.real(ifft(A_fft * G_fft, fft_size))[:n_lags]

        # Update mu: mu[k] proportional to mu[k] * exp(eta * ag[k])
        # Use log-space for stability
        log_mu = np.full(n_lags, np.log(1.0 / n_lags))
        log_mu += eta * ag
        log_mu -= np.max(log_mu)
        mu = np.exp(log_mu)
        mu_sum = mu.sum()
        if mu_sum < 1e-300:
            mu = np.ones(n_lags) / n_lags

        log_g = np.full(n, np.log(1.0 / n))

        g_new = np.maximum(g_new, 0.0)
        g_new_sum = g_new.sum()
        if g_new_sum < 1e-300:
            g_new = np.ones(n) / n
        else:
            g_new /= g_new_sum

        # Evaluate game value

            best_g = g_new.copy()
            best_mu = mu.copy()

            print(f"      Hedge it{t}: val={val:.10f}, eta={eta:.4f}, t={elapsed:.1f}s")

        for t_c in sorted(set(candidates_t)):
            a_try = (1.0 - t_c) * a + t_c * g_star
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t_c

        total_iters += 1

        if best_local_obj < best_obj - 1e-10:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a.copy()
            t_blend = best_local_t
            consecutive_worse = 0
            last_improve = total_iters
            if verbose and total_iters % 3 == 0:
                elapsed = time.monotonic() - start
                print(f"    HedgeOpt it{total_iters}: obj={best_obj:.10f}, t={t_blend:.2f}, game_val={game_val:.8f}, time={elapsed:.0f}s")
        else:
            consecutive_worse += 1
            # Perturb to escape
            if consecutive_worse >= 3:
                # Use mu_star to guide perturbation
                # mu_star tells us which lags are important
                scale = 0.03 * min(consecutive_worse, 5)
                noise = rng.normal(0, 1, n) * scale
                a_pert = a * np.exp(noise)
                a_pert = np.maximum(a_pert, 0)
                s_pert = float(np.sum(a_pert))
                if s_pert > 1e-10:
                    a_pert = a_pert / s_pert
                # Blend with best
                blend = 0.5 + 0.1 * min(consecutive_worse, 3)
                a = (1 - blend) * a_pert + blend * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()

            if consecutive_worse > 20:
                # More aggressive restart
                a = 0.5 * a + 0.5 * g_star
                a = np.maximum(a, 0)
                a = a / a.sum()
                consecutive_worse = 10

        # Periodic LP verification
        if total_iters > 0 and total_iters % 5 == 0 and remaining > 30:
            try:
                g_lp = _lp_solve(a, timeout_s=min(15.0, remaining * 0.2))
                if g_lp is not None:
                    obj_lp = compute_obj(g_lp)
                    if obj_lp < best_obj:
                        best_obj = obj_lp
                        best_a = g_lp.copy()
                        a = g_lp.copy()
                        consecutive_worse = 0
                        if verbose:
                            print(f"    LP verification: obj={best_obj:.10f} **")
            except Exception:
                pass

    return best_a


def _lp_solve(a, timeout_s=20.0):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0. Returns g or None."""
    n = len(a)
    a_norm = a / np.sum(a)

    # Build convolution matrix A (2n-1 x n)
    # A[k, j] = a[k-j] if 0 <= k-j < n
    n_lags = 2 * n - 1
    A_mat = np.zeros((n_lags, n))
    for j in range(n):
        for k in range(j, min(j + n, n_lags)):
            A_mat[k, j] = a_norm[k - j]

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A_mat @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def kkt_newton_refine(a, time_budget=100.0, verbose=False, seed=0):
    """
    KKT-based Newton refinement.
    At optimum: gradient of max_k (a*a)[k] is constant on support of a.
    Uses log-sum-exp smoothing with high tau, then Newton steps.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Start with high tau smoothing
    tau = 100.0
    lr = 1.0 / n

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        # Compute smoothed gradient
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / exp_vals.sum()

        # Gradient via FFT: grad[i] = 2 * sum_k p[k] * d(a*a)[k]/da[i]
        # d(a*a)[k]/da[i] = a[k-i] (if valid) + a[i-k] (if valid, and k!=2i)
        # More precisely: 2 * (p convolved with a, truncated) with correction
        fft_size = _next_pow2(3 * n)
        P_fft = fft(p, fft_size)
        A_fft = fft(a, fft_size)
        # The gradient of sum_k p[k]*(a*a)[k] w.r.t. a[i] is:
        # 2 * sum_k p[k] * a[k-i] (with bounds) ... approximate via circular conv
        grad_circ = np.real(ifft(P_fft * A_fft, fft_size))[:n]
        # This gives sum_k p[k]*a[k-i mod fft_size], which approximates the gradient
        # for linear conv when a has sufficient zero padding
        grad = 2.0 * grad_circ

        # Project gradient to tangent space of simplex
        grad_mean = float(np.mean(grad))
        grad_tan = grad - grad_mean

        # Adaptive step size (barzilai-borwein style)
        a_try = a - lr * grad_tan
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            lr *= 0.5
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()
            lr = min(lr * 1.1, 10.0 / n)
        else:
            lr *= 0.7

        # Slowly increase tau
        tau *= 1.005

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    KKT-Newton it{it}: obj={best_obj:.10f}, tau={tau:.1f}, lr={lr:.6f}, t={elapsed:.1f}s")

    return best_a


def smooth_gradient_fft(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / exp_vals.sum()

    fft_size = _next_pow2(3 * n)
    P_fft = fft(p, fft_size)
    A_fft = fft(a, fft_size)
    g_full = np.real(ifft(P_fft * A_fft, fft_size))[:n]
    return 2.0 * g_full, conv


def nesterov_descent(a, time_budget=120.0, tau_init=20.0, tau_end=500.0,
                     verbose=False, seed=0):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init + (tau_end - tau_init) * (1.0 - np.cos(np.pi * progress)) / 2.0

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smooth_gradient_fft(y, tau)

        # BB step
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.99995)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.arange(n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 10)
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        xx = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - xx ** 2 + 1e-8))
        a = a / a.sum()
    elif mode == 4:
        # Smooth random (low-pass filtered)
        a = rng.uniform(0, 1, n)
        k = max(2, n // 30)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 5:
        # Hann window
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
        a = a / a.sum()
    elif mode == 6:
        # Power law
        xx = np.linspace(0.01, 1, n)
        a = xx ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 7:
        # Random Fourier features
        freqs = rng.integers(1, 8, size=6)
        phases = rng.uniform(0, 2 * np.pi, size=6)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x / n + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 8:
        # Cosine taper
        a = np.cos(np.pi * x / (n - 1 + 1e-8)) ** 2
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def full_optimize(a, time_budget=300.0, verbose=False, seed=0):
    """
    Full optimization pipeline for a single sequence:
    1. Nesterov GD (quick warmup)
    2. Hedge-based main optimization
    3. KKT Newton polish
    4. LP verification
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Step 1: Quick Nesterov warmup
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        alloc = min(remaining * 0.15, 45.0)
        a1 = nesterov_descent(a, time_budget=alloc, tau_init=10.0, tau_end=300.0,
                              verbose=False, seed=seed)
        obj1 = compute_obj(a1)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()

    # Step 2: Hedge-based main optimization
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        alloc = min(remaining * 0.5, 200.0)
        a2 = hedge_main_optimize(best_a, time_budget=alloc, verbose=verbose, seed=seed + 100)
        obj2 = compute_obj(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()

    # Step 3: KKT Newton polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        alloc = min(remaining * 0.15, 60.0)
        a3 = kkt_newton_refine(best_a, time_budget=alloc, verbose=False, seed=seed + 200)
        obj3 = compute_obj(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()

    # Step 4: LP polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        try:
            g_lp = _lp_solve(best_a, timeout_s=min(20.0, remaining * 0.5))
            if g_lp is not None:
                obj_lp = compute_obj(g_lp)
                if obj_lp < best_obj:
                    best_obj = obj_lp
                    best_a = g_lp.copy()
        except Exception:
            pass

    # Step 5: Final Nesterov with high tau
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        alloc = min(remaining * 0.1, 30.0)
        a5 = nesterov_descent(best_a, time_budget=alloc, tau_init=50.0, tau_end=800.0,
                              verbose=False, seed=seed + 300)
        obj5 = compute_obj(a5)
        if obj5 < best_obj:
            best_obj = obj5
            best_a = a5.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses Hedge/MWU-based optimization with KKT Newton refinement,
    multi-length exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # --- Load parent ---
    parent_n = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                parent_n = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(parent_n) / parent_n
        best_score = compute_obj(best_seq)
        print(f"Default init (n={parent_n}): obj={best_score:.10f}")

    # === Phase 1: Main optimization on parent length ===
    print(f"\n=== Phase 1: Main Optimization (n={parent_n}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 500.0)
    if p1_budget > 30:
        a_opt, obj_opt = full_optimize(best_seq, time_budget=p1_budget,
                                        verbose=True, seed=seed)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase 1 done: obj={best_score:.10f} **")
        else:
            print(f"  Phase 1 done: obj={obj_opt:.10f}")

    # === Phase 2: Intensive Hedge on current best ===
    print(f"\n=== Phase 2: Intensive Hedge ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 280.0)
    if p2_budget > 20:
        a_hedge = hedge_main_optimize(best_seq, time_budget=p2_budget,
                                       verbose=True, seed=seed + 500)
        obj_hedge = compute_obj(a_hedge)
        if obj_hedge < best_score:
            best_score = obj_hedge
            best_seq = a_hedge.copy()
            print(f"  Phase 2 done: obj={best_score:.10f} **")
        else:
            print(f"  Phase 2 done: obj={obj_hedge:.10f}")

    # === Phase 3: KKT Newton + LP polish ===
    print(f"\n=== Phase 3: KKT Newton + LP Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    if p3_budget > 15:
        a_kkt = kkt_newton_refine(best_seq, time_budget=p3_budget * 0.6,
                                   verbose=True, seed=seed + 1000)
        obj_kkt = compute_obj(a_kkt)
        if obj_kkt < best_score:
            best_score = obj_kkt
            best_seq = a_kkt.copy()
            print(f"  KKT: obj={best_score:.10f} **")
        else:
            print(f"  KKT: obj={obj_kkt:.10f}")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            try:
                g_lp = _lp_solve(best_seq, timeout_s=min(25.0, remaining * 0.5))
                if g_lp is not None:
                    obj_lp = compute_obj(g_lp)
                    if obj_lp < best_score:
                        best_score = obj_lp
                        best_seq = g_lp.copy()
                        print(f"  LP: obj={best_score:.10f} **")
                    else:
                        print(f"  LP: obj={obj_lp:.10f}")
            except Exception:
                pass

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 100), n_current + 100, n_current + 200,
        n_current + 300, n_current + 500,
        600, 800, 1000, 1200, 1500, 2000
    ]))
    # Filter: only try lengths that differ by > 50 from current
    n_options = [n for n in n_options if 300 <= n <= 2500 and abs(n - n_current) > 50]
    n_options = n_options[:5]  # Limit to 5 attempts

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        alloc = min(remaining * 0.15, 120.0)
        if alloc < 20:
            break

        a_opt, obj_opt = full_optimize(a_try, time_budget=alloc,
                                        verbose=False, seed=seed + n_try)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.10f}")

    # === Phase 5: Diverse Restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = (r * 2 + 1) % 9
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.15 + 0.05 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.15, 80.0)
        if alloc < 15:
            break

        a_opt, obj_opt = full_optimize(a_r, time_budget=alloc,
                                        verbose=False, seed=seed + 2000 + r * 10)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.10f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_opt:.10f}")

    # === Phase 6: Final Intensive Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Final Hedge
    if remaining > 20:
        alloc = min(remaining * 0.25, 80.0)
        a_final = hedge_main_optimize(best_seq, time_budget=alloc,
                                       verbose=True, seed=seed + 5000)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final Hedge: obj={best_score:.10f} **")
        else:
            print(f"  Final Hedge: obj={obj_final:.10f}")

    # Final KKT
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.25, 50.0)
        a_kkt = kkt_newton_refine(best_seq, time_budget=alloc,
                                   verbose=False, seed=seed + 6000)
        obj_kkt = compute_obj(a_kkt)
        if obj_kkt < best_score:
            best_score = obj_kkt
            best_seq = a_kkt.copy()
            print(f"  Final KKT: obj={best_score:.10f} **")
        else:
            print(f"  Final KKT: obj={obj_kkt:.10f}")

    # Final Nesterov with very high tau
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.2, 30.0)
        a_ns = nesterov_descent(best_seq, time_budget=alloc,
                                 tau_init=100.0, tau_end=1000.0,
                                 verbose=False, seed=seed + 7000)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Final Nesterov: obj={best_score:.10f} **")
        else:
            print(f"  Final Nesterov: obj={obj_ns:.10f}")

    # === Final Validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
