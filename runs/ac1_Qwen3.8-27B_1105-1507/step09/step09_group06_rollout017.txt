```python
import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time

def compute_conv(a):
    n = len(a)
    if n == 0: return np.array([])
    size = 1
    while size < 2*n-1: size <<= 1
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    if n == 0: return np.inf
    s = float(np.sum(a))
    if s < 0.01: return np.inf
    a_n = a/s
    conv = compute_conv(a_n)
    return 2.0*n*float(np.max(conv))

def build_conv_matrix(a):
    """A_mat such that A_mat @ g = convolve(a, g)"""
    n = len(a)
    A_mat = np.zeros((2*n-1, n))
    for j in range(n):
        A_mat[j:j+min(n, 2*n-1-j), j] = a[:min(n, 2*n-1-j)]
    return A_mat

def solve_lp(a):
    """Solve min M s.t. A@g <= M, sum(g)=1, g>=0"""
    n = len(a)
    a_n = a / np.sum(a)
    A_mat = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val/s, float(M.value)
        except:
            continue
    return None, None

def line_search_blend(a, g, max_obj, iters=30):
    """Find best t in (0,1) for a_new = (1-t)*a + t*g"""
    best_t = 0.0
    best_obj = max_obj
    # Golden section search
    lo, hi = 0.0, 1.0
    gr = (np.sqrt(5)-1)/2
    a1 = hi - gr*(hi-lo)
    a2 = lo + gr*(hi-lo)
    f1 = compute_obj(np.maximum((1-a1)*a + a1*g, 0))
    f2 = compute_obj(np.maximum((1-a2)*a + a2*g, 0))
    for _ in range(iters):
        if f1 < f2:
            hi = a2; a2 = a1; f2 = f1; a1 = hi - gr*(hi-lo); f1 = compute_obj(np.maximum((1-a1)*a + a1*g, 0))
        else:
            lo = a1; a1 = a2; f1 = f2; a2 = lo + gr*(hi-lo); f2 = compute_obj(np.maximum((1-a2)*a + a2*g, 0))
    best_t = (lo+hi)/2
    best_obj = compute_obj(np.maximum((1-best_t)*a + best_t*g, 0))
    return best_t, best_obj
```

The active set approach:
1. Given current a, find active set S = {k : conv(a,a)[k] >= max_conv - eps}
