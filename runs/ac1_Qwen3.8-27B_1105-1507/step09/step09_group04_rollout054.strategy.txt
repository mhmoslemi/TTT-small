def objective_function(seq):
    """Compute score based on autocorrelation."""
    n = len(seq)
    # Clip negatives
    seq = clip_excess(seq)
    # Full convolution
    full_conv = np.convolve(seq, seq, mode='full')
    m_max = np.max(full_conv)
    s = sum(seq)
    if s >= 0.01:
        return 2.0 * n * m_max / (s**2)
    return np.inf

def clip_excess(seq):
