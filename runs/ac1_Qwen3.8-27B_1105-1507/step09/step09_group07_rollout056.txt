"""
Optimized step function search using accelerated LP alternating with convex line search,
peak-targeting heuristics, and adaptive n-sweep. Fixes bugs from previous version and
adds smarter perturbation strategies.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_fast(a_norm):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses scipy linprog with HiGHS. Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a_norm)
    A = build_conv_matrix(a_norm)
    
    # Variables: g[0..n-1], M
    # Minimize M -> c = [0, 0, ..., 0, 1]
    # Constraints: A@g - M <= 0 for each row -> A_ub = [A, -1-column]
    # Equality: sum(g) = 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def line_search_convex(a, g, n):
    """
    Find optimal t in [0,1] minimizing f(t) = 2n * max_k conv((1-t)*a + t*g, ...)[k]
    Uses ternary search since f is convex in t.
    """
    def f_of_t(t):
        a_t = (1 - t) * a + t * g
        conv = compute_conv(a_t)
        return 2.0 * n * float(np.max(conv))
    
    # Ternary search
    lo, hi = 0.0, 1.0
    for _ in range(60):  # sufficient precision
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        if f_of_t(m1) < f_of_t(m2):
            hi = m2
        else:
            lo = m1
    
    t_opt = (lo + hi) / 2
    
    # Also check endpoints and some key points
    candidates = [0.0, t_opt, 1.0, 0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9]
    best_t = t_opt
    best_f = f_of_t(t_opt)
    for t in candidates:
        f_val = f_of_t(t)
        if f_val < best_f:
            best_f = f_val
            best_t = t
    
    return best_t, best_f

        current_obj = compute_obj(a)

        similarity = 1.0 - np.linalg.norm(a - g_lp) / (np.linalg.norm(a) + np.linalg.norm(g_lp) + 1e-15)
        if similarity > 0.9999:
            stall += 2
        else:
            stall = max(0, stall - 1)

        # Convex line search for optimal t
        t_opt, f_opt = line_search_convex(a, g_lp, n)
        
        # Apply update
        a_new = (1 - t_opt) * a + t_opt * g_lp
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        iteration += 1
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                # Perturb and restart
                rng = np.random.default_rng(iteration * 104729 + n)
                scale = min(0.3, 0.02 + 0.01 * (stall // patience))
                noise = rng.normal(0, scale, n)
                a = best_a * np.exp(noise)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0

        if verbose and iteration % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"        LP-alt it{iteration}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                  f"stall={stall}, elapsed={elapsed:.1f}s")

    return best_a


def peak_targeting(a, time_budget, verbose=False):
    """
    Targeted peak reduction: identify the peak lags in autocorrelation,
    find the largest contributing pairs, and redistribute mass.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    no_improve = 0

    while time.monotonic() - start < time_budget and no_improve < 50:
        conv = compute_conv(a)
        peak_val = float(np.max(conv))
        peak_lags = np.argsort(conv)[::-1][:5]
        
        improved = False
        no_improve += 1
        
        for k_star in peak_lags:
            k_star = int(k_star)
            if k_star < 0 or k_star >= 2 * n - 1:
                continue
            
            # Find contributors to this lag
            if k_star == 0:
                # (a*a)[0] = sum a[i]^2
                # To reduce, spread mass from large a[i] to small a[j]
                idx_sorted = np.argsort(a)[::-1]
                src_candidates = idx_sorted[:10]
                dst_candidates = np.argsort(a)[:10]
            else:
                # (a*a)[k] = sum_i a[i]*a[i+k]
                contribs = []
                for i in range(n):
                    j = i + k_star
                    if j < n:
                        contribs.append((a[i] * a[j], i, j))
                contribs.sort(reverse=True)
                src_candidates = [c[1] for c in contribs[:10]]
                dst_candidates = np.argsort(a)[:10]
            
            for src in src_candidates:
                for dst in dst_candidates:
                    if src == dst:
                        continue
                    for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                        delta = frac * a[src]
                        if delta <= 0 or delta >= a[src] * 0.9:
                            continue
                        a[src] -= delta
                        a[dst] += delta
                        obj_try = compute_obj(a)
                        if obj_try < best_obj - 1e-12:
                            best_obj = obj_try
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break
                        else:
                            a[src] += delta
                            a[dst] -= delta
                            break
                if improved:
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break

    return best_a


def spectral_refine(a, time_budget, verbose=False):
    """
    Iterative spectral flattening to reduce autocorrelation peaks.
    Uses a better approach: gradient descent on the spectrum shaping.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n:
        size <<= 1
    
    for _ in range(100):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute current spectrum
        A = rfft(a, size)
        P = np.abs(A) ** 2
        
        # Target: flat spectrum (white noise)
        # Gradient: increase |A|^2 where it's below mean, decrease where above
        P_mean = np.mean(P)
        P_err = P - P_mean  # positive means too much power at this freq
        
        # Update in frequency domain
        # a_new = ifft(A * (1 - eta * P_err / (2*P)))
        eta = 0.1
        correction = 1.0 - eta * P_err / (2.0 * (P + 1e-15))
        A_new = A * correction
        a_new = irfft(A_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            if verbose:
                print(f"          Spectral: obj={best_obj:.8f}")
        else:
            # Try smaller step
            a_new = irfft(A * (1.0 - 0.01 * P_err / (2.0 * (P + 1e-15))), size)[:n]
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-10:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new.copy()

    return best_a


def diverse_init(n, rng, mode=0):

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimized step function search using accelerated LP alternating with convex line search,
peak-targeting heuristics, spectral refinement, and adaptive n-sweep.
Key improvements over previous: fixed bug, convex line search for optimal step,
faster LP via direct HiGHS, better perturbation strategies, wider n exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_fast(a_norm):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses scipy linprog with HiGHS. Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a_norm)
    A = build_conv_matrix(a_norm)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def line_search_convex(a, g, n):
    """
    Find optimal t in [0,1] minimizing f(t) = 2n * max_k conv((1-t)*a + t*g, ...)[k].
    Uses ternary search since f is convex in t (proven via sum of squared differences).
    """
    def f_of_t(t):
        a_t = (1.0 - t) * a + t * g
        conv = compute_conv(a_t)
        return 2.0 * n * float(np.max(conv))

    lo = 0.0
    hi = 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = f_of_t(m1)
        f2 = f_of_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t_opt = (lo + hi) / 2.0
    best_t = t_opt
    best_f = f_of_t(t_opt)

    # Check a few additional candidate points
    candidates = [0.0, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9, 0.95, 1.0]
    for t in candidates:
        f_val = f_of_t(t)
        if f_val < best_f:
            best_f = f_val
            best_t = t

    return best_t, best_f


def lp_alternate_accelerated(a, time_budget, verbose=False):
    """
    LP alternating with convex line search for optimal step size.
    Faster convergence than fixed-step-size approach.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    stall = 0
    patience = 30
    iteration = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_fast(a)
        if g_lp is None:
            break

        # Check convergence: if LP solution is very close to current, we're at a fixed point
        diff_norm = np.linalg.norm(a - g_lp)
        a_norm_val = np.linalg.norm(a)
        if a_norm_val > 1e-15:
            rel_diff = diff_norm / a_norm_val
        else:
            rel_diff = 1.0

        if rel_diff < 1e-6:
            stall += 3
        elif M_lp < float(np.max(compute_conv(a))) * 0.999:
            stall = max(0, stall - 2)
        else:
            stall = max(0, stall - 1)

        # Convex line search for optimal t
        t_opt, f_opt = line_search_convex(a, g_lp, n)

        # Apply update
        a_new = (1.0 - t_opt) * a + t_opt * g_lp
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        iteration += 1
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                # Perturb: use structured noise
                rng = np.random.default_rng(iteration * 104729 + n * 7)
                scale = min(0.25, 0.015 + 0.01 * (stall // patience))
                noise = rng.normal(0, scale, n)
                a = best_a * np.exp(noise)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0

        if verbose and iteration % 15 == 0:
            elapsed = time.monotonic() - start
            print(f"        LP-alt it{iteration}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                  f"stall={stall}, elapsed={elapsed:.1f}s")

    return best_a


def peak_targeting(a, time_budget, verbose=False):
    """
    Targeted peak reduction: identify peak lags, find largest contributors,
    redistribute mass from high-contribution positions to low-mass positions.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    no_improve_count = 0

    while time.monotonic() - start < time_budget and no_improve_count < 80:
        conv = compute_conv(a)
        peak_lags = np.argsort(conv)[::-1][:4]

        improved = False

        for k_star in peak_lags:
            k_star = int(k_star)
            if k_star < 0 or k_star >= 2 * n - 1:
                continue

            # Find source positions (high mass that contributes to peaks)
            if k_star == 0:
                # Diagonal: sum a[i]^2. Move from largest to smallest.
                src_idx = np.argsort(a)[::-1][:8]
                dst_idx = np.argsort(a)[:8]
            else:
                # Off-diagonal: sum a[i]*a[i+k_star]
                contribs = []
                for i in range(n):
                    j = i + k_star
                    if j < n:
                        contribs.append((a[i] * a[j], i, j))
                contribs.sort(reverse=True)
                src_idx = list(set([c[1] for c in contribs[:8]] + [c[2] for c in contribs[:8]]))[:8]
                dst_idx = np.argsort(a)[:8]

            for src in src_idx:
                for dst in dst_idx:
                    if src == dst:
                        continue
                    if a[src] < 1e-10:
                        continue
                    for frac in [0.003, 0.007, 0.015, 0.03, 0.05]:
                        delta = frac * a[src]
                        if delta <= 0 or delta >= a[src] * 0.8:
                            continue
                        a[src] -= delta
                        a[dst] += delta
                        obj_try = compute_obj(a)
                        if obj_try < best_obj - 1e-12:
                            best_obj = obj_try
                            best_a = a.copy()
                            improved = True
                            no_improve_count = 0
                            break
                        else:
                            a[src] += delta
                            a[dst] -= delta
                            break
                if improved:
                    break
            if improved:
                break

        if not improved:
            no_improve_count += 1

        if verbose and no_improve_count % 20 == 0:
            print(f"          Peak targeting: obj={best_obj:.8f}, no_improve={no_improve_count}")

    return best_a


def spectral_refine(a, time_budget, verbose=False):
    """
    Iterative spectral shaping: flatten the power spectrum to reduce autocorrelation peaks.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    step = 0.1
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        A = rfft(a, size)
        P = np.abs(A) ** 2
        P_mean = np.mean(P)
        P_err = P - P_mean

        correction = 1.0 - step * P_err / (2.0 * (P + 1e-15))
        correction = np.maximum(correction, 0.5)  # prevent sign flips
        A_new = A * correction
        a_new = irfft(A_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            step = min(0.5, step * 1.1)
            if verbose:
                print(f"          Spectral it{it}: obj={best_obj:.8f}, step={step:.3f}")
        else:
            step *= 0.8
            if step < 0.005:
                break

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode % 10 == 0:
        a = np.ones(n) / n
    elif mode % 10 == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode % 10 == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 3:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(4.0 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (x - 0.3) ** 2) + np.exp(-2.0 * (x - 0.7) ** 2)
        a = a / a.sum()
    elif mode % 10 == 5:
        x = np.linspace(0, 1, n)
        a = x ** 2 + (1.0 - x) ** 2 + 0.1
        a = a / a.sum()
    elif mode % 10 == 6:
        freqs = rng.integers(2, 8, size=3)
        phases = rng.uniform(0, 2.0 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for ii in range(3):
            a += 0.3 * np.sin(2.0 * np.pi * freqs[ii] * x + phases[ii])
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 7:
        a = rng.dirichlet(np.ones(n) * 0.5)
    elif mode % 10 == 8:
        x = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (x - 0.5) ** 2)
        a = a / a.sum()
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (x - 0.25) ** 2) + np.exp(-20.0 * (x - 0.75) ** 2)
        a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def two_phase_lp(a, time_budget):
    """
    Two-phase LP: first do LP alternating, then from the result solve LP again
    (using the LP solution as the new 'a' direction). Can escape some local minima.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: normal LP alternating
    a = lp_alternate_accelerated(a, time_budget=time_budget * 0.6, verbose=False)
    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    # Phase 2: Solve LP with current a, use solution as new starting a
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        g_lp, _ = solve_lp_fast(a)
        if g_lp is not None and np.linalg.norm(g_lp - a) > 1e-4:
            a2 = lp_alternate_accelerated(g_lp, time_budget=remaining * 0.8, verbose=False)
            obj2 = compute_obj(a2)
            if obj2 < best_obj:
                best_obj = obj2
                best_a = a2.copy()

    return best_a

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init n={n_init}: obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Quick LP polish (12%) ===
    print(f"\n=== Phase 1: Quick LP Polish ===")

        a_p1 = lp_alternate_accelerated(best_seq, time_budget=p1_budget, verbose=True)

    n_options = sorted(set([

        step = len(n_options_idx) // 15
        n_options_picked = [n_options_idx[i * step] for i in range(min(15, len(n_options_idx)))]
        n_options = [n_options[i] for i in n_options_picked]
        # Make sure n_current is included
        if n_current not in n_options:
            n_options.append(n_current)
        n_options = sorted(set(n_options))

    per_n_budget = p2_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_opt = lp_alternate_accelerated(a_try, time_budget=alloc, verbose=False)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"=== Target: 1.5027, Achieved: {best_score:.8f}, Gap: {best_score - 1.5027:.8f} ===")

    return result
