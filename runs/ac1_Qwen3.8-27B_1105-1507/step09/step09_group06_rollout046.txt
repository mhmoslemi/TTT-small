def fourier_perturb(a, freq, phase, scale):
    """Add a sinusoidal perturbation and project to simplex."""
    n = len(a)
    x = np.arange(n)
    pert = scale * np.sin(2*np.pi*freq*x/n + phase)
    a_new = a + pert
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a
    return a_new / s

def structured_search(a, time_budget=200.0, seed=0):
    """Try structured perturbations and polish each."""
    n = len(a)
    rng = np.random.default_rng(seed)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Try different Fourier modes
    for freq in range(1, 15):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 30:
            break
        
        for scale in [0.02, 0.05, 0.1]:
            remaining = time_budget - (time.monotonic() - start)
            if remaining < 20:
                break
            
            phases = rng.uniform(0, 2*np.pi, size=3)
            for phase in phases:
                remaining = time_budget - (time.monotonic() - start)
                if remaining < 15:
                    break
                
                a_pert = fourier_perturb(a, freq, phase, scale)
                # Quick L-BFGS polish
                a_polished, _ = lbfgs_optimize(a_pert, tau=100, max_iter=50, time_budget=10)
                obj = compute_obj(a_polished)
                if obj < best_obj - 1e-8:
                    best_obj = obj
                    best_a = a_polished.copy()
                    print(f"    Fourier f={freq}, s={scale}: obj={obj:.8f} **")
    
    # Try Gaussian bumps
    for _ in range(10):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 30:
            break
        c = rng.integers(0, n)
        w = rng.integers(max(5, n//50), n//5)
        scale = rng.uniform(0.02, 0.1)
        x = np.arange(n)
        pert = scale * np.exp(-0.5*((x-c)/w)**2)
        a_pert = a + pert - scale*np.exp(-0.5*((x-c)/w)**2).sum()/n  # maintain sum
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        a_polished, _ = lbfgs_optimize(a_pert, tau=100, max_iter=50, time_budget=10)
        obj = compute_obj(a_polished)
        if obj < best_obj - 1e-8:
            best_obj = obj
            best_a = a_polished.copy()
    
    return best_a

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
