Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast FFT-based transfer search, multiplicative peak reduction, spectral initialization,

import numpy as np

import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def transfer_delta(a, i, j, delta):
    """Compute the change in convolution when transferring delta from position i to position j."""
    n = len(a)
    # delta_conv[k] = 2*delta*(a[k-j] - a[k-i]) for valid k
    d = np.zeros(2 * n - 1)
    # Contribution from removing delta at i: -2*delta*a[k-i] for k in [i, i+n-1]
    k_start = i
    k_end = min(2 * n - 1, i + n)
    valid_a = np.zeros(n)
    for k in range(k_start, k_end):
        idx = k - i
        if 0 <= idx < n:
            d[k] -= 2 * delta * a[idx]
    # Contribution from adding delta at j: +2*delta*a[k-j] for k in [j, j+n-1]
    k_start = j
    k_end = min(2 * n - 1, j + n)
    for k in range(k_start, k_end):
        idx = k - j
        if 0 <= idx < n:
            d[k] += 2 * delta * a[idx]
    return d


def transfer_delta_fast(a, i, j, delta):
    """Vectorized computation of convolution change for transfer from i to j."""
    n = len(a)
    d = np.zeros(2 * n - 1)
    # -2*delta * (a shifted by i) placed at positions [i, i+n-1]
    seg1 = slice(i, min(2 * n - 1, i + n))
    seg1_len = min(n, 2 * n - 1 - i)
    d[seg1] -= 2 * delta * a[:seg1_len]
    # +2*delta * (a shifted by j) placed at positions [j, j+n-1]
    seg2 = slice(j, min(2 * n - 1, j + n))
    seg2_len = min(n, 2 * n - 1 - j)
    d[seg2] += 2 * delta * a[:seg2_len]
    return d


def fast_local_search(a, time_budget=100.0, verbose=False):
    """
    Fast transfer-based local search.
    For the dominant peak, find the best (source, target) transfer that reduces objective.
    Uses FFT for fast convolution recomputation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    it = 0
    stall = 0
    delta_scale = 0.05

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        k_star = int(np.argmax(conv))

        # Find top contributors to the peak
        i_lo = max(0, k_star - (n - 1))
        i_hi = min(n - 1, k_star)
        if i_lo > i_hi:
            break

        contribs = []
        for i in range(i_lo, i_hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contribs.append((a[i] * a[j], i, j))
        contribs.sort(reverse=True)

        # Try transfers from top contributors
        improved = False
        top_k = min(8, len(contribs))
        used_targets = set()

        for ci in range(top_k):
            _, src, partner = contribs[ci]
            if a[src] < 1e-12:
                continue

            # Determine best direction: reduce a[src] and increase a[target]
            # The change at peak: 2*delta*(a[k_star - target] - a[k_star - src])
            # We want a[k_star - target] < a[k_star - src]
            peak_src_val = a[k_star - src] if 0 <= k_star - src < n else 0.0

            # Candidate targets: positions where a[k_star - t] is small
            # k_star - t in [0, n-1] means t in [k_star-n+1, k_star]
            t_lo = max(0, k_star - (n - 1))
            t_hi = min(n - 1, k_star)

            # Score all potential targets
            best_target = -1
            best_score = np.inf
            for t in range(t_lo, t_hi + 1):
                if t == src or t in used_targets:
                    continue
                t_val = a[k_star - t] if 0 <= k_star - t < n else 0.0
                # We want t_val < peak_src_val, and also a[t] should not be too large
                score = t_val + 0.1 * a[t]
                if score < best_score:
                    best_score = score
                    best_target = t

            if best_target < 0:
                continue

            tgt = best_target
            # Try several delta values
            for delta_frac in [0.1, 0.05, 0.02, 0.01]:
                delta = delta_frac * a[src]
                if delta <= 0 or delta > a[src]:
                    continue

                a_try = a.copy()
                a_try[src] -= delta
                a_try[tgt] += delta
                if a_try[src] < -1e-15:
                    continue
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try

                conv_try = compute_conv(a_try)
                max_try = float(np.max(conv_try))
                obj_try = 2.0 * n * max_try

                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    used_targets.add(tgt)
                    # Decay delta
                    delta_scale = max(0.005, delta_scale * 0.95)
                    break

        # Also try random transfers for exploration
        if not improved and it % 3 == 0:
            for _ in range(10):
                src = rng.integers(0, n)
                tgt = rng.integers(0, n)
                if src == tgt or a[src] < 1e-12:
                    continue
                delta = 0.02 * a[src]
                a_try = a.copy()
                a_try[src] -= delta
                a_try[tgt] += delta
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 80:
                # Perturb to escape local minimum
                scale = 0.01 + 0.005 * min(stall / 20, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = float(np.sum(a))
                if s_a > 1e-10:
                    a = a / s_a
                else:
                    a = best_a.copy()
                stall = 0
                it = 0
                if time.monotonic() - start > time_budget * 0.8:
                    break

        if verbose and it % 50 == 0:
            print(f"    FLS it{it}: obj={best_obj:.8f}, stall={stall}")

    return best_a


def multiplicative_peak_reduction(a, time_budget=100.0, verbose=False):
    """
    Multiplicative update method: reduce elements that contribute to high peaks.
    a_i *= exp(-eta * sum over top peaks of weight * a[k-i])
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 1.0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        mean_val = float(np.mean(conv))

        # Find top peaks
        top_k = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:top_k]

        # Compute "pressure" on each element
        pressure = np.zeros(n)
        for k in peak_indices:
            weight = (conv[k] - mean_val) / max(max_val - mean_val, 1e-10)
            if weight <= 0:
                continue
            i_lo = max(0, k - (n - 1))
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += weight * a[j]

        # Multiplicative update
        a_new = a * np.exp(-eta * pressure)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Blend to avoid overshooting
        blend = 0.3
        a_new = (1 - blend) * a + blend * a_new
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Slowly decrease eta
            eta *= 0.998
        else:
            # Try smaller blend
            for b in [0.1, 0.05, 0.02]:
                a_try = (1 - b) * a + b * a_new
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        break
            else:
                # No improvement, perturb
                a = best_a * (1 + 0.01 * np.random.randn(n))
                a = np.maximum(a, 0)
                s_a = float(np.sum(a))
                if s_a > 1e-10:
                    a = a / s_a
                else:
                    a = best_a.copy()
                eta = max(0.01, eta * 0.5)

        it += 1
        if verbose and it % 20 == 0:
            print(f"    MPR it{it}: obj={best_obj:.8f}, eta={eta:.4f}")

    return best_a


def block_swap_search(a, time_budget=60.0, verbose=False):
    """Try swapping contiguous blocks of the sequence."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    it = 0
    while time.monotonic() - start < time_budget:
        # Try random block swaps
        improved = False
        for _ in range(50):
            block_len = rng.integers(1, max(2, n // 20))
            i = rng.integers(0, n - block_len)
            j = rng.integers(0, n - block_len)
            if np.abs(i - j) < block_len:
                continue

            a_try = a.copy()
            # Swap blocks
            a_try[i:i+block_len], a_try[j:j+block_len] = a_try[j:j+block_len].copy(), a_try[i:i+block_len].copy()
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        it += 1
        if not improved and it > 20:
            break

        if verbose and it % 5 == 0:
            print(f"    BSS it{it}: obj={best_obj:.8f}")

    return best_a


def solve_lp_minmax(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)

    # Build convolution matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def lp_polish(a, time_budget=60.0, verbose=False):
    """LP-based polishing with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    it = 0
    while time.monotonic() - start < time_budget:
        g_lp = solve_lp_minmax(a)
        if g_lp is None:
            break

        # Line search: find best t in [0,1]
        best_t = 0.0
        best_f = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            f_try = compute_obj(a_try)
            if f_try < best_f:
                best_f = f_try
                best_t = t

        if best_t > 0:
            a_new = (1 - best_t) * a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                if best_f < best_obj - 1e-12:
                    best_obj = best_f
                    best_a = a_new.copy()
                    a = a_new
        else:
            # Perturb
            rng = np.random.default_rng(it + 42)
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            a = a / a.sum()

        it += 1
        if verbose and it % 3 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}")

    return best_a


def spectral_init(n, rng, mode=0):
    """Generate initial sequences using spectral methods or other patterns."""
    if mode == 0:
        # Uniform
        a = np.ones(n)
    elif mode == 1:
        # Comb pattern
        a = np.ones(n)
        a[n//3::3] = 0.3
    elif mode == 2:
        # Sine-modulated
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.sin(2 * np.pi * rng.uniform(1, 8) * x + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
    elif mode == 3:
        # Random spectrum
        N = n
        spectrum = rng.uniform(0.8, 1.2, N)
        # Make it Hermitian
        full_spec = np.concatenate([spectrum, spectrum[::-1][1:-1]])
        a = np.fft.realfull(full_spec, 2*N)[:n] if N > 1 else np.array([1.0])
        a = np.maximum(a, 0)
        if a.sum() < 1e-10:
            a = np.ones(n)
    elif mode == 4:
        # Exponential decay from center
        mid = n // 2
        x = np.abs(np.arange(n) - mid)
        a = np.exp(-x / rng.uniform(n/4, n/2))
    elif mode == 5:
        # Piecewise constant with random breakpoints
        n_blocks = rng.integers(3, 10)
        breaks = sorted(rng.integers(1, n-1, n_blocks-1).tolist())
        starts = [0] + breaks
        ends = breaks + [n]
        a = np.zeros(n)
        for s, e in zip(starts, ends):
            a[s:e] = rng.uniform(0.2, 1.0)
    elif mode == 6:
        # Alternating high/low
        a = np.ones(n)
        a[::2] = rng.uniform(0.3, 0.7)
    else:
        a = rng.uniform(0.1, 1.0, n)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fine_grid_search(a, time_budget=60.0, verbose=False):
    """
    Fine-grained search over (source, delta) pairs for the dominant peak.
    Systematically tries reducing each contributor and distributing the mass.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(55555)

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        max_val = float(conv[k_star])

        # Get contributors sorted by contribution
        i_lo = max(0, k_star - (n - 1))
        i_hi = min(n - 1, k_star)
        contribs = []
        for i in range(i_lo, i_hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contribs.append((a[i] * a[j], i, j))
        contribs.sort(reverse=True)

        improved = False
        # For each top contributor, try multiple (target, delta) combinations
        for ci, (cval, src, partner) in enumerate(contribs[:6]):
            if a[src] < 1e-10:
                continue

            # Determine where in the convolution the source contributes
            # Reducing a[src] by delta reduces conv[k] by 2*delta*a[k-src]
            # We want to put the delta somewhere that minimizes new peaks

            # Precompute the "risk" of each target position
            # If we add delta at position t, conv[k] increases by 2*delta*a[k-t]
            # The peak after change at k_star: max_val - 2*delta*a[k_star-src] + 2*delta*a[k_star-t]
            # We want a[k_star-t] < a[k_star-src]

            peak_contrib = a[k_star - src] if 0 <= k_star - src < n else 0.0

            # Score targets: we want positions t where a[k_star-t] is small
            # and the position doesn't create new peaks elsewhere
            candidates = []
            for t in range(n):
                if t == src or t == partner:
                    continue
                # Risk at the dominant peak
                peak_risk = a[k_star - t] if 0 <= k_star - t < n else 0.0
                # Also check a few other high peaks
                other_risk = 0.0
                for k2 in range(len(conv)):
                    if k2 == k_star or conv[k2] < max_val * 0.95:
                        continue
                    other_risk += a[k2 - t] if 0 <= k2 - t < n else 0.0
                score = peak_risk + 0.3 * other_risk / max(n, 1)
                candidates.append((score, t))

            candidates.sort()
            used = set()

            for score, tgt in candidates[:8]:
                if tgt in used:
                    continue
                for delta_frac in [0.08, 0.04, 0.02]:
                    delta = delta_frac * a[src]
                    if delta <= 0 or delta >= a[src]:
                        continue

                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[tgt] += delta
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try

                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        used.add(tgt)
                        break

                if improved:
                    break

            if not improved and ci >= 3:
                break

        # Random exploration
        if not improved and it % 5 == 0:
            for _ in range(20):
                src = int(rng.integers(0, n))
                tgt = int(rng.integers(0, n))
                if src == tgt or a[src] < 1e-10:
                    continue
                delta = 0.03 * a[src]
                a_try = a.copy()
                a_try[src] -= delta
                a_try[tgt] += delta
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        it += 1
        if not improved and it % 30 == 0:
            # Perturb
            scale = min(0.05, 0.01 + 0.002 * (it // 30))
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s_a = float(np.sum(a))
            if s_a > 1e-10:
                a = a / s_a
            else:
                a = best_a.copy()
            it = 0

        if verbose and it % 20 == 0:
            print(f"    FGS it{it}: obj={best_obj:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-strategy optimization combining fast local search, multiplicative updates,
    spectral initialization, block swaps, and LP polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent available: {e}")

    if best_seq is None:
        best_seq = np.ones(800)
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Fast local search on parent ===
    print(f"\n=== Phase 1: Fast Local Search ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 30:
        a1 = fast_local_search(best_seq, time_budget=p1_budget, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  FLS: obj={best_score:.8f} **")
        else:
            print(f"  FLS: obj={obj1:.8f}")

    # === Phase 2: Fine grid search ===
    print(f"\n=== Phase 2: Fine Grid Search ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 250.0)
    if p2_budget > 30:
        a2 = fine_grid_search(best_seq, time_budget=p2_budget, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  FGS: obj={best_score:.8f} **")
        else:
            print(f"  FGS: obj={obj2:.8f}")

    # === Phase 3: Multiplicative peak reduction ===
    print(f"\n=== Phase 3: Multiplicative Peak Reduction ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 180.0)
    if p3_budget > 30:
        a3 = multiplicative_peak_reduction(best_seq, time_budget=p3_budget, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  MPR: obj={best_score:.8f} **")
        else:
            print(f"  MPR: obj={obj3:.8f}")

    # === Phase 4: Block swap search ===
    print(f"\n=== Phase 4: Block Swap ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.08, 80.0)
    if p4_budget > 15:
        a4 = block_swap_search(best_seq, time_budget=p4_budget, verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  BSS: obj={best_score:.8f} **")
        else:
            print(f"  BSS: obj={obj4:.8f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(100, n_best - 100),
        n_best + 50,
        n_best + 150,
        500, 700, 900, 1100, 1300
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1500 and abs(n - n_best) > 30]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break

        a_try = resample_sequence(best_seq, n_try)
        # Quick local search
        alloc = min(remaining * 0.15, 40.0)
        if alloc > 10:
            a_try = fast_local_search(a_try, time_budget=alloc)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse spectral restarts ===
    print(f"\n=== Phase 6: Spectral Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = r % 7
        a_r = spectral_init(n_best, rng, mode=mode)
        # Blend with best
        blend = 0.4
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        # Local search
        alloc = min(remaining * 0.15, 60.0)
        if alloc > 15:
            a_r = fast_local_search(a_r, time_budget=alloc)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Intensive final polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Fast local search (long)
    if remaining > 60:
        alloc = min(remaining * 0.3, 200.0)
        a_fp = fast_local_search(best_seq, time_budget=alloc, verbose=True)
        obj_fp = compute_obj(a_fp)
        if obj_fp < best_score:
            best_score = obj_fp
            best_seq = a_fp.copy()
            print(f"  Final FLS: obj={best_score:.8f} **")

    # Fine grid search
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.25, 120.0)
        a_fgs = fine_grid_search(best_seq, time_budget=alloc, verbose=True)
        obj_fgs = compute_obj(a_fgs)
        if obj_fgs < best_score:
            best_score = obj_fgs
            best_seq = a_fgs.copy()
            print(f"  Final FGS: obj={best_score:.8f} **")

    # LP polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 80.0)
        a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # One more round of fine search
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 60.0)
        a_final = fine_grid_search(best_seq, time_budget=alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Ultra-final FGS: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
