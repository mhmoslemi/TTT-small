```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: Unconstrained L-BFGS on log-parameterized simplex (softmax space)
with FFT-based gradients and adaptive tau schedule, combined with LP polishing,
multi-length exploration, and parametric initialization for diverse restarts.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n * max(conv) / (sum)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def softmax(x):
    """Numerically stable softmax."""
    x = x - np.max(x)
    e = np.exp(x)
    return e / e.sum()


def logsumexp(vals):
    """Numerically stable log-sum-exp."""
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(vals - m)))


def smoothed_obj_x(x, tau):
    """Smoothed objective in parameter space x (where a = softmax(x))."""
    a = softmax(x)
    conv = compute_conv(a)
    return float(logsumexp(tau * conv) / tau)


def smoothed_grad_x(x, tau):
    """Gradient of smoothed objective in parameter space."""
    a = softmax(x)
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    exp_vals = np.exp(tau * (conv - max_c))
    p = exp_vals / float(np.sum(exp_vals))

    # Compute grad_a[j] = 2 * sum_d p[j+d] * a[d] using FFT
    p_rev = p[::-1]
    size = 1
    while size < (2 * n - 1) + n - 1:
        size <<= 1
    P = fft(p_rev, size)
    A = fft(a, size)
    conv_pa = ifft(P * A).real[:2 * n - 1]
    C = conv_pa[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * C

    # Convert to x-space gradient: dF/dx[j] = a[j] * (grad_a[j] - dot(a, grad_a))
    dot_ga = float(np.dot(grad_a, a))
    grad_x = a * (grad_a - dot_ga)
    return grad_x


def lbfgs_multitau(a_init, time_budget=300.0, verbose=True, seed=42):
    """
    L-BFGS optimization with increasing tau schedule.
    Optimizes in the log-parameterized simplex (softmax space).
    """
    start = time.monotonic()
    a = np.maximum(a_init, 1e-15).copy().astype(np.float64)
    s = float(np.sum(a))
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    # Adaptive tau schedule
    tau_schedule = [3, 5, 8, 12, 20, 35, 60, 100, 160, 250, 400, 600, 900, 1200, 1600]
    # Dynamic max iterations based on remaining time
    for tau in tau_schedule:
        if time.monotonic() - start > time_budget - 5:
            break

        x0 = np.log(a)

        def fun(x):
            return smoothed_obj_x(x, tau)

        def grad(x):
            return smoothed_grad_x(x, tau)

        # Adaptive max iterations: more for smaller tau, fewer for larger
        if time.monotonic() - start < time_budget * 0.3:
            max_iter = 800
        elif time.monotonic() - start < time_budget * 0.6:
            max_iter = 400
        else:
            max_iter = 200

        try:
            result = minimize(fun, x0, jac=grad, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'gtol': 1e-14,
                                     'ftol': 1e-16, 'iprint': -1})
            x_opt = result.x
        except Exception:
            x_opt = x0

        a_new = softmax(x_opt)
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    L-BFGS tau={tau:.0f}: obj={best_obj:.8f}, t={elapsed:.1f}s")
        else:
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    L-BFGS tau={tau:.0f}: obj={obj_new:.8f} (no improve), t={elapsed:.1f}s")

        # Early stopping if objective is very good
        if best_obj < 1.45:
            break

    return best_a


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_sparse(a_norm):
    """Build sparse convolution matrix A where A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for i in range(n):
            k = i + j
            if k < 2 * n - 1:
                rows.append(k)
                cols.append(j)
                vals.append(a_norm[i])
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_minpeak(a, M_target=None, sparse=True):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / np.sum(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()

    if sparse:
        A = build_conv_matrix_sparse(a_norm)
    else:
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            r = min(n, 2 * n - 1 - j)
            A[j:j + r, j] = a_norm[:r]

    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue

    return None, None


def golden_section_search(f, a_lo, a_hi, tol=0.01, max_iter=40):
    """Golden section search for minimum of f on [a_lo, a_hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if abs(a_hi - a_lo) < tol:
            break
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = f(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = f(d)
    return (a_lo + a_hi) / 2.0


def lp_polish(a, time_budget=200.0, verbose=False, seed=42):
    """LP-based polishing with golden-section line search and constraint tightening."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    no_improve = 0
    last_t = 0.5

    while time.monotonic() - start < time_budget - 5:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 10:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        def blend_obj(t):
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return 1e10
            a_try = a_try / s_try
            return compute_obj(a_try)

        t_opt = golden_section_search(blend_obj, 0.005, 1.0, tol=0.02, max_iter=30)

        a_try = (1.0 - t_opt) * best_a + t_opt * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        iter_count += 1

        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a.copy()
            last_t = t_opt
            no_improve = 0
            if verbose and iter_count % 3 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={t_opt:.3f}, M={M_lp:.6f}")
        else:
            no_improve += 1
            if no_improve == 2:
                t_small = max(0.005, last_t * 0.15)
                a_try2 = (1.0 - t_small) * best_a + t_small * g_lp
                a_try2 = np.maximum(a_try2, 0)
                a_try2 = a_try2 / a_try2.sum()
                obj_try2 = compute_obj(a_try2)
                if obj_try2 < best_obj - 1e-14:
                    best_obj = obj_try2
                    best_a = a_try2.copy()
                    a = best_a.copy()
                    no_improve = 0
            elif no_improve >= 3:
                # Try constraint tightening
                if M_lp is not None:
                    for tight in [0.97, 0.94, 0.90]:
                        g_t, _ = solve_lp_minpeak(a, M_target=M_lp * tight)
                        if g_t is not None:
                            for t_try in [0.5, 1.0]:
                                a_try3 = (1.0 - t_try) * best_a + t_try * g_t
                                a_try3 = np.maximum(a_try3, 0)
                                a_try3 = a_try3 / a_try3.sum()
                                obj_try3 = compute_obj(a_try3)
                                if obj_try3 < best_obj - 1e-14:
                                    best_obj = obj_try3
                                    best_a = a_try3.copy()
                                    a = best_a.copy()
                                    no_improve = 0
                                    if verbose:
                                        print(f"    LP-tight it{iter_count}: obj={best_obj:.8f}")
                                    break
                            if no_improve == 0:
                                break

                if no_improve >= 5:
                    # Perturb
                    scale = 0.005 * min(no_improve, 10)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    no_improve = 0
                    if iter_count > 40:
                        break

    return best_a


def subgradient_refine(a, time_budget=60.0, K=5):
    """Subgradient method with top-K peaks and momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    momentum = 0.75
    v = np.zeros(n)
    stall = 0

    while time.monotonic() - start < time_budget - 3:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        weights = np.exp((peak_vals - max_pv) * 5.0 / max_pv)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break
            else:
                lr *= 0.997

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def parametric_init(n, rng):
    """Generate diverse parametric initializations."""
    x = np.linspace(0, 1, n)
    mode = rng.integers(0, 8)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = 1.0 / (1.0 + 4.0 * np.sin(np.pi * x) ** 2)
    elif mode == 2:
        c = rng.uniform(0.3, 0.7)
        w = rng.uniform(0.1, 0.3)
        a = np.exp(-((x - c) ** 2) / (2 * w ** 2))
    elif mode == 3:
        f = rng.uniform(1, 4)
        amp = rng.uniform(0.05, 0.3)
        a = 1.0 + amp * np.cos(2 * np.pi * f * x + rng.uniform(0, 6.28))
    elif mode == 4:
        f1 = rng.uniform(1, 3)
        f2 = rng.uniform(3, 7)
        a = 1.0 + 0.15 * np.cos(2 * np.pi * f1 * x) + 0.1 * np.cos(2 * np.pi * f2 * x)
    elif mode == 5:
        # Hann-like window
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif mode == 6:
        # Blackman-like
        a = 0.42 + 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    else:
        a = rng.uniform(0.5, 1.5, n)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: L-BFGS on smoothed objective with adaptive tau,
    LP polishing, multi-length exploration, and parametric restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: L-BFGS with adaptive tau (main workhorse) ===
    print(f"\n=== Phase 1: L-BFGS Smoothed Optimization ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 500.0)
    if p1_budget > 30:
        a_lb = lbfgs_multitau(best_seq, time_budget=p1_budget, verbose=True, seed=seed)
        obj_lb = compute_obj(a_lb)
        if obj_lb < best_score:
            best_score = obj_lb
            best_seq = a_lb.copy()
            print(f"  Phase1 L-BFGS: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 L-BFGS: obj={obj_lb:.8f} (no improve)")

    # === Phase 2: LP Polishing ===
    print(f"\n=== Phase 2: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 300.0)
    if p2_budget > 30:
        a_lp = lp_polish(best_seq, time_budget=p2_budget, verbose=True, seed=seed)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f}")

    # === Phase 3: Second L-BFGS pass (finer tau) ===
    print(f"\n=== Phase 3: Second L-BFGS Pass ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 180.0)
    if p3_budget > 20:
        # Fine-tune with high tau only
        a_fine = best_seq * np.exp(rng.normal(0, 0.003, len(best_seq)))
        a_fine = np.maximum(a_fine, 0)
        a_fine = a_fine / a_fine.sum()
        a_fine = lbfgs_multitau(a_fine, time_budget=p3_budget, verbose=False)
        obj_fine = compute_obj(a_fine)
        if obj_fine < best_score:
            best_score = obj_fine
            best_seq = a_fine.copy()
            print(f"  Phase3 L-BFGS2: obj={best_score:.8f} **")

    # === Phase 4: Subgradient refinement ===
    print(f"\n=== Phase 4: Subgradient Refine ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.07, 60.0)
    if p4_budget > 10:
        a_sg = subgradient_refine(best_seq, time_budget=p4_budget, K=8)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Phase4 SG: obj={best_score:.8f} **")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 150), n_current + 100, n_current + 250,
        n_current + 400, 1200, 1500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2200 and abs(n - n_current) > 80]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)

        # L-BFGS on new length
        alloc1 = min(remaining * 0.25, 80.0)
        if alloc1 > 15:
            a_try = lbfgs_multitau(a_try, time_budget=alloc1, verbose=False)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.2, 80.0)
        if alloc2 > 15:
            a_try = lp_polish(a_try, time_budget=alloc2, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Parametric restarts + quick optimization ===
    print(f"\n=== Phase 6: Parametric Restarts ===")
    remaining = deadline - time.monotonic()

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_p = parametric_init(n_current, rng)
        alloc = min(remaining * 0.2, 60.0)
        if alloc > 15:
            a_p = lbfgs_multitau(a_p, time_budget=alloc, verbose=False)
            obj_p = compute_obj(a_p)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Parametric {r}: obj={best_score:.8f} **")
            else:
                print(f"  Parametric {r}: obj={obj_p:.8f}")

    # === Phase 7: Final intensive LP + L-BFGS ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc_lp = min(remaining * 0.4, 150.0)
        if alloc_lp > 20:
            a_final = lp_polish(best_seq, time_budget=alloc_lp, verbose=True, seed=777)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc_lb = min(remaining * 0.4, 100.0)
            a_final2 = lbfgs_multitau(best_seq, time_budget=alloc_lb, verbose=False)
            obj_final2 = compute_obj(a_final2)
            if obj_final2 < best_score:
                best_score = obj_final2
                best_seq = a_final2.copy()
                print(f"  Final L-BFGS: obj={best_score:.8f} **")

    # Final quick subgradient
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_sg = subgradient_refine(best_seq, time_budget=min(remaining * 0.4, 30.0), K=10)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
