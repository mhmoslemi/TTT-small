def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smooth_gradient(a, tau):
    """Compute gradient of smoothed max of conv(a,a) using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    # Use log-sum-exp for numerical stability
    exp_vals = np.exp(tau * (conv - max_conv))
    weights = exp_vals / np.sum(exp_vals)
    # Gradient of log-sum-exp(tau * conv) = tau * weights
    # d(conv_k)/d(a_i) = a_{k-i} + a_{k-i} (for i != k-i) or 2*a_i (for i = k-i)
    # Using FFT: gradient = 2 * ifft(weights_padded * conj(a_padded))[:n]
    N = 1
    while N < 3*n:
        N <<= 1
    W = fft(weights, N)
    A = fft(a, N)
    grad = 2.0 * ifft(W * np.conj(A)).real[:n]
    return grad, conv, max_conv

def nesterov_smoothed_descent(a, time_budget, tau_start, tau_end):
    """Nesterov accelerated gradient descent with adaptive tau."""
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    lr = 1.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

            v = np.zeros(n)

        lr *= 0.9995

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def make_window(n, window_type, param=None):
    """Generate parametric window functions."""
    x = np.arange(n)
    if window_type == 'hann':
        w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    elif window_type == 'hamming':
        w = 0.54 + 0.46 * np.cos(2 * np.pi * x / (n - 1))
    elif window_type == 'blackman':
        w = 0.42 + 0.5 * np.cos(2 * np.pi * x / (n - 1)) + 0.08 * np.cos(4 * np.pi * x / (n - 1))
    elif window_type == 'kaiser':
        alpha = param if param is not None else 4.0
        if n == 1:
            w = np.ones(1)
        else:
            from scipy.special import i0
            t = 2 * x / (n - 1) - 1
            w = i0(alpha * np.sqrt(np.maximum(1 - t**2, 0))) / i0(alpha)
    elif window_type == 'exp':
        # Exponential taper
        c = param if param is not None else 0.5
        w = np.exp(-c * (2 * x / (n - 1) - 1) ** 2)
    elif window_type == 'raised_cos2':
        # Double raised cosine
        w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1))) ** 2
    elif window_type == 'optimal_shape':
        # Theoretical optimal: minimize peak autocorrelation
        # Approximation: w(x) = 1 - eps * T_2(2x-1) where T_2 is Chebyshev
        # T_2(t) = 2t^2 - 1
        t = 2 * x / (n - 1) - 1
        w = 1 + 0.3 * t ** 2  # Convex shape (more in middle)
        w = np.maximum(w, 0.01)
    else:
        w = np.ones(n)
    w = np.maximum(w, 0)
    s = w.sum()
    if s > 0:
        w = w / s
    return w


def nesterov_smoothed_descent(a, time_budget, tau_start=5.0, tau_end=200.0, verbose=True):

    Nesterov accelerated gradient descent on log-sum-exp smoothed max of conv(a,a).
    Uses plateau-grow tau schedule for better convergence.

    if S < 1e-10:
        return a
    a = a / S

    best_obj = compute_obj(best_a)
    start = time.monotonic()

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_grad = None

    best_it = 0

    while N < 3 * n:
        N <<= 1

        frac = min(1.0, elapsed / time_budget)
        # Plateau-grow schedule: stay at tau_start for first 30%, then grow
        if frac < 0.3:
            tau = tau_start
        else:
            growth_frac = (frac - 0.3) / 0.7
            tau = tau_start * (tau_end / tau_start) ** (growth_frac ** 0.8)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = y.sum()

        if prev_grad is not None and it % 2 == 0:
            da = y - prev_a
            dg = grad - prev_grad
            ss = float(np.dot(da, da))
            sy = float(np.dot(da, dg))
            if ss > 1e-14 and abs(sy) > 1e-12:
                lr_bb = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(lr_bb, 1e-8), 10.0 / np.sqrt(n))

        # Nesterov update
        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_grad = grad.copy()
        a = a_new
        it += 1

        # Restart if stuck
        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)
            beta = min(beta + 0.01, 0.95)
            best_it = it

        if verbose and it % 500 == 0:
            print(f"    NSD it={it}, tau={tau:.1f}, lr={lr:.4f}, obj={best_obj:.8f}")

    return best_a


def active_set_coord_descent(a, K=30, time_budget=200.0, verbose=True):
    """
    Fast coordinate descent using active set tracking.
    For each coordinate, only consider top-K convolution values.
    Uses golden section search in 1D.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0

    conv = compute_conv(a)
    S_full = a.sum()

    sweep = 0
    while time.monotonic() - start < time_budget:
        sweep += 1
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                return best_a

            a_i = a[i]
            if a_i < 1e-10:
                # Try setting a small value
                for trial in [1.0 / n, 0.5 / n, 2.0 / n]:
                    a[i] = trial
                    S_try = S_full + trial - a_i  # This is wrong, let me recompute
                    # Actually S doesn't change if we're doing it right
                    # Let me just recompute
                    break
                continue

            # Golden section search for optimal a[i]
            # The range: [0, a_i * 3] (don't go too far)
            lo = 0.0
            hi = min(a_i * 3.0, 1.0)

            # Evaluate at a few points to narrow down
            best_local = a_i
            best_local_obj = best_obj

            for _ in range(12):
                m1 = hi - 0.618 * (hi - lo)
                m2 = lo + 0.618 * (hi - lo)

                # Evaluate objective at a[i] = m1
                a[i] = m1
                S1 = a.sum()
                if S1 > 0.01:
                    a1_norm = a / S1
                    conv1 = compute_conv(a1_norm)
                    obj1 = 2.0 * n * float(np.max(conv1))
                else:
                    obj1 = np.inf

                # Evaluate objective at a[i] = m2
                a[i] = m2
                S2 = a.sum()
                if S2 > 0.01:
                    a2_norm = a / S2
                    conv2 = compute_conv(a2_norm)
                    obj2 = 2.0 * n * float(np.max(conv2))
                else:
                    obj2 = np.inf

                if obj1 < obj2:
                    hi = m2
                else:
                    lo = m1

            # Evaluate at the best point
            a[i] = (lo + hi) / 2
            S_best = a.sum()
            if S_best > 0.01:
                a_best_norm = a / S_best
                conv_best = compute_conv(a_best_norm)
                obj_best = 2.0 * n * float(np.max(conv_best))
            else:
                obj_best = np.inf

            if obj_best < best_obj - 1e-13:
                best_obj = obj_best
                best_a = a.copy()
                improved = True
            else:
                a[i] = a_i  # Restore
            conv = compute_conv(a / a.sum())
            S_full = a.sum()

        if not improved:
            stall += 1
            if stall > 3:
                break
        else:
            stall = 0

        if verbose and sweep % 2 == 0:
            print(f"    ASD sweep={sweep}, obj={best_obj:.8f}, improved={improved}")

    return best_a


def fast_coord_descent_increm(a, time_budget=300.0, verbose=True):
    """
    Coordinate descent with incremental convolution updates.
    When changing a_i by delta, update conv incrementally.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    best_conv = compute_conv(best_a)
    start = time.monotonic()
    rng = np.random.default_rng(77)
    stall = 0

    sweep = 0
    while time.monotonic() - start < time_budget:
        sweep += 1
        improved = False
        order = rng.permutation(n)
        conv = compute_conv(a)
        S_cur = a.sum()
        peak = float(np.max(conv))

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            a_i = a[i]
            if a_i < 1e-12:
                continue

            # Try a few candidate values for a[i]
            # The change in conv[k] when a[i] changes by delta is:
            # conv[k] += 2 * delta * a[k-i] - delta^2 (if k = 2i)
            # More precisely: conv[k] = sum_j a[j]*a[k-j]
            # d(conv[k])/d(a[i]) = 2*a[k-i] (for valid k-i)
            # So new_conv[k] = conv[k] + 2*delta*a[k-i] (approximately, ignoring delta^2)

            # Try golden section with fast evaluation
            lo = 0.0
            hi = min(a_i * 3.0, 1.0)

            # Quick scan to find approximate optimum
            fractions = [0.3, 0.5, 0.7, 1.0, 0.1, 0.9, 0.2, 0.4, 0.6, 0.8]
            best_frac = 1.0
            best_fast_obj = 2.0 * n * peak / (S_cur * S_cur)

            for frac in fractions:
                delta = (frac - 1.0) * a_i
                if a_i + delta < 0:
                    continue
                # Approximate new peak
                # conv[k] changes by 2*delta*a[k-i]
                # We only need to check if the peak changes significantly
                k_peak = int(np.argmax(conv))
                # New conv at peak position
                offset = k_peak - i
                if 0 <= offset < n:
                    new_peak_approx = conv[k_peak] + 2 * delta * a[offset]
                else:
                    new_peak_approx = conv[k_peak]
                obj_approx = 2.0 * n * max(new_peak_approx, 0) / (S_cur + delta) ** 2
                if obj_approx < best_fast_obj:
                    best_fast_obj = obj_approx
                    best_frac = frac

            # Fine-tune around best_frac
            delta_best = (best_frac - 1.0) * a_i
            # Try a few small perturbations
            for pert in [-0.05, -0.02, 0.0, 0.02, 0.05]:
                trial = a_i * best_frac + pert * a_i
                if trial < 0:
                    continue
                delta = trial - a_i
                a[i] = trial
                S_try = S_cur + delta
                if S_try < 0.01:
                    a[i] = a_i
                    continue
                a_norm = a / S_try
                conv_try = compute_conv(a_norm)
                obj_try = 2.0 * n * float(np.max(conv_try))
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a.copy()
                    conv = conv_try
                    S_cur = S_try
                    a_i = trial
                    peak = float(np.max(conv))
                    improved = True
            a[i] = best_a[i] if improved else a_i
            if time.monotonic() - start > time_budget:
                break

        if not improved:
            stall += 1
            if stall > 5:
                break
        else:
            stall = 0

        if verbose and sweep % 3 == 0:
            print(f"    FCD sweep={sweep}, obj={best_obj:.8f}")

    return best_a


def lp_polish(a, time_budget=100.0, verbose=False):
    """
    LP-based polishing: solve min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.
    Then blend and line-search.
    """
    n = len(a)
    try:
        from scipy.optimize import linprog
    except:
        return a

    a = np.maximum(a, 0).astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        return a
    a_norm = a / S

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    start = time.monotonic()

    # Build constraint matrix: conv(a_norm, g)[k] = sum_j A[k,j] * g[j] <= M
    # A[k,j] = a_norm[k-j] if 0<=k-j<n else 0
    # Using scipy's linprog: minimize c^T x s.t. A_ub x <= b_ub, A_eq x = b_eq, x >= 0
    # We want: min M s.t. A g - M <= 0, sum(g) = 1, g >= 0
    # Variables: [g_0, ..., g_{n-1}, M]
    # Minimize: [0, ..., 0, 1]
    
    m = 2 * n - 1  # number of convolution constraints
    A_full = np.zeros((m, n))
    for j in range(n):
        for k in range(m):
            idx = k - j
            if 0 <= idx < n:
                A_full[k, j] = a_norm[idx]

    # Sparse representation would be better, but for n=800, dense is OK
    # Actually for n=800, A is 1599 x 800 = 1.28M entries. That's fine.
    
    # linprog: min c^T x s.t. A_ub @ x <= b_ub, A_eq @ x = b_eq, bounds
    # x = [g (n vars), M (1 var)]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M

    # A_ub: for each k, sum_j A[k,j] g[j] - M <= 0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A_full
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)

    # A_eq: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    it = 0
    while time.monotonic() - start < time_budget:
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.status != 0 or res.x is None:
                break
            g_val = np.maximum(res.x[:n], 0)
            Sg = g_val.sum()
            if Sg < 1e-10:
                break
            g_val = g_val / Sg
        except:
            break

        # Golden section line search
        lo, hi = 0.0, 1.0
        phi = (np.sqrt(5) - 1) / 2
        for _ in range(8):
            t1 = hi - phi * (hi - lo)
            t2 = lo + phi * (hi - lo)
            a1 = np.maximum((1 - t1) * a + t1 * g_val, 0)
            s1 = a1.sum()
            obj1 = compute_obj(a1) if s1 > 1e-10 else np.inf
            a2 = np.maximum((1 - t2) * a + t2 * g_val, 0)
            s2 = a2.sum()
            obj2 = compute_obj(a2) if s2 > 1e-10 else np.inf
            if obj1 < obj2:
                hi = t2
            else:
                lo = t1

        t_best = (lo + hi) / 2
        a_new = np.maximum((1 - t_best) * a + t_best * g_val, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        it += 1
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if verbose and it % 3 == 0:
                print(f"    LP it={it}, obj={best_obj:.8f}")
        else:
            # Perturb
            rng = np.random.default_rng(it)
            a = best_a * np.exp(rng.normal(0, 0.03, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            if it > 20:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        best_obj = compute_obj(best_a)

    # Ensure non-negative

    elif val == np.inf:

        best_obj = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")
    return result
