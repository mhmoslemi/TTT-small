"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous:
1. U-shaped parametric initializations exploiting the insight that optimal sequences
   should be asymmetric in a specific way (high at edges to reduce middle peak).
2. Geometric blending (log-space interpolation) as alternative to arithmetic blending.
3. Dual-variable-guided subgradient refinement.
4. Faster LP with pre-allocated problems and warm-start-like behavior.
5. Adaptive multi-n exploration with quick screening.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_fast(a, solver_order=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal, dual_vals) or (None, None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if solver_order is None:
        solver_order = ['MOSEK', 'CLARABEL', 'ECOS', 'SCS', 'PDLP']

    solver_map = {
        'MOSEK': cp.MOSEK,
        'CLARABEL': cp.CLARABEL,
        'ECOS': cp.ECOS,
        'SCS': cp.SCS,
        'PDLP': cp.PDLP,
    }

    for sname in solver_order:
        try:
            solver = solver_map[sname]
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Try to extract dual values
                    dual_vals = None
                    try:
                        # The first constraint is A @ g <= M, dual is on that
                        dual_vals = prob.constraints[0].dual_value
                        if dual_vals is not None:
                            dual_vals = np.asarray(dual_vals).flatten()
                            if len(dual_vals) != 2 * n - 1:
                                dual_vals = None
                    except Exception:
                        dual_vals = None
                    return g_val, float(M.value), dual_vals
        except Exception:
            continue
    return None, None, None


def u_shaped_init(n, c, p=2.0):
    """
    Generate U-shaped initialization: a[i] ∝ (1 + c * ((2i/(n-1) - 1))^p)
    c > 0 gives U-shape (high at edges), c < 0 gives inverted U (high in middle).
    p controls the shape (p=2 is parabolic, p=1 is linear).
    """
    if n == 1:
        return np.array([1.0])
    x = 2.0 * np.arange(n) / (n - 1) - 1.0  # from -1 to 1
    a = 1.0 + c * np.abs(x) ** p
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        return u_shaped_init(n, 0.5, 2.0)
    elif mode == 2:
        return u_shaped_init(n, 1.0, 2.0)
    elif mode == 3:
        return u_shaped_init(n, 2.0, 2.0)
    elif mode == 4:
        return u_shaped_init(n, 1.0, 1.0)
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 6:
        # Exponential edge concentration
        a = np.exp(2.0 * np.abs(2.0 * np.arange(n) / (n - 1) - 1.0))
        return a / a.sum()
    elif mode == 7:
        # Sharp U
        a = np.abs(2.0 * np.arange(n) / (n - 1) - 1.0) ** 4 + 0.1
        return a / a.sum()
    elif mode == 8:
        # Random
        a = rng.uniform(0.1, 1.0, n)
        return a / a.sum()
    else:
        return np.ones(n) / n


def lp_polish_geometric(a, time_budget, verbose=False, solver_order=None):
    """
    LP polishing with geometric blending: a_new ∝ a^alpha * g^(1-alpha).
    This is done in log-space for numerical stability.
    Adaptive alpha: start with large (bold moves), decrease over time.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    s = float(np.sum(a))
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    iter_count = 0
    log_a = np.log(a)
    rng = np.random.default_rng(123)
    consecutive_fails = 0
    last_alpha = 0.5

    while time.monotonic() - start < time_budget:
        g_lp, M_lp, dual_vals = solve_lp_fast(a, solver_order)
        if g_lp is None:
            break

        log_g = np.log(np.maximum(g_lp, 1e-15))

        # Adaptive alpha: cool down over time
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1.0))
        # Alpha goes from 0.3 (bold) to 0.8 (conservative)
        base_alpha = 0.3 + 0.5 * progress

        # Try multiple alpha values
        candidates = [
            max(0.05, base_alpha - 0.2),
            base_alpha,
            min(0.95, base_alpha + 0.1),
            last_alpha,
        ]
        candidates = [c for c in candidates if 0.05 <= c <= 0.95]
        candidates = sorted(set([round(c, 3) for c in candidates]))

        best_alpha = last_alpha
        best_obj_local = best_obj
        best_a_local = best_a.copy()

        for alpha in candidates:
            # Geometric blend in log-space: log(a_new) = alpha*log_a + (1-alpha)*log_g
            log_new = alpha * log_a + (1.0 - alpha) * log_g
            # Normalize: a_new = exp(log_new) / sum(exp(log_new))
            log_new_max = float(np.max(log_new))
            a_new_unnorm = np.exp(log_new - log_new_max)
            s_new = float(np.sum(a_new_unnorm))
            if s_new < 1e-10:
                continue
            a_try = a_new_unnorm / s_new
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_local:
                best_obj_local = obj_try
                best_alpha = alpha
                best_a_local = a_try.copy()

        iter_count += 1
        if best_obj_local < best_obj - 1e-13:
            best_obj = best_obj_local
            best_a = best_a_local.copy()
            a = best_a
            log_a = np.log(np.maximum(a, 1e-15))
            last_alpha = best_alpha
            consecutive_fails = 0
            if verbose and iter_count % 10 == 0:
                print(f"    LPlip-geo it{iter_count}: obj={best_obj:.8f}, alpha={best_alpha:.3f}, M={M_lp:.6f}")
        else:
            consecutive_fails += 1
            # Perturb in log-space
            scale = 0.02 + 0.01 * min(consecutive_fails, 10)
            log_a = log_a + rng.normal(0, scale, n)
            a = np.exp(log_a - np.max(log_a))
            a = a / a.sum()
            if consecutive_fails > 30:
                scale = 0.08
                log_a = log_a + rng.normal(0, scale, n)
                a = np.exp(log_a - np.max(log_a))
                a = a / a.sum()
                consecutive_fails = 0
            if consecutive_fails > 60:
                break

    return best_a

def dualguided_subgradient(a, time_budget, K=15, verbose=False, solver_order=None):
    """
    Subgradient descent guided by LP dual variables.
    Uses the dual to weight which peaks to focus on.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.5 / np.sqrt(n)
    momentum = 0.5
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Try to get dual info from LP
        g_lp, M_lp, dual_vals = solve_lp_fast(a, solver_order)

        if dual_vals is not None and len(dual_vals) == 2 * n - 1:
            # Use dual to weight the peaks
            dual_pos = np.maximum(dual_vals, 0)
            dual_sum = float(np.sum(dual_pos))
            if dual_sum > 1e-10:
                weights = dual_pos / dual_sum
            else:
                top_idx = np.argsort(conv)[::-1][:K]
                weights = np.zeros(2 * n - 1)
                pv = conv[top_idx]
                weights[top_idx] = np.exp(3.0 * (pv - pv[0]) / (pv[0] + 1e-15))
                weights /= weights.sum()
        else:
            # Fallback: top-K peaks with softmax weighting
            top_idx = np.argsort(conv)[::-1][:K]
            weights = np.zeros(2 * n - 1)
            pv = conv[top_idx]
            weights[top_idx] = np.exp(3.0 * (pv - pv[0]) / (pv[0] + 1e-15))
            weights /= weights.sum()

        # Compute subgradient
        g = np.zeros(n)
        nz_weights = np.nonzero(weights)[0]
        for k in nz_weights:
            w = weights[k]
            if w < 1e-8:
                continue
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        v = momentum * v + g
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break

        a_new = a - lr * v / v_norm
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            momentum = min(0.8, momentum + 0.01)
        else:
            stall += 1
            a = best_a
            if stall % 20 == 0:
                lr *= 0.7
                v *= 0.3
                momentum = max(0.3, momentum - 0.05)
            if lr < 1e-12:
                break

        it += 1
        if verbose and it % 200 == 0:
            print(f"    DualSG it{it}: obj={best_obj:.8f}, lr={lr:.6f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.45, 500.0)

    if p2_budget > 30:
        # Multiple rounds with perturbation
        num_rounds = 5
        per_round = p2_budget / num_rounds
        for rnd in range(num_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(per_round * (1.0 - 0.1 * rnd), remaining * 0.4)
            if alloc < 10:
                break

            a_lp = lp_polish_arithmetic(best_seq, time_budget=alloc, verbose=(rnd == 0))
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP-arith round {rnd+1}: obj={best_score:.8f} **")
            else:
                print(f"  LP-arith round {rnd+1}: obj={obj_lp:.8f}")
                # Perturb for next round
                if rnd < num_rounds - 1:
                    scale = 0.01 * (rnd + 1)
                    best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                    best_seq = np.maximum(best_seq, 0)
                    best_seq = best_seq / best_seq.sum()

    # Phase 3: Geometric LP polishing (15%)
    print(f"\n=== Phase 3: Geometric LP Polishing ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)

    if p3_budget > 20:
        a_geo = lp_polish_geometric(best_seq, time_budget=p3_budget, verbose=True)
        obj_geo = compute_obj(a_geo)
        if obj_geo < best_score:
            best_score = obj_geo
            best_seq = a_geo.copy()
            print(f"  Geometric LP: obj={best_score:.8f} **")
        else:
            print(f"  Geometric LP: obj={obj_geo:.8f}")

    # Phase 4: Dual-guided subgradient (10%)
    print(f"\n=== Phase 4: Dual-Guided Subgradient ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.10, 100.0)

    if p4_budget > 15:
        a_dg = dualguided_subgradient(best_seq, time_budget=p4_budget, K=15, verbose=True)
        obj_dg = compute_obj(a_dg)
        if obj_dg < best_score:
            best_score = obj_dg
            best_seq = a_dg.copy()
            print(f"  DualSG: obj={best_score:.8f} **")
        else:
            print(f"  DualSG: obj={obj_dg:.8f}")

    # Phase 5: Final intensive polish (remaining)
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Another round of arithmetic LP
    if remaining > 30:
        alloc = min(remaining * 0.35, 100.0)
        a_lp = lp_polish_arithmetic(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Geometric LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.25, 60.0)
        a_geo = lp_polish_geometric(best_seq, time_budget=alloc)
        obj_geo = compute_obj(a_geo)
        if obj_geo < best_score:
            best_score = obj_geo
            best_seq = a_geo.copy()
            print(f"  Final Geo: obj={best_score:.8f} **")

    # One more arithmetic LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 50.0)
        a_lp = lp_polish_arithmetic(best_seq, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Try a different n one last time
    remaining = deadline - time.monotonic()
    if remaining > 30 and len(best_seq) != 900:
        n_alt = 900 if len(best_seq) != 900 else 1000
        a_alt = resample_sequence(best_seq, n_alt)
        alloc = min(remaining * 0.3, 40.0)
        a_alt = lp_polish_arithmetic(a_alt, time_budget=alloc)
        obj_alt = compute_obj(a_alt)
        if obj_alt < best_score:
            best_score = obj_alt
            best_seq = a_alt.copy()
            print(f"  Alt n={n_alt}: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
