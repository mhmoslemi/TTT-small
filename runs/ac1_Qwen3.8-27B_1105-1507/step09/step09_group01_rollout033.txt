import numpy as np
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        # Candidate deltas
        deltas = [step, -step, 2.0 * step, -2.0 * step, 0.5 * step, -0.5 * step]
        if use_momentum and abs(v[i]) > 1e-12:
            deltas.append(v[i] * step)

        for delta in deltas:
            new_ai = a[i] + delta
            if new_ai < -1e-12:
                continue
            new_ai = max(0.0, new_ai)
            delta = new_ai - a[i]
            if abs(delta) < 1e-15:
                continue

            # Compute new max efficiently
            # Affected slice: conv[i:i+n]
            new_slice = conv[i:i + n] + 2.0 * delta * a
            if 2 * i < 2 * n - 1:
                new_slice[i] += delta * delta  # self-term at position 2i in conv

            # Full new max: max of affected slice and unaffected parts
            new_max = float(new_slice.max())
            if i > 0:
                left_max = float(conv[:i].max())
                if left_max > new_max:
                    new_max = left_max
            end_pos = i + n
            if end_pos < 2 * n - 1:
                right_max = float(conv[end_pos:].max())
                if right_max > new_max:
                    new_max = right_max

            new_S = S + delta
            if new_S < 1e-10:
                continue
            new_obj = 2.0 * n * new_max / (new_S * new_S)

            if new_obj < best_new_obj:
                best_new_obj = new_obj
                best_delta = delta

        if best_delta != 0.0:
            # Accept the step
            conv[i:i + n] += 2.0 * best_delta * a
            if 2 * i < 2 * n - 1:
                conv[2 * i] += best_delta * best_delta
            a[i] += best_delta
            S += best_delta

            # Update momentum
            if use_momentum:
                momentum = 0.9
                v[i] = momentum * v[i] + best_delta
                # Occasionally decay all momentum
                if it % 500 == 0:
                    v *= 0.8

            best_obj = best_new_obj
            stall = 0
        else:
            stall += 1
            if stall > 2000:
                # Perturb and reset
                scale = 0.005 * min(stall / 2000.0, 3.0)
                a = a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                S = float(np.sum(a))
                a = a / S
                S = 1.0
                conv = compute_conv(a)
                best_obj = 2.0 * n * float(np.max(conv))
                stall = 0
                step = 0.02 / np.sqrt(n)
                v = np.zeros(n)

        # Track best
        cur_obj = 2.0 * n * float(np.max(conv)) / (S * S)
        if cur_obj < best_obj_val - 1e-14:
            best_obj_val = cur_obj
            best_a = a.copy()
            best_conv = conv.copy()
            best_S = S

        it += 1
        if verbose and it % 100000 == 0:
            print(f"      FCD it={it}: obj={best_obj_val:.8f}, step={step:.2e}, t={elapsed:.0f}s")

    # Normalize best
    S_best = float(np.sum(best_a))
    if S_best > 1e-10:
        best_a = best_a / S_best
    return best_a


def fast_pairwise_desc(a, time_budget, seed=42, verbose=False):
    """
    Pairwise coordinate descent: change two coordinates at once.
    More powerful than single-coordinate for escaping local optima.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        return a / n
    a = a / S
    S = 1.0
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv)) / (S * S)
    best_a = a.copy()
    best_obj_val = best_obj

    rng = np.random.default_rng(seed)
    step = 0.01 / np.sqrt(n)
    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = elapsed / time_budget
        step = (0.01 / np.sqrt(n)) * max(0.1, 1.0 - 0.6 * progress)

        # Find peak
        peak_idx = int(np.argmax(conv))
        p_lo = max(0, peak_idx - n + 1)
        p_hi = min(n - 1, peak_idx)

        # Pick two coordinates (one active, one random)
        i1 = int(rng.integers(p_lo, p_hi + 1))
        i2 = int(rng.integers(0, n))
        if i2 == i1:
            i2 = (i1 + 1) % n

        # Try various (d1, d2) combinations
        best_delta1, best_delta2 = 0.0, 0.0
        best_new_obj = best_obj

        for s1 in [1, -1]:
            for s2 in [1, -1]:
                d1 = s1 * step
                d2 = s2 * step
                if a[i1] + d1 < 0 or a[i2] + d2 < 0:
                    continue

                # Compute new conv max
                # Affected: conv[i1:i1+n] and conv[i2:i2+n]
                lo = min(i1, i2)
                hi = max(i1, i2) + n
                hi = min(hi, 2 * n - 1)

                new_slice = conv[lo:hi].copy()
                # Apply change at i1
                new_slice[i1 - lo:i1 - lo + n] += 2.0 * d1 * a
                if 2 * i1 < 2 * n - 1:
                    new_slice[2 * i1 - lo] += d1 * d1
                # Apply change at i2
                new_slice[i2 - lo:i2 - lo + n] += 2.0 * d2 * a
                if 2 * i2 < 2 * n - 1:
                    new_slice[2 * i2 - lo] += d2 * d2

                new_max = float(new_slice.max())
                if lo > 0:
                    lm = float(conv[:lo].max())
                    if lm > new_max:
                        new_max = lm
                if hi < 2 * n - 1:
                    rm = float(conv[hi:].max())
                    if rm > new_max:
                        new_max = rm

                new_S = S + d1 + d2
                if new_S < 1e-10:
                    continue
                new_obj = 2.0 * n * new_max / (new_S * new_S)

                if new_obj < best_new_obj:
                    best_new_obj = new_obj
                    best_delta1, best_delta2 = d1, d2

        if best_delta1 != 0.0 or best_delta2 != 0.0:
            # Apply
            if best_delta1 != 0:
                conv[i1:i1 + n] += 2.0 * best_delta1 * a
                if 2 * i1 < 2 * n - 1:
                    conv[2 * i1] += best_delta1 * best_delta1
                a[i1] += best_delta1
            if best_delta2 != 0:
                conv[i2:i2 + n] += 2.0 * best_delta2 * a
                if 2 * i2 < 2 * n - 1:
                    conv[2 * i2] += best_delta2 * best_delta2
                a[i2] += best_delta2
            S += best_delta1 + best_delta2
            best_obj = best_new_obj
            stall = 0
        else:
            stall += 1
            if stall > 3000:
                scale = 0.008 * min(stall / 3000.0, 3.0)
                a = a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                S = float(np.sum(a))
                a = a / S
                S = 1.0
                conv = compute_conv(a)
                best_obj = 2.0 * n * float(np.max(conv))
                stall = 0

        cur_obj = 2.0 * n * float(np.max(conv)) / (S * S)
        if cur_obj < best_obj_val - 1e-14:
            best_obj_val = cur_obj
            best_a = a.copy()

        it += 1
        if verbose and it % 50000 == 0:
            print(f"      FPD it={it}: obj={best_obj_val:.8f}, t={elapsed:.0f}s")

    S_best = float(np.sum(best_a))
    if S_best > 1e-10:
        best_a = best_a / S_best
    return best_a


def smoothed_gradient_fft(a, tau):
    """Compute gradient of smoothed max(conv) using FFT cross-correlation."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def nesterov_smoothed_descent(a, time_budget=300.0, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.95))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        if it % 2000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)

        # Barzilai-Borwein step size
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 20.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.99995)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 5.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def subgradient_refine(a, time_budget=100.0, K=8, verbose=False):
    """Subgradient method with top-K peaks, momentum, and adaptive step."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 3.0 / np.sqrt(n)
    momentum = 0.75
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        weights = np.exp((peak_vals - max_pv) * 8.0 / max_pv)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break
            else:
                lr *= 0.997

        it += 1

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, M_target=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    try:
        import cvxpy as cp
    except ImportError:
        return None, None

    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue

    return None, None


def lp_polish(a, time_budget=100.0, verbose=False, seed=42):
    """LP-based polishing with golden-section line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Golden section for blend
        def blend_obj(t):
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return 1e10
            a_try = a_try / s_try
            return compute_obj(a_try)

        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a_lo, a_hi = 0.01, 1.0
        c = a_hi - gr * (a_hi - a_lo)
        d = a_lo + gr * (a_hi - a_lo)
        fc, fd = blend_obj(c), blend_obj(d)
        for _ in range(25):
            if abs(a_hi - a_lo) < 0.02:
                break
            if fc < fd:
                a_hi, d, fd = d, c, fc
                c = a_hi - gr * (a_hi - a_lo)
                fc = blend_obj(c)
            else:
                a_lo, c, fc = c, d, fd
                d = a_lo + gr * (a_hi - a_lo)
                fd = blend_obj(d)
        t_opt = (a_lo + a_hi) / 2.0

        a_try = (1.0 - t_opt) * best_a + t_opt * g_lp
        a_try = np.maximum(a_try, 0)
        a_try = a_try / a_try.sum()
        obj_try = compute_obj(a_try)
        iter_count += 1

        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a.copy()
            no_improve = 0
            if verbose and iter_count % 3 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={t_opt:.3f}")
        else:
            no_improve += 1
            if no_improve >= 4:
                scale = 0.006 * min(no_improve, 6)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                if iter_count > 40:
                    break

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 7, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        #-arccos-like (sinc variant)
        x = np.linspace(0, np.pi, n)
        a = np.abs(np.sin(x) / (x + 1e-8) + 0.1)
        a = a / a.sum()
    elif mode == 7:
        # Bell-shaped
        c = n // 2
        w = n // 4
        a = np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining fast coordinate descent,
    pairwise descent, smoothed gradient, and LP polishing across multiple lengths.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_parent = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_parent) / n_parent
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_parent}): obj={best_score:.8f}")

    # === Phase 1: Fast coordinate descent on parent length ===
    print(f"\n=== Phase 1: Fast Coordinate Descent (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.15, 150.0)
    if p1_budget > 20:
        a1 = fast_coord_desc(best_seq, time_budget=p1_budget, seed=42, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1 FCD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 FCD: obj={obj1:.8f}")

    # Pairwise descent
    remaining = deadline - time.monotonic()
    p1b_budget = min(remaining * 0.10, 100.0)
    if p1b_budget > 15:
        a1b = fast_pairwise_desc(best_seq, time_budget=p1b_budget, seed=77, verbose=True)
        obj1b = compute_obj(a1b)
        if obj1b < best_score:
            best_score = obj1b
            best_seq = a1b.copy()
            print(f"  Phase1b FPD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1b FPD: obj={obj1b:.8f}")

    # === Phase 2: Multi-length exploration ===
    print(f"\n=== Phase 2: Multi-length Exploration ===")
    n_options = [1000, 1200, 1500, 1800, 2000]
    # Also try slightly different from parent
    n_options = [n for n in n_options if abs(n - len(best_seq)) > 50]
    n_options = sorted(set(n_options + [len(best_seq) + 200, len(best_seq) + 400]))
    n_options = [n for n in n_options if 600 <= n <= 2500]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)

        # Fast coordinate descent
        alloc1 = min(remaining * 0.20, 120.0)
        if alloc1 > 15:
            a_try = fast_coord_desc(a_try, time_budget=alloc1, seed=n_try, verbose=False)

        # Pairwise
        remaining = deadline - time.monotonic()
        alloc1b = min(remaining * 0.12, 60.0)
        if alloc1b > 10:
            a_try = fast_pairwise_desc(a_try, time_budget=alloc1b, seed=n_try + 1)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 3: Try different initializations at best length ===
    print(f"\n=== Phase 3: Diverse Initializations ===")
    n_best = len(best_seq)
    for mode in [1, 3, 4, 6]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        a_div = diverse_init(n_best, rng, mode)
        alloc = min(remaining * 0.15, 80.0)
        if alloc > 10:
            a_div = fast_coord_desc(a_div, time_budget=alloc, seed=200 + mode)
        obj_div = compute_obj(a_div)
        if obj_div < best_score:
            best_score = obj_div
            best_seq = a_div.copy()
            print(f"  ** mode={mode}: obj={best_score:.8f} **")
        else:
            print(f"  mode={mode}: obj={obj_div:.8f}")

    # === Phase 4: Smoothed gradient descent ===
    print(f"\n=== Phase 4: Smoothed Gradient Descent ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.20, 180.0)
    if p4_budget > 20:
        a4 = nesterov_smoothed_descent(best_seq, time_budget=p4_budget,
                                       tau_init=8.0, tau_end=400.0, verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4 NSD: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 NSD: obj={obj4:.8f}")

    # === Phase 5: LP polishing ===
    print(f"\n=== Phase 5: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.15, 120.0)
    if p5_budget > 20:
        a5 = lp_polish(best_seq, time_budget=p5_budget, verbose=True, seed=42)
        obj5 = compute_obj(a5)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  Phase5 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase5 LP: obj={obj5:.8f}")

    # === Phase 6: Subgradient refinement ===
    print(f"\n=== Phase 6: Subgradient Refinement ===")
    remaining = deadline - time.monotonic()
    p6_budget = min(remaining * 0.10, 80.0)
    if p6_budget > 15:
        a6 = subgradient_refine(best_seq, time_budget=p6_budget, K=8)
        obj6 = compute_obj(a6)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Phase6 SG: obj={best_score:.8f} **")
        else:
            print(f"  Phase6 SG: obj={obj6:.8f}")

    # === Phase 7: Final fast coordinate descent polish ===
    print(f"\n=== Phase 7: Final FCD Polish ===")
    remaining = deadline - time.monotonic()
    p7_budget = min(remaining * 0.25, 200.0)
    if p7_budget > 15:
        a7 = fast_coord_desc(best_seq, time_budget=p7_budget, seed=999, verbose=True)
        obj7 = compute_obj(a7)
        if obj7 < best_score:
            best_score = obj7
            best_seq = a7.copy()
            print(f"  Phase7 FCD: obj={best_score:.8f} **")
        else:
            print(f"  Phase7 FCD: obj={obj7:.8f}")

    # === Phase 8: Rapid try at new lengths ===
    print(f"\n=== Phase 8: Rapid Length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    for n_try in [n_current + 300, n_current + 500, max(600, n_current - 200)]:
        if remaining < 40:
            break
        if not (600 <= n_try <= 3000):
            continue
        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.20, 60.0)
        if alloc > 10:
            a_try = fast_coord_desc(a_try, time_budget=alloc, seed=n_try)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")
        remaining = deadline - time.monotonic()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
