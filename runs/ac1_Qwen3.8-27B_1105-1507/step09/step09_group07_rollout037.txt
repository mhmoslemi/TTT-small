"""
Optimized LP-alternating with simulated annealing, subgradient polishing,
and adaptive n-sweep for minimizing peak convolution of non-negative sequences.
Target: score < 1.5027.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def solve_lp_fast(a):
    """
    Fast LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Uses scipy linprog with HiGHS. Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build convolution matrix efficiently
    # A[k, i] = a_norm[k-i] if 0 <= k-i < n, else 0
    # A is (2n-1) x n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    
    # LP: min M s.t. A@g - M <= 0, sum(g) = 1, g >= 0, M free
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0  # A@g - M <= 0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0  # sum(g) = 1
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None

    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    stall = 0
    patience = 25
    temperature = temperature_start
    iteration = 0
    last_good_t = 0.5

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
            
        g_lp, M_lp = solve_lp_fast(a)
        if g_lp is None:
            break

        current_max_c = float(np.max(compute_conv(a)))
        
        # Check if we're at a fixed point
        diff_norm = np.linalg.norm(a - g_lp) / (np.linalg.norm(a) + 1e-15)
        if diff_norm < 1e-6:
            stall += 5
        else:
            stall += 0

        # Try step sizes: prefer t=1, then back off
        candidate_ts = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1]
        # Add adaptive step sizes around last_good_t
        if 0 < last_good_t < 1:
            candidate_ts.append(min(1.0, last_good_t * 1.5))
            candidate_ts.append(max(0.05, last_good_t * 0.7))
        candidate_ts = sorted(set(t for t in candidate_ts if 0 < t <= 1.0), reverse=True)

        best_t_obj = np.inf
        best_t_val = last_good_t
        best_a_local = a.copy()

        for t in candidate_ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
            # Early stop if we found a good improvement
            if obj_try < best_obj - 1e-8:
                break

        iteration += 1
        
        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local.copy()
            last_good_t = best_t_val
            stall = 0
            temperature = max(0.001, temperature * 0.97)
        else:
            stall += 1
            if stall > patience:
                # Annealing restart: structured perturbation
                rng = np.random.default_rng(iteration * 104729 + n)
                
                # Try different perturbation strategies
                strategy = iteration % 4
                if strategy == 0:
                    # Multiplicative noise
                    scale = min(0.4, 0.05 + 0.02 * (stall // patience))
                    a = best_a * np.exp(rng.normal(0, scale, n))
                elif strategy == 1:
                    # Move mass from peak-contributing positions
                    conv = compute_conv(best_a)
                    peak_lags = np.argsort(conv)[::-1][:5]
                    a = best_a.copy()
                    for _ in range(20):
                        i = rng.integers(n)
                        j = rng.integers(n)
                        if i != j and a[i] > 0.01:
                            delta = 0.05 * a[i]
                            a[i] -= delta
                            a[j] += delta
                    a = np.maximum(a, 0)
                elif strategy == 2:
                    # Blend with a different structured sequence
                    x = np.linspace(0, 1, n)
                    blend_seq = 1.0 + 0.5 * np.cos(2 * np.pi * rng.uniform(1, 5) * x)
                    blend_seq = np.maximum(blend_seq, 0.01)
                    blend_seq = blend_seq / blend_seq.sum()
                    mix = 0.2 + 0.1 * (stall // patience)
                    a = (1 - mix) * best_a + mix * blend_seq
                else:
                    # Random Dirichlet perturbation
                    alpha = 0.5 + 0.5 * (stall // patience)
                    a = rng.dirichlet(np.ones(n) * alpha)
                    a = (1 - 0.3) * best_a + 0.3 * a
                
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                stall = 0
                last_good_t = 0.5
                temperature = min(0.3, temperature * 3.0)
            else:
                temperature = max(0.005, temperature * 0.99)

        if verbose and iteration % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"        LP-annealed it{iteration}: obj={best_obj:.8f}, t={last_good_t:.2f}, "
                  f"stall={stall}, temp={temperature:.4f}, elapsed={elapsed:.0f}s")

    return best_a


def subgradient_polish(a, time_budget, verbose=False):
    """
    Subgradient method for polishing. Uses the subgradient of max_k(a*a)[k].
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 0.1
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Find active constraints (lags at max)
        tolerance = M * 0.01
        active = np.where(conv >= M - tolerance)[0]
        
        # Subgradient of M w.r.t. a[i]
        # dM/da[i] = (1/|active|) * sum_{k in active} sum_{j: i+j=k} a[j]
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            # a[i]*a[k-i] contributes to grad[i] and grad[k-i]
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]
                    if i != j:
                        grad[j] += a[i]
                    else:
                        grad[i] += a[i]  # a[i]^2 derivative is 2*a[i], counted once above
        grad /= len(active)
        
        # Actually, let's be more careful.
        # (a*a)[k] = sum_{i+j=k} a[i]*a[j]
        # d(a*a)[k]/da[i] = sum_{j: i+j=k} a[j] + sum_{j: j+i=k} a[j] (the two terms when i != j)
        # = 2 * sum_{j: i+j=k, j != i} a[j] + a[i] (if 2i = k)
        # But for the subgradient of max, we just need the gradient of each active (a*a)[k].
        
        # Let me redo this properly.
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                if 0 <= j < n:
                    # d(a*a)[k]/da[i] includes a[j] from the term a[i]*a[j]
                    # and a[i] from the term a[j]*a[i] (same term, but in convolution it's counted once)
                    # Actually (a*a)[k] = sum_i a[i]*a[k-i], so d/da[i] = a[k-i] (if 0 <= k-i < n)
                    # But also a[i] appears as a[k-i] when we set i' = k-i, giving d/da[i'] = a[i]
                    # So the total derivative is: a[k-i] + a[i] (if i != k-i, i.e., k != 2i)
                    # Or 2*a[i] if k = 2i
                    grad[i] += a[j]
                    if i != j:
                        grad[j] += a[i]
                    # When i == j (k = 2i), grad[i] gets a[i] from its own term
                    # which is already counted above
        grad /= float(len(active))
        
        # Objective gradient (with sum=1 normalization, the -2M/S term is constant)
        # df/da[i] = 2n * (grad[i] - 2*M) (for S=1)
        # We want to minimize, so move in negative gradient direction
        # But we need to maintain a >= 0
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            lr = min(lr * 1.05, 0.5)
        else:
            if obj_new < best_obj + 1e-10:
                a = a_new.copy()
            stall += 1
            lr = max(0.001, lr * 0.85)
        
        if stall > 50:
            break
        
        it += 1
        if verbose and it % 50 == 0:
            print(f"          Subgrad it{it}: obj={best_obj:.8f}, lr={lr:.4f}")

    return best_a


def spectral_balance(a, time_budget, verbose=False):
    """
    Try to balance the convolution spectrum by adjusting the sequence's frequency content.
    Uses the fact that flat convolution <=> specific spectral properties.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    for alpha in [0.05, 0.1, 0.2, 0.3, 0.5]:
        if time.monotonic() - start > time_budget:
            break
        for it in range(30):
            if time.monotonic() - start > time_budget:
                break
            A_spec = rfft(a, size)
            P = np.abs(A_spec) ** 2
            mean_P = np.mean(P)
            
            # Reduce peaks in power spectrum (which correspond to periodic structure)
            P_new = (1 - alpha) * P + alpha * mean_P
            P_new = np.maximum(P_new, 1e-20)
            
            # Reconstruct with original phase
            A_new = np.sqrt(P_new) * (A_spec / (np.abs(A_spec) + 1e-30))
            a_new = irfft(A_new, size)[:n]
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
            else:
                break

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode % 10 == 0:
        a = np.ones(n) / n
    elif mode % 10 == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode % 10 == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.5) ** 2)
        a = a / a.sum()
    elif mode % 10 == 4:
        x = np.linspace(0, 1, n)
        a = x * (1 - x) + 0.05
        a = a / a.sum()
    elif mode % 10 == 5:
        # Two bumps
        x = np.linspace(0, 1, n)
        a = np.exp(-8*(x-0.3)**2) + np.exp(-8*(x-0.7)**2) + 0.05
        a = a / a.sum()
    elif mode % 10 == 6:
        freqs = rng.integers(2, 10, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 7:
        # Geometric-like
        r = rng.uniform(0.95, 0.999)
        a = r ** np.arange(n)
        a = np.concatenate([a, a[::-1]])[:n]
        a = a / a.sum()
    elif mode % 10 == 8:
        # Power law
        a = np.power(np.linspace(0.01, 1, n), rng.uniform(0.5, 3))
        a = a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n) * 0.3)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_refine(a, time_budget, verbose=False):
    """
    Pairwise coordinate moves: try moving mass from position i to position j.
    Guided by the LP solution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    round_num = 0
    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 5:
            break
        round_num += 1
        
        # Get LP direction
        g_lp, _ = solve_lp_fast(a)
        if g_lp is None:
            break
        
        diff = g_lp - a
        # Positions where we should increase (positive diff) or decrease (negative diff)
        inc_positions = np.where(diff > 0.001)[0]
        dec_positions = np.where(diff < -0.001)[0]
        
        if len(inc_positions) == 0 or len(dec_positions) == 0:
            # LP says we're close to optimal, try random moves
            inc_positions = np.argsort(a)[::-1][:10]  # large positions
            dec_positions = np.argsort(a)[:10]  # small positions
        
        improved = False
        # Try top moves
        for frac in [0.02, 0.05, 0.1, 0.2]:
            for _ in range(30):
                if time.monotonic() - start > time_budget - 5:
                    break
                i = inc_positions[rng.integers(len(inc_positions))]
                j = dec_positions[rng.integers(len(dec_positions))]
                if a[j] < 1e-8:
                    continue
                delta = frac * a[j]
                if delta >= a[j]:
                    continue
                a[j] -= delta
                a[i] += delta
                s = float(np.sum(a))
                a = a / s
                obj = compute_obj(a)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] -= delta
                    a[j] += delta
                    a = a / s
        
        if not improved and round_num > 5:
            break
        if not improved:
            # Perturb and try again
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and round_num % 10 == 0:
            print(f"          Coord round {round_num}: obj={best_obj:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP-annealed alternating + subgradient polish + n-sweep + diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"\nInitial: n={n_current}, obj={best_score:.8f}, target=1.5027")

    # === Phase 1: Quick LP polish (15%) ===
    print(f"\n=== Phase 1: LP Polish ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.15, 150.0)
    if p1_budget > 30:
        a_p1 = lp_alternate_annealed(best_seq, time_budget=p1_budget, verbose=True)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_p1:.8f}")

    # === Phase 2: Subgradient polish (10%) ===
    print(f"\n=== Phase 2: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.10, 100.0)
    if p2_budget > 20:
        a_sg = subgradient_polish(best_seq, time_budget=p2_budget, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj_sg:.8f}")

    # === Phase 3: n-sweep (20%) ===
    print(f"\n=== Phase 3: n-Sweep ===")
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.20
    n_options = sorted(set([
        max(500, n_current - 100),
        max(500, n_current - 50),
        max(500, n_current - 20),
        max(500, n_current - 5),
        n_current,
        n_current + 5,
        n_current + 20,
        n_current + 50,
        n_current + 100,
        n_current + 200,
        600, 650, 700, 750, 850, 900, 950, 1000, 1100, 1200,
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1500]

    per_n_budget = p3_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(per_n_budget, remaining * 0.12)
        if alloc < 15:
            alloc = 15

        a_try = resample_sequence(best_seq, n_try)
        a_opt = lp_alternate_annealed(a_try, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)
        marker = " **" if obj_opt < best_score else ""
        print(f"  n={n_try}: obj={obj_opt:.8f}{marker}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try

    # === Phase 4: Intensive LP with annealing (30%) ===
    print(f"\n=== Phase 4: Intensive LP (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.30, 500.0)

    n_rounds = 5
    per_round = p4_budget / n_rounds
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        alloc = min(per_round, remaining * 0.2)
        if alloc < 30:
            break
        
        if rnd == 0:
            a_start = best_seq.copy()
        else:
            rng_r = np.random.default_rng(rnd * 104729 + n_current)
            scale = 0.02 + 0.015 * rnd
            a_start = best_seq * np.exp(rng_r.normal(0, scale, n_current))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()

        temp = 0.03 + 0.01 * rnd
        a_lp = lp_alternate_annealed(a_start, time_budget=alloc,
                                      temperature_start=temp, verbose=(rnd == 0))
        obj_lp = compute_obj(a_lp)
        marker = " **" if obj_lp < best_score else ""
        print(f"  LP round {rnd}: obj={obj_lp:.8f}{marker}")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()

        # Quick subgradient polish after LP
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_sg2 = subgradient_polish(best_seq, time_budget=min(remaining * 0.1, 30))
            obj_sg2 = compute_obj(a_sg2)
            if obj_sg2 < best_score:
                best_score = obj_sg2
                best_seq = a_sg2.copy()
                print(f"  Subgrad polish: obj={best_score:.8f} **")

    # === Phase 5: Coordinate refinement + spectral (15%) ===
    print(f"\n=== Phase 5: Coord + Spectral ===")
    remaining = deadline - time.monotonic()
    p5_budget = remaining * 0.15

    alloc_c = min(p5_budget * 0.6, 60.0)
    if alloc_c > 15:
        a_cd = coordinate_refine(best_seq, time_budget=alloc_c, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Coord: obj={best_score:.8f} **")
        else:
            print(f"  Coord: obj={obj_cd:.8f}")

    remaining = deadline - time.monotonic()
    alloc_s = min(remaining * 0.4, 30.0)
    if alloc_s > 10:
        a_spec = spectral_balance(best_seq, time_budget=alloc_s)
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Spectral: obj={best_score:.8f} **")

    # === Phase 6: Diverse restarts (10%) ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p6_budget = remaining * 0.10
    n_use = len(best_seq)
    n_restarts = 3
    per_restart = p6_budget / max(1, n_restarts)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(per_restart, remaining * 0.25)
        if alloc < 20:
            break

        mode = (r * 3 + 2) % 10
        a_r = diverse_init(n_use, rng, mode=mode)
        blend = 0.3 + 0.15 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        a_r = lp_alternate_annealed(a_r, time_budget=alloc, verbose=False)
        obj_r = compute_obj(a_r)
        marker = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}{marker}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()

    # === Phase 7: Final polish (10%) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    alloc = min(remaining * 0.4, 60.0)
    if alloc > 20:
        a_final = lp_alternate_annealed(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.3, 30.0)
    if alloc > 10:
        a_sg_f = subgradient_polish(best_seq, time_budget=alloc)
        obj_sg_f = compute_obj(a_sg_f)
        if obj_sg_f < best_score:
            best_score = obj_sg_f
            best_seq = a_sg_f.copy()
            print(f"  Final Subgrad: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = lp_alternate_annealed(best_seq, time_budget=min(remaining * 0.5, 30.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"=== Target: 1.5027, Achieved: {best_score:.8f}, Gap: {best_score - 1.5027:.8f} ===")

    return result
