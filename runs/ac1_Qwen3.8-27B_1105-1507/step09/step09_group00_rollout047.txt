"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses sparse LP-based power iteration (fix one factor, optimize other via LP),
with FFT-based objective evaluation, multi-n exploration, and tilted-uniform initialization.
Key: the LP directly minimizes max convolution for one factor, iterating to fixed point.
"""

import numpy as np
import cvxpy as cp
from scipy.sparse import lil_matrix, csc_matrix
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, g):
    """Full linear convolution a*g via FFT."""
    n = max(len(a), len(g))
    size = 1
    while size < len(a) + len(g) - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g, size)
    return irfft(fa * fg, size)[:len(a) + len(g) - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_sparse(a):
    """
    Build sparse matrix A (2n-1 x n) such that A @ g = conv(a, g).
    Row k: A[k,j] = a[j] if 0 <= k-j < n, i.e., max(0,k-n+1) <= j <= min(k,n-1).
    """
    n = len(a)
    rows = 2 * n - 1
    cols = n
    A = lil_matrix((rows, cols))
    for k in range(rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[j]
    return A.tocsr()


def solve_lp_minpeak(a, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (optimal_g, optimal_M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A_mat = build_conv_matrix_sparse(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    objective = cp.Minimize(M)
    prob = cp.Problem(objective, constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None


def tilted_uniform_init(n, tilt=1.0, rng=None):
    """Initialize with tilted uniform: a[j] proportional to exp(tilt * j/n)."""
    j = np.arange(n)
    a = np.exp(tilt * j / (n - 1) if n > 1 else 0)
    a = a / a.sum()
    return a


def baby_basket_init(n, rng):
    """Baby basket / Polak-type initialization."""
    a = np.ones(n) / n
    for _ in range(3):
        a = 1.0 - (n - 1) * abs(compute_conv(a / a.sum())[:n] - 1.0 / n)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
    return a


def lp_power_iteration(a, time_budget=600.0, verbose=True, seed=42):
    """
    Main LP-based power iteration: repeatedly solve min-peak LP and blend.
    Uses adaptive step size and persistence.
    """

    it = 0
    stalled = 0
    step = 0.3
    min_step = 0.02
    max_improvements = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        it += 1

        # Try multiple step sizes
        candidates = [step, step * 0.5, step * 0.7, step * 1.3, step * 1.5,
                      min(1.0, step * 2.0), 0.5, 0.3, 0.2, 0.1, 0.05]
        candidates = [t for t in candidates if 0.01 <= t <= 1.0]
        # Add some random

        candidates = sorted(set([round(t, 4) for t in candidates]))

                best_t_val = t

            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            step = best_t_val
            stalled = 0
            max_improvements += 1

        else:

                a_new = np.maximum(a_new, 0)
                s_new = a_new.sum()
                if s_new > 1e-10:
                    a = a_new / s_new
                    stalled = 0

    return best_a


def fast_smoothed_descent(a, time_budget=60.0, tau_init=20.0, tau_end=500.0, verbose=False):
    """
    Quick smoothed gradient descent for polishing.
    Uses FFT-based log-sum-exp smoothed gradient.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 2.0 / np.sqrt(n)
    it = 0

    # Precompute FFT size
    N = 1
    while N < 3 * n:
        N <<= 1

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress

        # Compute smoothed gradient
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient: g = 2 * cross_correlate(p, a) restricted to first n elements
        P = np.fft.fft(p, N)
        A = np.fft.fft(a, N)
        g_full = np.fft.ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        # Update
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr *= 0.999
        it += 1

        if verbose and it % 5000 == 0:
            print(f"      SGD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.0f}s")

    return best_a
