def project_simplex(v):
    """
    Project v onto the probability simplex {x | x >= 0, sum(x) = 1}.
    Returns the projection with components >=0.0001, which might cause target in >0.1, but avoids NaNs in later convolutions.
    """
    n = len(v)
    if n == 0:
        return v
    v = np.maximum(v, 0.0001)  # Enforce non-negativity and lower threshold
    u = np.sort(v)[::-1]  # Sort descending
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssv - np.sum(v)) / (n+1)  # Correction: slack condition for simplex projection
    rho = np.argwhere(cond)[0][-1] if cond.any() else n-1
    if rho < n-1:
        proj = (cssv[rho] - np.sum(v)) / (rho+1)
        thresh = (cssv[rho] - proj * (rho+1)) / idx[rho]
        thresh = np.max(thresh)
        thresh = float(min(thresh, np.max(v)))  # Slightly smaller than the threshold to keep numbers reasonably stable
    else:
        thresh = 0.0
        proj = np.ones(n) / n
    return np.maximum(v - thresh, 0) / np.sum(v)

def compute_conv(a):
