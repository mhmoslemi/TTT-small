"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses enhanced LP polishing with golden-section blend search, structured 
hot-spot perturbations, multi-length exploration, and adaptive time allocation.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal, dual_vars) or (None, None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Try to get dual variables
                    y = None
                    # The dual of the constraint A@k <= M
                    try:
                        dual = constraints[0].dual_value
                        if dual is not None:
                            y = np.maximum(dual, 0)
                    except:
                        pass
                    return g_val, float(M.value), y
        except Exception:
            continue
    return None, None, None


def golden_section_search(a, g_lp, obj_a, max_iter=30):
    """
    Golden-section search on blend parameter t in [0, 1].
    Minimizes compute_obj((1-t)*a + t*g_lp).
    Returns (best_t, best_a, best_obj).
    """
    golden = (np.sqrt(5) - 1) / 2  # ~0.618
    
    lo = 0.0
    hi = 1.0
    
    # Initial points
    d = hi - lo
    x1 = hi - golden * d
    x2 = lo + golden * d
    
    f1 = f2 = None
    
    for _ in range(max_iter):
        if f1 is None:
            a1 = (1 - x1) * a + x1 * g_lp
            a1 = np.maximum(a1, 0)
            s1 = float(np.sum(a1))
            if s1 < 1e-10:
                f1 = np.inf
            else:
                f1 = compute_obj(a1 / s1)
        
        if f2 is None:
            a2 = (1 - x2) * a + x2 * g_lp
            a2 = np.maximum(a2, 0)
            s2 = float(np.sum(a2))
            if s2 < 1e-10:
                f2 = np.inf
            else:
                f2 = compute_obj(a2 / s2)
        
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            d = hi - lo
            x1 = hi - golden * d
            f1 = None
        else:
            lo = x1
            x1 = x2
            f1 = f2
            d = hi - lo
            x2 = lo + golden * d
            f2 = None
    
    best_t = (lo + hi) / 2
    best_a = (1 - best_t) * a + best_t * g_lp
    best_a = np.maximum(best_a, 0)
    s = float(np.sum(best_a))
    if s < 1e-10:
        return 0.5, a.copy(), obj_a
    best_a = best_a / s
    best_obj = compute_obj(best_a)
    return best_t, best_a, best_obj

def hotspot_perturb(a, rng, strength=0.03):
    """
    Perturb based on convolution hot spots.
    Reduce mass at positions contributing to peaks, increase at flat positions.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv = compute_conv(a_norm)
    max_conv = float(np.max(conv))
    
    # Find hot spots (top 10% of conv values)
    threshold = max_conv * 0.9
    hot = conv > threshold
    
    # Compute contribution of each position j to hot spots
    n_hot = int(np.sum(hot))
    if n_hot == 0:
        return a.clone() if hasattr(a, 'clone') else a.copy()
    
    hot_indices = np.where(hot)[0]
    contrib = np.zeros(n)
    for k in hot_indices:
        k = int(k)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                contrib[j] += a_norm[idx]
    
    # Normalize contribution
    max_contrib = float(np.max(contrib))
    if max_contrib < 1e-10:
        return a.copy()
    contrib = contrib / max_contrib
    
    # Perturb: reduce where contribution is high, increase where low
    perturbation = 1.0 + strength * (0.5 - contrib)
    a_new = a * perturbation
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(n) / n
    return a_new / s

        a = a * np.exp(rng.normal(0, scale, n))
    s = float(np.sum(a))
    if s < 1e-10:

    return a / s


def smoothed_gradient(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    # Use a stable log-sum-exp
    shifted = conv - max_conv
    max_shift = float(np.max(shifted))
    exp_vals = np.exp(tau * (shifted - max_shift))
    p = exp_vals / float(np.sum(exp_vals))
    
    # Gradient of sum_k log(exp(tau*(conv_k - c))) w.r.t. a
    # = tau * sum_k p_k * d(conv_k)/da
    # d(conv_k)/da_j = a_{k-j} (if valid)
    # So gradient_j = tau * sum_k p_k * a_{k-j}
    
    # This is a cross-correlation of p and a
    from scipy.fft import fft, ifft
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * tau * g_full[:n]
    return g, conv

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Gaussian
        x = np.arange(n)
        center = n / 2
        sigma = n / 4
        a = np.exp(-(x - center) ** 2 / (2 * sigma ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Raised cosine
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 4:
        # Double Gaussian
        x = np.arange(n)
        a = np.exp(-(x - n/4) ** 2 / (2 * (n/8) ** 2)) + np.exp(-(x - 3*n/4) ** 2 / (2 * (n/8) ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Inverse Gaussian (heavy center)
        x = np.arange(n)
        center = n / 2
        a = 1.0 / (1.0 + ((x - center) / (n/6)) ** 2)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses enhanced LP polishing with golden-section blend search, structured hot-spot
perturbations, multi-length exploration, and adaptive time allocation.
Key improvements over previous: golden-section line search, dual-informed
perturbations, better time allocation, and more aggressive LP iterations.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def golden_section_search_obj(a, g_lp, max_iter=20):
    """
    Golden-section search on blend parameter t in [0, 1].
    Minimizes compute_obj((1-t)*a + t*g_lp).
    Returns (best_t, best_a, best_obj).
    """
    golden = (np.sqrt(5) - 1) / 2

    lo = 0.0
    hi = 1.0

    a_lo = a.copy()
    f_lo = compute_obj(a_lo)

    a_hi = g_lp.copy()
    s_hi = float(np.sum(a_hi))
    if s_hi < 1e-10:
        a_hi = a.copy()
        f_hi = f_lo
    else:
        a_hi = a_hi / s_hi
        f_hi = compute_obj(a_hi)

    for _ in range(max_iter):
        d = hi - lo
        if d < 1e-6:
            break
        x1 = hi - golden * d
        x2 = lo + golden * d

        a1 = (1 - x1) * a + x1 * g_lp
        a1 = np.maximum(a1, 0)
        s1 = float(np.sum(a1))
        f1 = compute_obj(a1 / s1) if s1 > 1e-10 else np.inf

        a2 = (1 - x2) * a + x2 * g_lp
        a2 = np.maximum(a2, 0)
        s2 = float(np.sum(a2))
        f2 = compute_obj(a2 / s2) if s2 > 1e-10 else np.inf

        if f1 < f2:
            hi = x2
            hi_a = a1
            f_hi = f1
        else:
            lo = x1
            lo_a = a2
            f_lo = f2

    best_t = (lo + hi) / 2.0
    best_a = (1 - best_t) * a + best_t * g_lp
    best_a = np.maximum(best_a, 0)
    s = float(np.sum(best_a))
    if s < 1e-10:
        return 0.5, a.copy(), compute_obj(a)
    best_a = best_a / s
    best_obj = compute_obj(best_a)
    return best_t, best_a, best_obj
