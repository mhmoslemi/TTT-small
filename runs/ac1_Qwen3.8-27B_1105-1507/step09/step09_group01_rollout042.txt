"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses spectral shaping initialization, FISTA on smoothed objective, LP polishing,
Frank-Wolfo steps, multi-length exploration, and adaptive restarts.
Key innovations: FFT-based spectral shaping for initialization, FISTA with
adaptive tau scheduling, and combined FW+LP polishing for final refinement.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient_fft(a, tau, conv=None):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*conv_k(a))) at point a.
    Returns the gradient vector of length n.
    """
    n = len(a)
    if conv is None:
        conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    p = exp_vals / float(np.sum(exp_vals))

    # Gradient: grad[i] = 2 * sum_k p[k]*(a[k-i] + a[i-k]) (with a[j]=0 outside [0,n))
    # S1[i] = sum_j a[j]*p[j+i] (cross-correlation)
    # S2[i] = sum_j p[j]*a[i-j] (convolution, first n terms)

    N = 1
    while N < 3 * n + 2:
        N <<= 1

    P = fft(p, N)
    A = fft(a, N)

    # S2[i] = (p*a)[i] = ifft(P*A)[i]
    conv_pa = ifft(P * A).real
    S2 = conv_pa[:n]

    # S1[i] = sum_j a[j]*p[j+i] = conv(a, p_rev)[2n-2+i] where p_rev = reversed(p)
    p_rev = p[::-1]
    P_rev = fft(p_rev, N)
    conv_ap = ifft(A * P_rev).real
    S1 = conv_ap[2 * n - 2: 2 * n - 2 + n]

    grad = 2.0 * (S1 + S2)
    return grad


def fista_smoothed(a, time_budget=300.0, tau_init=20.0, tau_end=500.0, verbose=False):
    """FISTA (accelerated proximal gradient) on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    v = np.zeros(n)
    prev_g = None
    prev_y = None
    lr = 2.0 / np.sqrt(n)
    it = 0
    best_it = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        conv_y = compute_conv(y)
        g = smoothed_gradient_fft(y, tau, conv_y)

        # Backtracking / BB step size
        if prev_g is not None and prev_y is not None and it % 3 == 0:
            sy = float(np.dot(y - prev_y, g - prev_g))
            ss = float(np.dot(y - prev_y, y - prev_y))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / max(abs(sy), 1e-10)
                lr = 0.4 * lr + 0.6 * min(max(bb, 1e-8), 20.0 / np.sqrt(n))

        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        theta = (1.0 + t_k) / t_new
        y_new = y + (theta - 1.0) * (y - a)
        y_new = y_new - lr * g
        a_new = project_simplex(y_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 5000:
                a = best_a.copy()
                y = best_a.copy()
                v = np.zeros(n)
                t_k = 1.0
                lr = 2.0 / np.sqrt(n)
                stall_count = 0

        a = a_new
        y = y_new
        t_k = t_new
        prev_y = y.copy()
        prev_g = g.copy()
        it += 1
        lr = max(1e-8, lr * 0.99995)

        if verbose and it % 20000 == 0:
            print(f"    FISTA it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.4f}, t={elapsed:.0f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        if r > 0:
            A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, M_target=None, verbose=False):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue
    return None, None


def golden_section_search(f, a_lo, a_hi, tol=0.01, max_iter=30):
    """Golden section search for minimum of f on [a_lo, a_hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if abs(a_hi - a_lo) < tol:
            break
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = f(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = f(d)
    return (a_lo + a_hi) / 2.0

def frank_wolvo_polish(a, time_budget=80.0, verbose=False):
    """Frank-Wolfo (conditional gradient) with pairwise steps."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]
        if max_val < 1e-15:
            break

        # Subgradient: direction to increase is where conv is peaked
        # We want to decrease the peak, so move mass away from contributing positions
        k = int(max_idx)
        # Contribution to conv[k] from a[i]: a[i]*a[k-i]
        # To decrease, we should decrease a at positions that contribute to this peak
        # Subgradient direction: d(conv[k])/da[i] = a[k-i] + a[i-k]
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start > i_end:
            break

        subgrad = np.zeros(n)
        for i in range(i_start, i_end + 1):
            contrib = 0.0
            if 0 <= k - i < n:
                contrib += a[k - i]
            if 0 <= i - k < n:
                contrib += a[i - k]
            subgrad[i] = contrib

        # FW step: find the vertex (unit vector) that minimizes subgrad^T e_i
        # i.e., find argmin of subgrad
        min_idx = np.argmin(subgrad)

        # Pairwise FW: optimize along the line from a to e_min_idx
        # a(t) = (1-t)*a + t*e_min_idx
        # Find t that minimizes the objective
        def line_obj(t):
            a_t = (1.0 - t) * a + t * (np.eye(1, n, min_idx)[:, 0])
            a_t = np.maximum(a_t, 0)
            s_t = float(np.sum(a_t))
            if s_t < 1e-10:
                return 1e10
            a_t = a_t / s_t
            return compute_obj(a_t)

        t_opt = golden_section_search(line_obj, 0.0, 1.0, tol=0.01, max_iter=20)

        a_new = (1.0 - t_opt) * a + t_opt * (np.eye(1, n, min_idx)[:, 0])
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        obj_new = compute_obj(a_new)

        it += 1
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break

        if verbose and it % 10000 == 0:
            print(f"    FW it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")

    return best_a


def spectral_shaping_init(n, n_iters=50, rng=None, verbose=False):
    """
    Generate a non-negative sequence with flat power spectrum using
    iterative spectral shaping (Gerchberg-Saxton variant).
    """
    if rng is None:
        rng = np.random.default_rng()

    # Initialize with random phase (flat magnitude)
    N = 1
    while N < 2 * n:
        N <<= 1

    # Start with a random non-negative sequence
    a = rng.uniform(0.5, 1.5, n)
    a = a / a.sum()

    # Target: flat |A[f]|^2 (proportional to 1/(2n-1) for all frequencies)
    # But we need to respect a >= 0 and sum(a) = 1

    for it in range(n_iters):
        # Compute spectrum
        A = fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Flatten the magnitude: raise low values, lower high values
        # Use a soft flattening: target magnitude is geometric mean
        target_mag = np.power(mag, 0.7)  # Compress dynamic range
        target_mag = target_mag / target_mag.sum() * mag.sum()

        # Blend towards flat
        alpha = 0.3
        new_mag = (1.0 - alpha) * mag + alpha * target_mag

        # Reconstruct with new magnitude, keep phase
        new_A = new_mag * np.exp(1j * phase)
        a_new = ifft(new_A).real[:n]

        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a = rng.uniform(0.5, 1.5, n)
            a = a / a.sum()
        else:
            a = a_new / s

    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        # Sine mixture
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        # Cosine taper
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        # Random bumps
        n_bumps = rng.integers(2, 5)
        a = np.zeros(n)
        for _ in range(n_bumps):
            c = rng.integers(0, n)
            w = max(1, n // rng.integers(5, 15))
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 8:
        # Spectral shaping
        a = spectral_shaping_init(n, n_iters=30, rng=rng)
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: spectral shaping init + FISTA + LP polish + FW + multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Multi-init spectral shaping + quick FISTA ===
    print(f"\n=== Phase 1: Multi-init Exploration ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.10, 100.0)

    if p1_budget > 20:
        n_curr = len(best_seq)
        modes = [0, 1, 3, 4, 5, 6, 7, 8]
        per_mode = max(5, p1_budget / len(modes) / 2)

        for mode in modes:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            a_init = diverse_init(n_curr, rng, mode=mode)
            # Quick FISTA
            a_opt = fista_smoothed(a_init, time_budget=min(per_mode, remaining * 0.8),
                                   tau_init=10.0, tau_end=200.0)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Mode {mode}: obj={best_score:.8f} **")
            else:
                print(f"  Mode {mode}: obj={obj:.8f}")

    # === Phase 2: Main FISTA on best sequence ===
    print(f"\n=== Phase 2: FISTA Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 300.0)
    if p2_budget > 30:
        a_fista = fista_smoothed(best_seq, time_budget=p2_budget,
                                 tau_init=15.0, tau_end=400.0, verbose=True)
        obj = compute_obj(a_fista)
        if obj < best_score:
            best_score = obj
            best_seq = a_fista.copy()
            print(f"  FISTA: obj={best_score:.8f} **")
        else:
            print(f"  FISTA: obj={obj:.8f}")

    # === Phase 3: LP Polishing (main workhorse) ===
    print(f"\n=== Phase 3: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.30, 350.0)
    if p3_budget > 30:
        a_lp = lp_polish(best_seq, time_budget=p3_budget, verbose=True, seed=42)
        obj = compute_obj(a_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj:.8f}")

    # === Phase 4: Frank-Wolvo refinement ===
    print(f"\n=== Phase 4: Frank-Wolvo ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.08, 80.0)
    if p4_budget > 15:
        a_fw = frank_wolvo_polish(best_seq, time_budget=p4_budget)
        obj = compute_obj(a_fw)
        if obj < best_score:
            best_score = obj
            best_seq = a_fw.copy()
            print(f"  FW: obj={best_score:.8f} **")
        else:
            print(f"  FW: obj={obj:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 200), max(400, n_current - 100),
        n_current + 100, n_current + 200, n_current + 400, n_current + 600,
        600, 800, 1000, 1200, 1400, 1600, 1800
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_current) > 80]
    # Sort by proximity to current best n
    n_options.sort(key=lambda x: abs(x - n_current))

    for n_try in n_options[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)

        # Spectral shaping refinement
        a_try = spectral_shaping_init(n_try, n_iters=20, rng=rng)
        # Blend resampled and shaped
        a_try = 0.5 * resample_sequence(best_seq, n_try) + 0.5 * a_try
        a_try = a_try / a_try.sum()

        # Quick FISTA
        alloc1 = min(remaining * 0.15, 30.0)
        if alloc1 > 8:
            a_try = fista_smoothed(a_try, time_budget=alloc1,
                                   tau_init=10.0, tau_end=200.0)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.20, 80.0)
        if alloc2 > 15:
            a_try = lp_polish(a_try, time_budget=alloc2, seed=n_try + 1)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Perturbation + re-optimize ===
    print(f"\n=== Phase 6: Perturbation + Re-LP ===")
    remaining = deadline - time.monotonic()

    for r in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        scale = 0.008 + 0.006 * r
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        # Quick FISTA warm
        alloc1 = min(remaining * 0.15, 20.0)
        if alloc1 > 5:
            a_pert = fista_smoothed(a_pert, time_budget=alloc1,
                                    tau_init=50.0, tau_end=200.0)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.25, 80.0)
        if alloc2 > 10:
            a_pert = lp_polish(a_pert, time_budget=alloc2, seed=200 + r)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Perturb {r} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Perturb {r} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 7: Final intensive LP ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 150.0)
        a_final = lp_polish(best_seq, time_budget=alloc, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # Final quick FISTA
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 40.0)
        a_sg = fista_smoothed(best_seq, time_budget=alloc,
                              tau_init=100.0, tau_end=500.0)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final FISTA: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
