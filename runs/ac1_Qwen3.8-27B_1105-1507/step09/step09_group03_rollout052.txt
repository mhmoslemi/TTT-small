```python
import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    # FFT size for cross-correlation and convolution
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    
    P = fft(p, N)
    A = fft(a, N)
    
    # g1 = cross-correlation: ifft(conj(P) * A).real[:n]
    g1 = ifft(np.conj(P) * A).real[:n]
```

```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated smoothed gradient descent with Nesterov acceleration,
adaptive tau scheduling, LP polishing, and multi-n exploration.
Key innovations: proper FFT cross-correlation gradient, aggressive tau annealing,
peak-targeted subgradient steps, and structured restarts.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient_fft(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*b_k)) where b = conv(a,a).
    Returns (gradient, convolution, peak_weights).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n - 1:
        N <<= 1

    P = fft(p, N)
    A = fft(a, N)

    # g1[i] = sum_k p[k] * a[k-i] = cross-correlation
    g1 = ifft(np.conj(P) * A).real[:n]
    # g2[i] = sum_k p[k] * a[i-k] = convolution
    g2 = ifft(P * A).real[:n]

    grad = (g1 + g2)  # The 1/tau factor is absorbed into step size
    return grad, conv, p


def subgradient_at_peak(a, peak_idx=None):
    """Compute subgradient at the maximum of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    if peak_idx is None:
        peak_idx = int(np.argmax(conv))

    # Subgradient of b_k at a[i] = a[k-i] + a[i-k] (with bounds)
    sg = np.zeros(n)
    for i in range(n):
        val = 0.0
        ki = peak_idx - i
        if 0 <= ki < n:
            val += a[ki]
        ik = i - peak_idx
        if 0 <= ik < n:
            val += a[ik]
        sg[i] = val
    return sg, conv, peak_idx


def nesterov_smoothed_descent(a, time_budget=200.0, tau_init=5.0, tau_end=500.0,
                               verbose=False, seed=123):
    """
    Nesterov accelerated gradient descent on smoothed objective over simplex.
    Uses adaptive tau and step size.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    lr = 4.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        # Exponential tau schedule
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)

        if it == 0 or it % 500 == 0:
            if verbose:
                y_temp = a + beta * v
                y_temp = np.maximum(y_temp, 0)
                s_y = float(np.sum(y_temp))
                if s_y > 1e-10:
                    y_temp = y_temp / s_y
                obj_y = compute_obj(y_temp)
                print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, obj_y={obj_y:.8f}")

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute gradient
        grad, conv, p = smoothed_gradient_fft(y, tau)

        # Adaptive step size based on gradient magnitude
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-10:
            break
        # Scale step by 1/sqrt(tau) since our gradient doesn't include 1/tau
        effective_lr = lr / np.sqrt(tau)
        effective_lr = min(effective_lr, 0.5 / grad_norm)

        v = beta * v - effective_lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        # Project to simplex to ensure sum=1
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        # Decay learning rate
        lr = max(0.1 / np.sqrt(n), lr * 0.9997)
        beta = min(0.95, 0.85 + 0.001 * it)

        # Stagnation handling
        if it - best_it > 8000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 4.0 / np.sqrt(n) * 0.3
            beta = 0.85
            # Add small perturbation
            if rng.random() < 0.5:
                a = best_a * np.exp(rng.normal(0, 0.01, n))
                a = project_simplex(a)
            best_it = it

    return best_a


def lp_polish(a, time_budget=100.0, verbose=False):
    """
    LP-based polishing: solve min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.
    Then blend with current a.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(5555)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        # Build constraint matrix: A @ g = conv(a, g)
        # A has shape (2n-1, n), A[k, j] = a[k-j] if 0<=k-j<n else 0
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            # conv(a,g)[k] = sum_j a[j] * g[k-j], so for fixed j, contributes a[j] to positions k=j..j+n-1
            for k in range(j, min(2 * n - 1, j + n)):
                A[k, j] = a[k - j]

        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        g_val = None
        for solver in [cp.GLPK, cp.CBC, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        g_val = g_val / s_g
                    break
            except Exception:
                continue

        if g_val is None:
            break

        # Try different blend ratios
        improved = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2]:
            a_try = (1 - t) * best_a + t * g_val
            a_try = np.maximum(a_try, 0)
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1
            # Perturb
            a_curr = best_a * np.exp(rng.normal(0, 0.02 * min(stall, 5), n))
            a_curr = project_simplex(a_curr)
            best_a_for_lp = a_curr  # Use perturbed for next LP
            # But keep best_a as the best found
            if stall > 30:
                break

        it += 1
        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, M={prob.value:.6f}")

    return best_a


def two_stage_lp(a, time_budget=80.0, verbose=False):
    """Two-stage LP: min-max then min-sum-of-active."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(7777)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        # Build A
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            for k in range(j, min(2 * n - 1, j + n)):
                A[k, j] = a[k - j]

        # Stage 1: min max
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        g1_val = None
        M1_val = None
        for solver in [cp.GLPK, cp.CBC, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g1_val = np.maximum(g.value, 0)
                    s_g = float(np.sum(g1_val))
                    if s_g > 1e-10:
                        g1_val = g1_val / s_g
                    M1_val = float(M.value)
                    break
            except Exception:
                continue

        if g1_val is None:
            break

        # Find active set
        conv_ag = A @ g1_val
        tol = 0.02 * max(float(np.max(conv_ag)), 1e-10)
        active = np.where(conv_ag >= float(np.max(conv_ag)) - tol)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv_ag))])

        # Stage 2: minimize sum at active positions
        g2 = cp.Variable(n, nonneg=True)
        active_A = A[active, :]
        prob2 = cp.Problem(cp.Minimize(cp.sum(active_A @ g2)), [cp.sum(g2) == 1, g2 >= 0])

        g2_val = None
        for solver in [cp.GLPK, cp.CBC, cp.ECOS]:
            try:
                prob2.solve(solver=solver)
                if g2.value is not None and prob2.status in ('optimal', 'optimal_inaccurate'):
                    g2_val = np.maximum(g2.value, 0)
                    s_g2 = float(np.sum(g2_val))
                    if s_g2 > 1e-10:
                        g2_val = g2_val / s_g2
                    break
            except Exception:
                continue

        # Blend
        g_blend = g1_val
        if g2_val is not None:
            # Check which gives better peak
            peak1 = float(np.max(A @ g1_val))
            peak2 = float(np.max(A @ g2_val))
            if peak2 < peak1:
                g_blend = g2_val
            else:
                g_blend = 0.5 * g1_val + 0.5 * g2_val

        improved = False
        for t in [1.0, 0.7, 0.5, 0.4, 0.3, 0.2, 0.1]:
            a_try = (1 - t) * best_a + t * g_blend
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 25:
                # Perturb and continue
                a_pert = best_a * np.exp(rng.normal(0, 0.015 * min(stall, 8), n))
                a_pert = project_simplex(a_pert)
                a = a_pert  # Use perturbed for next LP iteration
                if stall > 50:
                    break
            else:
                a = best_a

        it += 1
        if verbose and it % 5 == 0:
            print(f"    2SLP it{it}: obj={best_obj:.8f}, M1={M1_val:.6f}, |S|={len(active)}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def peak_splitting_descent(a, time_budget=60.0, verbose=False):
    """
    Targeted descent: repeatedly find the peak, compute its subgradient,
    and take a projected step. More aggressive than smoothed GD near convergence.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = project_simplex(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    it = 0
    stall = 0
    step_scale = 1.0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))
        peak_val = float(conv[peak_idx])

        # Subgradient at the peak
        sg = np.zeros(n)
        for i in range(n):
            ki = peak_idx - i
            if 0 <= ki < n:
                sg[i] += a[ki]
            ik = i - peak_idx
            if 0 <= ik < n:
                sg[i] += a[ik]

        # Also consider top few peaks
        top_k = np.argsort(conv)[-5:]
        for pk in top_k:
            for i in range(n):
                ki = pk - i
                if 0 <= ki < n:
                    sg[i] += 0.3 * a[ki]
                ik = i - pk
                if 0 <= ik < n:
                    sg[i] += 0.3 * a[ik]

        # Projected gradient step
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-10:
            break

        for scale in [step_scale, step_scale * 0.5, step_scale * 0.25, step_scale * 0.1]:
            a_try = a - scale * sg / (sg_norm * np.sqrt(n))
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                stall = 0
                break
        else:
            stall += 1
            step_scale *= 0.7
            if step_scale < 0.01 or stall > 100:
                break

        a = best_a
        it += 1
        if verbose and it % 20 == 0:
            print(f"    PSD it{it}: obj={best_obj:.8f}")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Comb-like
        a = np.ones(n)
        a[::2] = 2.0
        a = a / a.sum()
    elif mode == 3:
        # Decaying
        a = np.exp(-np.linspace(0, 2, n))
        a = a / a.sum()
    elif mode == 4:
        # Two bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Semicircle (archimedean spiral-like)
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x**2, 0.01))
        a = a / a.sum()
    elif mode == 6:
        # Random sparse-ish
        a = rng.uniform(0, 1, n)
        mask = rng.random(n) > 0.3
        a[~mask] = 0
        if a.sum() < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / a.sum()
    else:
        # Triangle
        x = np.arange(n) / n
        a = 1 - np.abs(2 * x - 1)
        a = a / a.sum()
    return a

        else:
            print(f"  GD low-tau: obj={obj_gd:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_gd2 = nesterov_smoothed_descent(best_seq, time_budget=min(remaining * 0.3, 80.0),
                                               tau_init=50.0, tau_end=1000.0,
                                               verbose=True, seed=200)
            obj_gd2 = compute_obj(a_gd2)
            if obj_gd2 < best_score:
                best_score = obj_gd2
                best_seq = a_gd2.copy()
                print(f"  GD high-tau: obj={best_score:.8f} **")
            else:
                print(f"  GD high-tau: obj={obj_gd2:.8f}")

    # === Phase 3: LP Polishing ===
    print(f"\n=== Phase 3: LP Polish (n={best_n}) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 150.0)
    if p3_budget > 20 and best_n <= 1200:
        a_lp = two_stage_lp(best_seq, time_budget=p3_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  2SLP: obj={best_score:.8f} **")
        else:
            print(f"  2SLP: obj={obj_lp:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_lp2 = lp_polish(best_seq, time_budget=min(remaining * 0.3, 60.0))
            obj_lp2 = compute_obj(a_lp2)
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  LP: obj={best_score:.8f} **")

    # === Phase 4: Peak-Splitting Descent ===
    print(f"\n=== Phase 4: Peak-Splitting Descent ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.1, 80.0)
    if p4_budget > 10:
        a_psd = peak_splitting_descent(best_seq, time_budget=p4_budget, verbose=True)
        obj_psd = compute_obj(a_psd)
        if obj_psd < best_score:
            best_score = obj_psd
            best_seq = a_psd.copy()
            print(f"  PSD: obj={best_score:.8f} **")
        else:
            print(f"  PSD: obj={obj_psd:.8f}")

    # === Phase 5: Diverse Restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = r % 8
        a_r = diverse_init(best_n, rng, mode=mode)
        # Blend with current best
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = project_simplex(a_r)

        # Quick optimization
        alloc = min(remaining * 0.2, 60.0)
        if alloc > 15:
            a_r = nesterov_smoothed_descent(a_r, time_budget=alloc * 0.5,
                                             tau_init=5.0, tau_end=300.0,
                                             verbose=False, seed=500 + r)
            remaining = deadline - time.monotonic()
            if remaining > 10 and best_n <= 1000:
                a_r = lp_polish(a_r, time_budget=min(remaining * 0.3, 30.0))

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Try nearby n values ===
    print(f"\n=== Phase 6: Nearby n Exploration ===")
    remaining = deadline - time.monotonic()
    for n_delta in [-100, -50, 50, 100, 150]:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        n_try = best_n + n_delta
        if n_try < 200 or n_try > 2000:
            continue

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.15, 40.0)
        if alloc > 10:
            a_try = nesterov_smoothed_descent(a_try, time_budget=alloc,
                                               tau_init=10.0, tau_end=400.0,
                                               verbose=False, seed=hash(n_try) % 10000)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            best_n = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish (n={best_n}) ===")
    remaining = deadline - time.monotonic()

    # High-tau GD
    if remaining > 20:
        a_final = nesterov_smoothed_descent(best_seq, time_budget=min(remaining * 0.3, 60.0),
                                             tau_init=200.0, tau_end=3000.0,
                                             verbose=False, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Peak splitting
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_final = peak_splitting_descent(best_seq, time_budget=min(remaining * 0.3, 40.0))
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final PSD: obj={best_score:.8f} **")

    # LP if n is small enough
    remaining = deadline - time.monotonic()
    if remaining > 20 and best_n <= 1000:
        a_final = two_stage_lp(best_seq, time_budget=min(remaining * 0.4, 50.0))
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # One more high-tau GD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_final = nesterov_smoothed_descent(best_seq, time_budget=min(remaining * 0.5, 40.0),
                                             tau_init=500.0, tau_end=5000.0,
                                             verbose=False, seed=1111)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Ultra-GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
