"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses QP-based target-descent (binary search on peak target with minimum-norm QP),
LP polishing with line search, frequency-domain perturbations, and multi-n exploration.
Key innovation: QP-based direct peak targeting instead of LP-then-blend.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(n, 2 * n - 1 - j)
        A[j:j + end, j] = a[:end]
    return A


def solve_qp_target(a, M_target):
    """
    Solve QP: min ||g-a||^2 s.t. conv(a,g)_k <= M_target for all k, sum(g)=1, g>=0.
    Returns g or None if infeasible.
    """
    n = len(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    obj = cp.sum_squares(g - a)
    constraints = [A @ g <= M_target, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(obj), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None) on failure.
    """
    n = len(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g, max_t=1.0, n_trials=12):
    """
    Find best t in [0, max_t] for a_new = (1-t)*a + t*g.
    Uses golden-section-like search + evaluation.
    Returns (best_a, best_obj, best_t).
    """
    best_obj = np.inf
    best_a = a.copy()
    best_t = 0.0

    # Evaluate at a grid + golden section refinement
    ts = np.concatenate([
        np.array([max_t]),
        np.linspace(0.05, max_t, n_trials),
        np.array([max_t * 0.1, max_t * 0.3, max_t * 0.5, max_t * 0.7])
    ])
    ts = np.unique(np.clip(ts, 0, max_t))

    for t in ts:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_a = a_try.copy()
            best_t = t

    return best_a, best_obj, best_t


def qp_descent(a, time_budget=300.0, verbose=True):
    """
    QP-based target descent: iteratively reduce the peak of conv(a,a)
    by solving QPs with decreasing target M.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    best_peak = float(np.max(compute_conv(a)))
    start = time.monotonic()
    it = 0
    stuck_count = 0
    last_improve_it = 0

    if verbose:
        print(f"    QP start: obj={best_obj:.8f}, peak={best_peak:.8f}, n={n}")

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if verbose and it % 50 == 0:
            print(f"    QP it{it}: obj={best_obj:.8f}, peak={best_peak:.8f}, t={elapsed:.1f}s")

        # Try a range of target reductions
        reduction_factors = [0.98, 0.95, 0.92, 0.89, 0.85, 0.80, 0.75]
        # Make more aggressive over time
        progress = min(1.0, elapsed / time_budget)
        if progress > 0.5:
            reduction_factors = [0.97, 0.94, 0.91, 0.88, 0.84, 0.80, 0.75, 0.70]

        found_update = False
        for factor in reduction_factors:
            M_target = best_peak * factor
            g = solve_qp_target(a, M_target)
            if g is None:
                continue

            # Line search for optimal blend
            a_new, obj_new, t_opt = line_search_blend(a, g, max_t=1.0, n_trials=10)

            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                best_peak = float(np.max(compute_conv(best_a)))
                a = best_a.copy()
                found_update = True
                stuck_count = 0
                last_improve_it = it
                break
            else:
                # Even if objective didn't improve, the peak might have decreased
                conv_new = compute_conv(a_new)
                peak_new = float(np.max(conv_new))
                if peak_new < best_peak - 1e-10:
                    best_peak = peak_new
                    a = a_new.copy()
                    found_update = True
                    stuck_count = 0
                    obj_check = compute_obj(a)
                    if obj_check < best_obj:
                        best_obj = obj_check
                        best_a = a.copy()
                    break

        it += 1

        if not found_update:
            stuck_count += 1
            if stuck_count > 5:
                # Perturb to escape
                rng_local = np.random.default_rng(it * 7919)
                pert_scale = 0.01 + 0.02 * min(stuck_count / 10.0, 1.0)
                a = a * np.exp(rng_local.normal(0, pert_scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stuck_count = 0

        if it - last_improve_it > 200:
            a = best_a.copy()

    return best_a


def lp_polish_round(a, max_iter=100, time_limit=100.0, verbose=False):
    """
    LP polishing: repeatedly solve LP for optimal g given a, then line-search blend.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf, 0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    no_improve = 0
    start = time.monotonic()
    it = 0

    while it < max_iter and (time.monotonic() - start) < time_limit:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        a_new, obj_new, t_opt = line_search_blend(a, g_lp, max_t=1.0, n_trials=15)
        it += 1

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = best_a
            no_improve = 0
            if verbose and it % 10 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={t_opt:.3f}, M={M_lp:.6f}")
        else:
            no_improve += 1
            if no_improve >= 5:
                break

    return best_a, best_obj, it


def smoothed_gradient_descent(a, time_budget=100.0, tau_init=10.0, tau_end=500.0):
    """Nesterov accelerated gradient descent on smoothed max of conv(a,a)."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(y, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.3 * lr + 0.7 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9995)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def perturb_sequence(a, rng, method=0, scale=0.03):
    """Apply a perturbation to escape local minima."""
    n = len(a)
    a = a.copy()
    if method == 0:
        a = a * np.exp(rng.normal(0, scale, n))
    elif method == 1:
        noise = rng.uniform(1 - scale, 1 + scale, n)
        a = a * noise
    elif method == 2:
        mask = rng.random(n) > scale * 3
        a = a * mask
        if a.sum() < 1e-10:
            a = np.ones(n) / n
    elif method == 3:
        shift = int(rng.integers(1, max(2, n // 10)))
        a = np.roll(a, shift)
    elif method == 4:
        a = (1 - scale) * a + scale / n
    elif method == 5:
        A = fft(a, 2 * n)
        A[1:-1] *= np.exp(1j * rng.normal(0, scale, len(A[1:-1])))
        a = ifft(A).real[:n]
        a = np.maximum(a, 0)
    elif method == 6:
        # Frequency domain amplitude perturbation
        A = fft(a, 2 * n)
        A[1:-1] *= (1 + rng.normal(0, scale * 0.5, len(A[1:-1])))
        a = ifft(A).real[:n]
        a = np.maximum(a, 0)
    elif method == 7:
        # Peak-targeted perturbation: reduce elements that contribute most to peaks
        conv = compute_conv(a / a.sum())
        top_k = np.argsort(conv)[::-1][:8]
        deltas = np.zeros(n)
        for k in top_k:
            k = int(k)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    deltas[j] += a[idx]
        max_d = np.max(np.abs(deltas))
        if max_d > 1e-10:
            a = a - scale * 0.3 * deltas / max_d
        a = np.maximum(a, 0)
    else:
        a = a * np.exp(rng.normal(0, scale, n))

    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        #Cosine window
        a = np.cos(np.pi * np.arange(n) / (2 * n - 2))
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)

                alloc2 = min(alloc * 0.5, remaining * 0.3)
                if alloc2 > 10:
                    a_pert, obj_pert, _ = lp_polish_round(a_current, max_iter=50,
                                                           time_limit=alloc2)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        a_current = a_pert.copy()
                        print(f"  LP round {rnd} (perturbed): obj={best_score:.8f} **")

    # === Phase 4: Another QP pass on the polished result ===
    print(f"\n=== Phase 4: QP Refinement ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.20, 200.0)
    if p4_budget > 20:
        a_qp2 = qp_descent(best_seq, time_budget=p4_budget, verbose=True)
        obj_qp2 = compute_obj(a_qp2)
        if obj_qp2 < best_score:
            best_score = obj_qp2
            best_seq = a_qp2.copy()
            print(f"  QP refinement: obj={best_score:.8f} **")
        else:
            print(f"  QP refinement: obj={obj_qp2:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(300, n_current - 150), max(300, n_current - 100),
        max(300, n_current - 50),
        n_current + 50, n_current + 100, n_current + 200,
        500, 600, 700, 800, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500 and n != n_current]
    n_options.sort(key=lambda x: abs(x - n_current))

    for n_try in n_options[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick QP
        alloc1 = min(remaining * 0.12, 60.0)
        if alloc1 > 15:
            a_try = qp_descent(a_try, time_budget=alloc1, verbose=False)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.10, 50.0)
        if alloc2 > 10:
            a_try, obj_try, _ = lp_polish_round(a_try, max_iter=40, time_limit=alloc2)
        else:
            obj_try = compute_obj(a_try)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        mode = r % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.10 + 0.05 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.2, 100.0)
        if alloc > 30:
            half = alloc * 0.5
            a_r = qp_descent(a_r, time_budget=half, verbose=False)
            remaining = deadline - time.monotonic()
            lp_alloc = min(remaining * 0.1, 40.0)
            if lp_alloc > 10:
                a_r, obj_r, _ = lp_polish_round(a_r, max_iter=30, time_limit=lp_alloc)
            else:
                obj_r = compute_obj(a_r)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Final QP
    if remaining > 40:
        alloc = min(remaining * 0.3, 100.0)
        a_sgd = qp_descent(best_seq, time_budget=alloc, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Final QP: obj={best_score:.8f} **")
        else:
            print(f"  Final QP: obj={obj_sgd:.8f}")

    # Final LP rounds
    for lp_rnd in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * 0.25, 60.0)
        if alloc < 10:
            break
        a_current = best_seq.copy()

        if lp_rnd > 0:
            a_current = perturb_sequence(a_current, rng, method=lp_rnd % 5, scale=0.01)

        a_lp, obj_lp, iters = lp_polish_round(a_current, max_iter=80, time_limit=alloc)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP {lp_rnd}: obj={best_score:.8f} ({iters} iters) **")
        else:
            print(f"  Final LP {lp_rnd}: obj={obj_lp:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
