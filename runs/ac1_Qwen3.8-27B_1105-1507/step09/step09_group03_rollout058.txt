    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smooth_gradient_fft(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*conv_k)) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    
    # Diagonal correction
    for j in range(n):
        k = 2 * j
        if 0 <= k < 2 * n - 1 and p[k] > 0:
            g[j] -= p[k] * a[j]
    
    return g, conv

def exact_subgradient(a, top_k=5):
    """Compute subgradient using top-k peaks with inverse-value weighting."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    # Get top-k peaks
    idx = np.argsort(conv)[::-1][:top_k]
    vals = conv[idx]
    # Weight by proximity to max
    weights = np.exp(50 * (vals - max_conv))
    weights = weights / weights.sum()
    
    p = np.zeros(2 * n - 1)
    for i, idx_i in enumerate(idx):
        p[idx_i] = weights[i]
    
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    
    for j in range(n):
        k = 2 * j
        if 0 <= k < 2 * n - 1 and p[k] > 0:
            g[j] -= p[k] * a[j]
    
    return g, conv

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def nesterov_smoothed(a, time_budget=200.0, tau_init=20.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed max-conv objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:

    a = a / s

    best_a = a.copy()

    beta = 0.85

    best_it = 0
    prev_a = a.copy()

    while time.monotonic() - start < time_budget:

        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)

        # Nesterov look-ahead

        y = np.maximum(y, 0)
        s_y = float(np.sum(y))

            y = best_a.copy()
        else:
            y = y / s_y

        else:

        # Barzilai-Borwein step size

            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.6 * lr + 0.4 * min(max(bb_lr, 1e-10), 5.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = y.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)

        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            beta = 0.8
            best_it = it

    return best_a

def anti_symmetric_init(n, rng):
    """Create initialization with strong anti-symmetric component."""
    a = np.ones(n) / n
    # Add anti-symmetric perturbation
    t = np.linspace(0, 1, n)
    # Anti-symmetric: f(t) = -f(1-t)
    anti = np.sin(np.pi * t)  # This is symmetric! sin(pi*(1-t)) = sin(pi*t)
    # Need truly anti-symmetric: f(t) - f(1-t)
    base = np.sin(2 * np.pi * t)  # sin(2pi(1-t)) = -sin(2pi*t), so this IS anti-symmetric
    a += 0.3 * base
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Anti-symmetric bias
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * t)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    elif mode == 3:
        # Two-bump anti-symmetric
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        a += 1.0 * np.exp(-((np.arange(n) - centers[0]) ** 2) / (2 * widths[0] ** 2))
        a += 0.5 * np.exp(-((np.arange(n) - centers[1]) ** 2) / (2 * widths[1] ** 2))
        a = a / a.sum()
    elif mode == 4:
        # Power law
        t = np.linspace(0.01, 1, n)
        a = t ** 1.5
        a = a / a.sum()
    elif mode == 5:
        # Random multi-frequency
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f in [2, 3, 5]:
            a += 0.2 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = anti_symmetric_init(n, rng)
    return a

def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    from scipy.interpolate import interp1d
    f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
    a_new = f(x_new)
    return np.maximum(a_new, 0)

Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated gradient descent with adaptive smoothing,
peak-tracking subgradients, non-symmetric structural priors, and

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.interpolate import interp1d

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def gradient_smoothed(a, tau):
    """Gradient of (1/tau)*logsumexp(tau * conv_k) - log(n) smoothing."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exps = np.exp(tau * (conv - max_c))
    p = exps / float(np.sum(exps))
    
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    gf = ifft(P * np.conj(A)).real
    g = 2.0 * gf[:n]
    
    # Diagonal correction
    diag_idx = np.arange(0, 2 * n - 1, 2)  # k = 2j for j = 0..n-1
    p_diag = np.zeros(2 * n - 1)
    p_diag[diag_idx] = p[diag_idx]
    corr = np.zeros(n)
    for j in range(n):
        k = 2 * j
        if 0 <= k < 2 * n - 1:
            corr[j] = p[k] * a[j]
    g -= corr
    
    return g, conv

def gradient_topk(a, k=10):
    """Subgradient using top-k peaks with soft weighting."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    
    # Soft top-k selection
    sorted_idx = np.argsort(conv)[::-1]
    topk_idx = sorted_idx[:k]
    topk_vals = conv[topk_idx]
    
    # Exponential weighting based on proximity to max
    weights = np.exp(30.0 * (topk_vals - max_c))
    weights = weights / float(np.sum(weights))
    
    p = np.zeros(2 * n - 1)
    for i in range(k):
        p[topk_idx[i]] = weights[i]
    
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    gf = ifft(P * np.conj(A)).real
    g = 2.0 * gf[:n]
    
    # Diagonal correction
    for j in range(n):
        k_val = 2 * j
        if 0 <= k_val < 2 * n - 1 and p[k_val] > 0:
            g[j] -= p[k_val] * a[j]
    
    return g, conv

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Combines two-stage LP polishing (fixed bug from previous code), Nesterov-accelerated

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf

def gradient_smoothed(a, tau):

def build_cross_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a  # conv(a,g)[k] = sum_j a[k-j]*g[j]
    # Actually: conv(a,g)[k] = sum_j a[j]*g[k-j] = sum_j a[k-j]*g[j]
    # So A[k,j] = a[k-j] if 0<=k-j<n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(2 * n - 1, j + n)
        for k in range(lo, hi):
            A[k, j] = a[k - j]
    return A


def solve_lp_direction(a, use_active=None, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k (or active set), sum(g)=1, g>=0.
    Returns g (normalized direction) and M.
    Uses cvxpy with multiple solver fallbacks.
    """
    import cvxpy as cp

    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_cross_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)

    if use_active is not None:
        # Minimize sum over active indices
        active_A = A[use_active, :]
        obj = cp.sum(active_A @ g)
        constraints = [cp.sum(g) == 1]
    else:
        M = cp.Variable()
        obj = M
        constraints = [A @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(obj), constraints)

    solvers = [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    if use_active is not None:
                        return g_val, None
                    else:
                        return g_val, float(M.value)
        except Exception:
            continue
    return None, None

        # Stagnation handling
        if it - best_it > 40000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = max(lr * 0.5, 1e-10)
            beta = 0.7
            best_it = it
        if it - best_it > 80000:
            rng_r = np.random.default_rng(it)
            a = best_a * np.exp(rng_r.normal(0, 0.04, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)
            beta = 0.85
            best_it = it

    return best_a


def peak_deflation(a, time_budget=50.0, verbose=False):
    """
    Peak deflation: repeatedly target the dominant peak with exact gradient,
    using backtracking line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_k = int(np.argmax(conv))

        # Compute exact gradient of conv[max_k]
        grad = np.zeros(n)
        for j in range(n):
            idx = max_k - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]
                if idx == j:
                    grad[j] = a[idx]

        # Vectorized version for speed
        # Actually let's use FFT
        p = np.zeros(2 * n - 1)
        p[max_k] = 1.0
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        gf = ifft(P * np.conj(A)).real
        grad = 2.0 * gf[:n]
        # Diagonal correction
        if max_k % 2 == 0:
            j = max_k // 2
            if 0 <= j < n:
                grad[j] -= a[j]

        # Line search in negative gradient direction
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break

        improved = False
        base_step = 0.5 / gn
        for alpha_mult in [1.0, 0.5, 0.2, 0.1, 0.05]:
            alpha = alpha_mult * base_step
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            stall += 1
            if stall > 40:
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if it > 500:
                    break
        else:
            stall = 0

        it += 1
        if verbose and it % 50 == 0:
            print(f"        PD it{it}: obj={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
    a_new = f(x_new)
    return np.maximum(a_new, 0)


def create_init(n, rng, mode=0):
    """Create diverse initializations with non-symmetric structure."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        return a / a.sum()
    elif mode == 2:
        # Anti-symmetric sine (key structural insight)
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.35 * np.sin(2 * np.pi * t)
        a = np.maximum(a, 0.05)
        return a / a.sum()
    elif mode == 3:
        # Skewed power law
        t = np.linspace(0.01, 1, n)
        a = t ** 1.8
        return a / a.sum()
    elif mode == 4:
        # Asymmetric two-bump
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-((x - 0.25 * n) ** 2) / (0.04 * n)) + 0.4 * np.exp(-((x - 0.7 * n) ** 2) / (0.02 * n))
        return a / a.sum()
    elif mode == 5:
        # Multi-frequency
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f in [2, 3, 5]:
            a += 0.12 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.02)
        return a / a.sum()
    elif mode == 6:
        # Comb with varying teeth
        a = np.ones(n)
        a[::3] = 0.1 + 0.1 * rng.random(1)
        return a / a.sum()
    else:
        return rng.dirichlet(np.ones(n) * 0.3)


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization:
    1. Two-stage LP polishing (fixed bug, robust solver selection)
    2. Nesterov accelerated smoothed gradient with adaptive tau
    3. Peak deflation with exact subgradient
    4. Multi-scale n exploration
    5. Diverse anti-symmetric restarts
    6. Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_default = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_default = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent available: {e}")

    if best_seq is None:
        best_seq = create_init(n_default, rng, mode=2)
        best_score = compute_obj(best_seq)
        print(f"Anti-symmetric init (n={n_default}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Two-stage LP polishing ===
    print(f"\n=== Phase 1: Two-Stage LP (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    budget = min(remaining * 0.22, 280.0)
    if budget > 20:
        a1 = two_stage_lp(best_seq, time_budget=budget, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  2SLP: obj={best_score:.8f} **")
        else:
            print(f"  2SLP: obj={obj1:.8f}")

    # === Phase 2: Nesterov smoothed gradient ===
    print(f"\n=== Phase 2: Nesterov Smoothed GD ===")
    remaining = deadline - time.monotonic()
    budget = min(remaining * 0.2, 250.0)
    if budget > 20:
        a2 = nesterov_smoothed(best_seq, time_budget=budget,
                               tau_init=15.0, tau_end=350.0, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  NSH: obj={best_score:.8f} **")
        else:
            print(f"  NSH: obj={obj2:.8f}")

    # === Phase 3: Peak deflation ===
    print(f"\n=== Phase 3: Peak Deflation ===")
    remaining = deadline - time.monotonic()
    budget = min(remaining * 0.08, 80.0)
    if budget > 10:
        a3 = peak_deflation(best_seq, time_budget=budget, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  PD: obj={best_score:.8f} **")
        else:
            print(f"  PD: obj={obj3:.8f}")

    # === Phase 4: Second LP pass ===
    print(f"\n=== Phase 4: Second LP Pass ===")
    remaining = deadline - time.monotonic()
    budget = min(remaining * 0.12, 150.0)
    if budget > 15:
        a4 = two_stage_lp(best_seq, time_budget=budget, verbose=False)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  2SLP2: obj={best_score:.8f} **")
        else:
            print(f"  2SLP2: obj={obj4:.8f}")

    # === Phase 5: Multi-scale n exploration ===
    print(f"\n=== Phase 5: Multi-scale n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Generate candidate n values
    n_candidates = set()
    for delta in [-200, -100, 100, 200, 400, 600, 800]:
        nc = n_best + delta
        if 200 <= nc <= 3000:
            n_candidates.add(nc)
    for fixed in [400, 600, 800, 1000, 1200, 1500, 2000]:
        if abs(fixed - n_best) > 50:
            n_candidates.add(fixed)
    n_candidates = sorted(n_candidates)

    n_tried = 0
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_tried >= 5:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.12, 50.0)
        if alloc > 10:
            a_try = nesterov_smoothed(a_try, time_budget=alloc,
                                      tau_init=30, tau_end=300)

        obj_try = compute_obj(a_try)
        n_tried += 1
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        mode = r % 7
        a_r = create_init(n_best, rng, mode=mode)
        # Blend with best for warm start
        blend = 0.35
        a_r = (1.0 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.12, 60.0)
        if alloc > 15:
            # First: quick Nesterov
            half1 = alloc * 0.5
            a_r = nesterov_smoothed(a_r, time_budget=half1,
                                    tau_init=20, tau_end=300)
            remaining = deadline - time.monotonic()
            if remaining > 10:
                half2 = min(remaining * 0.3, 25.0)
                a_r = peak_deflation(a_r, time_budget=half2)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-precision Nesterov
    if remaining > 20:
        alloc = min(remaining * 0.35, 100.0)
        a_f = nesterov_smoothed(best_seq, time_budget=alloc,
                                tau_init=100, tau_end=1500, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final NSH: obj={best_score:.8f} **")
        else:
            print(f"  Final NSH: obj={obj_f:.8f}")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 60.0)
        a_f = two_stage_lp(best_seq, time_budget=alloc, verbose=False)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final 2SLP: obj={best_score:.8f} **")

    # Final peak deflation
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.3, 40.0)
        a_f = peak_deflation(best_seq, time_budget=alloc)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final PD: obj={best_score:.8f} **")

    # Final Nesterov with very high tau
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.4, 50.0)
        a_f = nesterov_smoothed(best_seq, time_budget=alloc,
                                tau_init=300, tau_end=3000)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final NSH2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
