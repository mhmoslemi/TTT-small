"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based smoothed gradient descent with Nesterov acceleration,
LP-based local optimization (fixed from previous bug), subgradient methods,
and multi-start strategy. Key innovations: efficient O(n log n) gradient
computation via cross-correlation, adaptive tau schedule, and diverse
initialization with blending.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def compute_obj_normalized(a):
    """Compute objective assuming already normalized (sum=1)."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fft_gradient_smoothed(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT.
    Returns (gradient, conv) where gradient has same shape as a.
    """
    n = len(a)
    # Step 1: Compute convolution
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = fft(a, size)
    conv = ifft(A * A).real[:2 * n - 1]
    
    # Step 2: Compute softmax weights
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    # Step 3: Compute gradient via cross-correlation
    # grad[i] = 2 * sum_j p[i+j] * a[j] (with proper bounds)
    # This is the cross-correlation of p and a
    P = fft(p, size)
    g_full = ifft(P * np.conj(A)).real[:n]
    g = 2.0 * g_full
    return g, conv

def nesterov_smoothed_descent(a, time_budget=300.0, tau_init=5.0, tau_end=500.0, verbose=False):
    """
    Nesterov accelerated gradient descent on smoothed objective over simplex.
    The objective is max_k (a*a)_k subject to a>=0, sum(a)=1.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    beta = 0.7  # momentum
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    # For Barzilai-Borwein
    prev_a = a.copy()
    prev_g = None

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        # Exponential tau schedule
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)

        if it % 5000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute gradient at y
        g_y, conv_y = fft_gradient_smoothed(y, tau)

        # Barzilai-Borwein step size update
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g_y - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-18 and abs(sy) > 1e-15:
                bb_lr = ss / abs(sy)
                lr = 0.7 * lr + 0.3 * min(max(bb_lr, 1e-10), 50.0 / np.sqrt(n))

        # Gradient step
        v = beta * v - lr * g_y
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g_y.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9995)

        # Reset if stuck
        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 5.0 / np.sqrt(n) * 0.5
            best_it = it
            beta = max(0.3, beta * 0.9)

    return best_a

def build_conv_cross_matrix(a):
    """
    Build sparse matrix A (2n-1 x n) such that A @ g = conv(a, g).
    Uses the structure: (conv(a,g))[k] = sum_i a[i]*g[k-i]
    """
    n = len(a)
    # A[k, j] = a[j] if 0 <= k-j < n, else 0
    # Actually A[k, j] = a[k-j] if 0 <= k-j < n... 
    # Wait: conv(a,g)[k] = sum_j a[j]*g[k-j] (valid j, valid k-j)
    # So A[k,j] = a[j] if 0<=j<n and 0<=k-j<n, else 0
    # Which means for row k, column j is nonzero when 0<=j<n and 0<=k-j<n
    # i.e., k-n+1 <= j <= k, intersected with [0, n-1]
    # A[k,j] = a[j]
    
    # This is a banded matrix! Row k has nonzeros at columns max(0,k-n+1) to min(k, n-1)
    # For efficiency, build as dense for small n, or use scipy sparse
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # For column j, row k is nonzero when 0 <= k-j < n, i.e., j <= k <= j+n-1
        row_start = j
        row_end = min(j+n, 2*n-1)
        A[row_start:row_end, j] = a[:row_end-row_start]
    return A

def solve_lp_direction(a, time_per_lp=2.0, max_iters=50, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns the optimal g direction.
    """
    n = len(a)
    a = a / np.sum(a)  # normalize
    A = build_conv_cross_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    start = time.monotonic()
    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        if time.monotonic() - start > time_per_lp * 2:
            break
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def solve_lp_active_set(a, active_indices, time_per_lp=1.0):
    """
    Solve LP: min sum(conv(a,g)[k] for k in active_indices) s.t. sum(g)=1, g>=0.
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_cross_matrix(a)
    A_active = A[active_indices, :]

    g = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(g) == 1]
    obj = cp.sum(A_active @ g)

                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))

        except Exception:

    return None

def subgradient_descent(a, time_budget=100.0, verbose=False):
    """
    Exact subgradient method for the non-smooth problem min max_k (a*a)_k.
    Uses the fact that subgradient at active peaks is the average of their gradients.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    lr = 2.0 / n
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        
        # Find active peaks (within tolerance)
        tol = 1e-6 * max_val
        active = np.where(conv >= max_val - tol)[0]
        
        # Compute subgradient: average of gradients at active peaks
        # grad[i] = 2/a[i] * sum_{k in active} a[k-i] (if valid)
        # More precisely: subgrad[i] = (1/|active|) * sum_{k in active} 2*a[k-i] (if 0<=k-i<n)
        subgrad = np.zeros(n)
        for k in active:
            # grad_k[i] = 2*a[k-i] if 0<=k-i<n else 0
            valid = (k - np.arange(n) >= 0) & (k - np.arange(n) < n)
            idx = k - np.arange(n)
            subgrad[valid] += 2.0 * a[idx[valid]]
        subgrad /= len(active)

        # Gradient step with projection
        a_new = a - lr * subgrad
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            a = a_new
        else:
            # Try smaller step
            for scale in [0.5, 0.25, 0.1]:
                a_try = project_simplex(a - lr * scale * subgrad)
                obj_try = compute_obj_normalized(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    break

        it += 1
        lr = max(1e-8, lr * 0.999)

        if it - best_it > 5000:
            a = best_a.copy()
            lr = 2.0 / n * 0.5
            best_it = it

    return best_a

def coordinate_descent(a, time_budget=60.0, verbose=False):
    """
    Randomized coordinate descent: pick random pairs (i,j) and try to
    redistribute mass between them to reduce the max convolution peak.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        if it % 1000 == 0 and verbose:
            print(f"    CD it{it}: obj={best_obj:.8f}")
        
        # Try random coordinate moves
        improved = False
        for _ in range(20):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            # Try moving epsilon from i to j
            for eps_frac in [0.01, 0.005, 0.002]:
                eps = eps_frac * a[i]
                if eps > a[i]:
                    continue
                a_try = a.copy()
                a_try[i] -= eps
                a_try[j] += eps
                a_try = a_try / a_try.sum()
                obj_try = compute_obj_normalized(a_try)
                if obj_try < best_obj - 1e-15:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            if improved:
                break

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break

    return best_a

def multi_start_optimize(n, time_budget=60.0, verbose=False):
    """Try several different initializations and keep the best."""
    rng = np.random.default_rng(99)
    best_a = None
    best_obj = np.inf
    
    # Several initialization strategies
    inits = []
    inits.append(np.ones(n) / n)  # uniform
    inits.append(rng.uniform(0.1, 1, n)); inits[-1] /= inits[-1].sum()  # random
    x = np.linspace(0, 1, n); inits.append(np.exp(-((x-0.5)**2)/0.05)); inits[-1] /= inits[-1].sum()  # gaussian
    inits.append(1.0/np.sqrt(np.abs(x-0.5)+0.1)); inits[-1] /= inits[-1].sum()  # u-shaped
    inits.append(np.abs(np.sin(np.pi*x))); inits[-1] /= inits[-1].sum()  # sine

    for a_init in inits:
        if time.monotonic() > time_budget:

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based smoothed gradient descent with Nesterov acceleration and adaptive
tau schedule, LP-based local optimization (with active-set reduction), mirror
descent (exponential weights), subgradient methods, and multi-start/multi-length
exploration. Key innovations: O(n log n) gradient via cross-correlation FFT,
Barzilai-Borwein adaptive step size, and diverse initialization strategies.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(np.asarray(a, dtype=np.float64), size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1: returns 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj(a):
    """Compute full objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a))) / (s * s)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fft_gradient_smoothed(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*(a*a)_k)) using FFT.
    Returns (gradient, conv) where gradient has same shape as a.
    The gradient is: sum_k p_k * 2 * shift_reverse_k(a)
    where p_k = exp(tau*conv_k) / sum_j exp(tau*conv_j)
    """
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = fft(a, size)
    conv = ifft(A * A).real[:2 * n - 1]

    # Softmax weights (numerically stable)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    total = float(np.sum(exp_vals))
    if total < 1e-300:
        p = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        p = exp_vals / total

    # Cross-correlation: grad[i] = 2 * sum_j p[i+j] * a[j] (valid indices)
    P = fft(p, size)
    g_full = ifft(P * np.conj(A)).real[:n]
    g = 2.0 * g_full
    return g, conv

def build_conv_cross_matrix(a):
    """
    Build matrix A (2n-1 x n) such that A @ g = conv(a, g).
    (conv(a,g))[k] = sum_j a[j]*g[k-j] for valid indices.
    So A[k,j] = a[j] if 0<=j<n and 0<=k-j<n, else 0.
    Equivalently: for column j, rows j to j+n-1 get values a[0], a[1], ..., a[n-1].
    Wait, let me recheck: conv(a,g)[k] = sum_j a[j]*g[k-j].
    For fixed j, this contributes a[j]*g[k-j] to row k, where k-j must be valid (0<=k-j<n).
    So row k, column j: A[k,j] = a[j] if 0<=k-j<n, i.e., j<=k<=j+n-1.
    And the value is a[j].
    """
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        row_start = j
        row_end = min(j + n, 2 * n - 1)
        length = row_end - row_start
        A[row_start:row_end, j] = a[:length]
    return A


def solve_lp_direction(a, timeout=5.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_direction, M_value) or (None, None).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    a = a / float(np.sum(a))
    A = build_conv_cross_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    start = time.monotonic()
    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.MOSEK]
    for solver in solvers:
        if time.monotonic() - start > timeout:
            break
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_active(a, active_indices, timeout=3.0):
    """
    Solve LP: min sum_{k in active} conv(a,g)[k] s.t. sum(g)=1, g>=0.

                s_try = float(np.sum(a_try))
                if s_try > 1e-30:
                    a_try = a_try / s_try
                    obj_try = compute_obj_normalized(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        lr = lr_try
                        found = True
                        break
            if not found:
                lr *= 0.9

        it += 1
        lr = max(1e-10, lr * 0.9998)

        if it - best_it > 30000:
            a = best_a.copy()
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def subgradient_polish(a, time_budget=80.0, verbose=False):
    """
    Subgradient method targeting the exact max peak positions.
    Uses the analytical gradient formula: d/d a_i [(a*a)_k] = 2*a[k-i] (if valid).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    lr = 1.0 / n
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))

        # Find active peaks
        tol = max(1e-8, 1e-6 * max_val)
        active = np.where(conv >= max_val - tol)[0]

        # Compute subgradient
        subgrad = np.zeros(n)
        for k in active:
            # grad_k[i] = 2*a[k-i] if 0 <= k-i < n
            idx = k - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            subgrad[valid] += 2.0 * a[idx[valid]]
        subgrad /= float(len(active))

        # Try step
        a_new = project_simplex(a - lr * subgrad)
        obj_new = compute_obj_normalized(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            a = a_new
        else:
            # Backtrack
            found = False
            for scale in [0.5, 0.25, 0.1]:
                a_try = project_simplex(a - lr * scale * subgrad)
                obj_try = compute_obj_normalized(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    found = True
                    break
            if not found:
                lr *= 0.8

        it += 1
        lr = max(1e-10, lr * 0.9995)

        if it - best_it > 10000:
            a = best_a.copy()
            lr = 1.0 / n * 0.5
            best_it = it

    return best_a


def coordinate_descent(a, time_budget=40.0, verbose=False):
    """
    Randomized coordinate descent: try small mass transfers between pairs.
    Focuses on reducing the dominant peak.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        # Identify the peak position
        conv = compute_conv(a)
        peak_k = int(np.argmax(conv))

        # Find contributing indices to the peak
        i_lo = max(0, peak_k - (n - 1))
        i_hi = min(n - 1, peak_k)

        # Try reducing mass at high-contributing positions
        improved = False
        for _ in range(30):
            # Pick source from high contributors or random
            if rng.random() < 0.5 and i_lo <= i_hi:
                src = int(rng.integers(i_lo, i_hi + 1))
            else:
                src = int(rng.integers(0, n))

            if a[src] < 1e-6:
                continue

            # Pick target: prefer positions that don't contribute to the peak
            tgt = int(rng.integers(0, n))
            if tgt == src:
                continue

            for eps_frac in [0.005, 0.002, 0.001]:
                eps = eps_frac * a[src]
                if eps >= a[src]:
                    continue
                a_try = a.copy()
                a_try[src] -= eps
                a_try[tgt] += eps
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj_normalized(a_try)
                if obj_try < best_obj - 1e-15:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            if improved:
                break

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 200:
                break

        if verbose and it % 500 == 0:
            print(f"    CD it{it}: obj={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def make_init_uniform(n):
    return np.ones(n) / n


def make_init_random(n, rng):
    a = rng.uniform(0.1, 1.0, n)
    return a / a.sum()


def make_init_hann(n):
    x = np.arange(n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    return a / a.sum()


def make_init_semicircle(n):
    x = np.linspace(-1, 1, n)
    a = np.sqrt(np.maximum(1.0 - x * x, 0.01))
    return a / a.sum()


def make_init_bimodal(n, rng):
    c1 = n // 4
    c2 = 3 * n // 4
    w = n // 10
    x = np.arange(n)
    a = np.exp(-((x - c1) ** 2) / (2 * w * w)) + np.exp(-((x - c2) ** 2) / (2 * w * w))
    return a / a.sum()


def make_init_exponential(n, rng):
    x = np.linspace(0.01, 1, n)
    p = rng.uniform(-2, 2)
    a = x ** p
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization:
    1. FFT smoothed GD (Nesterov) - main workhorse
    2. LP-based local polishing (fixed active-set approach)
    3. Mirror descent (multiplicative updates)
    4. Subgradient polish
    5. Coordinate descent
    6. Multi-length exploration
    7. Multi-start with diverse initializations
    8. Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj_normalized(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent available: {e}")

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj_normalized(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: FFT Smoothed GD (main workhorse) ===
    print(f"\n=== Phase 1: FFT Smoothed GD (Nesterov) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.28, 280.0)
    if p1_budget > 30:
        a1 = nesterov_smoothed_descent(best_seq, time_budget=p1_budget,
                                       tau_init=5.0, tau_end=400.0, verbose=True)
        obj1 = compute_obj_normalized(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj1:.8f}")

    # === Phase 2: LP-based polishing ===
    print(f"\n=== Phase 2: LP Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.22, 180.0)
    if p2_budget > 30:
        a2 = lp_polish(best_seq, time_budget=p2_budget, verbose=True)
        obj2 = compute_obj_normalized(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj2:.8f}")

    # === Phase 3: Mirror Descent ===
    print(f"\n=== Phase 3: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.18, 150.0)
    if p3_budget > 20:
        a3 = mirror_descent(best_seq, time_budget=p3_budget,
                            tau_init=10.0, tau_end=800.0, verbose=True)
        obj3 = compute_obj_normalized(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj3:.8f}")

    # === Phase 4: Subgradient polish ===
    print(f"\n=== Phase 4: Subgradient ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.08, 60.0)
    if p4_budget > 15:
        a4 = subgradient_polish(best_seq, time_budget=p4_budget, verbose=True)
        obj4 = compute_obj_normalized(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase 4: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4: obj={obj4:.8f}")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coord Descent ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.06, 40.0)
    if p5_budget > 10:
        a5 = coordinate_descent(best_seq, time_budget=p5_budget, verbose=True)
        obj5 = compute_obj_normalized(a5)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  Phase 5: obj={best_score:.8f} **")
        else:
            print(f"  Phase 5: obj={obj5:.8f}")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(100, n_current - 200),
        n_current + 150,
        n_current + 300,
        500, 700, 1000, 1200
    ]))
    n_options = [n for n in n_options if 100 <= n <= 2000 and abs(n - n_current) > 50]

    for n_try in n_options[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        alloc = min(remaining * 0.15, 40.0)
        if alloc > 10:
            a_try = nesterov_smoothed_descent(a_try, time_budget=alloc,
                                              tau_init=10.0, tau_end=500.0)

        obj_try = compute_obj_normalized(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Multi-start with diverse initializations ===
    print(f"\n=== Phase 7: Multi-start ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    if remaining > 80:
        init_funcs = [
            make_init_random,
            make_init_hann,
            make_init_semicircle,
            make_init_bimodal,
            make_init_exponential,
        ]

        for r in range(min(5, len(init_funcs))):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            a_r = init_funcs[r](n_best, rng)
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r < 1e-10:
                continue
            a_r = a_r / s_r

            # Blend with current best for warm start effect
            blend = 0.3
            a_r = (1.0 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            s_r2 = float(np.sum(a_r))
            if s_r2 < 1e-10:
                continue
            a_r = a_r / s_r2

            obj_r_init = compute_obj_normalized(a_r)

            alloc = min(remaining * 0.2, 50.0)
            if alloc > 15:
                a_r = nesterov_smoothed_descent(a_r, time_budget=alloc,
                                                tau_init=5.0, tau_end=500.0)
                obj_r = compute_obj_normalized(a_r)
                if obj_r < best_score:
                    best_score = obj_r
                    best_seq = a_r.copy()
                    print(f"  Restart {r}: obj={best_score:.8f} **")
                else:
                    print(f"  Restart {r}: obj={obj_r:.8f} (init {obj_r_init:.8f})")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP
    if remaining > 25:
        alloc = min(remaining * 0.3, 60.0)
        a_f = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_f = compute_obj_normalized(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # High-tau GD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 40.0)
        a_f = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                        tau_init=200.0, tau_end=5000.0, verbose=False)
        obj_f = compute_obj_normalized(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Mirror descent
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.25, 30.0)
        a_f = mirror_descent(best_seq, time_budget=alloc,
                             tau_init=200.0, tau_end=3000.0, verbose=False)
        obj_f = compute_obj_normalized(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.2,
                20.0)
        a_f = coordinate_descent(best_seq, time_budget=alloc, verbose=False)
        obj_f = compute_obj_normalized(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Subgradient
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.2, 15.0)
        a_f = subgradient_polish(best_seq, time_budget=alloc, verbose=False)
        obj_f = compute_obj_normalized(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
