"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Combined LP (L-infinity) and QP (L2) direction finding with dual-guided
perturbation, symmetrization, and adaptive blending. Uses the Toeplitz structure
of the cross-convolution matrix for efficient QP solves. Multi-phase approach with
careful time allocation and periodic diversification to escape local minima.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, get_dual=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal, dual_values) or (None, None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    cons = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), cons)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    dual = None
                    if get_dual:
                        try:
                            dual = np.abs(cons[0].dual_value)
                        except Exception:
                            pass
                    return g_val / s, float(M.value), dual
        except Exception:
            continue
    return None, None, None


def solve_qp_l2(a):
    """
    Solve QP: min g^T (C_a^T C_a) g s.t. sum(g)=1, g>=0.
    This minimizes the L2 norm of conv(a, g).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    T = A.T @ A

    g = cp.Variable(n, nonneg=True)
    objective = cp.Minimize(cp.quad_form(g, cp.psd_wrap(T)))
    constraints = [cp.sum(g) == 1]
    prob = cp.Problem(objective, constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def solve_qp_weighted(a, weights):
    """
    Solve QP: min g^T W g s.t. sum(g)=1, g>=0
    where W[k] weights the peaks based on dual values.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    # Weighted: min sum_k w_k * (A g)_k^2 = g^T (A^T W A) g
    W = np.diag(weights)
    T = A.T @ W @ A

    g = cp.Variable(n, nonneg=True)
    objective = cp.Minimize(cp.quad_form(g, cp.psd_wrap(T)))
    constraints = [cp.sum(g) == 1]
    prob = cp.Problem(objective, constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def lp_qp_polish(a, time_budget, verbose=False, use_qp=True):

    Alternates between L-infinity (LP) and L2 (QP) directions.

    best_obj = compute_obj(a)

        if elapsed > time_budget - 5:
            break

        if use_qp and use_qp_next:
            g_dir = solve_qp_l2(a)
            method = "QP"
        else:

            method = "LP"

        use_qp_next = not use_qp_next

            if g_dir is None:
                break

def smoothed_gradient_descent(a, time_budget, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed max of conv(a,a)."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        if it % 5000 == 0 and verbose:
            print(f"    SGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        # Gradient: g_i = 2 * sum_k p[k] * y[k-i] (with bounds)
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        Y = fft(y, N)
        g_full = ifft(P * np.conj(Y)).real
        g = 2.0 * g_full[:n]

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)

        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def peak_shave_direct(a, time_budget, verbose=False):
    """
    Direct peak shaving: identify top peaks and solve a small QP to reduce them.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    lr = 0.2

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 2:
            break

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        top_idx = np.argsort(conv)[::-1]
        K = min(30, 2 * n - 1)
        top_K = top_idx[:K]

        # Compute weighted subgradient from top peaks
        peak_vals = conv[top_K]
        max_pv = peak_vals[0]
        weights = np.exp((peak_vals - max_pv) * 5.0 / (max_pv + 1e-15))
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_K):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        step = lr / g_norm
        a_new = a - step * g
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.01, 2.0)
        else:
            lr *= 0.7
            a = best_a
            if lr < 1e-10:
                break

        it += 1
        if verbose and it % 500 == 0:
            print(f"    PeakShave it{it}: obj={best_obj:.8f}, lr={lr:.4f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 3:
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    else:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    return a

            print(f"  Final LP+QP: obj={best_score:.8f} **")

    # Final peak shave
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 80.0)
        a_f2 = peak_shave_direct(best_seq, time_budget=alloc, verbose=False)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final PeakShave: obj={best_score:.8f} **")

    # Final short GD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 50.0)
        a_f3 = smoothed_gradient_descent(best_seq, time_budget=alloc,
                                         tau_init=50.0, tau_end=800.0, verbose=False)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # One last LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 50.0)
        a_f4 = lp_qp_polish(best_seq, time_budget=alloc, verbose=False, use_qp=False)
        obj_f4 = compute_obj(a_f4)
        if obj_f4 < best_score:
            best_score = obj_f4
            best_seq = a_f4.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
