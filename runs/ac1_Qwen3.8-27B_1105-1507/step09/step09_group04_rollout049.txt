    while N_fft < 3 * n:
        N_fft <<= 1
    
    W = np.fft.rfft(w, N_fft)
    A = np.fft.rfft(a, N_fft)
    
    # Second term: conv(w, a)
    term2 = np.fft.irfft(W * A, N_fft)[:n]
    
    # First term: cross-correlation
    # ifft(FFT(a) * conj(FFT(w))) gives circular cross-correlation
    # For linear, we need N_fft >= n + (2n-1) - 1 = 3n - 2, which is satisfied
    cross = np.fft.irfft(A * np.conj(W), N_fft)
    # The cross-correlation (a ⋆ w)[m] = sum_j a[j] * conj(w[j+m])
    # For real signals: = sum_j a[j] * w[j+m] = sum_j a[j] * w[m+j]
    # This is exactly our first term!
    term1 = cross[:n]
    
    grad = term1 + term2
    active_positions = np.where(active)[0].tolist()
    
    return grad, active_positions

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses intensive LP polishing, active-set subgradient descent with Nesterov acceleration,
adaptive coordinate descent, and multi-length exploration.
Key improvements: more aggressive LP iterations, better gradient tracking via
sliding-window active sets, and structured perturbations for escape from local minima.

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_set = np.where(cond)[0]
    if len(rho_set) == 0:
        return np.ones(n) / n
    rho = rho_set[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)

        prob2.solve(solver=cp.CLARABEL)

def compute_subgradient_fft(a, conv, rel_tol=0.005):
    """
    Compute subgradient of max(conv) using weighted active set.
    Uses FFT for efficiency.
    """
    n = len(a)
    max_conv = float(np.max(conv))
    if max_conv <= 0:
        return np.zeros(n)

    # Active set: positions where conv is within rel_tol of max
    threshold = max_conv * (1.0 - rel_tol)
    active_mask = conv >= threshold

    if not np.any(active_mask):
        k_max = int(np.argmax(conv))
        active_mask = np.zeros(2 * n - 1, dtype=bool)
        active_mask[k_max] = True

    # Weights proportional to how close to max
    weights_on_active = conv[active_mask] - threshold
    total_w = float(np.sum(weights_on_active))
    if total_w < 1e-30:
        weights_on_active = np.ones(np.sum(active_mask))
        total_w = len(weights_on_active)
    weights = weights_on_active / total_w

    # Build weight vector
    w = np.zeros(2 * n - 1)
    w[active_mask] = weights

    # Compute subgradient via FFT
    # subgrad[m] = sum_k w[k] * a[k-m] (if 0<=k-m<n) + sum_k w[k] * a[m-k] (if 0<=m-k<n)
    #
    # Term 1: cross-correlation (a ⋆ w)[m] = sum_j a[j] * w[j+m]
    # Term 2: convolution (w * a)[m] = sum_j w[j] * a[m-j]

    N_fft = 1
    while N_fft < 3 * n + 2:
        N_fft <<= 1

    W = np.fft.rfft(w, N_fft)
    A = np.fft.rfft(a, N_fft)

    # Term 2: conv(w, a) = ifft(W * A)
    term2 = np.fft.irfft(W * A, N_fft)[:n]

    # Term 1: cross-correlation (a ⋆ w)[m] = sum_j a[j]*w[j+m]
    # = ifft(FFT(a) * conj(FFT(w))) for circular; for linear need padding
    # With N_fft >= n + (2n-1) - 1 = 3n-2, the circular and linear agree for m in [0, n-1]
    term1 = np.fft.irfft(A * np.conj(W), N_fft)[:n]

    grad = term1 + term2
    return grad


def nesterov_subgradient(a, time_budget=100.0, verbose=False, rel_tol=0.005):
    """
    Nesterov-accelerated projected subgradient descent on the simplex.
    Minimize max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    conv = compute_conv(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov momentum
    beta = 0.9
    v = np.zeros(n)
    y = a.copy()
    lr = 5.0 / np.sqrt(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        # Compute subgradient at y
        conv_y = compute_conv(y)
        grad = compute_subgradient_fft(y, conv_y, rel_tol)

        # Update
        v = beta * v - lr * grad
        a_new = y + v
        a_new = project_simplex(a_new)

        # Evaluate
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        # Update y for next iteration
        y = best_a + beta * (a_new - best_a)
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y > 1e-10:
            y = y / s_y
        else:
            y = best_a.copy()

        a = a_new
        it += 1

        # Adaptive learning rate
        lr = max(1e-8, lr * 0.9995)

        # Reset if stuck
        if it - best_it > 5000:
            a = best_a.copy()
            y = a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n)
            best_it = it

    return best_a


def coord_descent(a, time_budget=100.0, rng=None, verbose=False):
    """
    Coordinate descent with golden section search on the 1D subproblem.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.ones(n) / n
    a = a / S
    conv = compute_conv(a)

    best_obj = float(np.max(conv))
    best_a = a.copy()
    gr = (np.sqrt(5) - 1) / 2.0
    stagnant = 0
    sweep = 0

    while time.monotonic() < time_budget + (time.monotonic() - 0) and False:
        pass  # placeholder

    start = time.monotonic()
    while time.monotonic() - start < time_budget:
        sweep += 1
        obj_before = float(np.max(conv))

        if rng is not None and sweep % 3 == 0:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for k in order:
            # Direction: changing a[k] by delta changes conv by 2*delta*d + delta^2 at position 2k
            # where d[m] = a[m-k] for m in [k, k+n-1]
            d = np.zeros(2 * n - 1)
            d[k:k + n] = a

            lo_delta = -a[k]
            # Upper bound
            hi_delta = 2.0 / n * 3

            # Golden section search
            a_gs = lo_delta
            b_gs = hi_delta
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)

            # Evaluate at interior points
            val_c = conv + 2.0 * c_gs * d
            val_c[2 * k] += c_gs * c_gs
            fc = float(np.max(val_c))

            val_d = conv + 2.0 * d_gs * d
            val_d[2 * k] += d_gs * d_gs
            fd = float(np.max(val_d))

            for _ in range(40):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    val_c = conv + 2.0 * c_gs * d
                    val_c[2 * k] += c_gs * c_gs
                    fc = float(np.max(val_c))
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    val_d = conv + 2.0 * d_gs * d
                    val_d[2 * k] += d_gs * d_gs
                    fd = float(np.max(val_d))

            delta_opt = (a_gs + b_gs) / 2.0

            # Check improvement
            val_new = conv + 2.0 * delta_opt * d
            val_new[2 * k] += delta_opt * delta_opt
            new_obj_est = float(np.max(val_new))

            if new_obj_est >= obj_before - 1e-14:
                continue

            # Apply update
            a[k] = max(0.0, a[k] + delta_opt)

            # Incremental convolution update
            conv += 2.0 * delta_opt * d
            conv[2 * k] += delta_opt * delta_opt
            # Fix small negatives from numerical error
            conv = np.maximum(conv, 0)

            # Re-normalize
            S_new = float(np.sum(a))
            if S_new < 1e-10:
                a[:] = 1.0 / n
                conv = compute_conv(a)
            else:
                a /= S_new
                conv /= (S_new * S_new)

        obj_after = float(np.max(conv))
        if obj_after < best_obj - 1e-13:
            best_obj = obj_after
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 8 and rng is not None:

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if new_n == 1:
        return np.array([a.sum()])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_init(n, rng, mode=0):
    """Generate diverse structured initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        # Sine window
        x = np.linspace(0, np.pi, n)
        a = np.sin(x)
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif mode == 2:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif mode == 3:
        # Two Gaussian bumps
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 10)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 4:
        # Power spectrum shaped (target flat autocorrelation)
        # Start with target autocorrelation = constant, take IFFT
        N = 2 * n - 1
        target = np.ones(N)
        target[0] = 2.0  # Make it positive definite
        # Add some structure
        for k in range(1, N // 2):
            target[k] = target[N - k] = 1.0 + 0.1 * np.cos(2 * np.pi * k / N)
        # IFFT to get time domain
        A = np.fft.rfft(target, N)
        # Magnitude squared spectrum -> time domain
        # Actually we want a such that (a*a) is flat
        # So |FFT(a)|^2 should be flat, meaning |FFT(a)| = constant
        # This gives a as a DFT of a constant-magnitude spectrum
        # But we need a >= 0, so this might not work directly
        # Just use a random phase
        phase = rng.uniform(0, 2 * np.pi, N // 2 + 1)
        a_full = np.fft.irfft(np.exp(1j * phase), N)[:n]
        a_full = np.maximum(a_full, 0)
        if a_full.sum() < 1e-10:
            return np.ones(n) / n
        return a_full / a_full.sum()
    elif mode == 5:
        # Random smooth
        a = rng.uniform(0.5, 1.5, n)
        # Smooth with a Gaussian
        kernel_size = max(3, n // 50)
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=kernel_size / 2)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 6:
        # Uniform with random bumps
        a = np.ones(n) / n
        for _ in range(3):
            c = rng.integers(0, n)
            w = rng.integers(n // 20, n // 5)
            a += 0.5 * np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 0)
        return a / a.sum()
    else:
        a = rng.uniform(0, 1, n)
        return a / a.sum()


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: intensive LP polishing + subgradient descent + coordinate descent
    + multi-length exploration + structured restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent loaded (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        # Try a few structured inits
        for mode in [1, 2, 5, 0]:
            a_try = structured_init(n_target, rng, mode=mode)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"Init mode {mode} (n={n_target}): obj={obj_try:.8f}")

    if best_seq is None:
        best_seq = np.ones(500) / 500
        best_score = compute_obj(best_seq)
        print(f"Default init (n=500): obj={best_score:.8f}")

    print(f"\nStarting optimization: n={len(best_seq)}, score={best_score:.8f}")

    # === Phase 1: Intensive LP Polishing (primary workhorse) ===
    print(f"\n=== Phase 1: Intensive LP Polishing ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 450.0)
    if p1_budget > 30:
        a_lp = lp_polish(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1 LP: obj={best_score:.8f} ** IMPROVED")
        else:
            print(f"  Phase1 LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: Coordinate Descent Fine-tuning ===
    print(f"\n=== Phase 2: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.20, 250.0)
    if p2_budget > 30:
        a_cd = coord_descent(best_seq, time_budget=p2_budget, rng=rng, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase2 CD: obj={best_score:.8f} ** IMPROVED")
        else:
            print(f"  Phase2 CD: obj={obj_cd:.8f}")

    # === Phase 3: More LP Polishing ===
    print(f"\n=== Phase 3: LP Polishing Round 2 ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.25, 300.0)
    if p3_budget > 30:
        a_lp2 = lp_polish(best_seq, time_budget=p3_budget, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase3 LP: obj={best_score:.8f} ** IMPROVED")
        else:
            print(f"  Phase3 LP: obj={obj_lp2:.8f}")

    # === Phase 4: Subgradient Descent ===
    print(f"\n=== Phase 4: Nesterov Subgradient ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 150.0)
    if p4_budget > 30:
        a_sgd = nesterov_subgradient(best_seq, time_budget=p4_budget, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase4 SGD: obj={best_score:.8f} ** IMPROVED")
        # Follow up with LP if improved
        if obj_sgd < best_score + 1e-8:
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_follow = lp_polish(best_seq, time_budget=min(60, remaining * 0.3), verbose=True)
                obj_follow = compute_obj(a_follow)
                if obj_follow < best_score:
                    best_score = obj_follow
                    best_seq = a_follow.copy()
                    print(f"  Phase4 follow-up LP: obj={best_score:.8f} ** IMPROVED")

    # === Phase 5: Multi-length Exploration ===
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Try neighboring lengths
    n_options = [n_current]
    for delta in [-200, -100, -50, 50, 100, 200, 300, 400]:
        n_try = n_current + delta
        if 300 <= n_try <= 2000 and n_try not in n_options:
            n_options.append(n_try)
    # Also try specific lengths
    for n_fixed in [400, 500, 600, 700, 900, 1000, 1200]:
        if 300 <= n_fixed <= 2000 and n_fixed not in n_options:
            n_options.append(n_fixed)

    n_options = sorted([n for n in n_options if abs(n - n_current) > 30 or n == n_current])
    n_options = n_options[:6]  # Limit number of trials

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        # LP polish
        alloc_lp = min(remaining * 0.4, 80.0)
        if alloc_lp > 20:
            a_try = lp_polish(a_try, time_budget=alloc_lp)

        # CD
        remaining = deadline - time.monotonic()
        alloc_cd = min(remaining * 0.2, 40.0)
        if alloc_cd > 15:
            a_try = coord_descent(a_try, time_budget=alloc_cd, rng=rng)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} ** IMPROVED")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final Intensive Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP
    if remaining > 40:
        alloc = min(remaining * 0.35, 120.0)
        a_final = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} ** IMPROVED")

    # CD
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 60.0)
        a_final2 = coord_descent(best_seq, time_budget=alloc, rng=rng)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final CD: obj={best_score:.8f} ** IMPROVED")

    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 40.0)
        a_final3 = lp_polish(best_seq, time_budget=alloc)
        obj_final3 = compute_obj(a_final3)
        if obj_final3 < best_score:
            best_score = obj_final3
            best_seq = a_final3.copy()
            print(f"  Final LP2: obj={best_score:.8f} ** IMPROVED")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure all values are valid
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
