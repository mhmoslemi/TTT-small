import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_cross_conv(a, b):
    """Full linear convolution a*b via FFT. Returns array of length len(a)+len(b)-1."""
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:na + nb - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, solver_prefer=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    if solver_prefer:
        solvers = [solver_prefer] + [s for s in solvers if s != solver_prefer]
    
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_min_l2(a):
    """
    Solve QP: min ||conv(a,g)||_2^2 s.t. sum(g)=1, g>=0.
    Returns g_optimal or None.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(g) == 1]
    obj = cp.sum_squares(A @ g)
    prob = cp.Problem(cp.Minimize(obj), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:

        except Exception:

def find_optimal_t(ca, cg, gg):
    """
    Find optimal t in [0,1] that minimizes max_k [(1-t)^2 * ca[k] + 2t(1-t)*cg[k] + t^2*gg[k]].
    Uses golden section search since f(t) is convex (max of convex quadratics).
    """
    # f(t) = max_k [ca[k]*(1-t)^2 + 2*cg[k]*t*(1-t) + gg[k]*t^2]
    #      = max_k [ca[k] - 2*ca[k]*t + ca[k]*t^2 + 2*cg[k]*t - 2*cg[k]*t^2 + gg[k]*t^2]
    #      = max_k [(ca[k] - gg[k] + cg[k]) * ... ]
    
    # Let's just evaluate directly
    def f(t):
        val = (1-t)**2 * ca + 2*t*(1-t) * cg + t**2 * gg
        return float(np.max(val))
    
    # Golden section search
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    a_lo, a_hi = 0.0, 1.0
    c = a_hi - phi * (a_hi - a_lo)
    d = a_lo + phi * (a_hi - a_lo)
    fc = f(c)
    fd = f(d)
    
    for _ in range(60):  # ~60 iterations gives precision ~1e-12
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - phi * (a_hi - a_lo)
            fc = f(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + phi * (a_hi - a_lo)
            fd = f(d)
    
    t_opt = (a_lo + a_hi) / 2
    return t_opt, f(t_opt)


def lp_iteration_with_optimal_t(a, time_budget, verbose=False):
    """
    Main iteration: solve LP, find optimal t, update.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    iter_count = 0
    rng = np.random.default_rng(12345)
    consecutive_fails = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if verbose and iter_count % 3 == 0:
            print(f"    LP-OptT it{iter_count}: obj={best_obj:.8f}, t={elapsed:.0f}s")
        
        # Solve LP
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Compute cross convolutions for optimal t
        ca = compute_conv(a)
        cg = compute_cross_conv(a, g_lp)
        gg = compute_conv(g_lp)
        
        # Find optimal t
        t_opt, f_opt = find_optimal_t(ca, cg, gg)
        
        # Clamp t to [0, 1]
        t_opt = np.clip(t_opt, 0.0, 1.0)
        
        # Update
        a_new = (1 - t_opt) * a + t_opt * g_lp
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        iter_count += 1
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            consecutive_fails = 0
            if verbose:
                print(f"      it{iter_count}: obj={best_obj:.8f}, t={t_opt:.4f}, M_lp={M_lp:.6f}")
        else:
            consecutive_fails += 1
            # Try smaller t
            for t_try in [t_opt * 0.5, t_opt * 0.1, 0.01, 0.005]:
                a_try = (1 - t_try) * a + t_try * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    consecutive_fails = 0
                    break
            else:
                # No improvement, perturb
                if consecutive_fails >= 3:
                    scale = 0.02 * np.sqrt(consecutive_fails)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    consecutive_fails = 0
                if consecutive_fails >= 10:
                    # Bigger jump
                    scale = 0.08
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    consecutive_fails = 0
                if consecutive_fails >= 25:
                    break

    return best_a


def lp_iteration_l2(a, time_budget, verbose=False):
    """
    LP iteration using L2 objective for smoother directions.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    iter_count = 0
    rng = np.random.default_rng(99999)
    
    while time.monotonic() - start < time_budget:
        if verbose and iter_count % 3 == 0:
            print(f"    LP-L2 it{iter_count}: obj={best_obj:.8f}")
        
        g_lp = solve_lp_min_l2(a)
        if g_lp is None:
            break
        
        # Find optimal t
        ca = compute_conv(a)
        cg = compute_cross_conv(a, g_lp)
        gg = compute_conv(g_lp)
        t_opt, _ = find_optimal_t(ca, cg, gg)
        t_opt = np.clip(t_opt, 0.0, 1.0)
        
        a_new = (1 - t_opt) * a + t_opt * g_lp
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        iter_count += 1
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Perturb
            scale = 0.03
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            if iter_count > 30:
                break

    return best_a


def spectral_smooth(a, alpha=0.5):
    """Apply spectral smoothing: blend with smoothed version in frequency domain."""
    n = len(a)
    N = 1
    while N < 2 * n:
        N <<= 1
    A = fft(a, N)
    # Low-pass filter
    freqs = np.fft.fftfreq(N)
    cutoff = 0.3 / n  # relative cutoff
    mask = np.exp(-freqs**2 / (2 * cutoff**2))
    A_smooth = A * mask
    a_smooth = ifft(A_smooth).real[:n]
    a_smooth = np.maximum(a_smooth, 0)
    s = a_smooth.sum()
    if s < 1e-10:
        return a
    a_smooth = a_smooth / s
    return (1 - alpha) * a + alpha * a_smooth


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    from scipy.interpolate import CubicSpline
    try:
        cs = CubicSpline(x_old, a, bc_type='not-a-knot')
        a_new = cs(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Gaussian mixture
        n_comp = rng.integers(2, 5)
        centers = rng.uniform(0.1, 0.9, n_comp) * n
        widths = rng.uniform(n/10, n/4, n_comp)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-0.999, 0.999, n)
        a = 1.0 / (np.sqrt(1 - x**2) + 1e-8)
        a = a / a.sum()
    elif mode == 4:
        # Low-frequency sinusoidal
        freqs = rng.integers(1, 4, size=rng.integers(2, 5))
        phases = rng.uniform(0, 2 * np.pi, size=len(freqs))
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.arange(1, n + 1) / n
        a = x ** rng.uniform(-1.5, 1.5)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def nesterov_smoothed_descent(a, time_budget, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        if it % 5000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Gradient of smoothed max
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(y, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.7 * lr + 0.3 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)

        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a
