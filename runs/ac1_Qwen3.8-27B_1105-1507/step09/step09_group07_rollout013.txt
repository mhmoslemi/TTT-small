"""
Optimized step function search for minimizing peak autocorrelation.
Uses two-sided LP alternating, smoothed gradient descent, spectral shaping,
adaptive n-sweep, and diverse restarts to break below 1.5027.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        row_start = max(0, j)
        row_end = min(m, j + n)
        col_idx = np.arange(row_start, row_end)
        A[col_idx, j] = a[col_idx - j]
    return A


def solve_lp_min_m(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, g >= 0, sum(g)=1.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    m = 2 * n - 1

    # LP: variables are [g_0, ..., g_{n-1}, M]
    # Minimize M (last variable)
    # Constraints: A @ g - M <= 0 for all rows, sum(g) = 1, g >= 0
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M

    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0  # A@g - M <= 0
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': 10000})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def solve_lp_max_sum(a, M_fixed):
    """
    Solve LP: max sum(g) s.t. conv(a,g)[k] <= M_fixed for all k, g >= 0.
    Returns g (unnormalized) or None.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    c = -np.ones(n)  # maximize sum = minimize -sum
    A_ub = A
    b_ub = np.full(2 * n - 1, M_fixed)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'maxiter': 10000})
        if res.status == 0:
            g_val = np.maximum(res.x, 0)
            if np.sum(g_val) > 1e-10:
                return g_val
    except Exception:
        pass

    return None


def two_sided_lp(a):
    """
    Two-sided LP: optimize both a and g via LP.
    Returns optimized a.
    """
    n = len(a)
    a = a / a.sum()

    # Step 1: Given a, find optimal g
    g, M_g = solve_lp_min_m(a)
    if g is None:
        return a

    # Step 2: Given g, find optimal a'
    a_new, M_a = solve_lp_min_m(g)
    if a_new is None:

    return a_new

def spectral_shaping(a, time_budget, verbose=False):
    """
    Iteratively flatten the power spectrum to reduce autocorrelation peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

        if time.monotonic() - start > time_budget:
            break
        for it in range(30):
            if time.monotonic() - start > time_budget:
                break
            A = rfft(a, size)
            P = np.abs(A) ** 2

            # Flatten: reduce dynamic range
            P_flat = (1 - alpha) * P + alpha * np.mean(P)
            P_flat = np.maximum(P_flat, 1e-30)

            # Invert preserving phase
            A_new = np.sqrt(P_flat) * (A / (np.abs(A) + 1e-30))
            a_new = irfft(A_new, size)[:n]
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))

            a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
                if verbose:
                    print(f"          Spectral alpha={alpha:.2f}, it{it}: obj={best_obj:.8f}")
            else:
                # Try smaller alpha
                break

    return best_a


def smoothed_gradient_descent(a, time_budget, verbose=False):
    """
    Smoothed gradient descent using log-sum-exp surrogate.
    Uses multiplicative (mirror) updates for the simplex constraint.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    lr = 2.0 / np.sqrt(n)
    beta = 50.0
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        # Compute convolution
        fa = rfft(a, size)
        P = np.abs(fa) ** 2
        conv_full = irfft(fa * fa, size)[:2 * n - 1]

        # Log-sum-exp smoothed max
        m = conv_full.max()
        shifted = beta * (conv_full - m)
        weights = np.exp(shifted)
        weights /= weights.sum()

        # Gradient of smoothed objective wrt a
        # d/d a[p] of sum_k w_k * (a*a)[k] = 2 * sum_k w_k * a[k-p] (if valid)
        # This is 2 * (weights * flip(a)) convolved appropriately
        # More precisely: grad[p] = 2 * sum_k weights[k] * a[k-p] = 2 * (w * flip(a))[p + n - 1]
        # where * is circular convolution of length size

        flip_a = np.zeros(size)
        flip_a[:n] = a[::-1]
        fa_flip = rfft(flip_a, size)
        fw = rfft(weights, size)
        grad = 2.0 * irfft(fw * fa_flip, size)[:n]

        # Mirror descent (exponential family)
        log_a = np.log(a)
        log_a_new = log_a - lr * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        it += 1

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
            lr = max(1e-6, lr * 0.999)
            beta = min(5000.0, beta * 1.01)
        else:
            no_improve += 1
            lr = max(1e-6, lr * 0.95)
            if no_improve > 50:
                # Restart with perturbation
                rng = np.random.default_rng(it * 7 + n)
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                no_improve = 0
                lr = 2.0 / np.sqrt(n)

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"          Smooth it{it}: obj={best_obj:.8f}, beta={beta:.0f}, "
                  f"lr={lr:.5f}, elapsed={elapsed:.1f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def targeted_moves(a, time_budget, verbose=False):
    """

    """

    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    iteration = 0

    while time.monotonic() - start < time_budget:
        iteration += 1
        conv = compute_conv(a)
        peak_val = float(np.max(conv))
        peak_lags = np.argsort(conv)[::-1][:5]

        # Find positions with high "activity" (contributing to peaks)
        activity = np.zeros(n)
        for k in peak_lags:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    activity[i] += a[i] * a[j]

        # Find "quiet" positions
        quiet_thresh = np.percentile(activity, 20)
        quiet_positions = np.where(activity < quiet_thresh)[0]
        if len(quiet_positions) == 0:
            quiet_positions = np.argsort(activity)[:10]

        # Active positions (high activity)
        active_positions = np.argsort(activity)[::-1][:20]

        improved = False
        for src in active_positions[:10]:
            if time.monotonic() - start > time_budget:
                break
            if a[src] < 1e-10:
                continue
            for target in quiet_positions[:10]:
                if target == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    delta = frac * a[src]
                    if delta <= 0 or delta >= 0.9 * a[src]:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    obj_new = compute_obj(a)
                    if obj_new < best_obj - 1e-12:
                        best_obj = obj_new
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break
                if improved:
                    break
            if verbose and iteration % 100 == 0:
                print(f"          Targeted it{iteration}: obj={best_obj:.8f}")

        if not improved and iteration > 20:
            # Perturb and try again
            rng = np.random.default_rng(iteration * 31 + n)
            a = best_a * np.exp(rng.normal(0, 0.03, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_a = a.copy()

    return best_a

        print(f"Default init: obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"\nInitial: n={n_current}, obj={best_score:.8f}, target=1.5027")

    # === Phase 1: Quick LP polish on parent (10% budget) ===
    print(f"\n=== Phase 1: Quick LP Polish ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.12, 120.0)
    if p1_budget > 30:
        a_p1 = lp_alternate(best_seq, time_budget=p1_budget, verbose=True)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Phase 1 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 done: obj={obj_p1:.8f}")

    # === Phase 2: Smoothed gradient descent (10% budget) ===
    print(f"\n=== Phase 2: Smoothed Gradient Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.10, 100.0)
    if p2_budget > 30:
        a_p2 = smoothed_gradient_descent(best_seq, time_budget=p2_budget, verbose=True)
        obj_p2 = compute_obj(a_p2)
        if obj_p2 < best_score:
            best_score = obj_p2
            best_seq = a_p2.copy()
            print(f"  Phase 2 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 done: obj={obj_p2:.8f}")

    # === Phase 3: n-sweep (20% budget) ===
    print(f"\n=== Phase 3: n-Sweep ===")
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.20

    n_options = sorted(set([
        n_current - 50, n_current - 30, n_current - 10,
        n_current,
        n_current + 10, n_current + 30, n_current + 50, n_current + 100,
        700, 720, 750, 768, 780, 800, 820, 840, 850, 860, 880, 900, 920, 960, 1000,
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1200]

    per_n_budget = p3_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(per_n_budget, remaining * 0.12)
        if alloc < 20:
            alloc = 20

        a_try = resample_sequence(best_seq, n_try)
        a_opt = lp_alternate(a_try, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)
        marker = " **" if obj_opt < best_score else ""
        print(f"  n={n_try}: obj={obj_opt:.8f}{marker}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try

    # === Phase 4: Intensive LP with perturbations (25% budget) ===
    print(f"\n=== Phase 4: Intensive LP (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.25, 400.0)

    n_rounds = 6
    per_round = p4_budget / n_rounds
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        alloc = min(per_round, remaining * 0.2)
        if alloc < 30:
            break

        if rnd == 0:
            a_start = best_seq.copy()
        elif rnd == 1:
            # Blend with spectral-shaped version
            a_spec = spectral_shaping(best_seq, time_budget=min(15, alloc/4))
            a_start = 0.5 * best_seq + 0.5 * a_spec
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        else:
            rng_r = np.random.default_rng(rnd * 7919 + n_current + seed)
            scale = 0.03 + 0.03 * (rnd - 1)
            a_start = best_seq * np.exp(rng_r.normal(0, scale, n_current))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()

        a_lp = lp_alternate(a_start, time_budget=alloc, verbose=(rnd == 0))
        obj_lp = compute_obj(a_lp)
        marker = " **" if obj_lp < best_score else ""
        print(f"  LP round {rnd}: obj={obj_lp:.8f}{marker}")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()

    # === Phase 5: Spectral + Targeted (15% budget) ===
    print(f"\n=== Phase 5: Spectral + Targeted ===")
    remaining = deadline - time.monotonic()
    p5_budget = remaining * 0.15

    alloc_s = min(p5_budget * 0.4, 60.0)
    if alloc_s > 15:
        a_spec = spectral_shaping(best_seq, time_budget=alloc_s, verbose=True)
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Spectral: obj={best_score:.8f} **")
        else:
            print(f"  Spectral: obj={obj_spec:.8f}")

    remaining = deadline - time.monotonic()
    alloc_t = min(remaining * 0.5, 60.0)
    if alloc_t > 15:
        a_targ = targeted_moves(best_seq, time_budget=alloc_t, verbose=True)
        obj_targ = compute_obj(a_targ)
        if obj_targ < best_score:
            best_score = obj_targ
            best_seq = a_targ.copy()
            print(f"  Targeted: obj={best_score:.8f} **")
        else:
            print(f"  Targeted: obj={obj_targ:.8f}")

    # === Phase 6: Diverse restarts (10% budget) ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p6_budget = remaining * 0.10
    n_use = len(best_seq)
    n_restarts = 5
    per_restart = p6_budget / max(1, n_restarts)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(per_restart, remaining * 0.25)
        if alloc < 20:
            break

        mode = (r * 3) % 10
        a_r = diverse_init(n_use, rng, mode=mode)
        # Blend with best
        blend = 0.4 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        # Quick gradient + LP
        half = alloc / 2
        if half > 10:
            a_r = smoothed_gradient_descent(a_r, time_budget=half, verbose=False)
        remaining = deadline - time.monotonic()
        lp_alloc = min(remaining * 0.12, 40.0)
        if lp_alloc > 15:
            a_r = lp_alternate(a_r, time_budget=lp_alloc, verbose=False)

        obj_r = compute_obj(a_r)
        marker = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}{marker}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()

    # === Phase 7: Final polish (remaining budget) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    alloc = min(remaining * 0.4, 80.0)
    if alloc > 20:
        a_final = lp_alternate(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # Smoothed gradient polish
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.3, 50.0)
    if alloc > 15:
        a_smooth = smoothed_gradient_descent(best_seq, time_budget=alloc, verbose=False)
        obj_smooth = compute_obj(a_smooth)
        if obj_smooth < best_score:
            best_score = obj_smooth
            best_seq = a_smooth.copy()
            print(f"  Final Smooth: obj={best_score:.8f} **")
        else:
            print(f"  Final Smooth: obj={obj_smooth:.8f}")

    # One more LP on the very best
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_last = lp_alternate(best_seq, time_budget=min(remaining * 0.6, 40.0), verbose=False)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"=== Target: 1.5027, Achieved: {best_score:.8f}, Gap: {best_score - 1.5027:.8f} ===")

    return result
