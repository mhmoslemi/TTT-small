def solve_lp(a):
    """

    Returns (g_normalized, M_opt) or (None, None).
    """

    a_norm = a / np.sum(a)
    A = build_conv_matrix_a(a_norm)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.PDLP]:
        try:
            G = cp.Variable(n, nonneg=True)
            M = cp.Variable()

            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if G.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(G.value, 0)
                s = float(np.sum(g_val))

                    return g_val / s, float(M.value)

        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

    except Exception:
        pass

    return None, None


def solve_qp_refine(a, n_active=30, slack=0.0):
    """
    QP refinement: minimize ||x - a||^2 s.t. linearized conv constraints at active lags.
    Returns refined sequence or None.
    """
    n = len(a)
    a = a / a.sum()
    conv = compute_conv(a)
    M = float(np.max(conv))

    active = np.argsort(conv)[::-1][:n_active]

    A_qp = np.zeros((n_active, n))
    b_qp = np.zeros(n_active)
    for idx, k in enumerate(active):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            indices = np.arange(i_start, i_end + 1)
            A_qp[idx, indices] = a[k - indices]
        b_qp[idx] = (M + conv[k]) / 2.0 + slack

    try:
        x = cp.Variable(n)
        diff = x - a
        obj = cp.sum_squares(diff)
        constraints = [A_qp @ x <= b_qp, cp.sum(x) == 1, x >= 0]
        prob = cp.Problem(cp.Minimize(obj), constraints)
        for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if x.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    x_val = np.maximum(x.value, 0)
                    s = float(np.sum(x_val))
                    if s > 1e-10:
                        return x_val / s
            except Exception:
                continue
    except Exception:
        pass
    return None

def gradient_descent_smooth(a, time_budget, verbose=False):
    """FFT-based Adam gradient descent with smoothed max objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = np.zeros(n)
    velocity = np.zeros(n)
    lr = 0.005
    beta1 = 0.9
    beta2 = 0.999
    t_count = 0

    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / max(1, time_budget))
        beta = 5.0 * (200.0 / 5.0) ** frac

        conv = compute_conv(a)
        shifted = beta * (conv - conv.max())
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(conv)) / len(conv)
        else:
            w = w / w_sum

        # Gradient: g[i] = 2 * sum_k w[k] * a[k-i]
        # = 2 * conv(w, a[::-1])[n-1 : n-1+n]
        size = 1
        while size < len(w) + n - 1:
            size <<= 1
        W_fft = rfft(w, size)
        A_rev_fft = rfft(a[::-1], size)
        conv_full = irfft(W_fft * A_rev_fft, size)
        g = 2.0 * conv_full[n - 1:n - 1 + n]

        # Adam update
        t_count += 1
        momentum = beta1 * momentum + (1 - beta1) * g
        velocity = beta2 * velocity + (1 - beta2) * (g * g)
        m_hat = momentum / (1 - beta1 ** t_count)
        v_hat = velocity / (1 - beta2 ** t_count)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = min(0.02, lr * 1.005)
        else:
            lr = max(0.0005, lr * 0.97)

        if verbose and t_count % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"      GD it{t_count}: obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    print(f"\n=== Phase 5: Gradient Descent (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.20, 250.0)
    if p5_budget > 20:
        a_gd = gradient_descent_smooth(best_seq, time_budget=p5_budget, verbose=True)
        obj_gd = compute_obj(a_gd)
        marker = " **" if obj_gd < best_score else ""
        print(f"  GD: obj={obj_gd:.8f}{marker}")
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()

        # QP polish after GD
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_qp2 = qp_refinement_loop(best_seq, time_budget=min(remaining * 0.1, 40.0), verbose=True)
            obj_qp2 = compute_obj(a_qp2)
            if obj_qp2 < best_score:
                best_score = obj_qp2
                best_seq = a_qp2.copy()
                print(f"  QP after GD: obj={best_score:.8f} **")

    # === Phase 6: Diverse restarts (10%) ===
    print("\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p6_budget = remaining * 0.10
    n_use = len(best_seq)
    n_restarts = 3
    per_restart = p6_budget / max(1, n_restarts)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(per_restart, remaining * 0.2)
        if alloc < 15:
            break

        mode = (r * 3 + 2) % 8
        a_r = diverse_init(n_use, rng, mode=mode)
        blend = 0.4 + 0.15 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        # Quick LP + QP
        lp_alloc = alloc * 0.6
        qp_alloc = alloc * 0.4
        a_r = lp_alternate(a_r, time_budget=lp_alloc, verbose=False)
        a_r = qp_refinement_loop(a_r, time_budget=qp_alloc, verbose=False)

        obj_r = compute_obj(a_r)
        marker = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}{marker}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()

    # === Phase 7: Final polish (5%) ===
    print("\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 20:
        a_final = lp_alternate(best_seq, time_budget=min(remaining * 0.4, 40.0), verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_qp_final = qp_refinement_loop(best_seq, time_budget=min(remaining * 0.4, 30.0), verbose=True)
        obj_qp_f = compute_obj(a_qp_final)
        if obj_qp_f < best_score:
            best_score = obj_qp_f
            best_seq = a_qp_final.copy()
            print(f"  Final QP: obj={best_score:.8f} **")

    # One last quick GD pass
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_gd_f = gradient_descent_smooth(best_seq, time_budget=min(remaining * 0.3, 25.0), verbose=False)
        obj_gd_f = compute_obj(a_gd_f)
        if obj_gd_f < best_score:
            best_score = obj_gd_f
            best_seq = a_gd_f.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"=== Target: 1.5027, Achieved: {best_score:.8f}, Gap: {best_score - 1.5027:.8f} ===")

    return result
