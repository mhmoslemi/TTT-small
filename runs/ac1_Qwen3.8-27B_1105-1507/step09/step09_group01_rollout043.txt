"""
Hybrid nonconvex search for nonnegative step-function heights minimizing
2 * n * max(conv(a, a)) / (sum(a))^2.

Main components:
1. A softmax + L-BFGS-B polish on the entropy simplex, optimizing an annealed
   log-sum-exp surrogate of the peak convolution term.
2. A mirror-Descent / exponentiated-gradient step using the exact active peaks.
3. Spectral square-root / flattening moves: blend the current convolution with
   a flat target, take a causally aligned spectral square root, clip to the
   simplex, and re-polish.
4. A limited LP polishing phase (minimize max(conv(a, g)) given a) for moderate
   lengths.
5. Multi-length exploration and perturbative restarts.
"""

import numpy as np
import time
import math

try:
    from scipy.fft import rfft, irfft, fft, ifft
except Exception:
    from numpy.fft import rfft, irfft, fft, ifft

try:
    from scipy import optimize
except Exception:
    optimize = None

    import cvxpy as cp
except Exception:
    cp = None

_OPT_N = 0
_OPT_TAU = 1.0
MAX_FINAL_N = 8000


def next_power_of_two(x):
    if x <= 1:
        return 1
    return 1 << (int(x) - 1).bit_length()


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.reshape(-1)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    n = a.size
    if s <= 1e-300 or n == 0:
        if n == 0:
            return np.array([1.0], dtype=np.float64)
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    m = 2 * n - 1
    size = next_power_of_two(m)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:m]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a = a / s
    c = compute_conv(a)
    c = np.maximum(c, 0.0)
    if c.size == 0:
        return np.inf
    return float(2.0 * n * np.max(c))


def resample_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        new_n = max(1, int(new_n))
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    if n == int(new_n):
        return a.copy()
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s > 0.0:
        a = a / s
    if new_n <= 1:
        return np.array([float(np.sum(a))], dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, int(new_n))
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def softmax_values(z):
    z = np.asarray(z, dtype=np.float64)
    n = z.size
    if n == 0:
        return np.array([], dtype=np.float64)
    z = np.nan_to_num(z, nan=0.0, posinf=700.0, neginf=-700.0)
    z = z - float(np.max(z))
    ez = np.exp(z)
    s = float(np.sum(ez))
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return ez / s


def weighted_shift(p, q, n):
    # Returns r[i] = sum_k q[k] * p[k-i] for i=0..n-1.
    m = len(q)
    L = next_power_of_two(max(3 * n, m + n + 1))
    Q = np.zeros(L, dtype=np.float64)
    P = np.zeros(L, dtype=np.float64)
    Q[:m] = q
    P[:n] = p
    FQ = fft(Q, L)
    FP = fft(P, L)
    r = ifft(FQ * np.conj(FP), L).real
    return r[:n]


def smooth_inf_obj_and_grad(z):
    n = int(_OPT_N)
    tau = float(_OPT_TAU)
    if n <= 0:
        z0 = np.asarray(z, dtype=np.float64)
        return 0.0, np.zeros_like(z0)
    p = softmax_values(z)
    c = compute_conv(p)
    c = np.maximum(c, 0.0)
    maxc = float(np.max(c))
    scaled = tau * (c - maxc)
    expv = np.exp(np.maximum(scaled, -745.0))
    total = float(np.sum(expv))
    if total <= 0.0:
        expv = np.zeros_like(c)
        expv[int(np.argmax(c))] = 1.0
        total = 1.0
    q = expv / total
    logsum = float(np.log(total))
    soft = maxc + logsum / tau
    f = 2.0 * float(n) * soft

    r = weighted_shift(p, q, n)
    grad_p = 4.0 * float(n) * r
    scalar = float(np.dot(p, grad_p))
    grad_z = p * (grad_p - scalar)
    grad_z = np.nan_to_num(grad_z, nan=0.0, posinf=0.0, neginf=0.0)
    return f, grad_z


def lbfgs_polish_from_p(p, tau_values, maxfun_per_tau=80, time_budget=120.0):
    if optimize is None:
        return normalize_nonneg(p)
    p = normalize_nonneg(p)
    n = len(p)
    best_p = p.copy()
    best_score = compute_obj(best_p)
    if not np.isfinite(best_score):
        return best_p

    z = np.log(np.maximum(p, 1e-300))
    start = time.monotonic()

    for tau in tau_values:
        if time.monotonic() - start > time_budget:
            break
        tau_use = float(max(1.0, tau))
        global _OPT_N, _OPT_TAU
        _OPT_N = n
        _OPT_TAU = tau_use

        if time_budget < 25.0:
            maxfun_use = min(maxfun_per_tau, 25)
        elif time_budget < 70.0:
            maxfun_use = min(maxfun_per_tau, 55)
        else:
            maxfun_use = maxfun_per_tau
        maxfun_use = max(10, int(maxfun_use))

        try:
            x0 = z
            if x0.size != n or not np.all(np.isfinite(x0)):
                x0 = np.log(np.maximum(best_p, 1e-300))
            res = optimize.minimize(
                fun=smooth_inf_obj_and_grad,
                x0=x0,
                jac=True,
                method="L-BFGS-B",
                options={
                    "maxiter": 120,
                    "maxfun": maxfun_use,
                    "ftol": 1e-14,
                    "gtol": 1e-8,
                    "maxls": 20,
                    "m": 20,
                },
            )
            if hasattr(res, "x") and res.x is not None:
                p_try = softmax_values(res.x)
                score_try = compute_obj(p_try)
                if np.isfinite(score_try) and score_try < best_score - 1e-13:
                    best_score = score_try
                    best_p = p_try.copy()
                    z = np.log(np.maximum(best_p, 1e-300))
                else:
                    z = np.log(np.maximum(best_p, 1e-300))
        except Exception:
            pass

    return best_p


def prune_small(p, frac=1e-9):
    p = normalize_nonneg(p)
    mx = float(np.max(p))
    if mx <= 0.0:
        return p
    p = p.copy()
    p[p < frac * mx] = 0.0
    return normalize_nonneg(p)


def mirror_refine(p, time_budget=30.0, K=5):
    p = normalize_nonneg(p)
    n = len(p)
    best_p = p.copy()
    best_score = compute_obj(best_p)
    if not np.isfinite(best_score):
        return best_p

    logits = np.log(np.maximum(p, 1e-300))
    step = 1.0 / math.sqrt(float(n))
    stall = 0
    start = time.monotonic()
    deadline = start + float(time_budget)

    while time.monotonic() < deadline:
        c = compute_conv(best_p)
        c = np.maximum(c, 0.0)
        maxc = float(np.max(c))
        if maxc <= 1e-300:
            break

        lc = len(c)
        if K < lc:
            top = np.argpartition(c, -K)[-K:]
        else:
            top = np.arange(lc)
        vals = c[top]
        maxv = float(np.max(vals))
        if maxv <= 1e-300:
            break
        w = np.exp(5.0 * (vals - maxv) / maxv)
        wsum = float(np.sum(w))
        if wsum <= 0.0:
            w = np.ones_like(vals) / float(len(vals))
        else:
            w = w / wsum

        q = np.zeros(lc, dtype=np.float64)
        q[top] = w
        r = weighted_shift(best_p, q, n)
        grad = 2.0 * float(n) * r
        gnorm = float(np.linalg.norm(grad))
        if gnorm > 20.0:
            grad *= 20.0 / gnorm
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

        logits = logits - step * grad
        lmax = float(np.max(logits))
        lmin = float(np.min(logits))
        rngz = lmax - lmin
        if rngz > 50.0:
            logits = lmin + (logits - lmin) * (50.0 / rngz)
        logits = logits - float(np.mean(logits))

        p_new = softmax_values(logits)
        score_new = compute_obj(p_new)
        if np.isfinite(score_new) and score_new < best_score - 1e-12:
            best_score = score_new
            best_p = p_new.copy()
            stall = 0
            step = min(step * 1.03, 3.0 / math.sqrt(float(n)))
        else:
            stall += 1
            step *= 0.93
            if stall > 100 or step < 1e-6 / math.sqrt(float(n)):
                break

    return best_p


def blend_objective(t, base, cand):
    base = np.asarray(base, dtype=np.float64)
    cand = np.asarray(cand, dtype=np.float64)
    a = (1.0 - float(t)) * base + float(t) * cand
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-300:
        return 1e12
    a = a / s
    return compute_obj(a)


def golden_section_search_params(func, lo, hi, tol, max_iter, *args):
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = func(c, *args)
    fd = func(d, *args)
    for _ in range(int(max_iter)):
        if abs(hi - lo) < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = func(c, *args)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = func(d, *args)
    return (lo + hi) / 2.0


def solve_lp_minpeak(a, M_target=None):
    if cp is None:
        return None, None
    try:
        n = len(a)
        if n == 0:
            return None, None
        a = normalize_nonneg(a)
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            r = min(n, 2 * n - 1 - j)
            if r > 0:
                A[j:j + r, j] = a[:r]

        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        if M_target is not None:
            constraints.append(M <= float(M_target))

        prob = cp.Problem(cp.Minimize(M), constraints)
        solver_list = []
        for name in ("MOSEK", "CLARABEL", "SCS", "ECOS", "GLPK", "CBC", "GLOP"):
            solver_obj = getattr(cp, name, None)
            if solver_obj is not None:
                solver_list.append(solver_obj)

        for solver in solver_list:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ("optimal", "optimal_inaccurate"):
                    gv = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
                    sgv = float(np.sum(gv))
                    if sgv > 1e-10:
                        return gv / sgv, float(M.value)
                if prob.status == "infeasible":
                    return None, None
            except Exception:
                continue
        return None, None
    except Exception:
        return None, None


def lp_polish(a, time_budget=100.0, seed=42):
    if cp is None:
        return normalize_nonneg(a)
    a = normalize_nonneg(a)
    n = len(a)
    best_p = a.copy()
    best_score = compute_obj(best_p)
    if not np.isfinite(best_score):
        return best_p

    start = time.monotonic()
    deadline = start + float(time_budget)
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        g_lp, _ = solve_lp_minpeak(best_p)
        if g_lp is None:
            break
        try:
            t_opt = golden_section_search_params(
                blend_objective,
                0.02,
                1.0,
                0.02,
                30,
                best_p,
                g_lp,
            )
        except Exception:
            t_opt = 0.25

        a_try = (1.0 - float(t_opt)) * best_p + float(t_opt) * g_lp
        a_try = normalize_nonneg(a_try)
        score_try = compute_obj(a_try)

        improved = False
        if np.isfinite(score_try) and score_try < best_score - 1e-13:
            best_score = score_try
            best_p = a_try.copy()
            no_improve = 0
            improved = True
        else:
            no_improve += 1
            if no_improve == 2:
                t_small = 0.08
                a_try2 = normalize_nonneg((1.0 - t_small) * best_p + t_small * g_lp)
                score_try2 = compute_obj(a_try2)
                if np.isfinite(score_try2) and score_try2 < best_score - 1e-13:
                    best_score = score_try2
                    best_p = a_try2.copy()
                    no_improve = 0
                    improved = True

        if not improved:
            if no_improve >= 5:
                if it > 50:
                    break
                scale = 0.004 + 0.002 * min(no_improve, 12)
                best_p = normalize_nonneg(best_p * np.exp(rng.normal(0.0, scale, n)))
                no_improve = 0

        it += 1

    return best_p


def spectral_from_conv(c, n, try_alt=True):
    c = np.asarray(c, dtype=np.float64)
    c = np.nan_to_num(c, nan=0.0, posinf=0.0, neginf=0.0)
    c = np.maximum(c, 0.0)
    m = c.size
    if m <= 0 or n <= 0:
        return np.ones(max(1, n), dtype=np.float64) / float(max(1, n))
    L = next_power_of_two(max(4 * n, 2 * m + 1))
    cpad = np.zeros(L, dtype=np.float64)
    cpad[:m] = c
    C = fft(cpad, L)
    max_abs = max(1.0, float(np.max(np.abs(C))))
    mask = np.abs(C) > (1e-14 * max_abs)

    half = L // 2
    C_half = C[:half + 1]
    abs_half = np.abs(C_half)
    phase = np.unwrap(np.angle(C_half), discont=np.pi)
    phase[~mask[:half + 1]] = 0.0
    P_half = np.sqrt(np.maximum(abs_half, 0.0)) * np.exp(0.5j * phase)

    P = np.zeros(L, dtype=np.complex128)
    P[:half + 1] = P_half
    if L % 2 == 0:
        P[half] = np.abs(P_half[half])
        if half > 1:
            P[half + 1:] = np.conj(P_half[1:half][::-1])
    else:
        if half > 0:
            P[half + 1:] = np.conj(P_half[1:half + 1][::-1])

    p_full = ifft(P, L).real
    p_full = np.nan_to_num(p_full, nan=0.0, posinf=0.0, neginf=0.0)

    ext = np.concatenate((p_full, p_full))
    cum = np.concatenate(([0.0], np.cumsum(ext)))
    pos_sums = cum[n:n + L] - cum[:L]
    o1 = int(np.argmax(pos_sums))
    q1 = normalize_nonneg(ext[o1:o1 + n])

    if try_alt:
        absp = np.abs(p_full)
        exta = np.concatenate((absp, absp))
        Cuma = np.concatenate(([0.0], np.cumsum(exta)))
        abs_sums = Cuma[n:n + L] - Cuma[:L]
        o2 = int(np.argmax(abs_sums))
        q2 = normalize_nonneg(exta[o2:o2 + n])
        s1 = compute_obj(q1)
        s2 = compute_obj(q2)
        if np.isfinite(s2) and (not np.isfinite(s1) or s2 < s1):
            q1 = q2

    return q1


def spectral_flat_candidate(n):
    m = 2 * int(n) - 1
    flat = np.full(m, 1.0 / float(m), dtype=np.float64)
    return spectral_from_conv(flat, int(n), try_alt=True)


def diverse_init(n, rng, mode):
    x = np.linspace(0.0, 1.0, int(n))
    mode = int(mode) % 10
    if mode == 0:
        a = np.ones(n, dtype=np.float64)
    elif mode == 1:
        a = rng.uniform(0.0, 1.0, n).astype(np.float64)
    elif mode == 2:
        c1 = rng.uniform(0.2, 0.5)
        c2 = rng.uniform(0.5, 0.8)
        w1 = max(0.02, rng.uniform(0.03, 0.12))
        w2 = max(0.02, rng.uniform(0.03, 0.12))
        a = (
            np.exp(-((x - c1) ** 2) / (2.0 * w1 * w1))
            + np.exp(-((x - c2) ** 2) / (2.0 * w2 * w2))
        )
    elif mode == 3:
        a = 1.0 / np.sqrt(x * (1.0 - x) + 1e-3) + 0.1
    elif mode == 4:
        f1 = int(rng.integers(1, 5))
        f2 = int(rng.integers(2, 8))
        p1 = float(rng.uniform(0.0, 2.0 * np.pi))
        p2 = float(rng.uniform(0.0, 2.0 * np.pi))
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * f1 * x + p1) + 0.35 * np.cos(
            2.0 * np.pi * f2 * x + p2
        )
    elif mode == 5:
        alpha = float(rng.uniform(0.2, 4.0))
        beta = float(rng.uniform(0.2, 4.0))
        a = np.power(np.maximum(x, 1e-3), alpha) + np.power(
            np.maximum(1.0 - x, 1e-3), beta
        )
    elif mode == 6:
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * x) + 0.3 * np.cos(4.0 * np.pi * x)
    elif mode == 7:
        sig = float(rng.uniform(0.02, 0.08))
        a = (
            1.0
            + 2.5 * np.exp(-((x - 0.0) ** 2) / (2.0 * sig * sig))
            + 2.5 * np.exp(-((x - 1.0) ** 2) / (2.0 * sig * sig))
            + 0.5 * np.exp(-((x - 0.5) ** 2) / (2.0 * (3.0 * sig) ** 2))
        )
    elif mode == 8:
        centers = sorted(rng.uniform(0.0, 1.0, size=3).tolist())
        widths = np.maximum(0.015, rng.uniform(0.02, 0.12, size=3))
        a = np.zeros(n, dtype=np.float64)
        for cc, ww in zip(centers, widths):
            a += np.exp(-((x - cc) ** 2) / (2.0 * ww * ww))
    else:
        a = np.ones(n, dtype=np.float64)
    return normalize_nonneg(a)


def consider_candidate(candidate, best_seq, best_score, tag="", verbose=False):
    if candidate is None:
        return best_seq, best_score
    cand = normalize_nonneg(candidate)
    if cand.size == 0:
        return best_seq, best_score
    score = compute_obj(cand)
    if not np.isfinite(score):
        return best_seq, best_score
    if score < best_score:
        if verbose:
            print(f"  update {tag}: n={cand.size}, score={score:.10f}")
        return cand, score
    return best_seq, best_score


def polish_length(p_init, n, time_budget, allow_lp=False, seed=0, verbose=False):
    p_init = normalize_nonneg(p_init)
    n = int(n)
    if p_init.size != n:
        p_init = resample_sequence(p_init, n)
    p_best = normalize_nonneg(p_init)
    best_score = compute_obj(p_best)
    start = time.monotonic()
    deadline = start + float(time_budget)

    # 1. L-BFGS-B softmax polish.
    rem = deadline - time.monotonic()
    if rem > 25.0:
        budget = min(rem * 0.62, 170.0)
        tau_values = [5 * n, 15 * n, 40 * n]
        if budget > 100.0:
            tau_values.append(100 * n)
        maxfun = 120 if n < 1000 else 75 if n < 3000 else 45
        p_l = lbfgs_polish_from_p(p_best, tau_values, maxfun_per_tau=maxfun, time_budget=budget)
        p_best, best_score = consider_candidate(p_l, p_best, best_score, tag="lbfgs", verbose=verbose)
        p_pr = prune_small(p_best, frac=1e-9)
        p_best, best_score = consider_candidate(p_pr, p_best, best_score, tag="prune0", verbose=verbose)

    # 2. LP polish for modest lengths.
    rem = deadline - time.monotonic()
    if allow_lp and cp is not None and n <= 1000 and rem > 30.0:
        budget = min(rem * 0.45, 140.0 if n < 800 else 100.0)
        p_lp = lp_polish(p_best, budget, seed=seed)
        p_best, best_score = consider_candidate(p_lp, p_best, best_score, tag="lp", verbose=verbose)
        p_pr = prune_small(p_best, frac=1e-9)
        p_best, best_score = consider_candidate(p_pr, p_best, best_score, tag="prune1", verbose=verbose)

    # 3. Exact peak mirror/exponentiated gradient.
    rem = deadline - time.monotonic()
    if rem > 20.0:
        budget = min(rem * 0.35, 45.0)
        p_m = mirror_refine(p_best, budget, K=5)
        p_best, best_score = consider_candidate(p_m, p_best, best_score, tag="mirror", verbose=verbose)

    # 4. Spectral flattening moves.
    rem = deadline - time.monotonic()
    if rem > 30.0 and n <= 6000:
        for alpha in (0.35, 1.0):
            rem = deadline - time.monotonic()
            if rem < 25.0:
                break
            c = compute_conv(p_best)
            flat = np.full(2 * n - 1, 1.0 / float(2 * n - 1), dtype=np.float64)
            h = (1.0 - float(alpha)) * c + float(alpha) * flat
            q = spectral_from_conv(h, n, try_alt=True)
            p_best, best_score = consider_candidate(q, p_best, best_score, tag=f"spectral={alpha}", verbose=verbose)
            if np.isfinite(best_score):
                rem = deadline - time.monotonic()
                if rem > 20.0:
                    p_q = lbfgs_polish_from_p(
                        p_best,
                        [15 * n, 50 * n],
                        maxfun_per_tau=60,
                        time_budget=min(rem * 0.35, 35.0),
                    )
                    p_best, best_score = consider_candidate(p_q, p_best, best_score, tag=f"spectpol={alpha}", verbose=verbose)

    p_pr = prune_small(p_best, frac=1e-10)
    p_best, best_score = consider_candidate(p_pr, p_best, best_score, tag="prune_final", verbose=verbose)
    return p_best


def propose_candidate(seed=42, budget_s=1500):
    global _OPT_N, _OPT_TAU
    _OPT_N = 0
    _OPT_TAU = 1.0

    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(int(seed))

    best_seq = None
    best_score = np.inf
    parent_n = 800

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.nan_to_num(parent_arr, nan=0.0, posinf=0.0, neginf=0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.size > 0 and float(np.sum(parent_arr)) > 1e-8:
            if parent_arr.size > MAX_FINAL_N:
                parent_arr = resample_sequence(parent_arr, MAX_FINAL_N)
            best_seq = normalize_nonneg(parent_arr)
            best_score = compute_obj(best_seq)
            parent_n = int(best_seq.size)
            print(f"Loaded parent: n={parent_n}, score={best_score:.10f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(parent_n, dtype=np.float64) / float(parent_n)
        best_score = compute_obj(best_seq)
        print(f"No parent; uniform init: n={parent_n}, score={best_score:.10f}")

    # Main polish at the parent length.
    rem = deadline - time.monotonic()
    if rem > 80.0:
        budget = min(rem * 0.28, 320.0)
        p_pol = polish_length(best_seq, len(best_seq), budget, allow_lp=True, seed=seed, verbose=True)
        best_seq, best_score = consider_candidate(p_pol, best_seq, best_score, tag="parent", verbose=False)
        print(f"After parent polish: n={len(best_seq)}, score={best_score:.10f}")

    # A few diverse initializations at the current length.
    for mode in (3, 7, 4):
        rem = deadline - time.monotonic()
        if rem < 140.0:
            break
        n_cur = len(best_seq)
        budget = min(70.0, rem * 0.12)
        p_div = diverse_init(n_cur, rng, mode)
        p_pol = polish_length(p_div, n_cur, budget, allow_lp=False, seed=seed + 100 + mode, verbose=False)
        best_seq, best_score = consider_candidate(p_pol, best_seq, best_score, tag=f"diverse={mode}", verbose=False)

    # Spectral flat starting point at a medium/large length later.

    # Multi-length exploration from the best sequence.
    current_n = len(best_seq)
    lengths = []
    for mult in (0.75, 1.25, 1.6, 2.2, 3.0, 4.5, 7.0):
        lengths.append(int(float(current_n) * float(mult)))
    lengths += [800, 1200, 2000, 3000, 5000, 8000]
    lengths = sorted(set(max(300, min(MAX_FINAL_N, int(v))) for v in lengths if int(v) >= 300))

    tried_lengths = {int(current_n)}
    for n in lengths:
        rem = deadline - time.monotonic()
        if rem < 140.0:
            break
        if n in tried_lengths:
            continue
        if any(abs(n - t) <= max(40, int(0.07 * t)) for t in tried_lengths):
            continue
        tried_lengths.add(n)

        if n < 1200:
            base_budget = 90.0
        elif n < 2500:
            base_budget = 110.0
        elif n < 5000:
            base_budget = 135.0
        else:
            base_budget = 160.0
        budget = min(base_budget, rem * 0.28)
        if budget < 45.0:
            continue

        p_init = resample_sequence(best_seq, n)
        score_init = compute_obj(p_init)

        # Sometimes a spectral-flat seed is better than direct resampling.
        rem2 = deadline - time.monotonic()
        if rem2 > budget + 45.0 and n <= MAX_FINAL_N:
            p_spec = spectral_flat_candidate(n)
            score_spec = compute_obj(p_spec)
            if score_spec < score_init:
                p_init = p_spec
                print(f"  spectral-flat seed better at n={n}: {score_spec:.10f}")

        p_pol = polish_length(p_init, n, budget, allow_lp=(n <= 1000), seed=seed + 1000 + n, verbose=False)
        old_n = len(best_seq)
        best_seq, best_score = consider_candidate(p_pol, best_seq, best_score, tag=f"n={n}", verbose=False)
        if len(best_seq) != old_n:
            current_n = len(best_seq)
            print(f"  improved to n={current_n}, score={best_score:.10f}")

    # Optional second multi-length sweep if enough time remains and length improved.
    rem = deadline - time.monotonic()
    if rem > 600.0 and abs(current_n - parent_n) > 50:
        extra_lengths = [int(current_n * 1.4), int(current_n * 1.8), int(current_n * 2.5), 4000, 6000, 8000]
        extra_lengths = sorted(set(max(300, min(MAX_FINAL_N, int(v))) for v in extra_lengths if int(v) >= 300))
        for n in extra_lengths:
            rem = deadline - time.monotonic()
            if rem < 160.0:
                break
            if n in tried_lengths:
                continue
            if any(abs(n - t) <= max(40, int(0.10 * t)) for t in tried_lengths):
                continue
            tried_lengths.add(n)
            budget = min(140.0, rem * 0.22)
            p_init = resample_sequence(best_seq, n)
            p_pol = polish_length(p_init, n, budget, allow_lp=(n <= 1000), seed=seed + 2000 + n, verbose=False)
            best_seq, best_score = consider_candidate(p_pol, best_seq, best_score, tag=f"n2={n}", verbose=False)

    # Perturbative restarts from the current best.
    for r in range(6):
        rem = deadline - time.monotonic()
        if rem < 120.0:
            break
        n_cur = len(best_seq)
        scale = 0.02 + 0.015 * r
        p_pert = best_seq * np.exp(rng.normal(0.0, scale, n_cur))
        budget = min(70.0, rem * 0.18)
        p_pol = polish_length(p_pert, n_cur, budget, allow_lp=False, seed=seed + 5000 + r, verbose=False)
        best_seq, best_score = consider_candidate(p_pol, best_seq, best_score, tag=f"restart={r}", verbose=False)

    # Final polish.
    rem = deadline - time.monotonic()
    if rem > 60.0:
        n_cur = len(best_seq)
        budget = min(rem * 0.55, 140.0)
        p_pol = polish_length(best_seq, n_cur, budget, allow_lp=(n_cur <= 1000), seed=seed + 9999, verbose=False)
        best_seq, best_score = consider_candidate(p_pol, best_seq, best_score, tag="final", verbose=False)

    p_pr = prune_small(best_seq, frac=1e-10)
    best_seq, best_score = consider_candidate(p_pr, best_seq, best_score, tag="final_prune", verbose=False)

    if len(best_seq) > MAX_FINAL_N:
        best_seq = resample_sequence(best_seq, MAX_FINAL_N)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if np.isfinite(val):
            best_score = float(val)
    except Exception:
        pass

    if not np.isfinite(best_score) or not result:
        result = [1.0] * 100
        try:
            best_score = float(evaluate_sequence(result))
        except Exception:
            best_score = np.inf

    elapsed = time.monotonic() - start_time
    print(f"FINAL: n={len(result)}, score={best_score:.10f}, elapsed={elapsed:.1f}s")
    return result
