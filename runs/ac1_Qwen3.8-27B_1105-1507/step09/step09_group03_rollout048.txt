import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_gradient(a, tau, N=None):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    if N is None:
        N = 1
        while N < 3 * n:
            N <<= 1
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    # Gradient: g_j = 2 * sum_k p_k * a_{k-j}
    # This is 2 * cross-correlation of a and p
    Pa = fft(p, N)
    A = fft(a, N)
    g_full = ifft(Pa * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv, p

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Combines fast FFT-based smooth gradient descent (Nesterov accelerated) with
targeted LP polishing on the active constraint set. Explores multiple sequence
lengths and uses circular-shift diversity to escape local minima.

import numpy as np

from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = int(rho[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(a, tau):
    """
    Compute gradient of smoothed max autocorrelation.
    f_tau(a) = (1/tau) * log(sum_k exp(tau * q_k(a)))
    Returns (gradient, convolution, peak_weights).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    ms = float(np.max(shifted))
    ev = np.exp(shifted - ms)
    p = ev / float(np.sum(ev))
    # g_j = 2 * sum_k p_k * a_{k-j} = 2 * (p * flip(a))_{n-1+j}
    c = a[::-1]
    N = 1
    while N < 3 * n:
        N <<= 1
    Pc = rfft(p, N)
    C = rfft(c, N)
    conv_result = irfft(Pc * C, N)
    g = 2.0 * conv_result[n - 1:2 * n - 1]
    return g, conv, p


def fast_gd(a_init, time_budget, tau_init=10.0, tau_end=300.0,
            lr_init=None, momentum=0.0, restart_period=3000,
            verbose=False, seed=42):

    Nesterov-accelerated gradient descent on smoothed max autocorrelation.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    if lr_init is None:
        lr_init = 2.0 / np.sqrt(n)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    v = np.zeros(n)
    it = 0
    best_it = 0
    lr = lr_init
    mom = momentum
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget * 0.85, 1e-6))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)
        
        # Nesterov lookahead
        y = a + mom * v
        y = np.maximum(y, 0)

        else:
            y = y / sy

        v = mom * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)

            v = np.zeros(n)

            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1

        lr *= 0.99995
        lr = max(lr, 1e-8)

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0], dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0).astype(np.float64)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def lp_minpeak(a, active_indices=None, time_limit=30.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M (for all k or active), sum(g)=1, g>=0.
    Returns g.
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    
    g = cp.Variable(n, nonneg=True)
    
    if active_indices is not None and len(active_indices) < 2 * n:
        A_sub = A[active_indices, :]
        M = cp.Variable()
        constraints = [A_sub @ g <= M, cp.sum(g) == 1]
    else:
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None
