import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize_scalar
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a, rows=None):
    """Build matrix A (len(rows) x n) such that A[k] @ g = conv(a, g)[rows[k]]."""
    n = len(a)
    if rows is None:
        rows = np.arange(2 * n - 1)
    A = np.zeros((len(rows), n))
    for idx, k in enumerate(rows):
        k = int(k)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # A[idx, j] = a[k - j] for j in [j_start, j_end]
            A[idx, j_start:j_end + 1] = a[k - j_start:k - j_end + 1][::-1]
    return A


def solve_lp_cutting(a, K=50, max_rounds=20):
    """
    Solve LP with cutting planes: min M s.t. conv(a,g)[k] <= M for selected k,
    sum(g)=1, g>=0. Iteratively adds violated constraints.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Start with the top K peaks
    conv = compute_conv(a_norm)
    top_k = np.argsort(conv)[::-1][:K]
    
    active_rows = list(np.sort(top_k))
    g_opt = None
    
    for round_idx in range(max_rounds):
        A = build_conv_matrix(a_norm, rows=active_rows)
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        solved = False
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_opt = np.maximum(g.value, 0)
                    s = float(np.sum(g_opt))
                    if s > 1e-10:
                        g_opt = g_opt / s
                        solved = True
                        break
            except Exception:
                continue
        
        if not solved:
            return None, None

        new_conv = compute_conv(a_norm)
        g_conv = np.convolve(a_norm, g_opt)  # This is conv(a, g)

        if worst not in active_rows:
            active_rows.append(worst)
            active_rows = sorted(active_rows)
    
    return g_opt, float(max(np.convolve(a_norm, g_opt)))

    # Actually: d/d a[i] sum_j a[j]*a[k-j] = a[k-i] (if 0<=k-i<n) from j=i
    #           + sum_j a[j] * d/d a[i] a[k-j] = a[k-i] (if 0<=k-i<n) from j=k-i
    # So total: 2*a[k-i] if 0<=k-i<n, else a[k-i] if only one term is valid.
    # Wait, if k-i is in [0,n-1], both terms contribute: a[k-i] + a[k-i] = 2*a[k-i].
    # If k-i is not in [0,n-1], neither term contributes (or only boundary cases).
    # Actually for the linear convolution, if k-i is in [0, 2n-2] but not in [0, n-1],
    # then a[k-i] doesn't exist, so the gradient is 0.
    
    # So grad[i] = 2 * sum_k p[k] * a[k-i] (where k-i is in [0, n-1])
    # This is 2 * (p * a)[i] in the appropriate sense.
    
    # Using FFT: grad = 2 * ifft(fft(p, N) * conj(fft(a, N)))[:n]
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    grad_full = ifft(P * np.conj(A)).real
    grad = 2.0 * grad_full[:n]
    
    return grad, conv

def structured_perturb(a, rng, mode=0):
    """Apply a structured perturbation to escape local minima."""
    n = len(a)
    if mode == 0:
        # Low-frequency perturbation
        freq = rng.integers(2, max(3, n // 50))
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n)
        pert = 0.05 * np.sin(2 * np.pi * freq * x / n + phase)
        a_new = a * np.exp(pert)
    elif mode == 1:
        # Block perturbation
        block_size = max(1, n // 20)
        n_blocks = n // block_size
        shifts = rng.normal(0, 0.05, n_blocks)
        a_new = a.copy()
        for b in range(n_blocks):
            a_new[b*block_size:(b+1)*block_size] *= np.exp(shifts[b])
    elif mode == 2:
        # Gaussian noise in frequency domain
        A = fft(a, n)
        noise = np.exp(rng.normal(0, 0.1, n) + 1j * rng.normal(0, 0.1, n))
        A_new = A * noise
        a_new = np.real(ifft(A_new))
    else:
        a_new = a * np.exp(rng.normal(0, 0.03, n))
    
    a_new = np.maximum(a_new, 1e-10)
    a_new = a_new / a_new.sum()
    return a_new


def frequency_domain_init(n, rng):
    """
    Initialize a sequence with a flat-ish power spectrum.
    Start from random phases in frequency domain, take magnitude, inverse FFT.
    """
    # Generate random phases
    phases = rng.uniform(0, 2 * np.pi, n)
    # Flat magnitude
    mag = np.ones(n)
    # Construct complex spectrum
    spec = mag * np.exp(1j * phases)
    # Make it conjugate symmetric for real output
    if n % 2 == 0:
        spec[n//2] = abs(spec[n//2])  # Nyquist must be real
        spec[:n//2] = spec[:n//2]
        spec[n//2+1:] = np.conj(spec[:n//2][::-1])
    else:
        spec[(n//2)+1:] = np.conj(spec[1:(n//2+1)][::-1])
    
    a = np.real(ifft(spec))
    a = np.maximum(a, 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    return a


def polynomial_init(n, rng):
    """Initialize with a polynomial-shaped sequence."""
    order = rng.integers(2, 8)
    coeffs = rng.uniform(-1, 1, order + 1)
    x = np.linspace(-1, 1, n)
    a = np.polynomial.polynomial.polyval(x, coeffs)
    a = np.maximum(a, 0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    return a


def sine_init(n, rng):
    """Initialize with a sine-modulated sequence."""

    a = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)

    return a

iterative peak targeting. Key innovations over previous: faster LP via constraint
selection, continuous adaptive blending, frequency-structured restarts to escape
local minima, and better time allocation across phases.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_ag(a, g):
    """Compute conv(a, g) via FFT for linear convolution."""
    n_a = len(a)
    n_g = len(g)
    size = 1
    while size < n_a + n_g - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g, size)
    return irfft(fa * fg, size)[:n_a + n_g - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a, rows):
    """Build matrix A (len(rows) x n) such that A[r] @ g = conv(a, g)[rows[r]]."""
    n = len(a)
    A = np.zeros((len(rows), n))
    for idx, k in enumerate(rows):
        k = int(k)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # A[idx, j] = a[k-j] for j in [j_start, j_end]
            # Values: a[k-j_start], a[k-j_start-1], ..., a[k-j_end]
            # = a[k-j_end : k-j_start+1] reversed
            A[idx, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_cutting(a, K_init=50, max_rounds=15, verbose=False):
    """
    Solve LP with cutting planes: min M s.t. conv(a,g)[k] <= M for selected k,
    sum(g)=1, g>=0. Iteratively adds violated constraints.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    if n < 2:
        return np.array([1.0]), float('inf')
    a_norm = a / np.sum(a)

    # Start with the top K peaks of conv(a,a)
    conv_aa = compute_conv(a_norm)
    top_k = np.argsort(conv_aa)[::-1][:K_init]
    active_rows = sorted(set([int(x) for x in top_k]))

    g_opt = None
    M_opt = float('inf')

    for round_idx in range(max_rounds):
        A = build_conv_matrix(a_norm, rows=active_rows)

        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        solved = False
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_opt = np.maximum(g.value, 0)
                    s = float(np.sum(g_opt))
                    if s > 1e-10:
                        g_opt = g_opt / s
                        M_opt = float(M.value)
                        solved = True
                        break
            except Exception:
                continue

        if not solved:
            return None, None

        # Check for violated constraints
        ag_conv = compute_conv_ag(a_norm, g_opt)
        M_actual = float(np.max(ag_conv))

        if M_actual <= M_opt + 1e-9:
            return g_opt, M_actual

        # Find and add worst violated constraint
        violated = np.where(ag_conv > M_opt + 1e-9)[0]
        if len(violated) > 0:
            # Add the top few violated
            worst_idx = violated[np.argsort(ag_conv[violated])[::-1][:max(1, len(violated)//4)]]
            for w in worst_idx:
                wi = int(w)
                if wi not in active_rows:
                    active_rows.append(wi)
                    active_rows = sorted(active_rows)

    return g_opt, M_opt


def line_search_blend(a, g, best_obj):
    """
    Find optimal blending parameter t via adaptive search.
    Returns (best_a, best_obj, best_t).
    """
    n = len(a)
    best_local_obj = best_obj
    best_local_a = a.copy()
    best_t = 0.0

    # Coarse scan
    ts = np.linspace(0.0, 1.0, 25)
    for t in ts:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_local_obj - 1e-14:
            best_local_obj = obj
            best_local_a = a_try.copy()
            best_t = float(t)

    # Fine scan around best
    if 0 < best_t < 1:
        t_fine = np.linspace(max(0, best_t - 0.04), min(1.0, best_t + 0.04), 17)
        for t in t_fine:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_local_obj - 1e-14:
                best_local_obj = obj
                best_local_a = a_try.copy()
                best_t = float(t)

    return best_local_a, best_local_obj, best_t


def lp_polish(a, time_budget, K=50, verbose=False):
    """
    LP polishing with cutting planes and adaptive line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        # Adaptively increase K
        K_cur = min(K + it * 2, 200)

        g_lp, M_lp = solve_lp_cutting(a, K_init=K_cur, max_rounds=12)
        if g_lp is None:
            break

        new_a, new_obj, best_t = line_search_blend(a, g_lp, best_obj)
        it += 1

        if new_obj < best_obj - 1e-13:
            best_obj = new_obj
            best_a = new_a.copy()
            a = new_a
            stall = 0
            if verbose and it % 2 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}")
        else:
            stall += 1
            if stall >= 3:
                scale = 0.008 + 0.004 * min(stall, 8)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
            if stall > 25:
                break

    return best_a


def smoothed_gradient(a, tau):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*conv)) w.r.t. a.
    Returns (gradient, conv).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    grad_full = ifft(P * np.conj(A)).real
    grad = 2.0 * grad_full[:n]
    return grad, conv


def nesterov_smoothed(a, time_budget, tau_init=8.0, tau_end=400.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed max of autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = 0.8
    lr = 2.5 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        if it % 2000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}")

        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        grad, conv = smoothed_gradient(y, tau)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        # Step
        a_new = y - (lr / gn) * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        # Update momentum
        v = momentum * v + (a_new - a)
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9997)

        # Staleness check
        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.5 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def directed_peak_shave(a, time_budget, verbose=False):
    """
    Iteratively target the worst peaks using subgradient descent.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.3
    stall = 0
    K = 10  # top K peaks to target

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]

        # Weighted subgradient from top peaks
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        weights = np.exp(5.0 * (peak_vals - max_pv) / (max_pv + 1e-15))
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    g[i] += w * 2.0 * a[j]

        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break

        a_new = a - lr * g / gn
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            lr *= 0.75
            stall += 1
            if stall > 30:
                break

        if verbose and (time.monotonic() - start) % 20 < 0.1:
            print(f"    PS: obj={best_obj:.8f}, lr={lr:.4f}")

    return best_a


def structured_perturb(a, rng, mode=0):
    """Apply a structured perturbation to escape local minima."""
    n = len(a)
    if mode == 0:
        # Low-frequency sinusoidal perturbation
        freq = rng.integers(2, max(3, n // 30))
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n)
        pert = 0.04 * np.sin(2 * np.pi * freq * x / n + phase)
        a_new = a * np.exp(pert)
    elif mode == 1:
        # Block-wise perturbation
        block_size = max(1, n // 15)
        n_blocks = n // block_size
        shifts = rng.normal(0, 0.04, n_blocks)
        a_new = a.copy()
        for b in range(n_blocks):
            a_new[b * block_size:(b + 1) * block_size] *= np.exp(shifts[b])
    elif mode == 2:
        # Frequency-domain multiplicative perturbation
        A = fft(a, n)
        mask = np.ones(n)
        # Perturb mid-frequencies more
        mid_start = n // 6
        mid_end = n - n // 6
        mask[mid_start:mid_end] = np.exp(rng.normal(0, 0.15, mid_end - mid_start))
        A_new = A * mask
        a_new = np.real(ifft(A_new))
    elif mode == 3:
        # Random local remix
        a_new = a.copy()
        num_swaps = max(1, n // 50)
        for _ in range(num_swaps):
            i = rng.integers(0, n - 10)
            j = rng.integers(i + 10, n)
            a_new[i], a_new[j] = a_new[j], a_new[i]
    elif mode == 4:
        # Power-law shaped perturbation
        alpha = rng.uniform(-1, 1)
        x = np.arange(n) / n
        pert = alpha * (x - 0.5)
        a_new = a * np.exp(2 * pert)
    else:
        a_new = a * np.exp(rng.normal(0, 0.02, n))

    a_new = np.maximum(a_new, 1e-10)
    a_new = a_new / a_new.sum()
    return a_new


def diverse_init(n, rng, mode=0):
    """Generate diverse initial sequences."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        c1, c2 = n // 4, 3 * n // 4
        w1, w2 = n // 10, n // 10
        x = np.arange(n)
        a = np.exp(-((x - c1) ** 2) / (2 * w1 ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w2 ** 2))
        return a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / np.sqrt(1 - x ** 2 + 1e-6)
        return a / a.sum()
    elif mode == 4:
        # Random Fourier
        freqs = rng.integers(1, 10, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.arange(n) / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 5:
        # Geometric progression
        r = rng.uniform(0.85, 1.15)
        a = r ** np.arange(n)
        return a / a.sum()
    elif mode == 6:
        # Sine squared
        freq = rng.uniform(1, 8)
        x = np.arange(n) / n
        a = np.sin(np.pi * freq * x) ** 2 + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses cutting-plane LP with adaptive line search,
    structured diversification, Nesterov smoothed descent, directed peak shaving,
    and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_parent = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_parent) / n_parent
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_parent}): obj={best_score:.8f}")

    n = len(best_seq)

    # === Phase 1: Initial LP polish (30% time) ===
    print(f"\n=== Phase 1: Initial LP Polish ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 350.0)
    if p1_budget > 30:
        a_lp = lp_polish(best_seq, time_budget=p1_budget, K=50, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 LP: obj={obj_lp:.8f}")

    # === Phase 2: Diversified LP rounds (30% time) ===
    print(f"\n=== Phase 2: Diversified LP ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.30, 350.0)
    if p2_budget > 30:
        n_rounds = 6
        per_round = p2_budget / n_rounds
        for rnd in range(n_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(per_round, remaining * 0.6)
            if alloc < 15:
                break

            mode = rnd % 5
            a_pert = structured_perturb(best_seq, rng, mode)
            a_result = lp_polish(a_pert, time_budget=alloc, K=40)
            obj_r = compute_obj(a_result)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_result.copy()
                print(f"  P2 r{rnd+1} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  P2 r{rnd+1} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 3: Nesterov smoothed descent (15% time) ===
    print(f"\n=== Phase 3: Nesterov Smoothed Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 180.0)
    if p3_budget > 20:
        a_ns = nesterov_smoothed(best_seq, time_budget=p3_budget,
                                 tau_init=8.0, tau_end=350.0, verbose=True)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Phase3 NSD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 NSD: obj={obj_ns:.8f}")

    # === Phase 4: Directed peak shaving (10% time) ===
    print(f"\n=== Phase 4: Directed Peak Shaving ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.10, 120.0)
    if p4_budget > 15:
        a_ps = directed_peak_shave(best_seq, time_budget=p4_budget)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  Phase4 PS: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 PS: obj={obj_ps:.8f}")

    # === Phase 5: Multi-length exploration (10% time) ===
    print(f"\n=== Phase 5: Multi-length ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.10, 120.0)
    if p5_budget > 30:
        n_options = sorted(set([
            n - 50, n + 50, n - 100, n + 100,
            700, 750, 850, 900, 950, 1000, 1050
        ]))
        n_options = [x for x in n_options if 500 <= x <= 1200 and abs(x - n) >= 40]
        n_options = n_options[:4]

        per_opt = p5_budget / max(1, len(n_options))
        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(per_opt, remaining * 0.5)
            if alloc < 15:
                break

            # Interpolate
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()

            a_try = lp_polish(a_try, time_budget=alloc, K=35)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Multi-start exploration (5% time) ===
    print(f"\n=== Phase 6: Multi-start ===")
    remaining = deadline - time.monotonic()

    if p6_budget > 20:
        n_starts = 3

        for ms in range(n_starts):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(per_start, remaining * 0.6)
            if alloc < 10:
                break

            mode = ms + 1  # skip mode 0 (uniform)
            a_init = diverse_init(n, rng, mode)
            a_init = lp_polish(a_init, time_budget=alloc, K=30)
            obj_ms = compute_obj(a_init)
            if obj_ms < best_score:
                best_score = obj_ms
                best_seq = a_init.copy()
                print(f"  ** Multi-start {ms+1} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Multi-start {ms+1}: obj={obj_ms:.8f}")

    # === Phase 7: Final intensive polish (remaining) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Alternating LP / peak shave / NSD
    for i in range(4):

        if remaining < 10:
            break

        if alloc_lp > 10:
            a_lp2 = lp_polish(best_seq, time_budget=alloc_lp, K=80)
            obj_lp2 = compute_obj(a_lp2)
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  Final {i+1} LP: obj={best_score:.8f} **")

        # Peak shave
        remaining = deadline - time.monotonic()
        alloc_ps = min(remaining * 0.2, 30.0)
        if alloc_ps > 8:
            a_ps2 = directed_peak_shave(best_seq, time_budget=alloc_ps)
            obj_ps2 = compute_obj(a_ps2)
            if obj_ps2 < best_score:
                best_score = obj_ps2
                best_seq = a_ps2.copy()
                print(f"  Final {i+1} PS: obj={best_score:.8f} **")

        # Small NSD
        remaining = deadline - time.monotonic()
        alloc_ns = min(remaining * 0.2, 30.0)
        if alloc_ns > 8:
            a_ns2 = nesterov_smoothed(best_seq, time_budget=alloc_ns,
                                      tau_init=50.0, tau_end=600.0)
            obj_ns2 = compute_obj(a_ns2)
            if obj_ns2 < best_score:
                best_score = obj_ns2
                best_seq = a_ns2.copy()
                print(f"  Final {i+1} NSD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
