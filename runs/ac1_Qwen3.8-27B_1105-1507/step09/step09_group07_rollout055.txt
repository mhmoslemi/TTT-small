Let me also think about whether there's a closed-form optimal solution.

The problem is: minimize max_k c[k] subject to sum a[i] = 1, a[i] >= 0.

This is a non-convex optimization (the constraint max_k c[k] <= t is quadratic). However, for a fixed "active set" of lags where the constraint is tight, it becomes a quadratically constrained quadratic program (QCQP), which is still non-convex in general.

The LP alternating approach works by fixing one variable (a) and optimizing the other (g) linearly, then alternating. This is similar to the method of alternating projections or block coordinate descent.

Let me think about a better approach:

**Approach: Parametric family + 1D optimization**

Consider a parametric family of sequences, e.g.:
- a[i] = (i+1)^p for p in some range (generalized ramp)
- a[i] = exp(-alpha*i) (exponential decay)
- a[i] = i^p * (n-1-i)^q (polynomial with free endpoints)

For each family, we can do a 1D (or low-dimensional) optimization over the parameters.

**Approach: Use the dual problem**

The LP for a given a has a dual. The dual variables correspond to the "active" lags. We can use the dual to guide the search.

**Approach: Subgradient method for the max**

The gradient of max_k c[k] is the indicator of the argmax. We can use this for a subgradient method, but it's not very efficient.

**Approach: Smooth max + gradient descent**

Use a smooth approximation like log-sum-exp: LSE(beta) = (1/beta) * log(sum exp(beta * c[k])). For large beta, this approximates the max. Then do gradient descent on the simplex.

"""
Optimization of non-negative step functions to minimize peak self-convolution.
Uses ramp-based initialization, LP alternating, FFT-based convolution,
and parameter sweeps. Target: score < 1.5027.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    # Use scipy linprog (HiGHS) - most reliable
    from scipy.optimize import linprog
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:

        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    
    # Fallback: try cvxpy

        g = cp.Variable(n, nonneg=True)

        prob = cp.Problem(cp.Minimize(M), constraints)

            g_val = np.maximum(g.value, 0)
            s = float(np.sum(g_val))
            if s > 1e-10:

    except:

            print(f"        LP-alt it{iteration}: obj={best_obj:.8f}, t={last_good_t:.3f}, "
                  f"stall={stall}, elapsed={elapsed:.1f}s")
    
    return best_a

def smooth_optimize(a, time_budget, verbose=False):
    """Smooth optimization using log-sum-exp surrogate."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 0.1
    beta = 50.0
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        m = conv.max()
        # Log-sum-exp gradient (softmax weights)
        shifted = beta * (conv - m)
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        # Gradient of LSE w.r.t. a[i]: sum_k weights[k] * d(conv[k])/d(a[i])
        # d(conv[k])/d(a[i]) = a[k-i] (if 0 <= k-i < n) + a[i-(k-n+1)] ... 
        # Actually conv[k] = sum_j a[j]*a[k-j], so d(conv[k])/d(a[i]) = a[k-i] (if valid)
        # The gradient of the full objective involves all k.
        
        # Simpler: compute gradient numerically or use the fact that
        # d(conv)/d(a) is a cross-correlation with the weighted sum
        grad = np.zeros(n)
        for i in range(n):
            # d(conv[k])/d(a[i]) = a[k-i] for valid k-i
            # Sum over k: grad[i] = sum_k weights[k] * a[k-i]
            # This is a cross-correlation of weights with a
            k_min = max(0, i)  # k-i >= 0 => k >= i
            k_max = min(2*n-2, i+n-1)  # k-i < n => k < i+n

        # d(conv[k])/d(a[i]) = a[k-i] (if 0<=k-i<n) + a[i] * (d(a[k-i])/d(a[i]) which is 0 unless k-i=i)

        # So d(conv[k])/d(a[i]) = a[k-i] (if 0<=k-i<n) + a[i-(k... 
        # Actually: conv[k] = sum_j a[j]*a[k-j]. The term involving a[i] appears when j=i (gives a[i]*a[k-i]) or when k-j=i i.e. j=k-i (gives a[k-i]*a[i]).
        # If k-i != i (i.e., k != 2i), these are two separate terms, each giving a[k-i].
        # If k = 2i, the term is a[i]*a[i] = a[i]^2, and d/d(a[i]) = 2*a[i].
        
        # So: d(conv[k])/d(a[i]) = a[k-i] * (1 if 0<=k-i<n and k-i != i) + a[i] * (1 if 0<=k-i<n and k-i == i) + ...
        # This is getting complicated. Let me just use: d(conv[k])/d(a[i]) = a[k-i] if 0<=k-i<n, counting the multiplicity.
        # For k != 2i: one term a[k-i] from j=i and one term a[k-i] from j=k-i (if k-i != i and 0<=k-i<n).
        # Wait no. conv[k] = sum_j a[j]*a[k-j]. The variable a[i] appears in this sum:
        # - When j = i: term is a[i]*a[k-i] (if 0 <= k-i < n)
        # - When k-j = i, i.e., j = k-i: term is a[k-i]*a[i] (if 0 <= k-i < n)
        # If k-i == i (i.e., k=2i), these are the same term: a[i]*a[i], derivative is 2*a[i].
        # If k-i != i, they're different terms, each contributing a[k-i] to the derivative.
        # So: d(conv[k])/d(a[i]) = a[k-i] if (k-i != i and 0<=k-i<n) [from j=i term]
        #                                   + a[k-i] if (k-i != i and 0<=k-i<n) [from j=k-i term]
        #                                   = 2*a[k-i] if k != 2i and 0<=k-i<n
        #                                   = 2*a[i] if k = 2i
        #                                   = 0 otherwise
        
        # Hmm, this is the same as saying d(conv[k])/d(a[i]) = a[k-i] * I(0<=k-i<n) + a[i] * I(k-i=i, 0<=k-i<n)
        # = a[k-i] * I(0<=k-i<n) + a[i] * I(k=2i)
        # But a[k-i] when k-i=i is a[i], so it's a[i]*I(k=2i) + a[i]*I(k=2i) = 2a[i] when k=2i.
        # And a[k-i] * I(0<=k-i<n) when k!=2i is just a[k-i].
        # Wait, I think the correct formula is simply:
        # d(conv[k])/d(a[i]) = a[k-i] if 0 <= k-i < n, else 0
        # This is because conv is bilinear in a, and the derivative w.r.t. a[i] picks out the term where either factor is a[i].
        
        # Actually for a bilinear form B(a,a) = sum_{j,l} C[j,l] a[j] a[l] where C is the convolution kernel,
        # d/da[i] = sum_l C[i,l] a[l] + sum_j C[j,i] a[j]
        # For convolution, C[j,l] = 1 if j+l=k... no, this isn't quite right.
        
        # Let me just use the simpler formula: for the full convolution vector c = a * a,
        # dc/da is a linear operator. The gradient of (1/beta)*log(sum exp(beta*c[k])) w.r.t. a is:
        # sum_k w[k] * (dc/d a) where w[k] = exp(beta*(c[k]-max))/sum(exp(...))
        # And (dc/da)[k,i] = a[k-i] if 0<=k-i<n, else 0 (for the "first" factor)
        #                    + a[k-i] if 0<=k-i<n, else 0 (for the "second" factor, by symmetry)
        # Wait, no. c[k] = sum_j a[j]*a[k-j]. 
        # dc[k]/da[i] = (d/d a[i]) sum_j a[j]*a[k-j]
        # = sum_j [delta(j,i)*a[k-j] + a[j]*delta(k-j,i)]
        # = a[k-i] * I(0<=k-i<n) + a[k-i] * I(0<=k-i<n)  [the second term: j=k-i, valid if 0<=k-i<n]
        # = 2*a[k-i] * I(0<=k-i<n) if k != 2i
        # = 2*a[i] if k = 2i (since both terms give a[i])
        # = a[i] + a[i] = 2*a[i]
        
        # Hmm wait, if k = 2i, then in the first sum, j=i gives a[k-i] = a[i].
        # In the second sum, j=k-i=i gives a[j] = a[i].
        # So total = a[i] + a[i] = 2*a[i]. Same as 2*a[k-i] when k-i=i.
        
        # So in all cases: dc[k]/da[i] = 2*a[k-i] if 0<=k-i<n, else 0.
        # Wait, that's not right either. If k-i = i (k=2i), we get 2*a[i]. If k-i != i, we get a[k-i] + a[k-i] = 2*a[k-i].
        # So it's always 2*a[k-i] when 0<=k-i<n.
        
        # But wait, that's only if both j=i and j=k-i are valid indices (0 to n-1).
        # j=i is valid if 0<=i<n (always true).
        # j=k-i is valid if 0<=k-i<n.
        # For the first term (j=i): we need a[k-i] to be valid, i.e., 0<=k-i<n.
        # For the second term (j=k-i): we need a[k-i] to be the coefficient, and a[j]=a[k-i] to be valid.
        # Actually, let me redo this more carefully.
        
        # c[k] = sum_{j=0}^{n-1} a[j] * a[k-j] (where a[k-j] is 0 if k-j not in [0,n-1])
        # dc[k]/da[i] = a[k-i] (from the j=i term in the first factor) + a[k-i] (from the j=k-i term in the second factor, if k-i is in [0,n-1])
        # But the second term: we need j = k-i to be in [0, n-1], and then the term is a[j]*a[i] = a[k-i]*a[i], derivative w.r.t. a[i] is a[k-i].
        # The first term: j = i, and we need k-i to be in [0, n-1], and the term is a[i]*a[k-i], derivative is a[k-i].
        # So dc[k]/da[i] = a[k-i]*I(0<=k-i<n) + a[k-i]*I(0<=k-i<n) = 2*a[k-i]*I(0<=k-i<n).
        
        # OK so the gradient is: grad[i] = sum_k w[k] * 2*a[k-i] * I(0<=k-i<n)
        # = 2 * sum_{k: 0<=k-i<n} w[k] * a[k-i]
        # = 2 * sum_{j: 0<=j<n} w[i+j] * a[j]  (where j = k-i)
        
        # This is 2 * (w * a)[i] (correlation of w with a, evaluated at lag i)
        
        # We can compute this efficiently with FFT!
        
        # Let me just compute it directly for now (O(n^2) per iteration, but n is at most ~1000)
        
        # Reset grad computation
        grad = np.zeros(n)
        for i in range(n):
            j_start = 0
            j_end = min(n, 2*n-1-i)  # k = i+j, need k < 2n-1, so j < 2n-1-i
            if j_start < j_end:
                grad[i] = 2.0 * np.dot(weights[i:i+j_end], a[:j_end])
        
        # Mirror descent (exponential weights) on the simplex
        log_a = np.log(a)
        log_a_new = log_a - lr * grad
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = max(1e-6, lr * 0.99)
            beta = min(5000, beta * 1.02)
        else:
            lr = max(1e-6, lr * 0.85)
        
        it += 1
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"        Smooth it{it}: obj={best_obj:.8f}, beta={beta:.0f}, lr={lr:.4f}, elapsed={elapsed:.1f}s")
    
    return best_a

def make_ramp(n, p=1.0):
    """Create a ramp-like sequence: a[i] = (i+1)^p, normalized."""
    a = np.arange(1, n+1, dtype=np.float64) ** p
    return a / a.sum()

def make_cosine_ramp(n):
    """Ramp with cosine taper: a[i] = (i+1) * cos^2(pi*i/(2n))"""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * np.cos(np.pi * i / (2 * n))**2
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def make_exp_ramp(n, alpha=0.5):
    """Exponentially weighted ramp: a[i] = (i+1) * exp(-alpha*i/n)"""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * np.exp(-alpha * i / n)
    return a / a.sum()

def make_arcoshape(n, p=0.5, q=1.0):
    """a[i] = (i+1)^p * (n-i)^q"""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p * (n - i) ** q
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
