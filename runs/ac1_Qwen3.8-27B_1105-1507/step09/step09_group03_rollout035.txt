import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a, active_indices=None, tol_factor=0.05):
    """
    Solve LP for direction finding.
    If active_indices is None: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0
    If active_indices given: min sum(conv(a,g)[k] for k in active), sum(g)=1, g>=0
    Returns g (normalized to sum=1) or None.
    """
    import cvxpy as cp
    n = len(a)
    a_norm = a / np.sum(a)
    
    # For efficiency, build the matrix once
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    
    if active_indices is not None:
        active_A = A[active_indices, :]
        obj = cp.sum(active_A @ g)
        constraints = [cp.sum(g) == 1]
    else:
        M = cp.Variable()
        obj = M
        constraints = [A @ g <= M, cp.sum(g) == 1]
    
    prob = cp.Problem(cp.Minimize(obj), constraints)
    
    solvers = [cp.CLBARAREL if hasattr(cp, 'CLBARAREL') else cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def golden_section_search(a, direction, f_eval, n_iter=8):
    """Golden section line search: find best t in [0,1] for a + t*(direction-a)."""
    phi = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    
    t1 = hi - phi * (hi - lo)
    t2 = lo + phi * (hi - lo)
    a1 = a + t1 * (direction - a)
    a1 = np.maximum(a1, 0)
    s1 = a1.sum()
    f1 = f_eval(a1 / s1) if s1 > 1e-10 else np.inf
    
    a2 = a + t2 * (direction - a)
    a2 = np.maximum(a2, 0)
    s2 = a2.sum()
    f2 = f_eval(a2 / s2) if s2 > 1e-10 else np.inf
    
    for _ in range(n_iter - 2):
        if f1 < f2:
            hi = t2
            t2 = t1
            a2 = a1
            f2 = f1
            t1 = hi - phi * (hi - lo)
            a1 = a + t1 * (direction - a)
            a1 = np.maximum(a1, 0)
            s1 = a1.sum()
            f1 = f_eval(a1 / s1) if s1 > 1e-10 else np.inf
        else:
            lo = t1
            t1 = t2
            a1 = a2
            f1 = f2
            t2 = lo + phi * (hi - lo)
            a2 = a + t2 * (direction - a)
            a2 = np.maximum(a2, 0)
            s2 = a2.sum()
            f2 = f_eval(a2 / s2) if s2 > 1e-10 else np.inf
    
    t_best = (lo + hi) / 2
    a_best = a + t_best * (direction - a)
    a_best = np.maximum(a_best, 0)
    s_best = a_best.sum()
    if s_best > 1e-10:
        a_best = a_best / s_best
    return a_best


def lp_polish(a, time_budget=100.0, verbose=False):
    """
    LP-based polishing with two stages.
    Stage 1: Find direction minimizing max conv(a,g)
    Stage 2: Among directions with similar max, minimize sum at active peaks
    """
    import cvxpy as cp
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        # Compute current conv
        conv_a = compute_conv(a)
        max_conv = float(np.max(conv_a))
        
        # Find active (near-maximal) convolution indices
        tol = 0.05 * max_conv
        active = np.where(conv_a >= max_conv - tol)[0]
        
        if len(active) == 0:
            active = np.array([int(np.argmax(conv_a))])
        
        # Stage 1: Min-max LP
        g1 = solve_lp_direction(a, active_indices=None)
        
        if g1 is None:
            # Fallback: gradient-based direction
            break
        
        # Stage 2: Min-sum at active indices
        g2 = solve_lp_direction(a, active_indices=active)
        
        # Choose better direction
        if g2 is not None:
            # Evaluate both directions quickly
            # Try a small step in each direction
            t_test = 0.1
            a1 = a + t_test * (g1 - a)
            a1 = np.maximum(a1, 0)
            s1 = a1.sum()
            obj1 = compute_obj(a1 / s1) if s1 > 1e-10 else np.inf
            
            a2 = a + t_test * (g2 - a)
            a2 = np.maximum(a2, 0)
            s2 = a2.sum()
            obj2 = compute_obj(a2 / s2) if s2 > 1e-10 else np.inf
            
            if obj2 < obj1:
                g_dir = g2
            else:
                g_dir = (0.7 * g1 + 0.3 * g2)
        else:
            g_dir = g1
        
        # Line search
        a_new = golden_section_search(a, g_dir, lambda x: compute_obj(x), n_iter=8)
        obj_new = compute_obj(a_new)
        
        it += 1
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, |active|={len(active)}")
        else:
            stall += 1
            # Perturb to escape
            if stall > 3:
                scale = min(0.05, 0.01 * stall)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                else:
                    a = best_a.copy()
                stall = 0
                if it > 50:
                    break
    
    return best_a

    # Using FFT: ifft(conj(W_fft) * A_fft)[j] gives sum_k w_k * conj(a[k-j])... 
    # Let me think more carefully.
    
    # Actually: grad[j] = 2 * sum_k w_k * a[j-k] (where a[m]=0 outside [0,n-1])
    # = 2 * (cross_correlation of w and a)[j] in a specific sense
    # = 2 * ifft(fft(w_padded) * conj(fft(a_padded)))[j] ... not quite
    
    # Let me just use the direct formula:
    # grad[j] = 2 * sum_k w_k * a[j-k]
    # For this, I can compute: for each k, add w_k * a shifted by k to grad
    # grad = sum_k w_k * shift(a, k)
    # where shift(a, k)[j] = a[j-k] if 0<=j-k<n else 0
    
    # Using FFT: pad a and w to length N >= 2n-1
    # shift(a, k) in FFT domain: multiply fft(a) by exp(-2πik/N)
    # So grad = ifft(sum_k w_k * exp(-2πik/N) * fft(a))
    # = ifft(fft(w) * fft(a)) ... no, that's convolution not correlation
    
    # Let me just be direct with FFT:
    # grad[j] = 2 * sum_k w_k * a[j-k]
    # Let's define b[m] = w[m] for m in [0, 2n-2], 0 otherwise
    # Then grad[j] = 2 * sum_m b[m] * a[j-m]
    # This is (b * a_rev)[j] where a_rev[m] = a[-m] ... getting confused
    
    # Simplest correct approach: use the fact that
    # d/dh[(a*h)[k]] at h=a is a convolved with delta at position k
    # = a shifted by k positions (to the right by k, with wrap-around in linear conv)
    
    # So grad = 2 * sum_k w_k * (a shifted right by k, zero-padded)
    # In FFT: if we zero-pad a to length N >= 2n-1, then a shifted right by k
    # corresponds to multiplying FFT(a) by exp(2πik/N) in frequency domain
    
    # Actually let me just use a different approach:
    # grad[j] = 2 * sum_k w_k * a[j-k] (with a out of range = 0)
    # = 2 * (w * a)[j] where * is the appropriate cross-correlation
    
    # Let's define: C[j] = sum_k w[k] * a[j-k]
    # Pad w to length N, pad a to length N
    # C = ifft(fft(w) * conj(fft(a)))... let me verify
    # fft(w)[m] = sum_k w[k] exp(-2πimk/N)
    # conj(fft(a))[m] = sum_j a[j] exp(2πimj/N)
    # product[m] = sum_k sum_j w[k] a[j] exp(2πim(j-k)/N)
    # ifft(product)[j] = (1/N) sum_m product[m] exp(2πimj/N)
    # = (1/N) sum_k sum_j' w[k] a[j'] exp(2πij'/N) exp(2πim(j'-k)/N) exp(2πimj/N)
    # Hmm this is getting complicated. Let me just use a simpler method.
    
    # Direct computation using the structure:
    # For each k, the contribution to grad from w[k] is:
    #   w[k] * 2 * (a shifted right by k)
    # i.e., grad[j] += 2 * w[k] * a[j-k] for valid j-k
    
    # For efficiency with moderate n, we can use FFT:
    # grad = 2 * ifft(conj(fft(w_padded)) * fft(a_padded))
    # where w_padded has w in positions 0..2n-2
    
    # Let me verify: ifft(conj(W) * A)[j] where W = fft(w_padded), A = fft(a_padded)
    # = sum_k w_padded[k] * a_padded[j+k] ... no
    
    # OK let me just use the most straightforward FFT approach:
    # We want C[j] = sum_k w[k] * a[j-k]
    # This is the cross-correlation of w and a, evaluated at lag j.
    # In FFT: C = ifft(conj(fft(w, N)) * fft(a, N))
    # where N >= max(len(w), len(a)) + min(len(w), len(a)) - 1 for linear correlation
    
    # Actually, for linear cross-correlation:
    # C[j] = sum_k w[k] * a[k+j]  (this is one convention)
    # or C[j] = sum_k w[k+j] * a[k]  (another)
    
    # Let me define what I need: grad[j] = 2 * sum_k w[k] * a[j-k]
    # Let m = j-k, so k = j-m: grad[j] = 2 * sum_m w[j-m] * a[m]
    # This is 2 * (w * a)[j] in the convolution sense (w convolved with a, evaluated at j)
    # = 2 * ifft(fft(w) * fft(a))[j]
    
    N = 1
    while N < 4 * n:
        N <<= 1
    
    w_padded = np.zeros(N, dtype=complex)
    w_padded[:2*n-1] = weights
    
    a_padded = np.zeros(N, dtype=complex)
    a_padded[:n] = a
    
    grad = 2.0 * np.real(ifft(fft(w_padded) * fft(a_padded))[:n])
    
    return grad, conv


def nesterov_smoothed(a, time_budget=200.0, tau_end=500.0, verbose=False):
    """
    Nesterov accelerated gradient descent on smoothed max(conv).
    Uses the log-sum-exp smoothing with increasing tau.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = 20.0 * (tau_end / 20.0) ** (progress ** 0.6)

        if verbose and it % 500 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}")

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        grad, conv = smoothed_gradient(y, tau)
        
        # Project gradient to simplex (subtract mean)
        grad_centered = grad - np.mean(grad)
        
        # Gradient step with Barzilai-Borwein step size adaptation
        a_new = y - lr * grad_centered
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
        
        # Update momentum
        v = a_new - y
        a = a_new
        it += 1
        
        # Adapt learning rate
        lr = max(0.1, min(5.0 / np.sqrt(n), lr * 0.9999))
        
        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)
            best_it = it
            stall = 0
            if stall > 20:
                break

    return best_a


def peak_shaving_cd(a, time_budget=60.0, verbose=False):
    """
    Coordinate descent that specifically targets reducing the maximum peak.
    Identifies top contributing pairs and tries to reduce them.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_val = float(np.max(conv))
        peak_k = int(np.argmax(conv))
        
        # Find all positions contributing to the peak
        # (a*a)[k] = sum_i a[i]*a[k-i] for valid i
        i_min = max(0, peak_k - (n - 1))
        i_max = min(n - 1, peak_k)
        
        if i_min > i_max:
            break
        
        contributions = []
        for i in range(i_min, i_max + 1):
            j = peak_k - i
            if 0 <= j < n:
                contributions.append((a[i] * a[j], i, j))
        
        contributions.sort(reverse=True)
        
        # Try to reduce the top contributions
        improved = False
        for contrib_val, src, dst in contributions[:5]:
            if contrib_val < 1e-15:
                continue
            if src == dst:
                # Can't reduce a[src]^2 by moving mass from src to dst (same)
                # Try moving mass from src to a random other position
                tgt = int(rng.integers(0, n))
                while tgt == src:
                    tgt = int(rng.integers(0, n))
                for frac in [0.3, 0.15, 0.05]:
                    delta = frac * a[src]
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[tgt] += delta
                    if a_try[src] < 0:
                        continue
                    s_try = a_try.sum()
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
            else:
                # Try reducing the product a[src]*a[dst]
                # Option 1: reduce a[src] (move to elsewhere)
                # Option 2: reduce a[dst] (move to elsewhere)
                for which in [0, 1]:
                    if which == 0 and a[src] > a[dst]:
                        source = src
                    elif which == 1 and a[dst] >= a[src]:
                        source = dst
                    else:
                        source = src if which == 0 else dst
                    
                    # Try several target positions
                    targets = []
                    # Positions that don't contribute to the same peak
                    for t in range(n):
                        if t == source:
                            continue
                        # Check if moving to t would create a new contribution to a peak
                        # For simplicity, just try a few random + structured targets
                    targets = [
                        int(rng.integers(0, n)),
                        int(rng.integers(0, n)),
                        max(0, source - max(1, n // 10)),
                        min(n - 1, source + max(1, n // 10)),
                        n - 1 - source  # mirror position
                    ]
                    
                    for frac in [0.3, 0.15, 0.05, 0.02]:
                        delta = frac * a[source]
                        if delta <= 0:
                            continue
                        for tgt in targets:
                            tgt = int(tgt) % n
                            if tgt == source:
                                continue
                            a_try = a.copy()
                            a_try[source] -= delta
                            a_try[tgt] += delta
                            if a_try[source] < 0:
                                continue
                            s_try = a_try.sum()
                            if s_try < 1e-10:
                                continue
                            a_try = a_try / s_try
                            obj_try = compute_obj(a_try)
                            if obj_try < best_obj - 1e-14:
                                best_obj = obj_try
                                best_a = a_try.copy()
                                a = a_try
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
            
            if improved:
                break

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                # Perturb
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if it > 200:
                    break

        if verbose and it % 20 == 0:
            print(f"    PCD it{it}: obj={best_obj:.8f}, peak={peak_val:.8f}")

    return best_a


def spectral_peak_reduction(a, time_budget=60.0, verbose=False):
    """
    Spectral approach: modify the spectrum to reduce time-domain peaks.
    Based on the CMOS/iterative peak reduction method.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 4 * n:
        N <<= 1

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_val = float(np.max(conv))
        
        if stall > 50:
            break
        
        # Identify peak positions in convolution
        peak_threshold = 0.8 * peak_val
        peak_positions = np.where(conv >= peak_threshold)[0]
        
        # In spectral domain, try to reduce energy at frequencies that contribute to these peaks
        # The peak at position k in time domain gets contribution from all frequencies
        # weighted by exp(i*omega*k). We want to reduce the coherent sum.
        
        # Simple approach: reduce the spectral magnitude at frequencies where
        # the phase aligns with the peak positions
        A_fft = fft(a, N)
        A_mag = np.abs(A_fft)
        A_phase = np.angle(A_fft)
        
        # For peaks at positions peak_positions, the phase alignment at frequency m is:
        # sum_k_in_peaks exp(i * 2*pi*m*k/N)
        # We want to reduce |A_fft[m]| where this sum has large magnitude
        # (i.e., where the frequency contributes coherently to the peak)
        
        # Compute the "peak alignment" spectrum
        peak_indicator = np.zeros(N)
        for k in peak_positions:
            if 0 <= k < N:
                peak_indicator[k] = 1.0
        
        alignment = np.abs(fft(peak_indicator, N)) / max(len(peak_positions), 1)
        
        # Reduce spectral magnitude where alignment is high (but keep DC)
        reduction = 1.0 - 0.1 * alignment
        reduction[0] = 1.0  # Keep DC (sum) unchanged
        A_modified = A_fft * reduction
        A_modified[0] = A_fft[0]  # Ensure sum stays the same
        
        a_new = np.real(ifft(A_modified, N)[:n])
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        it += 1
    
    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Asymmetric: first half larger
        x = np.arange(n)
        a = np.exp(-x / (n / 3))
        a = a / a.sum()
    elif mode == 3:
        # Asymmetric: second half larger
        x = np.arange(n)
        a = np.exp(-(n - 1 - x) / (n / 3))
        a = a / a.sum()
    elif mode == 4:
        # Bimodal asymmetric
        centers = [n // 5, 4 * n // 5]
        widths = [n // 10, n // 15]
        amps = [1.0, 0.6]
        a = np.zeros(n)
        for c, w, amp in zip(centers, widths, amps):
            a += amp * np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Random with asymmetry
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * x + 0.2 * rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 6:
        # Triangle (peak in middle) - symmetric baseline
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)
        a = a / a.sum()
    else:
        # Ramp
        a = np.arange(1, n + 1, dtype=float)
        a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization combining LP direction finding, smoothed gradient descent,
    peak-shaving coordinate descent, and spectral peak reduction.
    
    Phases:
    1. Quick smoothed GD to get into good basin
    2. LP polishing (two-stage)
    3. Peak shaving CD
    4. Spectral peak reduction
    5. Multi-n exploration
    6. Diverse restarts
    7. Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent available: {e}")

    if best_seq is None:
        best_seq = diverse_init(n_target, rng, mode=2)
        best_score = compute_obj(best_seq)
        print(f"Init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Quick smoothed GD ===
    print(f"\n=== Phase 1: Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.2, 200.0)
    if p1_budget > 20:
        a_gd = nesterov_smoothed(best_seq, time_budget=p1_budget, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  GD: obj={best_score:.8f} **")
        else:
            print(f"  GD: obj={obj_gd:.8f}")

    # === Phase 2: LP polishing ===
    print(f"\n=== Phase 2: LP Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 300.0)
    if p2_budget > 20:
        a_lp = lp_polish(best_seq, time_budget=p2_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 3: Peak shaving CD ===
    print(f"\n=== Phase 3: Peak Shaving CD ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 120.0)
    if p3_budget > 15:
        a_cd = peak_shaving_cd(best_seq, time_budget=p3_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 4: Spectral peak reduction ===
    print(f"\n=== Phase 4: Spectral Reduction ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.1, 60.0)
    if p4_budget > 10:
        a_sp = spectral_peak_reduction(best_seq, time_budget=p4_budget, verbose=True)
        obj_sp = compute_obj(a_sp)
        if obj_sp < best_score:
            best_score = obj_sp
            best_seq = a_sp.copy()
            print(f"  Spectral: obj={best_score:.8f} **")
        else:
            print(f"  Spectral: obj={obj_sp:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(200, n_current - 200), n_current + 150, n_current + 300,
        600, 1000, 1200, 1500, 2000
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000 and abs(n - n_current) > 100]

    for n_try in n_options[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        
        # Quick optimization
        alloc1 = min(remaining * 0.15, 60.0)
        alloc2 = min(remaining * 0.1, 40.0)
        if alloc1 > 10:
            a_try = nesterov_smoothed(a_try, time_budget=alloc1)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            # Further polish
            if alloc2 > 10:
                a_try = lp_polish(a_try, time_budget=alloc2)
            obj_try = compute_obj(a_try)
        
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = r % 7
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with current best
        blend = 0.3
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.15, 80.0)
        if alloc > 20:
            half = alloc / 2
            a_r = nesterov_smoothed(a_r, time_budget=half)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_r = lp_polish(a_r, time_budget=min(remaining * 0.2, 40.0))

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP
    if remaining > 30:
        alloc = min(remaining * 0.3, 80.0)
        a_lp = lp_polish(best_seq, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # CD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.25, 50.0)
        a_cd = peak_shaving_cd(best_seq, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # GD
    remaining = deadline - time.monotonic()

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses multi-phase optimization: smoothed gradient descent, two-stage LP direction
finding, peak-shaving coordinate descent, spectral peak reduction, penalty method,
and diverse restarts with multi-n exploration.
Key improvements: fixed LP solver, better active sets, penalty-based descent,
asymmetric initialization exploiting anti-correlation insight.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a, active_indices=None):
    """
    Solve LP for direction finding.
    If active_indices is None: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0
    If active_indices given: min sum(conv(a,g)[k] for k in active), sum(g)=1, g>=0
    Returns (g, M_or_obj) or (None, None).
    """
    import cvxpy as cp
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)

    if active_indices is not None:
        active_A = A[active_indices, :]
        obj = cp.sum(active_A @ g)
        constraints = [cp.sum(g) == 1]
    else:
        M = cp.Variable()
        obj = M
        constraints = [A @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(obj), constraints)

    solvers_to_try = []
    for sname in ['CLARABEL', 'SCS', 'ECOS', 'PDLP']:
        s = getattr(cp, sname, None)
        if s is not None:
            solvers_to_try.append(s)

    for solver in solvers_to_try:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    g_out = g_val / s_val
                    if active_indices is not None:
                        return g_out, None
                    else:
                        return g_out, float(M.value) if M.value is not None else None
        except Exception:
            continue
    return None, None


def golden_section_search(a, direction, n_iter=10):
    """Golden section line search: find best t in [0,1] for a + t*(direction-a)."""
    phi = (np.sqrt(5) - 1) / 2.0
    lo, hi = 0.0, 1.0

    t1 = hi - phi * (hi - lo)
    t2 = lo + phi * (hi - lo)

    a1 = a + t1 * (direction - a)
    a1 = np.maximum(a1, 0)
    s1 = a1.sum()
    f1 = compute_obj(a1 / s1) if s1 > 1e-10 else np.inf

    a2 = a + t2 * (direction - a)
    a2 = np.maximum(a2, 0)
    s2 = a2.sum()
    f2 = compute_obj(a2 / s2) if s2 > 1e-10 else np.inf

    for _ in range(n_iter - 2):
        if f1 < f2:
            hi = t2
            t2 = t1
            a2 = a1
            f2 = f1
            t1 = hi - phi * (hi - lo)
            a1 = a + t1 * (direction - a)
            a1 = np.maximum(a1, 0)
            s1 = a1.sum()
            f1 = compute_obj(a1 / s1) if s1 > 1e-10 else np.inf
        else:
            lo = t1
            t1 = t2
            a1 = a2
            f1 = f2
            t2 = lo + phi * (hi - lo)
            a2 = a + t2 * (direction - a)
            a2 = np.maximum(a2, 0)
            s2 = a2.sum()
            f2 = compute_obj(a2 / s2) if s2 > 1e-10 else np.inf

    t_best = (lo + hi) / 2.0
    a_best = a + t_best * (direction - a)
    a_best = np.maximum(a_best, 0)
    s_best = a_best.sum()
    if s_best > 1e-10:
        a_best = a_best / s_best
    return a_best

def smoothed_gradient(a, tau):
    """
    Compute smoothed gradient of max(conv(a,a)) using log-sum-exp.
    Returns (gradient, convolution_array).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    weights = np.exp(tau * (conv - max_c))
    weights = weights / float(np.sum(weights))

    # grad[j] = 2 * sum_k w[k] * a[j-k] (linear cross-correlation)
    N = 1
    while N < 4 * n:
        N <<= 1

    w_padded = np.zeros(N, dtype=complex)
    w_padded[:2 * n - 1] = weights

    a_padded = np.zeros(N, dtype=complex)
    a_padded[:n] = a

    grad = 2.0 * np.real(ifft(fft(w_padded) * fft(a_padded))[:n])

    return grad, conv

    best_it = 0

def penalty_descent(a, time_budget=120.0, verbose=False):

def peak_shaving_cd(a, time_budget=60.0, verbose=False):
    """
    Coordinate descent targeting the top peaks.
    Tries to reduce the maximum peak by moving mass from peak contributors.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_val = float(np.max(conv))
        peak_k = int(np.argmax(conv))

        i_min = max(0, peak_k - (n - 1))
        i_max = min(n - 1, peak_k)
        if i_min > i_max:
            break

        contributions = []
        for i in range(i_min, i_max + 1):
            j = peak_k - i
            if 0 <= j < n:
                contributions.append((a[i] * a[j], i, j))
        contributions.sort(reverse=True)

        improved = False
        for contrib_val, src, dst in contributions[:4]:
            if contrib_val < 1e-15:
                continue

            # Determine which element to reduce (the larger one)
            if src == dst:
                source = src
            else:
                source = src if a[src] >= a[dst] else dst

            # Try several target positions
            target_candidates = [
                int(rng.integers(0, n)),
                int(rng.integers(0, n)),
                max(0, source - max(1, n // 8)),
                min(n - 1, source + max(1, n // 8)),
                n - 1 - source,
                max(0, source - n // 4),
                min(n - 1, source + n // 4),
            ]

            for frac in [0.3, 0.15, 0.05, 0.02]:
                delta = frac * a[source]
                if delta <= 0:
                    continue
                for tgt in target_candidates:
                    tgt = int(tgt) % n
                    if tgt == source:
                        continue
                    a_try = a.copy()
                    a_try[source] -= delta
                    a_try[tgt] += delta
                    if a_try[source] < -1e-15:
                        continue
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 25:
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if it > 300:
                    break

        if verbose and it % 30 == 0:
            print(f"    PCD it{it}: obj={best_obj:.8f}, peak={peak_val:.8f}")

    return best_a


def spectral_peak_reduction(a, time_budget=60.0, verbose=False):
    """
    Spectral approach: modify the spectrum to reduce time-domain peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 4 * n:
        N <<= 1

    it = 0
    stall = 0
    rng = np.random.default_rng(555)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_val = float(np.max(conv))

        if stall > 40:
            break

        # Identify peak positions
        peak_threshold = 0.85 * peak_val
        peak_positions = np.where(conv >= peak_threshold)[0]

        A_fft = fft(a, N)
        A_abs = np.abs(A_fft)

        # Compute alignment spectrum
        peak_indicator = np.zeros(N)
        for k in peak_positions:
            if 0 <= k < N:
                peak_indicator[k] = 1.0

        alignment = np.abs(fft(peak_indicator, N)) / max(int(np.sum(peak_indicator)), 1)
        alignment = alignment / (max(alignment.max(), 1e-10))

        # Reduce spectral magnitude where alignment is high
        strength = 0.05 + 0.03 * min(it / 20.0, 1.0)
        reduction = 1.0 - strength * alignment
        reduction[0] = 1.0

        # Add small random phase perturbation for diversity
        if it % 10 == 0:
            phase_pert = rng.normal(0, 0.01, N)
        else:
            phase_pert = 0.0

        A_modified = A_fft * reduction
        A_modified[0] = A_fft[0]
        # Apply tiny phase perturbation (excluding DC)
        if phase_pert > 0:
            A_modified = A_modified * np.exp(1j * phase_pert)
            A_modified[0] = A_fft[0]

        a_new = np.real(ifft(A_modified, N)[:n])
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        it += 1

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.arange(n)
        a = np.exp(-x / (n / 3.0))
        a = a / a.sum()
    elif mode == 3:
        x = np.arange(n)
        a = np.exp(-(n - 1 - x) / (n / 3.0))
        a = a / a.sum()
    elif mode == 4:
        centers = [n // 5, 4 * n // 5]
        widths = [n // 10, n // 15]
        amps = [1.0, 0.6]
        a = np.zeros(n)
        for c, w, amp in zip(centers, widths, amps):
            a += amp * np.exp(-((np.arange(n) - c) ** 2) / (2.0 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * x + 0.2 * rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x) + 0.1 * rng.uniform(0, 1, n)
        a = a / a.sum()
    else:
        a = np.arange(1, n + 1, dtype=float)
        a = a / a.sum()
    return np.maximum(a, 0) / np.sum(np.maximum(a, 0))


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization combining LP direction finding, smoothed gradient
    descent, penalty method, peak-shaving CD, spectral reduction, and diverse restarts.
    
    Phases:
    1. Quick smoothed GD (warm start)
    2. LP polishing (two-stage with adaptive active sets)
    3. Penalty-based descent (hard threshold smoothing)
    4. Peak shaving coordinate descent
    5. Spectral peak reduction
    6. Multi-n exploration
    7. Diverse restarts
    8. Final intensive polish (cycle through all methods)
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent available: {e}")

    if best_seq is None:
        best_seq = diverse_init(n_target, rng, mode=2)
        best_score = compute_obj(best_seq)
        print(f"Init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Quick smoothed GD ===
    print(f"\n=== Phase 1: Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.18, 180.0)
    if p1_budget > 20:
        a_gd = nesterov_smoothed(best_seq, time_budget=p1_budget, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  GD: obj={best_score:.8f} **")
        else:
            print(f"  GD: obj={obj_gd:.8f}")

    # === Phase 2: LP polishing ===
    print(f"\n=== Phase 2: LP Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 300.0)
    if p2_budget > 30:
        a_lp = lp_polish(best_seq, time_budget=p2_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 3: Penalty descent ===
    print(f"\n=== Phase 3: Penalty Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    if p3_budget > 20:
        a_pen = penalty_descent(best_seq, time_budget=p3_budget, verbose=True)
        obj_pen = compute_obj(a_pen)
        if obj_pen < best_score:
            best_score = obj_pen
            best_seq = a_pen.copy()
            print(f"  Penalty: obj={best_score:.8f} **")
        else:
            print(f"  Penalty: obj={obj_pen:.8f}")

    # === Phase 4: Peak shaving CD ===
    print(f"\n=== Phase 4: Peak Shaving CD ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.12, 100.0)
    if p4_budget > 15:
        a_cd = peak_shaving_cd(best_seq, time_budget=p4_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 5: Spectral peak reduction ===
    print(f"\n=== Phase 5: Spectral Reduction ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.08, 50.0)
    if p5_budget > 10:
        a_sp = spectral_peak_reduction(best_seq, time_budget=p5_budget, verbose=True)
        obj_sp = compute_obj(a_sp)
        if obj_sp < best_score:
            best_score = obj_sp
            best_seq = a_sp.copy()
            print(f"  Spectral: obj={best_score:.8f} **")
        else:
            print(f"  Spectral: obj={obj_sp:.8f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(200, n_current - 200), max(300, n_current + 100),
        n_current + 250, 600, 1000, 1200, 1500, 2000
    ]))
    n_options = [x for x in n_options if 200 <= x <= 2000 and abs(x - n_current) > 100]

    for n_try in n_options[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc1 = min(remaining * 0.12, 60.0)
        alloc2 = min(remaining * 0.08, 40.0)

        if alloc1 > 10:
            a_try = nesterov_smoothed(a_try, time_budget=alloc1)
        if alloc2 > 10:
            a_try = lp_polish(a_try, time_budget=alloc2)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        mode = (r * 2 + 2) % 7
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.25 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = a_r.sum()
        if s_r > 1e-10:
            a_r = a_r / s_r

        alloc = min(remaining * 0.15, 80.0)
        if alloc > 20:
            third = alloc / 3.0
            a_r = nesterov_smoothed(a_r, time_budget=third)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_r = lp_polish(a_r, time_budget=min(third, remaining * 0.3))
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_r = peak_shaving_cd(a_r, time_budget=min(third, remaining * 0.3))

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Cycle: LP -> CD -> Penalty -> LP
    for cycle in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * 0.2, 60.0)

        if cycle == 0:
            a_new = lp_polish(best_seq, time_budget=alloc)
        elif cycle == 1:
            a_new = peak_shaving_cd(best_seq, time_budget=alloc)
        else:
            a_new = penalty_descent(best_seq, time_budget=alloc)

        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Final cycle {cycle}: obj={best_score:.8f} **")
        else:
            print(f"  Final cycle {cycle}: obj={obj_new:.8f}")

    # One last pass
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 30.0)
        a_final = lp_polish(best_seq, time_budget=alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if val < np.inf:
            best_score = val
        else:
            result = [1.0] * 100
            best_score = 2.0
    except NameError:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
