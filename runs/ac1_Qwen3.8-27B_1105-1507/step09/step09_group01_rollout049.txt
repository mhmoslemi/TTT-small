spectral-based initialization, symmetry exploitation, and multi-length exploration.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.sparse import csr_matrix
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_sparse(a):
    """Build sparse matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):
            i = k - j
            if i < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_minpeak(a, M_target=None, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses sparse matrix construction for efficiency.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_sparse(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue

    return None, None

        if len(peak_indices) == 0:
            peak_indices = [np.argmax(conv)]

        # Compute subgradient: weighted sum of peak gradients
        # Gradient of (a*a)[k] w.r.t. a[i] = a[k-i] (if valid) + a[i-k] (if valid)
        g = np.zeros(n)
        # Weight peaks by how much they exceed the mean
        mean_conv = np.mean(conv)
        weights = (conv[peak_indices] - mean_conv + 1e-10)
        weights = np.maximum(weights, 0)
        if weights.sum() < 1e-15:
            weights = np.ones(len(peak_indices))
        weights /= weights.sum()

        for w, k in zip(weights, peak_indices):
            k = int(k)
            # (a*a)[k] = sum_{i} a[i] * a[k-i] where 0<=i<n and 0<=k-i<n
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                # d/d a[i] of sum_j a[j]*a[k-j] = a[k-i] + a[i-k] (boundary terms)
                for i in range(i_start, i_end + 1):
                    j = k - i
                    if 0 <= j < n:
                        g[i] += w * a[j]
                    # Also the symmetric term if i != j
                    if i != j and 0 <= i < n:
                        pass  # already counted in the sum

        # Actually, for (a*a)[k] = sum_i a[i]*a[k-i], the gradient w.r.t a[m] is:
        # sum over i where i=m: a[k-m] + sum over i where k-i=m: a[i] = a[k-m] + a[k-m]
        # Wait no. (a*a)[k] = sum_{i=0}^{n-1} a[i] * a[k-i] (where a[j]=0 for j outside [0,n-1])
        # d/d(a[m]) (a*a)[k] = a[k-m] (from i=m term) + (if k-m != m, also from i=k-m term: a[m])
        # Actually: d/d(a[m]) sum_i a[i]*a[k-i] = a[k-m] + a[m] if k-m != m else 2*a[k-m]
        # More precisely: = a[k-m] (from derivative of a[i] w.r.t a[m] when i=m)
        #                   + a[m] (from derivative of a[k-i] w.r.t a[m] when k-i=m, i.e., i=k-m)
        # So if m != k-m: gradient = a[k-m] + a[m-k]... wait let me be more careful.

        # (a*a)[k] = sum_i a[i] * a[k-i]
        # d/d(a[m]) = d/d(a[m]) sum_i a[i]*a[k-i]
        # = a[k-m] * 1 (from i=m term, derivative of a[i] w.r.t a[m])
        #   + sum_i a[i] * d(a[k-i])/d(a[m])
        # = a[k-m] + a[m] (from the term where k-i=m, i.e., i=k-m, if k-m is in range)
        # But wait, if m == k-m (i.e., k=2m), then it's 2*a[m].

        # Let me just compute it directly for correctness:
        g = np.zeros(n)
        for w, k in zip(weights, peak_indices):
            k = int(k)
            for i in range(max(0, k - (n - 1)), min(n, k + 1)):
                j = k - i
                if 0 <= i < n and 0 <= j < n:
                    g[i] += w * a[j]

        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        # Multiplicative update (mirror descent with entropy regularizer)
        a_new = a * np.exp(-lr * g)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        # Also try additive update for comparison
        a_add = project_simplex(a - lr * g)

        obj_mult = compute_obj(a_new)
        obj_add = compute_obj(a_add)

        if obj_mult <= obj_add:
            a = a_new
            obj_cur = obj_mult
        else:
            a = a_add
            obj_cur = obj_add

        if obj_cur < best_obj - 1e-13:
            best_obj = obj_cur
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                lr *= 0.5
                stall = 0
                if lr < 1e-12:
                    break

        lr *= decay
        it += 1

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD it{it}: obj={best_obj:.8f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a


def lp_polish_improved(a, time_budget=300.0, verbose=False, seed=42):
    """
    Improved LP-based polishing with:
    - Wider step size search range
    - Sparse LP construction
    - Adaptive constraint tightening
    - Dual-guided perturbation
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    no_improve = 0
    last_t = 0.5

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Wider golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0

        def eval_t(t):
            a_try = best_a + t * (g_lp - best_a)
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return 1e10
            a_try = a_try / s_try
            return compute_obj(a_try)

        # Search in [0, 1.5] (allow extrapolation beyond g)
        a_lo, a_hi = 0.0, 1.5
        c = a_hi - gr * (a_hi - a_lo)
        d = a_lo + gr * (a_hi - a_lo)
        fc = eval_t(c)
        fd = eval_t(d)

        for _ in range(30):
            if abs(a_hi - a_lo) < 0.008:
                break
            if fc < fd:
                a_hi = d
                d = c
                fd = fc
                c = a_hi - gr * (a_hi - a_lo)
                fc = eval_t(c)
            else:
                a_lo = c
                c = d
                fc = fd
                d = a_lo + gr * (a_hi - a_lo)
                fd = eval_t(d)

        t_opt = (a_lo + a_hi) / 2.0
        t_opt = max(0.01, min(t_opt, 1.5))

        a_try = best_a + t_opt * (g_lp - best_a)
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            a_try = best_a.copy()
        else:
            a_try = a_try / s_try
        obj_try = compute_obj(a_try)

        iter_count += 1

        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a.copy()
            last_t = t_opt
            no_improve = 0
            if verbose and iter_count % 3 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={t_opt:.3f}, M={M_lp:.6f}, el={elapsed:.0f}s")
        else:
            no_improve += 1
            if no_improve == 2:
                # Try smaller step
                t_small = max(0.01, last_t * 0.15)
                a_try2 = best_a + t_small * (g_lp - best_a)
                a_try2 = np.maximum(a_try2, 0)
                a_try2 = a_try2 / a_try2.sum()
                obj_try2 = compute_obj(a_try2)
                if obj_try2 < best_obj - 1e-14:
                    best_obj = obj_try2
                    best_a = a_try2.copy()
                    a = best_a.copy()
                    no_improve = 0
            elif no_improve == 4:
                # Try negative step (move away from LP direction)
                t_neg = -0.3
                a_try2 = best_a + t_neg * (g_lp - best_a)
                a_try2 = np.maximum(a_try2, 0)
                s2 = float(np.sum(a_try2))
                if s2 > 1e-10:
                    a_try2 = a_try2 / s2
                    obj_try2 = compute_obj(a_try2)
                    if obj_try2 < best_obj - 1e-14:
                        best_obj = obj_try2
                        best_a = a_try2.copy()
                        a = best_a.copy()
                        no_improve = 0
            elif no_improve >= 6:
                # Perturb using peak information
                conv = compute_conv(best_a)
                max_conv = np.max(conv)
                peak_idx = np.argmax(conv)
                # Perturb coefficients that contribute to the peak
                n_p = len(best_a)
                i_start = max(0, int(peak_idx) - (n_p - 1))
                i_end = min(n_p - 1, int(peak_idx))
                pert = np.ones(n_p)
                scale = 0.02 * min(no_improve, 10)
                pert[i_start:i_end+1] = np.exp(rng.normal(0, scale, i_end - i_start + 1))
                pert[i_start:i_end+1] *= np.exp(rng.normal(0, scale, i_end - i_start + 1))
                a = best_a * pert
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                if iter_count > 50 and time.monotonic() - start > time_budget * 0.8:
                    break

        # Periodic constraint-tightening LP
        if iter_count % 10 == 5 and M_lp is not None:
            for tight_frac in [0.93, 0.97]:
                M_tight = M_lp * tight_frac
                g_tight, _ = solve_lp_minpeak(a, M_target=M_tight)
                if g_tight is not None:
                    for t_try in [0.3, 0.5, 0.7]:
                        a_try3 = best_a + t_try * (g_tight - best_a)
                        a_try3 = np.maximum(a_try3, 0)
                        s3 = float(np.sum(a_try3))
                        if s3 < 1e-10:
                            continue
                        a_try3 = a_try3 / s3
                        obj_try3 = compute_obj(a_try3)
                        if obj_try3 < best_obj - 1e-14:
                            best_obj = obj_try3
                            best_a = a_try3.copy()
                            a = best_a.copy()
                            no_improve = 0
                            if verbose:
                                print(f"    LP-tight it{iter_count}: obj={best_obj:.8f}, frac={tight_frac}, t={t_try}")
                            break

    return best_a


def subgradient_refine(a, time_budget=80.0, K=8, verbose=False):
    """Subgradient method with top-K peaks, adaptive weighting, and momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 3.0 / np.sqrt(n)
    momentum = 0.6
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = np.max(conv)
        if max_conv < 1e-15:
            break

        # Use all peaks above 80% of max
        threshold = 0.8 * max_conv
        peak_mask = conv >= threshold
        peak_indices = np.where(peak_mask)[0]

        if len(peak_indices) == 0:
            peak_indices = [np.argmax(conv)]

        # Weight by excess over mean
        mean_conv = np.mean(conv)
        weights = np.maximum(conv[peak_indices] - mean_conv, 0)
        if weights.sum() < 1e-15:
            weights = np.ones(len(peak_indices))
        weights /= weights.sum()

        # Compute gradient
        g = np.zeros(n)
        for w, k in zip(weights, peak_indices):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                for i in range(i_start, i_end + 1):
                    j = k - i
                    if 0 <= j < n:
                        g[i] += w * a[j]

        v = momentum * v + g
        a_new = project_simplex(a - lr * v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            lr *= 0.998
            if stall > 200:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break

        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation with smooth kernel."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    # Use a smoother resampling: pad with zeros, interpolate, crop
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def kernel_init(n, mode, rng):
    """Generate diverse initializations using various kernels."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Raised cosine
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 3:
        # Gaussian
        x = np.arange(n)
        a = np.exp(-((x - n/2)**2) / (2*(n/4)**2))
        a = a / a.sum()
    elif mode == 4:
        # Spectrally flat: random phases
        N = 4 * n
        phases = rng.uniform(0, 2*np.pi, N)
        A = np.ones(N) * np.exp(1j * phases)
        A[0] = 1.0  # DC component
        a = np.real(ifft(A)[:n])
        a = np.maximum(a, 0)
        if a.sum() < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / a.sum()
    elif mode == 5:
        # Two-bump
        c1 = n // 4
        c2 = 3 * n // 4
        w = n // 6
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1)**2) / (2*w**2))
        a += np.exp(-((np.arange(n) - c2)**2) / (2*w**2))
        a = a / a.sum()
    elif mode == 6:
        # Cosine window with ripple
        x = np.linspace(0, 1, n)
        freq = rng.uniform(2, 8)
        a = (0.5 + 0.5*np.cos(2*np.pi*x)) * (1 + 0.3*np.sin(2*np.pi*freq*x))
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 7:
        # Inverse Chebyshev-like
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(np.maximum(1 - x**2, 1e-8)))
        a = a / a.sum()
    elif mode == 8:
        # Polynomial window
        x = np.linspace(-1, 1, n)
        p = rng.uniform(1, 4)
        a = (1 - x**2)**p
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def spectral_polish(a, time_budget=60.0, verbose=False):
    """
    Frequency-domain inspired polishing: try to make the power spectrum |A(f)|^2
    as flat as possible, which should reduce autocorrelation peaks.
    Uses iterative phase adjustment in the frequency domain.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 4 * n:
        N <<= 1

    lr = 0.1
    it = 0

    while time.monotonic() - start < time_budget:
        # Compute FFT
        A = fft(a, N)
        # Power spectrum
        P = np.abs(A)**2
        P_max = np.max(P)
        if P_max < 1e-15:
            break

        # Gradient: we want to reduce peaks in the autocorrelation
        # which corresponds to reducing peaks in the time-domain autocorrelation
        # Alternative: directly minimize max of IDFT(P)
        conv_spec = irfft(P, N)[:2*n-1]
        max_conv = np.max(conv_spec)

        # Subgradient approach in frequency domain
        # The peak in time domain at lag k corresponds to a specific frequency component
        # Use IFFT of the peak indicator to get frequency-domain gradient
        peak_lag = int(np.argmax(conv_spec))
        indicator = np.zeros(2*n-1)
        indicator[peak_lag] = 1.0

        # FFT of the indicator gives us the frequency-direction to adjust
        # Actually, let's use a time-domain subgradient approach but with
        # frequency-domain regularization
        conv = compute_conv(a)
        max_c = np.max(conv)
        peak_idx = np.argmax(conv)

        # Compute subgradient
        g = np.zeros(n)
        i_start = max(0, peak_idx - (n-1))
        i_end = min(n-1, peak_idx)
        for i in range(i_start, i_end+1):
            j = peak_idx - i
            if 0 <= j < n:
                g[i] += a[j]

        # Add frequency-domain regularization: penalize non-flat spectrum
        P_flat = P / np.mean(P)
        P_excess = np.maximum(P_flat - 1, 0)
        # Gradient of sum(P_excess) w.r.t. a (through the FFT)
        # P = |A|^2, dP/da involves A conjugate
        dP_da = 2 * np.real(ifft(P_excess * np.conj(A), N)[:n])

        g_total = g + 0.1 * dP_da

        a_new = project_simplex(a - lr * g_total)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.95
            if lr < 1e-10:
                break

        lr = min(lr * 1.01, 0.5)
        it += 1

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining:
    1. Diverse kernel-based initialization at multiple lengths
    2. Mirror descent for global structure
    3. LP polishing with sparse matrices and improved step sizes
    4. Spectral polishing
    5. Multi-n exploration
    6. Perturbation and restart
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Quick diverse initialization search ===
    print(f"\n=== Phase 1: Diverse Init Search ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.1, 100.0)
    if p1_budget > 20:
        # Try a few different kernel inits at the current n
        for mode in [2, 3, 6, 8]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            a_try = kernel_init(n_target, mode, rng)

                print(f"  ** New best from kernel {mode}")

            a_md = mirror_descent_polish(best_seq, time_budget=min(remaining * 0.5, 60.0), seed=7)

            if obj_md < best_score:
                best_score = obj_md
                best_seq = a_md.copy()
                print(f"  Phase1 MD: obj={best_score:.8f} **")
            else:
                print(f"  Phase1 MD: obj={obj_md:.8f}")

    # === Phase 2: Intensive LP Polishing ===
    print(f"\n=== Phase 2: Intensive LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 500.0)
    if p2_budget > 50:
        a_lp = lp_polish_improved(best_seq, time_budget=p2_budget, verbose=True, seed=42)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f}")

    # === Phase 3: Mirror Descent Polish ===
    print(f"\n=== Phase 3: Mirror Descent Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 120.0)
    if p3_budget > 20:
        a_md = mirror_descent_polish(best_seq, time_budget=p3_budget, verbose=True, seed=88)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Phase3 MD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 MD: obj={obj_md:.8f}")

    # === Phase 4: Subgradient Refinement ===

        alloc2 = min(remaining * 0.2, 80.0)
        if alloc2 > 15:
            a_try = lp_polish_improved(a_try, time_budget=alloc2, seed=n_try + 100)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Perturbation + Re-optimization ===
    print(f"\n=== Phase 6: Perturbation + Re-LP ===")
    remaining = deadline - time.monotonic()

    for r in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        # Different perturbation strategies
        if r < 3:
            # Small Gaussian perturbation
            scale = 0.008 + 0.005 * r
            a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        elif r < 6:
            # Peak-targeted perturbation
            a_pert = best_seq.copy()
            conv = compute_conv(a_pert)
            peak_idx = int(np.argmax(conv))
            n_p = len(a_pert)
            i_start = max(0, peak_idx - (n_p - 1))
            i_end = min(n_p - 1, peak_idx)
            width = max(1, (i_end - i_start) // 4)
            a_pert[i_start:i_start+width] *= (1 + 0.05 * rng.standard_normal(width))
            a_pert[max(0, i_end-width+1):i_end+1] *= (1 + 0.05 * rng.standard_normal(min(width, i_end+1-max(0,i_end-width+1))))
        else:
            # Kernel blend
            a_kernel = kernel_init(len(best_seq), r - 6, rng)
            a_pert = 0.7 * best_seq + 0.3 * a_kernel

        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc = min(remaining * 0.2, 80.0)
        if alloc > 15:
            # Alternate between LP and mirror descent
            if r % 2 == 0:
                a_pert = lp_polish_improved(a_pert, time_budget=alloc, seed=200 + r)
            else:
                a_pert = mirror_descent_polish(a_pert, time_budget=alloc, seed=200 + r)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Perturb {r} (type={'G' if r<3 else 'P' if r<6 else 'K'}): obj={best_score:.8f} **")
            else:
                print(f"  Perturb {r} (type={'G' if r<3 else 'P' if r<6 else 'K'}): obj={obj_pert:.8f}")

    # === Phase 7: Final intensive LP ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.6, 200.0)
        a_final = lp_polish_improved(best_seq, time_budget=alloc, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # Quick final subgradient
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 30.0)
        a_sg = subgradient_refine(best_seq, time_budget=alloc, K=10)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # One more tiny LP pass
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final2 = lp_polish_improved(best_seq, time_budget=min(remaining * 0.5, 40.0), seed=777)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Extra LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
