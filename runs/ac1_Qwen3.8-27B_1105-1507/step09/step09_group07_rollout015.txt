import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_standard(a):
    """Standard LP: min M s.t. conv(a_norm, g) <= M, sum(g)=1, g>=0"""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    # Try cvxpy solvers
    for solver in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except:
            continue

        A_ub[:, n] = -1.0

        A_eq[0, :n] = 1.0

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_standard(a):
    """Standard LP: min M s.t. conv(a_norm, g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=np.zeros(2 * n - 1), A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_lp_reweighted(a, weights):
    """Reweighted LP: min M s.t. weights_k * conv(a_norm, g)_k <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    Aw = A * weights[:, np.newaxis]

    for solver in [cp.CLARABEL, cp.MOSEK]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [Aw @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:

        except Exception:
            continue

        # Actually, standard cross-correlation: (f ⋆ g)[i] = sum_j conj(f[j]) * g[j+i]
        # We want sum_j s[j+i] * a[j] = sum_j a[j] * s[j+i]
        # This is (a ⋆ s)[i] = sum_j a[j] * s[j+i] = (a ⋆ s)[i]

        # (a ⋆ s)[i] = sum_j a[j] * conj(s[j-i]) ... no that's not right either

def project_simplex(v):
    """Project vector v onto the simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def qp_flatten(a, time_budget, verbose=False):
    """
    QP: minimize variance of conv(a,g) subject to conv(a,g) <= M_lp, sum(g)=1, g>=0.
    This encourages a flatter cross-correlation profile.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Get LP bound
    _, M_lp = solve_lp_standard(a)
    if M_lp is None:
        return best_a

    A = build_conv_matrix(a)
    H = A.T @ A  # n x n

    success = False
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        if not success:
            try:
                g = cp.Variable(n, nonneg=True)
                constraints = [A @ g <= M_lp * 1.001, cp.sum(g) == 1]
                prob = cp.Problem(cp.Minimize(0.5 * cp.quad_form(g, cp.psd_wrap(H))), constraints)
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s_val = float(np.sum(g_val))
                    if s_val > 1e-10:
                        g_val = g_val / s_val
                        success = True
            except Exception:
                continue
        if success:
            break

    if not success:
        return best_a

    # Interpolate between a and QP solution
    for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
        if time.monotonic() - start > time_budget:
            break
        a_try = (1 - t) * a + t * g_val
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()

    if verbose:
        print(f"          QP-flatten: obj={best_obj:.8f}")

    return best_a


def spectral_flatten(a, time_budget, verbose=False):
    """More aggressive spectral flattening with multiple strategies."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))

        return a

    best_a = a.copy()

    while size < 2 * n:
        size <<= 1

        if time.monotonic() - start > time_budget:
            break
        for it in range(50):
            if time.monotonic() - start > time_budget:
                break
            A = rfft(a, size)
            P = np.abs(A) ** 2
            P_flat = (1 - alpha) * P + alpha * np.mean(P)
            P_flat = np.maximum(P_flat, 1e-20)
            A_new = np.sqrt(P_flat) * (A / (np.abs(A) + 1e-30))
            a_new = irfft(A_new, size)[:n]

                a = a_new.copy()
            else:
                break

    if verbose:
        print(f"          Spectral: obj={best_obj:.8f}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode % 10 == 0:
        a = np.ones(n) / n
    elif mode % 10 == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode % 10 == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 3:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 0.5
        a = a / a.sum()
    elif mode % 10 == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.5) ** 2)
        a = a / a.sum()
    elif mode % 10 == 5:
        x = np.linspace(0, 1, n)
        a = x * (1 - x) * 4 + 0.1
        a = a / a.sum()
    elif mode % 10 == 6:
        freqs = rng.integers(2, 10, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 7:
        # Biperiodic
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4 * np.pi * x) + 0.2 * np.cos(6 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 8:
        # Random with structure
        a = rng.dirichlet(np.ones(n) * 0.3)
    else:
        # Triangular
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2 * x - 1)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining smooth gradient descent, LP alternating with
    reweighting and momentum, QP flattening, spectral shaping, n-sweep, and
    diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"\nInitial: n={n_current}, obj={best_score:.8f}, target=1.5027")

    # === Phase 1: Smooth gradient descent (25%) ===
    print(f"\n=== Phase 1: Smooth Gradient Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.25
    if p1_budget > 30:
        a_gd = smooth_gradient_descent(best_seq, time_budget=p1_budget,
                                       beta_start=15.0, beta_end=3000.0,
                                       lr_init=0.03, momentum=0.85, verbose=True)
        obj_gd = compute_obj(a_gd)
        marker = " **" if obj_gd < best_score else ""
        print(f"  SmoothGD: obj={obj_gd:.8f}{marker}")
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()

    # === Phase 2: LP alternating improved (25%) ===
    print(f"\n=== Phase 2: LP Alternating Improved ===")
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.25
    if p2_budget > 30:
        a_lp = lp_alternate_improved(best_seq, time_budget=p2_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        marker = " **" if obj_lp < best_score else ""
        print(f"  LP-alt: obj={obj_lp:.8f}{marker}")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()

    # === Phase 3: QP flattening + Spectral (10%) ===
    print(f"\n=== Phase 3: QP + Spectral ===")
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.10
    if p3_budget > 20:
        half = p3_budget / 2
        a_qp = qp_flatten(best_seq, time_budget=half, verbose=True)
        obj_qp = compute_obj(a_qp)
        marker = " **" if obj_qp < best_score else ""
        print(f"  QP: obj={obj_qp:.8f}{marker}")
        if obj_qp < best_score:
            best_score = obj_qp
            best_seq = a_qp.copy()

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_spec = spectral_flatten(best_seq, time_budget=min(remaining * 0.3, 30), verbose=True)
            obj_spec = compute_obj(a_spec)
            marker = " **" if obj_spec < best_score else ""
            print(f"  Spectral: obj={obj_spec:.8f}{marker}")
            if obj_spec < best_score:
                best_score = obj_spec
                best_seq = a_spec.copy()

    # === Phase 4: n-sweep (25%) ===
    print(f"\n=== Phase 4: n-Sweep ===")
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.25

    n_options = sorted(set([
        500, 550, 600, 650, 700, 750, 800, 850, 900, 950,
        1000, 1050, 1100, 1150, 1200, 1300, 1400,
        n_current - 50, n_current - 20, n_current + 20, n_current + 50,
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500]

    per_n_budget = p4_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_n_budget, remaining * 0.12)
        if alloc < 10:
            alloc = 10
        if alloc > n_current:
            alloc = min(alloc, 40)

        a_try = resample_sequence(best_seq, n_try)
        
        # Quick smooth gradient + LP
        half = alloc / 2
        if half > 5:
            a_opt = smooth_gradient_descent(a_try, time_budget=half,
                                            beta_start=20, beta_end=2000,
                                            lr_init=0.03, momentum=0.85, verbose=False)
        else:
            a_opt = a_try
        remaining = deadline - time.monotonic()
        lp_alloc = min(remaining * 0.05, 20)
        if lp_alloc > 10:
            a_opt = lp_alternate_improved(a_opt, time_budget=lp_alloc, verbose=False)

        obj_opt = compute_obj(a_opt)
        marker = " **" if obj_opt < best_score else ""
        print(f"  n={n_try}: obj={obj_opt:.8f}{marker}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try

    # === Phase 5: Diverse restarts (10%) ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p5_budget = remaining * 0.10
    n_use = len(best_seq)
    n_restarts = 5
    per_restart = p5_budget / max(1, n_restarts)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_restart, remaining * 0.2)
        if alloc < 15:
            break

        mode = (r + 3) % 10
        a_r = diverse_init(n_use, rng, mode=mode)
        # Blend with best
        blend = 0.2 + 0.08 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        # Quick optimization
        third = alloc / 3
        if third > 5:
            a_r = smooth_gradient_descent(a_r, time_budget=third,
                                          beta_start=20, beta_end=2000,
                                          lr_init=0.03, momentum=0.85, verbose=False)
        remaining = deadline - time.monotonic()
        lp_alloc = min(remaining * 0.06, 25)
        if lp_alloc > 10:
            a_r = lp_alternate_improved(a_r, time_budget=lp_alloc, verbose=False)

        obj_r = compute_obj(a_r)
        marker = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}{marker}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()

    # === Phase 6: Final intensive polish (5% of remaining) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # One more round of smooth gradient with fine-tuned parameters
        alloc = min(remaining * 0.3, 60)
        a_final = smooth_gradient_descent(best_seq, time_budget=alloc,
                                          beta_start=100, beta_end=10000,
                                          lr_init=0.01, momentum=0.9, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.4, 60)
            a_final2 = lp_alternate_improved(best_seq, time_budget=alloc, verbose=True)
            obj_final2 = compute_obj(a_final2)
            if obj_final2 < best_score:
                best_score = obj_final2
                best_seq = a_final2.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.3, 30)
            a_final3 = smooth_gradient_descent(best_seq, time_budget=alloc,
                                              beta_start=500, beta_end=20000,
                                              lr_init=0.005, momentum=0.92, verbose=False)
            obj_final3 = compute_obj(a_final3)
            if obj_final3 < best_score:
                best_score = obj_final3
                best_seq = a_final3.copy()
                print(f"  Final fine GD: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"=== Target: 1.5027, Achieved: {best_score:.8f}, Gap: {best_score - 1.5027:.8f} ===")

    return result
