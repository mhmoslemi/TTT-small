        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, M_target=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.SCS, cp.MOSEK, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue
    return None, None


def golden_section_search(f, a_lo, a_hi, tol=0.01, max_iter=40):
    """Golden section search for minimum of f on [a_lo, a_hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if abs(a_hi - a_lo) < tol:
            break
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = f(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = f(d)
    return (a_lo + a_hi) / 2.0


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Sine-based: |sin| has relatively flat autocorrelation
        x = np.linspace(0, 1, n)
        a = np.abs(np.sin(np.pi * x)) + 0.1
        a = a / a.sum()
    elif mode == 3:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * (x - 0.5))
        a = a / a.sum()
    elif mode == 4:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Multi-frequency sine
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f in [2, 3, 5]:
            a += 0.2 * np.sin(2 * np.pi * f * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Beta-like distribution
        x = np.linspace(0.01, 0.99, n)
        a = x ** 0.5 * (1 - x) ** 0.5
        a = a / a.sum()
    elif mode == 7:
        # arcsine distribution
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-8))
        a = a / a.sum()
    elif mode == 8:
        # Log-normal-like
        a = np.exp(-0.5 * ((np.arange(n) - n/2) / (n/4)) ** 2)
        a = a / a.sum()
    elif mode == 9:
        # Sawtooth
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2 * x - 1)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def nesterov_smoothed_descent(a, time_budget=100.0, tau_init=8.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.85
    lr = 4.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget * 0.85, 1e-10))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient via FFT
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_fft = fft(y, N)
        g_full = ifft(P * np.conj(A_fft)).real
        g = 2.0 * g_full[:n]

        # Barzilai-Borwein adaptive step
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.4 * lr + 0.6 * min(max(bb_lr, 1e-8), 15.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9998)

        if it - best_it > 40000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 4.0 / np.sqrt(n) * 0.3
            best_it = it

        if verbose and it % 10000 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}")

    return best_a


def ips_optimize(a, time_budget=100.0, K=5, verbose=False, seed=42):
    """
    Iterative Peak Smoothing: multiplicative update that reduces
    convolution peaks by penalizing responsible elements.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    lr = 0.8
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)

        # Find top K peaks
        top_idx = np.argsort(conv)[::-1][:K]
        top_vals = conv[top_idx]
        max_val = top_vals[0]

        # Compute responsibility vector (weighted by peak excess)
        r = np.zeros(n)
        for idx in range(K):
            k = int(top_idx[idx])
            weight = top_vals[idx] / max_val
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                r[i_start:i_end + 1] += weight * a[j_start:j_end + 1][::-1]

        # Multiplicative update with temperature
        temp = max(0.1, 1.0 - (time.monotonic() - start) / time_budget)
        a_new = a * np.exp(-lr * temp * r)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                lr *= 0.7
            if stall > 300:
                # Perturb and restart
                scale = 0.02 * min(stall // 300, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                stall = 0
                lr = 0.5
                if it > 20000:
                    break
            else:
                lr *= 0.995

        it += 1
        if verbose and it % 5000 == 0:
            print(f"    IPS it{it}: obj={best_obj:.8f}, lr={lr:.4f}")

    return best_a


def lp_polish_aggressive(a, time_budget=200.0, verbose=False, seed=42):
    """
    Aggressive LP-based polishing with adaptive M-target sweeps,
    golden-section blending, and randomized LP perturbations.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    no_improve = 0
    last_t = 0.5
    lp_calls = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start

        # Occasionally use a perturbed a for the LP to escape local minima
        a_for_lp = best_a
        if no_improve > 6 and rng.random() < 0.3:
            scale = 0.01 * min(no_improve, 5)
            a_for_lp = best_a * np.exp(rng.normal(0, scale, n))
            a_for_lp = np.maximum(a_for_lp, 0)
            a_for_lp = a_for_lp / a_for_lp.sum()

        # Solve LP
        g_lp, M_lp = solve_lp_minpeak(a_for_lp)
        lp_calls += 1
        if g_lp is None:
            break

        # Golden section search for optimal blend
        def blend_obj(t):
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return 1e10
            a_try = a_try / s_try
            return compute_obj(a_try)

        t_opt = golden_section_search(blend_obj, 0.01, 1.0, tol=0.02, max_iter=30)
        a_try = (1.0 - t_opt) * best_a + t_opt * g_lp
        a_try = np.maximum(a_try, 0)
        a_try = a_try / a_try.sum()
        obj_try = compute_obj(a_try)

        iter_count += 1

        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            last_t = t_opt
            no_improve = 0
            if verbose and iter_count % 3 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={t_opt:.3f}")
        else:
            no_improve += 1
            if no_improve == 2:
                t_small = max(0.005, last_t * 0.1)
                a_try2 = (1.0 - t_small) * best_a + t_small * g_lp
                a_try2 = np.maximum(a_try2, 0)
                a_try2 = a_try2 / a_try2.sum()
                obj_try2 = compute_obj(a_try2)
                if obj_try2 < best_obj - 1e-14:
                    best_obj = obj_try2
                    best_a = a_try2.copy()
                    no_improve = 0
            elif no_improve >= 5:
                # M-target sweep: try tighter constraints
                if M_lp is not None:
                    for frac in [0.90, 0.85, 0.80, 0.75]:
                        M_tight = M_lp * frac
                        g_tight, _ = solve_lp_minpeak(a_for_lp, M_target=M_tight)
                        lp_calls += 1
                        if g_tight is not None:
                            for t_try in [0.3, 0.5, 0.7]:
                                a_try3 = (1.0 - t_try) * best_a + t_try * g_tight
                                a_try3 = np.maximum(a_try3, 0)
                                s3 = float(np.sum(a_try3))
                                if s3 > 1e-10:
                                    a_try3 = a_try3 / s3
                                    obj3 = compute_obj(a_try3)
                                    if obj3 < best_obj - 1e-14:
                                        best_obj = obj3
                                        best_a = a_try3.copy()
                                        no_improve = 0
                                        if verbose:
                                            print(f"    LP-tight it{iter_count}: obj={best_obj:.8f}, frac={frac}, t={t_try}")
                                        break
                            if no_improve == 0:
                                break

                # Larger perturbation
                scale = 0.03 * min(no_improve, 8)
                a_for_lp = best_a * np.exp(rng.normal(0, scale, n))
                a_for_lp = np.maximum(a_for_lp, 0)
                a_for_lp = a_for_lp / a_for_lp.sum()
                no_improve = 0
                if iter_count > 50:
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
