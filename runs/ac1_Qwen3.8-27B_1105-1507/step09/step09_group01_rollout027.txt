"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a multi-phase approach: diverse structured initialization, LP-based alternating
optimization with adaptive constraint tightening, spectral domain refinement,
differential mutation for diversity, and systematic length search.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft, next_fast_len
import time
from scipy.optimize import minimize


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = next_fast_len(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, M_target=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    If M_target given, add M <= M_target and minimize sum(g) (or just feasibility).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Use FFT-based approach for speed: we don't actually need to build the full matrix
    # But for LP, we need explicit constraints. Let's use a sparse representation.
    from scipy.sparse import csr_matrix, lil_matrix
    
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for i in range(2 * n - 1):
            if 0 <= i - j < n:
                rows.append(i)
                cols.append(j)
                vals.append(a_norm[i - j])
    
    # This is too slow for large n. Let's use a different approach.
    # Actually for n=800, 2n-1=1599 rows and n=800 cols, dense is 1.27M entries, fine.
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    result = None
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    result = (g_val / s, float(M.value))
                    break
        except Exception:
            continue

    return result if result else (None, None)


def lp_polish(a, time_budget=200.0, verbose=False, seed=42):
    """
    LP-based polishing with adaptive constraint tightening and 
    perturbative restarts. Improved version with better step control.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 5:
            break

        # Solve LP with current best
        result = solve_lp_minpeak(best_a)
        g_lp, M_lp = result
        if g_lp is None:
            break

        # Try different blend factors
        best_t = 0.0
        best_blend_obj = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_t = t

        iter_count += 1

        if best_blend_obj < best_obj - 1e-14:
            a_try = (1.0 - best_t) * best_a + best_t * g_lp
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            best_obj = best_blend_obj
            best_a = a_try.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve == 3:
                # Try constraint-tightening
                for tight_frac in [0.90, 0.93, 0.96, 0.98]:
                    M_tight = M_lp * tight_frac
                    result_t = solve_lp_minpeak(best_a, M_target=M_tight)
                    g_tight, _ = result_t
                    if g_tight is not None:
                        for t in [0.3, 0.5, 0.7]:
                            a_try = (1.0 - t) * best_a + t * g_tight
                            a_try = np.maximum(a_try, 0)
                            a_try = a_try / a_try.sum()
                            obj_try = compute_obj(a_try)
                            if obj_try < best_obj - 1e-14:
                                best_obj = obj_try
                                best_a = a_try.copy()
                                no_improve = 0
                                break
                        if no_improve == 0:
                            break
            elif no_improve >= 6:
                # Perturb
                scale = 0.01 * min(no_improve, 10)
                best_a = best_a * np.exp(rng.normal(0, scale, n))
                best_a = np.maximum(best_a, 0)
                best_a = best_a / best_a.sum()
                best_obj = compute_obj(best_a)
                no_improve = 0
                if iter_count > 80:
                    break

    return best_a


def subgradient_descent(a, time_budget=100.0, K=8, verbose=False):
    """Subgradient method focusing on top-K convolution peaks."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 5.0 / n
    momentum = 0.8
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        max_val = conv[top_idx[0]]
        if max_val < 1e-15:
            break

        # Weighted subgradient
        weights = np.exp(5.0 * (conv[top_idx] - max_val) / max_val)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            lr *= 0.99
            if stall > 200:
                v *= 0.1
                stall = 0
                if lr < 1e-12:
                    break

        it += 1

    return best_a


def smoothed_gradient_descent(a, time_budget=100.0, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov-accelerated gradient descent on log-sum-exp smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    it_since_best = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        # Nesterov step
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Smoothed gradient via FFT
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        # Gradient: d/dy [sum_k p_k * (y*y)_k] = 2 * sum_k p_k * (P_k * y) where P_k is the "convolution partner"
        # More precisely: (y*y)_k = sum_i y_i * y_{k-i}, so d/dy_j = sum_k p_k * sum_i [delta_{i,j} y_{k-i} + y_i delta_{k-i,j}]
        # = sum_k p_k * (y_{k-j} + y_{k-j}) ... = 2 * sum_k p_k * (reversed y)_{k-j}
        # Using FFT: gradient = 2 * ifft(fft(p, N) * conj(fft(y, N)))[:n]
        N = next_fast_len(3 * n)
        P = fft(p, N)
        A = fft(y, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            it_since_best = 0
        else:
            it_since_best += 1
            if it_since_best > 20000:
                a = best_a.copy()
                v = np.zeros(n)
                lr = 2.0 / np.sqrt(n) * 0.5
                it_since_best = 0

        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.99995)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Double bump
        centers = [n // 4, 3 * n // 4]
        widths = [max(1, n // 8), max(1, n // 8)]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / (np.sqrt(1 - x**2) + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        # Low-frequency sinusoidal
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        # Cosine bump (centered)
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        # Gaussian pair
        c1 = rng.integers(0, n // 3)
        c2 = rng.integers(2 * n // 3, n)
        w = max(1, n // 10)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 8:
        # Low-pass filtered noise
        a = rng.uniform(0, 1, n)
        # Apply a simple moving average
        k = max(1, n // 20)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 9:
        # Triangular
        a = np.abs(1 - 2 * np.linspace(0, 1, n))
        a = a / a.sum()
    elif mode == 10:
        # Quartic bump
        x = np.linspace(-1, 1, n)
        a = (1 - x**4) ** 2
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 11:
        # Symmetric low-pass
        freqs = rng.integers(1, 4, size=2)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f in freqs:
            a += 0.5 * np.cos(2 * np.pi * f * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def spectral_refine(a, time_budget=60.0, verbose=False):
    """
    Refine using spectral domain insights.
    The convolution peak relates to the power spectrum.
    We try to flatten the effective power spectrum by adjusting the sequence.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # In frequency domain, conv(a,a) = ifft(|fft(a)|^2)
    # We want to reduce the peak of ifft(|fft(a)|^2)
    # The peak corresponds to high |fft(a)|^2 energy at DC (for the center of convolution)
    # and other frequencies
    
    it = 0
    lr = 0.1
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]
        
        # Compute gradient with respect to a at the peak position
        # (a*a)[k] = sum_i a[i] * a[k-i]
        # d(da*a/dk)/da[j] = a[k-j] + a[j-(something)]
        # More precisely: d(ka)/da[j] where k = max_idx

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = next_fast_len(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_active(a, M_target=None, active_frac=1.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for active k, sum(g)=1, g>=0.
    active_frac: only use constraints where (a*a)_k >= active_frac * max(
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Compute current convolution to identify active constraints
    conv_a = compute_conv(a_norm)
    max_conv = float(np.max(conv_a))
    
    if active_frac < 1.0:
        # Only include constraints near the peak
        threshold = active_frac * max_conv
        active_mask = conv_a >= threshold
    else:
        active_mask = np.ones(len(conv_a), dtype=bool)
    
    # Build constraint matrix for active constraints only
    active_indices = np.where(active_mask)[0]
    n_active = len(active_indices)
    
    A = np.zeros((n_active, n))
    for idx, k in enumerate(active_indices):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[idx, j] = a_norm[i]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None


def solve_lp_full(a, M_target=None):
    """Solve LP with all constraints (slower but more accurate)."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None


def backtracking_blend(best_a, g_lp, max_iter=20):
    """Find optimal blend factor via backtracking line search."""
    best_t = 0.0
    best_obj = compute_obj(best_a)
    t = 1.0
    
    for _ in range(max_iter):
        a_try = (1.0 - t) * best_a + t * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            t *= 0.5
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_t = t
            # Try a larger step
            t_new = min(1.0, t * 1.3)
            if t_new <= t + 1e-6:
                break
            t = t_new
        else:
            t *= 0.6
            if t < 0.005:
                break
    
    return best_t, best_obj


def lp_polish_advanced(a, time_budget=200.0, verbose=False, seed=42):
    """
    Advanced LP polishing with active-set reduction, backtracking,
    constraint tightening sweeps, and perturbative restarts.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 5:
            break

        # Choose between active-set LP (faster) and full LP
        use_active = (iter_count % 3 != 2)  # Use full LP every 3rd iteration
        if use_active:
            g_lp, M_lp = solve_lp_active(best_a, active_frac=0.85)
        else:
            g_lp, M_lp = solve_lp_full(best_a)
        
        if g_lp is None:
            break

        # Backtracking line search for blend factor
        t_opt, obj_opt = backtracking_blend(best_a, g_lp, max_iter=15)

        iter_count += 1

        if t_opt > 0.01 and obj_opt < best_obj - 1e-14:
            a_try = (1.0 - t_opt) * best_a + t_opt * g_lp
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            best_obj = obj_opt
            best_a = a_try.copy()
            no_improve = 0
            if verbose and iter_count % 5 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={t_opt:.3f}, M={M_lp:.6f}, el={elapsed:.0f}s")
        else:
            no_improve += 1
            if no_improve == 2:
                # Try constraint tightening
                for tight_frac in [0.92, 0.95, 0.97]:
                    M_tight = M_lp * tight_frac
                    g_tight, _ = solve_lp_active(best_a, M_target=M_tight, active_frac=0.9)
                    if g_tight is not None:
                        t_try, obj_try = backtracking_blend(best_a, g_tight, max_iter=10)
                        if t_try > 0.01 and obj_try < best_obj - 1e-14:
                            a_try = (1.0 - t_try) * best_a + t_try * g_tight
                            a_try = np.maximum(a_try, 0)
                            a_try = a_try / a_try.sum()
                            best_obj = obj_try
                            best_a = a_try.copy()
                            no_improve = 0
                            break
            elif no_improve >= 5:
                # Perturb and restart
                scale = 0.008 * min(no_improve, 12)
                best_a = best_a * np.exp(rng.normal(0, scale, n))
                best_a = np.maximum(best_a, 0)
                best_a = best_a / best_a.sum()
                best_obj = compute_obj(best_a)
                no_improve = 0
                if iter_count > 100:
                    break

    return best_a


def subgradient_refine(a, time_budget=100.0, K=8, verbose=False):
    """Subgradient method with top-K peaks, momentum, and adaptive step."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 3.0 / np.sqrt(n)
    momentum = 0.75
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        max_pv = conv[top_idx[0]]
        if max_pv < 1e-15:
            break
        weights = np.exp((conv[top_idx] - max_pv) * 8.0 / max_pv)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.02, 10.0 / n)
        else:
            stall += 1
            lr *= 0.995
            if stall > 300:
                v *= 0.2
                stall = 0
                if lr < 1e-12:
                    break

        it += 1

    return best_a


def smoothed_gradient_descent(a, time_budget=100.0, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov-accelerated gradient descent on log-sum-exp smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    it_since_best = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = next_fast_len(3 * n)
        P = fft(p, N)
        A = fft(y, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            it_since_best = 0
        else:
            it_since_best += 1
            if it_since_best > 15000:
                a = best_a.copy()
                v = np.zeros(n)
                lr = 2.0 / np.sqrt(n) * 0.5
                it_since_best = 0

        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.99995)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [max(1, n // 8), max(1, n // 8)]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        xs = np.linspace(-0.99, 0.99, n)
        a = 1.0 / (np.sqrt(1 - xs**2) + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        xs = np.linspace(0.01, 1, n)
        a = xs ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        xs = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(xs)
        a = a / a.sum()
    elif mode == 7:
        c1 = rng.integers(0, n // 3)
        c2 = rng.integers(2 * n // 3, n)
        w = max(1, n // 10)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 8:
        a = rng.uniform(0, 1, n)
        k = max(1, n // 20)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 9:
        a = np.abs(1 - 2 * x)
        a = a / a.sum()
    elif mode == 10:
        xs = np.linspace(-1, 1, n)
        a = (1 - xs**4) ** 2
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 11:
        freqs = rng.integers(1, 4, size=2)
        a = np.ones(n)
        for f in freqs:
            a += 0.5 * np.cos(2 * np.pi * f * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 12:
        # Low-frequency dominant
        a = 1.0 + 0.4 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x + rng.uniform(0, np.pi))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 13:
        # Heavy-tailed
        a = 1.0 / (1.0 + 5.0 * (x - 0.5) ** 2)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

                print(f"  DE gen{gen} ind{i}: obj={best_score:.8f} **")
            
            obj_parent = compute_obj(population[i])
            new_pop.append(crossed if obj_c < obj_parent else population[i].copy())
        
        population = new_pop

    # === Phase 7: Perturbation + Re-LP (10% time) ===
    print(f"\n=== Phase 7: Perturbation + Re-LP ===")
    remaining = deadline - time.monotonic()

    for r in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        scale = 0.007 + 0.005 * r
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc = min(remaining * 0.25, 80.0)
        if alloc > 15:
            a_pert = lp_polish_advanced(a_pert, time_budget=alloc, seed=600 + r)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Perturb {r} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Perturb {r} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 8: Final intensive polish (remainder) ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.6, 120.0)
        a_final = lp_polish_advanced(best_seq, time_budget=alloc, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # Final subgradient
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 25.0)
        a_sg = subgradient_refine(best_seq, time_budget=alloc, K=12)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
