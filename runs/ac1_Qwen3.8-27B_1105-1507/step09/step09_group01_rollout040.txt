"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of LP polishing, annealed subgradient, multiplicative updates,
and multi-length exploration with simulated annealing restarts.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a, M_target=None, max_iter=200):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Use a more efficient construction
    # A[k, j] = a_norm[k-j] if 0 <= k-j < n else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(j, j + n):
            if k < 2 * n - 1:
                A[k, j] = a_norm[k - j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue
    
    return None, None

def annealed_subgradient(a, time_budget=200.0, K_init=3, K_max=20, verbose=False, seed=42):
    """
    Subgradient method with annealing over number of peak positions.
    Adapts step size and K (number of peaks to consider).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    lr = 5.0 / np.sqrt(n)
    momentum = 0.5
    v = np.zeros(n)
    stall = 0
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Anneal K
        K = int(K_init + (K_max - K_init) * progress)
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Get top-K peak positions
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        
        # Weight peaks by their value (exponential)
        temperature = max_conv * 0.1 * (1.0 - progress * 0.5)
        weights = peak_vals - peak_vals[0]
        weights = np.exp(weights / (temperature + 1e-15))
        weights /= weights.sum()
        
        # Compute subgradient (weighted average of peak gradients)
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            # Gradient of (a*a)[k] w.r.t. a[i] is a[k-i] (if valid)
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                g[i] += w * a[k - i]
        
        # Center the subgradient (project onto tangent space of simplex)
        support = a > 1e-12
        if support.any():
            g_centered = g.copy()
            g_centered[support] -= g[support].mean()
        else:
            g_centered = g
        
        v = momentum * v + g_centered
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            no_improve = 0
        else:
            stall += 1
            no_improve += 1
            if stall > 100:
                lr *= 0.7
                v *= 0.5
                stall = 0
            elif no_improve > 500:
                # Reheat
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                v = np.zeros(n)
                lr = 5.0 / np.sqrt(n)
                no_improve = 0
        
        # Adapt momentum
        momentum = 0.5 + 0.3 * progress
        lr = max(1e-10, 5.0 / np.sqrt(n) * (1.0 - 0.5 * progress))
        
        it += 1
        if verbose and it % 100 == 0:
            print(f"    ASG it{it}: obj={best_obj:.8f}, K={K}, lr={lr:.4f}, el={elapsed:.0f}s")
    
    return best_a

def multiplicative_update(a, time_budget=100.0, verbose=False, seed=42):
    """
    Multiplicative water-filling algorithm with annealing.
    Reduces mass at positions contributing to hot spots.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    lr = 1.0 / np.sqrt(n)
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Anneal the threshold fraction
        frac = 0.5 + 0.4 * progress  # from 0.5 to 0.9
        threshold = frac * max_conv
        
        # Find hot positions
        hot_mask = conv >= threshold
        n_hot = np.sum(hot_mask)
        if n_hot == 0:
            break
        
        # Compute pressure at each position
        pressure = np.zeros(n)
        hot_idx = np.where(hot_mask)[0]
        
        # For each hot position k, position i contributes a[k-i] to c[k]
        # Use FFT for efficiency
        # pressure[i] = sum_{k in hot} a[k-i]
        # This is a cross-correlation
        N = 1
        while N < 2 * n:
            N <<= 1
        hot_vec = np.zeros(N)
        hot_vec[:len(conv)] = hot_mask.astype(float)
        
        from scipy.fft import fft, ifft
        P = fft(hot_vec, N)
        A = fft(a, N)
        corr = ifft(np.conj(P) * A).real[:n]
        pressure = corr
        
        # Multiplicative update
        a_new = a * np.exp(-lr * pressure / (max_conv + 1e-15))
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                lr *= 0.7
                no_improve = 0
                if lr < 1e-8:
                    break
        
        lr = max(1e-8, lr * 0.999)
        it += 1
    
    return best_a

            break

                best_t = t

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        # Arcsine-like distribution
        x = np.linspace(0.001, 1, n)
        a = 1.0 / np.sqrt(x * (1 - x) + 1e-6)
        a = a / a.sum()
    elif mode == 7:
        # Two-peak
        c1 = rng.integers(0, n // 3)
        c2 = rng.integers(2 * n // 3, n)
        w = max(1, n // 10)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 8:

        c = n // 2
        w = n // 6
        a = np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 9:
        # Geometric
        r = rng.uniform(0.7, 0.99)
        a = r ** np.arange(n)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a = a / a.sum()
    return a

    """
    Fixed-point LP accounting for g*g term in blend.
    Iteratively solves LP with bias from previous g*g.
    """

    a_norm = a / float(np.sum(a))

    # Initial: standard LP
    g_cur, _ = solve_lp_minpeak(a_norm)
    if g_cur is None:
        return None

    A_base = build_conv_matrix(a_norm)

    for fp_iter in range(max_fp_iter):
        # Compute bias from g*g
        gg_conv = compute_conv(g_cur)
        # The effective objective for g in the blend is:
        # max_k (2*t*(1-t)*(a*g)[k] + t^2*(g*g)[k])
        # Approximate by adding t^2*gg_conv[k] as a bias to each constraint
        # min M s.t. 2*t*(1-t)*(a*g)[k] + t^2*gg_conv[k] <= M
        # i.e., (a*g)[k] <= (M - t^2*gg_conv[k]) / (2*t*(1-t))
        # which is equivalent to: weight*(a*g)[k] <= M with weight = 1,
        # and RHS offset.

        coeff = t_blend / (2.0 * (1.0 - t_blend))

        # This is: (a*g)[k] + bias[k] <= M

    stall = 0
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        K = int(K_init + (K_max - K_init) * progress)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]

        # Exponential weighting of peaks
        spread = max_conv * 0.05 * (1.0 - 0.5 * progress)
        if spread < 1e-15:
            weights = np.ones(K) / K
        else:
            weights = np.exp((peak_vals - peak_vals[0]) / spread)
            weights /= weights.sum()

        # Compute weighted subgradient
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            if i_min <= i_max:
                g[i_min:i_max + 1] += w * a[k - i_max:k - i_min + 1][::-1]

        # Center on support
        support = a > 1e-12
        if support.any():
            g[support] -= g[support].mean()

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            no_improve = 0
        else:
            stall += 1
            no_improve += 1
            if stall > 80:
                lr *= 0.6
                v *= 0.3
                stall = 0
            if no_improve > 400:
                # Reheat
                a = best_a * np.exp(rng.normal(0, 0.015, n))
                a = np.maximum(a, 0)
                a /= a.sum()
                v = np.zeros(n)
                lr = 4.0 / np.sqrt(n)
                no_improve = 0

        momentum = 0.5 + 0.3 * progress
        lr = max(1e-10, (4.0 / np.sqrt(n)) * (1.0 - 0.4 * progress))
        it += 1

    return best_a


def multiplicative_update(a, time_budget=80.0, verbose=False, seed=42):
    """
    Multiplicative water-filling with annealed threshold.
    Reduces mass at positions contributing to hot spots via FFT cross-correlation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.5 / np.sqrt(n)
    no_improve = 0
    it = 0

    N = 1
    while N < 2 * n:
        N <<= 1
    A_fac = fft(a, N)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Anneal threshold fraction
        frac = 0.4 + 0.5 * progress
        threshold = frac * max_conv

        hot_mask = (conv >= threshold).astype(float)
        if np.sum(hot_mask) == 0:
            break

        hot_fac = fft(hot_mask, N)
        # Cross-correlation: pressure[i] = sum_k hot[k] * a[k-i]
        # = IFFT(conj(hot_fac) * A_fac)[i]
        corr = ifft(np.conj(hot_fac) * A_fac).real[:n]
        pressure = np.maximum(corr, 0)

        a_new = a * np.exp(-lr * pressure / (max_conv + 1e-15))
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            A_fac = fft(a, N)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 150:
                lr *= 0.65
                no_improve = 0
                if lr < 1e-9:
                    break

        lr = max(1e-9, lr * 0.9995)
        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [max(1, n // 8), max(1, n // 8)]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
    elif mode == 6:
        x = np.linspace(0.001, 1, n)
        a = 1.0 / np.sqrt(x * (1 - x) + 1e-6)
    elif mode == 7:
        c1 = rng.integers(0, max(1, n // 3))
        c2 = rng.integers(2 * n // 3, n)
        w = max(1, n // 10)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
    elif mode == 8:
        c = n // 2
        w = max(1, n // 6)
        a = np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
    elif mode == 9:
        r = rng.uniform(0.7, 0.99)
        a = r ** np.arange(n)
    elif mode == 10:
        # Arcsine on [0, n]
        x = np.arange(n) / max(1, n - 1)
        x = np.clip(x, 0.001, 0.999)
        a = 1.0 / np.sqrt(x * (1 - x))
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 0:
        a = a / s
    return a


def local_search_combo(a, time_budget, seed=42):
    """Run a combination of local search methods on a."""
    remaining = time_budget
    # LP first (most effective)
    lp_budget = min(remaining * 0.5, 80.0)
    if lp_budget > 10:
        a = lp_polish_intensive(a, time_budget=lp_budget, seed=seed)
        remaining = time_budget - (80.0 if lp_budget == 80.0 else lp_budget)
        remaining = max(remaining, 0)

    # Subgradient
    sg_budget = min(remaining * 0.5, 40.0)
    if sg_budget > 5:
        a = annealed_subgradient(a, time_budget=sg_budget, seed=seed + 1)

    # Multiplicative
    mu_budget = min(remaining * 0.5, 20.0)
    if mu_budget > 5:
        a = multiplicative_update(a, time_budget=mu_budget, seed=seed + 2)

    return a


def propose_candidate(budget_s=1500):
    """
    Main search: LP polishing with fixed-point, quadratic line search,
    annealed subgradient, multiplicative updates, systematic n-sweep,
    and diverse restarts with local search.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(42)

    # Load parent
    n_init = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_init = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")

    # === Phase 1: Quick warmup (multiplicative + subgradient) ===
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.10, 100.0)
    if p1_budget > 15:
        print(f"\n=== Phase 1: Warmup ({p1_budget:.0f}s) ===")
        a1 = multiplicative_update(best_seq, time_budget=p1_budget * 0.5)
        o1 = compute_obj(a1)
        if o1 < best_score:
            best_score = o1
            best_seq = a1.copy()
            print(f"  MU: {best_score:.8f} **")
        a1 = annealed_subgradient(best_seq, time_budget=p1_budget * 0.5, K_init=5, K_max=20)
        o1 = compute_obj(a1)
        if o1 < best_score:
            best_score = o1
            best_seq = a1.copy()
            print(f"  ASG: {best_score:.8f} **")

    # === Phase 2: Intensive LP Polishing ===
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 400.0)
    if p2_budget > 30:
        print(f"\n=== Phase 2: Intensive LP ({p2_budget:.0f}s) ===")
        a2 = lp_polish_intensive(best_seq, time_budget=p2_budget, verbose=True, seed=42)
        o2 = compute_obj(a2)
        if o2 < best_score:
            best_score = o2
            best_seq = a2.copy()
            print(f"  Phase2 LP: {best_score:.8f} **")
        else:
            print(f"  Phase2 LP: {o2:.8f}")

    # === Phase 3: Subgradient + Multiplicative refinement ===
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 100.0)
    if p3_budget > 15:
        print(f"\n=== Phase 3: SG + MU ({p3_budget:.0f}s) ===")
        a3 = annealed_subgradient(best_seq, time_budget=p3_budget * 0.6, K_init=5, K_max=30, seed=123)
        o3 = compute_obj(a3)
        if o3 < best_score:
            best_score = o3
            best_seq = a3.copy()
            print(f"  ASG: {best_score:.8f} **")
        remaining = deadline - time.monotonic()
        p3b = min(remaining, 40.0)
        if p3b > 10:
            a3b = multiplicative_update(best_seq, time_budget=p3b, seed=456)
            o3b = compute_obj(a3b)
            if o3b < best_score:
                best_score = o3b
                best_seq = a3b.copy()
                print(f"  MU: {best_score:.8f} **")

    # === Phase 4: Systematic n-sweep ===
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    print(f"\n=== Phase 4: n-Sweep ===")

    n_candidates = []
    for delta in [-300, -200, -100, 100, 200, 300, 500]:
        n_try = n_current + delta
        if 400 <= n_try <= 2500:
            n_candidates.append(n_try)
    # Also try some absolute values
    for n_abs in [500, 600, 700, 900, 1000, 1100, 1200, 1500, 2000]:
        if 400 <= n_abs <= 2500 and abs(n_abs - n_current) > 80:
            n_candidates.append(n_abs)

    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        print(f"  n={n_try} (rem={remaining:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)

        # Quick local search
        alloc = min(remaining * 0.2, 60.0)
        if alloc > 15:
            a_try = local_search_combo(a_try, alloc, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # === Phase 5: Diverse restarts ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 5: Diverse Restarts ===")

    for r in range(12):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        n_div = len(best_seq)
        mode = r % 11

        if mode <= 2:
            # Small perturbation
            scale = 0.008 + 0.004 * r
            a_div = best_seq * np.exp(rng.normal(0, scale, n_div))
            a_div = np.maximum(a_div, 0)
            a_div /= a_div.sum()
        elif mode <= 5:
            # Blend diverse with best
            a_rand = diverse_init(n_div, rng, mode=mode)
            blend = 0.2 + 0.05 * (mode - 3)
            a_div = (1 - blend) * best_seq + blend * a_rand
            a_div = np.maximum(a_div, 0)
            a_div /= a_div.sum()
        else:
            # Pure diverse init (risky but might find new basin)
            a_div = diverse_init(n_div, rng, mode=mode)
            # Blend heavily with best to keep quality
            a_div = 0.7 * best_seq + 0.3 * a_div
            a_div = np.maximum(a_div, 0)
            a_div /= a_div.sum()

        alloc = min(remaining * 0.22, 55.0)
        if alloc > 15:
            a_div = local_search_combo(a_div, alloc, seed=500 + r)
            obj_div = compute_obj(a_div)
            if obj_div < best_score:
                best_score = obj_div
                best_seq = a_div.copy()
                print(f"  R{r} m={mode}: {best_score:.8f} **")
            else:
                print(f"  R{r} m={mode}: {obj_div:.8f}")

    # === Phase 6: Final intensive polish ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 6: Final Polish (rem={remaining:.0f}s) ===")

    if remaining > 40:
        # Long LP
        lp_budget = min(remaining * 0.45, 200.0)
        if lp_budget > 20:
            a_f = lp_polish_intensive(best_seq, time_budget=lp_budget, verbose=True, seed=9999)
            o_f = compute_obj(a_f)
            if o_f < best_score:
                best_score = o_f
                best_seq = a_f.copy()
                print(f"  Final LP: {best_score:.8f} **")

        # Short subgradient
        remaining = deadline - time.monotonic()
        sg_budget = min(remaining * 0.25, 50.0)
        if sg_budget > 10:
            a_f = annealed_subgradient(best_seq, time_budget=sg_budget, K_init=10, K_max=40, seed=7777)
            o_f = compute_obj(a_f)
            if o_f < best_score:
                best_score = o_f
                best_seq = a_f.copy()
                print(f"  Final SG: {best_score:.8f} **")

        # Short multiplicative
        remaining = deadline - time.monotonic()
        mu_budget = min(remaining * 0.2, 30.0)
        if mu_budget > 8:
            a_f = multiplicative_update(best_seq, time_budget=mu_budget, seed=8888)
            o_f = compute_obj(a_f)
            if o_f < best_score:
                best_score = o_f
                best_seq = a_f.copy()
                print(f"  Final MU: {best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
