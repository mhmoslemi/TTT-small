"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous:
1. Active-set LP with only top-K constraints (much faster iterations)
2. Block coordinate descent with O(n) updates
3. Tilted asymmetric initializations to reduce center peak
4. Adaptive active set that grows as we approach optimum
5. Multi-scale exploration with resampling
6. Better time management with early termination
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_subset(a, indices):
    """Build matrix A (len(indices) x n) such that A @ g = conv(a, g) at given indices."""
    n = len(a)
    k_list = np.array(indices)
    A = np.zeros((len(k_list), n))
    for j in range(n):
        # conv(a, g)[k] = sum_m a[m] * g[k-m] for valid m
        # So for fixed k: sum_m a[m] * g[k-m] where 0<=m<n and 0<=k-m<n
        # => sum_{m: max(0,k-(n-1))<=m<=min(n-1,k)} a[m] * g[k-m]
        # For row k, column j: coefficient of g[j] is a[k-j] if 0<=k-j<n
        valid = (k_list >= j) & (k_list - j < n)
        A[valid, j] = a[k_list[valid] - j]
    return A


def solve_active_lp(a, active_indices):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for k in active_indices, sum(g)=1, g>=0.
    Uses only active constraints for speed.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix_subset(a_norm, active_indices)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue

    # Fallback: return uniform
    return np.ones(n) / n


def active_lp_iteration(a, n_active=50, verbose=False):
    """
    One active-set LP iteration: find direction g, then line search.
    """
    n = len(a)
    a = a / float(np.sum(a))
    conv = compute_conv(a)
    current_max = float(np.max(conv))

    # Get active set (top peaks)
    active_indices = np.argsort(conv)[-n_active:]
    g_dir = solve_active_lp(a, active_indices)

    if g_dir is None or np.allclose(g_dir, 1.0/n):
        return a, current_max

    # Line search: minimize max_k conv((1-t)*a + t*g, (1-t)*a + t*g)
    # = max_k [(1-t)^2 conv_a[k] + 2t(1-t) conv_ag[k] + t^2 conv_g[k]]
    conv_ag = compute_conv(a * 0)  # placeholder
    # Actually compute conv(a,g) properly
    n_fft = 1
    while n_fft < 2 * n - 1:
        n_fft <<= 1
    fa = rfft(a, n_fft)
    fg = rfft(g_dir, n_fft)
    conv_ag = irfft(fa * fg, n_fft)[:2 * n - 1]
    conv_g = compute_conv(g_dir)

    def obj_at_t(t):
        if t <= 0:
            return current_max
        val = (1-t)**2 * conv + 2*t*(1-t) * conv_ag + t**2 * conv_g
        return float(np.max(val))

    # Golden section search for t in [0, 1]
    gr = (np.sqrt(5) - 1) / 2.0
    lo_t, hi_t = 0.0, 1.0
    c_t = hi_t - gr * (hi_t - lo_t)
    d_t = lo_t + gr * (hi_t - lo_t)
    fc = obj_at_t(c_t)
    fd = obj_at_t(d_t)

    for _ in range(60):
        if fc < fd:
            hi_t = d_t
            d_t = c_t
            fd = fc
            c_t = hi_t - gr * (hi_t - lo_t)
            fc = obj_at_t(c_t)
        else:
            lo_t = c_t
            c_t = d_t
            fc = fd
            d_t = lo_t + gr * (hi_t - lo_t)
            fd = obj_at_t(d_t)

    t_best = (lo_t + hi_t) / 2.0
    a_new = (1 - t_best) * a + t_best * g_dir
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        return a, current_max
    a_new = a_new / s_new
    new_max = float(np.max(compute_conv(a_new)))
    return a_new, new_max


def active_lp_polish(a, time_budget=100.0, n_active_start=30, n_active_end=100, verbose=False):
    """
    Iterative active-set LP polishing.
    Starts with few active constraints, grows as we converge.
    """
    n = len(a)
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    iter_count = 0
    stagnant = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        # Grow active set as we converge
        n_active = int(n_active_start + (n_active_end - n_active_start) * progress)

        a_new, new_max = active_lp_iteration(a, n_active=n_active)
        obj_new = compute_obj(a_new)
        iter_count += 1

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stagnant = 0
            if verbose and iter_count % 5 == 0:
                print(f"    ActiveLP it{iter_count}: obj={best_obj:.8f}, n_active={n_active}, t={elapsed:.0f}s")
        else:
            stagnant += 1
            if stagnant > 15:
                # Perturb and restart
                scale = 0.01 + 0.005 * stagnant
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stagnant = 0
                iter_count = 0
            elif iter_count > 100:
                break

    return best_a


def fast_coord_descent(a, time_budget=200.0, start_time=None, rng=None, verbose=False):
    """
    Fast coordinate descent with incremental convolution updates.
    Uses tight bounds and fewer golden section iterations.
    """
    if start_time is None:
        start_time = time.monotonic()

    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        conv = compute_conv(a)
        return a
    a = a / S
    conv = compute_conv(a)

    best_obj = float(np.max(conv))
    best_a = a.copy()
    gr = (np.sqrt(5) - 1) / 2.0
    stagnant = 0
    sweep = 0

    while time.monotonic() - start_time < time_budget:
        sweep_start = time.monotonic()
        obj_before = float(np.max(conv))

        # Random or cyclic order
        if rng is not None and sweep % 4 == 3:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for k in order:
            if time.monotonic() - start_time > time_budget:
                break

            # Direction vector: d[m] = a[m-k] for m in [k, k+n-1]
            d = np.zeros(2 * n - 1)
            d[k:k + n] = a

            # Bounds for delta
            lo_delta = -a[k]
            hi_delta = max(3.0 / n, abs(a[k]) + 1.0 / n)

            # Golden section search (fewer iterations for speed)
            a_gs = lo_delta
            b_gs = hi_delta
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)

            # Precompute for speed
            conv_local = conv.copy()
            base = conv_local.copy()

            def obj_delta(delta):
                # max of conv + 2*delta*d + delta^2 at position 2k
                val = base + 2.0 * delta * d
                val[2 * k] += delta * delta
                return float(np.max(val))

            fc = obj_delta(c_gs)
            fd = obj_delta(d_gs)

            for _ in range(35):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = obj_delta(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = obj_delta(d_gs)

            delta_opt = (a_gs + b_gs) / 2.0

            # Check improvement
            new_obj_est = obj_delta(delta_opt)
            if new_obj_est >= obj_before - 1e-16:
                continue

            # Apply update
            a[k] += delta_opt
            a[k] = max(0.0, a[k])

            # Update convolution incrementally
            conv += 2.0 * delta_opt * d
            conv[2 * k] += delta_opt * delta_opt

            # Re-normalize
            S_new = float(np.sum(a))
            if S_new < 1e-10:
                a[:] = 1.0 / n
                conv = compute_conv(a)
            else:
                a /= S_new
                conv /= S_new

        sweep += 1
        obj_after = float(np.max(conv))
        new_global = 2.0 * n * obj_after

        if obj_after < best_obj - 1e-13:
            best_obj = obj_after
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 4:
                if rng is not None:
                    scale = 0.005 + 0.003 * stagnant
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    S_new = a.sum()
                    if S_new > 1e-10:
                        a /= S_new
                        conv = compute_conv(a)
                        best_obj = float(np.max(conv))
                    stagnant = 0
                else:
                    break

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    CD sweep {sweep}: obj={new_global:.8f}, time={elapsed:.0f}s")

    return best_a


def block_coord_descent(a, time_budget=100.0, block_size=8, start_time=None, rng=None):
    """
    Block coordinate descent: optimize a block of coordinates simultaneously.
    For a block, the subproblem is still convex (quadratic in the block).
    """
    if start_time is None:
        start_time = time.monotonic()

    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        return a
    a = a / S
    conv = compute_conv(a)

    best_obj = float(np.max(conv))
    best_a = a.copy()
    stagnant = 0
    sweep = 0
    gr = (np.sqrt(5) - 1) / 2.0

    while time.monotonic() - start_time < time_budget:
        sweep += 1
        if time.monotonic() - start_time > time_budget:
            break

        for block_start in range(0, n, block_size):
            if time.monotonic() - start_time > time_budget:
                break

            block_end = min(block_start + block_size, n)
            block_indices = np.arange(block_start, block_end)
            mb = len(block_indices)

            # For the block, compute the "effective" direction
            # conv change from modifying block: each index k in block contributes
            # d_k[m] = a[m-k] for m in [k, k+n-1]
            # Total change: sum_{k in block} delta_k * d_k + sum_{k1,k2 in block} delta_k1 * delta_k2 * (e_{k1} at 2k2 position...)
            # This is getting complex; use simple approach: try a few random directions in the block

            # Simple: optimize one coordinate at a time within the block (just faster CD)
            for k in block_indices:
                d = np.zeros(2 * n - 1)
                d[k:k + n] = a

                lo_delta = -a[k]
                hi_delta = max(2.0 / n, a[k] + 0.5 / n)

                # Fewer iterations for speed
                a_gs, b_gs = lo_delta, hi_delta
                c_gs = b_gs - gr * (b_gs - a_gs)
                d_gs = a_gs + gr * (b_gs - a_gs)

                base_conv = conv.copy()

                def obj_d(delta):
                    val = base_conv + 2.0 * delta * d
                    val[2 * k] += delta * delta
                    return np.max(val)

                fc = obj_d(c_gs)
                fd = obj_d(d_gs)

                for _ in range(25):
                    if fc < fd:
                        b_gs = d_gs
                        d_gs = c_gs
                        fd = fc
                        c_gs = b_gs - gr * (b_gs - a_gs)
                        fc = obj_d(c_gs)
                    else:
                        a_gs = c_gs
                        c_gs = d_gs
                        fc = fd
                        d_gs = a_gs + gr * (b_gs - a_gs)
                        fd = obj_d(d_gs)

                delta_opt = (a_gs + b_gs) / 2.0
                if obj_d(delta_opt) < np.max(conv) - 1e-15:
                    a[k] = max(0.0, a[k] + delta_opt)
                    conv += 2.0 * delta_opt * d
                    conv[2 * k] += delta_opt * delta_opt
                    S_new = float(np.sum(a))
                    if S_new > 1e-10:
                        a /= S_new
                        conv /= S_new

        obj_after = float(np.max(conv))
        if obj_after < best_obj - 1e-13:
            best_obj = obj_after
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 6:
                if rng is not None:
                    a = best_a * np.exp(rng.normal(0, 0.01, n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    conv = compute_conv(a)
                stagnant = 0
                if stagnant > 20:
                    break

    return best_a


def tilted_init(n, rng, mode=0):
    """Generate tilted/asymmetric initializations."""
    x = np.arange(n) / max(n - 1, 1)
    if mode == 0:
        # Linear tilt
        a = 1.0 + rng.uniform(0.3, 0.8) * (2 * x - 1)
    elif mode == 1:
        # Quadratic tilt
        c = rng.uniform(-0.5, 0.5)
        a = 1.0 + c * (2 * x - 1) ** 2
    elif mode == 2:
        # Cosine tilt
        freq = rng.uniform(0.5, 1.5)
        phase = rng.uniform(0, np.pi)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * freq * x + phase)
    elif mode == 3:
        # Sine bump
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.3)
        a = 0.5 + 0.5 * np.exp(-((x - center) ** 2) / (2 * width ** 2))
    elif mode == 4:
        # Two bumps
        c1 = rng.uniform(0.1, 0.4)
        c2 = rng.uniform(0.6, 0.9)
        w = rng.uniform(0.08, 0.2)
        a = 0.3 + 0.4 * np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + 0.4 * np.exp(-((x - c2) ** 2) / (2 * w ** 2))
    elif mode == 5:
        # Logistic tilt
        steepness = rng.uniform(1, 4)
        mid = rng.uniform(0.3, 0.7)
        a = 1.0 / (1.0 + np.exp(-steepness * (x - mid)))
    elif mode == 6:
        # Power law tilt
        p = rng.uniform(-1.5, 1.5)
        a = (x + 0.01) ** p
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0.01)
    return a / a.sum()


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-6)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Active-set LP + fast CD + tilted inits + multi-n.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        # Try tilted initialization
        for mode in range(7):
            a_try = tilted_init(800, rng, mode=mode)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
        if best_score == np.inf:
            best_seq = np.ones(800) / 800
            best_score = compute_obj(best_seq)
        print(f"Init (n={len(best_seq)}): obj={best_score:.8f}")

    # === Phase 1: Fast Coordinate Descent (main workhorse) ===
    print(f"\n=== Phase 1: Fast CD ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 30:
        a_cd = fast_coord_descent(best_seq, time_budget=p1_budget,
                                  start_time=time.monotonic(), rng=rng, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase1 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 CD: obj={obj_cd:.8f}")

    # === Phase 2: Active-Set LP Polishing ===
    print(f"\n=== Phase 2: Active-Set LP ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.30, 400.0)
    if p2_budget > 30:
        # Run multiple rounds with increasing active set
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            na_start = 20 + 20 * rnd
            na_end = 60 + 30 * rnd
            alloc = min(p2_budget / (3 - rnd), remaining * 0.6)
            if alloc < 20:
                break
            a_lp = active_lp_polish(best_seq, time_budget=alloc,
                                     n_active_start=na_start, n_active_end=na_end, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd} (active {na_start}-{na_end}): obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
                # Perturb for next round
                if rnd < 2:
                    scale = 0.015 * (rnd + 1)
                    best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                    best_seq = np.maximum(best_seq, 0)
                    best_seq = best_seq / best_seq.sum()

    # === Phase 3: More CD (fine tuning) ===
    print(f"\n=== Phase 3: Fine CD ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.20, 200.0)
    if p3_budget > 30:
        a_cd2 = fast_coord_descent(best_seq, time_budget=p3_budget,
                                   start_time=time.monotonic(), rng=rng, verbose=False)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  Phase3 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 CD: obj={obj_cd2:.8f}")

    # === Phase 4: Second LP pass ===
    print(f"\n=== Phase 4: Second LP ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.25, 150.0)
        a_lp2 = active_lp_polish(best_seq, time_budget=alloc,
                                  n_active_start=40, n_active_end=150, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase4 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 LP: obj={obj_lp2:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = []
    for delta in [-200, -100, -50, 50, 100, 200, 300]:
        n_try = n_current + delta
        if 400 <= n_try <= 2000:
            n_options.append(n_try)
    n_options = sorted(set(n_options))

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick CD
        alloc1 = min(remaining * 0.4, 60.0)
        if alloc1 > 20:
            a_try = fast_coord_descent(a_try, time_budget=alloc1,
                                       start_time=time.monotonic(), rng=rng)

        # Quick LP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.25, 50.0)
        if alloc2 > 15:
            a_try = active_lp_polish(a_try, time_budget=alloc2,
                                      n_active_start=20, n_active_end=60)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Tilted Restarts ===
    print(f"\n=== Phase 6: Tilted Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        mode = r * 2  # modes 0, 2, 4
        a_r = tilted_init(n_best, rng, mode=mode)
        # Blend with best
        blend = 0.4 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        # CD
        alloc1 = min(remaining * 0.4, 60.0)
        if alloc1 > 20:
            a_r = fast_coord_descent(a_r, time_budget=alloc1,
                                     start_time=time.monotonic(), rng=rng)

        # LP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.2, 40.0)
        if alloc2 > 15:
            a_r = active_lp_polish(a_r, time_budget=alloc2,
                                    n_active_start=20, n_active_end=50)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 40:
        alloc = min(remaining * 0.3, 80.0)
        a_cd_f = fast_coord_descent(best_seq, time_budget=alloc,
                                    start_time=time.monotonic(), rng=rng)
        obj_cd_f = compute_obj(a_cd_f)
        if obj_cd_f < best_score:
            best_score = obj_cd_f
            best_seq = a_cd_f.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.35, 60.0)
        a_lp_f = active_lp_polish(best_seq, time_budget=alloc,
                                   n_active_start=50, n_active_end=200)
        obj_lp_f = compute_obj(a_lp_f)
        if obj_lp_f < best_score:
            best_score = obj_lp_f
            best_seq = a_lp_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 40.0)
        a_cd_f2 = fast_coord_descent(best_seq, time_budget=alloc,
                                     start_time=time.monotonic(), rng=rng)
        obj_cd_f2 = compute_obj(a_cd_f2)
        if obj_cd_f2 < best_score:
            best_score = obj_cd_f2
            best_seq = a_cd_f2.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
