def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None

def smoothed_gradient_fft(a, tau):
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    while N < 3 * n:
        N <<= 1
    
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv

    n = len(a)
    pert = np.zeros(n)

    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))

        return a.copy()

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Intensive LP-based direction finding with quadratic line search,
log-sum-exp smoothed gradient descent for local refinement, randomized LP
perturbations for escaping stagnation, coordinate sparsification, and
systematic multi-n exploration with resampling.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, random_vec=None):
    """
    Solve LP: min M (+ eps * rv'g if random_vec given)
    s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns optimal g or None.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    if random_vec is not None:
        objective = cp.Minimize(M + 0.001 * random_vec @ g)
    else:
        objective = cp.Minimize(M)
    
    prob = cp.Problem(objective, constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of autocorrelation using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1

    P = fft(p, N)
    A_fft = fft(a, N)
    g_full = ifft(P * np.conj(A_fft)).real
    g = 2.0 * g_full[:n]
    return g, conv


def lgd_refine(a, time_budget, tau_init=30.0, tau_end=600.0):
    """Nesterov-accelerated log-sum-exp smoothed gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 2.5 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    grad_norms = []

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * (progress ** 1.3)

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)
        g_norm = float(np.linalg.norm(g))
        grad_norms.append(g_norm)
        if len(grad_norms) > 300:
            grad_norms = grad_norms[-300:]

        # Adaptive LR
        if len(grad_norms) > 20:
            avg_gn = float(np.mean(grad_norms))
            if avg_gn > 1e-15:
                adaptive_lr = 0.8 / (avg_gn + 1e-10)
                adaptive_lr = min(max(adaptive_lr, 0.1 * base_lr), 3.0 * base_lr)
                lr = 0.6 * lr + 0.4 * adaptive_lr

        # Barzilai-Borwein
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                bb_lr = min(max(bb_lr, 1e-8), 4.0 / np.sqrt(n))
                lr = 0.5 * lr + 0.5 * bb_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = y.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)

        if it - best_it > 4000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it

    return best_a


def fourier_perturb(a, rng, strength=0.03):
    """Add low-frequency Fourier modes as structured perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 6):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def antisymmetric_perturb(a, rng, strength=0.02):
    """Add antisymmetric perturbation (preserves mirror-symmetric component)."""
    n = len(a)
    r = rng.standard_normal(n)
    r_anti = r - r[::-1]
    a_new = a + strength * r_anti
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n == 1 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def quadratic_line_search(a, g, obj0, max_iter=15):
    """
    Quadratic interpolation line search along direction (g - a).
    Returns (best_t, best_obj, best_a).
    """
    best_t = 0.0
    best_obj = obj0
    best_a = a.copy()

    # Initial evaluations
    t1, f1 = 0.0, obj0
    t2, f2 = 0.3, _eval_blend(a, g, 0.3)
    if f2 < best_obj:
        best_t, best_obj, best_a = 0.3, f2, _blend(a, g, 0.3)

    t_guess = 0.6
    for _ in range(max_iter):
        t3, f3 = t_guess, _eval_blend(a, g, t_guess)
        if f3 < best_obj:
            best_t, best_obj, best_a = t3, f3, _blend(a, g, t3)

        # Quadratic interpolation
        if abs(t3 - t2) < 1e-6 or abs(t2 - t1) < 1e-6:
            break
        denom = (t1 - t3) * (t1 - t2) * (t2 - t3)
        if abs(denom) < 1e-15:
            break
        A_num = (t3 - t2) * f1 - (t3 - t1) * f2 + (t2 - t1) * f3
        B_num = -(t3 * f1 - t1 * f3) - (t3 * f2 - t2 * f1) + (t2 * f3 - t1 * f2)
        t_new = (B_num * t1 + A_num * (t1 + t2 + t3)) / (2.0 * denom) if abs(denom) > 1e-15 else (t1 + t2 + t3) / 3.0
        t_new = max(0.01, min(1.0, t_new))

        if abs(t_new - t_guess) < 1e-4:
            break
        t1, t2, t_guess = t2, t3, t_new
        f1, f2 = f2, f3

    return best_t, best_obj, best_a


def _blend(a, g, t):
    a_new = (1.0 - t) * a + t * g
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _eval_blend(a, g, t):
    a_new = _blend(a, g, t)
    return compute_obj(a_new)


def sparsify_and_reoptimize(a, rng, keep_frac=0.7):
    """Zero out smallest coordinates, renormalize, then re-solve LP."""
    n = len(a)
    k = max(10, int(n * keep_frac))
    indices = np.argsort(a)[-k:]
    a_sparse = np.zeros(n)
    a_sparse[indices] = a[indices]
    a_sparse = a_sparse / a_sparse.sum()
    return a_sparse

    """
    n = len(a)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        a = np.exp(-((x - 0.3) ** 2) / 0.05) + np.exp(-((x - 0.7) ** 2) / 0.05)
    elif mode == 3:
        a = 1.0 + 0.5 * np.sin(2 * np.pi * 3 * x + rng.uniform(0, 6.28))
    elif mode == 4:
        a = 1.0 + 0.4 * np.cos(2 * np.pi * 5 * x + rng.uniform(0, 6.28))
    elif mode == 5:

    elif mode == 6:
        a = (1 - x) ** rng.uniform(0.5, 3.0)
    elif mode == 7:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 7 * x) + 0.2 * np.cos(2 * np.pi * 11 * x)
    elif mode == 8:
        a = np.maximum(1 - 4 * np.abs(x - 0.5), 0.01)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()

        n_try = n_current + dn
        if 400 <= n_try <= 2000:
            n_options.append(n_try)
    for n_fixed in [600, 700, 850, 950, 1000, 1100, 1200, 1400]:
        if 400 <= n_fixed <= 2000 and all(abs(n_fixed - x) > 30 for x in n_options):
            n_options.append(n_fixed)

    n_options = sorted(set(n_options))
    # Limit to at most 6 options, prioritize those close to current
    n_options = sorted(n_options, key=lambda x: abs(x - n_current))[:6]
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 70:
            break
        if n_try == n_current:
            continue

        a_try = resample(best_seq, n_try)

        # Quick LP
        g_lp = solve_lp(a_try)
        if g_lp is not None:
            t_best, obj_best, a_best = quadratic_line_search(a_try, g_lp, compute_obj(a_try), max_iter=10)
            if obj_best < compute_obj(a_try):
                a_try = a_best

        # Short GD
        alloc = min(remaining * 0.12, 45.0)
        if alloc > 15:
            a_try = lgd_refine(a_try, time_budget=alloc, tau_init=30, tau_end=500)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_use = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        mode = 1 + r % 8
        a_r = diverse_init(n_use, rng, mode=mode)
        # Blend with current best
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.15, 80.0)
        if alloc > 20:
            half = alloc * 0.5
            a_r = lgd_refine(a_r, time_budget=half, tau_init=20, tau_end=300)
            remaining = deadline - time.monotonic()
            lp_alloc = min(remaining * 0.08, 30.0)
            if lp_alloc > 10:
                rng_r = np.random.default_rng(seed + 200 + r)
                a_r = lp_polish_intensive(a_r, time_budget=lp_alloc, rng=rng_r, max_rounds=10)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau GD
    if remaining > 35:
        alloc = min(remaining * 0.2, 80.0)
        a_gd = lgd_refine(best_seq, time_budget=alloc, tau_init=100, tau_end=1500)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Multiple LP passes
    remaining = deadline - time.monotonic()
    for lp_rnd in range(4):
        if remaining <= 15:
            break
        alloc = min(remaining * 0.25, 60.0)
        rng_lp = np.random.default_rng(seed + 500 + lp_rnd * 50)
        a_lp = lp_polish_intensive(best_seq, time_budget=alloc, rng=rng_lp, max_rounds=15)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP {lp_rnd}: obj={best_score:.8f} **")
        else:
            if lp_rnd < 3:
                best_seq = fourier_perturb(best_seq, rng, strength=0.008)
            else:
                break

    # One more GD if time permits
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.3, 50.0)
        a_gd = lgd_refine(best_seq, time_budget=alloc, tau_init=200, tau_end=2000)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Ultra-final GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
