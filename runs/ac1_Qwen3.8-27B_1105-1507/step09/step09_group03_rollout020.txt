import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft, next_fast_len

import math

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = next_fast_len(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix_vec(a, n):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def solve_lp_direction(a, warm_start=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M, dual_vars).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix_vec(a_norm, n)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    if warm_start is not None:
        g.value = warm_start
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:

            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Get dual variables
                    dual = None
                    for c in constraints:
                        if c.dual_value is not None:
                            dual = c.dual_value
                    return g_val, float(M.value), dual
        except Exception:
            continue
    return None, None, None

    # = c[k] + t(2d[k]-2c[k]) + t^2(e[k]-2d[k]+c[k])
    
    # We want to minimize max_k of these quadratics over t in [0,1]
    # This is a 1D convex optimization (max of convex functions)
    
    # Use ternary search (since it's convex)
    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        
        # Evaluate max at m1 and m2
        t1 = m1
        t2 = m2
        
        val1 = np.max((1-t1)**2 * c + 2*t1*(1-t1)*d + t1**2 * e)
        val2 = np.max((1-t2)**2 * c + 2*t2*(1-t2)*d + t2**2 * e)
        
        if val1 < val2:
            hi = m2
        else:
            lo = m1
    
    t_opt = (lo + hi) / 2
    return t_opt

def spectral_perturbation(a, rng, alpha=0.1):
    """
    Add a small phase perturbation in frequency domain to escape local minima.
    """
    n = len(a)
    N = next_fast_len(2 * n)
    A = fft(a, N)
    
    # Add random phase perturbation
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + alpha * rng.normal(0, 1, N)
    
    A_new = mag * np.exp(1j * phase_pert)
    a_new = ifft(A_new).real[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new

def lp_polish(a, time_budget=100.0, verbose=False):
    """

    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    it = 0
    stall = 0
    warm_g = None
    
    while time.monotonic() - start < time_budget:
        # Solve LP for direction
        g, M_lp, dual = solve_lp_direction(a, warm_start=warm_g)
        if g is None:
            break
        
        # Exact line search
        t_opt = exact_line_search(a, g)
        
        # Try the exact t and a few nearby values
        candidates = [t_opt, max(0, t_opt - 0.1), min(1, t_opt + 0.1), 
                      0.5 * t_opt, min(1.0, 1.5 * t_opt), 1.0]
        
        best_t = t_opt
        best_local_obj = best_obj
        
        for t in candidates:

            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_t = t
        
        a_new = (1 - best_t) * a + best_t * g
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        it += 1
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            warm_g = g.copy()
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.4f}, M_lp={M_lp:.6f}")

                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0

def two_stage_lp(a, time_budget=100.0, verbose=False):
    """
    Two-stage LP: first min-max, then min-sum-active.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(7777)
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Stage 1: min max direction
        g1, M1, _ = solve_lp_direction(a)
        if g1 is None:
            break
        
        # Find active constraints
        conv_ag = build_conv_matrix_vec(a / float(np.sum(a)), n) @ g1
        max_val = float(np.max(conv_ag))
        tol = 0.005 * max_val
        active = np.where(conv_ag >= max_val - tol)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(conv_ag)])
        
        # Stage 2: min sum at active
        A = build_conv_matrix_vec(a / float(np.sum(a)), n)
        A_active = A[active, :]
        
        g2_var = cp.Variable(n, nonneg=True)
        constraints2 = [cp.sum(g2_var) == 1]
        obj2 = cp.sum(A_active @ g2_var)
        prob2 = cp.Problem(cp.Minimize(obj2), constraints2)
        
        g2 = None
        for solver in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS]:
            try:
                prob2.solve(solver=solver)
                if g2_var.value is not None and prob2.status in ('optimal', 'optimal_inaccurate'):
                    g2 = np.maximum(g2_var.value, 0)
                    s2 = float(np.sum(g2))
                    if s2 > 1e-10:
                        g2 = g2 / s2
                    break
            except:
                continue
        
        # Choose direction
        if g2 is not None:
            # Blend
            g_blend = 0.5 * g1 + 0.5 * g2
        else:
            g_blend = g1
        
        # Line search
        t_opt = exact_line_search(a, g_blend)
        a_new = (1 - t_opt) * a + t_opt * g_blend
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        it += 1
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if verbose and it % 5 == 0:
                print(f"    2SLP it{it}: obj={best_obj:.8f}, |S|={len(active)}")
        else:
            # Perturb
            scale = 0.02 + 0.005 * min(it, 20)
            a = best_a * np.exp(rng.normal(0, min(scale, 0.1), n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            if it > 50:
                break
    
    return best_a

def block_cd(a, time_budget=60.0, verbose=False):
    """
    Block coordinate descent: optimize groups of coordinates together.
    For each active peak, find the best pair (src, dst) to redistribute mass.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Top 3 peaks
        peak_indices = np.argsort(conv)[::-1][:3]
        
        improved = False
        for k in peak_indices:
            # Find contributing pairs
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            
            best_improvement = 0
            best_move = None
            
            # For the top contributing pair, try moving mass
            for i in range(i_start, i_end + 1):
                j = k - i
                if j < 0 or j >= n:
                    continue
                contrib = a[i] * a[j]
                if contrib < 0.01 * conv[k]:
                    continue
                
                # Try moving a small amount from the larger of a[i], a[j] to a random position
                src = i if a[i] >= a[j] else j
                if a[src] < 1e-10:
                    continue
                
                # Try multiple targets
                for _ in range(5):
                    tgt = int(rng.integers(0, n))
                    if tgt == src:
                        continue
                    
                    for delta_frac in [0.01, 0.03, 0.05, 0.1]:
                        delta = delta_frac * a[src]
                        a_try = a.copy()
                        a_try[src] -= delta
                        a_try[tgt] += delta
                        if a_try[src] < 0:
                            continue
                        a_try = a_try / a_try.sum()
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - best_improvement - 1e-14:
                            best_improvement = best_obj - obj_try
                            best_a = a_try.copy()
                            a = a_try
                            best_obj = obj_try
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Bi-modal
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        # Comb pattern
        a = np.zeros(n)
        a[::2] = 1.0
        a[::3] += 0.5
        a = a / a.sum()
    elif mode == 5:
        # Decaying
        a = np.exp(-np.arange(n) / (n * 0.3))
        a = a / a.sum()
    elif mode == 6:
        # Random smooth
        a = np.ones(n)
        for _ in range(3):
            center = int(rng.integers(0, n))
            width = int(rng.integers(n // 10, n // 3))
            a += 0.5 * np.exp(-((np.arange(n) - center) ** 2) / (2 * width ** 2))
        a = a / a.sum()
    else:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    return np.maximum(a, 0)

def momentum_lp(a, time_budget=100.0, momentum=0.5, verbose=False):
    """LP with momentum (heavy ball)."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        g, M_lp, _ = solve_lp_direction(a)
        if g is None:
            break
        
        # Momentum step
        v = momentum * v + (g - a)
        
        # Try different step sizes
        found = False
        for scale in [1.0, 0.5, 0.25, 0.1]:
            a_try = a + scale * v
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                stall = 0
                found = True
                break
        
        it += 1
        if not found:
            stall += 1
            v *= 0.5
            if stall > 30:
                a = best_a * np.exp(np.random.default_rng(it).normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                v = np.zeros(n)
                stall = 0
                if it > 100:
                    break
        
        # Gradually increase momentum
        momentum = min(0.7, 0.5 + 0.002 * it)
        
        if verbose and it % 10 == 0:
            print(f"    MOME it{it}: obj={best_obj:.8f}")
    
    return best_a

Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based direction finding with exact 1D minimization (ternary search on max-of-quadratics),
two-stage active-set LP, momentum-accelerated LP, spectral-domain perturbation for escaping

Key innovations: exact line search exploiting the quadratic structure along LP directions,

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft, next_fast_len
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = next_fast_len(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_vec(a, n):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_direction(a, warm_start=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix_vec(a_norm, n)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

    if warm_start is not None:
        g.value = warm_start

    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(M.value)
        except Exception:
            continue
    return None, None


def exact_line_search(a, g):
    """
    Minimize max_k [(1-t)^2 c_k + 2t(1-t) d_k + t^2 e_k] over t in [0,1].
    Uses ternary search since the function is convex in t (max of quadratics).
    """
    n = len(a)
    c = compute_conv(a)
    A = build_conv_matrix_vec(a, n)
    d = A @ g
    e = compute_conv(g)

    lo, hi = 0.0, 1.0
    for _ in range(40):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0

        val1 = np.max((1 - m1) ** 2 * c + 2 * m1 * (1 - m1) * d + m1 ** 2 * e)
        val2 = np.max((1 - m2) ** 2 * c + 2 * m2 * (1 - m2) * d + m2 ** 2 * e)

        if val1 < val2:
            hi = m2
        else:
            lo = m1

    return (lo + hi) / 2.0


def spectral_perturbation(a, rng, alpha=0.1):
    """Add small phase perturbation in frequency domain to escape local minima."""
    n = len(a)
    N = next_fast_len(2 * n)
    A = fft(a, N)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + alpha * rng.normal(0, 1, N)
    A_new = mag * np.exp(1j * phase_pert)
    a_new = ifft(A_new).real[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new

def block_cd(a, time_budget=60.0, verbose=False):
    """Block coordinate descent targeting top peaks."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_indices = np.argsort(conv)[::-1][:3]

        improved = False
        for k in peak_indices:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)

            for i in range(i_start, i_end + 1):
                j = k - i
                if j < 0 or j >= n:
                    continue
                contrib = a[i] * a[j]
                if contrib < 0.005 * conv[k]:
                    continue

                src = i if a[i] >= a[j] else j
                if a[src] < 1e-10:
                    continue

                for _ in range(4):
                    tgt = int(rng.integers(0, n))
                    if tgt == src:
                        continue
                    for delta_frac in [0.02, 0.05, 0.08]:
                        delta = delta_frac * a[src]
                        a_try = a.copy()
                        a_try[src] -= delta
                        a_try[tgt] += delta
                        if a_try[src] < 0:
                            continue
                        a_try = a_try / a_try.sum()
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 40:
                break

    return best_a
