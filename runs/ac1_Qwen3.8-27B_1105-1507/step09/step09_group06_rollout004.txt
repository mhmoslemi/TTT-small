def exact_subgradient(a, top_k=20):
    """
    Compute exact subgradient using the top_k largest convolution values.
    Returns gradient of sum_k p_k * conv_k where p is uniform over top_k.
    """
    n = len(a)
    conv = compute_conv(a)
    # Get indices of top_k largest conv values
    top_indices = np.argsort(conv)[::-1][:top_k]
    p = np.zeros(len(conv))
    p[top_indices] = 1.0 / top_k
    
    N = 1
    while N < 4 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    grad_full = ifft(A * np.conj(P)).real
    grad = 2.0 * grad_full[:n]
    
    return grad, conv, p


def build_conv_matrix(a):
    """Build matrix A_mat (2n-1 x n) such that A_mat @ g = convolve(a, g)."""
    n = len(a)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A_mat[j:j + r, j] = a[:r]
    return A_mat


def solve_lp_active(a, active_indices, time_limit=5.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for k in active_indices, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    # Build constraint matrix for active indices only
    # conv(a,g)[k] = sum_j A[k,j] * g[j]
    A_rows = []
    for k in active_indices:
        k = int(k)
        row = np.zeros(n)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                row[j] = a_norm[idx]
        A_rows.append(row)
    A_mat = np.array(A_rows)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def subgradient_descent(a, time_budget=60.0, seed=0, verbose=False):
    """
    Subgradient descent with Polyak step size and periodic resets.
    Uses exact subgradient (top-k peaks) and smoothed gradient (adaptive tau).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Estimate optimal step size using Polyak's formula
    # f(a) - f* <= ||a - a*||^2 / (2*step_sum)
    # We'll use a heuristic: step = (f(a) - f_lb) / ||g||^2
    # where f_lb is a lower bound (use 1.0 as a rough lower bound)
    f_lb = 1.4  # rough lower bound for the objective (after normalization)
    
    it = 0
    top_k = min(50, n)
    tau = 50.0  # smoothing parameter
    lr = 0.5 / math.sqrt(n)
    v = np.zeros(n)  # momentum
    beta = 0.7
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        
        # Adaptive tau: start smooth, end sharp
        tau = 20.0 + 200.0 * progress ** 1.5
        # Adaptive top_k: start with more, end with fewer
        top_k = max(10, int(50 * (1 - progress * 0.5)))
        
        # Compute gradient
        g, conv, _ = smoothed_max_gradient(a, tau)
        
        # Project gradient onto tangent space of simplex
        g_centered = g - np.mean(g)
        
        # Nesterov momentum
        v = beta * v - lr * g_centered
        a_new = a + v
        a_new = project_simplex(a_new)
        
        # Evaluate
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            a = a_new
        else:
            # Try a smaller step
            a_half = a + 0.5 * v
            a_half = project_simplex(a_half)
            obj_half = compute_obj(a_half)
            if obj_half < best_obj - 1e-12:
                best_obj = obj_half
                best_a = a_half.copy()
                best_it = it
                a = a_half
            else:
                a = best_a.copy()  # reset to best
                v = np.zeros(n)
                lr *= 0.8
        
        # Adaptive learning rate
        g_norm = np.linalg.norm(g_centered)
        if g_norm > 1e-10:
            polyak_lr = max(0.01, min(5.0, (best_obj - f_lb) / (2.0 * g_norm ** 2 * n)))
            lr = 0.5 * lr + 0.5 * polyak_lr * 0.1
        
        it += 1
        
        # Reset if stuck
        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 0.5 / math.sqrt(n)
            best_it = it
            # Add small perturbation
            noise = rng.normal(0, 0.01, n)
            a = a + noise
            a = project_simplex(a)

        if verbose and it % 1000 == 0:
            print(f"    SGD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a
