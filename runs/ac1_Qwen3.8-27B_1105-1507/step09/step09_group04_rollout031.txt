def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def solve_lp(a, max_time=30.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses multiple solvers with time limit.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_grad_and_obj(a, tau):
    """
    Compute gradient and value of log-sum-exp smoothed max of self-convolution.
    Returns (smoothed_value, gradient_wrta).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = conv - max_conv
    exp_vals = np.exp(tau * shifted)
    p = exp_vals / float(np.sum(exp_vals))
    smooth_val = max_conv + np.log(float(np.sum(exp_vals))) / tau

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g = ifft(P * np.conj(A)).real[:n]
    g *= 2.0
    return smooth_val, g


def compressed_tau_schedule(elapsed, total, tau_init, tau_end, exponent=1.2):
    """Non-linear tau schedule that stays low longer then ramps up."""
    progress = min(1.0, elapsed / total)
    if progress < 0.3:
        return tau_init
    adj = (progress - 0.3) / 0.7
    return tau_init * (tau_end / tau_init) ** (adj ** exponent)


def adaptive_smoothed_descent(a, time_budget=200.0, tau_init=5.0, tau_end=2000.0, verbose=False):
    """
    Smoothed gradient descent with adaptive tau, backtracking line search,
    and periodic frequency-domain refinement.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = float(np.max(compute_conv(a)))
    start = time.monotonic()

    lr = 1.0 / np.sqrt(n)
    it = 0
    best_it = 0
    no_improve = 0
    mom = np.zeros(n)
    beta = 0.7

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        tau = compressed_tau_schedule(elapsed, time_budget, tau_init, tau_end)

        smooth_val, g = smoothed_grad_and_obj(a, tau)
        g_tan = g - np.mean(g)

        # Nesterov-style momentum
        mom = beta * mom - lr * g_tan
        trial = a + mom
        trial = project_simplex(trial)

        # Backtracking line search
        alpha = 1.0
        obj_cur = float(np.max(compute_conv(a)))
        step_accepted = False

        for _ in range(15):
            trial = project_simplex(a + alpha * mom)
            trial_obj = float(np.max(compute_conv(trial)))
            if trial_obj < obj_cur - 1e-14:
                a = trial
                if trial_obj < best_obj - 1e-13:
                    best_obj = trial_obj
                    best_a = a.copy()
                    best_it = it
                    no_improve = 0
                else:
                    no_improve += 1
                step_accepted = True
                break
            alpha *= 0.5

        if not step_accepted:
            no_improve += 1
            lr *= 0.9

        # Periodic spectral equalization
        if it % 200 == 199 and no_improve > 20:
            N_spec = 1
            while N_spec < 2 * n:
                N_spec <<= 1
            A_fft = fft(a, N_spec)
            mag = np.abs(A_fft)
            phase = np.angle(A_fft)
            # Soft equalization: compress dynamic range
            med_mag = np.median(mag[mag > 0]) if np.any(mag > 0) else 1.0
            target = med_mag * 2.0
            factor = np.minimum(1.0, target / (mag + 1e-15))
            factor = 0.85 + 0.15 * factor
            A_new = factor * mag * np.exp(1j * phase)
            a_eq = ifft(A_new).real[:n]
            a_eq = np.maximum(a_eq, 0)
            s_eq = float(np.sum(a_eq))
            if s_eq > 1e-10:
                a_eq = a_eq / s_eq
                obj_eq = float(np.max(compute_conv(a_eq)))
                if obj_eq < best_obj:
                    best_obj = obj_eq
                    best_a = a_eq.copy()
                    a = a_eq.copy()
                    best_it = it
                    no_improve = 0

        # Restart mechanism
        if no_improve > 80:
            a = best_a.copy()
            mom = np.zeros(n)
            lr = 1.0 / np.sqrt(n)
            no_improve = 0
            beta = max(0.3, beta * 0.9)

        it += 1
        if verbose and it % 500 == 0:
            print(f"    SGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.2e}, t={elapsed:.0f}s")

    return best_a


def coord_descent_enhanced(a, time_budget=300.0, rng=None, verbose=False):
    """
    Enhanced coordinate descent with:
    - Gradient-informed search bounds
    - Adaptive precision (fewer GS iterations when close to optimum)
    - Smart anti-stagnation with structured perturbations
    - Periodic full-skip (skip coordinates unlikely to help)
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)

    a = a / S
    conv = compute_conv(a)

    best_obj = float(np.max(conv))

    stagnant = 0
    total_improvements = 0

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget:
            break

        obj_before = float(np.max(conv))

        if rng is not None and sweep % 5 == 4:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        sweep_improvements = 0

        for k in order:
            if time.monotonic() - start > time_budget:
                break

            # Build direction vector
            d = np.zeros(2 * n - 1)
            d[k:k + n] = a

            # Gradient-informed bounds

            mk = m_star - k
            gk = 0.0
            if 0 <= mk < n:
                gk = 2.0 * a[mk]

            # Compute subgradient magnitude to set bounds
            abs_gk = abs(gk)
            lo = -a[k] * 0.999
            # Upper bound: based on current value and gradient direction
            if gk < 0:
                hi = max(a[k] * 0.8, 2.0 / n)
            else:
                hi = max(1.5 / n, a[k] * 0.3)

            # If gradient is very small, use smaller bounds
            if abs_gk < 1.0 / (n * n):
                hi = min(hi, 3.0 / n)
                lo = max(lo, -3.0 / n)

            # Golden section search with adaptive iterations
            a_gs = lo
            b_gs = hi
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)

            def obj_eval(delta):
                val = conv + 2.0 * delta * d
                val[2 * k] += delta * delta
                return float(np.max(val))

            fd = obj_eval(d_gs)

            if stagnant > 10:

                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = obj_eval(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = obj_eval(d_gs)

            delta_opt = (a_gs + b_gs) / 2.0
            new_obj_est = obj_eval(delta_opt)

            if new_obj_est < obj_before - 1e-14:
                a[k] += delta_opt
                a[k] = max(0.0, a[k])
                conv += 2.0 * delta_opt * d
                conv[2 * k] += delta_opt * delta_opt

                S_new = float(np.sum(a))
                if S_new < 1e-10:
                    a[:] = 1.0 / n
                    conv = compute_conv(a)
                else:
                    a /= S_new
                    conv /= (S_new * S_new)
                sweep_improvements += 1

        sweep += 1
        total_improvements += sweep_improvements
        obj_after = float(np.max(conv))

        if obj_after < best_obj - 1e-12:
            best_obj = obj_after
            best_a = a.copy()

                    perturbation = rng.normal(0, 0.003, n) * (1.0 + 2.0 * peak_region[n//4:n//4+n])
                    a = best_a * np.exp(perturbation)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    conv = compute_conv(a)
                    best_obj = float(np.max(conv))
                    stagnant = 0
                else:
                    break

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={2.0*n*obj_after:.8f}, improvements={sweep_improvements}, t={elapsed:.0f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_fixed(a, max_time=20.0):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns optimal g.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def alternating_lp_optimize(a, time_budget=200.0, rng=None, verbose=False):
    """
    Alternating LP optimization with line search blending.
    Alternates: fix a, solve for g; then fix g, solve for a (symmetric).
    Uses golden section on blend parameter.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if rng is None:
        rng = np.random.default_rng(55555)

    it = 0
    gr_ls = (np.sqrt(5) - 1) / 2.0
    consecutive_no_improve = 0

    while time.monotonic() - start < time_budget:
        # Solve LP with current a fixed
        g_lp = solve_lp_fixed(a, max_time=15.0)
        if g_lp is None:
            break

        # Line search on blend parameter t
        t_lo = 0.0
        t_hi = 1.0

        def obj_at_blend(t):
            a_t = (1.0 - t) * a + t * g_lp
            a_t = np.maximum(a_t, 0)
            s_t = float(np.sum(a_t))
            if s_t < 1e-10:
                return np.inf
            a_t = a_t / s_t
            return compute_obj(a_t)

        t1 = t_lo + gr_ls * (t_hi - t_lo)
        t2 = t_hi - gr_ls * (t_hi - t_lo)
        f1 = obj_at_blend(t1)
        f2 = obj_at_blend(t2)

        for _ in range(25):
            if f1 < f2:
                t_hi = t2
                t2 = t1
                f2 = f1
                t1 = t_lo + gr_ls * (t_hi - t_lo)
                f1 = obj_at_blend(t1)
            else:
                t_lo = t1
                t1 = t2
                f1 = f2
                t2 = t_hi - gr_ls * (t_hi - t_lo)
                f2 = obj_at_blend(t2)

        t_opt = (t_lo + t_hi) / 2.0
        obj_opt = obj_at_blend(t_opt)

        it += 1
        if obj_opt < best_obj - 1e-12:
            a_t = (1.0 - t_opt) * a + t_opt * g_lp
            a_t = np.maximum(a_t, 0)
            a_t = a_t / a_t.sum()
            best_obj = obj_opt
            best_a = a_t.copy()
            a = a_t.copy()
            consecutive_no_improve = 0
            if verbose and it % 2 == 0:
                print(f"    AltLP it{it}: obj={best_obj:.8f}, t={t_opt:.3f}")
        else:
            consecutive_no_improve += 1

            a = np.maximum(a, 0)
            a = a / a.sum()

            if consecutive_no_improve > 25:
                break

    return best_a


def spectral_equalization(a, time_budget=80.0, verbose=False):
    """
    Spectral domain shaping: iteratively compress the dynamic range of |A(f)|
    to flatten the self-convolution. Uses soft thresholding in frequency domain.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    a = a / s

    best_a = a.copy()
    best_obj = float(np.max(compute_conv(a)))
    start = time.monotonic()

    N = 1
    while N < 2 * n:
        N <<= 1

    it = 0
    while time.monotonic() - start < time_budget:
        A = fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Compute current convolution peak
        conv_cur = compute_conv(a)
        max_conv = float(np.max(conv_cur))

        # Strategy: reduce frequencies that contribute most to the peak
        m_star = int(np.argmax(conv_cur))

        # Contribution of frequency k to conv[m_star]:
        # conv[m] = (1/N) * sum_k |A[k]|^2 * exp(2*pi*i*k*m/N)
        # We want to reduce |A[k]| where exp(2*pi*i*k*m_star/N) has large positive real part
        # This happens when k*m_star/N is near an integer

        # Simplified: just compress the overall dynamic range
        nonzero_mag = mag[mag > 1e-10]
        if len(nonzero_mag) > 0:
            ratio = np.max(nonzero_mag) / (np.min(nonzero_mag) + 1e-15)
            # Compress: mag_new = mag^alpha where alpha < 1
            alpha = max(0.3, 1.0 - 0.1 * np.log1p(np.log1p(ratio)))
            mag_new = mag ** alpha
            phase_new = phase

            A_new = mag_new * np.exp(1j * phase_new)
            a_new = ifft(A_new).real[:n]
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = float(np.max(compute_conv(a_new)))
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new.copy()

        # Also try: notch at peak-contributing frequencies
        # The peak at m_star gets contribution from all k, weighted by cos(2*pi*k*m_star/N)
        # Reduce |A[k]| where cos(2*pi*k*m_star/N) > 0.5
        freq_idx = np.arange(N)
        cos_weights = np.cos(2.0 * np.pi * freq_idx * m_star / N)
        notch_mask = cos_weights > 0.3
        mag_notch = mag.copy()
        mag_notch[notch_mask] *= 0.95  # Gentle reduction
        A_notch = mag_notch * np.exp(1j * phase)
        a_notch = ifft(A_notch).real[:n]
        a_notch = np.maximum(a_notch, 0)
        s_notch = float(np.sum(a_notch))
        if s_notch > 1e-10:
            a_notch = a_notch / s_notch
            obj_notch = float(np.max(compute_conv(a_notch)))
            if obj_notch < best_obj - 1e-14:
                best_obj = obj_notch
                best_a = a_notch.copy()
                a = a_notch.copy()

        it += 1
        if verbose and it % 20 == 0:
            print(f"    Spectral it{it}: obj={best_obj:.8f}")

        # Stop if no improvement for a while
        if it > 100 and it % 20 == 0:
            # Check if we're making progress
            if best_obj >= float(np.max(compute_conv(a))) - 1e-10:
                break

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        # Multi-frequency construction
        freqs = rng.integers(2, 8, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power-law (heavy-tailed)
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = np.maximum(a, 0)
        if a.sum() < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / a.sum()
    elif mode == 6:
        # Concentrated center
        x = np.linspace(0, np.pi, n)
        a = np.sin(x ** 2)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 7:
        # Two concentrated peaks
        c1 = n // 5
        c2 = 4 * n // 5
        w = max(1, n // 15)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 8:
        # Sinc-like (derivative of Gaussian)
        x = np.linspace(0, 1, n)
        center = 0.5
        width = 0.15
        a = np.sin(2 * np.pi * (x - center) / width) * np.exp(-((x - center) ** 2) / (2 * width ** 2))
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    else:
        # Random smooth (low-pass filtered noise)
        a = rng.normal(0, 1, n)
        # Simple moving average smoothing
        k = max(1, n // 50)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Quick CD warmup
    Phase 2: Alternating LP optimization
    Phase 3: Enhanced coordinate descent (main workhorse)
    Phase 4: Smoothed gradient descent
    Phase 5: More CD
    Phase 6: Spectral equalization
    Phase 7: Multi-n exploration
    Phase 8: Diverse restarts
    Phase 9: Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Quick CD warmup (60s) ===
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 60.0)
    if p1_budget > 20:
        print(f"\n=== Phase 1: CD Warmup ({p1_budget:.0f}s) ===")
        a_cd = coord_descent_enhanced(best_seq, time_budget=p1_budget, rng=rng, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj_cd:.8f}")

    # === Phase 2: Alternating LP (150s) ===
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 150.0)
    if p2_budget > 30:
        print(f"\n=== Phase 2: Alternating LP ({p2_budget:.0f}s) ===")
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(p2_budget / (3 - rnd) * (1 if rnd == 0 else 0.7), remaining * 0.8)
            if alloc < 20:
                break
            a_lp = alternating_lp_optimize(best_seq, time_budget=alloc, rng=rng, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                n_best = len(best_seq)
                print(f"  LP rnd{rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP rnd{rnd}: obj={obj_lp:.8f}")
                # Perturb for next round
                scale = 0.015 * (rnd + 1)
                best_seq = best_seq * np.exp(rng.normal(0, scale, n_best))
                best_seq = np.maximum(best_seq, 0)
                best_seq = best_seq / best_seq.sum()

    # === Phase 3: Main CD (300s) ===
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.25, 300.0)
    if p3_budget > 50:
        print(f"\n=== Phase 3: Main CD ({p3_budget:.0f}s) ===")
        a_cd = coord_descent_enhanced(best_seq, time_budget=p3_budget, rng=rng, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj_cd:.8f}")

    # === Phase 4: Smoothed Gradient (120s) ===
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.12, 120.0)
    if p4_budget > 40:
        print(f"\n=== Phase 4: Smoothed Gradient ({p4_budget:.0f}s) ===")
        a_sgd = adaptive_smoothed_descent(best_seq, time_budget=p4_budget,
                                          tau_init=10.0, tau_end=3000.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase4: obj={best_score:.8f} **")
        else:
            print(f"  Phase4: obj={obj_sgd:.8f}")

    # === Phase 5: Fine CD (250s) ===
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.20, 250.0)
    if p5_budget > 50:
        print(f"\n=== Phase 5: Fine CD ({p5_budget:.0f}s) ===")
        a_cd = coord_descent_enhanced(best_seq, time_budget=p5_budget, rng=rng, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase5: obj={best_score:.8f} **")
        else:
            print(f"  Phase5: obj={obj_cd:.8f}")

    # === Phase 6: Spectral Equalization (80s) ===
    remaining = deadline - time.monotonic()
    p6_budget = min(remaining * 0.08, 80.0)
    if p6_budget > 30:
        print(f"\n=== Phase 6: Spectral Equil ({p6_budget:.0f}s) ===")
        a_spec = spectral_equalization(best_seq, time_budget=p6_budget, verbose=True)
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Phase6: obj={best_score:.8f} **")
        else:
            print(f"  Phase6: obj={obj_spec:.8f}")
        # Quick CD after spectral
        remaining = deadline - time.monotonic()
        if remaining > 40:
            a_cd = coord_descent_enhanced(best_seq, time_budget=min(remaining * 0.15, 50.0),
                                          rng=rng)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Phase6 CD: obj={best_score:.8f} **")

    # === Phase 7: Multi-n exploration (180s) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 7: Multi-n ({remaining:.0f}s remaining) ===")
    n_current = len(best_seq)

    n_options = sorted(set([
        max(500, n_current - 100),
        max(500, n_current - 50),
        n_current + 50,
        n_current + 100,
        n_current + 200,
        700, 900, 1100, 1200, 1000
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1400 and abs(n - n_current) > 20]

    explored = 0
    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        explored += 1

        a_try = resample_sequence(best_seq, n_try)

        # Quick CD
        alloc1 = min(remaining * 0.3, 50.0)
        if alloc1 > 15:
            a_try = coord_descent_enhanced(a_try, time_budget=alloc1, rng=rng)

        # Quick LP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 35.0)
        if alloc2 > 15:
            a_try = alternating_lp_optimize(a_try, time_budget=alloc2, rng=rng)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Diverse restarts (120s) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 8: Diverse Restarts ({remaining:.0f}s remaining) ===")
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = (r + 3) % 10
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with current best
        blend = 0.4 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        # CD
        alloc1 = min(remaining * 0.4, 60.0)
        if alloc1 > 20:
            a_r = coord_descent_enhanced(a_r, time_budget=alloc1, rng=rng)

        # LP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.2, 30.0)
        if alloc2 > 15:
            a_r = alternating_lp_optimize(a_r, time_budget=alloc2, rng=rng)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 9: Final intensive polish ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 9: Final Polish ({remaining:.0f}s remaining) ===")

    # CD
    if remaining > 40:
        alloc = min(remaining * 0.30, 100.0)
        a_cd = coord_descent_enhanced(best_seq, time_budget=alloc, rng=rng, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # LP
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 70.0)
        a_lp = alternating_lp_optimize(best_seq, time_budget=alloc, rng=rng, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # One more CD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.40, 60.0)
        a_cd = coord_descent_enhanced(best_seq, time_budget=alloc, rng=rng)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")

    # Validate
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    check_val = evaluate_sequence(result)
    if check_val < np.inf:
        best_score = check_val
    else:
        result = [1.0] * 100
        best_score = compute_obj(np.ones(100))

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
