def build_conv_matrix(a):
    """Build matrix A_mat (2n-1 x n) such that A_mat @ g = convolve(a, g)."""
    n = len(a)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A_mat[j:j + r, j] = a[:r]
    return A_mat


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses a_mat precomputation for speed.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A_mat = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

        return obj_fn(x)
    
    c = lo + gr * (hi - lo)
    d = lo + (1.0 - gr) * (hi - lo)
    fc = eval_at(c)
    fd = eval_at(d)
    
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = eval_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = eval_at(d)
    
    t_best = (lo + hi) / 2.0
    return t_best


def nesterov_smoothed_descent(a, time_budget=300.0, tau_init=10.0, tau_end=500.0,
                               verbose=False, seed=0):
    """
    Nesterov accelerated gradient descent on smoothed objective over simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.8
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1.0))

        # Cosine annealing for tau
        tau = tau_init + (tau_end - tau_init) * (1.0 - np.cos(np.pi * progress)) / 2.0

        # Periodic tau reset
        if it > 0 and it % 4000 == 0:
            tau = tau_init * 1.5

        if it % 2000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        smoothed_val, g, p, conv = smoothed_obj_and_grad(y, tau)

        # Adaptive step size via背负
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.7 * lr + 0.3 * min(max(bb_lr, 1e-8), 20.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1

        # Decay learning rate
        lr = max(1e-8, lr * 0.99995)

        # Reset if stuck
        if it - best_it > 25000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 5.0 / np.sqrt(n)
            best_it = it

    return best_a


def lp_polish_with_linesearch(a, time_budget=200.0, verbose=False, seed=0):
    """
    LP polishing with golden-section line search for optimal blending.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    consecutive_fails = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Golden section line search along LP direction
        t_best = golden_section_line_search(a, g_lp, compute_obj, max_iter=25)
        a_try = (1.0 - t_best) * a + t_best * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            a_try = best_a.copy()
        else:
            a_try = a_try / s_try

        obj_try = compute_obj(a_try)
        iter_count += 1

        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()
            consecutive_fails = 0
            if verbose and iter_count % 5 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={t_best:.3f}, M={M_lp:.6f}")
        else:
            consecutive_fails += 1
            # Spectral-guided perturbation
            ps = compute_power_spectrum(a)
            # Compute how far from flat spectrum
            ps_mean = np.mean(ps[1:])
            ps_std = np.std(ps[1:]) / ps_mean if ps_mean > 0 else 1.0
            
            # Perturb in the direction that reduces spectral peaks
            noise = rng.normal(0, 1, n) * 0.03
            # Weight by "spectral importance" (peaks in autocorrelation)
            conv = compute_conv(a)
            top_positions = np.argsort(conv)[::-1][:5]
            weight = np.ones(n)
            for k in top_positions:
                k = int(k)
                for i in range(max(0, k - n + 1), min(n, k + 1)):
                    j = k - i
                    if 0 <= j < n:
                        weight[i] += 0.5
            noise = noise * (0.5 + 0.5 * weight / weight.max())
            
            a_new = a * np.exp(noise)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = 0.7 * a_new + 0.3 * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()

            if consecutive_fails > 30:
                break

    return best_a


def spectral_perturbation(a, scale=0.05, seed=0):
    """
    Perturb sequence based on power spectrum analysis.
    Directions that flatten the spectrum get more weight.
    """
    n = len(a)
    rng = np.random.default_rng(seed)
    
    # Compute current spectrum
    ps = compute_power_spectrum(a)
    ps_flat = np.mean(ps[1:])
    
    # Random perturbation in "spectral flattening" direction
    # Try several random directions and pick the one that best flattens
    best_a = a.copy()
    best_flatness = spectral_flatness(a)
    
    for trial in range(10):
        noise = rng.normal(0, 1, n) * scale
        a_try = a * np.exp(noise)
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        flatness = spectral_flatness(a_try)
        if flatness > best_flatness:
            best_flatness = flatness
            best_a = a_try.copy()
    
    return best_a


def multi_resolution_optimize(a, time_budget=300.0, verbose=False, seed=0):
    """
    Coarse-to-fine optimization: downsample, optimize, upsample.
    """
    n = len(a)
    start = time.monotonic()
    
    # Level 1: Full resolution GD
    alloc1 = min(time_budget * 0.3, time_budget - 10)
    a1 = nesterov_smoothed_descent(a, time_budget=alloc1,
                                    tau_init=5.0, tau_end=200.0,
                                    verbose=verbose, seed=seed)
    obj1 = compute_obj(a1)
    if verbose:
        print(f"  Multi-res L1 (n={n}): obj={obj1:.8f}")
    
    remaining = time_budget - (time.monotonic() - start)
    
    # Level 2: Downsample to n/2, optimize, upsample back
    if remaining > 30:
        n_coarse = n // 2
        if n_coarse >= 50:
            # Downsample by averaging pairs
            a_coarse = np.zeros(n_coarse)
            for i in range(n_coarse):
                a_coarse[i] = (a1[2*i] + a1[2*i+1]) / 2.0
            
            alloc2 = min(remaining * 0.3, 60.0)
            a_coarse_opt = nesterov_smoothed_descent(a_coarse, time_budget=alloc2,
                                                      tau_init=5.0, tau_end=100.0,
                                                      verbose=False, seed=seed + 100)
            
            # Upsample by interpolation
            x_old = np.linspace(0, 1, n_coarse)
            x_new = np.linspace(0, 1, n)
            a_upsampled = np.interp(x_new, x_old, a_coarse_opt)
            a_upsampled = np.maximum(a_upsampled, 0)
            a_upsampled = a_upsampled / a_upsampled.sum()
            
            # Blend with a1
            a_blend = 0.5 * a1 + 0.5 * a_upsampled
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()
            
            obj_blend = compute_obj(a_blend)
            if verbose:
                print(f"  Multi-res L2 (blend): obj={obj_blend:.8f}")
            
            if obj_blend < obj1:
                a1 = a_blend
    
    remaining = time_budget - (time.monotonic() - start)
    
    # Level 3: Fine-tune with LP
    if remaining > 20:
        alloc3 = min(remaining * 0.5, 80.0)
        a1 = lp_polish_with_linesearch(a1, time_budget=alloc3,
                                        verbose=verbose, seed=seed + 200)
    
    return a1
