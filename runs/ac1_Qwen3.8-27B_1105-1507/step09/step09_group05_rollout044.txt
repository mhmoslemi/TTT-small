import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.sparse import diags, csr_matrix
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_sparse_conv_matrix(a):
    """Build sparse matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # conv(a, g)[k] includes a[j] * g[k-j] for valid k-j
        # So for fixed j, contrib to row k = a[j] * g[k-j] where 0 <= k-j < n
        # i.e., k ranges from j to j+n-1, and the value is a[j]
        r_start = j
        r_end = min(2 * n - 1, j + n)
        for k in range(r_start, r_end):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_minpeak_sparse(a, warm_start=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses sparse matrix. Returns (g_optimal, M_optimal) or (None, None).
    """

    a_norm = a / float(np.sum(a))
    A = build_sparse_conv_matrix(a_norm)

    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:

    return None, None

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses sparse LP polishing with adaptive small steps, coordinate descent refinement,
multi-length exploration, and diverse restarts.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_sparse_conv_matrix(a):
    n = len(a)
    rows_list = []
    cols_list = []
    vals_list = []
    for j in range(n):
        r_start = j
        r_end = min(2 * n - 1, j + n)
        for k in range(r_start, r_end):
            rows_list.append(k)
            cols_list.append(j)
            vals_list.append(a[j])
    return csr_matrix((vals_list, (rows_list, cols_list)), shape=(2 * n - 1, n))


def solve_lp(a):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_sparse_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None
