import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # (a*g)[k] = sum_i a[i] * g[k-i]
        # For fixed j (index into g), a[i] * g[j] contributes to k = i + j
        # So A[i+j, j] = a[i] for valid i where 0 <= i < n and 0 <= i+j < 2n-1
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, max_time=30):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build the constraint matrix
    # (a*g)[k] = sum_j a[k-j] * g[j] for valid k-j
    # We need a matrix B such that B @ g = conv(a, g)
    # B has shape (2n-1, n)
    # B[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] contributes to positions k = j, j+1, ..., j+n-1
        # with coefficients a[0], a[1], ..., a[n-1]
        B[j:j+n, j] = a_norm
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    constraints = [
        B @ g <= M,
        cp.sum(g) == 1
    ]
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_minpeak_active(a, active_indices):
    """
    Solve LP: min sum(conv(a,g)[k] for k in active_indices) s.t. sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j+n, j] = a_norm
    
    # Only constrain active indices
    B_active = B[active_indices, :]
    
    g = cp.Variable(n, nonneg=True)
    obj = cp.sum(B_active @ g)
    constraints = [cp.sum(g) == 1]
    
    prob = cp.Problem(cp.Minimize(obj), constraints)
    
    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def golden_section_line_search(a, g, n_evals=8):
    """Find optimal t in [0,1] that minimizes compute_obj((1-t)*a + t*g)."""
    phi = (np.sqrt(5) - 1) / 2
    
    lo = 0.0
    hi = 1.0
    
    for _ in range(n_evals):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)
        
        a1 = (1 - t1) * a + t1 * g
        a1 = np.maximum(a1, 0)
        s1 = float(np.sum(a1))
        if s1 > 1e-10:
            a1 = a1 / s1
            f1 = compute_obj(a1)
        else:
            f1 = np.inf
        
        a2 = (1 - t2) * a + t2 * g
        a2 = np.maximum(a2, 0)
        s2 = float(np.sum(a2))
        if s2 > 1e-10:
            a2 = a2 / s2
            f2 = compute_obj(a2)
        else:
            f2 = np.inf
        
        if f1 < f2:
            hi = t2
        else:
            lo = t1
    
    t_best = (lo + hi) / 2
    a_best = (1 - t_best) * a + t_best * g
    a_best = np.maximum(a_best, 0)
    s_best = float(np.sum(a_best))
    if s_best > 1e-10:
        a_best = a_best / s_best
    return a_best

def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    else:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    return a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key fix over previous: corrected LP solver variable name bug (s_solver -> solver)
that caused all LP calls to silently fail. Now LP-based polishing actually works.
Combined with peak-targeted refinement, multi-n search, and Nesterov smoothed GD.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_cross(a, b):
    """Cross-correlation/convolution a*b via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a_norm):
    """Build matrix B (2n-1 x n) such that B @ g = conv(a_norm, g)."""
    n = len(a_norm)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = a_norm
    return B


def solve_lp_minpeak(a, timeout=60):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    B = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()

    constraints = [
        B @ g <= M,
        cp.sum(g) == 1
    ]

    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_minpeak_active(a, active_indices):
    """
    Solve LP: min sum(conv(a,g)[k] for k in active) s.t. sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    B = build_conv_matrix(a_norm)
    B_active = B[active_indices, :]

    g = cp.Variable(n, nonneg=True)
    obj = cp.sum(B_active @ g)
    constraints = [cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(obj), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]

                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))

                    return g_val / s

    return None


def golden_section_line_search(a, g, n_evals=8):
    """Find optimal t in [0,1] minimizing compute_obj((1-t)*a + t*g)."""
    phi = (np.sqrt(5) - 1) / 2
    lo = 0.0
    hi = 1.0

    for _ in range(n_evals):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)

        a1 = (1 - t1) * a + t1 * g
        a1 = np.maximum(a1, 0)
        s1 = float(np.sum(a1))
        f1 = compute_obj(a1 / s1) if s1 > 1e-10 else np.inf

        a2 = (1 - t2) * a + t2 * g
        a2 = np.maximum(a2, 0)
        s2 = float(np.sum(a2))
        f2 = compute_obj(a2 / s2) if s2 > 1e-10 else np.inf

        if f1 < f2:
            hi = t2
        else:
            lo = t1

    t_best = (lo + hi) / 2
    a_best = (1 - t_best) * a + t_best * g
    a_best = np.maximum(a_best, 0)
    s_best = float(np.sum(a_best))
    if s_best > 1e-10:
        a_best = a_best / s_best
    return a_best


def two_stage_lp_polish(a, time_budget=100.0, verbose=False):
    """Two-stage LP polishing: min-max then min-sum-of-active, with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(7777)

    iter_count = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        g1, M1 = solve_lp_minpeak(a)
        if g1 is None:
            break

        conv_ag1 = compute_cross(a, g1)
        max_conv_ag = float(np.max(conv_ag1))
        tol = 0.01 * max(max_conv_ag, 1e-10)
        active = np.where(conv_ag1 >= max_conv_ag - tol)[0]

        if len(active) == 0:
            active = np.array([np.argmax(conv_ag1)])

        g2 = solve_lp_minpeak_active(a, active)

        if g2 is not None:
            conv_a_g2 = compute_cross(a, g2)
            peak_g1 = max_conv_ag
            peak_g2 = float(np.max(conv_a_g2))
            if peak_g2 < peak_g1:
                g_blend = g2
            else:
                g_blend = (g1 + g2) / 2
        else:
            g_blend = g1

        a_new = golden_section_line_search(a, g_blend, n_evals=7)
        obj_new = compute_obj(a_new)

        iter_count += 1
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            if verbose and iter_count % 3 == 0:
                print(f"    2SLP it{iter_count}: obj={best_obj:.8f}, M1={M1:.6f}, |S|={len(active)}")
        else:
            stall_count += 1
            scale = 0.015 + 0.005 * min(stall_count, 8)
            a = best_a * np.exp(rng.normal(0, min(scale, 0.08), n))
            a = np.maximum(a, 0)
            s_a = float(np.sum(a))
            if s_a > 1e-10:
                a = a / s_a
            else:
                a = best_a.copy()
            if stall_count > 25:
                break

    return best_a


def accelerated_lp(a, time_budget=100.0, momentum=0.5, verbose=False):
    """Accelerated LP with heavy-ball momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    it = 0
    stall = 0
    base_momentum = momentum

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        v = momentum * v + (g_lp - a)
        a_new = project_simplex(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            found = False
            for scale in [0.5, 0.25, 0.1]:
                a_try = project_simplex(a + scale * v)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    v = scale * v
                    stall = 0
                    found = True
                    break
            if not found:
                stall += 1
                momentum = max(0.1, momentum * 0.85)
                v *= 0.3
                a = best_a.copy()
                if stall > 25:
                    rng = np.random.default_rng(it)
                    a = best_a * np.exp(rng.normal(0, 0.02, n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    v = np.zeros(n)
                    momentum = base_momentum
                    stall = 0
                    if it > 250:
                        break

        it += 1
        momentum = min(0.8, base_momentum + 0.001 * it)

        if verbose and it % 5 == 0:
            m_str = f"{M_lp:.6f}" if M_lp is not None else "N/A"
            print(f"    ALP it{it}: obj={best_obj:.8f}, M_lp={m_str}, mom={momentum:.3f}")

    return best_a


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def nesterov_smoothed_descent(a, time_budget=300.0, tau_init=10.0, tau_end=400.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        if it % 2000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)

        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9998)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def peak_targeted_refinement(a, time_budget=60.0, verbose=False):
    """Target the top peaks and try to reduce them via local moves."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(4242)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_peaks = np.argsort(conv)[::-1][:3]

        improved = False
        for max_k in top_peaks:
            max_k = int(max_k)
            i_start = max(0, max_k - (n - 1))
            i_end = min(n - 1, max_k)

            if i_start > i_end:
                continue

            best_delta = None
            best_new_obj = best_obj

            for i in range(i_start, i_end + 1):
                j = max_k - i
                if j < 0 or j >= n or i == j:
                    continue
                if a[i] < 1e-10:
                    continue

                for delta_frac in [0.008, 0.004, 0.002]:
                    delta = delta_frac * a[i]

                    for _ in range(10):
                        tgt = int(rng.integers(0, n))
                        if tgt == i or tgt == j:
                            continue
                        a_try = a.copy()
                        a_try[i] -= delta
                        a_try[tgt] += delta
                        if a_try[i] < 0:
                            continue
                        s_try = float(np.sum(a_try))
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_new_obj:
                            best_new_obj = obj_try
                            best_delta = (i, tgt, delta)

            if best_delta is not None and best_new_obj < best_obj - 1e-14:
                i_src, i_tgt, delta = best_delta
                a[i_src] -= delta
                a[i_tgt] += delta
                a = a / a.sum()
                best_obj = best_new_obj
                best_a = a.copy()
                improved = True
                break

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 25:
                break

        if verbose and it % 20 == 0:
            print(f"    PCD it{it}: obj={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    else:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    return a


def multi_n_explore(best_seq, best_score, deadline, verbose=True):
    """Try different sequence lengths."""
    n_current = len(best_seq)

    n_candidates = sorted(set([
        max(150, n_current - 200),
        max(150, n_current - 100),
        n_current + 100,
        n_current + 200,
        500, 700, 900, 1100, 1300
    ]))
    n_candidates = [n for n in n_candidates if 150 <= n <= 1500 and abs(n - n_current) > 50]
    n_candidates.sort(key=lambda n: abs(n - n_current))

    for n_try in n_candidates[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.35, 45.0)
        if alloc > 15:
            a_try = two_stage_lp_polish(a_try, time_budget=alloc)

        obj_try = compute_obj(a_try)
        if verbose:
            print(f"    n={n_try}: obj={obj_try:.8f}")

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            if verbose:
                print(f"    ** Improved: n={n_try}, obj={best_score:.8f}")
        elif obj_try < best_score * 1.005:
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_try = accelerated_lp(a_try, time_budget=min(remaining * 0.4, 30.0), momentum=0.6)
                obj_try = compute_obj(a_try)
                if obj_try < best_score:
                    best_score = obj_try
                    best_seq = a_try.copy()
                    if verbose:
                        print(f"    ** Improved after ALP: n={n_try}, obj={best_score:.8f}")

    return best_seq, best_score
