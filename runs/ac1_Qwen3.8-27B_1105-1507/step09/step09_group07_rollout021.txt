def build_cross_matrix(a):

        # So B[k, j] = a[k-j] if 0 <= k-j < n, else 0

def solve_lp(a, solver_list=None):
    """
    Solve LP: min M s.t. cross(a, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    B = build_cross_matrix(a_norm)
    
    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS]
    
    for solver in solver_list:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [B @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    # Fallback: scipy linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = B
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None

def gradient_polish(a, time_budget, verbose=False):

    Subgradient-based polishing: iteratively reduce the peak convolution value.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 0.1
    momentum = 0.9
    v = np.zeros(n)
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_val = float(np.max(conv))
        peak_lags = np.where(conv >= peak_val * 0.999)[0]
        
        # Compute subgradient: average of gradients at peak lags
        g = np.zeros(n)
        for k in peak_lags:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    g[i] += 2.0 * a[j]
        g /= len(peak_lags)
        
        # Normalization: subtract mean to maintain sum(a) = 1
        g = g - np.mean(g)
        
        # Momentum update
        v = momentum * v - lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = min(lr * 1.01, 1.0)
        else:
            lr *= 0.9
            v *= 0.5
        
        it += 1
        if verbose and it % 100 == 0:
            print(f"    GradPolish it{it}: obj={best_obj:.8f}")
    
    return best_a

    idx = np.arange(2*n-1)[:, None] - np.arange(n)[None, :]  # shape (2n-1, n)
