import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, g):
    """Compute conv(a, g) via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g, size)
    return irfft(fa * fg, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A such that (A@g)[k] = conv(a, g)[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # A[k, j] = a[k - j] if 0 <= k-j < n
        # k ranges from j to j+n-1
        k_start = j
        k_end = min(m, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_fast(a):
    """
    Solve LP: min M s.t. conv(a, g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses scipy HiGHS. Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    m = 2 * n - 1
    
    # Build constraint matrix: A @ g <= M  =>  A @ g - M <= 0
    A = build_conv_matrix(a_norm)
    
    # Variables: [g_0, ..., g_{n-1}, M]
    # Objective: minimize M => c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A @ g - M <= 0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={"presolve": True, "time_limit": 10})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None

        a = a / a.sum()
    elif mode % 10 == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + np.exp(-3 * (x - 0.7) ** 2) + 0.05
        a = a / a.sum()
    elif mode % 10 == 5:
        x = np.linspace(0, 1, n)
        a = x**2 + (1 - x)**2 + 0.1
        a = a / a.sum()
    elif mode % 10 == 6:
        freqs = rng.integers(2, 10, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n) * 0.5
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 7:
        a = rng.dirichlet(np.ones(n) * 0.3)
    elif mode % 10 == 8:
        # Lopsided: large on one side
        x = np.linspace(0, 1, n)
        a = np.exp(-2 * x) + 0.1
        a = a / a.sum()
    else:
        # Exponential decay from center
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.5)**2) + 0.05
        a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coarse_to_fine(a_coarse, n_fine, time_budget):
    """
    Start from a coarse solution, upsample to fine, and refine.
    This helps escape local minima of the fine-scale problem.
    """
    n_coarse = len(a_coarse)
    a_fine = resample_sequence(a_coarse, n_fine)
    a_fine = a_fine / a_fine.sum()
    
    # Refine with LP alternating
    a_opt = lp_alternate_simple(a_fine, time_budget=time_budget, verbose=False)
    return a_opt


def smooth_refine(a, time_budget):
    """
    Smooth refinement using log-sum-exp gradient with softmax updates.
    Good for final polish.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.1
    beta = 50.0
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        beta = min(500.0, 50.0 * (1.05 ** it))

        conv = compute_conv(a)
        m = conv.max()
        shifted = beta * (conv - m)
        shifted -= shifted.max()  # numerical stability
        weights = np.exp(shifted)
        weights = weights / weights.sum()

        # Gradient of sum_k weights[k] * conv(a,a)[k] with respect to a[i]
        # d(conv[a,a][k])/da[i] = a[k-i] + a[i-k] (if indices valid)
        # But since we're using weights on the full conv, and conv is symmetric:
        # grad[i] = 2 * sum_k weights[k] * a[k-i] (with appropriate bounds)
        
        # Efficient computation via cross-correlation
        # grad = 2 * conv(a, weights_padded) truncated
        # Actually: grad[i] = sum_k weights[k] * (a[k-i] + a[i-k]) for valid indices
        # = 2 * (a cross-correlated with weights)[i] (approximately, for the middle part)
        
        # Let's compute it directly using FFT
        # conv_grad = convolve(a, weights) but weights has length 2n-1 and a has length n
        # We need: for each i, sum_{k} weights[k] * a[k-i] where 0<=k-i<n and 0<=k<2n-1
        # This is like a cross-correlation
        
        size = 1
        while size < 3 * n:
            size <<= 1
        fa = rfft(a, size)
        fw = rfft(weights, size)
        # Cross-correlation: ifft(conj(fa) * fw) gives sum a[i+k] * w[k]
        # We want sum_k w[k] * a[k-i] = cross-correlation
        cross = irfft(np.conj(fa) * fw, size)[:2*n]
        # Actually let me just do it more carefully
        # For position i in a, the gradient contribution is:
        # d/da[i] of conv(a,a)[k] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
        # So grad[i] = sum_k weights[k] * (a[k-i] + a[i-k]) * [validity]
        
        # Use cross-correlation: c[i] = sum_k a[k] * w[k+i] (for appropriate ranges)
        # Let's just compute it directly for clarity (n is at most ~2000)
        
        grad = np.zeros(n)
        for i in range(n):
            # Contribution from a[k-i] term: k-i valid means k in [i, i+n-1]
            k1_start = i
            k1_end = min(2*n-1, i+n)
            if k1_start < k1_end:
                grad[i] += np.dot(weights[k1_start:k1_end], a[:k1_end-k1_start])
            # Contribution from a[i-k] term: i-k valid means k in [i-n+1, i], i.e., k in [max(0,i-n+1), i]
            # But by symmetry of conv, this is the same as the first term with k -> 2n-2-k
            # Actually for conv(a,a), it IS symmetric, so the gradient is:
            # grad[i] = 2 * sum_k weights[k] * a[k-i] (with bounds)
            # Let me just use the first term doubled (since conv(a,a) is symmetric)
            pass
        
        # Simpler: grad[i] = 2 * sum_{k} weights[k] * a[k-i] where 0 <= k-i < n
        # = 2 * (cross-correlation of a and weights)[i]
        # cross[i] = sum_k a[k] * weights[k + 2n - 2 - i] ... this is getting complicated
        # Let me just use the direct O(n^2) approach for now (n <= 2000, so n^2 = 4M, fine)
        
        grad = np.zeros(n)
        for i in range(n):
            total = 0.0
            for k in range(2*n - 1):
                j = k - i
                if 0 <= j < n:
                    total += weights[k] * a[j]
                j2 = i - k
                if 0 <= j2 < n:
                    total += weights[k] * a[j2]
            grad[i] = total
        
        # Mirror descent update
        log_a = np.log(a)
        log_a = log_a - lr * grad / (best_obj + 1e-10)
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = min(0.5, lr * 1.01)
        else:
            lr = max(0.001, lr * 0.9)

        if it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"            Smooth it{it}: obj={best_obj:.8f}, beta={beta:.0f}, lr={lr:.4f}, elapsed={elapsed:.1f}s")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines LP alternating, n-sweep, diverse restarts,
    and smooth refinement to minimize the peak autocorrelation score.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0  # leave 30s buffer

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): score={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800 uniform): score={best_score:.8f}")

    n_current = len(best_seq)
    print(f"\n=== Start: n={n_current}, score={best_score:.8f}, target=1.5027 ===")

    # === Phase 1: Quick LP polish on parent (5% of budget) ===
    print(f"\n--- Phase 1: Quick LP Polish ---")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.05, 60.0)
    if p1_budget > 15:
        a_p1 = lp_alternate_simple(best_seq, time_budget=p1_budget, verbose=False)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Phase 1: score={best_score:.8f} **")
        else:
            print(f"  Phase 1: score={obj_p1:.8f}")

    # === Phase 2: n-Sweep (25% of budget) ===
    print(f"\n--- Phase 2: n-Sweep ---")
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.25

    # Generate n candidates more aggressively
    base_n = n_current
    n_options = set()
    for delta in range(-200, 401, 25):
        n_try = base_n + delta
        if 300 <= n_try <= 2500:
            n_options.add(n_try)
    # Add some "magic" values
    for n_magic in [500, 512, 550, 600, 640, 650, 700, 750, 768, 800, 850, 900,
                     960, 1000, 1024, 1050, 1100, 1200, 1280, 1400, 1500, 1600, 2000]:
        n_options.add(n_magic)
    
    n_options = sorted(n_options)
    
    # Sort by proximity to current best n first, then by other values
    n_sorted = sorted(n_options, key=lambda x: (abs(x - base_n), x))
    
    per_n_time = p2_budget / max(1, len(n_sorted))
    # Give more time to n values near the current best
    scored_results = []
    
    for idx, n_try in enumerate(n_sorted):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        # Allocate time: more for nearby n, less for far
        dist = abs(n_try - base_n)
        alloc = min(per_n_time * (1.0 + max(0, 1.0 - dist / 500.0)), remaining * 0.08)
        alloc = max(alloc, 5.0)
        if alloc > 60:
            alloc = 60

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()
        a_opt = lp_alternate_simple(a_try, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)
        marker = " **" if obj_opt < best_score else ""
        print(f"  n={n_try}: score={obj_opt:.8f}{marker}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try
        scored_results.append((n_try, obj_opt))

    # === Phase 3: Intensive LP on best n (35% of budget) ===
    print(f"\n--- Phase 3: Intensive LP (n={n_current}) ---")
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.35

    n_rounds = 5
    per_round = p3_budget / n_rounds
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_round, remaining * 0.2)
        if alloc < 30:
            alloc = 30

        if rnd == 0:
            a_start = best_seq.copy()
        elif rnd == 1:
            # Slight perturbation
            rng_r = np.random.default_rng(31337 + rnd * 7 + n_current)
            a_start = best_seq * np.exp(rng_r.normal(0, 0.02, n_current))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        elif rnd == 2:
            # Medium perturbation
            rng_r = np.random.default_rng(54321 + rnd * 13 + n_current)
            a_start = best_seq * np.exp(rng_r.normal(0, 0.05, n_current))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        elif rnd == 3:
            # Blend with diverse init
            mode = (rnd * 7) % 10
            a_div = diverse_init(n_current, rng, mode=mode)
            a_start = 0.7 * best_seq + 0.3 * a_div
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        else:
            # Larger blend
            mode = (rnd * 11 + 3) % 10
            a_div = diverse_init(n_current, rng, mode=mode)
            a_start = 0.5 * best_seq + 0.5 * a_div
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()

        a_lp = lp_alternate_simple(a_start, time_budget=alloc, verbose=(rnd == 0))
        obj_lp = compute_obj(a_lp)
        marker = " **" if obj_lp < best_score else ""
        print(f"  Round {rnd} (perturb): score={obj_lp:.8f}{marker}")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()

    # === Phase 4: Coarse-to-fine restarts (20% of budget) ===
    print(f"\n--- Phase 4: Coarse-to-Fine Restarts ---")
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.20

    n_use = len(best_seq)
    n_restarts = 4
    per_restart = p4_budget / n_restarts

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(per_restart, remaining * 0.2)
        if alloc < 20:
            break

        # Coarse solution: work at n/4 resolution
        n_coarse = max(50, n_use // 4)
        a_coarse = diverse_init(n_coarse, rng, mode=(r * 3 + 1) % 10)
        # Quick LP on coarse
        coarse_alloc = alloc * 0.3
        a_coarse_opt = lp_alternate_simple(a_coarse, time_budget=coarse_alloc, verbose=False)
        
        # Upsample and refine
        fine_alloc = alloc * 0.7
        a_fine = coarse_to_fine(a_coarse_opt, n_use, time_budget=fine_alloc)
        obj_fine = compute_obj(a_fine)
        marker = " **" if obj_fine < best_score else ""
        print(f"  C2F restart {r} (n_coarse={n_coarse}): score={obj_fine:.8f}{marker}")
        if obj_fine < best_score:
            best_score = obj_fine
            best_seq = a_fine.copy()

    # === Phase 5: Direct diverse restarts at full n (10% of budget) ===
    print(f"\n--- Phase 5: Diverse Restarts ---")
    remaining = deadline - time.monotonic()
    p5_budget = remaining * 0.10

    n_direct = 5
    per_direct = p5_budget / n_direct
    for r in range(n_direct):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_direct, remaining * 0.15)
        if alloc < 15:
            break

        mode = (r * 4 + 2) % 10
        a_div = diverse_init(n_use, rng, mode=mode)
        # Blend with best
        blend = 0.2 + 0.1 * r
        a_mix = (1 - blend) * a_div + blend * best_seq
        a_mix = np.maximum(a_mix, 0)
        a_mix = a_mix / a_mix.sum()

        a_opt = lp_alternate_simple(a_mix, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)
        marker = " **" if obj_opt < best_score else ""
        print(f"  Direct restart {r} (mode={mode}): score={obj_opt:.8f}{marker}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()

    # === Phase 6: Final polish (5% of budget) ===
    print(f"\n--- Phase 6: Final Polish ---")
    remaining = deadline - time.monotonic()

    # One more LP round
    alloc = min(remaining * 0.5, 120.0)
    if alloc > 20:
        a_final = lp_alternate_simple(best_seq, time_budget=alloc, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: score={best_score:.8f} **")
        else:
            print(f"  Final LP: score={obj_final:.8f}")

    # Smooth refinement
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.5, 60.0)
    if alloc > 15:
        a_smooth = smooth_refine(best_seq, time_budget=alloc)
        obj_smooth = compute_obj(a_smooth)
        if obj_smooth < best_score:
            best_score = obj_smooth
            best_seq = a_smooth.copy()
            print(f"  Smooth: score={best_score:.8f} **")
        else:
            print(f"  Smooth: score={obj_smooth:.8f}")

    # One last LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = lp_alternate_simple(best_seq, time_budget=min(remaining * 0.7, 30.0), verbose=False)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: score={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"=== Target: 1.5027, Achieved: {best_score:.8f}, Gap: {best_score - 1.5027:.8f} ===")

    return result
