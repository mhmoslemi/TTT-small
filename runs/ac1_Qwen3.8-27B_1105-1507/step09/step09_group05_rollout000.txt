import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_conv_stats(a):
    """Compute convolution statistics for analysis."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv = compute_conv(a_norm)
    max_val = float(np.max(conv))
    mean_val = float(np.mean(conv))
    std_val = float(np.std(conv))
    return conv, max_val, mean_val, std_val


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of (a*a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def nesterov_smoothed_descent(a, time_budget=200.0, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        if it % 5000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient(y, tau)

        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.3 * lr + 0.7 * min(max(bb_lr, 1e-8), 15.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9997)

        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal, dual_vars) or (None, None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    duals = None
                    try:
                        duals = (A @ g <= M).dual_value
                    except Exception:
                        pass
                    return g_val / s, float(M.value), duals
        except Exception:
            continue
    return None, None, None


def solve_qp_uniform(a, target_spread=0.0):
    """
    Solve QP: minimize sum_k (conv(a,g)_k - target)^2 + lambda * M
    s.t. sum(g)=1, g>=0, conv(a,g)_k <= M
    This encourages uniformity while keeping the max bounded.
    Returns (g_optimal, obj_value) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    target = 1.0 / (2 * n - 1)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    conv_vals = A @ g
    obj = cp.sum_squares(conv_vals - target) + 0.1 * M
    constraints = [cp.sum(g) == 1, conv_vals <= M + 0.01]
    prob = cp.Problem(cp.Minimize(obj), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(obj.value)
        except Exception:
            continue
    return None, None


def adaptive_line_search_blend(a, g_candidate, a_prev_obj, max_evals=30):
    """
    Adaptive line search: find best t in (0,1) such that
    a(t) = (1-t)*a + t*g_candidate minimizes the objective.
    Uses golden section search.
    """
    a = np.maximum(a, 0)
    g_candidate = np.maximum(g_candidate, 0)
    g_candidate = g_candidate / float(np.sum(g_candidate))

    best_t = 0.5
    best_obj = a_prev_obj
    a_at_best = (1 - best_t) * a + best_t * g_candidate
    a_at_best = np.maximum(a_at_best, 0)
    a_at_best = a_at_best / float(np.sum(a_at_best))
    best_obj = compute_obj(a_at_best)

    # Golden section search on [0.05, 1.0]
    lo = 0.05
    hi = 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    t1 = hi - gr * (hi - lo)
    t2 = lo + gr * (hi - lo)

    a_t1 = (1 - t1) * a + t1 * g_candidate
    a_t1 = np.maximum(a_t1, 0)
    a_t1 = a_t1 / float(np.sum(a_t1))
    obj_t1 = compute_obj(a_t1)

    a_t2 = (1 - t2) * a + t2 * g_candidate
    a_t2 = np.maximum(a_t2, 0)
    a_t2 = a_t2 / float(np.sum(a_t2))
    obj_t2 = compute_obj(a_t2)

    n_evals = 2

    while hi - lo > 0.01 and n_evals < max_evals:
        if obj_t1 < obj_t2:
            hi = t2
            t2 = t1
            obj_t2 = obj_t1
            t1 = hi - gr * (hi - lo)
            a_t1 = (1 - t1) * a + t1 * g_candidate
            a_t1 = np.maximum(a_t1, 0)
            a_t1 = a_t1 / float(np.sum(a_t1))
            obj_t1 = compute_obj(a_t1)
            n_evals += 1
        else:
            lo = t1
            t1 = t2
            obj_t1 = obj_t2
            t2 = lo + gr * (hi - lo)
            a_t2 = (1 - t2) * a + t2 * g_candidate
            a_t2 = np.maximum(a_t2, 0)
            a_t2 = a_t2 / float(np.sum(a_t2))
            obj_t2 = compute_obj(a_t2)
            n_evals += 1

        # Track best
        if obj_t1 < best_obj:
            best_obj = obj_t1
            best_t = t1
            a_at_best = a_t1.copy()
        if obj_t2 < best_obj:
            best_obj = obj_t2
            best_t = t2
            a_at_best = a_t2.copy()

    return a_at_best, best_obj, best_t


def dual_guided_perturbation(a, duals, rng, scale=0.03):
    """
    Perturb a based on LP dual variables: increase weight where
    constraints are nearly active (high dual value), decrease elsewhere.
    """
    n = len(a)
    a = a.copy()
    if duals is not None and len(duals) > 0:
        # Duals correspond to conv constraints; high dual = tight constraint
        # Map back to which a indices contribute most to tight constraints
        conv = compute_conv(a)
        # Find which indices of conv have high dual values
        duals = np.asarray(duals)
        if len(duals) == 2 * n - 1:
            # Weight a indices by their contribution to high-dual conv positions
            weights = np.zeros(n)
            for k in range(2 * n - 1):
                dk = duals[k]
                if dk > 0:
                    for j in range(n):
                        idx = k - j
                        if 0 <= idx < n:
                            weights[j] += dk * a[idx]
            max_w = np.max(weights)
            if max_w > 1e-10:
                weights = weights / max_w
                # Increase a where weights are low (to reduce tight constraints)
                a = a * (1.0 + scale * (1.0 - weights))
    else:
        # Fallback: random perturbation
        a = a * np.exp(rng.normal(0, scale, n))

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def perturb_sequence(a, rng, method=0, scale=0.05):
    """Apply various perturbations to escape local minima."""
    n = len(a)
    a = a.copy()
    if method == 0:
        a = a * np.exp(rng.normal(0, scale, n))
    elif method == 1:
        noise = rng.uniform(1 - scale, 1 + scale, n)
        a = a * noise
    elif method == 2:
        mask = rng.random(n) > scale * 2
        a = a * mask
    elif method == 3:
        shift = int(rng.integers(1, max(2, n // 10)))
        a = np.roll(a, shift)
    elif method == 4:
        a = (1 - scale) * a + scale / n
    elif method == 5:
        A = fft(a, 2 * n)
        phases = 1j * rng.normal(0, scale, len(A[1:-1]))
        A[1:-1] *= np.exp(phases)
        a = ifft(A).real[:n]
        a = np.maximum(a, 0)
    elif method == 6:
        conv = compute_conv(a / a.sum())
        top_k = np.argsort(conv)[::-1][:8]
        deltas = np.zeros(n)
        for k in top_k:
            k = int(k)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    deltas[j] += a[idx] / a.sum()
        max_d = np.max(np.abs(deltas))
        if max_d > 1e-10:
            a = a - scale * 0.3 * deltas / max_d
        a = np.maximum(a, 0)
    elif method == 7:
        # Spectral perturbation: modify mid-frequency components
        A = fft(a, 2 * n)
        mid_start = n // 4
        mid_end = 3 * n // 4
        A[mid_start:mid_end] *= (1 + rng.normal(0, scale * 2, mid_end - mid_start))
        a = ifft(A).real[:n]
        a = np.maximum(a, 0)
    elif method == 8:
        # Anti-peak: reduce values at positions that contribute to conv peaks
        conv = compute_conv(a)
        peak_idx = np.argmax(conv)
        contrib = np.zeros(n)
        for j in range(n):
            idx = peak_idx - j
            if 0 <= idx < n:
                contrib[j] = a[j] * a[idx]
        max_c = np.max(contrib)
        if max_c > 1e-10:
            a = a - scale * 0.5 * contrib / max_c
        a = np.maximum(a, 0)
    elif method == 9:
        # Smooth then renormalize
        kernel = np.ones(3) / 3.0
        a_smooth = np.convolve(a, kernel, mode='same')
        a = (1 - scale) * a + scale * a_smooth
    else:
        a = a * np.exp(rng.normal(0, scale, n))

    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if n == 2:
        x_old = np.array([0.0, 1.0])
        x_new = np.linspace(0, 1, new_n)
        a_new = np.interp(x_new, x_old, a)
        return np.maximum(a_new, 0)
    # Use scipy for cubic
    from scipy.interpolate import CubicSpline
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    cs = CubicSpline(x_old, a, bc_type='not-a-knot')
    a_new = cs(x_new)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        # U-shaped (high at edges, low in middle) - compensates for triangular conv
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * (2 * x - 1) ** 2
        a = a / a.sum()
    elif mode == 8:
        # Inverted U (low at edges, high in middle)
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(np.pi * x)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)


def lp_polish_round(a, max_iter=80, time_limit=100.0, verbose=False, use_qp=False):
    """
    LP polishing with adaptive line search.
    Alternates between min-max LP and (optionally) QP uniformity.
    Returns (best_a, best_obj, num_iterations).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf, 0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    no_improve = 0
    last_t = 0.4
    start = time.monotonic()
    it = 0

    while it < max_iter and (time.monotonic() - start) < time_limit:
        remaining_in_round = time_limit - (time.monotonic() - start)
        if remaining_in_round < 5:
            break

        # Choose between LP and QP
        use_qp_this_iter = use_qp and (it % 3 == 2)

        if use_qp_this_iter:
            g_lp, qp_obj = solve_qp_uniform(a)
            duals = None
            M_lp = None
        else:
            g_lp, M_lp, duals = solve_lp_minpeak(a)

        if g_lp is None:
            break

        # Adaptive line search for blending
        a_best_blend, obj_blend, t_opt = adaptive_line_search_blend(
            a, g_lp, best_obj, max_evals=15)

        it += 1
        if obj_blend < best_obj - 1e-13:
            best_obj = obj_blend
            best_a = a_best_blend.copy()
            a = best_a
            last_t = t_opt
            no_improve = 0
            if verbose and it % 5 == 0:
                conv, mx, mn, sd = compute_conv_stats(a)
                print(f"      LP it{it}: obj={best_obj:.8f}, t={t_opt:.3f}, "
                      f"std/mn={sd/mn:.4f}")
        else:
            no_improve += 1
            if no_improve >= 4:
                break

    return best_a, best_obj, it


def spectral_analysis(a):
    """Analyze frequency content of the sequence."""
    n = len(a)
    A = fft(a, 2 * n)
    magnitude = np.abs(A[:n])
    # Spectral flatness measure
    log_mag = np.log(magnitude + 1e-10)
    arithmetic_mean = np.mean(log_mag)
    geometric_mean = np.exp(np.mean(np.log(magnitude + 1e-10)))
    flatness = geometric_mean / (arithmetic_mean + 1e-10)
    return flatness, magnitude


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP polishing with adaptive line search + QP uniformity +
    dual-guided perturbations + multi-n exploration + spectral analysis.
    Returns list of non-negative floats minimizing the evaluation objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # Initial spectral analysis
    flatness, mag = spectral_analysis(best_seq)
    print(f"Initial spectral flatness: {flatness:.4f}")

    # === Phase 1: Quick smoothed GD warm start ===
    print(f"\n=== Phase 1: Smoothed GD Warm Start ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 80.0)
    if p1_budget > 15:
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=p1_budget,
                                          tau_init=5.0, tau_end=300.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase1 SGD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 SGD: obj={obj_sgd:.8f}")

    # === Phase 2: Intensive LP + QP polishing ===
    print(f"\n=== Phase 2: LP+QP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 500.0)
    if p2_budget > 30:
        a_current = best_seq.copy()

        # Multiple rounds with increasing intensity
        for rnd in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = p2_budget / (6 - rnd) * 0.8
            alloc = min(alloc, remaining * 0.6)
            if alloc < 15:
                break

            use_qp = rnd >= 2  # Start using QP in later rounds
            a_new, obj_new, iters = lp_polish_round(a_current, max_iter=150,
                                                     time_limit=alloc,
                                                     verbose=(rnd == 0),
                                                     use_qp=use_qp)
            if obj_new < best_score - 1e-13:
                best_score = obj_new
                best_seq = a_new.copy()
                a_current = a_new.copy()
                print(f"  LP+QP round {rnd}: obj={best_score:.8f} ({iters} iters) **")
            else:
                print(f"  LP+QP round {rnd}: obj={obj_new:.8f} ({iters} iters)")

                # Perturb with dual-guided approach
                method = int(rng.integers(0, 10))
                scale = 0.015 + 0.01 * rnd
                a_current = perturb_sequence(a_new, rng, method=method, scale=scale)

                # Quick polish after perturbation
                remaining = deadline - time.monotonic()
                alloc2 = min(alloc * 0.4, remaining * 0.3)
                if alloc2 > 10:
                    a_pert, obj_pert, _ = lp_polish_round(a_current, max_iter=40,
                                                           time_limit=alloc2, use_qp=use_qp)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        a_current = a_pert.copy()
                        print(f"  LP round {rnd} (perturbed m{method}): obj={best_score:.8f} **")

    # === Phase 3: Dual-guided escape attempts ===
    print(f"\n=== Phase 3: Dual-Guided Escape ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 200.0)
    if p3_budget > 30:
        # Solve LP to get duals, then use them for targeted perturbation
        a_for_dual = best_seq.copy()
        g_lp, M_lp, duals = solve_lp_minpeak(a_for_dual)

        for attempt in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = p3_budget / (8 - attempt) * 0.7
            alloc = min(alloc, remaining * 0.5)
            if alloc < 10:
                break

            if duals is not None:
                a_pert = dual_guided_perturbation(a_for_dual, duals, rng, scale=0.02 + 0.005 * attempt)
            else:
                method = attempt % 10
                a_pert = perturb_sequence(a_for_dual, rng, method=method, scale=0.02)

            a_new, obj_new, iters = lp_polish_round(a_pert, max_iter=60,
                                                     time_limit=alloc, use_qp=True)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a_for_dual = a_new.copy()
                # Re-solve for new duals
                g_lp, M_lp, duals = solve_lp_minpeak(a_for_dual)
                print(f"  Dual escape {attempt}: obj={best_score:.8f} **")
            else:
                print(f"  Dual escape {attempt}: obj={obj_new:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        n_current - 100, n_current - 50, n_current + 50,
        n_current + 100, n_current + 200, n_current + 300,
        n_current + 500, n_current + 800,
        500, 600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600, 2000
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2500 and n != n_current]
    n_options.sort(key=lambda x: abs(x - n_current))

    for n_try in n_options[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick GD
        alloc1 = min(remaining * 0.05, 30.0)
        if alloc1 > 10:
            a_sgd = nesterov_smoothed_descent(a_try, time_budget=alloc1,
                                              tau_init=10.0, tau_end=200.0)
            if compute_obj(a_sgd) < compute_obj(a_try):
                a_try = a_sgd

        # LP+QP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.08, 60.0)
        if alloc2 > 10:
            a_try, obj_try, _ = lp_polish_round(a_try, max_iter=50,
                                                 time_limit=alloc2, use_qp=True)
        else:
            obj_try = compute_obj(a_try)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Final intensive refinement ===
    print(f"\n=== Phase 5: Final Refinement ===")
    remaining = deadline - time.monotonic()

    # High-precision LP polishing (multiple rounds)
    a_final = best_seq.copy()
    for fr in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining * 0.12, 60.0)
        if alloc < 10:
            break

        if fr > 0 and fr % 2 == 1:
            a_final = perturb_sequence(a_final, rng, method=fr % 10, scale=0.008)

        a_new, obj_new, iters = lp_polish_round(a_final, max_iter=120,
                                                 time_limit=alloc, use_qp=True)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_final = a_new.copy()
            print(f"  Final {fr}: obj={best_score:.8f} ({iters} iters) **")
        else:
            print(f"  Final {fr}: obj={obj_new:.8f} ({iters} iters)")
            if fr > 3:
                break

    # Final high-tau GD
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 50.0)
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                          tau_init=100.0, tau_end=2000.0)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Final SGD: obj={best_score:.8f} **")
        else:
            print(f"  Final SGD: obj={obj_sgd:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    # Final spectral analysis
    final_flatness, final_mag = spectral_analysis(np.array(result))
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, "
          f"flatness={final_flatness:.4f}, time={elapsed:.1f}s ===")

    return result
