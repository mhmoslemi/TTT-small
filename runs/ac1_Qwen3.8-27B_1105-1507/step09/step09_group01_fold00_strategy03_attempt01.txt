def golden_section_line_search(f, low, high, tol=0.1, max_iter=40):
    # Avoids scientific notation and uses precise constraints
    # Returns x such that f(x) is minimized in [low, high]
    gr = (np.sqrt(5) - 1) / 2
    c = high - gr*(high-low)
    d = low + gr*(high-low)
    fc = f(c)
    fd = f(d)
    mid_val = (low+high)/2
    for _ in range(max_iter):
        if abs(high-low) < tol:
            break

            d = low + gr*(high-low)
            fd_new = f(d)

                low = mid_val
        elif fc > fd:

def compute_conv_derivative(a, mode='full'):
    # Analytic gradient for the convolution objective
    n = len(a)
    conv = np.convolve(a, a, mode=mode)
    conv_full = np.pad(conv, n, mode='constant', constant_values=0)
    conv_mid = conv_full[numdiff]
    # Use central difference for gradient computation
    d = 1.0
    grad = np.zeros(n)
    for i in range(n):
        # Central difference for interior points, forward for boundaries
        if i == 0:
            perturbed = a.copy()
            perturbed[0] += d
            conv_pert = np.convolve(perturbed, a, mode='full')
            grad[i] = (conv_pert[0] - conv_pert[0-n+2])/d
        elif i == n-1:
            perturbed = a.copy()
            perturbed[i] += d
            conv_pert = np.convolve(a, perturbed, mode='full')
            # Using the symmetry of the convolution
            grad[i] = (conv_pert[2*n-1] - conv_pert[0])/d
        else:
            perturbed = a.copy()
            perturbed[i] += d
            conv_pert = np.convolve(a, perturbed, mode='full')
            grad[i] = (conv_pert[i+1] - conv_pert[i-1])/ (2*d)
    return grad, conv

def tensioned_optimization(a, alpha=0.3):
    # Adaptive smoothing parameter bringing in outside restraints
    # This function reshapes the sequence using tension parameters
    n = len(a)
    b = a + 0
    for i in range(1, n-1):
        b[i] = b[i] * alpha + 0.7 * (b[i+1] + b[i-1]) * 0.3
    # Apply sigmoid transformation for sharper transitions
    for j in range(n):
        b[j] = 1/(1+np.exp(-b[j]))
    # Normalize
    total = np.sum(b)
    return b / total
