```python
def solve_lp_fast(a):
    n = len(a)
    a_norm = a / np.sum(a)
    # Build A matrix (2n-1 x n)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        r = min(n, 2*n-1-j)
        A[j:j+r, j] = a_norm[:r]
    
    # LP: min M
    # Variables: [g_0, ..., g_{n-1}, M]
    # c = [0, ..., 0, 1]
    c = np.zeros(n+1)
    c[-1] = 1
    
    # Inequality constraints: A@g - M <= 0 for each k
    # => A@g + [-1, -1, ..., -1]*M <= 0
    # Matrix form: [A | -1_{(2n-1)x1}] @ [g; M] <= 0
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) + 0*M = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1
    b_eq = np.array([1.0])
    
    # Bounds: g_i >= 0, M unbounded (but will be positive)
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if result.success:
        g = np.maximum(result.x[:n], 0)
        M = result.x[-1]
        s = g.sum()
        if s > 0:
            g = g / s
        return g, M
    return None, None
```

**Mathematical insight**: The objective 2n*max(conv(a,a))/(sum(a))^2 is related to the "aperiodic autocorrelation" problem. In signal processing, the goal is to design a sequence with low sidelobes. For non-negative sequences, this is related to "M-sequence" or "Barker sequence" design.

One known result: for a sequence of length n with values in {0,1}, the minimum peak sidelobe is O(sqrt(n)). For continuous non-negative sequences, we can do better.

The theoretical minimum of max(conv(a,a)) for sum(a)=1, a>=0, length n is at least 1/(2n-1) (by averaging), achieved only if conv is perfectly flat. For non-negative a, perfect flatness is impossible (I believe), but we can get close.

Actually, wait. Can we achieve perfectly flat convolution? We need (conv(a,a))_k = 1/(2n-1) for all k. This means the autocorrelation is a "boxcar" function. In the frequency domain, this means |A(f)|^2 = (2n-1) * sinc^2(f) or something like that... Actually the DFT of a boxcar of length 2n-1 is a Dirichlet kernel. So we'd need |A(f)|^2 to be proportional to the Dirichlet kernel, which means A(f) would need to be complex-valued (phase matters). For real non-negative a, A(f) is real and non-negative (well, not necessarily non-negative, but real), so |A(f)|^2 = A(f)^2. This means A(f) = ±sqrt(Dirichlet kernel), which is not a valid DFT of a non-negative sequence (I think).

So perfect flatness is impossible, and the question is how close we can get.

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast HiGHS-based LP polishing, multi-scale n search, diverse initialization,
and adaptive perturbation restarts. Key improvement over previous: much faster LP
solves via scipy HiGHS, broader n exploration, and fixed-point detection.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses scipy HiGHS for speed.
    Returns (g, M) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: A@g - M <= 0  =>  [A | -1] @ [g; M] <= 0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs')
        if result.success:
            g = np.maximum(result.x[:n], 0)
            M = float(result.x[-1])
            s = float(np.sum(g))
            if s > 1e-15:
                g = g / s
            return g, M
    except Exception as e:
        if verbose:
            print(f"    LP error: {e}")
    
    return None, None

def lp_polish(a, time_budget=100.0, verbose=False, seed=42):
    """
    LP-based polishing: iteratively solve LP, blend with golden-section.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    iter_count = 0
    no_improve = 0
    last_t = 0.5
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 5:
            break
        
        g_lp, M_lp = solve_lp_minpeak(best_a)
        if g_lp is None:
            break
        
        # Check fixed point: if max(conv(a,a)) ≈ M_lp and a ≈ g_lp
        conv_a = compute_conv(best_a)
        max_conv_a = float(np.max(conv_a))
        
        if abs(max_conv_a - M_lp) < 1e-10 * max(1, M_lp):
            # Very close to fixed point
            no_improve += 1
            if no_improve > 3:
                # Try perturbation
                scale = 0.01 * min(no_improve, 10)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                best_a = a_pert.copy()
                no_improve = 0
                continue
        
        # Golden section search for blend factor
        def blend_obj(t):
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return 1e10
            a_try = a_try / s_try
            return compute_obj(a_try)
        
        # Use fewer iterations for speed
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a_lo, a_hi = 0.0, 1.0
        c = a_hi - gr * (a_hi - a_lo)
        d = a_lo + gr * (a_hi - a_lo)
        fc = blend_obj(c)
        fd = blend_obj(d)
        for _ in range(15):
            if abs(a_hi - a_lo) < 0.02:
                break
            if fc < fd:
                a_hi = d
                d = c
                fd = fc
                c = a_hi - gr * (a_hi - a_lo)
                fc = blend_obj(c)
            else:
                a_lo = c
                c = d
                fc = fd
                d = a_lo + gr * (a_hi - a_lo)
                fd = blend_obj(d)
        t_opt = (a_lo + a_hi) / 2.0
        
        a_try = (1.0 - t_opt) * best_a + t_opt * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            break
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        
        iter_count += 1
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            last_t = t_opt
            no_improve = 0
            if verbose and iter_count % 5 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={t_opt:.3f}, M={M_lp:.6f}")
        else:
            no_improve += 1
            if no_improve >= 5:
                scale = 0.008 * min(no_improve, 8)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                best_a = a_pert.copy()
                no_improve = 0
                if iter_count > 40:
                    break
    
    return best_a

def subgradient_refine(a, time_budget=80.0, K=8, verbose=False):
    """Subgradient method with top-K peaks, momentum, adaptive step."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 2.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        # Sharp weights on top peaks
        weights = np.exp((peak_vals - max_pv) * 10.0 / max_pv)
        weights /= weights.sum()
        
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]
        
        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break
            else:
                lr *= 0.997
        
        it += 1
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like (1/sqrt)
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        # Cosine bell
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Sinc-like
        x = np.arange(n) - n / 2
        a = np.sinc(x / (n * 0.3))
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 7:
        # Comb pattern
        period = max(2, n // 20)
        a = np.ones(n)
        a[::period] = 2.0
        a = a / a.sum()
    elif mode == 8:
        # Smooth random (low-pass filtered)
        a = rng.uniform(0, 1, n)
        # Simple moving average smoothing
        k = max(1, n // 50)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = a / a.sum()
    elif mode == 9:
        # Airy-like (optimal for some related problems)
        x = np.linspace(0, 1, n)
        a = x ** 0.5 * (1 - x) ** 0.5  # Beta(1.5, 1.5)-like
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

def coordinate_descent_peak(a, time_budget=30.0, verbose=False):
    """
    Direct coordinate descent: at each step, find the coordinate whose
    change most reduces the current peak convolution value.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    conv = compute_conv(a)
    peak_k = int(np.argmax(conv))
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Recompute peak periodically
        if it % 10 == 0:
            conv = compute_conv(a)
            peak_k = int(np.argmax(conv))
        
        # Gradient of conv[peak_k] w.r.t. a_i:
        # d(conv[peak_k])/da_i = 2 * a[peak_k - i] (if valid)
        i_start = max(0, peak_k - (n - 1))
        i_end = min(n - 1, peak_k)
        
        # The "partner" values
        partners = np.zeros(n)
        if i_start <= i_end:
            j_start = peak_k - i_end
            j_end = peak_k - i_start
            partners[i_start:i_end+1] = a[j_start:j_end+1][::-1]
        
        # To reduce conv[peak_k], we want to decrease a_i where partners[i] is large
        # and increase a_i where partners[i] is small.
        # Direction: -partners (decrease where partner is large)
        # But we need to maintain sum(a) = 1 and a >= 0.
        # Projected gradient step:
        direction = -partners
        # Add small lifting to avoid all going to 0
        direction += 0.1 * np.mean(partners)
        
        # Try a few step sizes
        improved = False
        for lr in [0.01, 0.005, 0.001, 0.0005]:
            a_try = a + lr * direction
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        
        if not improved:
            # Try random coordinate swap
            i = rng.integers(0, n) if 'rng' in dir() else np.random.randint(n)
            j = np.random.randint(n)
            if i != j:
                delta = 0.001 * a[i]
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] >= 0 and a_try[j] >= 0:
                    s_try = a_try.sum()
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
        
        if not improved and it > 100:
            break
        it += 1
    
    return best_a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast scipy HiGHS LP polishing, aggressive multi-n exploration, diverse
initialization, and adaptive perturbation restarts. Key improvement: much faster
LP solves via HiGHS (vs cvxpy), broader n search, and smarter time allocation.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses scipy HiGHS for speed. Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(-np.inf, np.inf)]

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'maxiter': 5000, 'dual_feasibility_tolerance': 1e-9,
                                  'primal_feasibility_tolerance': 1e-9})
        if result.success or hasattr(result, 'status') and result.status == 1:
            g = np.maximum(result.x[:n], 0)
            M = float(result.x[-1])
            s = float(np.sum(g))
            if s > 1e-15:
                g = g / s
            return g, M
    except Exception:
        pass

    return None, None


def lp_polish(a, time_budget=100.0, verbose=False, seed=42):
    """
    LP-based polishing: iteratively solve LP, blend with adaptive step.
    Uses fixed-point detection and perturbation restarts.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    no_improve = 0
    last_t = 0.5
    fail_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 3:
            break

        g_lp, M_lp = solve_lp_minpeak(best_a)
        if g_lp is None:
            fail_count += 1
            if fail_count > 3:
                break
            continue
        fail_count = 0

        # Check fixed point
        conv_a = compute_conv(best_a)
        max_conv_a = float(np.max(conv_a))

        # If very close to fixed point, perturb
        if abs(max_conv_a - M_lp) < 1e-8 * max(1.0, M_lp):
            no_improve += 3
        if no_improve >= 3:
            scale = 0.01 * min(no_improve, 12)
            a_pert = best_a * np.exp(rng.normal(0, scale, n))
            a_pert = np.maximum(a_pert, 0)
            s_p = float(np.sum(a_pert))
            if s_p < 1e-10:
                a_pert = np.ones(n) / n
            else:
                a_pert = a_pert / s_p
            best_a = a_pert.copy()
            no_improve = 0
            continue

        # Try several blend factors (faster than full golden section)
        best_t = last_t
        best_blend_obj = compute_obj(best_a)
        for t in [0.3, 0.5, 0.7, 0.85, 1.0, last_t * 0.5, last_t * 1.5]:
            t = min(max(t, 0.05), 1.0)
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_t = t

        iter_count += 1
        if best_blend_obj < best_obj - 1e-14:
            best_obj = best_blend_obj
            a_new = (1.0 - best_t) * best_a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            best_a = a_new.copy()
            last_t = best_t
            no_improve = 0
            if verbose and iter_count % 3 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t:.3f}, M={M_lp:.6f}")
        else:
            no_improve += 1

                scale = 0.01 * min(no_improve, 10)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                best_a = a_pert.copy()
                no_improve = 0
                if iter_count > 50:
                    break

    return best_a


def subgradient_refine(a, time_budget=60.0, K=8, verbose=False):
    """Subgradient method with top-K peaks, momentum, adaptive step."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.5 / np.sqrt(n)
    momentum = 0.75
    v = np.zeros(n)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]

        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        weights = np.exp((peak_vals - max_pv) * 8.0 / max_pv)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 150:
                lr *= 0.7
                v *= 0.5
                stall = 0
                if lr < 1e-12:
                    break
            else:
                lr *= 0.998

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:

        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:

        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 5:

        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        x = np.arange(n) - n / 2.0
        a = np.sinc(x / (n * 0.3))
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    elif mode == 7:
        period = max(2, n // 15)
        a = np.ones(n)
        a[::period] = 1.5
        a = a / a.sum()
    elif mode == 8:
        a = rng.uniform(0, 1, n)
        k = max(1, n // 40)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = a / a.sum()
    elif mode == 9:
        x = np.linspace(0, 1, n)
        a = np.sqrt(x * (1 - x) + 1e-10)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

            print(f"  Pert {r}: obj={best_score:.8f} **")
        else:
            print(f"  Pert {r}: obj={obj_pert:.8f}")

    # Phase 7: Final intensive polish
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.6, 120.0)
        a_final = lp_polish(best_seq, time_budget=alloc, verbose=True, seed=9999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # Final subgradient
    remaining = deadline - time.monotonic()
    if remaining > 8:
        alloc = min(remaining * 0.3, 25.0)
        a_sg = subgradient_refine(best_seq, time_budget=alloc, K=12)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
