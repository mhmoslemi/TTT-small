import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv

def multiplicative_peak_balance(a, n_iter=100, alpha=0.5, verbose=False):
    """
    Multiplicative update to balance the convolution peak.
    At each step, compute the 'pressure' on each coordinate and reduce it.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(n_iter):
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        # Compute pressure: for each j, how much it contributes to high-conv positions
        # p[j] = sum_k w[k] * a[k-j] where w[k] emphasizes high conv

        w = np.where(conv > threshold, conv, 0.0)
        w = w / float(np.sum(w) + 1e-30)
        
        # pressure[j] = sum_k w[k] * a[k-j] (valid indices)
        # This is a cross-correlation
        N = 1
        while N > len(w) + n - 1:
            N >>= 1
        N = max(N, 1)
        while N < len(w) + n - 1:
            N <<= 1
        W = fft(w, N)
        A = fft(a, N)
        pressure_full = ifft(W * np.conj(A)).real

        pressure = np.maximum(pressure, 0)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, active_only=False, threshold_frac=0.95):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    If active_only, only constrain conv positions above threshold_frac * max.
    Returns (g_optimal, M_optimal) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    if active_only:
        conv_mask = compute_conv(a_norm)
        max_c = float(np.max(conv_mask))
        active = conv_mask >= threshold_frac * max_c
        A = A[active, :]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def peak_shave(a, n_rounds=50, verbose=False):
    """
    Direct peak shaving: identify coordinates contributing most to the peak
    convolution and redistribute their mass to low-pressure coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    for rnd in range(n_rounds):
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        k_peak = int(np.argmax(conv))

        # Find which coordinates contribute most to the peak
        contributions = np.zeros(n)
        for j in range(n):
            idx = k_peak - j
            if 0 <= idx < n:
                contributions[j] = a[j] * a[idx]
            else:
                contributions[j] = 0.0
        # Also count contribution to near-peak positions
        n_near = max(3, n // 50)
        top_ks = np.argsort(conv)[::-1][:n_near]
        for k in top_ks[1:]:
            k = int(k)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    contributions[j] += 0.3 * a[j] * a[idx]

        total_contrib = float(np.sum(contributions))
        if total_contrib < 1e-15:
            break

        # Identify "donor" coordinates (high contribution) and "receiver" (low contribution)
        threshold_hi = np.percentile(contributions, 70)
        threshold_lo = np.percentile(contributions, 30)
        donors = contributions > threshold_hi
        receivers = contributors_lo = contributions < threshold_lo

        if not np.any(donors) or not np.any(receivers):
            break

        # Redistribute: take mass from donors, give to receivers proportional to a
        donor_mass = float(np.sum(a[donors])) * 0.05
        recv_a = a[receivers].copy()
        recv_total = float(np.sum(recv_a))
        if recv_total < 1e-15:
            break

        a[donors] -= donor_mass * a[donors] / float(np.sum(a[donors]))
        a[receivers] += donor_mass * recv_a / recv_total
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            break
        a = a / s

        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
        elif verbose and rnd % 10 == 0:
            print(f"    PS rnd{rnd}: obj={obj:.8f} (best={best_obj:.8f})")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / (a.sum() + 1e-30)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
        a = a / (a.sum() + 1e-30)
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = a / (a.sum() + 1e-30)
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = a / (a.sum() + 1e-30)
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = 1.0 / (x + 1.0 - x)  # uniform but computed differently
        freq = rng.integers(2, 8)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        a = a / (a.sum() + 1e-30)
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines mirror descent, Nesterov smoothed descent,
    LP polishing, peak shaving, and multi-n exploration.
    Returns list of non-negative floats minimizing the evaluation objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Mirror descent (new approach) ===
    print(f"\n=== Phase 1: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.12, 120.0)
    if p1_budget > 30:
        a_md = mirror_descent(best_seq, time_budget=p1_budget,
                              tau_init=5.0, tau_end=300.0, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Phase1 MD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 MD: obj={obj_md:.8f}")

    # === Phase 2: Nesterov smoothed descent ===
    print(f"\n=== Phase 2: Nesterov Smoothed Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 150.0)
    if p2_budget > 30:
        a_nsd = nesterov_smoothed_descent(best_seq, time_budget=p2_budget,
                                          tau_init=8.0, tau_end=500.0, verbose=True)
        obj_nsd = compute_obj(a_nsd)
        if obj_nsd < best_score:
            best_score = obj_nsd
            best_seq = a_nsd.copy()
            print(f"  Phase2 NSD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 NSD: obj={obj_nsd:.8f}")

    # === Phase 3: LP Polishing (intensive) ===
    print(f"\n=== Phase 3: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.30, 300.0)
    if p3_budget > 30:
        a_current = best_seq.copy()
        n_rounds = 6
        round_budget = p3_budget / n_rounds

        for rnd in range(n_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(round_budget, remaining * 0.45)
            if alloc < 15:
                break

            use_active = (rnd >= 2)  # Use active-set LP after a few rounds
            a_new, obj_new, iters = lp_polish_round(a_current, max_iter=80,
                                                     time_limit=alloc, verbose=(rnd == 0),
                                                     use_active=use_active)
            if obj_new < best_score - 1e-12:
                best_score = obj_new
                best_seq = a_new.copy()
                a_current = a_new.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} ({iters} iters) **")
            else:
                print(f"  LP round {rnd}: obj={obj_new:.8f} ({iters} iters)")
                # Perturb and try again
                a_pert = a_current * np.exp(rng.normal(0, 0.02 + 0.01 * rnd, len(a_current)))
                a_pert = np.maximum(a_pert, 0)
                s_p = float(np.sum(a_pert))
                if s_p > 1e-10:
                    a_pert = a_pert / s_p
                    remaining = deadline - time.monotonic()
                    alloc2 = min(alloc * 0.5, remaining * 0.2)
                    if alloc2 > 10:
                        a_pert2, obj_pert, _ = lp_polish_round(a_pert, max_iter=30,
                                                               time_limit=alloc2)
                        if obj_pert < best_score:
                            best_score = obj_pert
                            best_seq = a_pert2.copy()
                            a_current = a_pert2.copy()
                            print(f"  LP round {rnd} (perturbed): obj={best_score:.8f} **")

    # === Phase 4: Peak Shaving ===
    print(f"\n=== Phase 4: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_ps = peak_shave(best_seq, n_rounds=100, verbose=True)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  Phase4 PS: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 PS: obj={obj_ps:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 100), max(400, n_current - 50),
        max(400, n_current - 200),
        n_current + 50, n_current + 100, n_current + 200, n_current + 400,
        500, 600, 700, 800, 900, 1000, 1200, 1400
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1600 and n != n_current]
    n_options.sort(key=lambda x: abs(x - n_current))

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick mirror descent
        alloc1 = min(remaining * 0.08, 40.0)
        if alloc1 > 10:
            a_try = mirror_descent(a_try, time_budget=alloc1,
                                   tau_init=10.0, tau_end=300.0)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.12, 60.0)
        if alloc2 > 10:
            a_try, obj_try, _ = lp_polish_round(a_try, max_iter=40, time_limit=alloc2)
        else:
            obj_try = compute_obj(a_try)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau mirror descent
    if remaining > 40:
        alloc = min(remaining * 0.2, 60.0)
        a_md2 = mirror_descent(best_seq, time_budget=alloc,
                               tau_init=50.0, tau_end=800.0)
        obj_md2 = compute_obj(a_md2)
        if obj_md2 < best_score:
            best_score = obj_md2
            best_seq = a_md2.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # Multiple LP polish rounds
    for lp_rnd in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * 0.2, 60.0)
        if alloc < 10:
            break
        a_current = best_seq.copy()

        if lp_rnd > 0:
            # Small perturbation
            a_current = a_current * np.exp(rng.normal(0, 0.01, len(a_current)))
            a_current = np.maximum(a_current, 0)
            s_c = float(np.sum(a_current))
            if s_c > 1e-10:
                a_current = a_current / s_c

        a_lp, obj_lp, iters = lp_polish_round(a_current, max_iter=60,
                                               time_limit=alloc, use_active=True)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP {lp_rnd}: obj={best_score:.8f} ({iters} iters) **")
        else:
            print(f"  Final LP {lp_rnd}: obj={obj_lp:.8f}")

    # === Phase 7: Diverse restarts (if time remains) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 7: Diverse Restarts ===")
        n_best = len(best_seq)
        for r in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            mode = r
            a_r = diverse_init(n_best, rng, mode=mode)
            blend = 0.2 + 0.1 * r
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r

            alloc_md = min(remaining * 0.15, 40.0)
            if alloc_md > 15:
                a_r = mirror_descent(a_r, time_budget=alloc_md,
                                     tau_init=10.0, tau_end=200.0)
                remaining = deadline - time.monotonic()
                alloc_lp = min(remaining * 0.1, 30.0)
                if alloc_lp > 10:
                    a_r, obj_r, _ = lp_polish_round(a_r, max_iter=30, time_limit=alloc_lp)
                else:
                    obj_r = compute_obj(a_r)
                if obj_r < best_score:
                    best_score = obj_r
                    best_seq = a_r.copy()
                    print(f"  Restart {r}: obj={best_score:.8f} **")
                else:
                    print(f"  Restart {r}: obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
