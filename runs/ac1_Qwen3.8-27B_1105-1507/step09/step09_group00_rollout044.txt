import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, M_cap=None, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (optimal_g, optimal_M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None


def solve_lp_weighted(a, weights, verbose=False):

def line_search_lp(a, g, max_iter=30):
    """
    Proper line search along direction (1-t)*a + t*g.
    Uses golden section search to find optimal t.

            if o1 < o2:
                hi = t2
            else:
                lo = t1
            if hi - lo < 1e-5:

        best_t = (lo + hi) / 2
        a_best = np.maximum((1 - best_t) * a + best_t * g, 0)
        s = float(np.sum(a_best))
        if s > 1e-10:

def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1

    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]

    return g, conv


def accelerated_smoothed_descent(a, time_budget=300.0, tau_init=10.0, tau_end=1000.0, verbose=False):
    """
    Nesterov accelerated gradient descent on log-sum-exp smoothed objective.
    With adaptive learning rate and cosine tau schedule.
    """

    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 3.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Cosine schedule for tau
        tau = tau_init + (tau_end - tau_init) * (1 - np.cos(np.pi * progress)) / 2

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break

        # Adaptive LR
        lr = min(max(1.0 / (g_norm + 1e-8), 0.05 * base_lr), 2.0 * base_lr)

        # Update
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr *= 0.9999

        # Reset if stuck
        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it

        if verbose and it % 2000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")

    """
    LP polishing with proper line search and adaptive strategies.
    Key improvement: uses golden section line search and weighted LP variants.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall_count = 0
    iter_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        # Get LP direction
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            # Try weighted LP
            conv = compute_conv(a / np.sum(a))
            max_c = np.max(conv)
            weights = np.exp(5.0 * (conv - max_c))
            weights = weights / np.sum(weights)
            g_lp = solve_lp_weighted(a, weights)
            if g_lp is None:
                break
            M_lp = float(np.max(compute_conv(a / np.sum(a) * 0 + g_lp)))

        iter_count += 1

        # Line search
        best_t, best_new_obj = line_search_lp(a, g_lp)

        if best_new_obj < best_obj - 1e-14:
            best_obj = best_new_obj
            best_a = np.maximum((1 - best_t) * a + best_t * g_lp, 0)
            s = float(np.sum(best_a))
            if s > 1e-10:
                best_a = best_a / s
            a = best_a.copy()
            stall_count = 0
            if verbose and iter_count % 5 == 0:
                print(f"    LP+LS it{iter_count}: obj={best_obj:.8f}, t={best_t:.4f}, M_lp={M_lp:.8f}")
        else:
            stall_count += 1
            # Try perturbation
            if stall_count >= 2:
                # Add small noise to escape
                noise = rng.normal(0, 0.01 / np.sqrt(n), n)
                a_pert = a * np.exp(noise)
                a_pert = np.maximum(a_pert, 0)
                s_p = float(np.sum(a_pert))
                if s_p > 1e-10:
                    a = a_pert / s_p
                stall_count = 0

                # Try weighted LP with emphasis on peak positions
                conv = compute_conv(a / np.sum(a))
                max_c = np.max(conv)
                # Heavy weight on peaks
                weights = np.where(conv > max_c * 0.95, 1.0, 0.01)
                weights = weights / np.sum(weights)
                g_w = solve_lp_weighted(a, weights)
                if g_w is not None:
                    t_w, obj_w = line_search_lp(a, g_w)
                    if obj_w < best_obj - 1e-14:
                        best_obj = obj_w
                        best_a = np.maximum((1 - t_w) * a + t_w * g_w, 0)
                        s = float(np.sum(best_a))
                        if s > 1e-10:
                            best_a = best_a / s
                        a = best_a.copy()

                if iter_count > 50:
                    break

    return best_a


def fourier_kick(a, rng, strength=0.03):
    """Add structured Fourier perturbation to escape local structure."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 4
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
        pert += amp * 0.5 * np.sin(freq * np.arange(n) + phase + 0.7)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 3:
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        a = np.exp(-np.abs(np.arange(n) - n / 2) / (n / 4))
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 4 * np.abs(x - 0.5), 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Spectral init
        size = 1
        while size < 2 * n:
            size <<= 1
        freqs = np.fft.rfftfreq(size)
        mag = np.exp(-freqs * n * 0.3)
        phases = rng.uniform(0, 2 * np.pi, size=size // 2 + 1)
        F = mag * np.exp(1j * phases)
        F[0] = abs(F[0])
        a = np.fft.irfft(F, size)[:n]
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            return np.ones(n) / n
        a = a / s
    elif mode == 7:
        # Concentrated at edges
        x = np.linspace(0, 1, n)
        a = x**2 + (1-x)**2 + 0.1
        a = a / a.sum()
    elif mode == 8:
        # Concentrated at center
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5)**2) / 0.05)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def kkt_gap(a, tol=1e-4):
    """
    Estimate the KKT gap for the convex problem.
    Returns the maximum violation of optimality conditions.
    """
    n = len(a)
    a = a / np.sum(a)
    conv = compute_conv(a)
    max_val = np.max(conv)

    # Compute gradient at each critical position
    # grad_k[j] = a[k-j] + a[j-k] (with bounds checking)
    critical = np.where(conv >= max_val - tol)[0]

    # The subgradient is the convex hull of {2*P_k(a) : k in critical}
    # where P_k shifts a
    # For a point a to be optimal on the simplex, we need:
    # There exists mu such that for all i: sum_k lambda_k * grad_k[i] + mu >= 0
    # and for i with a[i]=0: sum_k lambda_k * grad_k[i] + mu = 0 (complementary slackness)
    # Simplified: just check if the directional derivative in some direction is negative

    # Compute gradient of smoothed max
    tau = 50.0
    exp_vals = np.exp(tau * (conv - max_val))
    p = exp_vals / np.sum(exp_vals)

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]

    # Project gradient onto tangent space of simplex
    g_proj = g - np.sum(g * a)  # Remove component along 'a' direction
    # Actually for simplex, the normal cone condition is:
    # g[i] <= mu for all i with a[i] > 0 (mu = max of g[i] for a[i]>0)
    # g[i] <= mu for all i with a[i] = 0
    # The violation is: max_i (g[i] - mu) where mu = max(g[i] for a[i]>0)

    active = a > 1e-10
    if np.any(active):
        mu = np.max(g[active])
        violation = np.max(g) - mu
    else:
        violation = 0

    return max(0, violation)
