"""

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.maximum(v, 0)
    s = float(np.sum(v))
    if s < 1e-15:
        return np.ones(n) / n
    return v / s


def smoothed_obj_grad(a, tau):
    """
    Compute smoothed objective (log-sum-exp / tau) and gradient.
    a is assumed to be on the simplex (sum=1, a>=0).
    Returns (obj, grad) where grad has same shape as a.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    obj = max_conv + np.log(sum_exp) / tau
    p = exp_vals / sum_exp

    # Gradient: 2 * cross_correlation(p, a)
    # cross[i] = sum_j p[i+j] * a[j]
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(np.conj(A) * P, N).real[:n]
    grad = 2.0 * cross
    return obj, grad


def lbfgs_smoothed_optimize(a, time_budget=200.0, verbose=False, seed=42):
    """
    L-BFGS-B optimization on smoothed objective with annealed tau.
    Works on the simplex using the simplex as the domain (with projection).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Anneal tau from low to high
    tau_start = 5.0
    tau_end = 2000.0
    total_time = time_budget
    phase1_time = total_time * 0.4
    phase2_time = total_time * 0.35
    phase3_time = total_time * 0.25

    # Phase 1: Low tau (smoother landscape), more iterations
    tau1 = tau_start
    elapsed = time.monotonic() - start

    def rounded_obj_grad(x, tau):
        x_pos = np.maximum(x, 0)
        s = float(np.sum(x_pos))
        if s < 1e-10:
            return 1e10, np.zeros_like(x)
        x_norm = x_pos / s
        obj, grad = smoothed_obj_grad(x_norm, tau)
        # Chain rule: d/dx of f(x/max(x,0)/sum)
        # For simplicity, just use the projected gradient
        return obj, grad

    # Use a parameterization: x = exp(y) / sum(exp(y)) (softmax) to handle positivity
    # But for L-BFGS-B, it's easier to work with x directly and use bounds [0, inf]
    # with a penalty for sum != 1, or use the simplex constraint

    # Actually, let's work on the simplex directly using the "log-barrier" approach:
    # Minimize f(a) + mu * sum(-log(a_i))  for a_i > 0
    # This keeps us in the interior

    def barrier_obj_grad(x, tau, mu):
        x_pos = np.maximum(x, 1e-15)
        s = float(np.sum(x_pos))
        x_norm = x_pos / s
        obj, grad = smoothed_obj_grad(x_norm, tau)
        # Barrier term
        barrier = -mu * np.sum(np.log(np.maximum(x, 1e-15)))
        barrier_grad = -mu / np.maximum(x, 1e-15)
        return obj + barrier, grad + barrier_grad

    for phase, (phase_budget, tau) in enumerate([
        (phase1_time, tau_start),
        (phase2_time, (tau_start * tau_end) ** 0.5),
        (phase3_time, tau_end)
    ]):
        phase_start = time.monotonic()
        mu = 0.01 / (1.0 + phase * 0.5)

        def obj_fn(x):
            x_pos = np.maximum(x, 1e-15)
            s = float(np.sum(x_pos))
            if s < 1e-10:
                return 1e10
            x_norm = x_pos / s
            o, _ = smoothed_obj_grad(x_norm, tau)
            return o - mu * np.sum(np.log(np.maximum(x, 1e-15)))

        def grad_fn(x):
            x_pos = np.maximum(x, 1e-15)
            s = float(np.sum(x_pos))
            if s < 1e-10:
                return np.zeros(len(x))
            x_norm = x_pos / s
            _, g = smoothed_obj_grad(x_norm, tau)
            return g - mu / np.maximum(x, 1e-15)

        res = minimize(
            obj_fn, a, jac=grad_fn, method='L-BFGS-B',
            bounds=[(1e-10, None)] * n,
            options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-10, 'maxfun': 500}
        )

        if res.x is not None:
            x_new = np.maximum(res.x, 0)
            s_new = float(np.sum(x_new))
            if s_new > 1e-10:
                x_new = x_new / s_new
                obj_new = compute_obj(x_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = x_new.copy()
                    a = x_new.copy()
                    if verbose:
                        print(f"    L-BFGS phase {phase+1} (tau={tau:.1f}): obj={best_obj:.8f}")

        elapsed = time.monotonic() - start
        if elapsed >= total_time:
            break

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, M_target=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue

    return None, None


def lp_alternating_optimize(a, time_budget=200.0, verbose=True, seed=42):
    """
    LP-based alternating (Sinkhorn-like) optimization.
    Alternates: given a, find optimal b via LP, then set a = blend(a, b).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    no_improve = 0
    last_t = 0.5

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start

        # Solve LP with current a
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            if iter_count > 5:
                break
            a = best_a * np.exp(rng.normal(0, 0.01, n))
            a = project_simplex(a)
            continue

        # Try multiple blend factors
        best_t_obj = best_obj
        best_t_val = 0
        for t in [0.1, 0.25, 0.5, 0.75, 1.0]:
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t

        iter_count += 1

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = ((1.0 - best_t_val) * best_a + best_t_val * g_lp)
            best_a = np.maximum(best_a, 0)
            best_a = best_a / best_a.sum()
            a = best_a.copy()
            last_t = best_t_val
            no_improve = 0
            if verbose and iter_count % 3 == 0:
                print(f"    LP-alt it{iter_count}: obj={best_obj:.8f}, t={best_t_val:.2f}, M={M_lp:.6f}")
        else:
            no_improve += 1
            if no_improve == 2:
                # Try perturbation
                scale = 0.005 + 0.003 * min(no_improve, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = project_simplex(a)
                no_improve = 0
            elif no_improve >= 5:
                if iter_count > 10:
                    break
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = project_simplex(a)
                no_improve = 0

        # Periodic: try constrained LP with tighter M
        if iter_count % 6 == 3 and M_lp is not None:
            for tight_frac in [0.93, 0.96, 0.99]:
                g_tight, _ = solve_lp_minpeak(a, M_target=M_lp * tight_frac)
                if g_tight is not None:
                    for t in [0.3, 0.6, 0.9]:
                        a_try = (1.0 - t) * best_a + t * g_tight
                        a_try = np.maximum(a_try, 0)
                        a_try = a_try / a_try.sum()
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try.copy()
                            no_improve = 0
                            if verbose:
                                print(f"    LP-tight it{iter_count}: obj={best_obj:.8f}, frac={tight_frac}")
                            break
                    else:
                        continue
                    break

    return best_a


def coord_descent_optimize(a, time_budget=100.0, verbose=True, seed=42):
    """
    Cyclic coordinate descent with incremental convolution updates.
    For each coordinate, finds the optimal 1D value given all others fixed.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    conv = compute_conv(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    sweep = 0
    while time.monotonic() - start < time_budget:
        old_best_obj = best_obj
        for i in range(n):
            elapsed = time.monotonic() - start
            if elapsed >= time_budget:
                break

            old_val = a[i]

            # Find optimal value for coordinate i
            # conv_k as function of t=a[i]:
            # k = 2i: t^2 + (conv[2i] - old_val^2)  [if 2i < 2n-1]
            # k where 0 <= k-i < n, k != 2i: 2*t*a[k-i] + (conv[k] - 2*old_val*a[k-i])
            # otherwise: conv[k] (constant)

            def one_d_obj(t):
                t = max(0.0, t)
                # Compute new conv values (only those affected by a[i])
                max_v = 0.0
                for k in range(2 * n - 1):
                    if k == 2 * i:
                        val = t * t + (conv[k] - old_val * old_val)
                    elif 0 <= k - i < n:
                        val = 2.0 * t * a[k - i] + (conv[k] - 2.0 * old_val * a[k - i])
                    else:
                        val = conv[k]
                    if val > max_v:
                        max_v = val
                return max_v

            # Use ternary search on [0, 4*old_val + 0.01]
            lo = 0.0
            hi = max(4.0 * old_val + 0.01, 0.01)
            # Check if hi is good enough
            f_hi = one_d_obj(hi)
            f_mid = one_d_obj(hi / 2)
            if f_hi < f_mid:
                lo = hi / 2
                hi *= 2

            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = one_d_obj(c)
            fd = one_d_obj(d)

            for _ in range(20):
                if hi - lo < 1e-10:
                    break
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = one_d_obj(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = one_d_obj(d)

            t_opt = (lo + hi) / 2.0
            t_opt = max(0.0, t_opt)

            # Update a[i] and conv
            delta = t_opt - old_val
            if abs(delta) > 1e-12:
                a[i] = t_opt
                # Update convolution
                for k in range(2 * n - 1):
                    if k == 2 * i:
                        conv[k] += 2 * old_val * delta + delta * delta
                    elif 0 <= k - i < n:
                        conv[k] += 2.0 * a[k - i] * delta
                # Rescale to maintain sum=1
                s_new = float(np.sum(a))
                if abs(s_new - 1.0) > 1e-10:
                    a /= s_new
                    conv /= (s_new * s_new)

        sweep += 1
        obj_now = compute_obj(a)
        if obj_now < best_obj:
            best_obj = obj_now
            best_a = a.copy()
            if verbose and sweep % 5 == 0:
                print(f"    CD sweep{sweep}: obj={best_obj:.8f}")
        elif sweep > 20 and abs(obj_now - old_best_obj) < 1e-12:
            # Stagnation: try perturbation
            a = best_a * np.exp(rng.normal(0, 0.01, n))
            a = project_simplex(a)
            conv = compute_conv(a)
            best_obj = compute_obj(a)
            if verbose:
                print(f"    CD restart sweep{sweep}: obj={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations on the simplex."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * (x - 0.5))
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 3:
        # Gaussian center
        x = np.arange(n)
        a = np.exp(-((x - n / 2) ** 2) / (2 * (n / 4) ** 2))
        a = a / a.sum()
    elif mode == 4:
        # Two Gaussians
        x = np.arange(n)
        a = np.exp(-((x - n * 0.3) ** 2) / (2 * (n * 0.1) ** 2))
        a += np.exp(-((x - n * 0.7) ** 2) / (2 * (n * 0.1) ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Sinc-like (optimal for flat spectrum in unconstrained case)
        x = np.linspace(-3, 3, n)
        a = np.sinc(x) ** 2
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 6:
        # Polynomial (Bernstein-like)
        x = np.linspace(0, 1, n)
        p = rng.uniform(2, 8)
        a = x ** p * (1 - x) ** p
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
    elif mode == 7:
        # Flat with noise
        a = 1.0 + 0.3 * rng.normal(0, 1, n)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def palindromic_fold(a):
    """Fold a sequence to be palindromic."""
    n = len(a)
    half = (n + 1) // 2
    a_folded = np.zeros(n)
    for i in range(half):
        a_folded[i] = (a[i] + a[n - 1 - i]) / 2.0
        a_folded[n - 1 - i] = a_folded[i]
    s = float(np.sum(a_folded))
    if s > 0:
        a_folded /= s
    return a_folded


def nesterov_smoothed_descent(a, time_budget=100.0, tau_init=5.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        _, g = smoothed_obj_grad(y, tau)

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining L-BFGS-B, LP alternating, coordinate descent,
    and multi-length exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Quick warmup with Nesterov (5% time) ===
    print(f"\n=== Phase 1: Nesterov Warmup ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.05, 60.0)
    if p1_budget > 10:
        a_warm = nesterov_smoothed_descent(best_seq, time_budget=p1_budget,
                                           tau_init=5.0, tau_end=200.0)
        obj_warm = compute_obj(a_warm)
        if obj_warm < best_score:
            best_score = obj_warm
            best_seq = a_warm.copy()
            print(f"  Warmup: obj={best_score:.8f} **")
        else:
            print(f"  Warmup: obj={obj_warm:.8f}")

    # === Phase 2: L-BFGS-B on smoothed objective (25% time) ===
    print(f"\n=== Phase 2: L-BFGS-B Smoothed Optimization ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 300.0)
    if p2_budget > 30:
        a_lbfgs = lbfgs_smoothed_optimize(best_seq, time_budget=p2_budget, verbose=True)
        obj_lbfgs = compute_obj(a_lbfgs)
        if obj_lbfgs < best_score:
            best_score = obj_lbfgs
            best_seq = a_lbfgs.copy()
            print(f"  L-BFGS-B: obj={best_score:.8f} **")
        else:
            print(f"  L-BFGS-B: obj={obj_lbfgs:.8f}")

    # === Phase 3: LP Alternating Optimization (30% time) ===
    print(f"\n=== Phase 3: LP Alternating ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.30, 350.0)
    if p3_budget > 30:
        a_lp = lp_alternating_optimize(best_seq, time_budget=p3_budget, verbose=True, seed=42)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP-alt: obj={best_score:.8f} **")
        else:
            print(f"  LP-alt: obj={obj_lp:.8f}")

    # === Phase 4: L-BFGS-B second pass (15% time) ===
    print(f"\n=== Phase 4: L-BFGS-B Refinement ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 180.0)
    if p4_budget > 20:
        a_lbfgs2 = lbfgs_smoothed_optimize(best_seq, time_budget=p4_budget, verbose=True, seed=123)
        obj_lbfgs2 = compute_obj(a_lbfgs2)
        if obj_lbfgs2 < best_score:
            best_score = obj_lbfgs2
            best_seq = a_lbfgs2.copy()
            print(f"  L-BFGS-B2: obj={best_score:.8f} **")
        else:
            print(f"  L-BFGS-B2: obj={obj_lbfgs2:.8f}")

    # === Phase 5: Coordinate Descent (10% time) ===
    print(f"\n=== Phase 5: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.10, 100.0)
    if p5_budget > 15:
        a_cd = coord_descent_optimize(best_seq, time_budget=p5_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CoordDesc: obj={best_score:.8f} **")
        else:
            print(f"  CoordDesc: obj={obj_cd:.8f}")

    # === Phase 6: Multi-length exploration (10% time) ===
    print(f"\n=== Phase 6: Multi-Length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 100),
        n_current + 50, n_current + 100, n_current + 200,
        900, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and n != n_current]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)

        # Quick optimization sequence
        alloc1 = min(remaining * 0.2, 30.0)
        if alloc1 > 10:
            a_try = nesterov_smoothed_descent(a_try, time_budget=alloc1,
                                              tau_init=10.0, tau_end=300.0)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.3, 60.0)
        if alloc2 > 20:
            a_try = lp_alternating_optimize(a_try, time_budget=alloc2, verbose=False, seed=n_try)

        remaining = deadline - time.monotonic()
        alloc3 = min(remaining * 0.2, 40.0)
        if alloc3 > 15:
            a_try = lbfgs_smoothed_optimize(a_try, time_budget=alloc3, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse restarts with different initializations ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()

    for mode in [2, 3, 5, 6]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_init = diverse_init(n_current, rng, mode=mode)

        # Blend with current best
        a_blend = 0.5 * a_init + 0.5 * best_seq
        a_blend = project_simplex(a_blend)

        alloc = min(remaining * 0.25, 80.0)
        if alloc > 20:
            a_blend = lbfgs_smoothed_optimize(a_blend, time_budget=alloc * 0.5, verbose=False)
            remaining = deadline - time.monotonic()
            alloc2 = min(remaining * 0.3, 60.0)
            if alloc2 > 15:
                a_blend = lp_alternating_optimize(a_blend, time_budget=alloc2, verbose=False,
                                                   seed=mode * 100)

        obj_blend = compute_obj(a_blend)
        if obj_blend < best_score:
            best_score = obj_blend
            best_seq = a_blend.copy()
            print(f"  Restart mode{mode}: obj={best_score:.8f} **")
        else:
            print(f"  Restart mode{mode}: obj={obj_blend:.8f}")

    # === Phase 8: Final intensive LP polish ===
    print(f"\n=== Phase 8: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.7, 200.0)
        a_final = lp_alternating_optimize(best_seq, time_budget=alloc, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # Final coordinate descent pass
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 50.0)
        a_cd_final = coord_descent_optimize(best_seq, time_budget=alloc, verbose=False)
        obj_cd_final = compute_obj(a_cd_final)
        if obj_cd_final < best_score:
            best_score = obj_cd_final
            best_seq = a_cd_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        # Fallback
        result = [1.0 / 100.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
