"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast projected subgradient descent with FFT convolution, peak-targeted
coordinate moves, LP polishing, spectral shaping, and multi-scale exploration.
Key innovations: exact subgradient with active set tracking, adaptive step size
with backtracking, structured perturbations, and beeswarm exploration.
"""

import numpy as np
import time
import math


def compute_conv_fft(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fft(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_of_max_conv(a):
    """
    Compute subgradient of max(conv(a,a)) with respect to a.
    Returns (gradient, max_value, active_indices)

"""

Uses fast projected subgradient descent with FFT convolution, peak-targeted
coordinate moves, adaptive step size with backtracking, structured perturbations,

import numpy as np

def compute_conv_fft(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fft(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fft_subgradient(a, active):
    """
    Compute subgradient of max(conv(a,a)) using FFT.
    For each active lag k, d conv[k]/d a[j] = 2*a[k-j] (if 0<=k-j<n).
    Result: subgrad[j] = sum_{k in active} 2*a[k-j] (bounds checked).
    """
    n = len(a)
    if len(active) == 0:
        return np.zeros(n)

    # Build indicator of active set, padded
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    # indicator[k] = 1 if k in active
    indicator = np.zeros(size)
    for k in active:
        indicator[k] = 1.0

    # We want: subgrad[j] = sum_{d=0}^{n-1} 2*a[d] * indicator[j+d]
    # This is a cross-correlation of a with indicator, taken at shifts 0..n-1.
    # Using FFT: ifft(fft(a) * conj(fft(indicator))) gives the cross-correlation.
    A = np.fft.rfft(a, size)
    I = np.fft.rfft(indicator, size)
    # Cross-correlation: C[m] = sum_d a[d] * indicator[d+m]
    # In FFT: ifft(A * conj(I)) gives sum_d a[d] * conj(indicator[d+m])
    # Since indicator is real: ifft(A * conj(I)) = cross-correlation
    cc = np.fft.irfft(A * np.conj(I), size)
    # cc[m] = sum_d a[d] * indicator[(d+m) mod size]
    # We want subgrad[j] = sum_{d=0}^{n-1} 2*a[d] * indicator[j+d]
    # = 2 * cc[j] (for j in 0..n-1, assuming size is large enough to avoid wraparound)
    subgrad = 2.0 * cc[:n]
    return subgrad

def peak_targeted_moves(a, time_budget=120.0, verbose=False):
    """
    Coordinate descent that targets the dominant peaks.
    For each peak, find the best (source, target) mass transfer to reduce it.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(424242)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv_fft(a)
        max_val = float(np.max(conv))

        # Find top peaks
        top_k = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:top_k]

        improved = False
        for pk in peak_indices:
            pk = int(pk)
            # Contributions to this peak: a[i]*a[pk-i] for valid i
            i_start = max(0, pk - (n - 1))
            i_end = min(n - 1, pk)

            if i_start > i_end:
                continue

            # Find the largest contributing pair
            contribs = []
            for i in range(i_start, i_end + 1):
                j = pk - i
                if 0 <= j < n:
                    contribs.append((a[i] * a[j], i, j))

            if not contribs:
                continue

            contribs.sort(reverse=True)

            # Try moving mass from the largest contributor
            for contrib_val, src, dst in contribs[:5]:
                if a[src] < 1e-10:
                    continue

                # Try several fraction moves to random and structured targets
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    delta = frac * a[src]
                    if delta >= a[src]:
                        continue

                    # Candidate targets
                    targets_set = set()
                    targets_set.add(int(rng.integers(0, n)))
                    targets_set.add((src + n // 3) % n)
                    targets_set.add((src + 2 * n // 3) % n)
                    targets_set.add(n - 1 - src)
                    targets_set.add(int(np.argmax(a)))

                    for tgt in targets_set:
                        tgt = int(tgt) % n
                        if tgt == src:
                            continue
                        a_try = a.copy()
                        a_try[src] -= delta
                        a_try[tgt] += delta
                        if a_try[src] < -1e-15:
                            continue
                        a_try = np.maximum(a_try, 0)
                        s_try = float(np.sum(a_try))
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break

        if verbose and it % 50 == 0:
            print(f"    PTC it{it}: obj={best_obj:.8f}, stall={stall}")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def lp_polish(a, time_budget=100.0, verbose=False):
    """
    LP-based polishing: solve min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.
    Then move towards g with golden section line search.
    """
    try:
        import cvxpy as cp
    except ImportError:
        return a

    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    it = 0
    stall = 0
    rng = np.random.default_rng(99999)

    while time.monotonic() - start < time_budget:
        if stall > 60:
            break

        a_norm = a / float(np.sum(a))
        A_mat = build_conv_matrix(a_norm)

        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A_mat @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        g_val = None
        M_val = None
        solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
        for solver in solvers:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        g_val = g_val / s_g
                        M_val = float(M.value)
                    break
            except Exception:
                continue

        if g_val is None:
            break

        # Golden section line search
        phi = (np.sqrt(5) - 1) / 2
        lo = 0.0
        hi = 1.0
        for _ in range(7):
            t1 = hi - phi * (hi - lo)
            t2 = lo + phi * (hi - lo)
            a1 = np.maximum((1 - t1) * a + t1 * g_val, 0)
            s1 = float(np.sum(a1))
            f1 = compute_obj(a1) if s1 > 1e-10 else np.inf
            a2 = np.maximum((1 - t2) * a + t2 * g_val, 0)
            s2 = float(np.sum(a2))
            f2 = compute_obj(a2) if s2 > 1e-10 else np.inf
            if f1 < f2:
                hi = t2
            else:
                lo = t1

        t_best = (lo + hi) / 2
        a_new = np.maximum((1 - t_best) * a + t_best * g_val, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
        else:
            obj_new = np.inf

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            # Perturb
            noise = 0.01 + 0.005 * min(stall, 10)
            a = best_a * np.exp(rng.normal(0, noise, n))
            a = np.maximum(a, 0)
            s_a = float(np.sum(a))
            if s_a > 1e-10:
                a = a / s_a
            else:
                a = best_a.copy()

        it += 1
        if verbose and it % 3 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, M={M_val:.6f}, stall={stall}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / (a.sum() + 1e-15)
    elif mode == 2:
        # Comb pattern: alternating high and low
        a = np.ones(n)
        a[::2] = 2.0
        a = a / a.sum()
    elif mode == 3:
        # Triangular distribution
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)
        a = a / (a.sum() + 1e-15)
    elif mode == 4:
        # Cosine bump
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / (a.sum() + 1e-15)
    elif mode == 5:
        # Exponential decay from center
        center = n // 2
        a = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (n * 0.1))
        a = a / (a.sum() + 1e-15)
    elif mode == 6:
        # Two peaks
        a = np.zeros(n)
        a[n // 4] = 1.0
        a[3 * n // 4] = 1.0
        # Spread
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=n * 0.05)
        a = a / (a.sum() + 1e-15)
    elif mode == 7:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1, 3)
        a = a / (a.sum() + 1e-15)
    else:
        a = rng.dirichlet(np.ones(n))
    return a


def beeswarm_explore(a_best, n, time_budget=120.0, verbose=False):
    """
    Beeswarm-style exploration: generate perturbations around the best point,
    evaluate them, and keep the best.
    """
    start = time.monotonic()
    rng = np.random.default_rng(55555)
    best_a = a_best.copy()
    best_obj = compute_obj(a_best)

    iteration = 0
    while time.monotonic() - start < time_budget:
        # Generate a swarm of perturbations
        swarm_size = 5
        candidates = []

        for _ in range(swarm_size):
            # Random perturbation with Gaussian noise
            scale = 0.005 + 0.02 * np.exp(-iteration / 200.0)
            noisy = best_a * np.exp(rng.normal(0, scale, n))
            noisy = np.maximum(noisy, 0)
            s = float(np.sum(noisy))
            if s > 1e-10:
                noisy = noisy / s
                candidates.append(noisy)

        # Also try a "flip" perturbation: swap two random elements
        i, j = rng.integers(0, n, 2)
        if i != j:
            flip = best_a.copy()
            flip[i], flip[j] = flip[j], flip[i]
            candidates.append(flip)

        # Try a "shift" perturbation
        shift = int(rng.integers(1, max(2, n // 20)))
        shifted = np.roll(best_a, shift)
        s = float(np.sum(shifted))
        if s > 1e-10:
            candidates.append(shifted / s)

        # Evaluate all candidates
        for cand in candidates:
            obj = compute_obj(cand)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = cand.copy()
                if verbose:
                    print(f"    Beeswarm it{iteration}: improved to {best_obj:.8f}")

        iteration += 1

        if time.monotonic() - start > time_budget * 0.8:
            break

    return best_a


def smoothed_gradient_descent(a, time_budget=150.0, tau_init=20.0, tau_end=200.0, verbose=False):
    """
    Nesterov-accelerated gradient descent on log-sum-exp smoothed objective.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.8
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        # Compute smoothed gradient
        conv = compute_conv_fft(a)
        max_conv = float(np.max(conv))
        # Log-sum-exp gradient
        shifted = tau * (conv - max_conv)
        # Clip for numerical stability
        shifted = np.maximum(shifted, -50)
        exp_vals = np.exp(shifted)
        p = exp_vals / float(np.sum(exp_vals))

        # Gradient of sum_k p[k] * conv[k] with respect to a
        # = sum_k p[k] * grad conv[k]
        # Use FFT: for each k, grad conv[k][j] = 2*a[k-j]
        # So total grad[j] = sum_k p[k] * 2*a[k-j] = 2 * (p cross-correlate a)[j]
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        P = np.fft.rfft(p, size)
        A = np.fft.rfft(a, size)
        # Cross-correlation of p and a: ifft(P * conj(A))
        cc = np.fft.irfft(P * np.conj(A), size)
        grad = 2.0 * cc[:n]

        # Scale by 2n
        grad = 2.0 * n * grad

        # Nesterov step
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y > 1e-10:
            y = y / s_y
        else:
            y = a.copy()

        # Recompute gradient at y for Nesterov
        conv_y = compute_conv_fft(y)
        max_conv_y = float(np.max(conv_y))
        shifted_y = tau * (conv_y - max_conv_y)
        shifted_y = np.maximum(shifted_y, -50)
        exp_y = np.exp(shifted_y)
        p_y = exp_y / float(np.sum(exp_y))
        Py = np.fft.rfft(p_y, size)
        Ay = np.fft.rfft(y, size)
        cc_y = np.fft.irfft(Py * np.conj(Ay), size)
        grad_y = 2.0 * n * 2.0 * cc_y[:n]

        v = beta * v - lr * grad_y
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9997)

        if it - best_it > 10000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            beta = 0.7
            best_it = it

        if verbose and it % 2000 == 0:
            print(f"    SGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization combining fast subgradient descent, peak-targeted
    coordinate moves, LP polishing, smoothed GD, beeswarm exploration, and
    multi-scale length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent available: {e}")

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Fast Subgradient Descent (main workhorse) ===
    print(f"\n=== Phase 1: Fast Subgradient Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 500.0)
    if p1_budget > 30:
        a_fsd = fast_subgradient_descent(best_seq, time_budget=p1_budget, verbose=True)
        obj_fsd = compute_obj(a_fsd)
        if obj_fsd < best_score:
            best_score = obj_fsd
            best_seq = a_fsd.copy()
            print(f"  FSD: obj={best_score:.8f} **")
        else:
            print(f"  FSD: obj={obj_fsd:.8f}")

    # === Phase 2: Peak-Targeted Coordinate Moves ===
    print(f"\n=== Phase 2: Peak-Targeted Moves ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 200.0)
    if p2_budget > 20:
        a_ptc = peak_targeted_moves(best_seq, time_budget=p2_budget, verbose=True)
        obj_ptc = compute_obj(a_ptc)
        if obj_ptc < best_score:
            best_score = obj_ptc
            best_seq = a_ptc.copy()
            print(f"  PTC: obj={best_score:.8f} **")
        else:
            print(f"  PTC: obj={obj_ptc:.8f}")

    # === Phase 3: LP Polish ===
    print(f"\n=== Phase 3: LP Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 250.0)
    if p3_budget > 30:
        a_lp = lp_polish(best_seq, time_budget=p3_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 4: Smoothed Gradient Descent ===
    print(f"\n=== Phase 4: Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 180.0)
    if p4_budget > 30:
        a_sgd = smoothed_gradient_descent(best_seq, time_budget=p4_budget,
                                          tau_init=30.0, tau_end=300.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  SGD: obj={best_score:.8f} **")
        else:
            print(f"  SGD: obj={obj_sgd:.8f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(100, n_current - 100), max(100, n_current - 200),
        n_current + 100, n_current + 200,
        400, 600, 1000, 1200
    ]))
    n_options = [n for n in n_options if 100 <= n <= 2000 and abs(n - n_current) > 30]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try > 1e-10:
            a_try = a_try / s_try

        # Quick FSD on this length
        alloc = min(remaining * 0.15, 60.0)
        if alloc > 15:
            a_try = fast_subgradient_descent(a_try, time_budget=alloc)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Beeswarm + FSD around best ===
    print(f"\n=== Phase 6: Beeswarm Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 100.0)
        a_bee = beeswarm_explore(best_seq, len(best_seq), time_budget=alloc)
        obj_bee = compute_obj(a_bee)
        if obj_bee < best_score:
            best_score = obj_bee
            best_seq = a_bee.copy()
            print(f"  Beeswarm: obj={best_score:.8f} **")

        # Follow up with FSD
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.2, 80.0)
            a_fsd2 = fast_subgradient_descent(best_seq, time_budget=alloc)
            obj_fsd2 = compute_obj(a_fsd2)
            if obj_fsd2 < best_score:
                best_score = obj_fsd2
                best_seq = a_fsd2.copy()
                print(f"  FSD2: obj={best_score:.8f} **")

    # === Phase 7: Diverse restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = r % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with current best (70% best, 30% diversity)
        blend = 0.25 + 0.1 * r
        a_r = (1 - blend) * best_seq + blend * a_r
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        alloc = min(remaining * 0.2, 60.0)
        if alloc > 15:
            a_r = fast_subgradient_descent(a_r, time_budget=alloc)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Multiple rounds of the best methods
    for round_idx in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # FSD
        alloc = min(remaining * 0.3, 80.0)
        if alloc > 15:
            a_f = fast_subgradient_descent(best_seq, time_budget=alloc)
            obj_f = compute_obj(a_f)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final FSD r{round_idx}: obj={best_score:.8f} **")

        # PTC
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.2, 40.0)
        if alloc > 10:
            a_p = peak_targeted_moves(best_seq, time_budget=alloc)
            obj_p = compute_obj(a_p)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Final PTC r{round_idx}: obj={best_score:.8f} **")

        # Beeswarm
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.15, 30.0)
        if alloc > 10:
            a_b = beeswarm_explore(best_seq, len(best_seq), time_budget=alloc)
            obj_b = compute_obj(a_b)
            if obj_b < best_score:
                best_score = obj_b
                best_seq = a_b.copy()
                print(f"  Final Bee r{round_idx}: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with evaluate_sequence
    try:
        val = evaluate_sequence(result)
        if val < np.inf and val < best_score + 1e-10:
            best_score = val
    except Exception:
        pass

    if best_score >= np.inf or not np.isfinite(best_score):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
