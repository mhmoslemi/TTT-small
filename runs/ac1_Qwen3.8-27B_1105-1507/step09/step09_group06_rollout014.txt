```python
import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho = np.arange(1, n+1)
    cond = mu * rho > (cssv - 1.0)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_val = idx[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt).
    """
    n = len(a)
    # Build constraint matrix A_mat (2n-1) x n
    # A_mat[k, i] = a[i] if k-i is in [0, n-1], else 0
    # i.e., A_mat[k, i] = a[i] if 0 <= k-i <= n-1, i.e., k-(n-1) <= i <= k
    A_mat = np.zeros((2*n-1, n))
    for i in range(n):
        rows = slice(i, i + n)  # k ranges from i to i+n-1
        A_mat[rows, i] = a[i]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except:
            continue
    return None, None

def line_search(a, g, max_iters=50):
    """
    Find optimal t in [0,1] minimizing f((1-t)*a + t*g).
    Uses golden section search.
    """
    a_norm = a / a.sum()
    g_norm = g / g.sum()
    
    # Evaluate at boundaries
    f0 = compute_obj(a_norm)
    f1 = compute_obj(g_norm)
    
    if f1 < f0:
        return 1.0  # moving to g is always better
    # If f0 is already better, we still want to find the best in between
    # but if f1 > f0, the minimum might be at t=0 or somewhere in between
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618
    a_bs, b_bs = 0.0, 1.0
    c_bs = b_bs - gr * (b_bs - a_bs)
    d_bs = a_bs + gr * (b_bs - a_bs)
    fc = compute_obj((1-c_bs)*a_norm + c_bs*g_norm)
    fd = compute_obj((1-d_bs)*a_norm + d_bs*g_norm)
    
    for _ in range(max_iters):
        if fc < fd:
            b_bs = d_bs
            d_bs = c_bs
            fd = fc
            c_bs = b_bs - gr * (b_bs - a_bs)
            fc = compute_obj((1-c_bs)*a_norm + c_bs*g_norm)
        else:
            a_bs = c_bs
            c_bs = d_bs
            fc = fd
            d_bs = a_bs + gr * (b_bs - a_bs)
            fd = compute_obj((1-d_bs)*a_norm + d_bs*g_norm)
        if b_bs - a_bs < 1e-6:
            break
    
    t_opt = (a_bs + b_bs) / 2
    return t_opt
```

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)

    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_val = idx[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

    A_ub = np.column_stack([A_dense, -np.ones(2*n-1)])
    b_ub = np.zeros(2*n-1)
    A_eq = np.column_stack([np.ones(n), [0.0]])
    b_eq = np.array([1.0])
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize M
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        g = res.x[:n]
        g = np.maximum(g, 0)
        s = g.sum()
        if s > 1e-10:
            return g / s
    return None


def solve_lp_cvxpy(a):
    """Solve LP using cvxpy."""
    n = len(a)
    A_mat = np.zeros((2*n-1, n))
    for i in range(n):
        if a[i] > 1e-15:
            A_mat[i:i+n, i] = a[i]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = g_val.sum()
                if s > 1e-10:
                    return g_val / s
        except:
            continue
    return None


def line_search_1d(a, g, f0):
    """
    Golden section search for optimal t in [0,1] minimizing f((1-t)*a + t*g).
    f(t) = 2n * max_k conv((1-t)*a + t*g, (1-t)*a + t*g)[k]
    """
    n = len(a)
    
    # Evaluate endpoints
    f1 = compute_obj(g)
    if f1 <= f0:
        return 1.0
    
    # Golden section
    gr = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = compute_obj((1-c)*a + c*g)
    fd = compute_obj((1-d)*a + d*g)
    
    for _ in range(40):
        if hi - lo < 1e-7:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = compute_obj((1-c)*a + c*g)
        else:
            lo = c
            c = d
            fc = fd
            d = lo
