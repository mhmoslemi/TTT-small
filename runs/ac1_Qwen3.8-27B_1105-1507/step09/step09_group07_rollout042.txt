Hybrid optimization for minimizing peak autocorrelation of non-negative step functions.
Uses LP alternating with Nesterov momentum, smoothed gradient descent with simplex

Target: score < 1.5027.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None

    Uses gradient-based approach since the quadratic makes it non-linear.

    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M + lam * cp.sum_squares(g - a_norm)), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

                s = float(np.sum(g_val))

                    a_new = best_a.copy()

                a_prev = a.copy()
                y = a.copy()
                stall = 0
                last_good_t = 0.4
                mu = 0.8

        if verbose and iteration % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"        LP-mom it{iteration}: obj={best_obj:.8f}, t={last_good_t:.3f}, "
                  f"mu={mu:.3f}, stall={stall}, elapsed={elapsed:.1f}s")

    return best_a


def gradient_descent_smoothed(a, time_budget, beta_start=20.0, beta_end=2000.0, verbose=False):
    """
    Smoothed gradient descent using log-sum-exp surrogate with simplex projection.
    More efficient than the previous mirror descent version.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 5.0 / n
    it = 0
    total_iters = max(1, int(time_budget * 10))

    while time.monotonic() - start < time_budget:
        frac = min(1.0, it / total_iters * 2)
        beta = beta_start * (beta_end / beta_start) ** frac

        conv = compute_conv(a)
        conv_max = conv.max()
        
        # Softmax weights (numerically stable)
        shifted = beta * (conv - conv_max)
        weights = np.exp(shifted)
        weights /= weights.sum()

        # Gradient of sum_k w_k * (a*a)[k] w.r.t. a
        # = 2 * cross_correlation(a, weights)
        # (a*a)[k] = sum_i a_i * a[k-i]
        # d/d a_j [sum_k w_k * (a*a)[k]] = 2 * sum_k w_k * a[k-j]
        # = 2 * (a reversed and correlated with weights)
        
        # Compute efficiently via FFT
        size = 1
        while size < 2 * n:
            size <<= 1
        A = rfft(a, size)
        W = rfft(np.concatenate([weights, np.zeros(size - len(weights))]), size)
        grad = 2.0 * irfft(A * W, size)[:n]
        # Actually need to be more careful. Let me compute directly.
        # d/d a_j sum_k w_k (a*a)[k] = 2 * sum_k w_k a[k-j] (for valid k-j)
        # This is 2 * (w * a_reversed)[j] approximately
        
        # Simpler: use the formula gradient[i] = 2 * sum_k w[k] * a[k-i] where a[m]=0 for m out of range
        grad = np.zeros(n)
        for i in range(n):
            # a[k-i] is valid when 0 <= k-i < n, i.e., i <= k <= i+n-1
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            if k_lo <= k_hi:
                grad[i] = 2.0 * np.dot(weights[k_lo:k_hi+1], a[0:(k_hi-k_lo+1)])

        # Gradient descent with projection
        a_new = a - lr * grad
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = min(10.0/n, lr * 1.01)
        else:
            lr = max(0.1/n, lr * 0.85)

        it += 1
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"          Grad it{it}: obj={best_obj:.8f}, beta={beta:.0f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")

    return best_a


def spectral_flatten(a, time_budget, verbose=False):
    """
    Iteratively flatten the power spectrum to reduce autocorrelation peaks.
    Uses a more careful approach: only modify phases, keep magnitudes.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
            
        # Compute current convolution
        conv = compute_conv(a)
        peak = conv.max()
        mean_conv = conv.mean()
        
        # Compute FFT of a
        A = rfft(a, size)
        
        # The convolution in frequency domain is A^2
        # We want |A^2| to be flat, i.e., |A| to be flat
        # But we need a >= 0, so we can't freely change A
        
        # Instead, compute the desired magnitude: sqrt(mean(|A|^2))
        # and blend the current magnitude with the target
        mag = np.abs(A)
        target_mag = np.sqrt(np.mean(mag ** 2))
        
        for alpha in [0.05, 0.1, 0.2]:
            new_mag = (1 - alpha) * mag + alpha * target_mag
            new_mag = np.maximum(new_mag, 0)
            phase = A / (mag + 1e-30)
            A_new = new_mag * phase
            a_new = irfft(A_new, size)[:n]
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
                break

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"          Spectral it{it}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_a


def targeted_refine(a, time_budget, verbose=False):
    """
    Targeted refinement: identify peak lags and move mass from contributors to quiet positions.
    More efficient than previous version by using vectorized operations.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    round_num = 0
    max_rounds = 500
    while time.monotonic() - start < time_budget and round_num < max_rounds:
        round_num += 1
        conv = compute_conv(a)
        peak_val = float(np.max(conv))
        
        # Find top 2 peak lags
        peak_lags = np.argsort(conv)[::-1][:2]
        
        # For each peak lag, find the main contributors
        improved = False
        for k_star in peak_lags:
            k_star = int(k_star)
            # Contributors: pairs (i, k_star-i) with 0 <= i < n, 0 <= k_star-i < n
            i_lo = max(0, k_star - n + 1)
            i_hi = min(n - 1, k_star)
            
            if i_lo > i_hi:
                continue
                
            indices = np.arange(i_lo, i_hi + 1)
            contribs = a[indices] * a[k_star - indices]
            
            # Target positions: where a is small and far from peak-contributing positions
            # "Quiet" = positions that don't contribute to top peaks
            quiet_score = a.copy()
            for kp in peak_lags[:3]:
                kp = int(kp)
                ilo = max(0, kp - n + 1)
                ihi = min(n - 1, kp)
                quiet_score[ilo:ihi+1] -= 0.5 * a[ilo:ihi+1] * a[kp - np.arange(ilo, ihi+1)]
            
            quiet_pos = np.argsort(quiet_score)[-8:]
            
            # Try moving mass from top contributors to quiet positions
            top_contrib_idx = indices[np.argsort(contribs)[::-1][:8]]
            
            for src in top_contrib_idx:
                if time.monotonic() - start > time_budget:
                    break
                if a[src] < 1e-8:
                    continue
                for target in quiet_pos:
                    if target == src:
                        continue
                    for frac in [0.002, 0.005, 0.01, 0.02]:
                        delta = frac * a[src]
                        if delta <= 0 or delta >= a[src] * 0.9:
                            continue
                        a[src] -= delta
                        a[target] += delta
                        new_obj = compute_obj(a)
                        if new_obj < best_obj - 1e-14:
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[src] += delta
                            a[target] -= delta
                            break
                if improved:
                    break
            
            if improved:
                break

        if not improved and round_num > 20:
            break

        if verbose and round_num % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"          Targeted round {round_num}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode % 10 == 0:
        a = np.ones(n) / n
    elif mode % 10 == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode % 10 == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 3:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + np.exp(-3 * (x - 0.7) ** 2) + 0.05
        a = a / a.sum()
    elif mode % 10 == 5:
        x = np.linspace(0, 1, n)
        a = x**2 + (1 - x)**2 + 0.1
        a = a / a.sum()
    elif mode % 10 == 6:
        freqs = rng.integers(2, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        amplitudes = rng.uniform(0.2, 0.5, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p, amp in zip(freqs, phases, amplitudes):
            a += amp * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 7:
        a = rng.dirichlet(np.ones(n) * 0.3)
    elif mode % 10 == 8:
        # Triangular
        x = np.linspace(0, 1, n)
        a = 1 - 2 * np.abs(x - 0.5)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        # Parabolic
        x = np.linspace(0, 1, n)
        a = 4 * x * (1 - x) + 0.1
        a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if n == 2:
        x_old = np.linspace(0, 1, 2)
        x_new = np.linspace(0, 1, new_n)
        a_new = np.interp(x_new, x_old, a)
        return np.maximum(a_new, 0)
    
    from scipy.interpolate import interp1d
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
    a_new = f(x_new)
    return np.maximum(a_new, 0)


def coordinate_descent_lp(a, time_budget, verbose=False):
    """
    Coordinate descent guided by LP solution.
    Move mass between coordinate pairs to reduce objective.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for rnd in range(50):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        # Find coordinates where we should move mass
        diff = g_lp - a
        # Sort by absolute difference
        idx = np.argsort(np.abs(diff))[::-1]

        improved_any = False
        for i in idx[:20]:
            if time.monotonic() - start > time_budget:
                break
            for j in idx[:20]:
                if j == i:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    delta = frac * abs(diff[i])
                    if delta <= 0 or delta < 1e-10:
                        continue
                    # Move towards LP solution
                    if diff[i] > 0:
                        a[i] += delta
                        a[j] -= delta
                    else:
                        a[i] -= delta
                        a[j] += delta
                    
                    if np.any(a < 0):
                        a[i] -= delta if diff[i] > 0 else delta
                        a[j] += delta if diff[i] > 0 else -delta
                        continue
                    
                    a = np.maximum(a, 0)
                    s2 = float(np.sum(a))
                    if s2 < 1e-10:
                        continue
                    a = a / s2
                    obj = compute_obj(a)
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_a = a.copy()
                        improved_any = True
                        break
                    else:
                        # Revert
                        if diff[i] > 0:
                            a[i] -= delta
                            a[j] += delta
                        else:
                            a[i] += delta
                            a[j] -= delta
                        a = np.maximum(a, 0)
                        break

        if not improved_any:
            break

        if verbose and rnd % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"          CD-LP round {rnd}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_a

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).

def gradient_descent_smoothed(a, time_budget, beta_start=20.0, beta_end=3000.0, verbose=False):
    """
    Smoothed gradient descent using log-sum-exp surrogate with simplex projection.
    Uses np.correlate for efficient gradient computation.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 3.0 / n
    it = 0

    while time.monotonic() - start < time_budget:
        frac = min(1.0, it / max(1, int(time_budget * 5)))
        beta = beta_start * (beta_end / beta_start) ** frac

        conv = compute_conv(a)
        conv_max = conv.max()

        # Softmax weights (numerically stable)
        shifted = beta * (conv - conv_max)
        weights = np.exp(shifted)
        weights /= weights.sum()

        # Gradient: d/d a_j [sum_k w_k (a*a)[k]] = 2 * sum_k w_k * a[k-j]
        # = 2 * (w correlated with a) at position j
        # np.correlate(w, a, 'full')[j] = sum_m w[m+j] * a[m] for valid m
        grad_half = np.correlate(weights, a, mode='full')[:n]
        grad = 2.0 * grad_half

        # Gradient descent with projection onto simplex
        a_new = a - lr * grad
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = min(20.0 / n, lr * 1.02)
        else:
            lr = max(0.05 / n, lr * 0.88)

        it += 1
        if verbose and it % 30 == 0:
            elapsed = time.monotonic() - start
            print(f"          Grad it{it}: obj={best_obj:.8f}, beta={beta:.0f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")

    return best_a

            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
                break

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"          Spectral it{it}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_a


def targeted_refine(a, time_budget, verbose=False):
    """
    Targeted refinement: identify peak lags and move mass from top contributors
    to quiet positions. Vectorized for efficiency.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    round_num = 0
    max_rounds = 1000
    while time.monotonic() - start < time_budget and round_num < max_rounds:
        round_num += 1
        conv = compute_conv(a)
        peak_val = float(np.max(conv))

        # Find top 2 peak lags
        peak_lags = np.argsort(conv)[::-1][:2]

        # Compute "activity" score: how much each position contributes to peaks
        activity = np.zeros(n)
        for kp in peak_lags:
            kp = int(kp)
            ilo = max(0, kp - n + 1)
            ihi = min(n - 1, kp)
            if ilo <= ihi:
                activity[ilo:ihi + 1] += conv[kp] * a[ilo:ihi + 1] * a[kp - np.arange(ilo, ihi + 1)]

        # Quiet positions = low activity
        quiet_pos = np.argsort(activity)[:10]

        improved = False
        for k_star in peak_lags:
            if time.monotonic() - start > time_budget:
                break
            k_star = int(k_star)
            ilo = max(0, k_star - n + 1)
            ihi = min(n - 1, k_star)
            if ilo > ihi:
                continue

            indices = np.arange(ilo, ihi + 1)
            contribs = a[indices] * a[k_star - indices]
            top_src = indices[np.argsort(contribs)[::-1][:6]]

            for src in top_src:
                if time.monotonic() - start > time_budget:
                    break
                if a[src] < 1e-10:
                    continue
                for target in quiet_pos:
                    if target == src:
                        continue
                    for frac in [0.002, 0.005, 0.01, 0.015, 0.025]:
                        delta = frac * a[src]
                        if delta <= 0 or delta >= a[src] * 0.8:
                            continue
                        a[src] -= delta
                        a[target] += delta
                        new_obj = compute_obj(a)
                        if new_obj < best_obj - 1e-14:
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[src] += delta
                            a[target] -= delta
                            break
                if improved:
                    break
            if improved:
                break

        if not improved and round_num > 30:
            break

        if verbose and round_num % 30 == 0:
            elapsed = time.monotonic() - start
            print(f"          Targeted round {round_num}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode % 10 == 0:
        a = np.ones(n) / n
    elif mode % 10 == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode % 10 == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 3:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + np.exp(-3 * (x - 0.7) ** 2) + 0.05
        a = a / a.sum()
    elif mode % 10 == 5:
        x = np.linspace(0, 1, n)
        a = x**2 + (1 - x)**2 + 0.1
        a = a / a.sum()
    elif mode % 10 == 6:
        x = np.linspace(0, 1, n)
        a = 4 * x * (1 - x) + 0.1
        a = a / a.sum()
    elif mode % 10 == 7:
        a = rng.dirichlet(np.ones(n) * 0.3)
    elif mode % 10 == 8:
        x = np.linspace(0, 1, n)
        a = 1 - 2 * np.abs(x - 0.5)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        freqs = rng.integers(2, 10, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        amplitudes = rng.uniform(0.2, 0.5, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p, amp in zip(freqs, phases, amplitudes):
            a += amp * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    if n >= 4:
        from scipy.interpolate import interp1d
        f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
        a_new = f(x_new)
    else:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_descent_lp(a, time_budget, verbose=False):
    """
    Coordinate descent guided by LP solution.
    Moves mass between pairs of coordinates toward the LP solution direction.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for rnd in range(30):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        diff = g_lp - a
        idx = np.argsort(np.abs(diff))[::-1]

        improved_any = False
        for i in idx[:15]:
            if time.monotonic() - start > time_budget:
                break
            for j in idx[:15]:
                if j == i:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.04]:
                    delta = frac * abs(diff[i])
                    if delta <= 0 or delta < 1e-12:
                        continue
                    if diff[i] > 0:
                        a[i] += delta
                        a[j] -= delta
                    else:
                        a[i] -= delta
                        a[j] += delta

                    if np.any(a < 0):
                        if diff[i] > 0:
                            a[i] -= delta
                            a[j] += delta
                        else:
                            a[i] += delta
                            a[j] -= delta
                        continue

                    a_rng = float(a.max())
                    a = a / a.sum()
                    obj = compute_obj(a)
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_a = a.copy()
                        improved_any = True
                        break
                    else:
                        if diff[i] > 0:
                            a[i] -= delta
                            a[j] += delta
                        else:
                            a[i] += delta
                            a[j] -= delta
                        a = np.maximum(a, 0)
                        break

        if not improved_any:
            break

        if verbose and rnd % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"          CD-LP round {rnd}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating with momentum, smoothed gradient, spectral,
    targeted refinement, n-sweep, diverse restarts, and coordinate descent.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"\nInitial: n={n_current}, obj={best_score:.8f}, target=1.5027")

    # === Phase 1: Quick LP polish on parent (7%) ===
    print(f"\n=== Phase 1: Quick LP Polish ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.07, 70.0)
    if p1_budget > 20:
        a_p1 = lp_alternate_momentum(best_seq, time_budget=p1_budget, verbose=True, momentum=0.85)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_p1:.8f}")

    # === Phase 2: n-sweep with wider range (18%) ===
    print(f"\n=== Phase 2: n-Sweep ===")
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.18

    n_base = n_current
    n_options = sorted(set([
        max(500, n_base - 200), max(500, n_base - 100), max(500, n_base - 50),
        max(500, n_base - 20), n_base, n_base + 20, n_base + 50,
        n_base + 100, n_base + 200, n_base + 400, n_base + 500,
        600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600, 1800, 2000, 2500,
    ]))
    n_options = [n for n in n_options if 500 <= n <= 2500]

    per_n_budget = p2_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = max(10.0, min(per_n_budget * 1.5, remaining * 0.12))

        a_try = resample_sequence(best_seq, n_try)
        a_opt = lp_alternate_momentum(a_try, time_budget=alloc, verbose=False, momentum=0.8)
        obj_opt = compute_obj(a_opt)
        marker = " **" if obj_opt < best_score else ""
        print(f"  n={n_try}: obj={obj_opt:.8f}{marker}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try

    # === Phase 3: Intensive LP alternating on best n (30%) ===
    print(f"\n=== Phase 3: Intensive LP (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.30, 450.0)

    n_rounds = 5
    per_round = p3_budget / n_rounds
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        alloc = min(per_round, remaining * 0.25)
        if alloc < 25:
            break
        if rnd == 0:
            a_start = best_seq.copy()
        elif rnd == 1:
            # Structured low-frequency perturbation
            rng_r = np.random.default_rng(rnd * 104729 + n_current)
            x = np.linspace(0, 1, n_current)
            freq = rng_r.integers(2, max(3, n_current // 100))
            phase = rng_r.uniform(0, 2 * np.pi)
            pert = 0.04 * np.sin(2 * np.pi * freq * x + phase)
            a_start = best_seq * np.exp(pert)
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        elif rnd == 2:
            # Higher frequency structured perturbation
            rng_r = np.random.default_rng(rnd * 20917 + n_current)
            x = np.linspace(0, 1, n_current)
            freq = rng_r.integers(n_current // 30, max(n_current // 10, n_current // 30 + 2))
            phase = rng_r.uniform(0, 2 * np.pi)
            pert = 0.03 * np.cos(2 * np.pi * freq * x + phase)
            a_start = best_seq * np.exp(pert)
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        else:
            # Random multiplicative perturbation
            rng_r = np.random.default_rng(rnd * 7919 + n_current)
            scale = 0.03 + 0.02 * (rnd - 2)
            a_start = best_seq * np.exp(rng_r.normal(0, scale, n_current))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()

        momentum = 0.85 + 0.03 * rnd
        a_lp = lp_alternate_momentum(a_start, time_budget=alloc, verbose=True, momentum=momentum)
        obj_lp = compute_obj(a_lp)
        marker = " **" if obj_lp < best_score else ""
        print(f"  LP round {rnd}: obj={obj_lp:.8f}{marker}")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()

    # === Phase 4: Gradient + Spectral + Targeted (15%) ===
    print(f"\n=== Phase 4: Gradient + Spectral + Targeted ===")
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.15

    # Smoothed gradient descent
    alloc_g = min(p4_budget * 0.35, 80.0)
    if alloc_g > 15:
        a_grad = gradient_descent_smoothed(best_seq, time_budget=alloc_g, beta_start=20, beta_end=3000, verbose=True)
        obj_grad = compute_obj(a_grad)
        marker = " **" if obj_grad < best_score else ""
        print(f"  Gradient: obj={obj_grad:.8f}{marker}")
        if obj_grad < best_score:
            best_score = obj_grad
            best_seq = a_grad.copy()

    # Spectral flattening
    remaining = deadline - time.monotonic()
    alloc_s = min(remaining * 0.2, 50.0)
    if alloc_s > 10:
        a_spec = spectral_flatten(best_seq, time_budget=alloc_s, verbose=True)
        obj_spec = compute_obj(a_spec)
        marker = " **" if obj_spec < best_score else ""
        print(f"  Spectral: obj={obj_spec:.8f}{marker}")
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()

    # Targeted refinement
    remaining = deadline - time.monotonic()
    alloc_t = min(remaining * 0.35, 60.0)
    if alloc_t > 15:
        a_targ = targeted_refine(best_seq, time_budget=alloc_t, verbose=True)
        obj_targ = compute_obj(a_targ)
        marker = " **" if obj_targ < best_score else ""
        print(f"  Targeted: obj={obj_targ:.8f}{marker}")
        if obj_targ < best_score:
            best_score = obj_targ
            best_seq = a_targ.copy()

    # === Phase 5: Diverse restarts (12%) ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p5_budget = remaining * 0.12
    n_use = len(best_seq)
    n_restarts = 5
    per_restart = p5_budget / max(1, n_restarts)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = max(15.0, min(per_restart, remaining * 0.2))

        mode = (r * 2 + 3) % 10
        a_r = diverse_init(n_use, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        half = alloc / 2
        if half > 10:
            a_r = lp_alternate_momentum(a_r, time_budget=half, verbose=False, momentum=0.85)
        remaining = deadline - time.monotonic()
        grad_alloc = min(remaining * 0.08, 30.0)
        if grad_alloc > 10:
            a_r = gradient_descent_smoothed(a_r, time_budget=grad_alloc, beta_start=30, beta_end=1500)

        obj_r = compute_obj(a_r)
        marker = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}{marker}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()

    # === Phase 6: Final polish (13%) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP alternating polish with high momentum
    alloc = min(remaining * 0.35, 100.0)
    if alloc > 20:
        a_final = lp_alternate_momentum(best_seq, time_budget=alloc, verbose=True, momentum=0.92)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Targeted refinement
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.25, 60.0)
    if alloc > 15:
        a_targ = targeted_refine(best_seq, time_budget=alloc, verbose=True)
        obj_targ = compute_obj(a_targ)
        if obj_targ < best_score:
            best_score = obj_targ
            best_seq = a_targ.copy()
            print(f"  Final Targeted: obj={best_score:.8f} **")

    # Coordinate descent with LP
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.2, 40.0)
    if alloc > 15:
        a_cd = coordinate_descent_lp(best_seq, time_budget=alloc, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final short gradient pass
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.15, 30.0)
    if alloc > 10:
        a_gf = gradient_descent_smoothed(best_seq, time_budget=alloc, beta_start=200, beta_end=5000)
        obj_gf = compute_obj(a_gf)
        if obj_gf < best_score:
            best_score = obj_gf
            best_seq = a_gf.copy()
            print(f"  Final Grad: obj={best_score:.8f} **")

    # One last LP round
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_last = lp_alternate_momentum(best_seq, time_budget=min(remaining * 0.5, 30.0), verbose=False, momentum=0.9)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
        print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f} ===")
    else:
        result = [1.0] * 100
        best_score = 2.0
        print(f"\n=== ERROR: fallback to default ===")

    elapsed = time.monotonic() - start_time
    print(f"=== Time: {elapsed:.1f}s, Target: 1.5027, Gap: {best_score - 1.5027:.8f} ===")

    return result
