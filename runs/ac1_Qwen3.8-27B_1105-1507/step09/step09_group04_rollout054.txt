def proj_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]

                else:

                    conv /= (s_new*s_new)
        
        obj_after = float(np.max(conv))
        if obj_after < best_obj - 1e-12:
            best_obj = obj_after
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 8:
                if rng is not None:
                    scale = 0.02 * (1 + stagnant*0.3)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    ss = float(np.sum(a))
                    if ss > 1e-10:
                        a /= ss
                        conv = compute_conv(a)
                        best_obj = float(np.max(conv))
                        stagnant = 0
                else:
                    break
    
    return best_a

def build_conv_matrix(a):
    """Build sparse-ish matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A

def lp_solve(a, time_budget=60.0):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0. Returns g or None."""
    n = len(a)
    a_n = a / np.sum(a)
    A = build_conv_matrix(a_n)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, **{'gap_rel': 1e-4} if solver in [cp.SCS] else {})
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None
