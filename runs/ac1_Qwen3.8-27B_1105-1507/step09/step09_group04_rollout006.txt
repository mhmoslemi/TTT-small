Let me focus on what's most likely to work:

**Improved LP iteration with multiplicative updates:**

Instead of `a_new = (1-t)*a + t*g`, try:
- `a_new = a * g^t` (element-wise, then normalize)
- Or `a_new = a^(1-t) * g^t` (geometric mean)

The geometric mean update preserves the "shape" better and might converge faster.

**Better subgradient with projection:**

The subgradient of `max(conv(a,a))` at `a` is in the convex hull of the gradients at the peak positions. If the peak is at positions `S`, the subgradient is:
`grad = sum_{m in S} w_m * d(conv_m)/da` where `w_m >= 0, sum w_m = 1`.

`d(conv_m)/da[j] = 2 * a[m-j]` (if `0 <= m-j < n`).

We can do:
1. Find peak set S (positions within epsilon of max).
2. Choose weights w_m (e.g., uniform, or proportional to the "excess" over the second-max).
3. Compute subgradient g.
4. Take a step: `a_new = a - eta * g`.
5. Project onto simplex.

With a good step size (e.g., backtracking or Barzilai-Borwein), this could converge well.

**Another idea: Use the connection to "balancing"**

The problem is equivalent to: find a probability vector a of length n that minimizes the max coefficient of A(x)^2 where A(x) = sum a[i] x^i.

This is related to the "flatness" of polynomial squares. There's a rich literature on this.

One approach from the literature: **greedy balancing**. Start with a uniform vector, then iteratively adjust the coordinates that most reduce the peak.

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def cd_refine(a, time_budget, rng, verbose=False):
    """Coordinate descent with precise golden section search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s
    conv = compute_conv(a)
    best_obj = float(np.max(conv))
    best_a = a.copy()
    sr = (np.sqrt(5) - 1) / 2.0
    stagnant = 0
    start = time.monotonic()
    sweep = 0
    
    while time.monotonic() - start < time_budget:
        obj_before = float(np.max(conv))
        
        if rng is not None and sweep % 5 == 4:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        
        for k in order:
            d = np.zeros(2 * n - 1)
            d[k:k+n] = a
            
            lo = -a[k]
            # Upper bound: ensure conv stays reasonable
            hi = max(2.0 / n, -lo + 1.0 / n)
            
            # Golden section
            a_lo = lo
            a_hi = hi
            c = a_hi - sr * (a_hi - a_lo)
            d_ = a_lo + sr * (a_hi - a_lo)
            
            def f(delta):
                val = conv + 2.0 * delta * d
                val[2*k] += delta * delta
                return float(np.max(val))
            
            fc = f(c)
            fd = f(d_)
            
            for _ in range(60):
                if fc < fd:
                    a_hi = d_
                    d_ = c
                    fd = fc
                    c = a_hi - sr * (a_hi - a_lo)
                    fc = f(c)
                else:
                    a_lo = c
                    c = d_
                    fc = fd
                    d_ = a_lo + sr * (a_hi - a_lo)
                    fd = f(d_)
            
            delta = (a_lo + a_hi) / 2.0
            new_peak = f(delta)
            
            if new_peak < obj_before - 1e-14:
                a[k] = max(0.0, a[k] + delta)
                conv += 2.0 * delta * d
                conv[2*k] += delta * delta
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a /= s_new
                    conv /= (s_new * s_new)
        
        sweep += 1
        obj_after = float(np.max(conv))
        
        if obj_after < best_obj - 1e-12:
            best_obj = obj_after
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 8:
                if rng is not None:
                    scale = 0.005 + 0.003 * min(stagnant, 20)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    conv = compute_conv(a)
                    best_obj = float(np.max(conv))
                    stagnant = 0
                else:
                    break
        
        if verbose and sweep % 5 == 0:
            print(f"    CD sweep {sweep}: obj={2*n*best_obj:.8f}, t={time.monotonic()-start:.0f}s")
    
    return best_a

def build_conv_matrix(a):
    """Build matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A

def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    # Get current peak as initial bound
    conv_self = compute_conv(a_norm)
    M_init = float(np.max(conv_self))
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1, M <= M_init * 1.5]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_g = float(np.sum(g_val))
                if s_g > 1e-10:
                    return g_val / s_g, float(M.value)
        except Exception:
            continue
    return None, None

def lp_polish_v2(a, time_budget, rng, verbose=False):

            best_a = best_local_a.copy()

def bb_subgradient_descent(a, time_budget, verbose=False):
    """Barzilai-Borwein subgradient descent with simplex projection."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    prev_a = a.copy()
    prev_g = None
    s_step = 0.0
    y_step = np.zeros(n)
    eta = 1.0 / np.sqrt(n)
    
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        # Compute subgradient
        conv = compute_conv(a)
        peak = float(np.max(conv))
        # Find active set (positions within 1% of peak)
        tol = 0.01 * peak
        active = np.where(conv >= peak - tol)[0]
        
        # Subgradient: average of gradients at active positions
        g = np.zeros(n)
        for m in active:
            for j in range(n):
                if 0 <= m - j < n:
                    g[j] += a[m - j]
        g /= len(active)
        g *= 2.0  # Factor of 2 from self-convolution
        
        # Barzilai-Borwein step size
        if prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-20 and abs(sy) > 1e-15:
                eta_bb = ss / abs(sy)
                eta = 0.5 * eta + 0.5 * np.clip(eta_bb, 0.01 * eta, 10 * eta)
        
        # Take step
        a_new = a - eta * g
        a_new = project_simplex(a_new)
        
        # Stagnation detection
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        
        if it - best_it > 5000:
            a = best_a.copy()
            eta = 1.0 / np.sqrt(n)
            prev_g = None
            best_it = it
        
        if verbose and it % 1000 == 0:
            print(f"    BB it{it}: obj={best_obj:.8f}, eta={eta:.6f}, t={time.monotonic()-start:.0f}s")
    
    return best_a

def resample_sequence(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
