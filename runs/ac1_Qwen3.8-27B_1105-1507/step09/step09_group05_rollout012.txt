"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def solve_lp_active(a, top_k=200, time_limit=None):
    """
    Solve LP with only the top-k active constraints.
    min M s.t. conv(a,g)[k] <= M for k in active_set, sum(g)=1, g>=0
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv_a = compute_conv(a_norm)  # not directly used but for reference
    
    # Get the top-k positions of the convolution of a with itself
    # (These are the likely active constraints)
    if top_k < 2 * n - 1:
        top_indices = np.argsort(conv_a)[::-1][:top_k]
    else:
        top_indices = np.arange(2 * n - 1)
    
    # Build constraint matrix (only for top_k constraints)
    # conv(a, g)[k] = sum_j a[j] * g[k-j]
    # For each k in top_indices, the constraint is: sum_j A[k,j] * g[j] <= M
    # where A[k,j] = a[j] if 0 <= k-j < n, else 0
    
    A_active = np.zeros((len(top_indices), n))
    for i, k in enumerate(top_indices):
        k = int(k)
        # g[j] contributes to position k when k-j is a valid index of a
        # conv(a,g)[k] = sum_{j: 0<=k-j<n} a[k-j] * g[j]
        # = sum_j a[k-j] * g[j] (where a[k-j] = 0 if k-j out of range)
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A_active[i, j] = a_norm[k - j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_active @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value), top_indices
        except Exception:
            continue
    return None, None, None

        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        if max_conv < 1e-30:
            break
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(y, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]
        
        # Barzilai-Borwein step size
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.3 * lr + 0.7 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9997)
        
        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it
    
    return best_a


def perturb_and_reoptimize(a, rng, time_budget=30.0):
    """Perturb the sequence and re-optimize."""
    n = len(a)
    a = a.copy()
    
    method = rng.integers(0, 6)
    if method == 0:
        scale = 0.03
        a = a * np.exp(rng.normal(0, scale, n))
    elif method == 1:
        # Swap random pairs
        for _ in range(n // 5):
            i, j = rng.choice(n, 2, replace=False)
            a[i], a[j] = a[j], a[i]
    elif method == 2:
        # Local smoothing
        kernel = np.array([0.2, 0.6, 0.2])
        a = np.convolve(a, kernel, mode='same')
    elif method == 3:
        # Frequency domain perturbation
        A = fft(a, 2 * n)
        A[1:-1] *= np.exp(1j * rng.normal(0, 0.05, len(A[1:-1])))
        a = ifft(A).real[:n]
        a = np.maximum(a, 0)
    elif method == 4:
        # Add uniform noise
        a = a + rng.uniform(-0.005, 0.005, n)
        a = np.maximum(a, 0)
    else:
        # Concentrate on peaks
        conv = compute_conv(a / a.sum())
        peak_idx = np.argmax(conv)
        # Reduce values that contribute to the peak
        for j in range(n):
            idx = peak_idx - j
            if 0 <= idx < n:
                a[j] *= 0.95
        a = np.maximum(a, 0)
    
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s
    
    # Quick re-optimization
    a = smoothed_gradient_descent(a, time_budget=min(20, time_budget * 0.4), tau_init=20, tau_end=200)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

Let me also reconsider the LP formulation. The key insight is:

Given a fixed `a`, the LP:
min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0

finds the "best" g to blend with a. The optimal M tells us how much we can reduce the peak by moving towards g.

But there's a subtlety: we want to minimize max(conv(a_new, a_new)) where a_new = (1-t)*a + t*g. The LP gives us min max(conv(a,g)), which is different from min max(conv((1-t)a+tg, (1-t)a+tg)).

Actually, conv((1-t)a+tg, (1-t)a+tg) = (1-t)^2 conv(a,a) + 2t(1-t) conv(a,g) + t^2 conv(g,g)

So the objective is NOT linear in t! The LP only helps with the cross term conv(a,g). The t^2 conv(g,g) term is quadratic.

This means the line search is important, and the LP solution might not be the best direction for large t values.

For small t, the improvement is approximately:
obj(t) ≈ (1-t)^2 * obj_a + 2t(1-t) * M_lp + t^2 * obj_g

where obj_a = max(conv(a,a)), M_lp = min_g max(conv(a,g)), obj_g = max(conv(g,g)).

The derivative at t=0 is: -2*obj_a + 2*M_lp = 2*(M_lp - obj_a)

If M_lp < obj_a (which it should be, since the LP is minimizing the cross term), then the objective decreases for small t.

This is good - the LP always gives a descent direction (for small t).

Now, for the improvement: the issue is that the line search might not find the optimal t, and the quadratic term t^2 * obj_g might dominate for large t.

**Better approach**: Instead of a fixed set of t values, use a proper line search with backtracking:
- Start with t = min(1, 2*(obj_a - M_lp) / obj_g) (from the quadratic model)
- If obj(t) > obj_a, halve t and retry

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses active-set LP polishing with adaptive constraint selection, smoothed gradient
descent with Nesterov momentum, coordinate refinement, and multi-scale exploration.
Key improvements: active-set LP for faster iterations, quadratic-model line search,
adaptive constraint management, and better time allocation.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_row(a_norm, k):
    """Build the k-th row of the convolution matrix: conv(a, g)[k] for variable g."""
    n = len(a_norm)
    row = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    for j in range(lo, hi):
        row[j] = a_norm[k - j]
    return row


def solve_lp_full(a_norm, constraint_indices):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for k in constraint_indices, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a_norm)
    num_constraints = len(constraint_indices)
    
    A = np.zeros((num_constraints, n))
    for i, k in enumerate(constraint_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[i, j] = a_norm[k - j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def quadratic_line_search(a, g_lp, obj_a, M_lp):
    """
    Line search using quadratic model: obj(t) ~= (1-t)^2*obj_a + 2t(1-t)*M_lp + t^2*obj_g
    Returns best (a_new, obj_new, t).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    g_norm = g_lp / float(np.sum(g_lp))
    
    # Compute obj_g = max(conv(g,g))
    conv_g = compute_conv(g_norm)
    obj_g = float(np.max(conv_g))
    
    # Quadratic model: q(t) = (1-t)^2 * obj_a + 2*t*(1-t) * M_lp + t^2 * obj_g
    # q(t) = obj_a + t*(2*M_lp - 2*obj_a) + t^2*(obj_a - 2*M_lp + obj_g)
    # q'(t) = 2*(M_lp - obj_a) + 2*t*(obj_a - 2*M_lp + obj_g)
    # Optimal t from model:
    c2 = obj_a - 2.0 * M_lp + obj_g
    c1 = 2.0 * (M_lp - obj_a)
    if c2 > 1e-15:
        t_model = max(0.0, -c1 / (2.0 * c2))
    else:
        t_model = 1.0
    t_model = min(t_model, 1.0)
    
    # Try candidates around t_model
    candidates = set()
    candidates.add(max(0.01, t_model))
    candidates.add(max(0.01, t_model * 0.5))
    candidates.add(max(0.01, t_model * 0.7))
    candidates.add(max(0.01, t_model * 1.3))
    candidates.add(max(0.01, t_model * 1.5))
    candidates.add(0.05)
    candidates.add(0.1)
    candidates.add(0.2)
    candidates.add(0.3)
    candidates.add(0.5)
    candidates.add(0.7)
    candidates.add(0.9)
    candidates.add(1.0)
    
    candidates = [t for t in candidates if 0.01 <= t <= 1.0]
    
    best_obj = np.inf
    best_a = a_norm.copy()
    best_t = 0.0
    
    for t in sorted(candidates):
        a_try = (1.0 - t) * a_norm + t * g_norm
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            best_t = t
    
    return best_a, best_obj, best_t

    """
    LP polishing with adaptive active set and quadratic line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf, 0
    a = a / s
    
    if initial_top_k is None:
        initial_top_k = min(200, 2 * n - 1)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    no_improve = 0
    last_t = 0.4
    start = time.monotonic()
    it = 0
    current_top_k = initial_top_k
    
    while it < max_iter and (time.monotonic() - start) < time_limit:
        a_norm = a / float(np.sum(a))
        conv_a = compute_conv(a_norm)
        
        # Select active constraints: top-k positions in conv(a,a)
        if current_top_k < 2 * n - 1:
            top_indices = np.argsort(conv_a)[::-1][:current_top_k]
        else:
            top_indices = np.arange(2 * n - 1)
        
        g_lp, M_lp = solve_lp_full(a_norm, top_indices)
        if g_lp is None:
            current_top_k = min(current_top_k * 2, 2 * n - 1)
            no_improve += 2
            if no_improve >= 8:
                break
            continue
        
        # Quadratic line search
        a_new, obj_new, t_used = quadratic_line_search(a, g_lp, float(np.max(conv_a)), M_lp)
        
        it += 1
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            last_t = t_used
            no_improve = 0
            # Shrink active set when making progress
            current_top_k = max(100, current_top_k // 2)
            if verbose and it % 5 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={t_used:.3f}, top_k={current_top_k}")
        else:
            no_improve += 1
            # Expand active set when stuck
            current_top_k = min(current_top_k * 2, 2 * n - 1)
            if no_improve >= 8:
                break
    
    return best_a, best_obj, it


def smoothed_gradient_descent(a, time_budget=100.0, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    lr = 2.5 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget * 0.9, 1e-10))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)
        
        if verbose and it % 10000 == 0 and it > 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}")
        
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        # Compute smoothed gradient via FFT
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        if max_conv < 1e-30:
            break
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(y, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]
        
        # Barzilai-Borwein step size
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.3 * lr + 0.7 * min(max(bb_lr, 1e-8), 12.0 / np.sqrt(n))
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9997)
        
        if it - best_it > 25000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.5 / np.sqrt(n) * 0.5
            best_it = it
    
    return best_a


def coordinate_refinement(a, time_limit=60.0, verbose=False):
    """
    Local coordinate optimization: try adjusting individual and pairs of coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)
    it = 0
    no_improve = 0
    
    while (time.monotonic() - start) < time_limit and no_improve < 50:
        # Try a random coordinate
        idx = rng.integers(0, n)
        old_val = a[idx]
        
        best_tried = True
        if old_val > 1e-10:
            factors = [0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0]
        else:
            factors = [0.001, 0.005, 0.01, 0.02, 0.05]
        
        for f in factors:
            a_try = best_a.copy()
            if old_val > 1e-10:
                a_try[idx] = old_val * f
            else:
                a_try[idx] = f / n * 10
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                no_improve = 0
                best_tried = False
                break
        
        if best_tried:
            no_improve += 1
        
        # Also try pair optimization every 5 iterations
        if it % 5 == 4:
            idx1 = rng.integers(0, n)
            idx2 = rng.integers(0, n)
            if idx1 != idx2:
                v1, v2 = a[idx1], a[idx2]
                for f1 in [0.5, 1.0, 1.5, 2.0]:
                    for f2 in [0.5, 1.0, 1.5, 2.0]:
                        a_try = best_a.copy()
                        a_try[idx1] = max(0, v1 * f1)
                        a_try[idx2] = max(0, v2 * f2)
                        s_try = float(np.sum(a_try))
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-13:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            no_improve = 0
                            break
                    else:
                        continue
                    break
        
        it += 1
    
    if verbose:
        print(f"    Coord refine: obj={best_obj:.8f} after {it} iters")
    return best_a, best_obj


def perturb_seq(a, rng, method=0):
    """Apply a perturbation to escape local minima."""
    n = len(a)
    a = a.copy()
    if method == 0:
        a = a * np.exp(rng.normal(0, 0.03, n))
    elif method == 1:
        for _ in range(n // 5):
            i, j = rng.choice(n, 2, replace=False)
            a[i], a[j] = a[j], a[i]
    elif method == 2:
        kernel = np.array([0.15, 0.7, 0.15])
        a = np.convolve(a, kernel, mode='same')
    elif method == 3:
        A = fft(a, 2 * n)
        A[1:-1] *= np.exp(1j * rng.normal(0, 0.04, len(A[1:-1])))
        a = ifft(A).real[:n]
        a = np.maximum(a, 0)
    elif method == 4:
        conv = compute_conv(a / max(a.sum(), 1e-10))
        peak_idx = int(np.argmax(conv))
        for j in range(n):
            idx = peak_idx - j
            if 0 <= idx < n:
                a[j] *= 0.93
    else:
        a = a * np.exp(rng.normal(0, 0.05, n))
    
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, mode, rng):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = np.exp(-3.0 * x)
        a = a / a.sum()
    elif mode == 2:
        a = np.exp(-3.0 * (1.0 - x))
        a = a / a.sum()
    elif mode == 3:
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * x)
        a = a / a.sum()
    elif mode == 4:
        a = np.sin(np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    return a
