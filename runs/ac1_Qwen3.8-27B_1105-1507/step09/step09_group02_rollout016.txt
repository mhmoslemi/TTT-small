import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n - 1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        r = min(n, 2*n-1-j)
        A[j:j+r, j] = a[:r]
    return A

def solve_lp_with_dual(a, use_dual=True):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.

    """

    a_norm = a / s
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    # Get dual values for the inequality constraints
    A_dual = None
    for con in constraints:
        if hasattr(con, 'dual_value') and con.dual_value is not None:
            if con.dual_value.shape == (2*n-1,):
                A_dual = con.dual_value
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:

            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)

                            if con.dual_value.shape == (2*n-1,):
                                dual_y = np.abs(con.dual_value)
                                break
                    return g_val, float(M.value), dual_y
        except Exception:
            continue
    return None, None, None

def fourier_smoothing(a, bandwidth, time_budget):
    """
    Use Fourier domain to smooth the sequence while preserving non-negativity.
    Iterate: FFT -> zero out high freq -> IFFT -> project to non-negative -> renormalize
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for bw in bandwidths_if_provided(bandwidth):
        # Zero out frequencies above bw
        fa = np.fft.fft(a)
        mask = np.zeros(n, dtype=bool)
        mask[:bw+1] = True
        mask[-bw:] = True
        mask[n//2 - bw//2 : n//2 + bw//2 + 1] = True
        fa_zeroed = fa * mask
        a_smooth = np.fft.ifft(fa_zeroed).real
        a_smooth = np.maximum(a_smooth, 0)
        s_s = float(np.sum(a_smooth))
        if s_s > 1e-10:
            a_smooth = a_smooth / s_s
            obj_smooth = compute_obj(a_smooth)
            if obj_smooth < best_obj:
                best_obj = obj_smooth
                best_a = a_smooth.copy()
    
    return best_a

def bandwidths_if_provided(bandwidth):
    return bandwidth if isinstance(bandwidth, list) else [bandwidth]

def pairwise_swap(a, i, j, delta):
    """Try moving delta mass from position i to position j."""
    a_new = a.copy()
    if a_new[i] >= delta:
        a_new[i] -= delta
        a_new[j] += delta
    else:
        a_new[j] += a_new[i]
        a_new[i] = 0
    return a_new

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def nesterov_descent_v2(a, time_budget, tau_init=20.0, tau_end=500.0, verbose=False):
    """
    Improved Nesterov smoothed gradient descent with better

    """

    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)
        
        y = a + beta * v
        y = np.maximum(y, 0)

        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / (float(np.sum(exp_vals)) + 1e-30)
        
        # Gradient: grad_i = 2 * sum_k p[k] * y[k-i] (for k-i in [0,n-1])
        # This is a cross-correlation of p and y

        P = np.fft.fft(p, N)
        Y = np.fft.fft(y, N)
        g_full = np.fft.ifft(P * np.conj(Y)).real
        g = 2.0 * g_full[:n]
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
            lr *= 0.5
        else:
            a_new = a_new / s_new

            best_a = a_new.copy()

            v = np.zeros(n)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous:
1. Golden-section line search for LP blending (instead of fixed t grid)
2. Symmetry-constrained LP variant
3. SQP-style local refinement
4. Better time allocation and multi-start strategy

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time
import math

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:min(j+n, 2*n-1), j] = a[:min(n, 2*n-1-j)]
    return A

def solve_lp(a, symmetry=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    If symmetry=True, also constrain g[i] = g[n-1-i].
    """
    n = len(a)
    s = float(np.sum(a))
    a_norm = a / s
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    if symmetry and n > 1:
        for i in range(n // 2):
            constraints.append(g[i] == g[n-1-i])
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            if solver == cp.MOSEK:
                prob.solve(solver=solver, msdk_params={"MSDK_OPTIMALITY_TOL": 1e-9})
            else:
                prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    return g_val, float(M.value)
        except Exception:
            continue
    return None, None

def golden_search_blend(a, g, best_obj):
    """
    Golden section search over t in [0,1] to find best blend (1-t)*a + t*g.
    """
    def f(t):
        a_try = (1-t)*a + t*g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            return np.inf
        return compute_obj(a_try / s)
    
    GR = (math.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    
    lo, hi = 0.0, 1.0
    c = hi - GR * (hi - lo)
    d = lo + GR * (hi - lo)
    fc = f(c)
    fd = f(d)
    
    # Also check t=0 and t=1
    f0 = f(0.0)
    f1 = f(1.0)
    
    best_t = 0.0
    best_f = min(f0, fc, fd, f1)
    if best_f == f0:
        best_t = 0.0
    elif best_f == f1:
        best_t = 1.0
    elif best_f == fc:
        best_t = c
    else:
        best_t = d
    
    for _ in range(20):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - GR * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + GR * (hi - lo)
            fd = f(d)
        
        # Track best
        if fc < best_f:
            best_f = fc
            best_t = c
        if fd < best_f:
            best_f = fd
            best_t = d
    
    if best_f >= best_obj:
        return None, best_obj
    return best_t, best_f

def multiplicative_descent(a, time_budget, verbose=False):
    """
    Multiplicative weight / mirror descent with negative entropy.
    Update: a[i] *= exp(-eta * grad[i])
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.1
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        
        # Softmax weights
        tau = 50.0
        exp_vals = np.exp(tau * (conv - max_c))
        p = exp_vals / (float(np.sum(exp_vals)) + 1e-30)
        
        # Gradient of smoothed max: grad_i = 2 * sum_k p[k] * a[k-i]
        # This is cross-correlation of p and reversed a
        # (a * p)[k] = sum_i a[i] p[k-i]
        # But we want sum_k p[k] a[k-i] which is (a flipped * p)[i]... 
        # Actually: sum_k p[k] * a[k-i] for k-i in [0,n-1]
        # = sum_j p[i+j] * a[j]  where j = k-i
        # This is the cross-correlation of p and a evaluated at lag i
        
        # Use FFT: ifft(fft(p) * conj(fft(a)))... 
        # Actually let's just do it directly for clarity (n is at most ~1200)
        N = 1
        while N < 3*n:
            N <<= 1
        P = np.fft.fft(p, N)
        A = np.fft.fft(a, N)
        # cross-correlation: ifft(conj(P) * A) gives sum_k p[k] a[k+n-i]... 
        # Let me be more careful.
        # We want g[i] = sum_k p[k] a[k-i] for valid k-i
        # = sum_j p[i+j] a[j]
        # This is (p * a)[i] where * is cross-correlation
        # In FFT: ifft(conj(FFT(p)) * FFT(a))
        g_full = np.fft.ifft(np.conj(P) * A).real
        g = 2.0 * g_full[:n]
        
        # Multiplicative update
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = best_a
            stall += 1
            if stall % 50 == 0:
                eta *= 0.7
                stall = 0
            if eta < 1e-6:
                break
        
        it += 1
        if verbose and it % 200 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, eta={eta:.6f}")
    
    return best_a

def resample_sequence(a, new_n):
    """Resample via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def local_refinement(a, time_budget, verbose=False):
    """
    Coordinate descent: try adjusting individual elements.
    For each i, try increasing and decreasing a[i] while maintaining sum=1.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(321)
    
    while time.monotonic() - start < time_budget:
        # Pick a random set of coordinates to try
        num_coords = min(n, 50)
        indices = rng.choice(n, size=num_coords, replace=False)
        
        improved = False
        for i in indices:
            # Try moving mass from i to a random other position
            j = rng.integers(0, n)
            if i == j:
                continue
            
            for frac in [0.01, 0.05, 0.1, 0.2]:
                delta = frac * a[i]
                if delta < 1e-10:
                    continue
                
                a_new = best_a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                s_new = float(np.sum(a_new))
                if s_new < 1e-10:
                    continue
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    improved = True
                    break
        
        if not improved:
            # Try small random perturbations
            a_try = best_a + rng.normal(0, 0.001, n)
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 0:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-15:
                    best_obj = obj_try
                    best_a = a_try.copy()
    
    return best_a
