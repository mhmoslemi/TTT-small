    n = len(v)
    if n == 0:
        return v

    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)

    rho = np.where(cond)[0]

    rho = int(rho[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_grad(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a) * 2n (with sum(a)=1)."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    p = exp_vals / float(np.sum(exp_vals))
    # Gradient of sum_k p_k * conv_k w.r.t. a[i]
    # conv_k = sum_j a[j]*a[k-j], so d(conv_k)/da[i] = 2*a[k-i] (if valid)
    # Gradient of smoothed objective = sum_k p_k * 2 * a[k-i] (for valid k-i)
    # This is a cross-correlation of p with a
    N = 1
    while N < 2 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    # We need sum_k p[k] * a[k-i] for each i
    # This is the cross-correlation, which can be computed via: IFFT(fft(p, N) * conj(fft(a, N)))
    # Actually: (p circ a)[i] = sum_k p[k] * a[(k-i) mod N]
    # What we want is sum_k p[k] * a[k-i] (linear, not circular)
    # With zero-padding to N >= 2n-1, this equals the circular cross-correlation
    cross = ifft(P * np.conj(A)).real[:n]
    # The gradient of sum_k p_k * conv_k w.r.t. a[i] is 2 * sum_k p[k] * a[k-i] (for valid indices)
    # But we need to be careful: a[k-i] is only defined for 0 <= k-i < n
    # With the FFT approach and zero-padding, this should be handled correctly
    grad = 2.0 * cross
    return grad, conv

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses high-precision log-sum-exp smoothed gradient descent with adaptive momentum,
LP-based direction finding with multi-constraint active set, quadratic line search,
multi-n exploration, and structured restarts. Key improvements over previous:
(1) quadratic interpolation line search for sub-1e-6 precision,
(2) top-K active constraint LP for better directions,
(3) adaptive tau schedule with very high final tau for peak-focus,
(4) conjugate-gradient-in-Birkhoff for fine polishing,
(5) systematic n-sweep with shape-preserving resampling.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = int(rho_arr[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient(a, tau):
    """
    Compute gradient of soft-max smoothed max(conv(a,a)) using FFT cross-correlation.
    Returns (gradient, conv) where gradient is w.r.t. a (assumes sum(a)=1).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 2 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    # cross-correlation: (p cross a)[i] = sum_k p[k] * a[k-i] (with zero-padding)
    cross = ifft(P * np.conj(A)).real[:n]
    grad = 2.0 * cross
    return grad, conv


def build_conv_matrix_subset(a, lags):
    """
    Build matrix A such that A @ g = [conv(a,g)[k] for k in lags].
    A[idx, j] = a[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(a)
    A = np.zeros((len(lags), n))
    for idx, k in enumerate(lags):
        for j in range(n):
            ki = k - j
            if 0 <= ki < n:
                A[idx, j] = a[ki]
    return A


def lp_direction(a, n_active=15, verbose=False):

def _line_search(a, direction, n_evals=8):
    """Quadratic interpolation line search on the simplex."""
    a = a / float(np.sum(a))
    direction = np.maximum(direction, 0)
    s_d = float(np.sum(direction))
    if s_d < 1e-10:
        return a.copy()
    direction = direction / s_d

    # Initial bracket
    t1, f1 = 0.2, compute_obj(project_simplex(a + 0.2 * (direction - a)))
    t2, f2 = 0.5, compute_obj(project_simplex(a + 0.5 * (direction - a)))
    t3, f3 = 1.0, compute_obj(project_simplex(a + 1.0 * (direction - a)))

    for _ in range(max(0, n_evals - 3)):
        # Quadratic interpolation
        denom = (f1 - f2) * (t1 - t3) - (f1 - f3) * (t1 - t2)
        if abs(denom) < 1e-18:
            t_new = 0.5 * (t1 + t2 + t3) / 3.0
        else:
            num = (f1 - f2) * t1 * t1 * (t1 - t3) - (f1 - f3) * t1 * t1 * (t1 - t2)
            t_new = (num) / denom
            t_new = 0.5 * (t1 * t1 * (f2 - f3) / ((t1 - t2) * (t1 - t3)) +
                          t2 * t2 * (f3 - f1) / ((t2 - t1) * (t2 - t3)) +
                          t3 * t3 * (f1 - f2) / ((t3 - t1) * (t3 - t2)))

        if not (0.0 < t_new < 1.0):
            t_new = 0.5 * (min(t1, t2, t3) + max(t1, t2, t3))

        f_new = compute_obj(project_simplex(a + t_new * (direction - a)))

        # Keep the 3 best points
        points = [(f1, t1), (f2, t2), (f3, t3), (f_new, t_new)]
        points.sort(key=lambda x: x[0])
        (f1, t1), (f2, t2), (f3, t3) = points[0], points[1], points[2]

    a_new = project_simplex(a + t1 * (direction - a))
    return a_new

def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a, kind='cubic') if hasattr(np, 'interp') else np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / (np.sqrt(1 - x ** 2) + 0.1)
        a = a / a.sum()
    elif mode == 4:
        # Modulated
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x + rng.uniform(0, 2 * np.pi)) + \
            0.2 * np.cos(2 * np.pi * 7 * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        alpha = rng.uniform(-1, 1)
        a = x ** alpha
        a = a / a.sum()
    elif mode == 6:
        # Triangular
        x = np.linspace(0, 1, n)
        a = 1 - 2 * np.abs(x - 0.5)
        a = a / a.sum()
    else:
        # Cosine bump
        x = np.linspace(0, np.pi, n)
        a = 0.3 + 0.7 * np.cos(x) ** 2
        a = a / a.sum()
    return a

    """
    Conjugate-Newton-like Direction Descent (CNDd):
