import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed objective using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

    direction = direction - direction.sum() / n  # Make it sum-preserving... actually no

def nesterov_smoothed_gd(a, time_budget, tau_init=10.0, tau_end=2000.0, verbose=False):
    """Nesterov accelerated smoothed gradient descent with cosine tau annealing."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 3.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    it = 0
    best_it = 0
    grad_history = []

    while time.monotonic() - start < time_budget:

        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        
        # Cosine tau annealing
        tau = tau_init + (tau_end - tau_init) * (1 - np.cos(np.pi * progress)) / 2

            print(f"      GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")

            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)
        g_norm = float(np.linalg.norm(g))
        
        if it % 2 == 0:
            grad_history.append(g_norm)
            if len(grad_history) > 200:
                grad_history = grad_history[-200:]
            avg_gn = np.mean(grad_history)
            if avg_gn > 1e-15:
                adaptive_lr = 0.5 / (avg_gn + 1e-10)
                lr = 0.5 * lr + 0.5 * min(max(adaptive_lr, 0.1 * base_lr), 5.0 * base_lr)

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        lr = max(1e-10, lr * 0.99995)
        it += 1

        # Reset on stagnation
        if it - best_it > 50000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it

    return best_a


def lp_polish_with_linesearch(a, time_budget, verbose=False, seed=42):
    """LP polishing with proper line search and adaptive perturbations."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    iter_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        # Direction
        direction = g_lp - a

        # Line search
        best_alpha = 0
        best_alpha_obj = best_obj
        alphas = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]
        for alpha in alphas:
            a_try = a + alpha * direction
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_alpha_obj:
                best_alpha_obj = obj_try
                best_alpha = alpha

        iter_count += 1
        if best_alpha_obj < best_obj - 1e-12:
            best_obj = best_alpha_obj
            a = best_a = (a + best_alpha * direction)
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0
            if verbose and iter_count % 5 == 0:
                print(f"      LP it{iter_count}: obj={best_obj:.8f}, alpha={best_alpha:.3f}")
        else:

                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if iter_count > 100:
                    break

    return best_a


def palindromic_init(n, rng):
    """Create a palindromic random initialization."""
    half = (n + 1) // 2
    a_half = rng.uniform(0.5, 1.5, half)
    a = np.zeros(n)
    a[:half] = a_half
    a[half:] = a_half[:n-half][::-1]
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def fourier_init(n, rng, n_modes=5):
    """Initialize using sum of cosines (palindromic by construction)."""
    a = np.ones(n) / n
    for k in range(1, n_modes + 1):
        amp = rng.uniform(-0.3, 0.3) / k
        a += amp * np.cos(2 * np.pi * k * np.arange(n) / n)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with adaptive line search and dual-informed perturbations
2. Nesterov smoothed gradient descent with aggressive cosine tau annealing
3. Palindromic structure exploitation
4. Multi-n systematic exploration with diverse restarts
5. Convolution equalization and active-set refinement
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_gradient_fft(a, tau):
    """Compute gradient of LSE-smoothed max-conv using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def build_conv_matrix(a):
    """Build matrix (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k]<=M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def nesterov_gd(a, time_budget, tau_init=20.0, tau_end=3000.0, verbose=False):
    """Nesterov accelerated smoothed GD with cosine tau annealing."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    base_lr = 3.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    it = 0
    best_it = 0
    gn_hist = []

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * (1 - np.cos(np.pi * progress)) / 2

        if it % 5000 == 0 and verbose:
            print(f"      GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.5f}")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)
        if it % 2 == 0:
            gn_hist.append(float(np.linalg.norm(g)))
            if len(gn_hist) > 200:
                gn_hist = gn_hist[-200:]
            avg_gn = np.mean(gn_hist)
            if avg_gn > 1e-15:
                adapt_lr = 0.5 / (avg_gn + 1e-10)
                lr = 0.5 * lr + 0.5 * min(max(adapt_lr, 0.1 * base_lr), 5.0 * base_lr)

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        lr = max(1e-10, lr * 0.99997)
        it += 1

        if it - best_it > 40000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it

    return best_a


def lp_polish(a, time_budget, verbose=False, seed=42):
    """LP best-response polishing with adaptive line search and perturbation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    it = 0
    last_alpha = 0.3

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break
        t0 = time.monotonic()
        g_lp, M_lp = solve_lp(a)
        solve_time = time.monotonic() - t0
        if g_lp is None:
            break

        direction = g_lp - a
        # Adaptive line search centered around last good alpha
        best_alpha = 0.0
        best_alpha_obj = best_obj
        alphas_base = [0.02, 0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.8, 1.0]
        # Add alphas near last good value
        alphas = list(set(alphas_base + [
            max(0.01, last_alpha * 0.5), max(0.01, last_alpha * 0.8),
            last_alpha, min(1.0, last_alpha * 1.2), min(1.0, last_alpha * 1.5)
        ]))
        alphas.sort()
        for alpha in alphas:
            a_try = a + alpha * direction
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_alpha_obj:
                best_alpha_obj = obj_try
                best_alpha = alpha

        it += 1
        if best_alpha_obj < best_obj - 1e-13:
            best_obj = best_alpha_obj
            a = (a + best_alpha * direction)
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_a = a.copy()
            last_alpha = best_alpha
            stall = 0
            if verbose and it % 3 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, alpha={best_alpha:.3f}, M={M_lp:.6f}")
        else:
            stall += 1
            if stall > 2:
                # Guided perturbation: push mass toward low-conv regions
                conv = compute_conv(a)
                max_c = conv.max()
                min_c = conv.min()
                if max_c > min_c + 1e-10:
                    # Perturb in direction that raises low conv values
                    strength = 0.008 + 0.003 * min(stall, 15)
                    # Use Fourier perturb for smoothness
                    pert = np.zeros(n)
                    for k in range(1, 4):
                        pert += strength * rng.uniform(0.5, 1.5) * np.cos(
                            2 * np.pi * k * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
                    a = best_a * np.exp(pert)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                else:
                    a = best_a + rng.normal(0, 0.005, n)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                stall = 0
                if it > 200:
                    break

    return best_a


def palindromic_init(n, rng):
    """Palindromic random initialization."""
    half = (n + 1) // 2
    a_half = rng.uniform(0.5, 1.5, half)
    a = np.zeros(n)
    a[:half] = a_half
    a[half:] = a_half[:n - half][::-1]
    a = np.maximum(a, 0.01)
    return a / a.sum()


def fourier_init(n, rng, n_modes=5):
    """Sum-of-cosines initialization (automatically palindromic)."""
    a = np.ones(n) / n
    for k in range(1, n_modes + 1):
        amp = rng.uniform(-0.35, 0.35) / k
        a += amp * np.cos(2 * np.pi * k * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0.01)
    return a / a.sum()


def resample(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def equalize_convolution(a, time_budget, n_iter=50):
    """Try to equalize the convolution by GD on sum of squared deviations from mean."""
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.02 / np.sqrt(n)

    for it in range(n_iter):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        target = np.mean(conv)
        # Gradient of sum((conv[k]-target)^2) w.r.t. a
        # d(sum(conv^2))/da = 2 * sum_k conv[k] * d(conv[k])/da
        # d(conv[k])/da[i] = 2*a[k-i] (approx)
        # Use FFT: the gradient is essentially the cross-correlation of conv with a, doubled
        N = 1
        while N < 3 * n:
            N <<= 1
        C = fft(conv[:N], N)
        A = fft(a, N)
        g = ifft(C * np.conj(A)).real[:n] * 4.0
        # Subtract mean of conv * 2 to account for target
        g -= 2.0 * target * 2.0 * a  # approximation
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
        lr *= 0.999

    return best_a
