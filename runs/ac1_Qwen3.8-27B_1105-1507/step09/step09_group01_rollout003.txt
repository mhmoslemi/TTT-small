"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a multi-phase approach: smoothed gradient descent with FISTA acceleration,
LP-based alternating minimization with dual-guided perturbations, multi-length
exploration, and targeted peak reduction. Key improvements over previous:
dual-informed LP perturbations, FISTA smoothing, adaptive backtracking,
and frequency-domain diversification.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt, dual_weights) or (None, None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0).copy()
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Get dual values for the constraint A@g <= M
                    dual_w = None
                    try:
                        # First constraint is A@g <= M
                        if len(prob.constraints) >= 1:
                            dc = prob.constraints[0].dual_value
                            if dc is not None:
                                dual_w = np.maximum(np.asarray(dc).ravel(), 0)
                    except Exception:
                        dual_w = None
                    return g_val, float(M.value), dual_w
            elif prob.status == 'infeasible':
                return None, None, None
        except Exception:
            continue
    return None, None, None

def smoothed_objective_and_grad(a, tau):
    """
    Compute smoothed objective (log-sum-exp of conv) and its gradient.
    Returns (smoothed_val, gradient).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    logsumexp = max_c + np.log(float(np.sum(exp_shifted))) / tau
    p = exp_shifted / float(np.sum(exp_shifted))

    # Gradient of smoothed max wrt a:
    # d/d a_i [sum_k p_k * (a*a)[k]] = 2 * sum_k p_k * a[k-i] (with valid indices)
    # = 2 * (p * a)[i] (cross-correlation)
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    # cross-correlation: ifft(conj(P) * A)
    g_full = ifft(np.conj(P) * A).real
    grad = 2.0 * g_full[:n]
    return logsumexp, grad


def fista_smoothed(a, time_budget=120.0, tau_init=8.0, tau_end=500.0, verbose=False):
    """FISTA on the smoothed max-convolution objective over the simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = a.copy()
    y = a.copy()
    t_k = 1.0
    lr = 2.0 / np.sqrt(n)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.95))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        _, grad = smoothed_objective_and_grad(y, tau)

        # Gradient step with backtracking
        step = lr
        x_new = y - step * grad
        x_new = project_simplex(x_new)

        # Check improvement in smoothed objective
        obj_old, _ = smoothed_objective_and_grad(y, tau)
        obj_new, _ = smoothed_objective_and_grad(x_new, tau)

        if obj_new > obj_old + 1e-10:
            # Backtrack
            for _ in range(10):
                step *= 0.5
                x_new = y - step * grad
                x_new = project_simplex(x_new)
                obj_new, _ = smoothed_objective_and_grad(x_new, tau)
                if obj_new <= obj_old + 1e-10:
                    break
            else:
                x_new = y.copy()

        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k ** 2)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        x = x_new
        t_k = t_new

        # Track best by true objective
        if int(x.sum() * 10000) % 50 == 0 or elapsed < 1.0:
            obj_true = compute_obj(x)
            if obj_true < best_obj - 1e-12:
                best_obj = obj_true
                best_a = x.copy()

        # Periodic true objective check
        if time.monotonic() - start > 5 and (int(elapsed * 2) != int((time.monotonic() - start - 0.01) * 2)):
            obj_true = compute_obj(x)
            if obj_true < best_obj - 1e-12:
                best_obj = obj_true
                best_a = x.copy()

        lr = max(1e-10, lr * 0.9998)

    return best_a

    best_dual = None

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start

        # Solve LP with current a
        g_lp, M_lp, dual_w = solve_lp_minpeak(best_a)
        if g_lp is None:
            break

        if dual_w is not None:
            best_dual = dual_w

        # Golden section search for blend factor
        gr = (np.sqrt(5.0) - 1.0) / 2.0

        def blend_obj(t):
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return 1e10
            a_try = a_try / s_try
            return compute_obj(a_try)

        a_lo, a_hi = 0.01, 1.0
        c = a_hi - gr * (a_hi - a_lo)
        d = a_lo + gr * (a_hi - a_lo)
        fc = blend_obj(c)
        fd = blend_obj(d)
        for _ in range(30):
            if abs(a_hi - a_lo) < 0.012:
                break
            if fc < fd:
                a_hi = d
                d = c
                fd = fc
                c = a_hi - gr * (a_hi - a_lo)
                fc = blend_obj(c)
            else:
                a_lo = c
                c = d
                fc = fd
                d = a_lo + gr * (a_hi - a_lo)
                fd = blend_obj(d)
        t_opt = (a_lo + a_hi) / 2.0

        a_try = (1.0 - t_opt) * best_a + t_opt * g_lp
        a_try = np.maximum(a_try, 0)
        a_try = a_try / a_try.sum()
        obj_try = compute_obj(a_try)

        iter_count += 1

        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            last_t = t_opt
            no_improve = 0
            if verbose and iter_count % 3 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={t_opt:.4f}, M={M_lp:.7f}, el={elapsed:.0f}s")
        else:
            no_improve += 1

            if no_improve == 2:
                # Smaller step
                t_small = max(0.02, last_t * 0.15)
                a_try2 = (1.0 - t_small) * best_a + t_small * g_lp
                a_try2 = np.maximum(a_try2, 0)
                a_try2 = a_try2 / a_try2.sum()
                obj_try2 = compute_obj(a_try2)
                if obj_try2 < best_obj - 1e-14:
                    best_obj = obj_try2
                    best_a = a_try2.copy()
                    no_improve = 0
            elif no_improve >= 4:
                # Dual-guided perturbation: push mass away from high-dual positions
                if best_dual is not None and len(best_dual) == 2 * n - 1:
                    # Map dual weights to a-space: for each i, the "pressure" is
                    # sum_k dual[k] * a[k-i] (roughly)
                    pressure = np.zeros(n)
                    for i in range(n):
                        k_lo = max(0, i - (n - 1))
                        k_hi = min(2 * n - 2, i + (n - 1))
                        j_lo = k_lo - i
                        j_hi = k_hi - i
                        if j_lo >= 0 and j_hi < n:
                            pressure[i] = np.sum(best_dual[k_lo:k_hi + 1] * a[j_lo:j_hi + 1])
                    # Push a down where pressure is high
                    a_pert = best_a * np.exp(-0.02 * pressure / (np.max(pressure) + 1e-10))
                    a_pert = np.maximum(a_pert, 0)
                    a_pert = a_pert / a_pert.sum()
                else:
                    scale = 0.006 * min(no_improve, 10)
                    a_pert = best_a * np.exp(rng.normal(0, scale, n))
                    a_pert = np.maximum(a_pert, 0)
                    a_pert = a_pert / a_pert.sum()

                # Quick LP from perturbed point
                g_pert, M_pert, _ = solve_lp_minpeak(a_pert)
                if g_pert is not None:
                    a_pert2 = 0.5 * best_a + 0.5 * g_pert
                    a_pert2 = np.maximum(a_pert2, 0)
                    a_pert2 = a_pert2 / a_pert2.sum()
                    obj_pert2 = compute_obj(a_pert2)
                    if obj_pert2 < best_obj - 1e-14:
                        best_obj = obj_pert2
                        best_a = a_pert2.copy()
                        no_improve = 0
                    else:
                        # Try the perturbed point itself
                        obj_pert = compute_obj(a_pert)
                        if obj_pert < best_obj - 1e-14:
                            best_obj = obj_pert
                            best_a = a_pert.copy()
                            no_improve = 0

                if no_improve >= 8:
                    # More aggressive perturbation
                    scale = 0.02
                    a_pert = best_a * np.exp(rng.normal(0, scale, n))
                    a_pert = np.maximum(a_pert, 0)
                    a_pert = a_pert / a_pert.sum()
                    # Quick polish from here
                    for _ in range(3):
                        g_tmp, _, _ = solve_lp_minpeak(a_pert)
                        if g_tmp is None:
                            break
                        a_pert = 0.5 * a_pert + 0.5 * g_tmp
                        a_pert = np.maximum(a_pert, 0)
                        a_pert = a_pert / a_pert.sum()
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_obj - 1e-14:
                        best_obj = obj_pert
                        best_a = a_pert.copy()
                    no_improve = 0

                if iter_count > 50 and no_improve >= 12:
                    break

        # Periodic constraint tightening
        if iter_count % 6 == 3 and M_lp is not None:
            for tight_frac in [0.94, 0.97]:
                M_tight = M_lp * tight_frac
                g_tight, M_ach = solve_lp_constrained(best_a, M_tight)
                if g_tight is not None:
                    for t_try in [0.3, 0.5, 0.7]:
                        a_try3 = (1.0 - t_try) * best_a + t_try * g_tight
                        a_try3 = np.maximum(a_try3, 0)
                        a_try3 = a_try3 / a_try3.sum()
                        obj_try3 = compute_obj(a_try3)
                        if obj_try3 < best_obj - 1e-14:
                            best_obj = obj_try3
                            best_a = a_try3.copy()
                            no_improve = 0
                            if verbose:
                                print(f"    LP-tight it{iter_count}: obj={best_obj:.8f}, frac={tight_frac}, t={t_try}")
                            break

    return best_a


def targeted_peak_reduction(a, time_budget=100.0, verbose=False):
    """
    Directly target the peak of the convolution by reducing contributions.
    Uses a form of cyclic coordinate descent focused on peak minimization.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)

    lr = 1.0 / np.sqrt(n)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_k = int(np.argmax(conv))
        peak_val = float(conv[peak_k])

        # Gradient of (a*a)[peak_k] wrt a[i]:
        # d/d a[i] sum_j a[j]*a[peak_k - j] = a[peak_k - i] + (if peak_k != 2i) a[peak_k - i]
        # More carefully: (a*a)[k] = sum_j a[j] a[k-j]
        # d/d a[i] = a[k-i] + a[k-i] (if i != k-i) or a[k-i] (if i == k-i, i.e., k=2i)
        # Actually: d/d a[i] sum_j a[j] a[k-j]
        # = a[k-i] (from j=i term) + a[i] (from k-j=i, i.e., j=k-i term)
        # If i = k-i (i.e., k=2i): just a[i] (single term)
        # If i != k-i: a[k-i] + a[i]... no wait.
        # (a*a)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]
        # d/d a[i]: we get contributions from j=i (gives a[k-i]) and from k-j=i i.e. j=k-i (gives a[i])
        # But if i = k-i, these are the same term, so we just get a[i]
        # If i != k-i and both valid, we get a[k-i] + a[i]... 
        # Wait no. Let me be more careful.
        # (a*a)[k] = sum_j a[j] a[k-j]
        # d/d a[i] of this = sum_j [delta(j,i) a[k-j] + a[j] delta(k-j, i)]
        # = a[k-i] + a[k-i] (from the second sum when j = k-i)
        # Wait: second sum: sum_j a[j] delta(k-j, i) = a[k-i] (if k-i is valid)
        # First sum: sum_j delta(j,i) a[k-j] = a[k-i] (if k-i is valid)
        # So total = 2*a[k-i] if i != k-i and both i and k-i are valid
        #          = a[k-i] if i == k-i (i.e., k=2i)
        
        # Actually for the full convolution (a*a)[k] where a is defined on [0,n-1]:
        # (a*a)[k] = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[j] * a[k-j]
        # The derivative wrt a[i]:
        # Term j=i: a[k-i] (valid if k-i in [0,n-1])
        # Term j=k-i: a[i] (valid if k-i in [0,n-1], same condition)
        # If i = k-i (k=2i): only one term, derivative = a[i]
        # If i != k-i: two terms, derivative = 2*a[k-i]... 
        # Wait no. Let me redo.
        # sum_j a[j] * a[k-j], derivative wrt a[i]:
        # = a[k-i] (when j=i) + a[i] (when k-j=i, i.e., j=k-i)
        # If i = k-i: both give a[i], but it's the same j, so just a[i]
        # If i != k-i: gives a[k-i] + a[i]... no!
        # When j=i: term is a[i]*a[k-i], derivative wrt a[i] is a[k-i]
        # When j=k-i: term is a[k-i]*a[i], derivative wrt a[i] is a[k-i]
        # So total: if i != k-i: a[k-i] + a[k-i] = 2*a[k-i]
        #           if i == k-i: a[k-i] = a[i/2]... wait i=k/2, so a[k-i]=a[i]
        # So: derivative = 2*a[k-i] if i*2 != k, a[k-i] if i*2 == k
        # But we need k-i to be in [0, n-1]

        grad = np.zeros(n)
        for i in range(n):
            ki = k_peak - i if (k_peak := peak_k) else 0
            ki = peak_k - i
            if 0 <= ki < n:
                if 2 * i == peak_k:
                    grad[i] = a[ki]
                else:
                    grad[i] = 2.0 * a[ki]

        # Also add gradients from nearby peaks (top-5)
        top_idx = np.argsort(conv)[::-1][:5]
        for k2 in top_idx[1:]:
            w = conv[k2] / (conv[top_idx[0]] + 1e-15) * 0.3
            for i in range(n):
                ki = k2 - i
                if 0 <= ki < n:
                    if 2 * i == k2:
                        grad[i] += w * a[ki]
                    else:
                        grad[i] += w * 2.0 * a[ki]

        # Update
        a_new = a - lr * grad
        a_new = project_simplex(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.02, 5.0 / np.sqrt(n))
        else:
            lr *= 0.95
            if lr < 1e-8:
                break

    return best_a


def spectral_diversify(a, n_target=None, rng=None):
    """
    Create a new sequence by modifying the spectral content of a.
    Adds random phase perturbations to the DFT that preserve non-negativity.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    if n_target is not None and n_target != n:
        # Resample first
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, n_target)
        a = np.interp(x_new, x_old, a)
        n = n_target

    a = a / a.sum()

    # Get DFT
    N = 1
    while N < 2 * n:
        N <<= 1
    A = fft(a, N)

    # Perturb phases of non-DC components
    for f in range(1, N // 2):
        if rng.random() < 0.5:
            angle = rng.normal(0, 0.1)
            A[f] *= np.exp(1j * angle)
            A[N - f] *= np.exp(-1j * angle)

    # Inverse FFT
    a_new = ifft(A).real[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a
    a_new = a_new / s
    return a_new


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining FISTA smoothing,
    LP alternating minimization with dual-guided perturbations,
    targeted peak reduction, spectral diversification, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent if available
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        # Try several initializations
        for n_init in [400, 600, 800, 1000, 1200]:
            a_init = np.ones(n_init) / n_init
            obj_init = compute_obj(a_init)
            if obj_init < best_score:
                best_score = obj_init
                best_seq = a_init.copy()
                n_target = n_init
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: FISTA Smoothed Descent ===
    print(f"\n=== Phase 1: FISTA Smoothed Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.1, 120.0)
    if p1_budget > 15:
        a_fista = fista_smoothed(best_seq, time_budget=p1_budget,
                                 tau_init=8.0, tau_end=400.0, verbose=True)
        obj_fista = compute_obj(a_fista)
        if obj_fista < best_score:
            best_score = obj_fista
            best_seq = a_fista.copy()
            print(f"  Phase1 FISTA: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 FISTA: obj={obj_fista:.8f}")

    # === Phase 2: LP Alternating Minimization ===
    print(f"\n=== Phase 2: LP Alternating Minimization ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.3, 420.0)
    if p2_budget > 30:
        a_lp = lp_alternating(best_seq, time_budget=p2_budget, verbose=True, seed=42)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f}")

    # === Phase 3: Targeted Peak Reduction ===
    print(f"\n=== Phase 3: Targeted Peak Reduction ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.08, 80.0)
    if p3_budget > 10:
        a_tpr = targeted_peak_reduction(best_seq, time_budget=p3_budget, verbose=True)
        obj_tpr = compute_obj(a_tpr)
        if obj_tpr < best_score:
            best_score = obj_tpr
            best_seq = a_tpr.copy()
            print(f"  Phase3 TPR: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 TPR: obj={obj_tpr:.8f}")

    # === Phase 4: Spectral Diversification + Polish ===
    print(f"\n=== Phase 4: Spectral Diversification ===")
    remaining = deadline - time.monotonic()

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_spec = spectral_diversify(best_seq, rng=rng)
        obj_spec = compute_obj(a_spec)

        if obj_spec < best_score + 0.01:  # Only polish if close
            alloc = min(remaining * 0.2, 80.0)
            if alloc > 15:
                a_spec = lp_alternating(a_spec, time_budget=alloc, seed=200 + r)
                obj_spec = compute_obj(a_spec)
                if obj_spec < best_score:
                    best_score = obj_spec
                    best_seq = a_spec.copy()
                    print(f"  Spectral {r}: obj={best_score:.8f} **")
                else:
                    print(f"  Spectral {r}: obj={obj_spec:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(300, n_current - 200),
        max(300, n_current - 100),
        n_current + 100,
        n_current + 200,
        n_current + 400,
        500, 700, 1000, 1200, 1500, 1800
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000 and abs(n - n_current) > 80]
    n_options = sorted(n_options)[:5]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)

        # Quick FISTA
        alloc1 = min(remaining * 0.12, 30.0)
        if alloc1 > 8:
            a_try = fista_smoothed(a_try, time_budget=alloc1,
                                   tau_init=10.0, tau_end=200.0)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 80.0)
        if alloc2 > 15:
            a_try = lp_alternating(a_try, time_budget=alloc2, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Multi-scale Perturbation + LP ===
    print(f"\n=== Phase 6: Perturbation + LP ===")
    remaining = deadline - time.monotonic()

    for r in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        scale = 0.005 + 0.005 * r
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc = min(remaining * 0.2, 80.0)
        if alloc > 15:
            a_pert = lp_alternating(a_pert, time_budget=alloc, seed=500 + r)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Perturb {r} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                if r < 4:
                    print(f"  Perturb {r}: obj={obj_pert:.8f}")

    # === Phase 7: Final Intensive LP ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 150.0)
        a_final = lp_alternating(best_seq, time_budget=alloc, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

        # Quick TPR after
        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc_tpr = min(remaining * 0.4, 40.0)
            a_tpr = targeted_peak_reduction(best_seq, time_budget=alloc_tpr)
            obj_tpr = compute_obj(a_tpr)
            if obj_tpr < best_score:
                best_score = obj_tpr
                best_seq = a_tpr.copy()
                print(f"  Final TPR: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
