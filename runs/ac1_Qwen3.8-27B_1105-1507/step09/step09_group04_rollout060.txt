    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if n == 1:
        return np.maximum(v, 0) / max(1e-30, np.abs(v).sum() + 1e-30) if np.sum(np.maximum(v,0)) > 0 else np.ones(1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_idx = np.where(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(x):
    x = x - np.max(x)
    e = np.exp(x)
    return e / np.sum(e)


def compute_grad_fft(a, p):
    """Compute gradient of sum_m p[m]*conv(a,a)[m] w.r.t. a, using FFT.
    Returns grad[k] = 2 * sum_j p[k+j] * a[j] for k=0..n-1.
    """
    n = len(a)
    N = 1
    while N < 3 * n:
        N <<= 1
    # grad[k] = 2 * sum_{j=0}^{n-1} a[j] * p[k+j]
    # This is cross-correlation of a and p
    # Use FFT: correlate(a, p) at appropriate lags
    ar = a[::-1]
    fa = rfft(ar, N)
    fp = rfft(p, N)
    corr = irfft(fa * fp, N)
    # corr[m] = sum_j ar[j] * p[m-j] = sum_j a[n-1-j] * p[m-j]
    # We want c[k] = sum_{j=0}^{n-1} a[j] * p[k+j]
    # c[k] = corr[k + n - 1]
    c = corr[n - 1:2 * n - 1]
    return 2.0 * c


def fista_smoothed(a_init, time_budget=200.0, tau_init=5.0, tau_end=500.0, verbose=True):
    """FISTA on the log-sum-exp smoothed objective over the simplex."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_val = float(np.max(compute_conv(a)))
    start = time.monotonic()

    t = 1.0
    a_prev = a.copy()
    y = a.copy()
    lr = 1.0 / (4.0 * n)  # initial step size (Lipschitz constant ~ 4n for the gradient)
    it = 0
    best_it = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        # Compute gradient at y
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy < 1e-10:
            y = best_a.copy()
            sy = 1.0
        y = y / sy

        conv_y = compute_conv(y)
        max_c = float(np.max(conv_y))
        exp_vals = np.exp(tau * (conv_y - max_c))
        p = exp_vals / float(np.sum(exp_vals))

        # Gradient of (1/tau)*log(sum exp(tau*conv)) w.r.t. a
        # = sum_m p[m] * d(conv[m])/da
        # d(conv[m])/da[k] = 2*a[m-k] if 0<=m-k<n
        grad_a = compute_grad_fft(y, p)

        # Projected gradient step
        a_new = project_simplex(y - lr * grad_a)

        # Compute actual objective (unsmoothed) for tracking
        conv_new = compute_conv(a_new)
        val_new = float(np.max(conv_new))

        if val_new < best_val - 1e-12:
            best_val = val_new
            best_a = a_new.copy()
            best_it = it
            stall_count = 0
        else:
            stall_count += 1

        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        a_next = a_new + beta * (a_new - a_prev)
        a_next = project_simplex(a_next)

        # Barzilai-Borwein step size update (every 50 iters)
        if it > 50 and it % 50 == 0:
            s_vec = a_new - a_prev
            y_vec = grad_a  # approximate
            ss = float(np.dot(s_vec, s_vec))
            sy_dot = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and sy_dot > 1e-10:
                lr_bb = ss / sy_dot
                lr = 0.7 * lr + 0.3 * np.clip(lr_bb, 1e-6, 10.0 / n)

        a_prev = a_new.copy()
        a_new_copy = a_next.copy()
        a = a_new_copy
        y = a_next.copy()
        t = t_new
        it += 1

        # Reset if stalled too long
        if stall_count > 1000:
            a = best_a * np.exp(np.random.default_rng(it).normal(0, 0.01, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            a_prev = a.copy()
            y = a.copy()
            t = 1.0
            lr = 1.0 / (4.0 * n)
            stall_count = 0
            best_it = it

        if verbose and it % 500 == 0:
            obj = 2.0 * n * best_val
            print(f"    FISTA it{it}: obj={obj:.8f}, tau={tau:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a


def active_set_cd(a_init, time_budget=300.0, K=30, verbose=True, seed=42):
    """Coordinate descent that only checks top-K convolution positions."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s
    conv = compute_conv(a)

    best_obj = float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5) - 1) / 2.0

    # Get initial active set
    active = np.argsort(conv)[::-1][:K]
    sweep = 0
    stagnant = 0

    while time.monotonic() - start < time_budget:
        obj_before = float(np.max(conv))
        improved_any = False

        # Random order every few sweeps
        if sweep % 4 == 3:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for k in order:
            # Direction vector (only need values at active positions + 2k)
            # conv[m] changes by 2*delta*a[m-k] for m in [k, k+n-1], m!=2k
            # and by 2*a[k]*delta + delta^2 for m = 2k

            # Build check set
            check_pos = np.unique(np.concatenate([active, [2 * k]]))
            check_pos = check_pos[(check_pos >= 0) & (check_pos < 2 * n - 1)]

            # Get current values at check positions
            conv_check = conv[check_pos]

            # Direction at check positions
            d_check = np.zeros(len(check_pos))
            for i, m in enumerate(check_pos):
                if m == 2 * k:
                    d_check[i] = 2.0 * a[k]  # linear part; quadratic handled separately
                else:
                    diff = m - k
                    if 0 <= diff < n:
                        d_check[i] = 2.0 * a[diff]
                    else:
                        d_check[i] = 0.0

            # Golden section search
            lo_delta = -a[k]
            hi_delta = 2.0 / n
            # Ensure the interval brackets the minimum
            a_gs = lo_delta
            b_gs = hi_delta
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)

            def obj_at_delta(delta):
                vals = conv_check + delta * d_check
                # Add quadratic term at position 2k
                idx_2k = np.where(check_pos == 2 * k)[0]
                if len(idx_2k) > 0:
                    vals[idx_2k] += delta * delta
                return float(np.max(vals))

            fc = obj_at_delta(c_gs)
            fd = obj_at_delta(d_gs)

            for _ in range(25):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = obj_at_delta(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = obj_at_delta(d_gs)

            delta_opt = (a_gs + b_gs) / 2.0
            new_obj_est = obj_at_delta(delta_opt)

            if new_obj_est < obj_before - 1e-14:
                # Apply update
                a[k] = max(0.0, a[k] + delta_opt)

                # Incremental convolution update (full, to keep conv accurate)
                d_full = np.zeros(2 * n - 1)
                d_full[k:k + n] = 2.0 * a
                # Fix: d_full[m] should be 2*a[m-k] for m in [k, k+n-1]
                # But we already added to a[k], so use the old a for direction
                # Actually, let's recompute direction with old a
                # d_full[m] = 2*a_old[m-k] for m in [k, k+n-1]
                d_full = np.zeros(2 * n - 1)
                d_full[k:k + n] = 2.0 * (a - delta_opt)  # old a
                conv += delta_opt * d_full
                conv[2 * k] += delta_opt * delta_opt

                improved_any = True

        # Renormalize
        s_new = float(np.sum(a))
        if s_new < 1e-10:
            a = best_a.copy()
            conv = compute_conv(a)
        else:
            a /= s_new
            conv /= (s_new * s_new)

        sweep += 1
        obj_after = float(np.max(conv))

        if obj_after < best_obj - 1e-13:
            best_obj = obj_after
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 8:
                # Perturb
                scale = 0.005 * (1 + stagnant * 0.3)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv = compute_conv(a)
                best_obj = float(np.max(conv))
                stagnant = 0

        # Update active set every 5 sweeps
        if sweep % 5 == 0:
            active = np.argsort(conv)[::-1][:K]

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    ACD sweep {sweep}: obj={2*n*best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def full_cd(a_init, time_budget=200.0, verbose=True, seed=123):
    """Standard coordinate descent with full convolution check."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s
    conv = compute_conv(a)

    best_obj = float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5) - 1) / 2.0
    sweep = 0
    stagnant = 0

    while time.monotonic() - start < time_budget:
        obj_before = float(np.max(conv))

        if sweep % 3 == 2:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for k in order:
            d = np.zeros(2 * n - 1)
            d[k:k + n] = 2.0 * a

            lo_delta = -a[k]
            hi_delta = max(1.5 / n, 0.01 / max(np.max(a), 1e-10))

            a_gs = lo_delta
            b_gs = hi_delta
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)

            def obj_at_delta(delta):
                val = conv + delta * d
                val[2 * k] += delta * delta
                return float(np.max(val))

            fc = obj_at_delta(c_gs)
            fd = obj_at_delta(d_gs)

            for _ in range(30):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = obj_at_delta(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = obj_at_delta(d_gs)

            delta_opt = (a_gs + b_gs) / 2.0
            new_obj_est = obj_at_delta(delta_opt)

            if new_obj_est < obj_before - 1e-14:
                a[k] = max(0.0, a[k] + delta_opt)
                conv += delta_opt * d
                conv[2 * k] += delta_opt * delta_opt

        s_new = float(np.sum(a))
        if s_new < 1e-10:
            a = best_a.copy()
            conv = compute_conv(a)
        else:
            a /= s_new
            conv /= (s_new * s_new)

        sweep += 1
        obj_after = float(np.max(conv))

        if obj_after < best_obj - 1e-13:
            best_obj = obj_after
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 10:
                scale = 0.003 * (1 + stagnant * 0.2)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv = compute_conv(a)
                best_obj = float(np.max(conv))
                stagnant = 0

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={2*n*best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def lbfgs_softmax(a_init, time_budget=150.0, verbose=True):
    """L-BFGS on smoothed objective with softmax parameterization."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    # Convert to theta (log-space for softmax)
    a_clipped = np.maximum(a, 1e-15)
    theta0 = np.log(a_clipped / a_clipped.sum())

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    class Closure:
        def __init__(self, n, n_factor):
            self.n = n
            self.n_factor = n_factor
            self.tau = 20.0

        def obj_grad(self, theta):
            a = softmax(theta)
            conv = compute_conv(a)
            max_c = float(np.max(conv))
            exp_vals = np.exp(self.tau * (conv - max_c))
            p = exp_vals / float(np.sum(exp_vals))
            log_se = max_c + np.log(float(np.sum(exp_vals))) / self.tau
            # Objective: 2n * (1/tau) * log(sum exp(tau*conv))
            # = 2n * (max_c + log(sum(exp(tau*(conv-max_c))))/tau)
            obj_val = 2.0 * self.n * (max_c + np.log(float(np.sum(exp_vals))) / self.tau)

            # Gradient w.r.t. a
            grad_a = compute_grad_fft(a, p)
            # Gradient w.r.t. theta: a_j * (grad_a_j - sum_k grad_a_k * a_k)
            dot = float(np.dot(grad_a, a))
            grad_theta = a * (grad_a - dot)

            return obj_val, grad_theta

    # Use multiple tau values
    tau_schedule = [10.0, 30.0, 80.0, 200.0, 500.0]
    theta = theta0.copy()

    for tau in tau_schedule:
        if time.monotonic() - start > time_budget:
            break
        alloc = min(60.0, (time_budget - (time.monotonic() - start)) * 0.5)
        if alloc < 5:
            break

        closure = Closure(n, 2.0 * n)
        closure.tau = tau

        def fun_and_grad(x):
            obj, grad = closure.obj_grad(x)
            return obj, grad

        try:
            res = minimize(fun_and_grad, theta, method='L-BFGS-B',
                           options={'maxiter': 500, 'maxfun': 2000,
                                    'maxls': 50, 'gtol': 1e-8})
            if res.x is not None:
                a_try = softmax(res.x)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    theta = res.x.copy()
        except Exception as e:
            pass

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def lp_polish_linearized(a_init, time_budget=100.0, verbose=True):
    """LP polishing using linearized self-convolution constraints.
    Solves: min M s.t. 2*conv(a,g)[m] - 0*conv(a,a)[m] <= M + conv(a,a)[m]
    which approximates conv(g,g)[m] <= M via first-order Taylor at a.
    Actually: conv(g,g)[m] ≈ 2*conv(a,g)[m] - conv(a,a)[m] (first order)
    So constraint: 2*conv(a,g)[m] - conv(a,a)[m] <= M
    """
    try:
        import cvxpy as cp
    except ImportError:
        return a_init

    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(7777)

    A = build_conv_matrix(a)
    conv_aa = compute_conv(a)

    it = 0
    while time.monotonic() - start < time_budget:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g - conv_aa <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        g_val = None
        for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.CBC, cp.GLPK]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    break
            except Exception:
                continue

        if g_val is None:
            break

        s_g = float(np.sum(g_val))
        if s_g < 1e-10:
            break
        g_val = g_val / s_g

        # Try different blend factors
        best_t_obj = best_obj
        best_t_a = best_a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_a = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = best_a
            # Rebuild LP with updated a
            A = build_conv_matrix(a)
            conv_aa = compute_conv(a)
            if verbose and it % 3 == 0:
                print(f"    LP-Lin it{it}: obj={2*n*best_obj/ (2*n):.8f} -> score={best_obj:.8f}")
        else:
            # Perturb and try again
            scale = 0.01 * (1 + it * 0.1)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            if it > 20:
                break

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Concentrated at edges
        a = np.zeros(n)
        w = max(1, n // 10)
        a[:w] = np.exp(np.arange(w) / w)
        a[-w:] = np.exp(-np.arange(w) / w)
        a = a / a.sum()
    elif mode == 3:
        # Raised cosine
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 4:
        # Semicircle
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x**2, 0))
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 5:
        # Two peaks
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 12)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2)
