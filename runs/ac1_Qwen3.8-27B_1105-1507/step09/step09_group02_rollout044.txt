import numpy as np

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_obj_topk(a, K=5):
    """Compute objective with top-K peaks penalty."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    sorted_conv = np.sort(conv)[::-1]
    top_k = sorted_conv[:min(K, len(conv))]
    return 2.0 * n * float(np.max(top_k)) + 0.01 * float(np.sum(top_k[1:]))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, topK=1):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for top-K positions, sum(g)=1, g>=0.
    For topK=1, this is the standard min-max LP.
    For topK>1, this minimizes the sum of the top-K peaks.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    if topK == 1:
        A = build_conv_matrix(a_norm)
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
    else:
        # Minimize sum of top-K peaks using auxiliary variables
        A = build_conv_matrix(a_norm)
        g = cp.Variable(n, nonneg=True)
        # Use the trick: min sum of top K = min over t of (K*t + sum max(0, x_i - t))
        # Or simpler: use a large M and minimize weighted sum
        # Actually, let's just use the standard LP but with a modified cost
        # min M s.t. A@g <= M, sum(g)=1, g>=0 (same as before)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]

                if s > 1e-10:

def solve_lp_weighted(a, weights):
    """
    Solve LP: min sum_k weights[k] * (conv(a,g))[k] s.t. sum(g)=1, g>=0.
    This is a linear objective, so it's a pure LP.
    """

    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    c = A.T @ weights  # objective coefficients for g
    
    g = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Minimize(cp.sum(cp.multiply(c, g))), [cp.sum(g) == 1])

                s = float(np.sum(g_val))

def moment_matching_init(n, mode=0, rng=None):
    """
    Generate initializations with specific moment properties.
    The key insight: for the autocorrelation to be flat, the sequence should
    have specific statistical properties.
    """
    if rng is None:
        rng = np.random.default_rng()
    
    if mode == 0:
        # Uniform
        return np.ones(n) / n
    elif mode == 1:
        # Raised cosine
        x = np.arange(n) / (n - 1)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        return a / a.sum()
    elif mode == 2:
        # Truncated Gaussian (narrow)
        x = np.linspace(-1, 1, n)
        a = np.exp(-x**2 / (2 * 0.3**2))
        return a / a.sum()
    elif mode == 3:
        # Truncated Gaussian (wide)
        x = np.linspace(-1, 1, n)
        a = np.exp(-x**2 / (2 * 0.7**2))
        return a / a.sum()
    elif mode == 4:
        # Sinc-like (main lobe of sinc)
        x = np.linspace(-2.5, 2.5, n)
        a = np.sinc(x / np.pi)
        a = np.maximum(a, 0)
        if a.sum() < 1e-10:
            return np.ones(n) / n
        return a / a.sum()
    elif mode == 5:
        # Parabolic
        x = np.linspace(-1, 1, n)
        a = 1 - x**2
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 6:
        # Two-lobe
        x = np.linspace(-1, 1, n)
        a = np.exp(-((x-0.4)**2) / 0.1) + np.exp(-((x+0.4)**2) / 0.1)
        return a / a.sum()
    elif mode == 7:
        # Power law edge decay
        x = np.linspace(0.01, 1, n)
        a = x ** 0.5 * (1 - x) ** 0.5  # Beta(1.5, 1.5)-like
        return a / a.sum()
    elif mode == 8:
        # Flat top with smooth edges
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        edge = 0.2
        mask_left = x < edge
        mask_right = x > 1 - edge
        a[mask_left] = 0.5 * (1 + np.cos(np.pi * x[mask_left] / edge))
        a[mask_right] = 0.5 * (1 + np.cos(np.pi * (x[mask_right] - (1-edge)) / edge))
        return a / a.sum()
    elif mode == 9:
        # Random smooth
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.05)
        return a / a.sum()
    elif mode == 10:
        # Exponential decay from center
        x = np.arange(n)
        center = n // 2
        a = np.exp(-np.abs(x - center) / (n * 0.3))
        return a / a.sum()
    elif mode == 11:
        # Cosine window (Hamming-like)
        x = np.arange(n)
        a = 0.54 + 0.46 * np.cos(2 * np.pi * x / (n - 1))
        return a / a.sum()
    elif mode == 12:
        # Kaiser-like
        x = np.linspace(-1, 1, n)
        beta = 5.0
        from scipy.special import i0
        a = i0(beta * np.sqrt(1 - x**2)) / i0(beta)
        return a / a.sum()
    else:
        return np.ones(n) / n


def spectral_refine(a, time_budget, verbose=False):
    """
    Spectral refinement: identify frequency components contributing to time-domain
    peaks and suppress them.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Get FFT of current sequence
    N = 1
    while N < 2 * n:
        N <<= 1
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        peak_idx = np.argmax(conv)
        
        # The peak at position peak_idx in convolution corresponds to
        # a specific frequency in the |FFT|^2 domain
        # IFFT(|A|^2)[peak_idx] is the peak
        # To reduce it, we need to adjust |A|^2 at frequencies that 
        # contribute to IFFT position peak_idx
        
        # IFFT(|A|^2)[k] = (1/N) sum_m |A|^2[m] * exp(2*pi*i*m*k/N)
        # To reduce IFFT[peak_idx], reduce |A|^2[m] where exp(2*pi*i*m*peak_idx/N) > 0
        
        A = np.zeros(N, dtype=complex)
        A[:n] = a
        A_abs_sq = np.abs(A)**2
        
        # Phase factor for the peak
        k = peak_idx
        phase = np.cos(2 * np.pi * k * np.arange(N) / N)
        
        # Identify frequencies that positively contribute to the peak
        positive_freqs = np.where(phase > 0.5)[0]
        positive_freqs = positive_freqs[(positive_freqs < n) | (positive_freqs > N - n)]
        
        # Reduce those frequency components slightly
        A_new = A.copy()
        for m in positive_freqs[:10]:  # Top 10
            if 0 <= m < n or N - n < m < N:
                A_new[m] *= 0.995
        
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0)

def subgradient_descent(a, time_budget, K=5, verbose=False):
    """Subgradient descent with top-K peak weighting and momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 0.5 / np.sqrt(n)
    momentum = 0.5
    v = np.zeros(n)
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        
        if max_c < 1e-15:
            break
        
        # Weighted subgradient from top K peaks
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        weights = np.exp((peak_vals - peak_vals[0]) * 5.0)
        weights /= weights.sum()
        
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]
        
        v = momentum * v + g
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        
        a_new = a - lr * v / v_norm
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                lr *= 0.8
                v *= 0.5
                stall = 0
                if lr < 1e-8:
                    break
    
    return best_a


def coordinate_descent(a, time_budget, verbose=False):
