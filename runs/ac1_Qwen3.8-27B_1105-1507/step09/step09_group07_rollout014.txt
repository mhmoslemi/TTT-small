def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_fast(a, A=None):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0 using HiGHS."""
    n = len(a)
    if A is None:
        A = build_conv_matrix(a)
    
    # Variables: g_0, ..., g_{n-1}, M
    # Minimize M
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # A_ub: A@g - M <= 0  =>  [A, -1] @ [g; M] <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, A=None):
    """
    Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt, dual_vars) or (None, None, None).
    """
    n = len(a)
    if A is None:
        A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status != 0:
            return None, None, None
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s < 1e-10:
            return None, None, None
        g_val = g_val / s
        M_opt = float(res.x[n])
        # Try to get dual variables
        dual = None
        try:
            if hasattr(res, 'marginals') and res.marginals is not None:
                if hasattr(res.marginals, 'ineqlin') and res.marginals.ineqlin is not None:
                    dual = np.abs(res.marginals.ineqlin)
        except Exception:
            pass
        return g_val, M_opt, dual
    except Exception:
        return None, None, None


def find_best_step(a, g_lp, time_limit=0.5):

        obj_try = compute_obj(a_try)

def lp_alternate(a, time_budget, verbose=False, seed=0):
    """
    LP alternating with adaptive step size, dual-guided perturbation,
    and stall detection.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    patience = 30
    iteration = 0
    perturbation_counter = 0

    while time.monotonic() - start < time_budget:
        A = build_conv_matrix(a)
        g_lp, M_lp, dual = solve_lp(a, A)
        if g_lp is None:
            break

        # Check if we've converged (g_lp ≈ a)
        diff_norm = np.linalg.norm(a - g_lp)
        if diff_norm < 1e-8:
            stall += 5
        else:
            # Find best step
            best_t, best_t_obj = find_best_step(a, g_lp, time_limit=0.3)

            if best_t_obj < best_obj - 1e-13:
                a_new = (1.0 - best_t) * a + best_t * g_lp
                a_new = np.maximum(a_new, 0)
                s_new = float(np.sum(a_new))
                if s_new > 1e-10:
                    a_new = a_new / s_new
                best_obj = best_t_obj
                best_a = a_new.copy()
                a = a_new.copy()
                stall = 0
            else:
                stall += 1

        iteration += 1

        # Stall handling: perturb and continue
        if stall > patience:
            perturbation_counter += 1
            scale = 0.03 + 0.02 * min(perturbation_counter, 10)

            if dual is not None and len(dual) == 2 * n - 1:
                # Dual-guided perturbation: focus on coordinates that affect
                # high-dual (important) lags
                importance = np.zeros(n)
                for k in range(2 * n - 1):
                    d_k = dual[k]
                    if d_k < 1e-10:
                        continue
                    for i in range(n):
                        j = k - i
                        if 0 <= j < n:
                            importance[i] += d_k * a[j]
                importance = importance / (np.max(importance) + 1e-15)
                noise = rng.normal(0, scale, n) * (0.5 + importance)
            else:
                noise = rng.normal(0, scale, n)

            a_pert = best_a * np.exp(noise)
            a_pert = np.maximum(a_pert, 0)
            s_pert = float(np.sum(a_pert))
            if s_pert > 1e-10:
                a_pert = a_pert / s_pert
            else:
                a_pert = best_a.copy()

            # Try the perturbed point
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_obj:
                best_obj = obj_pert
                best_a = a_pert.copy()
            a = a_pert.copy()
            stall = 0

        if verbose and iteration % 30 == 0:
            elapsed = time.monotonic() - start
            print(f"          LP it{iteration}: obj={best_obj:.8f}, stall={stall}, "
                  f"pert={perturbation_counter}, elapsed={elapsed:.1f}s")

    return best_a


def lbfgsb_refine(a, time_budget=60):
    """
    L-BFGS-B refinement using a smooth approximation of the max convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for beta in [30, 50, 100, 200]:
        if time.monotonic() - start > time_budget:
            break

        def obj_and_grad(x):
            x = np.maximum(x, 1e-15)
            s_x = float(np.sum(x))
            if s_x < 1e-15:
                return 1e10, np.zeros(len(x))
            a_x = x / s_x
            conv = compute_conv(a_x)
            # Softmax smooth max
            shifted = beta * (conv - conv.max())
            exp_vals = np.exp(shifted)
            w = exp_vals / exp_vals.sum()
            smooth_max = conv.max() + np.log(exp_vals.sum()) / beta

            # Gradient: d(smooth_max)/dx_i
            # smooth_max = f(a) where a = x/s, f(a) = (1/beta)*log(sum exp(beta*conv_k))
            # d f / d a_i = sum_k w_k * 2 * a[k-i] (if 0<=k-i<n)
            # d a / d x_i = (1/s - x_i/s^2) = (s - x_i)/s^2 (for the i-th component in a)
            # Actually a_j = x_j / s, so d a_j / d x_i = delta_{ji}/s - x_j/s^2
            # d f / d x_i = sum_j (d f / d a_j) * (d a_j / d x_i)
            # = sum_j g_j * (delta_{ji}/s - x_j/s^2)
            # = g_i/s - sum_j g_j * x_j / s^2
            # where g_j = sum_k w_k * 2 * a[k-j] (if 0<=k-j<n)

            grad_a = np.zeros(n)
            for i in range(n):
                total = 0.0
                # grad_a[i] = sum_k w[k] * 2 * a[k-i] (if 0 <= k-i < n)
                k_min = i
                k_max = min(i + n - 1, 2 * n - 2)
                if k_min <= k_max:
                    total = 2.0 * np.sum(w[k_min:k_max + 1] * a[k_max - k_min - 1 : -1 if k_max == k_min else (k_max - i) : -1])
                # Let me redo this more carefully
                # grad_a[i] = sum_{k: 0<=k-i<n} w[k] * 2 * a[k-i]
                # Let j = k-i, so k = j+i, j ranges over [0, n-1] and k ranges over [i, i+n-1]
                # but k also must be in [0, 2n-2]
                j_vals = np.arange(n)
                k_vals = j_vals + i
                valid = (k_vals <= 2 * n - 2)
                if np.any(valid):
                    total = 2.0 * np.sum(w[k_vals[valid]] * a[j_vals[valid]])
                grad_a[i] = total

            # Chain rule for normalization
            s_x = float(np.sum(x))
            grad_x = grad_a / s_x - np.sum(grad_a * x) / (s_x * s_x)

            return smooth_max, grad_x

        # Run L-BFGS-B
        try:
            result = minimize(
                obj_and_grad,
                a.copy(),
                jac=True,
                method='L-BFGS-B',
                bounds=[(0, None)] * n,
                options={'maxiter': 100, 'maxfun': 500, 'ftol': 1e-15, 'gtol': 1e-10}
            )
            if result.x is not None:
                a_new = np.maximum(result.x, 0)
                s_new = float(np.sum(a_new))
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new.copy()
                        if verbose:
                            print(f"          L-BFGS-B beta={beta}: obj={best_obj:.8f}")
        except Exception:
            pass

    return best_a


def spectral_init(n, rng, mode=0):
    """Generate diverse spectral-based initializations."""
    if mode % 6 == 0:
        # Uniform
        a = np.ones(n) / n
    elif mode % 6 == 1:
        # Cosine weighting
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 6 == 2:
        # Multiple frequency components
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f in rng.integers(2, 10, size=3):
            p = rng.uniform(0, 2 * np.pi)
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 6 == 3:
        # Bimodal
        x = np.linspace(0, 1, n)
        a = np.exp(-10 * (x - 0.3) ** 2) + np.exp(-10 * (x - 0.7) ** 2) + 0.01
        a = a / a.sum()
    elif mode % 6 == 4:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(0.5, 2.0)
        a = a / a.sum()
    else:
        # Random with Dirichlet
        a = rng.dirichlet(np.ones(n) * 0.3)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating + n-sweep + L-BFGS-B + diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"\nInitial: n={n_current}, obj={best_score:.8f}, target=1.5027")

    # === Phase 1: Quick LP polish on parent (8%) ===
    print(f"\n=== Phase 1: Quick LP Polish ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 100.0)
    if p1_budget > 20:
        a_p1 = lp_alternate(best_seq, time_budget=p1_budget, verbose=True, seed=seed)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_p1:.8f}")

    # === Phase 2: n-sweep (22%) ===
    print(f"\n=== Phase 2: n-Sweep ===")
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.22

    n_options = sorted(set([
        500, 550, 600, 640, 680, 700, 720, 750, 768, 780,
        800, 820, 832, 850, 860, 880, 900, 920, 960, 1000,
        1024, 1050, 1100, 1200, 1280,
        n_current - 50, n_current + 50, n_current + 100,
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1400]

    per_n_budget = p2_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_n_budget, remaining * 0.12)
        if alloc < 10:
            alloc = 10

        a_try = resample_sequence(best_seq, n_try)
        a_opt = lp_alternate(a_try, time_budget=alloc, verbose=False, seed=seed + n_try)
        obj_opt = compute_obj(a_opt)
        marker = " **" if obj_opt < best_score else ""
        print(f"  n={n_try}: obj={obj_opt:.8f}{marker}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try

    # === Phase 3: Intensive LP on best n with perturbation (35%) ===
    print(f"\n=== Phase 3: Intensive LP (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.35, 500.0)

    # Run multiple rounds with increasing perturbation
    n_rounds = 5
    per_round = p3_budget / n_rounds
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        alloc = min(per_round, remaining * 0.2)
        if alloc < 20:
            break
        if rnd == 0:
            a_start = best_seq.copy()
        else:
            rng_r = np.random.default_rng(seed + rnd * 104729)
            scale = 0.02 + 0.015 * rnd
            a_start = best_seq * np.exp(rng_r.normal(0, scale, n_current))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()

        a_lp = lp_alternate(a_start, time_budget=alloc, verbose=True, seed=seed + rnd * 7919)
        obj_lp = compute_obj(a_lp)
        marker = " **" if obj_lp < best_score else ""
        print(f"  LP round {rnd}: obj={obj_lp:.8f}{marker}")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()

    # === Phase 4: L-BFGS-B refinement (12%) ===
    print(f"\n=== Phase 4: L-BFGS-B Refinement ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.12, 120.0)
    if p4_budget > 15:
        # Try from current best
        a_lb = lbfgsb_refine(best_seq, time_budget=p4_budget)
        obj_lb = compute_obj(a_lb)
        marker = " **" if obj_lb < best_score else ""
        print(f"  L-BFGS-B (best): obj={obj_lb:.8f}{marker}")
        if obj_lb < best_score:
            best_score = obj_lb
            best_seq = a_lb.copy()

        # Try from slightly perturbed versions
        remaining = deadline - time.monotonic()
        if remaining > 30:
            for pert in [0.01, 0.02]:
                remaining = deadline - time.monotonic()
                if remaining < 15:
                    break
                rng_p = np.random.default_rng(seed + 999 + int(pert * 1000))
                a_pert = best_seq * np.exp(rng_p.normal(0, pert, len(best_seq)))
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                a_lb2 = lbfgsb_refine(a_pert, time_budget=min(remaining * 0.4, 40.0))
                obj_lb2 = compute_obj(a_lb2)
                marker = " **" if obj_lb2 < best_score else ""
                print(f"  L-BFGS-B (pert={pert}): obj={obj_lb2:.8f}{marker}")
                if obj_lb2 < best_score:
                    best_score = obj_lb2
                    best_seq = a_lb2.copy()

    # === Phase 5: Diverse spectral restarts (10%) ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p5_budget = remaining * 0.10
    n_use = len(best_seq)
    n_restarts = 5
    per_restart = p5_budget / max(1, n_restarts)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_restart, remaining * 0.2)
        if alloc < 15:
            break

        mode = (r * 2 + 1) % 6
        a_r = spectral_init(n_use, rng, mode=mode)
        # Blend with best
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        # Quick LP
        a_r = lp_alternate(a_r, time_budget=alloc, verbose=False, seed=seed + r * 331)
        obj_r = compute_obj(a_r)
        marker = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}{marker}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()

    # === Phase 6: Final polish (remaining time) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Another round of LP on the best
    alloc = min(remaining * 0.5, 80.0)
    if alloc > 15:
        a_final = lp_alternate(best_seq, time_budget=alloc, verbose=True, seed=seed + 42)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # One more L-BFGS-B pass
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_lb_final = lbfgsb_refine(best_seq, time_budget=min(remaining * 0.4, 40.0))
        obj_lb_final = compute_obj(a_lb_final)
        if obj_lb_final < best_score:
            best_score = obj_lb_final
            best_seq = a_lb_final.copy()
            print(f"  Final L-BFGS-B: obj={best_score:.8f} **")

    # Last LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = lp_alternate(best_seq, time_budget=min(remaining * 0.6, 30.0), verbose=False, seed=seed + 7)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"=== Target: 1.5027, Achieved: {best_score:.8f}, Gap: {best_score - 1.5027:.8f} ===")

    return result
