```python
import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A
```

```python
def freq_domain_correct(a, time_budget, verbose=False):
    """
    Iteratively reduce the autocorrelation peak by adjusting the power spectrum.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    
    it = 0
    lr = 0.01
    
    while time.monotonic() - start < time_budget:
        A = fft(a, L)
        P = np.abs(A) ** 2
        r = irfft(P, L)  # This gives the circular autocorrelation
        # Actually, irfft of P gives the full autocorrelation
        # r[k] = sum_j a[j] * a[j+k] (circular)
        # For L >= 2n-1, this equals the linear autocorrelation for k = 0..n-1
        
        # Find the peak
        max_r = float(np.max(r[:2*n-1]))
        peak_idx = int(np.argmax(r[:2*n-1]))
        
        # Gradient of max_r with respect to P[m]:
        # r[k] = (1/L) * sum_m P[m] * exp(2πimk/L)
        # dr[k]/dP[m] = (1/L) * exp(2πimk/L)
        # So gradient of max_r w.r.t. P is exp(2πi*peak_idx*m/L) / L
        
        m_vals = np.arange(L)
        grad = np.real(np.exp(2j * np.pi * peak_idx * m_vals / L)) / L
        
        # Update P: reduce where grad is positive (contributes to peak)
        P_new = P - lr * np.abs(grad)  # Simplified: reduce all non-DC
        # Actually, we should only reduce where grad > 0
        P_new = P - lr * np.maximum(grad, 0)
        P_new = np.maximum(P_new, 0)
        
        # Keep DC
        P_new[0] = P[0]
        
        # Update A: scale magnitudes, keep phases
        P_old = np.maximum(P, 1e-30)
        scale = np.sqrt(P_new / P_old)
        A_new = A * scale
        
        # Back to time domain
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.7
        
        it += 1
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    Freq it{it}: obj={best_obj:.8f}, lr={lr:.4f}, time={elapsed:.0f}s")
        
        if lr < 1e-6:
            break
    
    return best_a
```

**Idea: Coordinate descent on the peaks.**
Instead of optimizing all of `a` simultaneously, focus on the peaks of the autocorrelation. For each peak, try to reduce it by adjusting the relevant elements of `a`.

For peak at lag `k`, the autocorrelation value is `(a*a)[k] = sum_i a[i] * a[k-i]`. To reduce this, we can decrease `a[i]` for the `i` values that contribute most.

This is a form of **coordinate descent** on the peaks.

Algorithm:
1. Compute `r = a * a`.
2. Find the top-K peaks.
3. For each peak (in order of magnitude), compute the contribution of each `a[i]` to that peak.
4. Decrease the `a[i]` with the largest contribution.
5. Renormalize.
6. Repeat.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using parametric LP blending with
line search, coordinate peak shaving, and smoothed gradient descent.
Key innovation over previous: parametric LP (solving different LPs for different
blending parameters t for more accurate linearizations) with line search along
the LP direction, and coordinate-wise peak shaving as a complementary method.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, t, conv_aa, A_mat):
    """
    Solve parametric LP:
    min M s.t. 2t*(A@g)[k] - M <= (2t-1)*conv_aa[k]/t ... 
    Actually: (1-2t)*conv_aa[k] + 2t*(A@g)[k] <= M
    => A@g[k] - M/(2t) <= (2t-1)*conv_aa[k]/(2t)
    """
    n = len(a)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    tt = 2.0 * t
    # A@g - M/tt <= ((tt-1)/tt) * conv_aa
    constraints = [
        A_mat @ g - (1.0 / tt) * M <= ((tt - 1.0) / tt) * conv_aa,
        cp.sum(g) == 1
    ]
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                sv = float(np.sum(gv))
                if sv > 1e-10:
                    return gv / sv
        except Exception:
            continue
    return None
```
