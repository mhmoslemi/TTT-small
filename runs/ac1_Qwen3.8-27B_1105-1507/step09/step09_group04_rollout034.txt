"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternation (block coordinate descent), FFT-based smoothed gradient descent,
and multi-length exploration with diverse restarts.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # conv(a,g)[k] = sum_j a[j] * g[k-j]
        # So for fixed j, a[j] contributes to positions k where 0 <= k-j < n
        # i.e., j <= k <= j+n-1
        # And the coefficient of g[k-j] is a[j]
        # So A[k, j] = a[j] for j <= k <= j+n-1
        lo = j
        hi = min(j + n, m)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_lp(a, time_limit=30.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None) if infeasible.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    m = 2 * n - 1
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    # conv(a, g)[k] <= M for all k
    A = build_conv_matrix(a_norm)
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS]:
        try:
            prob.solve(solver=solver, time_limit=time_limit)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def smoothed_gradient(a, tau):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*conv_k)) w.r.t. a.
    Uses FFT for efficiency.
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    
    # Stability: shift to avoid overflow
    exp_vals = np.exp(tau * (conv - max_c))
    total = float(np.sum(exp_vals))
    h = exp_vals / total  # probability distribution over lags
    
    # Gradient: grad[j] = tau * sum_k h[k] * (a[k-j] + a[j-k])
    # where a[i] = 0 for i outside [0, n-1]
    
    # sum_k h[k] * a[k-j] = correlation of h and a
    # This is (h convolved with a reversed) or (a correlated with h)
    
    # Let's use: sum_k h[k] * a[k-j]
    # For j >= 0: this is sum_{k=j}^{j+n-1} h[k] * a[k-j] (if k-j in [0,n-1] and k in [0,2n-2])
    # = (h * a)[j] where * denotes cross-correlation... 
    
    # Actually let me think of it as:
    # f[j] = sum_k h[k] * a[k-j]
    # This is the cross-correlation (h ⊛ a)[j] = sum_k h[k] a[k-j]
    # = FFT-based: IFFT(FFT(h) * conj(FFT(a)))
    
    # And g[j] = sum_k h[k] * a[j-k]
    # = (h * a)[j] = convolution
    # = IFFT(FFT(h) * FFT(a))
    
    # Total N size for FFT
    N = 1
    while N < 3 * n:
        N <<= 1
    
    H = fft(h, N)
    A = fft(a, N)
    
    # f[j] = sum_k h[k]*a[k-j] = correlation
    f = ifft(H * np.conj(A)).real[:n]
    
    # g[j] = sum_k h[k]*a[j-k] = convolution
    g = ifft(H * A).real[:n]
    
    # Gradient = tau * (f + g)
    grad = tau * (f + g)
    
    return grad, conv, max_c


def smoothed_descent(a, time_budget=200.0, tau_init=50.0, tau_end=500.0, verbose=False):
    """
    Mirror descent (multiplicative weights) on the smoothed max objective.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    it = 0
    best_it = 0
    lr = 2.0 / tau_init  # initial learning rate
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(1, time_budget))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)
        
        if it % 5000 == 0 and verbose:
            print(f"  SD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.0f}s")
        
        grad, conv, max_c = smoothed_gradient(a, tau)
        
        # Mirror descent step: a_new[j] = a[j] * exp(-lr * grad[j])
        # But gradient is with respect to the unnormalized a.
        # Since we're on the simplex, subtract the mean gradient (projection).
        grad_centered = grad - np.mean(grad)
        
        # Adaptive learning rate
        lr_step = 1.0 / tau  # this is the natural scale
        
        # Multiplicative update
        update = np.exp(-lr_step * grad_centered)
        a_new = a * update
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        
        # Reset if stuck
        if it - best_it > 20000:
            a = best_a.copy()
            best_it = it

    return best_a, best_obj


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.arange(n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        # Beta-like distribution
        a = (x + 0.5) ** 0.5 * (n - x - 0.5) ** 0.5
        a = np.maximum(a, 0.01)
    elif mode == 3:
        # Arcsine distribution
        xn = x / (n - 1)
        a = 1.0 / np.sqrt(xn * (1 - xn) + 1e-10)
    elif mode == 4:
        # Exponential decay
        a = np.exp(-x / (n * rng.uniform(0.5, 2.0)))
    elif mode == 5:
        # Two peaks
        c1 = n // 4
        c2 = 3 * n // 4
        w = n // 10
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
    elif mode == 6:
        # Cosine window
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
        a = np.maximum(a, 0.001)
    elif mode == 7:
        # Random smooth
        freqs = rng.uniform(0.5, 3.0, 5)
        phases = rng.uniform(0, 2 * np.pi, 5)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x / n + p)
        a = np.maximum(a, 0.01)
    elif mode == 8:
        # Geometric
        r = rng.uniform(0.85, 0.98)
        a = r ** x
    elif mode == 9:
        # Sine squared
        a = np.sin(np.pi * x / n) ** 2
        a = np.maximum(a, 0.001)
    else:
        a = np.ones(n)
    
    a = a.astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

        for k in order:

            # conv_new = conv + 2*delta*(a shifted) + delta^2 at position 2k
            # Specifically: d[m] = a[m-k] for m in [k, k+n-1] (within [0, 2n-2])

            # But we also need to maintain sum(a) = 1, so changing a[k] by delta
            # means we need to adjust. Actually, since the objective is scale-invariant,
            # let's just work with the unnormalized version.
            # 
            # Actually it's easier to think of it as: we're on the simplex, so if we
            # increase a[k] by delta, we must decrease something else. But for a
            # 1D coordinate search, we can parameterize as:
            # a_new[k] = a[k] + delta, and renormalize by dividing by (1+delta).
            # This is equivalent to: a_new[k] = a[k] + delta, a_new[j] = a[j]/(1+delta) for j≠k

            # Simplest approach: just try the direct search.
            # f(delta) = max(conv + 2*delta*d + delta^2*e_{2k}) / (1+delta)^2

            if time.monotonic() - start > time_budget:

            lo_d = -a[k] + 1e-12
            hi_d = max(0.5 / n, 2 * a[k])
            
            # Quick check: try delta = 0
            f_mid = float(np.max(conv))

            test_deltas = [lo_d, hi_d, (lo_d + hi_d) / 2, -a[k] * 0.5, a[k] * 0.5]
            best_fd = f_mid
            best_d = 0.0

                if td <= lo_d or td >= hi_d:
                    continue
                val = conv + 2.0 * td * d
                val[2 * k] += td * td
                obj_val = float(np.max(val)) / (1.0 + td) ** 2
                if obj_val < best_fd:
                    best_fd = obj_val
                    best_d = td
            
            if best_d == 0.0:
                continue
            
            # Golden section around best_d
            span = max(hi_d - lo_d, 1e-6)
            a_gs = best_d - span * 0.3
            b_gs = best_d + span * 0.3
            a_gs = max(a_gs, lo_d)
            b_gs = min(b_gs, hi_d)
            
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)
            
            def obj_delta(delta):
                val = conv + 2.0 * delta * d
                val[2 * k] += delta * delta
                return float(np.max(val)) / (1.0 + delta) ** 2
            
            fc = obj_delta(c_gs)
            fd = obj_delta(d_gs)
            
            for _ in range(30):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = obj_delta(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = obj_delta(d_gs)
            
            delta_opt = (a_gs + b_gs) / 2.0
            new_obj = obj_delta(delta_opt)
            
            if new_obj < best_obj - 1e-14:
                a[k] += delta_opt
                a[k] = max(a[k], 0.0)
                s_new = a.sum()
                a = a / s_new
                conv = compute_conv(a)
                best_obj = float(np.max(conv))
                improved = True
        
        iters += 1
        if verbose and iters % 3 == 0:
            elapsed = time.monotonic() - start
            print(f"  CD round {iters}: obj={2*n*best_obj:.8f}, t={elapsed:.0f}s")
        
        if not improved and iters > 1:
            break

    return a
