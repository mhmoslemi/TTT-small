    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Phase 0: Load parent
    print("=== Phase 0: Load Parent ===")
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"  Parent (n={n_target}): obj={obj:.8f}")
    except Exception as e:
        print(f"  Warning: could not load parent: {e}")

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"  Default init (n={n_target}): obj={best_score:.8f}")

    # Phase 1: Multi-n exploration
    print("\n=== Phase 1: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.1, 120.0)

    n_options = sorted(set([
        max(400, n_target - 200), max(400, n_target - 100),
        n_target, n_target + 100, n_target + 200, n_target + 400,
        600, 800, 1000, 1200, 1400, 1600
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1600]
    n_options.sort(key=lambda x: abs(x - n_target))

    # Always include current best n first
    if n_target not in n_options:
        n_options.insert(0, n_target)

    n_per_trial = max(5, int(p1_budget / len(n_options[:6])))
    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 10 or n_per_trial < 5:
            break

        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(n_per_trial, remaining * 0.4)
        a_opt, obj_opt = alternating_optimize(a_try, time_budget=alloc,
                                              verbose=False, rng=rng)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # Phase 2: Intensive alternating optimization on best n
    print(f"\n=== Phase 2: Intensive Optimization (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.45, 700.0)

    # Run in sub-rounds with periodic perturbation
    n_sub_rounds = 5
    sub_budget = p2_budget / n_sub_rounds

    for rnd in range(n_sub_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(sub_budget, remaining * 0.6)
        if alloc < 10:
            break

        a_start = best_seq.copy()
        if rnd > 0:
            # Light perturbation between sub-rounds
            method = rnd % 8
            scale = 0.015 + 0.005 * rnd
            a_start = perturb_sequence(a_start, rng, method=method, scale=scale)

        a_opt, obj_opt = alternating_optimize(a_start, time_budget=alloc,
                                              verbose=(rnd == 0), rng=rng)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Sub-round {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  Sub-round {rnd}: obj={obj_opt:.8f}")

    # Phase 3: Diverse restarts
    print(f"\n=== Phase 3: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.25, 350.0)
    n_best = len(best_seq)

    n_restarts = max(3, int(p3_budget / 70))
    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        mode = r % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend_frac = 0.2 + 0.05 * r
        a_r = (1 - blend_frac) * a_r + blend_frac * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r < 1e-10:
            a_r = np.ones(n_best) / n_best
        else:
            a_r = a_r / s_r

        alloc = min(70.0, remaining * 0.3)
        a_opt, obj_opt = alternating_optimize(a_r, time_budget=alloc,
                                              verbose=False, rng=rng)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_opt:.8f}")

    # Phase 4: Final polish
    print(f"\n=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Run a few more alternating cycles on the best
    if remaining > 30:
        alloc = min(remaining * 0.5, 120.0)
        a_opt, obj_opt = alternating_optimize(best_seq, time_budget=alloc,
                                              verbose=True, rng=rng)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final polish: obj={best_score:.8f} **")
        else:
            print(f"  Final polish: obj={obj_opt:.8f}")

    # Try slightly different n values near the best
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_cur = len(best_seq)
        for dn in [-50, 50, 100, -100]:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            n_new = n_cur + dn
            if n_new < 300 or n_new > 2000:
                continue
            a_try = resample_sequence(best_seq, n_new)
            alloc = min(30.0, remaining * 0.3)
            a_opt, obj_opt = alternating_optimize(a_try, time_budget=alloc,
                                                  verbose=False, rng=rng)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Final n={n_new}: obj={best_score:.8f} **")

    # Final validation and output
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
