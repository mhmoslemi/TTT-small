"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: LP-based optimization with golden-section line search, spectral escape
(Gerchberg-Saxton style) for local minima, lag-targeted subgradient polish,
and systematic multi-length exploration. Key innovations:
1. Full-replacement first (t=1) with fallback line search
2. Spectral/domain alternating projections when stuck
3. Weighted LP that emphasizes current peak lags

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, lag_weights=None):
    """
    Solve LP: min M s.t. A @ g <= M (weighted by lag_weights if provided),
    sum(g)=1, g>=0. Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    m = 2 * n - 1

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()

    if lag_weights is not None:
        # Weighted: (A@g)[j] / w_j <= M  =>  (A@g)[j] <= w_j * M
        # Use diag(1/w) @ A @ g <= M
        inv_w = 1.0 / np.maximum(lag_weights, 1e-6)
        A_w = A * inv_w[:, None]
        constraints = [A_w @ g <= M, cp.sum(g) == 1]
    else:
        constraints = [A @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def spectral_escape(a, num_iters=50, rng=None):
    """
    Gerchberg-Saxton style alternating projections:
    - Frequency domain: flatten the magnitude spectrum
    - Time domain: enforce non-negativity and normalize
    This can escape local minima by changing the phase structure.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.copy(), 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    N = 1
    while N < 2 * n:
        N <<= 1

    # Get current spectrum
    A_spec = fft(a, N)
    mag = np.abs(A_spec)
    phase = np.angle(A_spec)

    # Flattening factor: blend magnitude toward its root-mean-square
    mag_rms = np.sqrt(np.mean(mag ** 2))
    alpha = 0.3  # flattening strength

    for it in range(num_iters):
        # Frequency domain: blend magnitude toward flat
        mag = (1 - alpha) * mag + alpha * mag_rms
        # Randomize phase slightly (more randomization for early iterations)
        noise_scale = 0.5 * max(0, 1 - it / num_iters)
        phase = phase + noise_scale * rng.normal(0, 1, N)
        # Ensure Hermitian symmetry for real signal
        A_spec = mag * np.exp(1j * phase)
        A_spec = (A_spec + np.conj(A_spec[N - np.arange(N)][::-1][:N])) / 2  # approximate symmetry

        # Time domain: inverse FFT, enforce non-negativity
        a_new = ifft(A_spec).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a = a_new / s_new

    return a


def lag_targeted_subgradient(a, time_budget, K=15, lr_init=0.5, rng=None):
    """
    Subgradient method focusing on top-K convolution peaks with
    adaptive step size and momentum.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init / np.sqrt(n)
    momentum = 0.8
    v = np.zeros(n)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        max_pv = conv[top_idx[0]]
        if max_pv < 1e-15:
            break
        # Softmax weights over top peaks
        weights = np.exp((conv[top_idx] - max_pv) * 5.0 / (max_pv + 1e-15))
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr *= 1.01
        else:
            stall += 1
            if stall % 50 == 0:
                lr *= 0.7
                v *= 0.5
            if stall > 500:
                # Perturb to escape
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                v = np.zeros(n)
                lr = lr_init / np.sqrt(n)
                stall = 0

    return best_a


def block_coordinate_descent(a, time_budget, num_blocks=8, rng=None):
    """
    Block coordinate descent: divide sequence into blocks, optimize
    each block via LP while fixing others. Helps escape global local minima.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    block_size = n // num_blocks
    blocks = [(i * block_size, min((i + 1) * block_size, n)) for i in range(num_blocks)]
    blocks[-1] = (blocks[-1][0], n)

    iteration = 0
    while time.monotonic() - start < time_budget:
        progress = (time.monotonic() - start) / time_budget
        # Cycle through blocks
        for bi in range(num_blocks):
            if time.monotonic() - start >= time_budget:
                break
            lo, hi = blocks[bi]
            block_len = hi - lo
            if block_len < 2:
                continue

            # Solve LP for the block: given a with block fixed at current,
            # find best block values g_block minimizing max conv
            a_fixed = a.copy()
            a_norm = a_fixed / np.sum(a_fixed)

            # Build restricted LP: only optimize the block
            A = build_conv_matrix(a_norm)
            m = 2 * n - 1
            g_block = cp.Variable(block_len, nonneg=True)
            M = cp.Variable()

            # conv(a_new, a_new) where a_new differs from a in the block
            # This is complex; approximate by fixing the non-block part
            # and optimizing the block contribution
            # Simplified: use the full LP but constrain non-block to a
            g_full = cp.Variable(n, nonneg=True)
            constraints = [cp.sum(g_full) == 1]
            for j in range(lo):
                constraints += [g_full[j] == a[j] / np.sum(a)]
            for j in range(hi, n):
                constraints += [g_full[j] == a[j] / np.sum(a)]
            constraints += [A @ g_full <= M]

            prob = cp.Problem(cp.Minimize(M), constraints)
            solved = False
            for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
                try:
                    prob.solve(solver=solver)
                    if g_full.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                        g_val = np.maximum(g_full.value, 0)
                        s_val = float(np.sum(g_val))
                        if s_val > 1e-10:
                            g_val = g_val / s_val
                            # Blend
                            a_trial = a.copy()
                            t = 0.5
                            a_trial[lo:hi] = (1 - t) * a[lo:hi] + t * g_val[lo:hi]
                            a_trial = np.maximum(a_trial, 0)
                            s_trial = float(np.sum(a_trial))
                            if s_trial < 1e-10:
                                continue
                            a_trial = a_trial / s_trial
                            obj_trial = compute_obj(a_trial)
                            if obj_trial < best_obj - 1e-13:
                                best_obj = obj_trial
                                best_a = a_trial.copy()
                                a = a_trial
                            solved = True
                            break
                except Exception:
                    continue
            if not solved:
                continue

        iteration += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via spline interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    from scipy.interpolate import interp1d
    f = interp1d(x_old, a, kind='cubic', fill_value='extrap', bounds_error=False)
    a_new = f(x_new)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two-gaussian
        centers = [n // 3, 2 * n // 3]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine
        x = np.linspace(-0.999, 0.999, n)
        a = 1.0 / np.sqrt(1 - x ** 2 + 1e-8)
        a = a / a.sum()
    elif mode == 4:
        # Random frequency mixture
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Periodic with random offset
        period = rng.integers(n // 5, n // 2)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * np.arange(n) / period)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def lp_loop(a, time_budget, use_weighted=True, verbose=False, rng=None):

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = elapsed / time_budget

        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Solve LP
        if use_weighted and iter_count > 20:

            weights = np.ones(2 * n - 1)
            top_frac = int(0.1 * (2 * n - 1))
            top_idx = np.argsort(conv)[::-1][:top_frac]
            weights[top_idx] = 2.0

        else:

        if g_lp is None:
            break

        # Line search: try full replacement first, then fractions
        best_t = 0.0
        best_t_obj = best_obj

        for t in candidates:

        else:
            print(f"  BCD: obj={obj_bcd:.8f}")

    # === Phase 5: Multi-n exploration (20%) ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Generate candidate n values
    n_options = set()
    for delta in [-200, -100, -50, 0, 50, 100, 200, 300, 400, 500]:
        n_try = n_current + delta
        if 400 <= n_try <= 2000:
            n_options.add(n_try)
    # Also add some absolute values
    for n_abs in [500, 600, 700, 800, 900, 1000, 1200, 1500]:
        if abs(n_abs - n_current) > 30:
            n_options.add(n_abs)

    n_options = sorted(n_options)
    # Limit to 5 most promising (closest to current + a few far)
    n_options_filtered = []
    for n_try in n_options:
        if n_try != n_current and n_try not in n_options_filtered:
            n_options_filtered.append(n_try)
        if len(n_options_filtered) >= 6:
            break

    for n_try in n_options_filtered:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.18, 80.0)
        if alloc > 15:
            a_try = lp_loop(a_try, time_budget=alloc, rng=rng)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final intensive polish (remaining) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP round 1
    if remaining > 30:
        alloc = min(remaining * 0.35, 120.0)
        a_lp = lp_loop(best_seq, time_budget=alloc, verbose=True, rng=rng)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP1: obj={best_score:.8f} **")
        else:
            print(f"  Final LP1: obj={obj_lp:.8f}")

    # Subgradient
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.2, 50.0)
        a_sg = lag_targeted_subgradient(best_seq, time_budget=alloc, K=8, rng=rng)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")
        else:
            print(f"  Final SG: obj={obj_sg:.8f}")

    # LP round 2
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 80.0)
        a_lp = lp_loop(best_seq, time_budget=alloc, rng=rng)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")
        else:
            print(f"  Final LP2: obj={obj_lp:.8f}")

    # One more spectral escape attempt
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_spec = spectral_escape(best_seq, num_iters=50, rng=rng)
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Final Spectral: obj={best_score:.8f} **")
        # Quick LP after spectral
        if remaining > 30:
            alloc = min(remaining * 0.4, 40.0)
            if alloc > 10:
                a_lp = lp_loop(best_seq, time_budget=alloc, rng=rng)
                obj_lp = compute_obj(a_lp)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    print(f"  Final LP3: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
