def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a / s)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g) for g of same length as a."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        # C[k, j] = a[k - j] if 0 <= k-j < n, else 0

        # i.e., rows j to j+n-1, column j gets a[0] to a[n-1]
        C[j:j+n, j] = a
    return C

                t_par = -B / (2 * A) if abs(A) > 1e-15 else t2
                t_par = max(0.01, min(1.0, t_par))
                a_par = (1.0 - t_par) * a + t_par * g_lp
                a_par = np.maximum(a_par, 0)
                s_par = float(np.sum(a_par))
                if s_par > 1e-10:
                    a_par = a_par / s_par
                    obj_par = compute_obj(a_par)
                    if obj_par < best_t_obj:
                        best_t = t_par
                        best_t_obj = obj_par
                        best_idx = -1  # use parabolic result

        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            break
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()
            # Update momentum
            momentum_t = 0.7 * momentum_t + 0.3 * best_t
            stall = 0
        else:
            stall += 1
            if stall > 20:
                # Restart with perturbation
                rng = np.random.default_rng(iteration * 7919)
                scale = min(0.3, 0.05 + 0.02 * (stall // 20))
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                momentum_t = 0.5

        if verbose and iteration % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{iteration}: obj={best_obj:.8f}, t={momentum_t:.3f}, "
                  f"stall={stall}, elapsed={elapsed:.1f}s")

    return best_a


def smooth_grad_descent(a, time_budget, verbose=False):
    """
    Smooth optimization using log-sum-exp surrogate on the peak.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.5
    beta = 50.0
    beta_growth = 1.05
    convergence_count = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = conv.max()
        
        # Log-sum-exp weights
        shifted = beta * (conv - max_c)
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        # Gradient of sum(weights[k] * conv[k]) w.r.t. a[j]
        # conv[k] = sum_i a[i] * a[k-i]
        # d(conv[k])/d(a[j]) = a[k-j] + a[j-k] (for valid indices)
        # So grad[j] = sum_k weights[k] * (a[k-j] + a[j-k])
        # This is essentially the cross-correlation of weights with a, doubled
        
        # Use FFT for efficiency
        size = 1
        while size < 2 * n:
            size <<= 1
        W = rfft(weights, size)
        A = rfft(a, size)
        # Cross-correlation: ifft(conj(W) * A) gives sum_k weights[k] * a[k-j]
        # Actually we need: sum_k weights[k] * a[k-j] for k-j valid
        # This is the cross-correlation of weights and a
        cross = irfft(W * A, size)[:n]  # This gives sum_k weights[k] * a[?]
        
        # Actually, let me compute the gradient more directly
        # grad[j] = sum_k weights[k] * (a[k-j] + a[j-k])
        # First term: sum_k weights[k] * a[k-j] = (weights * a)[j] in cross-correlation sense
        # = sum_k weights[k] * a[k-j]
        # In FFT: this is the cross-correlation of weights and a
        # cross_corr[j] = sum_k weights[k+j] * a[k] = ... 
        
        # Let me just do it with the convolution matrix approach for clarity
        # grad[j] = 2 * sum_k weights[k] * a[|k-j| adjusted]
        
        # Simpler: grad = 2 * C^T @ (weights * a_adjusted)
        # where C is the convolution matrix
        # Actually: conv[k] = sum_j C[k,j] * a[j] where C[k,j] = a[k-j] if valid
        # d(conv[k])/d(a[j]) = C[k,j] + C[k,j] = 2*a[k-j] (by symmetry of convolution)
        
        # So grad[j] = sum_k weights[k] * 2 * a[k-j] (for valid k-j)
        # = 2 * cross_correlation(weights, a)[j]

        # cross_correlation[i] = sum_k weights[k] * a[k-i]

        # Actually, let's use FFT properly
        # sum_k weights[k] * a[k-j]: for each j, sum over k where 0<=k<2n-1 and 0<=k-j<n
        # = sum_i a[i] * weights[i+j] (substituting i = k-j)
        # This is the cross-correlation of a and weights at lag j
        # = ifft(conj(FFT(a)) * FFT(weights)): but need to be careful about lengths

        # grad[j] = 2 * sum_{i: 0<=i<n, 0<=i+j<2n-1} a[i] * weights[i+j]

        # C @ a = conv(a, a) where C[k,j] = a[k-j] if 0<=k-j<n else 0
        # d(conv[k])/d(a[j]) = a[k-j] if 0<=k-j<n (from the first a) + a[j-k] if 0<=j-k<n (from the second a)
        # = a[k-j] + a[j-k] (both valid when in range)

        # Let me compute this using the structure

        # grad[j] = sum_k weights[k] * (a[k-j] + a[j-k])
        # Note that a[j-k] = a[(-k)-j + 2k]... no that's not right
        # a[j-k] is just a[j-k] if 0<=j-k<n
        # The matrix for the second term: C2[k,j] = a[j-k] if 0<=j-k<n
        # C2 is the flip of C: C2[k,j] = C[j,k]
        # So grad = C^T @ weights + C2^T @ weights = C^T @ weights + C @ weights
        
        # C @ weights: (C @ weights)[j] = sum_k C[k,j] * weights[k]
        # Wait, C has shape (2n-1, n), so C @ weights doesn't make sense (weights has length 2n-1)
        # C^T @ weights: shape (n,) ✓
        # C @ weights: doesn't work dimensionally
        # What we need is the matrix for the second term
        # Let's define C2[k,j] = a[j-k] if 0<=j-k<n else 0
        # C2 is actually the transpose of C! Because C[k,j] = a[k-j] and C2[k,j] = a[j-k] = C[j,k]
        # So C2 = C^T
        # And C2^T = C
        # So grad[j] = (C^T @ weights)[j] + (C^T @ C^T @ ... no

        # Term 1: (C^T @ weights)[j] = sum_k C[k,j] * weights[k] = sum_k a[k-j] * weights[k] (valid k-j)
        # Term 2: sum_k a[j-k] * weights[k] (valid j-k)
        #         = sum_k C[j,k] * weights[k] ... no, C[j,k] = a[j-k] if valid
        #         = (C @ something)... 
        # Actually C[j,k] is defined for j in 0..2n-2 and k in 0..n-1
        # So term 2 at position j: sum_{k=0}^{n-1} C[j,k] * weights[k]... but weights has length 2n-1
        # Hmm, this is getting confusing. Let me just do it the brute force way for n=800.
        
        # For n=800, the convolution has 1599 elements. For each j in 0..799:
        # grad[j] = 2 * sum over valid pairs
        # This is O(n^2) = 640000 operations, which is fast.
        
        # Actually, I realize I can just use the fact that:
        # grad[j] = 2 * (cross_correlation of a and weights evaluated at position j)
        # where cross_correlation is computed via FFT
        
        # Let me use a cleaner FFT approach:
        # We want: g[j] = sum_k weights[k] * a[k-j] for valid indices
        # This is the cross-correlation of (weights, a) at lag j
        # = ifft(conj(rfft(a)) * rfft(weights_padded))
        
        a_padded = np.zeros(2*n-1)
        a_padded[:n] = a
        a_f = rfft(a_padded, 2*n-1)
        w_f = rfft(weights, 2*n-1)
        # Cross-correlation: ifft(conj(a_f) * w_f) gives sum_k a[k]*weights[k+j]
        # We want sum_k weights[k]*a[k-j] = sum_k a[k]*weights[k-j] (reindexing)
        # = conj part... 
        # Actually: sum_k weights[k] * a[k-j] = sum_m weights[m+j] * a[m] (let m=k-j)
        # This is the cross-correlation of weights and a at lag j
        # = ifft(conj(rfft(a)) * rfft(weights)) at position j
        
        corr = irfft(np.conj(a_f) * w_f, 2*n-1)
        # corr[j] = sum_m a[m] * weights[m+j] for valid m
        # But we want sum_k weights[k] * a[k-j] = sum_m a[m] * weights[m+j] = corr[j] ✓
        
        # Similarly for the other term: sum_k weights[k] * a[j-k]
        # = sum_m a[m] * weights[j-m] (let m=j-k)
        # = cross-correlation at lag -j = corr[-j] = corr[2n-1-j] (circular)
        # But since we're doing linear, it's just corr_flipped
        
        # For autocorrelation (a*a), the gradient simplifies because of symmetry:
        # conv[k] = sum_i a[i]*a[k-i]
        # d(conv[k])/d(a[j]) = a[k-j] (from first a) + a[j-k] (from second a, since d(a[k-i])/d(a[j]) = delta(k-i-j))

        # d/d(a[j]): sum_i delta(i-j)*a[k-i] + sum_i a[i]*delta(k-i-j)
        # = a[k-j] + a[j-k] (if both indices valid)
        
        # So grad[j] = sum_k weights[k] * (a[k-j] + a[j-k])
        # First part: sum_k weights[k] * a[k-j] = corr1[j] (as computed above)

        # Let's compute: sum_k weights[k] * a[j-k] = sum_i a[i] * weights[j-i] (let i=j-k)
        # = cross-correlation of a and weights at lag -j
        # Using FFT: ifft(conj(rfft(weights)) * rfft(a_padded)) at position j? No...
        # sum_i a[i] * weights[j-i] = (a * weights_flipped)[j]... 
        # Actually this is just the regular convolution of a and reversed weights
        # = convolve(a, weights[::-1])... but lengths differ
        
        # You know what, for n=800 this brute force is fast enough:
        # grad = 2*corr where corr[j] = sum_k weights[k]*a[k-j] (first term)
        # The second term is similar. Due to symmetry of the autocorrelation, 
        # if weights is symmetric (which it won't be exactly), the two terms would be equal.
        
        # For simplicity and correctness, let me compute the full gradient:
        grad = np.zeros(n)
        for j in range(n):
            # First term: sum_k weights[k] * a[k-j]
            # k-j must be in [0, n-1], so k in [j, j+n-1]

            k_end = min(2*n-1, j+n)

            # Second term: sum_k weights[k] * a[j-k]

            k_end2 = min(2*n-1, j+1)
            if k_start2 < k_end2:
                grad[j] += np.dot(weights[k_start2:k_end2], a[j:k_end2-k_start2][::-1])
        
        # Project gradient to simplex (maintain sum=1, nonnegativity)
        # Mirror descent step
        log_a = np.log(a)
        log_a -= lr * grad / (a + 1e-15)  # normalized gradient for multiplicative update
        log_a -= np.max(log_a)
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode % 8 == 0:
        a = np.ones(n) / n
    elif mode % 8 == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode % 8 == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 8 == 3:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 8 == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-2 * (x - 0.3) ** 2) + np.exp(-2 * (x - 0.7) ** 2)
        a = a / a.sum()
    elif mode % 8 == 5:
        x = np.linspace(0, 1, n)
        a = x**2 + (1 - x)**2 + 0.1
        a = a / a.sum()
    elif mode % 8 == 6:
        freqs = rng.integers(2, 8, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)
    return a

"""
Optimized step function search using LP alternating with parabolic step-size
interpolation, adaptive momentum, multi-length sweep, and periodic restarts.
Target: score < 1.5027 (previous best: 1.506328).

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a / s)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix C (shape 2n-1 x n) such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def solve_lp(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    C = build_conv_matrix(a_norm)

    # Variables: [g_0, ..., g_{n-1}, M] -> n+1 total
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Inequality: C @ g - M <= 0  =>  [C, -1] @ [g; M] <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = C
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n == 1 else 0.0)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

                        if new_obj < best_obj - 1e-13:
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[src] += delta
                            a[target] -= delta
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: LP alternating with parabolic step sizes,
    smooth gradient refinement, multi-n sweep, peak reduction,
    and periodic restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent loaded (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"\nInitial: n={n_current}, obj={best_score:.8f}, target=1.5027, gap={best_score-1.5027:.6f}")

    # === Phase 1: Intensive LP Alternating (35% of time) ===
    print(f"\n{'='*50}")
    print("Phase 1: Intensive LP Alternating")
    print(f"{'='*50}")
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.35

    # Multiple LP passes with increasing aggressiveness
    n_passes = 4
    per_pass = p1_budget / n_passes

    for pass_idx in range(n_passes):
        remaining = deadline - time.monotonic()
        alloc = min(per_pass, remaining * 0.35)
        if alloc < 20:
            break

        if pass_idx == 0:
            a_start = best_seq.copy()
        else:
            rng_p = np.random.default_rng(seed + pass_idx * 31337)
            scale = 0.04 * (pass_idx + 1)
            a_start = best_seq * np.exp(rng_p.normal(0, scale, n_current))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()

        a_lp = lp_alternate(a_start, time_budget=alloc, verbose=(pass_idx < 2))
        obj_lp = compute_obj(a_lp)
        marker = " **" if obj_lp < best_score - 1e-10 else ""
        print(f"  LP pass {pass_idx+1}/{n_passes}: obj={obj_lp:.8f}{marker}")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()

    # === Phase 2: Smooth Refinement (10%) ===
    print(f"\n{'='*50}")
    print("Phase 2: Smooth Gradient Refinement")
    print(f"{'='*50}")
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.10
    if p2_budget > 15:
        a_sm = smooth_refine(best_seq, time_budget=p2_budget, verbose=True)
        obj_sm = compute_obj(a_sm)
        marker = " **" if obj_sm < best_score - 1e-10 else ""
        print(f"  Smooth: obj={obj_sm:.8f}{marker}")
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()

    # === Phase 3: N-Sweep (20%) ===
    print(f"\n{'='*50}")
    print("Phase 3: Multi-Length Sweep")
    print(f"{'='*50}")
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.20

    # Generate n options
    n_base = n_current
    n_options = sorted(set([
        n_base - 100, n_base - 50, n_base - 30, n_base - 10,
        n_base, n_base + 10, n_base + 30, n_base + 50,
        n_base + 100, n_base + 200,
        600, 650, 700, 750, 768, 800, 840, 880, 900, 960, 1000, 1024, 1100, 1200,
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1500]
    # Sort by priority: closest to current first
    n_options.sort(key=lambda x: abs(x - n_base))

    per_n = p3_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = max(8, min(per_n, remaining * 0.08))

        a_try = resample_sequence(best_seq, n_try)
        a_opt = lp_alternate(a_try, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)
        marker = " **" if obj_opt < best_score - 1e-10 else ""
        print(f"  n={n_try}: obj={obj_opt:.8f}{marker}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try

    # === Phase 4: Focused Peak Reduction (10%) ===
    print(f"\n{'='*50}")
    print("Phase 4: Focused Peak Reduction")
    print(f"{'='*50}")
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.10
    if p4_budget > 10:
        a_foc = focused_peak_reduction(best_seq, time_budget=p4_budget)
        obj_foc = compute_obj(a_foc)
        marker = " **" if obj_foc < best_score - 1e-10 else ""
        print(f"  Focused: obj={obj_foc:.8f}{marker}")
        if obj_foc < best_score:
            best_score = obj_foc
            best_seq = a_foc.copy()

    # === Phase 5: Final Intensive LP (20%) ===
    print(f"\n{'='*50}")
    print("Phase 5: Final Intensive LP")
    print(f"{'='*50}")
    remaining = deadline - time.monotonic()
    p5_budget = remaining * 0.20
    if p5_budget > 20:
        a_fin = lp_alternate(best_seq, time_budget=p5_budget, verbose=True)
        obj_fin = compute_obj(a_fin)
        marker = " **" if obj_fin < best_score - 1e-10 else ""
        print(f"  Final LP: obj={obj_fin:.8f}{marker}")
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()

    # === Phase 6: Last-chance polish (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = lp_alternate(best_seq, time_budget=min(remaining * 0.6, 40.0), verbose=False)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last polish: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"Target: 1.5027 | Achieved: {best_score:.8f} | Gap: {best_score - 1.5027:.8f}")
    print(f"{'='*50}")

    return result
