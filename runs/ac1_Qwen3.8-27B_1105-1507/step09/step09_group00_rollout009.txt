    size = next_fast_len(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, solver_list=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]

    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None


def subgradient_step(a, lr):
    """
    One subgradient step: find argmax of conv(a,a), compute subgradient, step.
    """
    n = len(a)
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    # Subgradient of (a*a)[k_star] w.r.t. a:
    # d/d a[j] of sum_i a[i]*a[k_star-i] = a[k_star-j] + a[j] (if both valid)
    # More carefully: (a*a)[k] = sum_{i} a[i] * a[k-i]
    # d/d a[j] = a[k-j] (from the i=j term) + a[j] (from the k-i=j term, i.e., i=k-j)
    # But only if 0 <= k-j < n
    subgrad = np.zeros(n)
    for j in range(n):
        val = 0.0
        # Term where i=j: contributes a[k_star - j] if 0 <= k_star-j < n
        if 0 <= k_star - j < n:
            val += a[k_star - j]
        # Term where k_star - i = j, i.e., i = k_star - j: contributes a[j] if 0 <= k_star-j < n
        # This is the same condition, so it's 2*a[k_star-j] if valid... 
        # Wait, let me be more careful.
        # (a*a)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * a[k-i]
        # d/d a[j] of this:
        # If j is in the range of i: contributes a[k-j]
        # If j = k-i for some valid i (i.e., i=k-j is valid): contributes a[j]... no.
        # Let me just do it numerically: d/d a[j] sum_i a[i]*a[k-i]
        # = sum_i [delta_{i,j} * a[k-i] + a[i] * delta_{k-i,j}]
        # = a[k-j] (if 0<=k-j<n) + a[j] (if 0<=k-j<n)  [same condition]
        # Wait no: delta_{k-i,j} means i = k-j, so the sum becomes a[k-j] (evaluated at i=k-j)
        # = a[k-j] if 0 <= k-j < n
        # Total: a[k-j] + a[k-j] = 2*a[k-j] if 0 <= k-j < n
        # BUT: if j = k-j (i.e., 2j = k), we're counting the same term twice.
        # Actually no, the two sums are independent: first sum is over i, second is over i.
        # First sum: i=j gives a[k-j]
        # Second sum: i=k-j gives a[k-j]... no, a[i] where i = k-j, so a[k-j]
        # So total = a[k-j] + a[k-j] = 2*a[k-j] (when 0 <= k-j < n)
        # UNLESS k-j = j, in which case it's just a[j] (since the term a[j]*a[j] only appears once in the sum)
        # Hmm, actually in the sum sum_i a[i]*a[k-i], when i=j and k-i=j (i.e., k=2j),
        # the term a[j]*a[j] appears only once (at i=j).
        # So d/d a[j] of a[j]*a[j] = 2*a[j] (product rule).
        # In general: d/d a[j] sum_i a[i]*a[k-i] = 2*a[k-j] if k != 2j, else 2*a[j]
        # Which is just 2*a[k-j] in all cases (since k-j = j when k=2j).
        # So subgrad[j] = 2*a[k_star - j] if 0 <= k_star - j < n, else 0.
        if 0 <= k_star - j < n:
            subgrad[j] = 2.0 * a[k_star - j]

    a_new = a - lr * subgrad
    a_new = project_simplex(a_new)
    return a_new, k_star


def smoothed_grad(a, tau):
    """Compute log-sum-exp smoothed gradient of max conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    # Gradient: d/da[j] of sum_k p[k] * (a*a)[k] = 2 * sum_k p[k] * a[k-j] (valid indices)
    # This is 2 * cross-correlation of p with a, restricted
    # Use FFT for speed
    size = next_fast_len(3 * n)
    P = fft(p, size)
    A = fft(a, size)
    # We need: for each j, sum_k p[k] * a[k-j] where 0<=k-j<n and 0<=k<2n-1
    # This is the cross-correlation: (p * flip(a)) or similar
    # Let's use: result[j] = sum_{k=0}^{2n-2} p[k] * a[k-j] (a[m]=0 outside [0,n-1])
    # = sum_{k=j}^{j+n-1} p[k] * a[k-j] (since a[m] != 0 only for 0<=m<n)
    # Let m = k-j: sum_{m=0}^{n-1} p[m+j] * a[m]
    # This is the cross-correlation of p and a at lag j.
    # cross_corr[j] = sum_m a[m] * p[m+j]
    # = ifft(fft(a) * conj(fft(p))) ... let me just do it directly for clarity
    # Actually for speed with FFT:
    # cross_corr = ifft(conj(fft(a, size)) * fft(p, size)).real
    # Hmm, let me just build it the straightforward way for correctness
    g = np.zeros(n)
    for j in range(n):
        # sum_m a[m] * p[m+j] for m=0..n-1 where m+j < 2n-1
        valid = min(n, len(p) - j)
        if valid > 0:
            g[j] = np.dot(a[:valid], p[j:j+valid])
    return 2.0 * g


def smoothed_grad_fft(a, tau):
    """Faster smoothed gradient using FFT cross-correlation."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = np.clip(tau * (conv - max_conv), -50, 0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    # cross_corr[j] = sum_{m=0}^{n-1} a[m] * p[m+j]
    # Pad p to length at least 2n-1 (already is)
    # Use FFT: pad a to length size, pad p to length size
    size = next_fast_len(2 * n + len(p) - 1)
    A = fft(a, size)
    P = fft(p, size)
    # cross_corr = ifft(A * conj(P)) ... let me verify
    # ifft(A * conj(P))[k] = sum_n A[n] * conj(P[(k-n) mod size])
    # This is circular cross-correlation. For our linear case, we need to be careful.
    # Actually: cross_corr[j] = sum_m a[m] * p[m+j]
    # Let's define q[m] = p[m+j] for fixed j. Then cross_corr[j] = sum_m a[m]*q[m]
    # This is the inner product, not a convolution.
    # Alternative: cross_corr[j] = sum_m a[m] * p[m+j] = sum_m a[m+j] * p[j+m]... 
    # Let me just use the direct computation but vectorized:
    # For efficiency, use the FFT-based convolution approach:
    # cross_corr[j] = (a * flip(p))[j + n - 1] ... 
    # Actually, let's use: flip_p = p[::-1], then conv(a, flip_p)[j + n - 1] = sum_m a[m] * flip_p[n-1-j+m] = sum_m a[m]*p[n-1-(n-1-j+m)] = sum_m a[m]*p[j-m]
    # Hmm this is getting confusing. Let me just do the direct O(n^2) computation since n<=2000.
    
    g = np.zeros(n)
    for j in range(n):
        end = min(n, len(p) - j)
        if end > 0:
            g[j] = np.dot(a[:end], p[j:j+end])
    return 2.0 * g


def lp_iteration(a, t_schedule=None, n_steps=50, time_budget=60.0, verbose=False):
    """
    Perform LP-based best response iteration with adaptive step size.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    last_t = 0.5
    stall = 0

    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        # Try multiple step sizes (line search)
        candidates_t = [last_t * 0.5, last_t * 0.75, last_t, last_t * 1.25,
                        last_t * 1.5, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        candidates_t = [t for t in candidates_t if 0.01 <= t <= 1.0]
        candidates_t = sorted(set(round(t, 4) for t in candidates_t))

        best_t_obj = best_obj
        best_t_val = last_t
        best_a_local = best_a

        for t in candidates_t:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall = 0
            if verbose and it % 20 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}")
        else:
            stall += 1
            if stall > 5:
                # Try subgradient steps to escape
                for _ in range(10):
                    a_sub, _ = subgradient_step(best_a, 1.0 / (np.sqrt(n) * 5))
                    obj_sub = compute_obj(a_sub)
                    if obj_sub < best_obj:
                        best_obj = obj_sub
                        best_a = a_sub.copy()
                        a = best_a
                stall = 0
                if it > 30:
                    break

    return best_a


def gradient_polish(a, time_budget=60.0, tau_init=50.0, tau_end=800.0, verbose=False):
    """
    Gradient descent with cosine annealing on the smoothed objective.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    beta = 0.9
    patience = 0
    best_patience = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress

        y = a + beta * v
        y = project_simplex(y)

        g = smoothed_grad_fft(y, tau)
        g_norm = float(np.linalg.norm(g))

        # Adaptive lr
        if g_norm > 1e-10:
            lr = np.clip(lr * (0.5 / (g_norm + 0.5)), 0.1 / np.sqrt(n), 5.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = project_simplex(y + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        a = a_new

        if patience > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)
            patience = 0

        if verbose:
            print(f"    GD: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.4f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 3:
        # Sinc-like
        x = np.linspace(-1, 1, n)
        a = np.sinc(x * 2) ** 2
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 4:
        # Gaussian.centered
        x = np.arange(n) - n / 2
        a = np.exp(-x**2 / (2 * (n/4)**2))
        a = a / a.sum()
    elif mode == 5:
        # Modulated cosine
        x = np.linspace(0, 1, n)
        a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) * (1 + 0.3 * np.sin(4 * np.pi * x))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Two bumps
        c1, c2 = n // 4, 3 * n // 4
        w = n // 10
        a = np.exp(-((np.arange(n) - c1)**2) / (2*w**2)) + np.exp(-((np.arange(n) - c2)**2) / (2*w**2))
        a = a / a.sum()
    elif mode == 7:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 4)
        a = a / a.sum()
    elif mode == 8:
        # Random Fourier
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines LP best-response iteration, gradient polishing,
    multi-n exploration, and diverse restarts to minimize the objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0  # safety buffer

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        best_seq = np.ones(n_current) / n_current
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")

    # === Phase 1: LP iteration from parent (warm start) ===
    print(f"\n=== Phase 1: LP Warm Start ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.15, 200.0)
    if p1_budget > 30:
        a_lp = lp_iteration(best_seq, n_steps=200, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: Gradient polish ===
    print(f"\n=== Phase 2: Gradient Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 180.0)
    if p2_budget > 30:
        a_gd = gradient_polish(best_seq, time_budget=p2_budget, tau_init=30.0, tau_end=500.0)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Gradient: obj={best_score:.8f} **")
        else:
            print(f"  Gradient: obj={obj_gd:.8f}")

    # === Phase 3: Multi-n exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Generate candidate n values
    n_candidates = []
    for delta in [-200, -100, -50, 50, 100, 200, 400, 600, 1000]:
        n_try = n_best + delta
        if 400 <= n_try <= 3000:
            n_candidates.append(n_try)
    # Also try some absolute values
    for n_abs in [500, 700, 1000, 1200, 1500, 2000, 2500]:
        if n_abs not in n_candidates and abs(n_abs - n_best) > 50:
            n_candidates.append(n_abs)

    n_candidates = sorted(set(n_candidates))
    # Limit to 6 best candidates (closest to current first, then spread)
    n_candidates.sort(key=lambda x: abs(x - n_best))
    n_candidates = n_candidates[:6]

    print(f"  Trying n values: {n_candidates}")

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break

        # Resample best_seq to new length
        a_try = resample_sequence(best_seq, n_try)
        s = float(np.sum(a_try))
        if s > 1e-10:
            a_try = a_try / s

        # LP iteration
        alloc_lp = min(remaining * 0.25, 120.0)
        if alloc_lp > 30:
            a_try = lp_iteration(a_try, n_steps=100, time_budget=alloc_lp)

        # Gradient polish
        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.15, 60.0)
        if alloc_gd > 20:
            a_try = gradient_polish(a_try, time_budget=alloc_gd, tau_init=50.0, tau_end=600.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Diverse restarts ===
    print(f"\n=== Phase 4: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        mode = r % 9
        a_r = diverse_init(n_current, rng, mode=mode)
        # Blend with best for warm start
        blend = 0.3 + 0.1 * min(r, 3)
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        # LP + GD
        alloc = min(remaining * 0.2, 100.0)
        if alloc > 40:
            half = alloc * 0.55
            a_r = lp_iteration(a_r, n_steps=80, time_budget=half)
            remaining = deadline - time.monotonic()
            gd_alloc = min(remaining * 0.1, 40.0)
            if gd_alloc > 15:
                a_r = gradient_polish(a_r, time_budget=gd_alloc, tau_init=50.0, tau_end=500.0)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 5: Intensive final polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Multiple LP rounds with decreasing step
    for lp_round in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.2, 80.0)
        a_lp = lp_iteration(best_seq, n_steps=100, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP {lp_round}: obj={best_score:.8f} **")
        else:
            # Small perturbation
            if lp_round < 3:
                noise = rng.normal(0, 0.01, len(best_seq))
                best_seq = np.maximum(best_seq + noise * best_seq, 0)
                best_seq = best_seq / best_seq.sum()
            break

    # Final gradient with high tau
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.3, 100.0)
        a_gd = gradient_polish(best_seq, time_budget=alloc, tau_init=100.0, tau_end=2000.0)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_lp = lp_iteration(best_seq, n_steps=50, time_budget=min(remaining * 0.5, 50.0))
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Ultra-final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
