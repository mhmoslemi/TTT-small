import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coord_descent_sweep(a, conv, time_budget=None, start_time=0, rng=None, verbose=False, max_gs_iters=30):
    """
    Fast coordinate descent on normalized self-convolution peak.
    Uses tighter brackets and adaptive ordering.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        conv = compute_conv(a)
        return a, conv
    a = a / S
    conv = compute_conv(a)

    best_obj = float(np.max(conv))
    best_a = a.copy()
    gr = (np.sqrt(5) - 1) / 2.0
    total_sweeps = 0
    stagnant = 0

    # Precompute direction vectors for efficiency
    # d_k[m] = a[m-k] for m in [k, k+n-1], 0 otherwise
    # We'll compute on-the-fly for memory efficiency but cache the structure

    for sweep in range(1000000):
        if time_budget is not None and time.monotonic() - start_time > time_budget:
            break

        obj_before = float(np.max(conv))

        # Adaptive ordering: start from coordinates near the peak
        peak_idx = int(np.argmax(conv))
        # Coordinates that affect the peak: k where |peak_idx - k - (n-1)| is small
        # Actually, coordinate k affects conv[m] for m in [k, k+n-1]
        # So coordinate k affects the peak at peak_idx if k <= peak_idx <= k+n-1
        # i.e., peak_idx - n + 1 <= k <= peak_idx
        # Process these first for faster convergence
        center_coords = list(range(max(0, peak_idx - n + 1), min(n, peak_idx + 1)))
        other_coords = [k for k in range(n) if k not in set(center_coords)]

        if rng is not None and sweep % 5 == 4:
            rng.shuffle(center_coords)
            rng.shuffle(other_coords)
        else:
            # Process from center outward
            center_coords.sort(key=lambda k: abs(k - peak_idx))
            other_coords.sort(key=lambda k: min(abs(k - peak_idx), abs(k - (peak_idx - n + 1))))

        order = center_coords + other_coords

        for k in order:
            # Direction: d[m] = a[m-k] for m in [k, k+n-1]
            d = np.zeros(2 * n - 1)
            d[k:k + n] = a

            # Tighter bracket using current peak
            max_conv_cur = float(np.max(conv))
            lo_delta = -a[k] * 0.999
            # Upper bound: we need conv[2k] + 2*delta*a[k] + delta^2 <= max_conv_cur + margin
            # delta^2 + 2*delta*a[k] <= max_conv_cur - conv[2k]
            rhs = max_conv_cur - conv[2 * k]
            if rhs > 0:
                upper_est = np.sqrt(a[k]**2 + rhs) - a[k]
            else:
                upper_est = 0.0
            hi_delta = max(upper_est * 1.2, 2.0 / n)
            hi_delta = max(hi_delta, lo_delta + 1.0 / n)
            hi_delta = min(hi_delta, 3.0 / n)  # Cap to avoid huge jumps

            # Golden section search with fewer iterations
            a_gs = lo_delta
            b_gs = hi_delta
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)

            def obj_at_delta(delta):
                val = conv + 2.0 * delta * d
                val[2 * k] += delta * delta
                return float(np.max(val))

            fc = obj_at_delta(c_gs)
            fd = obj_at_delta(d_gs)

            for _ in range(max_gs_iters):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = obj_at_delta(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = obj_at_delta(d_gs)

            delta_opt = (a_gs + b_gs) / 2.0

            # Skip if no improvement
            new_obj_est = obj_at_delta(delta_opt)
            if new_obj_est >= obj_before - 1e-14:
                continue

            # Apply update
            a[k] += delta_opt
            a[k] = max(0.0, a[k])

            # Incremental convolution update
            conv += 2.0 * delta_opt * d
            conv[2 * k] += delta_opt * delta_opt

            # Re-normalize
            S_new = float(np.sum(a))
            if S_new < 1e-10:
                a[:] = 1.0 / n
                conv = compute_conv(a)
            else:
                scale = 1.0 / S_new
                a *= scale
                conv *= (scale * scale)

        total_sweeps += 1
        obj_after = float(np.max(conv))

        if obj_after < best_obj - 1e-12:
            best_obj = obj_after
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 8:
                if rng is not None:
                    scale = 0.005 + 0.003 * min(stagnant, 10)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    s = a.sum()
                    if s < 1e-10:
                        a = np.ones(n) / n
                    else:
                        a /= s
                    conv = compute_conv(a)
                    best_obj = float(np.max(conv))
                    stagnant = 0
                else:
                    break

        if verbose and total_sweeps % 5 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    CD sweep {total_sweeps}: obj={2.0*n*best_obj:.8f}, time={elapsed:.0f}s")

    return best_a, compute_conv(best_a)


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(n, 2 * n - 1 - j)
        if end > 0:
            A[j:j + end, j] = a_norm[:end]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None


def lp_polish(a, time_budget=100.0, verbose=False, rng=None):
    """
    LP-based polishing with golden-section line search on blend parameter.
    """
    if rng is None:
        rng = np.random.default_rng(42)

    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    iter_count = 0
    last_t = 0.3

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Golden section search on t in [0, 1] for optimal blend
        a_current = a.copy()

        def obj_at_t(t):
            a_try = (1.0 - t) * a_current + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return np.inf
            a_try = a_try / s_try
            return compute_obj(a_try)

        # Golden section on [0, 1]
        gr = (np.sqrt(5) - 1) / 2.0
        lo = 0.0
        hi = 1.0
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)
        fc = obj_at_t(c)
        fd = obj_at_t(d)

        for _ in range(25):
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = hi - gr * (hi - lo)
                fc = obj_at_t(c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
                fd = obj_at_t(d)

        t_opt = (lo + hi) / 2.0
        obj_opt = obj_at_t(t_opt)

        iter_count += 1
        if obj_opt < best_obj - 1e-13:
            a_try = (1.0 - t_opt) * a_current + t_opt * g_lp
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            best_obj = obj_opt
            best_a = a_try.copy()
            a = best_a
            last_t = t_opt
            if verbose and iter_count % 2 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={t_opt:.4f}")
        else:
            # Perturb and try again
            scale = 0.01 + 0.005 * min(iter_count, 15)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-10:
                break
            a /= s
            if iter_count > 50:
                break

    return best_a

    for k in range(0, n - 1, 2):

def spectral_init(n, rng):
    """
    Initialize using flat power spectrum approach.
    Try to make |FFT(a)|^2 as flat as possible.
    """
    # Start with random phases and inverse FFT
    for attempt in range(50):
        # Random non-negative initialization
        phases = rng.uniform(0, 2 * np.pi, n)
        # Amplitudes: try to be flat
        amps = np.ones(n) * 0.8 + rng.uniform(0, 0.4, n)
        # IFFT to get time domain
        spec = amps * np.exp(1j * phases)
        a = np.real(ifft(spec))
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            continue
        a /= s

        # Check if this is good (low peak)
        obj = compute_obj(a)
        if obj < 1.8:  # Better than uniform (which gives 2.0)
            return a

    # Fallback: raised cosine
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0.001)
    a /= a.sum()
    return a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 10)
        a = np.exp(-((np.arange(n) - c1)**2) / (2 * w**2))
        a += np.exp(-((np.arange(n) - c2)**2) / (2 * w**2))
        a = a / a.sum()
    elif mode == 3:
        # Semicircle (arcsine density)
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(np.maximum(1 - x**2, 1e-8)))
        a = a / a.sum()
    elif mode == 4:
        # Multi-frequency
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for _ in range(3):
            f = rng.integers(2, 8)
            p = rng.uniform(0, 2 * np.pi)
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1.5, 1.5)
        a = a / a.sum()
    elif mode == 6:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a / a.sum()
    elif mode == 7:
        # Rectangular with tapered edges
        a = np.ones(n)
        taper = max(1, n // 20)
        for i in range(taper):
            frac = (taper - i) / taper
            a[i] = frac
            a[n-1-i] = frac
        a = a / a.sum()
    elif mode == 8:
        # Spectral init
        a = spectral_init(n, rng)
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def subgradient_descent(a, time_budget=50.0, verbose=False):
    """
    Subgradient method with averaging for min max_k conv(a,a)[k] over simplex.
    Uses Nesterov acceleration.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute subgradient
        conv = compute_conv(y)
        k_star = int(np.argmax(conv))

        # Subgradient of max conv at a=y:
        # grad[i] = 2 * y[k_star - i] (if 0 <= k_star - i < n)
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if hi >= lo:
            grad[lo:hi+1] = 2.0 * y[k_star - hi:k_star - lo + 1][::-1]

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr *= 0.9999

        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 5.0 / np.sqrt(n) * 0.5
            best_it = it

        if verbose and it % 1000 == 0:
            print(f"    SGD it{it}: obj={best_obj:.8f}, lr={lr:.4f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining fast CD, block-2 CD, LP polishing,
    subgradient descent, spectral init, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={n_target}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = spectral_init(n_target, rng)
        best_score = compute_obj(best_seq)
        print(f"Spectral init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Fast Coordinate Descent ===
    print(f"\n=== Phase 1: Fast CD ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 30:
        a_cd, _ = coord_descent_sweep(best_seq, None, time_budget=p1_budget,
                                      start_time=time.monotonic(), rng=rng, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase1 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 CD: obj={obj_cd:.8f}")

    # === Phase 2: LP Polishing (with golden section line search) ===
    print(f"\n=== Phase 2: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.30, 350.0)
    if p2_budget > 30:
        for rnd in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(p2_budget / (6 - rnd), remaining * 0.6)
            if alloc < 20:
                break
            a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True, rng=rng)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
                if rnd < 5:
                    scale = 0.01 * (rnd + 1)
                    best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                    best_seq = np.maximum(best_seq, 0)
                    best_seq = best_seq / best_seq.sum()

    # === Phase 3: Block-2 Coordinate Descent ===
    print(f"\n=== Phase 3: Block-2 CD ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    if p3_budget > 30:
        a_b2, _ = block2_cd(best_seq, None, time_budget=p3_budget,
                            start_time=time.monotonic(), rng=rng, verbose=True)
        obj_b2 = compute_obj(a_b2)
        if obj_b2 < best_score:
            best_score = obj_b2
            best_seq = a_b2.copy()
            print(f"  Phase3 B2CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 B2CD: obj={obj_b2:.8f}")

    # === Phase 4: More CD (fine tuning) ===
    print(f"\n=== Phase 4: Fine CD ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 120.0)
    if p4_budget > 20:
        a_cd2, _ = coord_descent_sweep(best_seq, None, time_budget=p4_budget,
                                       start_time=time.monotonic(), rng=rng, verbose=True,
                                       max_gs_iters=35)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  Phase4 CD: obj={best_score:.8f} **")

    # === Phase 5: Subgradient descent ===
    print(f"\n=== Phase 5: Subgradient Descent ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.08, 60.0)
    if p5_budget > 15:
        a_sgd = subgradient_descent(best_seq, time_budget=p5_budget, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase5 SGD: obj={best_score:.8f} **")
        else:
            print(f"  Phase5 SGD: obj={obj_sgd:.8f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        n_current - 100, n_current - 50,
        n_current + 50, n_current + 100, n_current + 200,
        600, 700, 800, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1400 and abs(n - n_current) > 30]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        # CD
        alloc1 = min(remaining * 0.30, 60.0)
        if alloc1 > 15:
            a_try, _ = coord_descent_sweep(a_try, None, time_budget=alloc1,
                                           start_time=time.monotonic(), rng=rng,
                                           max_gs_iters=25)

        # LP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 40.0)
        if alloc2 > 10:
            a_try = lp_polish(a_try, time_budget=alloc2, rng=rng)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        mode = (r + 2) % 9
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc1 = min(remaining * 0.30, 60.0)
        if alloc1 > 15:
            a_r, _ = coord_descent_sweep(a_r, None, time_budget=alloc1,
                                         start_time=time.monotonic(), rng=rng,
                                         max_gs_iters=25)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.12, 30.0)
        if alloc2 > 10:
            a_r = lp_polish(a_r, time_budget=alloc2, rng=rng)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        alloc = min(remaining * 0.30, 80.0)
        a_cd_f, _ = coord_descent_sweep(best_seq, None, time_budget=alloc,
                                        start_time=time.monotonic(), rng=rng, verbose=True,
                                        max_gs_iters=40)
        obj_cd_f = compute_obj(a_cd_f)
        if obj_cd_f < best_score:
            best_score = obj_cd_f
            best_seq = a_cd_f.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.35, 70.0)
        a_lp_f = lp_polish(best_seq, time_budget=alloc, verbose=True, rng=rng)
        obj_lp_f = compute_obj(a_lp_f)
        if obj_lp_f < best_score:
            best_score = obj_lp_f
            best_seq = a_lp_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.40, 50.0)
        a_cd_f2, _ = coord_descent_sweep(best_seq, None, time_budget=alloc,
                                         start_time=time.monotonic(), rng=rng,
                                         max_gs_iters=45)
        obj_cd_f2 = compute_obj(a_cd_f2)
        if obj_cd_f2 < best_score:
            best_score = obj_cd_f2
            best_seq = a_cd_f2.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 30.0)
        a_lp_f2 = lp_polish(best_seq, time_budget=alloc, rng=rng)
        obj_lp_f2 = compute_obj(a_lp_f2)
        if obj_lp_f2 < best_score:
            best_score = obj_lp_f2
            best_seq = a_lp_f2.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
