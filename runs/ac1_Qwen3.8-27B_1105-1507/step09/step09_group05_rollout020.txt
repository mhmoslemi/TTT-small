"""
Optimizes non-negative step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated evaluation, MOSEK-based LP polishing with adaptive line search,
dual-variable-guided perturbation for escaping local minima, and multi-length exploration.
Key improvements over parent: adaptive golden-section step size, dual-guided search direction,
larger n exploration, symmetry exploitation, and better time allocation.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a_norm):
    """Build dense matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        length = min(n, 2 * n - 1 - j)
        A[j:j + length, j] = a_norm[:length]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal, dual_vars) or (None, None, None) on failure.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.GUROBI, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Try to extract dual variables for the inequality constraints
                    duals = None
                    try:
                        # The dual variables correspond to A @ g <= M constraints
                        # In cvxpy, we can access them via the constraint's dual value
                        duals = np.maximum(0, float(M.value) - A @ g_val + 1e-10)
                    except Exception:
                        duals = None
                    return g_val, float(M.value), duals
        except Exception:
            continue
    return None, None, None


def adaptive_lp_step(a, g_lp, obj_current, a_init, max_tries=8):
    """
    Find good blending factor t via adaptive search.
    Returns (best_a, best_obj, best_t).
    """
    best_obj = obj_current
    best_a = a_init.copy()
    best_t = 0.0

    # Start with aggressive step, back off if needed
    t_candidates = [1.0, 0.7, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02]
    
    # Try a few key values
    for t in t_candidates[:4]:
        a_try = (1.0 - t) * a_init + t * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            best_t = t

    # If we found improvement at t=1, try to go even further isn't possible (t max is 1)
    # If improvement at intermediate t, try to find the peak
    if best_t > 0 and best_t < 1.0:
        # Golden section search around best_t
        lo = best_t * 0.5
        hi = min(1.0, best_t * 2.0)
        gr = (np.sqrt(5) - 1) / 2
        for _ in range(4):
            t1 = hi - gr * (hi - lo)
            t2 = lo + gr * (hi - lo)
            a1 = (1 - t1) * a_init + t1 * g_lp
            a1 = np.maximum(a1, 0)
            a1 = a1 / max(1e-10, a1.sum())
            o1 = compute_obj(a1)

            a2 = (1 - t2) * a_init + t2 * g_lp
            a2 = np.maximum(a2, 0)
            a2 = a2 / max(1e-10, a2.sum())
            o2 = compute_obj(a2)

            if o1 < o2:
                hi = t2
            else:
                lo = t1

            t_mid = (lo + hi) / 2
            a_mid = (1 - t_mid) * a_init + t_mid * g_lp
            a_mid = np.maximum(a_mid, 0)
            a_mid = a_mid / max(1e-10, a_mid.sum())
            o_mid = compute_obj(a_mid)
            if o_mid < best_obj:
                best_obj = o_mid
                best_a = a_mid.copy()
                best_t = t_mid

    return best_a, best_obj, best_t


def lp_polish_round(a, max_iter=100, time_limit=100.0, verbose=False):
    """
    LP polishing with adaptive step sizes.
    Returns (best_a, best_obj, num_iterations, duals_last).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf, 0, None
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    no_improve = 0
    start = time.monotonic()
    it = 0
    last_duals = None

    while it < max_iter and (time.monotonic() - start) < time_limit:
        g_lp, M_lp, duals = solve_lp_minpeak(a)
        if g_lp is None:
            break
        last_duals = duals

        a_new, obj_new, t_best = adaptive_lp_step(a, g_lp, best_obj, a)

        it += 1
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
            if verbose and it % 5 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={t_best:.3f}, M={M_lp:.6f}")
        else:
            no_improve += 1
            if no_improve >= 4:
                break

    return best_a, best_obj, it, last_duals


def dual_guided_perturb(a, duals, rng, scale=0.03):
    """Perturb a in a direction guided by the LP dual variables (peak lags)."""
    n = len(a)
    a = a.copy()
    if duals is not None:
        # duals[k] > 0 means lag k is at the peak
        # Reduce contribution to peak lags
        # For lag k, contribution from a[j] is a[j] * a[k-j]
        # We want to reduce a[j] where it contributes most to peak lags
        peak_mask = duals > np.max(duals) * 0.9
        delta = np.zeros(n)
        for k in np.where(peak_mask)[0]:
            k = int(k)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    delta[j] += a[idx] * duals[k]
        max_d = np.max(np.abs(delta))
        if max_d > 1e-10:
            a = a - scale * 0.5 * delta / max_d
            a = np.maximum(a, 0)

    # Add small random perturbation
    a = a * np.exp(rng.normal(0, scale * 0.5, n))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_perturb(a, rng, method=0, scale=0.04):

    elif method == 4:
        # Add uniform component

    elif method == 5:

        phases = 1j * rng.normal(0, scale * 0.3, len(A))
        A *= np.exp(phases)

        a = np.maximum(a, 0)
    elif method == 6:

        a = a * (1 + scale * np.sin(2 * np.pi * rng.uniform(1, 4) * x + rng.uniform(0, 2 * np.pi)))

        # Block perturbation

            start = int(rng.integers(0, n - block))
            a[start:start+block] *= rng.uniform(1 - scale, 1 + scale)
    else:
        a = a * np.exp(rng.normal(0, scale, n))

    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def smoothed_gradient_descent(a, time_budget=50.0, tau_init=10.0, tau_end=300.0):
    """Simple smoothed gradient descent for quick warm-up and refinement."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    momentum = 0.9
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** progress

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        # Project gradient onto tangent space of simplex
        g = g - np.mean(g)

        v = momentum * v - lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1
        lr = max(1e-6, lr * 0.999)

        if it % 20000 == 0:
            a = best_a.copy()
            v = np.zeros(n)

    return best_a

        a = np.ones(n)
        for scale in [n//4, n//8, n//16]:
            a += 0.3 * np.cos(2 * np.pi * np.arange(n) / scale + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 7:
        # Symmetric random
        half = n // 2
        h = rng.uniform(0.1, 1, half)
        a = np.concatenate([h, h[::-1]])
        if n % 2 == 1:
            a = np.concatenate([h, [h[-1]], h[::-1]])
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP polishing with adaptive steps + dual-guided perturbation + multi-n exploration.
    Returns list of non-negative floats minimizing the evaluation objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Quick warm-up with smoothed GD (30s) ===
    print(f"\n=== Phase 1: Warm-up GD ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.03, 40.0)
    if p1_budget > 10:
        a_warm = smoothed_gradient_descent(best_seq, time_budget=p1_budget,
                                           tau_init=5.0, tau_end=200.0)
        obj_warm = compute_obj(a_warm)
        if obj_warm < best_score:
            best_score = obj_warm
            best_seq = a_warm.copy()
            print(f"  Warm-up GD: obj={best_score:.8f} **")
        else:
            print(f"  Warm-up GD: obj={obj_warm:.8f}")

    # === Phase 2: Intensive LP polishing (500s) ===
    print(f"\n=== Phase 2: Intensive LP Polish (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.38, 500.0)
    if p2_budget > 30:
        a_current = best_seq.copy()

        # Multiple rounds with perturbation
        num_rounds = 4
        round_budget = p2_budget / num_rounds

        for rnd in range(num_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(round_budget, remaining * 0.5)
            if alloc < 10:
                break

            if rnd > 0:
                # Perturb to escape local minimum
                if rnd == 1:
                    a_current = random_perturb(best_seq, rng, method=0, scale=0.03)
                elif rnd == 2:
                    a_current = random_perturb(best_seq, rng, method=4, scale=0.05)
                else:
                    a_current = random_perturb(best_seq, rng, method=int(rng.integers(0, 8)), scale=0.03)

            a_new, obj_new, iters, duals = lp_polish_round(a_current, max_iter=200,
                                                           time_limit=alloc, verbose=(rnd == 0))
            if obj_new < best_score - 1e-12:
                best_score = obj_new
                best_seq = a_new.copy()
                a_current = a_new.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} ({iters} iters) **")
            else:
                print(f"  LP round {rnd}: obj={obj_new:.8f} ({iters} iters)")

            # Dual-guided refinement
            remaining = deadline - time.monotonic()
            if duals is not None and remaining > 30:
                a_pert = dual_guided_perturb(a_current if obj_new >= best_score else best_seq,
                                             duals, rng, scale=0.02)
                alloc2 = min(remaining * 0.2, 60.0)
                if alloc2 > 10:
                    a_p, obj_p, it_p, _ = lp_polish_round(a_pert, max_iter=50, time_limit=alloc2)
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_p.copy()
                        print(f"  LP round {rnd} (dual-guided): obj={best_score:.8f} **")

    # === Phase 3: Multi-n exploration (350s) ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 100), max(400, n_current - 50),
        n_current + 50, n_current + 100, n_current + 200,
        n_current + 400, n_current + 600,
        600, 700, 900, 1000, 1200, 1400, 1600
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and n != n_current]
    n_options.sort(key=lambda x: abs(x - n_current))

    num_n_tries = min(len(n_options), int(remaining // 70))
    for i, n_try in enumerate(n_options[:num_n_tries]):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick GD warm-up
        alloc1 = min(remaining * 0.08, 30.0)
        if alloc1 > 8:
            a_sgd = smoothed_gradient_descent(a_try, time_budget=alloc1,
                                              tau_init=10.0, tau_end=200.0)
            if compute_obj(a_sgd) < compute_obj(a_try):
                a_try = a_sgd

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.12, 70.0)
        if alloc2 > 10:
            a_try, obj_try, it_try, _ = lp_polish_round(a_try, max_iter=80, time_limit=alloc2)
        else:
            obj_try = compute_obj(a_try)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} ({it_try} LP iters) **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f} ({it_try} LP iters)")

    # === Phase 4: Diverse restarts (200s) ===
    print(f"\n=== Phase 4: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    num_restarts = min(4, int(remaining // 60))
    for r in range(num_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = (r + 2) % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.25, 80.0)
        if alloc > 20:
            half = alloc / 2
            a_r = smoothed_gradient_descent(a_r, time_budget=half,
                                            tau_init=10.0, tau_end=150.0)
            remaining = deadline - time.monotonic()
            lp_alloc = min(remaining * 0.1, 40.0)
            if lp_alloc > 10:
                a_r, obj_r, it_r, _ = lp_polish_round(a_r, max_iter=50, time_limit=lp_alloc)
            else:
                obj_r = compute_obj(a_r)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 5: Final intensive polish (150s) ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Extended LP polishing
    for lp_rnd in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining * 0.3, 60.0)
        if alloc < 8:
            break

        a_current = best_seq.copy()
        if lp_rnd > 0:
            a_current = random_perturb(a_current, rng, method=lp_rnd % 3, scale=0.015)

        a_lp, obj_lp, iters, _ = lp_polish_round(a_current, max_iter=100, time_limit=alloc)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP {lp_rnd}: obj={best_score:.8f} ({iters} iters) **")
        else:
            print(f"  Final LP {lp_rnd}: obj={obj_lp:.8f} ({iters} iters)")

    # Final high-resolution GD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 40.0)
        a_sgd = smoothed_gradient_descent(best_seq, time_budget=alloc,
                                          tau_init=50.0, tau_end=800.0)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Final validation with evaluate_sequence
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
