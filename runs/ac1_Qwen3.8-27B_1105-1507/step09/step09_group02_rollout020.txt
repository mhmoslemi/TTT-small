```python
import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    # A is (2n-1) x n, A @ g = conv(a, g)
    # conv(a,g)[k] = sum_{i+j=k} a[i]*g[j]
    # For fixed a, this is linear in g
    # A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a[k-j] for k = j, j+1, ..., j+n-1
        for k_offset in range(n):
            k = j + k_offset
            if k < 2 * n - 1:
                A[k, j] = a[k_offset]
    return A

def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M)."""
    n = len(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def line_search(a, g, num_points=100):
    """Find best t in [0,1] to minimize f(t) = compute_obj((1-t)*a + t*g)."""
    best_t = 0.0
    best_obj = compute_obj(a)
    
    # First try a coarse grid
    for t in np.linspace(0, 1, num_points):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    return best_t, best_obj, (1 - best_t) * a + best_t * g
```

    conv_aa = compute_conv(a)
    conv_ag = np.convolve(a, g)  # or use FFT
    conv_gg = compute_conv(g)
    
    # Find top K peaks of conv_aa
    top_idx = np.argsort(conv_aa)[::-1][:K]

    B_k = conv_ag[top_idx]
    C_k = conv_gg[top_idx]

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: LP Frank-Wolfe with quadratic line search, perturbation, multi-n,
and gradient-based fine-tuning.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):
    """Compute conv(a, b) via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:len(a) + len(b) - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(j + n, 2 * n - 1)
        A[j:end, j] = a[:end - j]
    return A

def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def quadratic_line_search(a, g, conv_aa, K=30, num_points=300):
    """Find best t using quadratic model of top-K peaks."""
    conv_ag = compute_conv_cross(a, g)
    conv_gg = compute_conv(g)
    
    top_idx = np.argsort(conv_aa)[::-1][:K]
    
    A_k = conv_aa[top_idx]
    B_k = conv_ag[top_idx]
    C_k = conv_gg[top_idx]
    
    ts = np.linspace(0, 1, num_points)
    # Q_k(t) = A_k*(1-t)^2 + 2*B_k*t*(1-t) + C_k*t^2
    # = A_k + (2B_k - 2A_k)*t + (A_k - 2B_k + C_k)*t^2
    # Vectorized: for each t, compute Q_k(t) for all k and take max
    
    best_t = 0.0
    best_val = np.inf
    
    # Compute for all t at once (memory: K x num_points)
    t_col = ts[:, np.newaxis]  # (num_points, 1)
    A_row = A_k[np.newaxis, :]  # (1, K)
    B_row = B_k[np.newaxis, :]
    C_row = C_k[np.newaxis, :]
    
    Q = A_row * (1 - t_col)**2 + 2 * B_row * t_col * (1 - t_col) + C_row * t_col**2  # (num_points, K)
    max_Q = Q.max(axis=1)  # (num_points,)
    
    best_idx = np.argmin(max_Q)
    best_t = ts[best_idx]
    best_val = max_Q[best_idx]
    
    return best_t, best_val

def lp_frank_wolfe(a, time_budget, verbose=True):
    """Main LP Frank-Wolfe optimization loop."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - start < time_budget:
        t_iter_start = time.monotonic()
        
        # Solve LP
        g, M_lp = solve_lp(a)
        if g is None:
            if verbose:
                print(f"  LP failed at it {it}")
            break
        
        # Compute current conv for line search
        conv_aa = compute_conv(a)
        
        # Quadratic line search
        t_best, t_val = quadratic_line_search(a, g, conv_aa)
        
        # Also verify with a few direct evaluations
        direct_candidates = [t_best, 0, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        direct_candidates = [t for t in direct_candidates if 0 <= t <= 1]
        
        best_t = t_best
        best_obj_try = best_obj
        
        for t_try in direct_candidates:
            if abs(t_try - t_best) < 0.01 and t_try != t_best:
                continue
            a_try = (1 - t_try) * a + t_try * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_try:
                best_obj_try = obj_try
                best_t = t_try
        
        # Update
        if best_obj_try < best_obj - 1e-14:
            a_new = (1 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            best_obj = best_obj_try
            best_a = a_new.copy()
            a = a_new
            stall = 0
            it += 1
            
            if verbose and it % 50 == 0:
                elapsed = time.monotonic() - start
                print(f"  FW it{it}: obj={best_obj:.8f}, t={best_t:.4f}, M_lp={M_lp:.6f}, "
                      f"time={elapsed:.0f}s, iter_time={time.monotonic()-t_iter_start:.3f}s")
        else:
            stall += 1
            # Perturb to escape
            if stall == 5:
                scale = 0.02
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
            elif stall == 15:
                scale = 0.05
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
            elif stall == 30:
                scale = 0.1
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
            elif stall > 50:
                break
        
        # Check time
        if time.monotonic() - start > time_budget * 0.95:
            break
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def smoothed_gradient_fft(a, tau):
    """Compute 2 * sum_k p[k] * a[k-j] efficiently using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    # We want grad[j] = 2 * sum_{m} p[m+j] * a[m] for m in [0, n-1], m+j in [0, 2n-2]
    # This is 2 * (reverse correlation of p and a at lag j)
    # = 2 * (ifft(fft(p, 2n) * conj(fft(a, 2n)))[j]) ... let me think.
    
    # Actually, sum_m p[m+j] * a[m] is the cross-correlation of p and a at lag j.
    # cross_correlation[j] = sum_m p[m+j] * a[m]
    # = sum_m a[m] * p[j+m]
    # This is conv(a, p_reversed)? No...
    
    # Let me define: corr[j] = sum_m a[m] * p[j+m]
    # = sum_m a[m] * p_rev[n-1-(j+m)]  ... no this is getting complicated.
    
    # Let me use the definition: corr(j) = E[p(X+j) * a(X)] for X in [0, n-1]
    # = sum_{m=0}^{n-1} a[m] * p[m+j]
    
    # Using FFT: corr = ifft(conj(fft(a, N)) * fft(p, N)) where N >= n + (2n-2) = 3n-2
    # Actually, the cross-correlation via FFT is:
    # corr = ifft(fft(a, N) * conj(fft(p, N)))  ... no.
    
    # The standard formula: (f * g)[n] = sum_k f[k] * g[n-k] (convolution)
    # Cross-correlation: (f ⋆ g)[n] = sum_k f[k+n] * g[k] = sum_k f[k] * g[k-n]
    # Using FFT: f ⋆ g = ifft(conj(fft(f)) * fft(g))
    
    # So corr[j] = sum_m p[m+j] * a[m] = (p ⋆ a)[j]  (cross-correlation of p and a at lag j)
    # Wait: (p ⋆ a)[j] = sum_m p[m+j] * conj(a[m]) ... in real case = sum_m p[m+j] * a[m]
    # Yes! So corr = ifft(conj(fft(p, N)) * fft(a, N))
    
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    corr = ifft(np.conj(P) * A).real[:n]
    
    grad = 2.0 * corr
    return grad, conv

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = int(rho[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def gradient_polish(a, time_budget, verbose=False):
    """Smoothed gradient descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    tau = 50.0
    lr = 1.0 / np.sqrt(n)
    v = np.zeros(n)
    beta = 0.5  # momentum
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        grad, conv = smoothed_gradient_fft(a, tau)
        
        # Normalized gradient descent with momentum
        v = beta * v - lr * grad
        a_new = a + v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            tau *= 1.01  # increase temperature (sharpen)
        else:
            stall += 1
            if stall > 50:
                lr *= 0.5
                v *= 0.5
                a = best_a.copy()
                stall = 0
                if lr < 1e-10:
                    break
        
        it += 1
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}, time={elapsed:.0f}s")
    
    return best_a
