def compute_conv(a):

    n = len(a)
    if n == 0:

    while size < 2*n - 1:

    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # conv(a,g)[k] = sum_j a[j]*g[k-j]
        # For fixed j, contributes a[j]*g[k-j] to position k = j + i where i is the g-index
        # So A[j+i, i] += a[j] for valid k=j+i
        for i in range(n):
            k = j + i
            if k < 2 * n - 1:
                A[k, i] = a[j]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    # Build constraint matrix efficiently
    # conv(a, g)[k] = sum_j a[j] * g[k-j]
    # For each k, the coefficient of g[i] is a[k-i] if 0 <= k-i < n
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        # g[i] appears in conv[a, g][k] with coefficient a[k-i]
        # k ranges from i to i+n-1 (where k-i ranges from 0 to n-1)
        A[i:i + n, i] = a_norm

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None

def sine_window_init(n, rng):
    """Initialize with a half-sine window (good for uniform autocorrelation)."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x)
    a = np.maximum(a, 0.001)
    # Add slight random perturbation
    a *= np.exp(rng.normal(0, 0.01, n))
    a = a / a.sum()
    return a


def raised_cosine_init(n, rng):
    """Initialize with a raised cosine window."""
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0.001)
    a *= np.exp(rng.normal(0, 0.01, n))
    a = a / a.sum()
    return a


def fourier_flattened_init(n, rng):
    """
    Initialize using Fourier-domain flattening.
    Start with uniform, randomize phases in Fourier domain, IFFT, clip to non-negative.
    """
    x = np.ones(n) / n
    X = rfft(x, 2 * n)
    # Randomize phases (keep DC component)
    phases = np.exp(1j * rng.uniform(0, 2 * np.pi, len(X)))
    X[0] = abs(X[0])  # Keep DC real and positive
    X *= phases
    a = irfft(X, 2 * n)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        a = sine_window_init(n, rng)
    elif mode == 3:
        a = raised_cosine_init(n, rng)
    elif mode == 4:
        a = fourier_flattened_init(n, rng)
    elif mode == 5:
        # Two-peak initialization
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 8)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 6:

        a = x ** rng.uniform(-1.5, 1.5)
        a = a / a.sum()
    elif mode == 7:
        # Smoothed Dirichlet
        a = rng.dirichlet(np.ones(n) * 0.5)
    else:

    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Multi-phase optimization combining coordinate descent,
    LP polishing with line search, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent if available
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = sine_window_init(n_target, rng)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Coordinate Descent ===
    print(f"\n=== Phase 1: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 30:
        a_cd, _ = coord_descent(best_seq, None, time_budget=p1_budget,
                                start_time=time.monotonic(), rng=rng, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase1 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 2: LP Polishing with Line Search ===
    print(f"\n=== Phase 2: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.30, 400.0)
    if p2_budget > 30:
        # Run LP polish in rounds
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(p2_budget / (3 - rnd), remaining * 0.7)
            if alloc < 20:
                break
            a_lp = lp_polish(best_seq, time_budget=alloc, rng=rng, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
                if rnd < 2:
                    scale = 0.015 * (rnd + 1)
                    perturbed = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                    perturbed = np.maximum(perturbed, 0)
                    perturbed = perturbed / perturbed.sum()
                    a_lp2 = lp_polish(perturbed, time_budget=min(alloc * 0.5, 60.0), rng=rng)
                    obj_lp2 = compute_obj(a_lp2)
                    if obj_lp2 < best_score:
                        best_score = obj_lp2
                        best_seq = a_lp2.copy()
                        print(f"  LP round {rnd} (perturbed): obj={best_score:.8f} **")

    # === Phase 3: More Coordinate Descent ===
    print(f"\n=== Phase 3: Fine Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.20, 200.0)
    if p3_budget > 20:
        a_cd2, _ = coord_descent(best_seq, None, time_budget=p3_budget,
                                 start_time=time.monotonic(), rng=rng, verbose=True)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  Phase3 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 CD: obj={obj_cd2:.8f}")

    # === Phase 4: LP Polish Again ===
    print(f"\n=== Phase 4: LP Polish (second pass) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.25, 150.0)
        a_lp2 = lp_polish(best_seq, time_budget=alloc, rng=rng, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase4 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 LP: obj={obj_lp2:.8f}")

    # === Phase 5: Multi-length Exploration ===
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Try a few nearby lengths
    n_options = []
    for dn in [-100, -50, 50, 100, 150, -200, 200]:
        n_try = n_current + dn
        if 300 <= n_try <= 2000 and abs(dn) >= 50:
            n_options.append(n_try)

    # Remove duplicates and sort
    n_options = sorted(set(n_options))

    tried_lengths = 0
    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick coordinate descent
        alloc1 = min(remaining * 0.30, 60.0)
        if alloc1 > 15:
            a_try, _ = coord_descent(a_try, None, time_budget=alloc1,
                                     start_time=time.monotonic(), rng=rng)

        # Quick LP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 40.0)
        if alloc2 > 10:
            a_try = lp_polish(a_try, time_budget=alloc2, rng=rng)

        obj_try = compute_obj(a_try)
        tried_lengths += 1
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        mode = (r + 2) % 8  # Skip uniform(0) and random(1)
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        # Coordinate descent
        alloc1 = min(remaining * 0.35, 60.0)
        if alloc1 > 15:
            a_r, _ = coord_descent(a_r, None, time_budget=alloc1,
                                   start_time=time.monotonic(), rng=rng)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.20, 40.0)
        if alloc2 > 10:
            a_r = lp_polish(a_r, time_budget=alloc2, rng=rng)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Coordinate descent
    if remaining > 30:
        alloc = min(remaining * 0.25, 80.0)
        a_cd_f, _ = coord_descent(best_seq, None, time_budget=alloc,
                                  start_time=time.monotonic(), rng=rng, verbose=True)
        obj_cd_f = compute_obj(a_cd_f)
        if obj_cd_f < best_score:
            best_score = obj_cd_f
            best_seq = a_cd_f.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # LP
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.30, 60.0)
        a_lp_f = lp_polish(best_seq, time_budget=alloc, rng=rng, verbose=True)
        obj_lp_f = compute_obj(a_lp_f)
        if obj_lp_f < best_score:
            best_score = obj_lp_f
            best_seq = a_lp_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # One more CD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.40, 50.0)
        a_cd_f2, _ = coord_descent(best_seq, None, time_budget=alloc,
                                   start_time=time.monotonic(), rng=rng)
        obj_cd_f2 = compute_obj(a_cd_f2)
        if obj_cd_f2 < best_score:
            best_score = obj_cd_f2
            best_seq = a_cd_f2.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")

    # Final LP squeeze
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.50, 30.0)
        a_lp_f2 = lp_polish(best_seq, time_budget=alloc, rng=rng)
        obj_lp_f2 = compute_obj(a_lp_f2)
        if obj_lp_f2 < best_score:
            best_score = obj_lp_f2
            best_seq = a_lp_f2.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
