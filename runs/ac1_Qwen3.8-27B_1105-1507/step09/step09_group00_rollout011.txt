"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses quadratic-linearization LP polishing (new approach), cross-LP alternation,
FFT-accelerated smoothed gradient descent, spectral exploration for diversity,
and aggressive multi-length search. The key innovation is solving a LP that
directly linearizes the quadratic objective a^T Q_k a at the current point,
giving a better descent direction than the standard cross-correlation LP.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_cross_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M for all k, sum(g)=1, g>=0.
    Returns (optimal_g, optimal_M) or (None, None)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_quad_linearization_lp(a):
    """Solve LP that linearizes the quadratic objective at point a.
    min t s.t. 2*conv(a0, a_new) - conv(a0,a0) <= t, sum(a_new)=1, a_new>=0
    This gives a better descent direction than the cross LP."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    c = compute_conv(a_norm)
    a_var = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    constraints = [2 * (A @ a_var) - t <= c, cp.sum(a_var) == 1]
    prob = cp.Problem(cp.Minimize(t), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if a_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.maximum(a_var.value, 0)
                s = float(np.sum(a_val))
                if s > 1e-10:
                    return a_val / s, float(t.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a) components."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n], conv

def spectral_exploration(a, rng, strength=0.03):
    """Perturb in frequency domain to escape local minima."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    A = fft(a, size)
    n_modes = 4
    mask = np.zeros(size // 2 + 1, dtype=bool)
    for k in range(1, n_modes + 1):
        mask[k] = True
        mask[size - k] = True if size - k <= size // 2 else False
    # Perturb low-frequency components
    for k in range(1, n_modes + 1):
        if k < len(A):
            phase_shift = rng.uniform(-strength, strength)
            A[k] = A[k] * np.exp(1j * phase_shift)
    a_new = irfft(A, size)[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):

            print(f"  Phase3 Cross-LP: obj={obj_cross:.8f}")

    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(400, n_best - 100), max(400, n_best - 50),
        n_best + 50, n_best + 100, n_best + 200, n_best + 400,
        600, 700, 800, 900, 1000, 1100, 1200, 1400
    ]))
    # Filter: keep distinct values far from current
    n_options = [n for n in n_options if 500 <= n <= 1600 and abs(n - n_best) > 30]
    # Also include some close values
    n_options = sorted(set(n_options + [n_best - 20, n_best + 20]))
    n_options = [n for n in n_options if 400 <= n <= 1600]
    n_options = n_options[:8]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick GD
        alloc1 = min(remaining * 0.12, 40.0)
        if alloc1 > 15:
            a_try = fast_gd(a_try, time_budget=alloc1, tau_init=10.0, tau_end=300.0)

        # Short LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.1, 40.0)
        if alloc2 > 12:
            a_try = lp_polish_dual(a_try, time_budget=alloc2, verbose=False, use_quad_lp=True)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        mode = 1 + r % 5
        a_r = diverse_init(n_cur, rng, mode=mode)
        blend = 0.15 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.15, 80.0)
        if alloc > 20:
            half = alloc * 0.5
            a_r = fast_gd(a_r, time_budget=half, tau_init=10.0, tau_end=300.0)
            remaining = deadline - time.monotonic()
            lp_alloc = min(remaining * 0.1, 30.0)
            if lp_alloc > 10:
                a_r = lp_polish_dual(a_r, time_budget=lp_alloc, verbose=False, use_quad_lp=True)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Intensive Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau GD
    if remaining > 40:
        alloc = min(remaining * 0.2, 80.0)
        a_final = fast_gd(best_seq, time_budget=alloc, tau_init=50.0, tau_end=1000.0)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final GD: obj={best_score:.8f} **")
        else:
            print(f"  Final GD: obj={obj_final:.8f}")

    # Multiple LP passes (alternating types)
    remaining = deadline - time.monotonic()
    for lp_rnd in range(5):
        if remaining <= 15:
            break
        alloc = min(remaining * 0.25, 60.0)
        use_q = (lp_rnd % 2 == 0)
        a_lp = lp_polish_dual(best_seq, time_budget=alloc, verbose=False, use_quad_lp=use_q)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP {lp_rnd} ({'quad' if use_q else 'cross'}): obj={best_score:.8f} **")
        else:
            if lp_rnd < 3:
                best_seq = spectral_exploration(best_seq, rng, strength=0.005)
            else:
                break

    # Ultra-final: one more GD with very high tau
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.3, 40.0)
        a_uf = fast_gd(best_seq, time_budget=alloc, tau_init=200.0, tau_end=2000.0)
        obj_uf = compute_obj(a_uf)
        if obj_uf < best_score:
            best_score = obj_uf
            best_seq = a_uf.copy()
            print(f"  Ultra-final GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
