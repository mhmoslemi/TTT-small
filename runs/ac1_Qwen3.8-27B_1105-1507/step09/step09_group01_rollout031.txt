def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, 2 * j + n)
        A[j:end, j] = a[0:end - j]
    return A

from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a[j]
    return A


def solve_lp_minpeak(a, M_target=None):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue
    
    return None, None


def smoothed_grad_fft(a, tau):
    """Gradient of (1/tau)*log(sum_k exp(tau*conv_k)) w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    # g[i] = 2 * sum_k p_k * a_{k-i} (with a_j=0 outside [0,n-1])
    # = 2 * (flip(p) * a)[2n-2-i]
    flip_p = p[::-1]
    N = 1
    while N < len(flip_p) + n - 1:
        N <<= 1
    conv_full = ifft(fft(flip_p, N) * fft(a, N)).real
    # g[i] = 2 * conv_full[2n-2-i] for i=0..n-1
    g = 2.0 * conv_full[2*n-2 : n-1 : -1]
    return g, conv, p


def nesterov_descent(a, time_budget=60.0, tau_init=5.0, tau_end=500.0):
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0  # will update
    lr = 2.0 / (1.0 + tau_init)  # initial step
    v = np.zeros(n)
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)
        
        g, conv, p = smoothed_grad_fft(a, tau)
        
        # Nesterov momentum
        t_k = 1.0 + np.sqrt(1.0 + 4.0 * it * it) / 2.0
        t_prev = 1.0 + np.sqrt(1.0 + 4.0 * (it-1) * (it-1)) / 2.0 if it > 0 else 1.0
        beta_k = (1.0 - t_k) / (t_prev + 1.0) if it > 0 else 0.0
        
        y = a + beta_k * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            s_y = 1.0
        y = y / s_y
        
        g_y, _, _ = smoothed_grad_fft(y, tau)
        
        a_new = y - lr * g_y / tau
        a_new = project_simplex(a_new)
        
        v = a_new - y
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        
        # Adaptive step size
        if it % 100 == 0:
            if it - best_it > 500:
                lr *= 0.7
                v = np.zeros(n)
                a = best_a.copy()
                best_it = it
            elif obj_new < best_obj:
                lr = min(lr * 1.05, 5.0 / np.sqrt(n))
        
        # Symmetrize periodically
        if it % 200 == 0:
            a = (a + a[::-1]) / 2.0
            a = a / a.sum()
            obj_sym = compute_obj(a)
            if obj_sym < best_obj:
                best_obj = obj_sym
                best_a = a.copy()
    
    return best_a


def lp_polish(a, time_budget=200.0, seed=42, verbose=False):
    """LP polishing with backtracking line search."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    iter_count = 0
    no_improve = 0
    last_t = 0.5
    
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Try full step first, then backtracking
        improved = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.15, 0.08]:
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = best_a.copy()
                last_t = t
                no_improve = 0
                improved = True
                break
        
        iter_count += 1
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                # Perturb
                scale = 0.005 + 0.003 * min(no_improve, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                if iter_count > 50:
                    break
            else:
                # Try symmetrization
                a = (best_a + best_a[::-1]) / 2.0
                a = a / a.sum()
                obj_sym = compute_obj(a)
                if obj_sym < best_obj - 1e-14:
                    best_obj = obj_sym
                    best_a = a.copy()
        
        if verbose and iter_count % 10 == 0:
            print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={last_t:.2f}, M={M_lp:.6f}")
    
    return best_a


def peak_directed_perturb(a, rng, K=5, alpha=0.05):
    """Perturb a by moving mass from peak-contributing coords to others."""
    n = len(a)
    a = a.copy()
    conv = compute_conv(a)
    
    # Find top K peaks
    top_peaks = np.argsort(conv)[::-1][:K]
    
    # For each peak, find the most contributing coordinate
    for k in top_peaks:
        k = int(k)
        # conv[k] = sum_j a[j] * a[k-j]
        # Contribution of a[j] is a[j] * a[k-j]
        j_range = np.arange(max(0, k - n + 1), min(n, k + 1))
        contribs = a[j_range] * a[k - j_range]
        
        # Find the coordinate with max contribution
        j_max = j_range[np.argmax(contribs)]
        
        # Move some mass from j_max to a random other coordinate
        other_idx = rng.integers(0, n)
        while other_idx == j_max:
            other_idx = rng.integers(0, n)
        
        move = alpha * a[j_max]
        a[j_max] -= move
        a[other_idx] += move
    
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def kernel_init(n, rng, mode=0):
    """Generate initial sequences using various kernels."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        # Bimodal Gaussian
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 10)
        a = np.exp(-((np.arange(n) - c1)**2) / (2*w**2)) + np.exp(-((np.arange(n) - c2)**2) / (2*w**2))
        return a / a.sum()
    elif mode == 3:
        # Semicircle
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x**2, 0))
        return a / (a.sum() + 1e-15)
    elif mode == 4:
        # Multi-frequency
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for _ in range(rng.integers(2, 5)):
            f = rng.integers(1, 8)
            p = rng.uniform(0, 2*np.pi)
            a += 0.3 * np.sin(2*np.pi*f*x + p)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        return a / a.sum()
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def symmetrize(a):
    """Symmetrize: (a + flip(a)) / 2."""
    return (a + a[::-1]) / 2.0

Key improvements over previous:

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a[j]
    return A


def solve_lp_minpeak(a, M_target=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue

    return None, None


def smoothed_grad_fft(a, tau):
    """Gradient of (1/tau)*log(sum_k exp(tau*conv_k)) w.r.t. a, via FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    # g[i] = 2 * sum_k p_k * a_{k-i} = 2 * (flip(p) * a)[2n-2-i]
    flip_p = p[::-1]
    N = 1
    while N < len(flip_p) + n - 1:
        N <<= 1
    conv_full = ifft(fft(flip_p, N) * fft(a, N)).real
    g = 2.0 * conv_full[2 * n - 2: n - 1: -1]
    return g, conv, p


def nesterov_descent(a, time_budget=60.0, tau_init=5.0, tau_end=400.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    t = 1.0
    t_prev = 1.0
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        if verbose and it % 5000 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        # Nesterov momentum
        t = 1.0 + np.sqrt(1.0 + 4.0 * t * t) / 2.0
        beta_k = (1.0 - t_prev) / t

        y = a + beta_k * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv, p = smoothed_grad_fft(y, tau)

        a_new = y - (lr / tau) * g
        a_new = project_simplex(a_new)

        v = a_new - y
        a = a_new
        t_prev = t
        it += 1

        obj_new = compute_obj(a)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a.copy()
            best_it = it

        # Adaptive learning rate
        if it % 200 == 0:
            if it - best_it > 2000:
                lr *= 0.6
                v = np.zeros(n)
                a = best_a.copy()
                best_it = it
            else:
                lr = min(lr * 1.02, 5.0 / np.sqrt(n))

        # Symmetrize periodically
        if it % 500 == 0:
            a_sym = (a + a[::-1]) / 2.0
            a_sym = a_sym / a_sym.sum()
            obj_sym = compute_obj(a_sym)
            if obj_sym < best_obj - 1e-14:
                best_obj = obj_sym
                best_a = a_sym.copy()
                a = a_sym.copy()

    return best_a


def lp_polish(a, time_budget=200.0, seed=42, verbose=False):
    """
    LP-based polishing with adaptive backtracking step size.
    Uses simulated-annealing style: start with large steps, shrink over time.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    no_improve = 0
    last_t = 0.5
    T = 1.0  # "temperature" for step size exploration

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Temperature decays over time (simulated annealing style)
        progress = min(1.0, elapsed / (time_budget * 0.8))
        T = 1.0 * (0.3) ** (progress ** 0.5)

        # Try step sizes: full step, then backtracking
        improved = False
        # Main candidate: blend with temperature-adjusted step
        t_candidates = [min(1.0, last_t * 1.5), last_t, last_t * 0.5,
                        last_t * 0.25, 0.1, 0.05]
        # Add temperature: sometimes try a random larger step
        if rng.random() < T:
            t_candidates = [1.0] + t_candidates

        for t_try in t_candidates:
            t_try = max(0.01, min(1.0, t_try))
            a_try = (1.0 - t_try) * best_a + t_try * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = best_a.copy()
                last_t = t_try
                no_improve = 0
                improved = True
                break

        iter_count += 1

        if not improved:
            no_improve += 1
            if no_improve == 2:
                # Symmetrization attempt
                a_sym = (best_a + best_a[::-1]) / 2.0
                a_sym = a_sym / a_sym.sum()
                obj_sym = compute_obj(a_sym)
                if obj_sym < best_obj - 1e-14:
                    best_obj = obj_sym
                    best_a = a_sym.copy()
                    a = best_a.copy()
                    no_improve = 0
                    if verbose:
                        print(f"    LP it{iter_count}: sym obj={best_obj:.8f}")
            elif no_improve >= 5:
                # Perturb with adaptive scale
                scale = 0.004 * (1 + 0.3 * no_improve)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                if iter_count > 80:
                    break

        if verbose and iter_count % 20 == 0:
            print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={last_t:.3f}, M={M_lp:.6f}, T={T:.3f}")

    return best_a


def peak_directed_perturb(a, rng, K=5, alpha=0.05):
    """
    Perturb a by moving mass from high-convolution-contribution coordinates
    to low-contribution coordinates, targeting the peak locations.
    """
    n = len(a)
    a = a.copy()
    conv = compute_conv(a)

    top_peaks = np.argsort(conv)[::-1][:K]

    for k in top_peaks:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue

        j_range = np.arange(j_start, j_end + 1)
        contribs = a[j_range] * a[k - j_range]
        total_contrib = float(np.sum(contribs))
        if total_contrib < 1e-15:
            continue

        # Find the coordinate with max contribution
        j_max = j_range[np.argmax(contribs)]

        # Find a low-contribution coordinate to receive mass
        # Choose from coordinates that don't contribute to any top peak
        peak_j_set = set()
        for kp in top_peaks:
            kp = int(kp)
            js = max(0, kp - n + 1)
            je = min(n - 1, kp)
            peak_j_set.update(range(js, je + 1))

        available = [j for j in range(n) if j not in peak_j_set and j != j_max]
        if not available:
            available = [j for j in range(n) if j != j_max]
        if not available:
            continue

        # Weight by inverse contribution
        other_idx = available[rng.integers(0, len(available))]

        move = alpha * a[j_max]
        a[j_max] -= move
        a[other_idx] += move

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    return a / s


def kernel_init(n, rng, mode=0):
    """Generate diverse initial sequences using various kernels."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        return a / a.sum()
    elif mode == 2:
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 8)
        x = np.arange(n).astype(np.float64)
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x ** 2, 0.0))
        return a / (a.sum() + 1e-15)
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for _ in range(rng.integers(2, 6)):
            f = rng.integers(1, 10)
            p = rng.uniform(0, 2 * np.pi)
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.05)
        return a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        return a / a.sum()
    elif mode == 6:
        # Bell-shaped with noise
        x = np.arange(n).astype(np.float64)
        c = n / 2
        w = n / 4
        a = np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a += 0.1 * rng.standard_normal(n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def symmetrize(a):
    """Symmetrize: (a + flip(a)) / 2."""
    return (a + a[::-1]) / 2.0


def spectral_rebalance(a, tau=50.0):
    """
    Rebalance a in the spectral domain to flatten |A|^2.
    This helps make the convolution more uniform.
    """
    n = len(a)
    A = fft(a, 2 * n)
    # Flatten the magnitude
    mag = np.abs(A)
    phase = np.angle(A)
    # Apply soft flattening
    target_mag = np.ones_like(mag)
    smoothed_mag = (1.0 - 0.3) * mag + 0.3 * target_mag.astype(np.float64)
    A_new = smoothed_mag * np.exp(1j * phase)
    a_new = ifft(A_new).real[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a
    return a_new / s

        a_try = a_try / a_try.sum()

        # Nesterov warmup
        alloc1 = min(remaining * 0.1, 25.0)
        if alloc1 > 10:
            a_try = nesterov_descent(a_try, time_budget=alloc1, tau_init=10.0, tau_end=300.0)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 80.0)
        if alloc2 > 20:
            a_try = lp_polish(a_try, time_budget=alloc2, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Random restarts + LP ===
    print(f"\n=== Phase 6: Random Restarts ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        scale = 0.004 + 0.005 * r
        a_r = best_seq * np.exp(rng.normal(0, scale, n_final))
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()
        a_r = symmetrize(a_r)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.18, 60.0)
        if alloc > 15:
            a_r = lp_polish(a_r, time_budget=alloc, seed=600 + r)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (scale={scale:.4f}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive LP ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.6, 180.0)
        a_final = lp_polish(best_seq, time_budget=alloc, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # Quick final Nesterov with high tau
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 40.0)
        a_final2 = nesterov_descent(best_seq, time_budget=alloc, tau_init=50.0, tau_end=1000.0)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final NSD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
