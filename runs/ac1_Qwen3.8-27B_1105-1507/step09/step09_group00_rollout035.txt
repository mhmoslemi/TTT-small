"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key approach: Fast LP-based iterative polishing with line search, random restarts,
and FFT-based gradient descent for final polishing. Uses warm-start LP and adaptive
step sizes to converge quickly to high-precision solutions.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_lp_matrix(a):
    """Build the convolution matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def solve_lp(a, a_norm=None):
    """Solve LP: min M s.t. conv(a,g) <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    if a_norm is None:
        s = float(np.sum(a))
        a_norm = a / s if s > 0 else a
    
    A = build_lp_matrix(a_norm)
    
    # Variables: [M, g_0, ..., g_{n-1}]
    c = np.zeros(n + 1)
    c[0] = 1.0
    
    # Inequality: A @ g - M <= 0 → [-1, A] @ [M; g] <= 0
    A_ub = np.hstack([-np.ones((2 * n - 1, 1)), A])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        M_opt = float(res.x[0])
        g_opt = np.maximum(res.x[1:], 0)
        s = float(np.sum(g_opt))
        if s > 1e-10:
            g_opt = g_opt / s
        return g_opt, M_opt
    return None, None

        return a

    best_t = 0.3

    t_candidates = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 1.0]
