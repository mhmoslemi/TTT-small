def compute_conv(a):
    n = len(a)

**Peak Shaving Algorithm:**
1. Compute conv = a * a
2. Find the top K positions in conv (the "peaks")
3. For each peak position k, the contribution is `sum_j a_j * a_{k-j}`
4. To reduce conv[k], we want to decrease some `a_j * a_{k-j}` products

Actually, a better approach: Use a quadratic programming subproblem.

Given the current `a`, want to find a small perturbation `d` such that:
- `a + d >= 0`
- `sum(d) = 0` (preserve normalization)
- `conv(a+d)[k] <= M` for all k (or at least for the top peaks)
- Minimize `||d||^2` (small change)

The constraint `conv(a+d)[k] <= M` can be linearized:
`conv(a)[k] + 2 * (a * d)[k] <= M` (first-order approximation)

This gives a QP:
- Minimize `||d||^2`
- Subject to: `sum(d) = 0`, `a + d >= 0`, `2*(a*d)[k] <= M - conv(a)[k]` for peak positions k

This is a quadratic program that can be solved efficiently!

Let me implement this:

```python
def peak_shave_qp(a, M, top_k=10):
    """Solve QP to find smallest perturbation d that brings peaks below M."""
    n = len(a)
    conv = compute_conv(a)
    
    # Find the top positions
    peak_positions = np.argsort(conv)[::-1][:top_k]
    
    # Build the constraint matrix for (a*d)[k] <= (M - conv[k])/2
    # (a*d)[k] = sum_j a_j * d_{k-j}
    # So the row for position k has a_j at position (k-j) for valid j
    
    # ... this gets complex, let me use cvxpy
    import cvxpy as cp
    
    d = cp.Variable(n)
    constraints = [cp.sum(d) == 0, a + d >= 0]
    
    for k in peak_positions:
        k = int(k)
        # (a*d)[k] = sum_j a_j * d_{k-j} where 0 <= j < n and 0 <= k-j < n
        # = sum_j a_j * d_{k-j} for j in [max(0, k-n+1), min(k, n-1)]
        terms = None
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                if terms is None:
                    terms = a[j] * d[idx]
                else:
                    terms = terms + a[j] * d[idx]
        if terms is not None:
            constraints.append(terms <= (M - conv[k]) / 2)
    
    prob = cp.Problem(cp.Minimize(cp.sum_squares(d)), constraints)
    prob.solve()
    
    if d.value is not None:
        return a + d.value
    return a
```

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_gradient(a, tau):
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv

def solve_lp_given_a(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns (g, M, binding_k)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    conv_ag = A @ g_val
                    binding = np.where(conv_ag >= M.value - 1e-10)[0]
                    return g_val, float(M.value), binding
        except Exception:
            continue
    return None, None, None


def bcd_step(a, max_iterations=200, time_limit=100.0, verbose=False):
    """Block coordinate descent alternating LP solves. Returns (best_a, best_obj, iters)."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf, 0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    no_improve = 0
    it = 0

    while it < max_iterations and (time.monotonic() - start) < time_limit:
        # Step 1: Given a, find optimal g
        g, M_g, binding_g = solve_lp_given_a(a)
        if g is None:
            break

        # Step 2: Given g, find optimal a'
        a_new, M_a, binding_a = solve_lp_given_a(g)
        if a_new is None:
            break

        it += 1
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            if verbose and it % 20 == 0:
                print(f"      BCD it{it}: obj={best_obj:.8f}, M_a={M_a:.6f}, binding={len(binding_a)}")
        else:
            no_improve += 1
            # Try a blend
            for t in [0.3, 0.5, 0.7]:

                blend = np.maximum(blend, 0)

                        best_a = blend.copy()
                        a = blend
                        no_improve = 0
                        break

                break

    return best_a, best_obj, it
