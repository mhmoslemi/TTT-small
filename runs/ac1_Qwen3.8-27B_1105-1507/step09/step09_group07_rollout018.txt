"""
Optimized search for non-negative step functions minimizing peak autocorrelation.
Uses: (1) Fast Frank-Wolfe inner solver, (2) Log-sum-exp smoothing, 
(3) Adaptive length sweep, (4) Diverse restarts with anti-palindromic structure.

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_cross(a, b):
    """Cross-convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u * np.arange(1, n + 1) < cssv)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def fw_direction(a, max_iter=500, time_limit=3.0):
    """
    Frank-Wolfe to find direction g that minimizes max_k (a*g)[k]
    over the simplex. Returns (g, best_val).
    """
    n = len(a)
    a = a / np.sum(a)  # normalize
    g = np.ones(n) / n
    best_g = g.copy()
    best_val = np.inf
    start = time.monotonic()
    
    for it in range(max_iter):
        if time.monotonic() - start > time_limit:
            break
        
        # Cross-convolution a * g
        s = compute_cross(a, g)
        val = float(np.max(s))
        
        if val < best_val:
            best_val = val
            best_g = g.copy()
        
        # Subgradient w.r.t. g: grad[j] = a[k*-j] where k* = argmax
        k_star = int(np.argmax(s))
        # grad[j] = a[k_star - j] if 0 <= k_star - j < n else 0
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        grad = np.zeros(n)
        grad[lo:hi+1] = a[k_star - hi:k_star - lo + 1][::-1]
        
        # Frank-Wolfe step
        j_star = int(np.argmin(grad))
        t = 2.0 / (it + 2)
        g = (1 - t) * g
        g[j_star] += t
    
    s = float(np.sum(g))
    if s > 1e-10:
        g = g / s
    return g, best_val

def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g)."""
    n = len(a)
    # A has shape (2n-1, n)
    # A[k, j] = a[k-j] if 0 <= k-j < n else 0
    # i.e., A[k, j] = a[k-j] for max(0, k-n+1) <= j <= min(n-1, k)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # a is placed at positions j, j+1, ..., j+n-1 in the convolution
        A[j:j+n, j] = a
    return A

def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.MOSEK]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None

def line_search_blend(a, g, best_obj):
    """Find optimal blend parameter t for a_new = (1-t)*a + t*g."""
    n = len(a)
    best_t = 0.0
    best_local_obj = best_obj

            obj = compute_obj(a_try)

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
