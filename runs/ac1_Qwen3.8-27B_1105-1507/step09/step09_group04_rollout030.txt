def project_simplex(v):

The function is h(delta) = max_k (conv[k] + delta * dir[k]) + delta^2 (for k=2k, i.e., add delta^2 to the k=2k element).

2. The upper envelope changes at the intersections of consecutive lines (in slope order)
3. The minimum is at the intersection where the slope goes from negative to positive

To find this efficiently:

2. This is still O(n^2) in the worst case

"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import scipy.sparse as sp

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_sparse(a_norm):
    """Build sparse matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        lo_k = j
        hi_k = min(j + n - 1, 2 * n - 2)
        length = hi_k - lo_k + 1
        rows.extend(range(lo_k, hi_k + 1))
        cols.extend([j] * length)
        vals.extend(a_norm[:length].tolist())
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_standard(a):
    """Solve LP: min M s.t. conv(a_norm, g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix_sparse(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_l2(a, weight=1.0):
    """Solve QP: min weight*M + 0.5*||conv(a_norm,g)||_2^2 s.t. sum(g)=1, g>=0, conv<=M."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix_sparse(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    conv_expr = A @ g
    constraints = [conv_expr <= M, cp.sum(g) == 1]
    objective = cp.Minimize(weight * M + 0.5 * cp.sum_squares(conv_expr))
    prob = cp.Problem(objective, constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def subgradient_nesterov(a, time_budget, rng, verbose=False):
    """
    Nesterov accelerated subgradient descent with averaging.
    Uses the structure: subgradient of max_k (a*a)[k] is 2*P_k a where k=argmax.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a /= float(np.sum(a))

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    v = np.zeros(n)
    beta = 0.5
    lr = 2.0 / np.sqrt(n)
    avg = a.copy()
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        it += 1

        # Compute convolution and find active lag
        conv = compute_conv(y)
        max_idx = int(np.argmax(conv))
        max_val = float(conv[max_idx])

        # Subgradient: grad[m] = 2 * (y[max_idx - m] + y[m - max_idx]) with bounds
        grad = np.zeros(n)
        for m in range(n):
            g = 0.0
            idx1 = max_idx - m
            if 0 <= idx1 < n:
                g += y[idx1]
            idx2 = m - max_idx
            if 0 <= idx2 < n:
                g += y[idx2]
            grad[m] = 2.0 * g

        # Nesterov update
        beta = 2.0 / (it + 2.0)
        y_new = y + beta * v
        y_new = np.maximum(y_new, 0)
        s_y = float(np.sum(y_new))
        if s_y < 1e-10:
            y_new = y.copy()
        else:
            y_new /= s_y

        v = beta * v - lr * grad
        a_new = y_new + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = y_new.copy()
        else:
            a_new /= s_new

        # Running average
        avg = (it * avg + a_new) / (it + 1)

        obj_new = compute_obj(a_new)
        obj_avg = compute_obj(avg)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        if obj_avg < best_obj:
            best_obj = obj_avg
            best_a = avg.copy()
            best_it = it

        y = y_new
        lr = max(1e-6, lr * 0.999)

        # Restart if stuck
        if it - best_it > 5000:
            y = best_a.copy()
            v = np.zeros(n)
            avg = best_a.copy()
            lr = 2.0 / np.sqrt(n) * 0.5
            beta = 0.5
            best_it = it

        if verbose and it % 5000 == 0:
            print(f"    NS it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")

    return best_a


def simulated_annealing(a, time_budget, rng, verbose=False):
    """
    Simulated annealing with adaptive temperature.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a /= float(np.sum(a))

    best_a = a.copy()
    best_obj = compute_obj(a)
    cur_a = a.copy()
    cur_obj = best_obj
    start = time.monotonic()

    T_init = 0.05
    T_end = 0.001
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        T = T_init * (T_end / T_init) ** progress

        # Generate perturbation
        n_perturb = max(1, n // (20 + int(50 * progress)))
        indices = rng.integers(0, n, size=n_perturb)
        cur_a[indices] *= np.exp(rng.normal(0, T, n_perturb))
        cur_a = np.maximum(cur_a, 0)
        s = float(np.sum(cur_a))
        if s < 1e-10:
            cur_a = best_a.copy()
            continue
        cur_a /= s

        new_obj = compute_obj(cur_a)
        delta = new_obj - cur_obj

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = cur_a.copy()

        it += 1
        if verbose and it % 10000 == 0:
            print(f"    SA it{it}: best={best_obj:.8f}, T={T:.5f}")

        # Periodic reset to best
        if it % 20000 == 0:
            cur_a = best_a.copy()
            cur_obj = best_obj

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** rng.uniform(0.5, 3.0)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 5:
        # Two-bump
        c1 = rng.integers(0, n // 3)
        c2 = rng.integers(2 * n // 3, n)
        w = max(1, n // rng.integers(8, 16))
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 6:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 1)
        a = a / a.sum()
    elif mode == 7:
        # Cosine window
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 8:
        # Sinc-like
        x = np.linspace(-3, 3, n)
        a = np.sinc(x / (n * 0.02)) ** 2
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    else:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining LP polishing, coordinate descent, subgradient,
    simulated annealing, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_target = 800
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Intensive LP Polishing (most effective per second) ===
    print(f"\n=== Phase 1: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.40, 550.0)
    if p1_budget > 30:
        a_lp = lp_polish_enhanced(best_seq, p1_budget, rng, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp
            print(f"  LP Phase1: obj={best_score:.8f} **")
        else:
            print(f"  LP Phase1: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: Coordinate Descent Refinement ===
    print(f"\n=== Phase 2: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.20, 250.0)
    if p2_budget > 20:
        a_cd, _ = coord_descent_fast(best_seq, p2_budget, rng, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd
            print(f"  CD Phase2: obj={best_score:.8f} **")
        else:
            print(f"  CD Phase2: obj={obj_cd:.8f}")

    # === Phase 3: LP + CD Alternating ===
    print(f"\n=== Phase 3: LP+CD Alternating ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.20, 250.0)
    if p3_budget > 30:
        n_rounds = 3
        round_budget = p3_budget / n_rounds
        for rnd in range(n_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(round_budget, remaining * 0.7)
            if alloc < 20:
                break

            # LP step
            a_lp = lp_polish_enhanced(best_seq, alloc * 0.5, rng)
            obj = compute_obj(a_lp)
            if obj < best_score:
                best_score = obj
                best_seq = a_lp

            # CD step
            remaining = deadline - time.monotonic()
            alloc_cd = min(alloc * 0.4, remaining * 0.5)
            if alloc_cd > 15:
                a_cd, _ = coord_descent_fast(best_seq, alloc_cd, rng)
                obj = compute_obj(a_cd)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_cd
                    print(f"  Alt rnd{rnd}: obj={best_score:.8f} **")

    # === Phase 4: Subgradient with Nesterov ===
    print(f"\n=== Phase 4: Subgradient Nesterov ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.10, 120.0)
    if p4_budget > 20:
        a_sub = subgradient_nesterov(best_seq, p4_budget, rng, verbose=True)
        obj_sub = compute_obj(a_sub)
        if obj_sub < best_score:
            best_score = obj_sub
            best_seq = a_sub
            print(f"  Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj_sub:.8f}")

    # === Phase 5: Simulated Annealing ===
    print(f"\n=== Phase 5: Simulated Annealing ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.08, 100.0)
    if p5_budget > 20:
        a_sa = simulated_annealing(best_seq, p5_budget, rng, verbose=True)
        obj_sa = compute_obj(a_sa)
        if obj_sa < best_score:
            best_score = obj_sa
            best_seq = a_sa
            print(f"  SA: obj={best_score:.8f} **")
        else:
            print(f"  SA: obj={obj_sa:.8f}")

    # === Phase 6: Multi-length Exploration ===
    print(f"\n=== Phase 6: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 100),
        n_current + 100,
        n_current + 200,
        600, 900, 1100, 1300,
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500 and abs(n - n_current) > 50]

    for n_try in n_options[:3]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        # CD
        alloc1 = min(remaining * 0.30, 60.0)
        if alloc1 > 15:
            a_try, _ = coord_descent_fast(a_try, alloc1, rng)

        # LP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.30, 50.0)
        if alloc2 > 15:
            a_try = lp_polish_enhanced(a_try, alloc2, rng)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse Restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        mode = (r + 2) % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r /= a_r.sum()

        alloc1 = min(remaining * 0.40, 60.0)
        if alloc1 > 15:
            a_r, _ = coord_descent_fast(a_r, alloc1, rng)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.30, 40.0)
        if alloc2 > 15:
            a_r = lp_polish_enhanced(a_r, alloc2, rng)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Final Intensive Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # CD
    if remaining > 30:
        alloc = min(remaining * 0.30, 80.0)
        a_cd_f, _ = coord_descent_fast(best_seq, alloc, rng, verbose=True)
        obj = compute_obj(a_cd_f)
        if obj < best_score:
            best_score = obj
            best_seq = a_cd_f
            print(f"  Final CD: obj={best_score:.8f} **")

    # LP
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.40, 70.0)
        a_lp_f = lp_polish_enhanced(best_seq, alloc, rng, verbose=True)
        obj = compute_obj(a_lp_f)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp_f
            print(f"  Final LP: obj={best_score:.8f} **")

    # One more CD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.50, 50.0)
        a_cd_f2, _ = coord_descent_fast(best_seq, alloc, rng)
        obj = compute_obj(a_cd_f2)
        if obj < best_score:
            best_score = obj
            best_seq = a_cd_f2
            print(f"  Final CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
