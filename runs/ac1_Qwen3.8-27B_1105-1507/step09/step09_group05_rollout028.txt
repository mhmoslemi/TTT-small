import numpy as np

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, solver_list=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal, active_set) or (None, None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Find active set
                    conv_ag = compute_conv(a_norm * g_val)  # Wait, this is conv(a*g, ...) no
                    # Actually we need A @ g_val to get conv(a, g_val)
                    conv_vals = A @ g_val
                    M_val = float(M.value)
                    tol = 1e-6 * max(1.0, abs(M_val))
                    active = np.where(conv_vals >= M_val - tol)[0]
                    return g_val, M_val, active
        except Exception:
            continue
    return None, None, None


def lp_polish(a, max_iter=50, time_limit=60.0, verbose=False):
    """
    LP polishing: repeatedly solve LP for optimal g given a, then blend.
    Returns (best_a, best_obj, M_lp_history).

    if s < 1e-10:
        return a, np.inf, []
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    M_history = []
    no_improve = 0
    last_t = 0.5
    start = time.monotonic()
    it = 0

    while it < max_iter and (time.monotonic() - start) < time_limit:
        g_lp, M_lp, active_set = solve_lp(a)
        if g_lp is None:
            break

        M_history.append(M_lp)
        
        # Try multiple blend factors

            t_candidates.append(max(0.05, last_t * 0.7))
            t_candidates.append(min(1.0, last_t * 1.3))

        best_t_obj = np.inf
        best_t_val = last_t
        best_a_local = best_a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:

            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            no_improve = 0

            if no_improve >= 4:

def deflation_perturb(a, conv, scale=0.03):

        k = int(k)
        for j in range(n):
            idx1 = k - j
            idx2 = j
            if 0 <= idx1 < n:
                deltas[j] += a[idx1]
            # For the other direction in convolution
    # Actually, d/d a[j] of (a*a)[k] = a[k-j] if 0<=k-j<n, plus a[j-k] if 0<=j-k<n
    # But since a*a is symmetric, d/d a[j] of (a*a)[k] = a[k-j] + (a[j-k] if j-k != k-j)
    # Wait, (a*a)[k] = sum_j a[j]*a[k-j], so d/d a[j] = a[k-j] (if 0<=k-j<n)

    # And if k = 2j, there's an extra a[j] term (from the j factor)
    # Actually for the product rule: d/d a[j] (a[j] * a[k-j]) = a[k-j] + a[j] * d(a[k-j])/d(a[j])
    # But a[k-j] only depends on a[j] if k-j = j, i.e., k = 2j
    # So for general j: d/d a[j] (a*a)[k] = a[k-j] (if 0 <= k-j < n)
    # Plus a[j] if k = 2j (the cross term)
    
    # Simpler: just use the gradient from the smoothed version
    # For deflation, we want to decrease a[j] where it most contributes to the max

    deltas = np.zeros(n)

def smooth_perturb(a, rng, scale=0.03, method=0):
    """Apply a smooth perturbation."""
    n = len(a)
    a = a.copy()
    if method == 0:
        # Multiplicative noise
        a = a * np.exp(rng.normal(0, scale, n))
    elif method == 1:
        # Additive noise with projection
        noise = rng.normal(0, scale, n)
        a = a + noise
        a = np.maximum(a, 0)
    elif method == 2:
        # Low-pass filtered noise
        noise = rng.normal(0, scale, n)
        A = fft(noise, 2 * n)
        cutoff = max(1, n // 4)
        A[cutoff:-cutoff] = 0
        noise_filtered = ifft(A).real[:n]
        a = a + noise_filtered
        a = np.maximum(a, 0)
    elif method == 3:
        # Random coordinate scaling
        mask = rng.random(n) < 0.3
        a[mask] *= rng.uniform(0.5, 1.5, mask.sum())
    elif method == 4:
        # Mirror + blend
        a2 = a[::-1].copy()
        t = rng.uniform(0.2, 0.5)
        a = (1 - t) * a + t * a2
    else:
        a = a * np.exp(rng.normal(0, scale, n))
    
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_solutions(a1, a2, t, n):
    """Interpolate between two solutions of potentially different lengths."""
    if len(a1) != len(a2):
        a2 = resample_sequence(a2, len(a1))
    result = (1 - t) * a1 + t * a2
    result = np.maximum(result, 0)
    s = float(np.sum(result))
    if s > 1e-10:
        result = result / s
    return result


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)

    elif mode == 3:

    elif mode == 4:
        # Exponential decay from center

        a = a / a.sum()
    elif mode == 5:

        # Smooth
        kernel = np.exp(-np.arange(-5, 6)**2 / 4)
        a = np.convolve(a, kernel, 'same')
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 6:

    else:

def smoothed_gradient_descent(a, time_budget=60.0, tau_init=5.0, tau_end=500.0):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient via FFT
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_fft = fft(y, N)
        g_full = ifft(P * np.conj(A_fft)).real
        g = 2.0 * g_full[:n]

        # Barzilai-Borwein step size
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.3 * lr + 0.7 * min(max(bb_lr, 1e-8), 20.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9995)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 5.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a

Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal, active_indices) or (None, None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    conv_vals = A @ g_val
                    tol = max(1e-8, 1e-6 * abs(M_val))
                    active = np.where(conv_vals >= M_val - tol)[0]
                    return g_val, M_val, active
        except Exception:
            continue
    return None, None, None

        current_max = float(np.max(compute_conv(a)))

        best_t = last_t

        # Fine search around best coarse t
        if 0.05 < best_t < 0.95 and (time.monotonic() - start) < time_limit - 2:
            lo = max(0.0, best_t - 0.15)
            hi = min(1.0, best_t + 0.15)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_lo, a_hi = lo, hi
            for _ in range(8):
                t1 = a_hi - gr * (a_hi - a_lo)
                t2 = a_lo + gr * (a_hi - a_lo)
                a1 = np.maximum((1.0 - t1) * a + t1 * g_lp, 0)
                a2 = np.maximum((1.0 - t2) * a + t2 * g_lp, 0)
                s1, s2 = float(np.sum(a1)), float(np.sum(a2))
                obj1 = compute_obj(a1 / s1) if s1 > 1e-10 else np.inf
                obj2 = compute_obj(a2 / s2) if s2 > 1e-10 else np.inf
                if obj1 < obj2:
                    a_hi = t2
                    if obj1 < best_t_obj:
                        best_t_obj = obj1
                        best_t = t1
                else:
                    a_lo = t1
                    if obj2 < best_t_obj:
                        best_t_obj = obj2
                        best_t = t2

        a_new = np.maximum((1.0 - best_t) * a + best_t * g_lp, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new

        it += 1
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = a_new.copy()
            a = best_a
            last_t = best_t
            no_improve = 0
            if verbose and it % 5 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, gap={dual_gap:.6f}")
        else:
            no_improve += 1
            if no_improve >= 4:
                break

    return best_a, best_obj, final_gap


def deflation_perturb(a, conv, scale=0.03):
    """
    Deflation: reduce a where it contributes most to peak convolution values.
    Uses the gradient structure of the max convolution.
    """
    n = len(a)
    a = a.copy()
    max_conv = float(np.max(conv))
    if max_conv < 1e-15:
        return a

    # Find near-peak positions
    threshold = max_conv - 0.15 * max_conv
    peak_positions = np.where(conv >= threshold)[0]

    # Compute negative gradient at peak positions
    deltas = np.zeros(n)
    for k in peak_positions:
        k = int(k)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                deltas[j] += a[idx]

    max_d = float(np.max(np.abs(deltas)))
    if max_d > 1e-15:
        a = a - scale * deltas / max_d
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s
    return a


def smooth_perturb(a, rng, scale=0.03, method=0):
    """Apply various smooth perturbations."""
    n = len(a)
    a = a.copy()
    if method == 0:
        a = a * np.exp(rng.normal(0, scale, n))
    elif method == 1:
        noise = rng.normal(0, scale, n)
        a = a + noise
        a = np.maximum(a, 0)
    elif method == 2:
        # Low-pass filtered noise (preserves smooth structure)
        noise = rng.normal(0, scale * 2, n)
        N = 1
        while N < 2 * n:
            N <<= 1
        A_noise = fft(noise, N)
        cutoff = max(2, n // 5)
        A_noise[cutoff:-cutoff] = 0
        noise_filtered = ifft(A_noise).real[:n]
        a = a + noise_filtered
        a = np.maximum(a, 0)
    elif method == 3:
        # Sparse coordinate modification
        n_mod = max(1, n // 10)
        idx = rng.choice(n, size=n_mod, replace=False)
        a[idx] *= rng.uniform(0.4, 1.6, n_mod)
        a = np.maximum(a, 0)
    elif method == 4:
        # Mirror blend
        a2 = a[::-1].copy()
        t = rng.uniform(0.1, 0.4)
        a = (1 - t) * a + t * a2
    elif method == 5:
        # Phase perturbation in frequency domain
        N = 1
        while N < 2 * n:
            N <<= 1
        A = fft(a, N)
        phase_noise = np.exp(1j * rng.normal(0, scale * 0.5, N))
        phase_noise[0] = 1.0
        if N % 2 == 0:
            phase_noise[N // 2] = 1.0
        A *= phase_noise
        a = ifft(A).real[:n]
        a = np.maximum(a, 0)
    else:
        a = a * np.exp(rng.normal(0, scale, n))

    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2 + 0.01
        a = a / a.sum()
    elif mode == 3:
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
        a = a / a.sum()
    elif mode == 4:
        center = n // 2
        a = np.exp(-np.abs(np.arange(n) - center) / (n / 4))
        a = a / a.sum()
    elif mode == 5:
        a = np.zeros(n)
        a[n // 4] = 1.0
        a[3 * n // 4] = 1.0
        kernel = np.exp(-np.arange(-5, 6) ** 2 / 4.0)
        a = np.convolve(a, kernel, 'same')
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 6:
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)


def smoothed_gradient_descent(a, time_budget=60.0, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed max-convolution."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(1e-10, time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        if verbose and it % 5000 == 0:
            print(f"      SGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_fft = fft(y, N)
        g_full = ifft(P * np.conj(A_fft)).real
        g = 2.0 * g_full[:n]

        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.3 * lr + 0.7 * min(max(bb_lr, 1e-8), 25.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9995)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 5.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Adaptive LP polishing with dual-gap tracking, deflation,
    multi-length exploration, population interpolation, and high-precision tuning.
    Returns list of non-negative floats minimizing the evaluation objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 12.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # Store good solutions for interpolation
    good_solutions = [(best_seq.copy(), best_score)]

    # === Phase 1: Initial intensive LP polish ===
    print(f"\n=== Phase 1: Initial LP Polish ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.22, 220.0)
    if p1_budget > 30:
        a_lp, obj_lp, gap = lp_polish(best_seq, max_iter=100, time_limit=p1_budget, verbose=True)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            good_solutions.append((a_lp.copy(), obj_lp))
            print(f"  Phase1 LP: obj={best_score:.8f}, gap={gap:.6f} **")
        else:
            print(f"  Phase1 LP: obj={obj_lp:.8f}, gap={gap:.6f}")

    # === Phase 2: Smoothed GD + LP ===
    print(f"\n=== Phase 2: Smoothed GD + LP ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.12, 100.0)
    if p2_budget > 25:
        a_sgd = smoothed_gradient_descent(best_seq, time_budget=p2_budget * 0.5,
                                          tau_init=5.0, tau_end=300.0)
        remaining = deadline - time.monotonic()
        lp_alloc = min(p2_budget * 0.5, remaining * 0.4)
        if lp_alloc > 10:
            a_sgd, obj_sgd, _ = lp_polish(a_sgd, max_iter=60, time_limit=lp_alloc)
        else:
            obj_sgd = compute_obj(a_sgd)

        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            good_solutions.append((a_sgd.copy(), obj_sgd))
            print(f"  Phase2: obj={best_score:.8f} **")
        else:
            print(f"  Phase2: obj={obj_sgd:.8f}")

    # === Phase 3: Deflation + Perturbation + LP ===
    print(f"\n=== Phase 3: Deflation & Perturbation ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.22, 220.0)
    if p3_budget > 30:
        a_cur = best_seq.copy()
        for d_iter in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break

            conv = compute_conv(a_cur)

            if d_iter % 3 == 0:
                # Deflation
                scale = 0.02 + 0.01 * (d_iter // 3)
                a_try = deflation_perturb(a_cur, conv, scale=scale)
            elif d_iter % 3 == 1:
                # Smooth perturbation
                a_try = smooth_perturb(a_cur, rng, scale=0.03, method=d_iter % 6)
            else:
                # Sparse modification
                a_try = smooth_perturb(a_cur, rng, scale=0.04, method=3)

            alloc = min(p3_budget / 8.0, remaining * 0.45)
            if alloc < 10:
                break

            a_new, obj_new, _ = lp_polish(a_try, max_iter=60, time_limit=alloc)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a_cur = a_new.copy()
                good_solutions.append((a_new.copy(), obj_new))
                print(f"  Def/Pert it{d_iter}: obj={best_score:.8f} **")
            else:
                print(f"  Def/Pert it{d_iter}: obj={obj_new:.8f}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length Search ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(300, n_best - 100), max(300, n_best - 50),
        n_best + 50, n_best + 100, n_best + 200, n_best + 400,
        500, 600, 700, 750, 800, 850, 900, 1000, 1100, 1200, 1400
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1800 and n != n_best]
    n_options.sort(key=lambda x: abs(x - n_best))

    for n_try in n_options[:7]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick GD
        alloc1 = min(remaining * 0.06, 25.0)
        if alloc1 > 8:
            a_try = smoothed_gradient_descent(a_try, time_budget=alloc1,
                                              tau_init=10.0, tau_end=200.0)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.1, 50.0)
        if alloc2 > 8:
            a_try, obj_try, _ = lp_polish(a_try, max_iter=60, time_limit=alloc2)
        else:
            obj_try = compute_obj(a_try)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            good_solutions.append((a_try.copy(), obj_try))
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Population interpolation search ===
    print(f"\n=== Phase 5: Population Interpolation ===")
    remaining = deadline - time.monotonic()

    # Generate diverse candidates and interpolate
    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = (r * 2 + 1) % 7
        n_cur = len(best_seq)
        a_r = diverse_init(n_cur, rng, mode=mode)
        blend = 0.12 + 0.04 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.14, 55.0)
        if alloc > 20:
            a_r = smoothed_gradient_descent(a_r, time_budget=alloc * 0.35,
                                            tau_init=10.0, tau_end=200.0)
            remaining = deadline - time.monotonic()
            lp_alloc = min(alloc * 0.65, remaining * 0.35)
            if lp_alloc > 8:
                a_r, obj_r, _ = lp_polish(a_r, max_iter=50, time_limit=lp_alloc)
            else:
                obj_r = compute_obj(a_r)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                good_solutions.append((a_r.copy(), obj_r))
                print(f"  Pop {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Pop {r} (mode={mode}): obj={obj_r:.8f}")

        # Interpolation with another good solution
        if len(good_solutions) >= 2:
            remaining = deadline - time.monotonic()
            if remaining > 30:
                idx = min(r + 1, len(good_solutions) - 1)
                if idx != 0:
                    other_seq, other_score = good_solutions[idx]
                    for t_interp in [0.3, 0.5, 0.7]:
                        remaining = deadline - time.monotonic()
                        if remaining < 15:
                            break
                        a_interp = (1 - t_interp) * best_seq + t_interp * resample_sequence(other_seq, len(best_seq))
                        a_interp = np.maximum(a_interp, 0)
                        s_i = float(np.sum(a_interp))
                        if s_i < 1e-10:
                            continue
                        a_interp = a_interp / s_i
                        alloc_i = min(remaining * 0.12, 30.0)
                        if alloc_i > 8:
                            a_interp, obj_i, _ = lp_polish(a_interp, max_iter=40, time_limit=alloc_i)
                        else:
                            obj_i = compute_obj(a_interp)
                        if obj_i < best_score:
                            best_score = obj_i
                            best_seq = a_interp.copy()
                            good_solutions.append((a_interp.copy(), obj_i))
                            print(f"  Interp (t={t_interp}): obj={best_score:.8f} **")
                            break

    # === Phase 6: High-precision final polish ===
    print(f"\n=== Phase 6: High-Precision Final Polish ===")
    remaining = deadline - time.monotonic()

    # Very high tau smoothed GD (close to exact max)
    if remaining > 35:
        alloc = min(remaining * 0.2, 60.0)
        a_precise = smoothed_gradient_descent(best_seq, time_budget=alloc,
                                              tau_init=100.0, tau_end=2000.0)
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_precise, obj_precise, _ = lp_polish(a_precise, max_iter=60,
                                                   time_limit=min(remaining * 0.4, 40.0))
            if obj_precise < best_score:
                best_score = obj_precise
                best_seq = a_precise.copy()
                print(f"  High-prec GD+LP: obj={best_score:.8f} **")
            else:
                print(f"  High-prec GD+LP: obj={obj_precise:.8f}")

    # Multiple short LP rounds with varied perturbations
    for final_rnd in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        a_cur = best_seq.copy()

        if final_rnd == 1:
            conv = compute_conv(a_cur)
            a_cur = deflation_perturb(a_cur, conv, scale=0.01)
        elif final_rnd == 2:
            a_cur = smooth_perturb(a_cur, rng, scale=0.015, method=2)
        elif final_rnd == 3:
            a_cur = smooth_perturb(a_cur, rng, scale=0.02, method=0)
        elif final_rnd == 4:
            # Slightly different length
            new_n = max(300, len(a_cur) - 20)
            a_cur = resample_sequence(a_cur, new_n)
        elif final_rnd == 5:
            new_n = max(300, len(a_cur) + 20)
            a_cur = resample_sequence(a_cur, new_n)

        alloc = min(remaining * 0.18, 50.0)
        if alloc < 8:
            break

        a_final, obj_final, _ = lp_polish(a_cur, max_iter=80, time_limit=alloc)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP {final_rnd}: obj={best_score:.8f} **")
        else:
            print(f"  Final LP {final_rnd}: obj={obj_final:.8f}")

    # === Phase 7: Ultra-fine single-pass ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # One more LP pass with maximum iterations
        alloc = min(remaining * 0.5, 40.0)
        a_ultra, obj_ultra, _ = lp_polish(best_seq, max_iter=100, time_limit=alloc)
        if obj_ultra < best_score:
            best_score = obj_ultra
            best_seq = a_ultra.copy()
            print(f"  Ultra: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
