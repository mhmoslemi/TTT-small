```python
def solve_direction_lp(a, M_target=None):
    """
    Solve LP to find direction δ that minimizes the linearized peak.
    Linearized objective: conv(a,a)[k] + 2*conv(a,δ)[k] <= M
    Constraints: sum(δ) = 0, a+δ >= 0 (i.e., δ >= -a)
    """
    n = len(a)
    conv_aa = np.convolve(a, a, mode='full')  # length 2n-1
    
    # conv(a, δ)[k] = sum_i a[i] * δ[k-i] (valid indices)
    # This is linear in δ. Build the matrix.
    
    # For each k in 0..2n-2, conv(a,δ)[k] = sum_{i} a[i]*δ[k-i]
    # where 0 <= i < n and 0 <= k-i < n
    
    # Let's build the constraint matrix A where A[k, j] = coefficient of δ[j] in conv(a,δ)[k]
    # conv(a,δ)[k] = sum_j a[k-j] * δ[j] (where 0 <= k-j < n and 0 <= j < n)
    # So A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    A = np.zeros((2*n-1, n))
    for j in range(n):
        for k in range(2*n-1):
            i = k - j
            if 0 <= i < n:
                A[k, j] = a[i]
    
    # LP: min M
    # s.t. 2*A @ δ + conv_aa <= M  (for all k)
    #      sum(δ) = 0
    #      δ >= -a
    
    δ = cp.Variable(n)
    M = cp.Variable()
    
    constraints = [
        2 * A @ δ + conv_aa <= M,
        cp.sum(δ) == 0,
        δ >= -a
    ]
    
    if M_target is not None:
        constraints.append(M <= M_target)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    prob.solve(solver=cp.MOSEK)
    
    if δ.value is not None:
        return np.asarray(δ.value).flatten(), float(M.value)
    return None, None
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an LP direction-finding method
with line search. The LP finds the optimal first-order direction δ that
minimizes the linearized peak of conv(a+δ, a+δ). Multi-n exploration,
perturbation restarts, and adaptive step control are used to escape local
minima and explore the landscape broadly.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_cross_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ delta = conv(a, delta)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        A[k_start:k_end, j] = a[0:k_end - j]
    return A

        if delta_norm < 1e-10:
            no_improve += 2
        else:
            t_opt, obj_at_t = line_search(a, delta, max_t)

            it += 1

            if t_opt > 0 and obj_at_t < best_obj - 1e-13:
                a_new = a + t_opt * delta
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                a = a_new
                best_obj = obj_at_t
                best_a = a.copy()
                no_improve = 0

                # Adapt max_t based on found step
                if t_opt < 0.3:
                    max_t = max(0.3, max_t * 0.8)
                elif t_opt > 0.7:
                    max_t = min(1.0, max_t * 1.1)

                obj_history.append(best_obj)

                if verbose and it % 10 == 0:
                    print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.3f}, "
                          f"dnorm={delta_norm:.3e}, el={elapsed:.0f}s")
            else:
                no_improve += 1
        # End if/else for delta_norm

        if no_improve >= 3:
            # Perturb to escape
            scale = 0.015 * min(no_improve, 6)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
            max_t = 1.0

            if it > 80 and len(obj_history) > 30:
                recent = obj_history[-30:]
                if recent[0] - recent[-1] < 1e-10:
                    break
        elif no_improve >= 1:
            # Try smaller step
            if delta is not None and delta_norm > 1e-10:
                t_small = 0.1
                a_try = a + t_small * delta
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    no_improve = 0

    return best_a


def lp_descent_weighted(a, time_budget=300.0, verbose=False, seed=42):
    """
    Variant: LP with weighted constraints emphasizing current peaks more.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    no_improve = 0
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start

        conv_aa = compute_conv(a)
        peak = np.max(conv_aa)

        # Create weights: emphasize peaks near the maximum
        weights = np.ones(2 * n - 1)
        above_90 = conv_aa > 0.9 * peak
        weights[above_90] = 5.0
        above_99 = conv_aa > 0.99 * peak
        weights[above_99] = 20.0

        A = build_cross_matrix(a)

        delta = cp.Variable(n)
        M = cp.Variable()

        constraints = [
            weights * (2.0 * (A @ delta) + conv_aa) <= M,
            cp.sum(delta) == 0,
            delta >= -a - 1e-12
        ]

        prob = cp.Problem(cp.Minimize(M), constraints)

        solved = False
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    d = np.asarray(delta.value).flatten()
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            if it > 10:
                break
            a = best_a.copy()
            continue

        delta_norm = np.linalg.norm(d)
        if delta_norm < 1e-10:
            no_improve += 2
        else:
            # Line search
            best_t = 0.0
            best_t_obj = best_obj
            for t in [0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
                a_try = a + t * d
                if np.any(a_try < -1e-10):
                    continue
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                obj_try = compute_obj(a_try)
                if obj_try < best_t_obj - 1e-15:
                    best_t_obj = obj_try
                    best_t = t

            it += 1

            if best_t > 0 and best_t_obj < best_obj - 1e-13:
                a = a + best_t * d
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_obj = best_t_obj
                best_a = a.copy()
                no_improve = 0

                if verbose and it % 10 == 0:
                    print(f"    LPW it{it}: obj={best_obj:.8f}, t={best_t:.3f}, el={elapsed:.0f}s")
            else:
                no_improve += 1

        if no_improve >= 4:
            scale = 0.02 * min(no_improve, 5)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
            if it > 60:
                break

    return best_a


def subgradient_descent(a, time_budget=100.0, K=5, verbose=False):
    """Subgradient method targeting top-K peaks with momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        weights = np.exp((peak_vals - max_pv) * 5.0 / max_pv)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v

        # Project onto simplex
        u = np.sort(a_new)[::-1]
        cssv = np.cumsum(u)
        idx = np.arange(1, n + 1)
        cond = u * idx > (cssv - 1.0)
        rho_arr = np.where(cond)[0]
        if len(rho_arr) > 0:
            rho = rho_arr[-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_new - theta, 0.0)
        else:
            a_new = np.ones(n) / n

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-10:
                    break
            else:
                lr *= 0.995

        it += 1

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Arcsine distribution
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-8))
        a = a / a.sum()
    elif mode == 3:
        # Two bumps
        c1, c2 = n // 4, 3 * n // 4
        w = n // 8
        a = np.exp(-((np.arange(n) - c1)**2) / (2*w**2)) + np.exp(-((np.arange(n) - c2)**2) / (2*w**2))
        a = a / a.sum()
    elif mode == 4:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1.5, 1.5)
        a = a / a.sum()
    elif mode == 5:
        # Cosine window
        a = 0.5 + 0.5 * np.cos(np.pi * np.arange(n) / (n - 1 + 1e-10))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP direction method with line search as the primary optimizer,
    combined with multi-n exploration, weighted LP variants, subgradient
    refinement, and perturbation-based restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_a = None
    best_obj = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_a = parent.copy()
                best_obj = compute_obj(parent)
                print(f"Parent (n={parent.size}): obj={best_obj:.8f}")
    except Exception:
        pass

    if best_a is None:
        best_a = np.ones(800) / 800
        best_obj = compute_obj(best_a)
        print(f"Default init: n={len(best_a)}, obj={best_obj:.8f}")

    n_current = len(best_a)

    # === Phase 1: Primary LP descent on parent length ===
    print(f"\n=== Phase 1: LP Descent (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 250.0)
    if p1_budget > 30:
        a_p1 = lp_descent(best_a, time_budget=p1_budget, verbose=True, seed=42)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_obj:
            best_obj = obj_p1
            best_a = a_p1.copy()
            print(f"  Phase 1: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_p1:.8f}")

    # === Phase 2: Weighted LP variant ===
    print(f"\n=== Phase 2: Weighted LP ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 150.0)
    if p2_budget > 30:
        a_p2 = lp_descent_weighted(best_a, time_budget=p2_budget, verbose=True, seed=77)
        obj_p2 = compute_obj(a_p2)
        if obj_p2 < best_obj:
            best_obj = obj_p2
            best_a = a_p2.copy()
            print(f"  Phase 2: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 2: obj={obj_p2:.8f}")

    # === Phase 3: Multi-n exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_options = []
    for dn in [-200, -100, 100, 200, 400, 600, 800]:
        n_try = n_current + dn
        if 400 <= n_try <= 3000 and n_try != n_current:
            n_options.append(n_try)
    # Add some absolute values
    for n_abs in [600, 1000, 1400, 1800, 2400]:
        if n_abs not in n_options and 400 <= n_abs <= 3000:
            n_options.append(n_abs)
    n_options = sorted(set(n_options))

    # Prioritize closer lengths first
    n_options_sorted = sorted(n_options, key=lambda x: abs(x - n_current))

    for n_try in n_options_sorted[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample(best_a, n_try)

        # Quick LP descent
        alloc = min(remaining * 0.18, 100.0)
        if alloc > 20:
            a_try = lp_descent(a_try, time_budget=alloc, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Subgradient refinement ===
    print(f"\n=== Phase 4: Subgradient Refinement ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.08, 80.0)
    if p4_budget > 20:
        a_sg = subgradient_descent(best_a, time_budget=p4_budget, K=8)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()
            print(f"  Phase 4: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 4: obj={obj_sg:.8f}")

    # === Phase 5: Perturbation + LP restarts ===
    print(f"\n=== Phase 5: Perturbation + LP Restarts ===")
    remaining = deadline - time.monotonic()

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        scale = 0.015 + 0.01 * r
        a_pert = best_a * np.exp(rng.normal(0, scale, len(best_a)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc = min(remaining * 0.2, 90.0)
        if alloc > 20:
            a_pert = lp_descent(a_pert, time_budget=alloc, seed=500 + r)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_obj:
                best_obj = obj_pert
                best_a = a_pert.copy()
                print(f"  Perturb {r} (scale={scale:.4f}): obj={best_obj:.8f} **")
            else:
                print(f"  Perturb {r} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 6: Final intensive LP polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.5, 200.0)
        a_final = lp_descent(best_a, time_budget=alloc, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
            print(f"  Final LP: obj={best_obj:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

        # Quick subgradient on top
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc2 = min(remaining * 0.3, 50.0)
            a_sg2 = subgradient_descent(best_a, time_budget=alloc2, K=10)
            obj_sg2 = compute_obj(a_sg2)
            if obj_sg2 < best_obj:
                best_obj = obj_sg2
                best_a = a_sg2.copy()
                print(f"  Final SG: obj={best_obj:.8f} **")

    # === Phase 7: Try diverse initializations at best_n ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 7: Diverse Inits at n={n_current} ===")
        for mode in [2, 3, 5]:
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break
            a_div = diverse_init(n_current, rng, mode)
            # Quick LP
            alloc = min(remaining * 0.15, 50.0)
            if alloc > 15:
                a_div = lp_descent(a_div, time_budget=alloc, seed=800 + mode)
                obj_div = compute_obj(a_div)
                if obj_div < best_obj:
                    best_obj = obj_div
                    best_a = a_div.copy()
                    print(f"  Diverse mode {mode}: obj={best_obj:.8f} **")
                else:
                    print(f"  Diverse mode {mode}: obj={obj_div:.8f}")

    # Final validation
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")

    return result
```
