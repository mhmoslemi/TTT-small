def compute_conv(a):

        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

    if s < 0.01:

    conv = compute_conv(a_n)

    """Solve min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M, dual_vars) or None."""
    n = len(a)

    # conv(a_n, g)[k] = sum_j a_n[j] * g[k-j] for valid j, k-j
    # We need: for each k, sum_j a_n[j] * g[max(0,k-n+1):min(n,k+1)] <= M

    from scipy.sparse import lil_matrix, csr_matrix
    A = lil_matrix((2*n-1, n))
    for j in range(n):
        for k in range(j, j+n):
            A[k, j] = a_n[j]
    A = A.tocsr()
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based fixed-point iteration with dual-guided exact subgradients,
adaptive line search, Nesterov acceleration on the fixed-point map,
and multi-phase time allocation with local search polishing.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.sparse import lil_matrix, csr_matrix
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        aj = a[j]
        if abs(aj) < 1e-15:
            continue
        for k in range(j, j + n):
            rows.append(k)
            cols.append(j)
            vals.append(aj)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_minpeak(a, solver_list=None, warm_start=None):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    if warm_start is not None:
        g_var.value = warm_start
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solver_list:
        try:
            extra = {}
            if solver == cp.MOSEK:
                extra = {'msk.IMATRIX_OPT’: 20, 'msk.PRESOLVE_OPT': 0}
            prob.solve(solver=solver, **extra)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M_var.value)
                    dual_y = None
                    try:
                        dual_y = np.array(constraints[0].dual_value).flatten()
                    except Exception:
                        pass
                    return g_val, M_val, dual_y
        except Exception:
            continue
    return None, None, None


def dual_subgradient(a, dual_y):
    if dual_y is None or len(dual_y) == 0:
        return None
    n = len(a)
    N = 1
    while N < 4 * n:
        N <<= 1
    A_fft = fft(a, N)
    Y_fft = fft(dual_y, N)
    cross = ifft(A_fft * Y_fft).real
    s = 2.0 * cross[:n]
    return s


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def line_search_blend(a, g, num_points=20):
    n = len(a)
    best_t = 0.0
    best_obj = compute_obj(a)
    ts = np.linspace(0, 1, num_points)
    objs = np.zeros(num_points)
    for i, t in enumerate(ts):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            objs[i] = np.inf
            continue
        a_try = a_try / s
        conv = compute_conv(a_try)
        objs[i] = 2.0 * n * float(np.max(conv))
    idx = np.argmin(objs)
    best_t = ts[idx]
    best_obj = objs[idx]
    if 0 < idx < num_points - 1:
        t0, t1, t2 = ts[idx - 1], ts[idx], ts[idx + 1]
        f0, f1, f2 = objs[idx - 1], objs[idx], objs[idx + 1]
        denom = (f0 - 2 * f1 + f2)
        if abs(denom) > 1e-15:
            t_parab = t1 - 0.5 * (t0 - t1) * (f0 - f1) / (f0 - 2 * f1 + f2)
            if t0 < t_parab < t2:
                a_try = (1 - t_parab) * a + t_parab * g
                a_try = np.maximum(a_try, 0)
                s = float(np.sum(a_try))
                if s > 1e-10:

                    conv = compute_conv(a_try)
                    f_parab = 2.0 * n * float(np.max(conv))
                    if f_parab < best_obj:
                        best_t = t_parab
                        best_obj = f_parab
    a_best = (1 - best_t) * a + best_t * g
    a_best = np.maximum(a_best, 0)
    s = float(np.sum(a_best))
    if s > 1e-10:
        a_best = a_best / s
    return a_best, best_t, best_obj


def smoothed_gradient(a, tau):
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv

        if it - best_it > 50000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it
    return best_a


def lp_fixed_point_iteration(a, time_budget=500.0, max_iter=5000, verbose=True):
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf, 0
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    no_improve = 0
    prev_a = a.copy()
    y_prev = None
    momentum = 0.3
    last_t = 0.5
    while (time.monotonic() - start) < time_budget and it < max_iter:
        g_lp, M_lp, dual_y = solve_lp_minpeak(a)
        if g_lp is None:
            if verbose:
                print(f"    LP failed at iter {it}")
            break
        a_blend, t_opt, obj_blend = line_search_blend(a, g_lp, num_points=15)
        if obj_blend < best_obj - 1e-13:
            if no_improve > 5 or it % 10 == 0:
                a_mom = a + momentum * (a - prev_a)
                a_mom = np.maximum(a_mom, 0)
                s_mom = float(np.sum(a_mom))
                if s_mom > 1e-10:
                    a_mom = a_mom / s_mom
                    g_lp2, M_lp2, _ = solve_lp_minpeak(a_mom)
                    if g_lp2 is not None:
                        a_blend2, t_opt2, obj_blend2 = line_search_blend(a_mom, g_lp2, num_points=10)
                        if obj_blend2 < obj_blend:
                            a_blend = a_blend2
                            obj_blend = obj_blend2
            best_obj = obj_blend
            best_a = a_blend.copy()
            prev_a = a.copy()
            a = a_blend
            last_t = t_opt
            no_improve = 0
            if verbose and it % 20 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.3f}, M={M_lp:.8f}")
        else:
            no_improve += 1
            if dual_y is not None and no_improve <= 5:
                s_grad = dual_subgradient(a, dual_y)
                if s_grad is not None and np.max(np.abs(s_grad)) > 1e-15:
                    lr_grad = 0.5 / (np.max(np.abs(s_grad)) + 1e-10)
                    a_step = a - lr_grad * s_grad
                    a_step = project_simplex(a_step)
                    s_step = float(np.sum(a_step))
                    if s_step > 1e-10:
                        a_step = a_step / s_step
                        obj_step = compute_obj(a_step)
                        if obj_step < best_obj + 1e-12:
                            a = a_step
            if no_improve >= 8:
                break
        it += 1
    return best_a, best_obj, it


def perturb_sequence(a, rng, method=0, scale=0.03):
    n = len(a)
    a = a.copy()
    if method == 0:
        a = a * np.exp(rng.normal(0, scale, n))
    elif method == 1:
        noise = rng.uniform(1 - scale, 1 + scale, n)
        a = a * noise
    elif method == 2:
        mask = rng.random(n) > scale
        a = np.where(mask, a, 0.0)
    elif method == 3:
        shift = int(rng.integers(1, max(2, n // 20)))
        a = np.roll(a, shift)
    elif method == 4:
        a = (1 - scale) * a + scale / n
    elif method == 5:
        A = fft(a, 2 * n)
        A[1:-1] *= np.exp(1j * rng.normal(0, scale, len(A[1:-1])))
        a = ifft(A).real[:n]
        a = np.maximum(a, 0)
    elif method == 6:
        conv = compute_conv(a / max(a.sum(), 1e-10))
        top_k = np.argsort(conv)[::-1][:min(10, n)]
        deltas = np.zeros(n)
        for k in top_k:
            k = int(k)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    deltas[j] += a[idx]
        max_d = np.max(np.abs(deltas))
        if max_d > 1e-10:
            a = a - scale * 0.3 * deltas / max_d
        a = np.maximum(a, 0)
    elif method == 7:
        idx = rng.integers(0, n, size=min(20, n))
        a[idx] *= rng.uniform(0.5, 1.5, size=len(idx))
    else:
        a = a * np.exp(rng.normal(0, scale, n))
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def local_search_polish(a, time_budget=60.0):
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    active_lags = np.where(conv >= max_val * 0.999)[0]
    while (time.monotonic() - start) < time_budget:
        improved = False
        n_active = len(active_lags)
        if n_active == 0:
            break
        for k in active_lags[:min(n_active, 30)]:
            k = int(k)
            if (time.monotonic() - start) >= time_budget:
                break
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                idx1 = j
                idx2 = k - j
                if idx1 == idx2 or idx1 < 0 or idx1 >= n or idx2 < 0 or idx2 >= n:
                    continue
                current_val = conv[k]
                delta_options = [0.01, -0.01, 0.02, -0.02, 0.05, -0.05]
                best_delta = 0
                best_new_obj = best_obj
                for d in delta_options:
                    a_try = a.copy()
                    a_try[idx1] += d / n
                    a_try[idx2] -= d / n
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    conv_try = compute_conv(a_try)
                    obj_try = 2.0 * n * float(np.max(conv_try))
                    if obj_try < best_new_obj:
                        best_new_obj = obj_try
                        best_delta = d
                        a_best_try = a_try.copy()
                if best_delta != 0 and best_new_obj < best_obj - 1e-14:
                    a = a_best_try
                    best_a = a.copy()
                    best_obj = best_new_obj
                    conv = compute_conv(a)
                    max_val = float(np.max(conv))
                    active_lags = np.where(conv >= max_val * 0.999)[0]
                    improved = True
                    break
            if improved:
                break
        if not improved:
            n Moves = 10
            for _ in range(n_moves):
                if (time.monotonic() - start) >= time_budget:
                    break
                j1 = int(np.random.randint(0, n))
                j2 = int(np.random.randint(0, n))
                if j1 == j2:
                    continue
                d = np.random.choice([0.05, -0.05, 0.1, -0.1]) / n
                a_try = a.copy()
                a_try[j1] += d
                a_try[j2] -= d
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                conv_try = compute_conv(a_try)
                obj_try = 2.0 * n * float(np.max(conv_try))
                if obj_try < best_obj - 1e-14:
                    a = a_try
                    best_a = a.copy()
                    best_obj = obj_try
                    conv = compute_conv(a)
                    max_val = float(np.max(conv))
                    active_lags = np.where(conv >= max_val * 0.999)[0]
                    break
            if (time.monotonic() - start) >= time_budget:
                break
            small_improvement = (best_obj - compute_obj(a)) < 1e-12
            if small_improvement and len(active_lags) > 50:
                break
    return best_a, best_obj


def diverse_init(n, rng, mode=0):
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 6, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(x) + 0.2 * np.cos(2 * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 7:
        a = np.ones(n)
        a[::2] *= 1.5
        a[1::3] *= 0.5
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)
