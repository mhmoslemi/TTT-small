"""
LP-dominant optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses aggressive LP iteration with backtracking line search, dual-guided
perturbation, spectral initialization, and multi-length search.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def solve_lp(a):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    return g_val / s_val, float(M.value), prob
        except Exception:
            continue
    return None, None, None

def spectral_init(n, n_iters=15, alpha=0.3):
    a = np.ones(n) / n
    size = 1
    while size < 2 * n:
        size <<= 1
    for _ in range(n_iters):
        F = np.fft.fft(a, size)
        mag = np.abs(F).copy()
        if len(mag) > 1:
            target = np.mean(mag[1:])
            mag[1:] = (1 - alpha) * mag[1:] + alpha * target
            new_F = mag * np.exp(1j * np.angle(F))
            a = np.fft.irfft(new_F, size)[:n]
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
    return a

def quick_gd(a, time_budget=30.0, tau=50.0):
    """Quick log-sum-exp smoothed gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 2.0 / np.sqrt(n)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient via cross-correlation
        N = 1
        while N < 3 * n:
            N <<= 1
        P = np.fft.fft(p, N)
        A = np.fft.fft(a, N)
        g_full = np.fft.ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]
        
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new
        lr *= 0.999
    
    return best_a

def dual_guided_perturbation(a, prob, rng, strength=0.05):
    """
    Use LP dual variables to create targeted perturbation.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    # Get dual variables for the inequality constraints (conv(a,g) <= M)
    # In cvxpy, these are the dual values for the constraint A @ g <= M
    try:
        # The constraint is A @ g <= M, which in cvxpy is a single constraint
        # with dual variables of length (2n-1)
        constraint = prob.constraints[0]  # A @ g <= M
        dual_val = constraint.dual_value
        if dual_val is None:
            return fourier_perturb(a, rng, strength)
        dual_val = np.maximum(dual_val, 0)
        dsum = float(np.sum(dual_val))
        if dsum < 1e-10:
            return fourier_perturb(a, rng, strength)
        lambda_star = dual_val / dsum
    except Exception:
        return fourier_perturb(a, rng, strength)
    
    # Find the top contributing lags
    top_k = np.argsort(lambda_star)[::-1][:5]
    
    # Create perturbation: decrease a at positions that contribute to top lags
    pert = np.zeros(n)
    for k_idx in top_k:
        k_val = int(k_idx)
        for j in range(n):
            idx = k_val - j
            if 0 <= idx < n:
                # a[j] * a[idx] contributes to conv[k_val]
                # To reduce this, decrease the larger of a[j] and a[idx]
                pert[j] -= a_norm[j] * a_norm[idx] * lambda_star[k_val]
    
    # Apply perturbation
    a_new = a * np.exp(strength * pert / (np.max(np.abs(pert)) + 1e-10))
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new > 1e-10:
        a_new = a_new / s_new
    else:
        a_new = fourier_perturb(a, rng, strength)
    return a_new

def fourier_perturb(a, rng, strength=0.03):
    """Add random Fourier modes as perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 4):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s
