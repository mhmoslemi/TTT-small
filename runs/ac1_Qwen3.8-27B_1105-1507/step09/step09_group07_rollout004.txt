def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lse_mirror_descent(a, time_budget, beta_start=5.0, beta_end=200.0, verbose=False):
    """
    Log-sum-exp with mirror descent (exponential steps) on the simplex.
    This preserves positivity naturally and can be more effective.
    """
    from scipy.fft import rfft, irfft
    
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size_corr = 1
    while size_corr < 3 * n:
        size_corr <<= 1
    
    it = 0
    lr = 2.0 / np.sqrt(n)
    total_iters = max(100, int(time_budget * 5))
    
    while time.monotonic() - start < time_budget:
        frac = min(1.0, it / max(1, total_iters))
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Compute convolution
        fa = rfft(a, size_corr)
        conv = irfft(fa * fa, size_corr)[:2 * n - 1]
        
        # Softmax weights
        conv_shifted = beta * (conv - conv.max())
        w = np.exp(conv_shifted)
        w = w / w.sum()
        
        # Gradient via cross-correlation
        fw = rfft(w, size_corr)
        corr = irfft(fw * np.conj(fa), size_corr)[:n]
        g = 2.0 * corr
        
        # Mirror descent: a_new[i] = a[i] * exp(-lr * g[i])
        log_a = np.log(a)
        log_a -= lr * g
        # Normalize to stay on simplex (approximately)
        log_a -= np.max(log_a)
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = min(lr * 1.005, 5.0 / np.sqrt(n))
        else:
            lr *= 0.97
        
        it += 1
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"          Mirror it{it}: obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.4f}, elapsed={elapsed:.1f}s")
    
    return best_a


def diverse_init(n, rng, mode=0):

    elif mode == 4:
        freqs = rng.integers(2, 10, size=3)

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def targeted_moves(a, time_budget):
    """
    Targeted single-coordinate moves: for the peak lag, try moving mass from large
    contributors to quiet positions.
    """
    from scipy.fft import rfft, irfft
    
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    round_num = 0
    while time.monotonic() - start < time_budget:
        round_num += 1
        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]
        
        if round_num % 10 == 0:
            print(f"            Targeted r{round_num}: obj={best_obj:.8f}, peak_lag={peak_idx}, peak_val={peak_val:.6f}")
        
        # Find quiet positions (small a values that don't contribute to peaks)
        # Score position i by how much it contributes to the top 5 lags
        top_lags = np.argsort(conv)[::-1][:5]
        contrib = np.zeros(n)
        for k in top_lags:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contrib[i] += a[j]
        
        # Quiet positions: low contribution and low value
        quiet_score = contrib + 0.5 * a
        quiet_positions = np.argsort(quiet_score)[:10]
        
        # For the peak lag, find the largest contributors
        peak_i = int(peak_idx)
        contributors = []
        for i in range(n):
            j = peak_i - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)
        
        improved = False
        for contrib_val, i, j in contributors[:5]:
            if time.monotonic() - start > time_budget:
                break
            if contrib_val < 1e-10:
                break
            src = i if a[i] >= a[j] else j
            for target in quiet_positions:
                if target == src:
                    continue
                for frac in [0.01, 0.02, 0.05, 0.1]:
                    delta = frac * a[src]
                    if delta <= 0 or delta >= 0.9 * a[src]:
                        continue
                    old_src = a[src]
                    old_tgt = a[target]
                    a[src] -= delta
                    a[target] += delta
                    obj_try = compute_obj(a)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] = old_src
                        a[target] = old_tgt
                        break
            if improved:
                break
        
        if not improved and round_num > 50:
            break
    
    return best_a
