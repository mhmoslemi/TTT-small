def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-8))
        a = a / a.sum()
    elif mode == 4:
        # Low-frequency modulation
        freqs = rng.integers(1, 4, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1, 1)
        a = a / a.sum()
    elif mode == 6:
        # Comb pattern
        a = np.zeros(n)
        tooth_width = max(1, n // 40)
        for i in range(0, n, 2 * tooth_width):
            a[i:i+tooth_width] = 1.0
        a = a / a.sum()
    else:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n) / n)
        a = a / a.sum()
    return a

def lp_polish(a, time_budget, verbose=False):
    """LP-based direction finding with golden section line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        # Get LP direction
        g = solve_lp_direction(a, verbose=verbose)
        if g is None:
            # Try QP
            g = solve_qp_direction(a, reg=0.01)
        if g is None:
            break
        
        # Golden section line search
        phi = (np.sqrt(5) - 1) / 2
        lo, hi = 0.0, 1.0
        
        for _ in range(8):
            t1 = hi - phi * (hi - lo)
            t2 = lo + phi * (hi - lo)
            
            a1 = np.maximum((1 - t1) * a + t1 * g, 0)
            s1 = float(np.sum(a1))
            f1 = compute_obj(a1 / s1) if s1 > 1e-10 else np.inf
            
            a2 = np.maximum((1 - t2) * a + t2 * g, 0)
            s2 = float(np.sum(a2))
            f2 = compute_obj(a2 / s2) if s2 > 1e-10 else np.inf
            
            if f1 < f2:
                hi = t2
            else:
                lo = t1
        
        t_best = (lo + hi) / 2
        a_new = np.maximum((1 - t_best) * a + t_best * g, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        it += 1
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}")
        else:
            stall += 1
            # Perturb
            scale = 0.02 + 0.005 * min(stall, 10)
            a = best_a * np.exp(rng.normal(0, min(scale, 0.1), n))
            a = np.maximum(a, 0)
            s_a = float(np.sum(a))
            if s_a > 1e-10:
                a = a / s_a
            else:
                a = best_a.copy()
            if stall > 30:
                break
    
    return best_a

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based direction finding with golden-section line search,
Nesterov-accelerated smoothed gradient descent with adaptive tau,
2-coordinate peak tapping, and multi-length exploration.
Key improvements: QP-regularized LP direction, synchronized subgradient
at high tau, analytical 2-coord moves targeting top peaks, and
aggressive multi-length search with re-optimization.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns the optimal g (normalized to sum=1) or None.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def solve_qp_direction(a, reg=0.005):
    """
    Solve QP: min (1/2)*reg*||g-a||^2 + M
    s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    obj = cp.quad_form(g - a_norm, cp.psd_wrap(np.eye(n) * reg)) + M
    prob = cp.Problem(cp.Minimize(obj), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def solve_lp_active(a, active_idx):
    """
    Solve LP: min sum(conv(a,g)[k] for k in active_idx)
    s.t. sum(g)=1, g>=0
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    obj = cp.sum(A[active_idx, :] @ g)
    constraints = [cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(obj), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def smoothed_gradient(a, tau):
    """
    Compute gradient of (1/tau)*log(sum(exp(tau*conv(a,a)[k]))) via FFT.
    Returns (gradient, conv, soft_weights).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    grad = 2.0 * g_full[:n]
    return grad, conv, p


def nesterov_smoothed(a, time_budget, tau_init=20.0, tau_end=500.0, verbose=False):
    """
    Nesterov accelerated gradient descent on log-sum-exp smoothed objective.
    Uses adaptive learning rate and periodic restarts.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget * 0.85, 1e-6))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        grad, conv, p = smoothed_gradient(y, tau)

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        # Adaptive LR decay
        lr = max(0.3 / np.sqrt(n), lr * 0.9997)

        # Periodic restart if stalled
        if it - best_it > 8000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)
            beta = max(0.5, beta * 0.9)
            best_it = it

        if verbose and it % 3000 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.5f}")

    return best_a


def two_stage_lp_polish(a, time_budget, verbose=False):
    """
    Two-stage LP: (1) min-max to get active set, (2) min-sum on active set.
    Blend and line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(54321)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        # Stage 1: min max
        g1 = solve_lp_direction(a)
        if g1 is None:
            break

        # Find active set
        A = build_conv_matrix(a / np.sum(a))
        conv_a_g1 = A @ g1
        max_val = float(np.max(conv_a_g1))
        tol = 0.05 * max_val + 1e-10
        active = np.where(conv_a_g1 >= max_val - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv_a_g1)])
        # Keep at most 20 most active
        if len(active) > 20:
            active_sorted = active[np.argsort(conv_a_g1[active])[::-1][:20]]
            active = active_sorted

        # Stage 2: min sum on active
        g2 = solve_lp_active(a, active)

        # Choose direction
        if g2 is not None:
            conv_a_g2 = A @ g2
            peak_g2 = float(np.max(conv_a_g2))
            peak_g1 = max_val
            if peak_g2 < peak_g1 * 0.99:
                g_dir = g2
            else:
                g_dir = 0.5 * g1 + 0.5 * g2
        else:
            g_dir = g1

        # Also try QP direction occasionally
        if it % 5 == 3:
            g_qp = solve_qp_direction(a, reg=0.003)
            if g_qp is not None:
                g_dir = 0.5 * g_dir + 0.5 * g_qp

        # Golden section line search
        phi = (np.sqrt(5) - 1) / 2
        lo, hi = 0.0, 1.0
        for _ in range(8):
            t1 = hi - phi * (hi - lo)
            t2 = lo + phi * (hi - lo)

            a1 = np.maximum((1 - t1) * a + t1 * g_dir, 0)
            s1 = float(np.sum(a1))
            f1 = compute_obj(a1 / s1) if s1 > 1e-10 else np.inf

            a2 = np.maximum((1 - t2) * a + t2 * g_dir, 0)
            s2 = float(np.sum(a2))
            f2 = compute_obj(a2 / s2) if s2 > 1e-10 else np.inf

            if f1 < f2:
                hi = t2
            else:
                lo = t1

        t_best = (lo + hi) / 2
        a_new = np.maximum((1 - t_best) * a + t_best * g_dir, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        it += 1

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 3 == 0:
                print(f"    2SLP it{it}: obj={best_obj:.8f}, |active|={len(active)}")
        else:
            stall += 1
            scale = 0.015 + 0.005 * min(stall, 12)
            a = best_a * np.exp(rng.normal(0, min(scale, 0.1), n))
            a = np.maximum(a, 0)
            s_a = float(np.sum(a))
            if s_a > 1e-10:
                a = a / s_a
            else:
                a = best_a.copy()

def peak_tap_2coord(a, time_budget, verbose=False):
    """
    2-coordinate descent: for each top peak, find best (src, tgt) mass transfer.
    Uses analytical insight: moving mass from a[src] to a[tgt] changes
    conv at position k by -delta*a[k-src] + delta*a[k-tgt].
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(77777)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_k = 5
        peak_indices = np.argsort(conv)[::-1][:top_k]

        improved = False
        for pk in peak_indices:
            pk = int(pk)

            # Find contributing pairs for this peak
            i_start = max(0, pk - (n - 1))
            i_end = min(n - 1, pk)

            if i_start > i_end:
                continue

            # Get top contributing source positions
            contribs = []
            for i in range(i_start, i_end + 1):
                j = pk - i
                if 0 <= j < n:
                    contribs.append((a[i] * a[j], i, j))
            contribs.sort(reverse=True)

            for contrib_val, src, _ in contribs[:8]:
                if a[src] < 1e-8:
                    continue

                # Candidate targets: positions that minimize increase in peak
                # a[k - tgt] should be small for the peak position k
                candidates = set()
                # Far from peak
                candidates.add((src + n // 4) % n)
                candidates.add((src + n // 2) % n)
                candidates.add((src + 3 * n // 4) % n)
                # Mirror
                candidates.add(n - 1 - src)
                # Random
                for _ in range(6):
                    candidates.add(int(rng.integers(0, n)))
                # Near but offset
                for off in [2, 3, n // 15, n // 8]:
                    candidates.add((src + off) % n)
                    candidates.add((src - off) % n)

                for tgt in candidates:
                    tgt = int(tgt) % n

                    # We want a[pk-tgt] to be small

                    if 0 <= idx_partner < n:
                        score = a[idx_partner]
                    else:
                        score = -1  # Best case: no contribution
                    scored.append((score, tgt))

                scored.sort()

                for score, tgt in scored[:6]:
                    for frac in [0.15, 0.08, 0.04, 0.02, 0.01]:

                        a_try[src] -= delta
                        a_try[tgt] += delta
                        a_try = np.maximum(a_try, 0)

                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a_try.copy()

                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 150:
                break

        if verbose and it % 200 == 0:

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = np.full(new_n, 1.0 / new_n)
    return a_new


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [max(1, n // 10), max(1, n // 10)]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-8))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 4, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1.5, 1.5)
        a = a / a.sum()
    elif mode == 6:
        # Sparse comb
        num_teeth = max(3, n // 30)
        a = np.zeros(n)
        for i in range(num_teeth):
            pos = int(i * n / num_teeth + rng.random() * (n // num_teeth))
            pos = min(pos, n - 1)
            a[pos] = 1.0
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 7:
        # Exponential decay from center
        center = n // 2
        a = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (n // 6) ** 2)
        a = a / a.sum()
    else:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n) / n)
        a = a / a.sum()
    return a


def block_cd(a, time_budget, verbose=False):
    """
    Block coordinate descent: optimize pairs of adjacent coordinates
    to minimize the max convolution value.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(31415)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_k = int(np.argmax(conv))
        max_val = float(conv[max_k])

        # Find the top contributor to this peak
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)

        best_improvement = best_obj
        best_move = None

        # Try moving mass from high positions to low positions
        # Focus on the dominant pair
        sorted_idx = np.argsort(a)[::-1]
        top_positions = sorted_idx[:max(5, n // 50)]

        for src in top_positions:
            src = int(src)
            if a[src] < 1e-8:
                continue
            # Partner position for the peak
            partner = max_k - src
            if partner < 0 or partner >= n:
                continue

            # Try transferring to various targets
            targets = [
                int(rng.integers(0, n)),
                int(rng.integers(0, n)),
                (src + n // 3) % n,
                (src + n // 2) % n,
                n - 1 - src,
            ]

            for tgt in targets:
                tgt = int(tgt) % n
                if tgt == src:
                    continue
                for frac in [0.2, 0.1, 0.05]:
                    delta = frac * a[src]
                    if delta > a[src]:
                        continue
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[tgt] += delta
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_improvement - 1e-14:
                        best_improvement = obj_try
                        best_move = (src, tgt, delta, a_try)

        it += 1
        if best_move is not None:
            best_obj = best_improvement
            best_a = best_move[3].copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break

        if verbose and it % 200 == 0:
            print(f"    BCD it{it}: obj={best_obj:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization:
    1. Two-stage LP polishing (min-max + min-active)
    2. Nesterov smoothed GD with adaptive tau
    3. 2-coordinate peak tapping
    4. Block coordinate descent
    5. Multi-length exploration with re-optimization
    6. Diverse restarts
    7. Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent: {e}")

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Two-stage LP ===
    print(f"\n=== Phase 1: Two-Stage LP (budget available: {deadline - time.monotonic():.0f}s) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 280.0)
    if p1_budget > 30:
        a1 = two_stage_lp_polish(best_seq, time_budget=p1_budget, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  2SLP: obj={best_score:.8f} **")
        else:
            print(f"  2SLP: obj={obj1:.8f}")

    # === Phase 2: Nesterov Smoothed GD ===
    print(f"\n=== Phase 2: Nesterov Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.22, 250.0)
    if p2_budget > 30:
        a2 = nesterov_smoothed(best_seq, time_budget=p2_budget,
                               tau_init=15.0, tau_end=400.0, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  NSD: obj={best_score:.8f} **")
        else:
            print(f"  NSD: obj={obj2:.8f}")

    # === Phase 3: 2-Coordinate Peak Tapping ===
    print(f"\n=== Phase 3: Peak Tapping ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 120.0)
    if p3_budget > 20:
        a3 = peak_tap_2coord(best_seq, time_budget=p3_budget, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  PTC: obj={best_score:.8f} **")
        else:
            print(f"  PTC: obj={obj3:.8f}")

    # === Phase 4: Block CD ===
    print(f"\n=== Phase 4: Block CD ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.1, 80.0)
    if p4_budget > 15:
        a4 = block_cd(best_seq, time_budget=p4_budget, verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  BCD: obj={best_score:.8f} **")
        else:
            print(f"  BCD: obj={obj4:.8f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(200, n_current - 100),
        n_current + 80,
        n_current + 150,
        max(300, n_current - 200),
        500, 700, 900, 1100, 1300, 1400
    ]))
    n_options = [n for n in n_options if 150 <= n <= 1500 and abs(n - n_current) > 40]

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break

        a_try = resample_sequence(best_seq, n_try)

        alloc_lp = min(remaining * 0.3, 50.0)
        if alloc_lp > 20:
            a_try = two_stage_lp_polish(a_try, time_budget=alloc_lp)

        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.3, 35.0)
        if alloc_gd > 15:
            a_try = nesterov_smoothed(a_try, time_budget=alloc_gd,
                                      tau_init=25.0, tau_end=300.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        mode = (r * 2 + 1) % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.35 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r
        else:
            a_r = best_seq.copy()

        alloc = min(remaining * 0.25, 70.0)
        if alloc > 25:
            half = alloc / 2
            a_r = two_stage_lp_polish(a_r, time_budget=half)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_r = nesterov_smoothed(a_r, time_budget=min(remaining * 0.3, 30.0),
                                        tau_init=30.0, tau_end=350.0)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 35:
        alloc = min(remaining * 0.35, 90.0)
        a7 = two_stage_lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj7 = compute_obj(a7)
        if obj7 < best_score:
            best_score = obj7
            best_seq = a7.copy()
            print(f"  Final 2SLP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.3, 55.0)
        a7 = nesterov_smoothed(best_seq, time_budget=alloc,
                               tau_init=50.0, tau_end=800.0, verbose=True)
        obj7 = compute_obj(a7)
        if obj7 < best_score:
            best_score = obj7
            best_seq = a7.copy()
            print(f"  Final NSD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 18:
        alloc = min(remaining * 0.35, 35.0)
        a7 = peak_tap_2coord(best_seq, time_budget=alloc)
        obj7 = compute_obj(a7)
        if obj7 < best_score:
            best_score = obj7
            best_seq = a7.copy()
            print(f"  Final PTC: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 12:
        alloc = min(remaining * 0.4, 25.0)
        a7 = block_cd(best_seq, time_budget=alloc)
        obj7 = compute_obj(a7)
        if obj7 < best_score:
            best_score = obj7
            best_seq = a7.copy()
            print(f"  Final BCD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
