  (The Hessian of `(a*a)[k*]` applied to (d,d) gives `2*(d*d)[k*]`)
  
  Wait, let me redo. `(a+td)*(a+td))[k*] = (a*a)[k*] + 2t*(a*d)[k*] + t^2*(d*d)[k*]`
  
  So `f(a+td) = 2n * ((a*a)[k*] + 2t*(a*d)[k*] + t^2*(d*d)[k*])` (assuming k* stays the active peak)
  
  `f'(0) = 4n * (a*d)[k*]`
  `f''(0) = 4n * (d*d)[k*]`
  
  Optimal t: `t* = -f'(0) / f''(0) = -(a*d)[k*] / (d*d)[k*]`
  
  But we need `f'(0) < 0` for this to be a descent direction. If `(a*d)[k*] > 0`, then the LP direction actually increases the peak! In that case, we should use a smaller step or a different direction.

  Actually, the LP minimizes `max_k (a*g)[k]`, and we know that `(a*a)[k*] = M` (the current peak). The LP ensures `(a*g)[k] <= M_LP <= M` for all k.   But we need `d*d)[k*] > 0` for this to make sense. If `(d*d)[k*] <= 0`, we can't use the Newton step (the function is concave in that direction), and we should fall back to a line search.

  Also, we need `t* > 0` and `t* < 1` (since we're blending between a and g, we want 0 < t < 1). If `t* > 1`, we cap it at 1. If `t* <= 0`, something is wrong.

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Newton-accelerated LP blending, FFT-based convolution, multi-length exploration,
and spectral initialization for improved local optimization.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build dense matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, M_target=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver, **{'tol_abs': 1e-8, 'tol_rel': 1e-8} if solver != cp.PDLP else {})
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue
    
    return None, None


def newton_blend_step(a, g, k_star):
    """Compute Newton-optimal blend factor for a + t*(g-a)."""
    d = g - a
    # (a*d)[k_star] and (d*d)[k_star]
    # (a*d)[k] = sum_i a[i] * d[k-i]
    n = len(a)
    # Compute (a*d) at k_star using FFT
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fd = rfft(d, size)
    ad = irfft(fa * fd, size)[:2 * n - 1]
    dd = irfft(fd * fd, size)[:2 * n - 1]
    
    ad_k = ad[k_star]
    dd_k = dd[k_star]
    
    if dd_k > 1e-12:
        t_newton = -ad_k / dd_k
        t_newton = max(0.01, min(1.0, t_newton))
        return t_newton
    else:
        # Fallback: use 0.5
        return 0.5


def nesterov_smoothed_descent(a, time_budget=120.0, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)
        
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_fft = fft(y, N)
        g_full = ifft(P * np.conj(A_fft)).real
        g = 2.0 * g_full[:n]
        
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9998)
        
        if it - best_it > 50000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 5.0 / np.sqrt(n) * 0.3
            best_it = it
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Spectral init: random phases, flat magnitude
        N = n
        A = np.ones(N, dtype=complex)
        phases = rng.uniform(0, 2 * np.pi, N)
        A = np.exp(1j * phases)
        # Enforce Hermitian symmetry for real output
        A = (A + np.conj(A[::-1])) / 2
        a = np.real(ifft(A))
        a = np.maximum(a, 0)
        if a.sum() < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / a.sum()
    elif mode == 3:
        # Bump functions
        centers = rng.integers(0, n, size=3)
        widths = rng.integers(n // 10, n // 4, size=3)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 4:
        # Cosine with random frequency
        x = np.linspace(0, 1, n)
        f = rng.uniform(1, 10)
        p = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

The objective is `2n * max_k (a*a)[k] / (sum(a))^2`. After normalization (sum=1), it's `2n * max_k (a*a)[k]`.

The minimum possible value of `max_k (a*a)[k]` for a non-negative sequence of length n with sum 1 is achieved when the autocorrelation is as flat as possible.

For a "random" non-negative sequence, the expected value of `(a*a)[k]` for k > 0 is approximately `1/n` (by the law of large numbers, since each term is a product of two independent-ish random variables with mean 1/n). The peak at k=0 is always `sum(a^2) >= 1/n` (by Cauchy-Schwarz).

So the "random" baseline gives `max_k (a*a)[k] ≈ sum(a^2) ≈ 1/n * (1 + 1/n)` (for a roughly uniform distribution). The objective would be `2n * 1/n ≈ 2`.

To get below 2, we need to make the autocorrelation "flatter" than random, which means the k=0 peak (energy) should be close to the off-peak values. This requires the sequence to be "spread out" (low energy concentration).

The theoretical minimum is related to the "flat-top" autocorrelation, which would give `max_k (a*a)[k] = 1/(2n-1)` (perfectly flat autocorrelation over all 2n-1 lags). This would give an objective of `2n * 1/(2n-1) ≈ 1`. But this is achievable only for complex sequences (like Zadoff-Chu), not real non-negative ones.

For real non-negative sequences, the constraint `a >= 0` is very restrictive. The best we can do is probably around 1.5 (as confirmed by the target).

def solve_lp_minpeak_subset(a, M_target=None, subset_indices=None):
    """Solve LP with subset of constraints."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    if subset_indices is not None:
        rows = subset_indices
    else:
        rows = np.arange(2 * n - 1)
    
    # Build sub-matrix
    A_sub = np.zeros((len(rows), n))
    for j in range(n):
        for idx, k in enumerate(rows):
            if 0 <= k - j <= n - 1:
                A_sub[idx, j] = a_norm[k - j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_sub @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None
