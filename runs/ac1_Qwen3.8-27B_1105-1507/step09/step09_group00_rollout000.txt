"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Explore larger sequence lengths (up to 5000) using FFT-accelerated
smoothed gradient descent, exact subgradient steps, LP polishing for moderate n,
arcsine-based initializations, and Polyak step-size adaptation.
Key difference from previous: focuses on much larger n values where the theoretical
lower bound 2n/(2n-1) is closer to 1, combined with structured initializations
(arcsine distribution) that give flatter convolutions.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_grad_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1

    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def exact_subgradient(a):
    """Compute exact subgradient using the argmax of convolution."""
    n = len(a)
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    # Subgradient: g_i = 2 * a_{k_star - i} if 0 <= k_star - i < n
    g = np.zeros(n)
    # Vectorized: for each i, j = k_star - i, need 0 <= j < n
    # i ranges from max(0, k_star-n+1) to min(n-1, k_star)
    i_lo = max(0, k_star - n + 1)
    i_hi = min(n - 1, k_star)
    if i_lo <= i_hi:
        j_vals = k_star - np.arange(i_lo, i_hi + 1)
        g[i_lo:i_hi + 1] = 2.0 * a[j_vals]
    return g, k_star


def polyak_lr(f_val, f_lower, g_norm_sq):
    """Polyak step size."""
    if g_norm_sq < 1e-30:
        return 1e-8
    return max(1e-10, (f_val - f_lower) / g_norm_sq)


def arcsine_init(n):
    """Arcsine distribution initialization - gives flatter convolution."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    # Avoid division by zero at endpoints
    denom = np.sqrt(np.maximum(j * (n - 1 - j), 1e-10))
    a = 1.0 / denom
    # Also add a small uniform component to avoid zero at edges
    a = 0.9 * a / a.sum() + 0.1 * np.ones(n) / n
    return a


def uniform_init(n):
    return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def build_conv_matrix(a):
    """Build convolution matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK, cp.CBC]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

        f_val = float(np.max(conv))

        # Polyak-inspired step size
        lr_polyak = polyak_lr(f_val, f_lower, g_norm_sq)
        lr = min(max(lr_polyak, 0.1 * base_lr), 5.0 * base_lr)

        # Update with momentum
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        # Decay momentum slightly
        beta = max(0.5, beta * 0.9999)

        if verbose and it % 5000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        # Reset if stuck
        if it - best_it > 50000:
            a = best_a.copy()
            v = np.zeros(n)
            beta = 0.7
            best_it = it

    return best_a


def subgradient_refine(a, time_budget=100.0, verbose=False):
    """
    Exact subgradient method with Polyak step size.
    Very cheap per iteration (one FFT + array ops).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = uniform_init(n)
    else:
        a = a / s

    f_lower = 1.0 / (2 * n - 1)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.0 / np.sqrt(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        g, k_star = exact_subgradient(a)
        g_norm_sq = float(np.dot(g, g))
        if g_norm_sq < 1e-30:
            break

        f_val = float(np.max(compute_conv(a)))
        # Polyak step
        lr = polyak_lr(f_val, f_lower, g_norm_sq)
        lr = min(max(lr, 1e-8), 5.0 / np.sqrt(n))

        a_new = a - lr * g
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        if verbose and it % 10000 == 0:
            print(f"    SG it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")

        if it - best_it > 100000:
            a = best_a.copy()
            best_it = it

    return best_a


def combined_optimize(a, time_budget=300.0, verbose=False):
    """
    Combined optimization: smoothed GD followed by subgradient refinement.
    """
    start = time.monotonic()
    n = len(a)

    # Phase A: Smoothed GD (70% of time)
    gd_budget = min(time_budget * 0.7, time_budget - 30)
    if gd_budget > 20:
        a = smoothed_gd_optimize(a, time_budget=gd_budget,
                                 tau_init=5.0, tau_end=400.0, verbose=verbose)

    # Phase B: Subgradient refinement (remaining time)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        a = subgradient_refine(a, time_budget=remaining, verbose=verbose)

    return a


def optimize_for_n(n, a_init, time_budget=300.0, use_lp=True, verbose=False):
    """Full optimization pipeline for a given n."""
    start = time.monotonic()
    a = a_init.copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = uniform_init(n)
    else:
        a = a / s

    # Phase 1: Smoothed GD
    gd_budget = time_budget * 0.4
    if gd_budget > 20:
        a = smoothed_gd_optimize(a, time_budget=gd_budget,
                                 tau_init=5.0, tau_end=500.0, verbose=verbose)

    # Phase 2: LP polishing (if n is moderate)
    remaining = time_budget - (time.monotonic() - start)
    if use_lp and n <= 2500 and remaining > 30:
        lp_budget = min(remaining * 0.5, 120.0)
        a = lp_polish(a, time_budget=lp_budget, verbose=verbose)

    # Phase 3: Subgradient refinement
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        a = subgradient_refine(a, time_budget=remaining, verbose=verbose)

    return a


def diverse_init(n, mode, rng):
    """Generate diverse initializations."""
    if mode == 0:
        return uniform_init(n)
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        # Arcsine
        return arcsine_init(n)
    elif mode == 3:
        # Two Gaussians
        c1 = n // 4
        c2 = 3 * n // 4
        w = n // 10
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 4:
        # Cosine window
        x = np.arange(n) / n
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif mode == 5:
        # Polynomial bump
        x = np.linspace(-1, 1, n)
        a = (1 - x ** 2) ** 3
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif mode == 6:
        # Multi-peak
        n_peaks = max(2, n // 200)
        a = np.zeros(n)
        for i in range(n_peaks):
            c = int((i + 0.5) * n / n_peaks)
            w = max(1, n // (n_peaks * 4))
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 0.001)
        return a / a.sum()
    else:
        return uniform_init(n)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Explores multiple sequence lengths with
    structured initializations and combined GD+LP+subgradient optimization.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if parent is None:
        n_init = 800
        parent = uniform_init(n_init)
        best_score = compute_obj(parent)
        best_seq = parent.copy()
        print(f"Default init (n={n_init}): obj={best_score:.8f}")

    # === Phase 1: Optimize parent at original n ===
    print(f"\n=== Phase 1: Optimize parent (n={len(parent)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.15, 200.0)
    if p1_budget > 50:
        a_opt = combined_optimize(parent, time_budget=p1_budget, verbose=True)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj:.8f} (no improvement)")

    # === Phase 2: Explore larger n values ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    n_parent = len(best_seq)

    # Determine n values to try
    n_candidates = sorted(set([
        n_parent,
        n_parent + 200,
        n_parent + 500,
        n_parent + 1000,
        n_parent + 2000,
        n_parent + 3000,
        1500, 2000, 2500, 3000, 4000, 5000,
    ]))
    n_candidates = [n for n in n_candidates if 500 <= n <= 6000]
    n_candidates = sorted(set(n_candidates))

    # Try each n value with time allocation based on promise
    n_results = {}  # n -> (obj, sequence)

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        # Time allocation: more time for promising (larger) n
        # but ensure we can test at least 4-5 values
        n_remaining = len([x for x in n_candidates if x >= n_try])
        alloc_per_n = min(remaining / max(n_remaining, 1) * 0.9, 300.0)
        if alloc_per_n < 40:
            alloc_per_n = min(remaining * 0.7, 200.0)

        print(f"  Trying n={n_try}, budget={alloc_per_n:.0f}s")

        # Multiple initializations
        best_a_for_n = None
        best_obj_for_n = np.inf

        inits_to_try = [
            # Resampled from best
            resample_sequence(best_seq, n_try),
            # Arcsine
            arcsine_init(n_try),
            # Multi-peak
            diverse_init(n_try, 6, rng),
        ]

        # If we have a previous result for a nearby n, use it
        for n_prev in n_results:
            if 0 < n_try - n_prev <= 500:
                inits_to_try.append(resample_sequence(n_results[n_prev][1], n_try))
                break

        # Time per init
        n_inits = len(inits_to_try)
        time_per_init = alloc_per_n / n_inits

        for init_idx, a_init in enumerate(inits_to_try):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(time_per_init, remaining * 0.4)
            if alloc < 20:
                break

            use_lp = n_try <= 2500
            a_opt = optimize_for_n(n_try, a_init, time_budget=alloc,
                                   use_lp=use_lp, verbose=False)
            obj = compute_obj(a_opt)

            if obj < best_obj_for_n:
                best_obj_for_n = obj
                best_a_for_n = a_opt.copy()

        if best_a_for_n is not None:
            n_results[n_try] = (best_obj_for_n, best_a_for_n)
            if best_obj_for_n < best_score:
                best_score = best_obj_for_n
                best_seq = best_a_for_n.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={best_obj_for_n:.8f}")

    # === Phase 3: Intensive polish on best ===
    print(f"\n=== Phase 3: Intensive Polish ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    print(f"  Best so far: n={n_best}, obj={best_score:.8f}")

    if remaining > 100:
        # Intensive combined optimization
        intensive_budget = min(remaining * 0.5, 400.0)
        a_polished = combined_optimize(best_seq, time_budget=intensive_budget,
                                       verbose=True)
        obj = compute_obj(a_polished)
        if obj < best_score:
            best_score = obj
            best_seq = a_polished.copy()
            print(f"  Intensive GD+SG: obj={best_score:.8f} **")
        else:
            print(f"  Intensive GD+SG: obj={obj:.8f}")

    # LP polish if n allows
    remaining = deadline - time.monotonic()
    if n_best <= 2500 and remaining > 50:
        lp_budget = min(remaining * 0.4, 200.0)
        a_lp = lp_polish(best_seq, time_budget=lp_budget, verbose=True)
        obj = compute_obj(a_lp)
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            print(f"  LP Polish: obj={best_score:.8f} **")
        else:
            print(f"  LP Polish: obj={obj:.8f}")

    # Subgradient refinement
    remaining = deadline - time.monotonic()
    if remaining > 30:
        sg_budget = min(remaining * 0.5, 150.0)
        a_sg = subgradient_refine(best_seq, time_budget=sg_budget, verbose=True)
        obj = compute_obj(a_sg)
        if obj < best_score:
            best_score = obj
            best_seq = a_sg.copy()
            print(f"  Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj:.8f}")

    # === Phase 4: Try slightly different n around best ===
    print(f"\n=== Phase 4: Neighboring n ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    neighbor_ns = sorted(set([n_final - 100, n_final - 50, n_final + 50,
                              n_final + 100, n_final + 200]))
    neighbor_ns = [n for n in neighbor_ns if 500 <= n <= 6000 and n != n_final]

    for n_nb in neighbor_ns:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(remaining * 0.3, 120.0)
        if alloc < 25:
            break

        a_nb = resample_sequence(best_seq, n_nb)
        a_nb = combined_optimize(a_nb, time_budget=alloc)
        obj = compute_obj(a_nb)
        if obj < best_score:
            best_score = obj
            best_seq = a_nb.copy()
            print(f"  ** n={n_nb}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_nb}: obj={obj:.8f}")

    # === Phase 5: Final subgradient pass ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = subgradient_refine(best_seq, time_budget=remaining * 0.8)
        obj = compute_obj(a_final)
        if obj < best_score:
            best_score = obj
            best_seq = a_final.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = uniform_init(100)
        best_score = compute_obj(best_seq)

    # Ensure all values are valid for evaluation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        final_score = val
    else:
        # Fallback
        result = [1.0] * 100
        final_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={final_score:.8f}, time={elapsed:.1f}s ===")

    return result
