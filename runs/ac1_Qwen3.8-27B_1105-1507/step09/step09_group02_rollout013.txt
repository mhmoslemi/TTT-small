"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: QP-based flattening + LP best response + subgradient polishing.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_ag(a, g):
    """Compute convolution a*g via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g, size)
    return irfft(fa * fg, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_toeplitz(a):
    """Build the n x n Toeplitz matrix A^T A where (A^T A)[i,j] = (a*a)[j-i]."""
    n = len(a)
    conv = compute_conv(a)  # length 2n-1
    # (A^T A)[i,j] = (a*a)[|j-i|] for |j-i| < n, but actually it's (a*a)[j-i]
    # For linear convolution, (a*a)[k] is defined for k in [-(n-1), n-1]... 
    # Wait, np.convolve gives the full linear convolution of length 2n-1.
    # conv[k] = sum_i a[i]*a[k-i] for k = 0, ..., 2n-2
    # (A^T A)[i,j] = sum_k A[k,i]*A[k,j]
    # A[k,i] = a[k-i] if 0 <= k-i <= n-1
    # So (A^T A)[i,j] = sum_{k: max(i, j) <= k <= min(i+n-1, j+n-1)} a[k-i]*a[k-j]
    # Let t = k-i, then k = t+i, and we need max(i,j) <= t+i <= min(i+n-1, j+n-1)
    # i.e., max(0, j-i) <= t <= min(n-1, n-1+j-i)
    # So (A^T A)[i,j] = sum_{t=max(0,j-i)}^{n-1} a[t]*a[t+j-i]
    # Let m = j-i. If m >= 0: sum_{t=0}^{n-1-m} a[t]*a[t+m] = conv[m]
    # If m < 0: sum_{t=-m}^{n-1} a[t]*a[t+m] = sum_{t=0}^{n-1+m} a[t+m]*a[t] = conv[|m|]
    # So (A^T A)[i,j] = conv[|j-i|] for |j-i| < n.
    
    # Build Toeplitz matrix
    from scipy.linalg import toeplitz
    first_col = conv[:n]  # conv[0], conv[1], ..., conv[n-1]
    return toeplitz(first_col)


def solve_qp_flatten(a, M_bound):
    """
    Solve QP: min g^T Q g s.t. A@g <= M_bound, sum(g)=1, g>=0
    where Q = A^T A (Toeplitz from a's autocorrelation).
    This encourages the convolution a*g to be flat.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build Q = A^T A
    conv_a = compute_conv(a_norm)
    Q = cp.toeplitz(conv_a[:n])  # This should work if cvxpy supports toeplitz
    
    # Actually, let's build it manually to be safe
    Q_mat = np.zeros((n, n))
    for k in range(n):
        for i in range(n - k):
            Q_mat[i, i+k] = conv_a[k]
            Q_mat[i+k, i] = conv_a[k]
    
    # Build A matrix (2n-1 x n)
    A_mat = np.zeros((2*n-1, n))
    for j in range(n):
        A_mat[j:j+n, j] = a_norm
    
    g = cp.Variable(n, nonneg=True)
    M = M_bound  # Fixed bound
    
    objective = cp.Minimize(cp.quad_form(g, Q_mat) + 1e-6 * cp.sum_squares(g))
    constraints = [A_mat @ g <= M * cp.ones(2*n-1), cp.sum(g) == 1]
    
    prob = cp.Problem(objective, constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    A_mat = np.zeros((2*n-1, n))
    for j in range(n):
        A_mat[j:j+n, j] = a_norm
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_qp_combined(a):
    """
    Combined LP+QP: First find M* from LP, then solve QP with that M bound.
    Returns (g_lp, g_qp, M_opt).
    """
    g_lp, M_opt = solve_lp_minpeak(a)
    if g_lp is None:
        return None, None, None
    
    # Now solve QP with M bound slightly above M_opt
    M_bound = M_opt * 1.05 + 1e-8
    g_qp = solve_qp_flatten(a, M_bound)
    
    return g_lp, g_qp, M_opt

def subgradient_refine(a, time_budget, K=10, verbose=False):
    """Subgradient method with top-K peaks and momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]

        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        
        # Exponential weighting for peaks
        temp = max_pv * 0.1
        weights = np.exp((peak_vals - max_pv) / temp)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                j = k - i
                if 0 <= j < n:
                    g[i] += w * a[j]

        # Project gradient onto tangent space of simplex
        g_centroid = g - g.mean()
        
        v = momentum * v + g_centroid
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                lr *= 0.8
                v *= 0.5
                a = best_a
            if stall > 500:
                break
        if lr < 1e-10:
            break

    return best_a


def nesterov_smoothed_descent(a, time_budget, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed max of conv(a,a)."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        # Gradient of smoothed max w.r.t. a
        # d/da_i [sum_k p_k * conv(a,a)[k]] = 2 * sum_k p_k * a[k-i] (with bounds)
        # This is 2 * (p * a) circular... no, linear
        # Actually: conv(a,a)[k] = sum_i a[i]*a[k-i]
        # d/da_m conv(a,a)[k] = sum_i (delta_{im}*a[k-i] + a[i]*delta_{i,k-m})
        # = a[k-m] + a[m-k] (if valid)
        # So grad_m = 2 * sum_k p_k * (a[k-m] + a[m-k]) (with valid index checks)
        # = 2 * sum_k p_k * a[|k-m|]... no, that's for circular.
        # For linear: grad_m = 2 * sum_{k: max(0,m) <= k <= min(2n-2, 2n-2-m+m)} ... 
        # Let's just compute it directly using the correlation
        # grad = 2 * (p * a) where * is cross-correlation
        # (p * a)[m] = sum_k p[k] * a[k-m] (with valid indices)
        # This is the cross-correlation of p and a
        
        # Use FFT for efficiency
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        # Barzilai-Borwein step size
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 20.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9995)

        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 5.0 / np.sqrt(n) * 0.5
            best_it = it

        if verbose and it % 5000 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = a / a.sum()
    elif mode == 3:
        # Triangular (tent)
        x = np.linspace(0, 1, n)
        a = 1 - 2 * np.abs(x - 0.5)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Gaussian bump
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.5)**2)
        a = a / a.sum()
    elif mode == 5:
        # Two peaks
        x = np.linspace(0, 1, n)
        a = np.exp(-50 * (x - 0.3)**2) + np.exp(-50 * (x - 0.7)**2)
        a = a / a.sum()
    elif mode == 6:
        # Sinc-like (windowed)
        x = np.linspace(-0.99, 0.99, n)
        a = np.sinc(x) ** 2
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 7:
        # Parabolic
        x = np.linspace(-1, 1, n)
        a = 1 - x**2
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a
