The gradient of $b_k = \sum_j a_j a_{k-j}$ with respect to $a$ is:

For a fixed $k$: $\nabla_a b_k$ has component $j$ equal to $2a_{k-j}$ if $0 \leq k-j \leq n-1$, else 0.

For the weighted sum: $g_j = 2 \sum_k \lambda_k a_{k-j} \mathbf{1}[0 \leq k-j \leq n-1]$

Let $\lambda$ be a vector of length $2n-1$ (the weights on each bin).

$g_j = 2 \sum_{k=j}^{j+n-1} \lambda_k a_{k-j}$ (where $\lambda_k = 0$ for $k \geq 2n-1$)

Define $m = k-j$: $g_j = 2 \sum_{m=0}^{n-1} \lambda_{j+m} a_m$

This is the cross-correlation of $\lambda$ and $a$: $(\lambda \star a)_j = \sum_m \lambda_{j+m} a_m$

Cross-correlation can be computed as: $\text{IFFT}(\text{FFT}(\lambda) \cdot \overline{\text{FFT}(a)})$

Let me verify: if $X = \text{FFT}(x)$ and $Y = \text{FFT}(y)$, then $\text{IFFT}(X \cdot \bar{Y})_n = \sum_k x_{(n+k) \mod N} \bar{y_k}$

For real signals: $\text{IFFT}(\text{FFT}(\lambda) \cdot \overline{\text{FFT}(a)})_n = \sum_k \lambda_{(n+k) \mod N} a_k$

So for $n = j$: $\sum_k \lambda_{(j+k) \mod N} a_k$

If $N$ is large enough (no wraparound), this equals $\sum_k \lambda_{j+k} a_k$ which is exactly what we want!

So: $g = 2 \cdot \text{IFFT}(\text{FFT}(\lambda, N) \cdot \overline{\text{FFT}(a, N)})[:n]$

Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated proximal gradient with FFT-based smoothed gradients,
LP-based alternating with quadratic line search, coordinate descent polishing,
and multi-length exploration. Key improvements: proper Nesterov schedule,
quadratic line search in LP blending, and adaptive n-search with resampling.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """
    Compute smoothed subgradient of max(conv) w.r.t. a using FFT.
    g_j = 2 * sum_k p_k * a_{k-j} * 1[0<=k-j<=n-1]
    where p_k = exp(tau*(conv_k - max_conv)) / sum_m exp(tau*(conv_m - max_conv))

    n = len(a)
    max_c = float(np.max(conv))
    exp_v = np.exp(tau * (conv - max_c))
    p = exp_v / float(np.sum(exp_v))

    # g_j = 2 * sum_{m=0}^{n-1} p_{j+m} * a_m
    # = 2 * cross_correlation(p, a)_j
    # Use FFT: IFFT(FFT(p) * conj(FFT(a)))[:n]

    N = 1
    while N < 3 * n:
        N <<= 1

    P = fft(p, N)
    A = fft(a, N)
    c = ifft(P * np.conj(A)).real[:n]

    return 2.0 * c


def nesterov_prox_gradient(a, time_budget, tau_init=5.0, tau_end=300.0, verbose=True):
    """
    Nesterov-accelerated proximal gradient on smoothed max-conv objective.
    a should be normalized (sum=1, non-negative).
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    lr = 3.0 / np.sqrt(n)
    it = 0
    stagnation = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        # Compute gradient at y
        conv_y = compute_conv(y)
        g = fft_subgradient(y, conv_y, tau)

        # Nesterov step
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        beta_k = (1.0 - t_k) / (t_k + 1.0)

        a_old = a.copy()
        a_new = y - lr * g
        a_new = project_simplex(a_new)

        y = a_new + beta_k * (a_new - a_old)
        y = project_simplex(y)

        # Evaluate
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Adapt learning rate and reset if stagnating
        if stagnation > 300:
            lr = max(lr * 0.85, 0.1 / np.sqrt(n))
            if stagnation > 1000:
                # Partial reset toward best
                a = best_a + 0.05 * (a_new - best_a)
                a = project_simplex(a)
                y = a.copy()
                lr = 2.0 / np.sqrt(n)
            stagnation = 0

        a = a_new
        it += 1

        if verbose and it % 5000 == 0:
            print(f"    NG it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.5f}, t={elapsed:.0f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. (a*g)_k <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None


def quadratic_line_search(a, g, b_a, b_g, c_cross, n):
    """
    Find t in [0,1] minimizing max_k [(1-t)^2 * b_a_k + t(1-t) * c_k + t^2 * b_g_k].
    Uses golden section search on the 1D piecewise-quadratic function.
    """
    def f_of_t(t):
        val = (1.0 - t) ** 2 * b_a + t * (1.0 - t) * c_cross + t * t * b_g
        return float(np.max(val))

    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f_of_t(c)
    fd = f_of_t(d)

    for _ in range(60):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f_of_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f_of_t(d)

    t_opt = (lo + hi) / 2.0
    # Also check t=0 and t=1
    t0 = f_of_t(0.0)
    t1 = f_of_t(1.0)
    topt_val = f_of_t(t_opt)
    if t0 <= topt_val and t0 <= t1:
        return 0.0
    if t1 <= topt_val and t1 <= t0:
        return 1.0
    return t_opt


def lp_alternating(a, time_budget, verbose=True):
    """
    LP-based alternating: solve LP direction, do quadratic line search blend.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stagnant = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break

        # Compute convolutions for line search
        b_a = compute_conv(a)
        b_g = compute_conv(g_lp)
        # Cross term: c_k = sum_j (a_j * g_{k-j} + a_{k-j} * g_j) = 2*(a*g)_k
        # Build (a*g) via matrix or FFT
        A_mat = build_conv_matrix(a)
        cross = 2.0 * (A_mat @ g_lp)

        # Line search
        t_opt = quadratic_line_search(a, g_lp, b_a, b_g, cross, n)

        if t_opt <= 1e-10:
            stagnant += 1
        else:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = best_a.copy()
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stagnant = 0
            else:
                stagnant += 1
                # Try smaller step
                for t_try in [t_opt * 0.5, t_opt * 0.25, 0.05, 0.02]:
                    a_try = (1.0 - t_try) * a + t_try * g_lp
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        stagnant = 0
                        break

        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, M_lp={M_lp:.6f}, t={elapsed:.0f}s")

        if stagnant > 15:
            # Perturb to escape
            rng = np.random.default_rng(it * 7919)
            scale = 0.01 + 0.005 * min(stagnant, 20)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            stagnant = 0

    return best_a


def coord_descent(a, time_budget, verbose=True, rng=None):
    """
    Coordinate descent with golden section on normalized self-conv peak.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = compute_conv(a)
    best_a = a.copy()
    best_obj = float(np.max(conv))
    start = time.monotonic()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0
    stagnant = 0

    while time.monotonic() - start < time_budget:
        obj_before = float(np.max(conv))

        if rng is not None and sweep % 4 == 3:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for k in order:
            # Direction: d[m] = a[m-k] for valid m
            # conv change: conv + 2*delta*d + delta^2 at position 2k
            d = np.zeros(2 * n - 1)
            lo_idx = max(0, k - (n - 1))
            hi_idx = min(2 * n - 2, k + (n - 1))
            # d[m] = a[m-k] if 0 <= m-k <= n-1
            a_lo = max(0, -k)  # not needed, a is indexed 0..n-1
            a_hi = min(n, 2 * n - 1 - k)
            # For m in [k, k+n-1] intersect [0, 2n-2]: d[m] = a[m-k]
            m_start = max(0, k)
            m_end = min(2 * n - 1, k + n)
            idx = np.arange(m_start, m_end)
            a_idx = idx - k
            # a_idx ranges from m_start-k to m_end-1-k, should be in [0, n-1]
            a_idx = np.clip(a_idx, 0, n - 1)
            d[idx] = a[a_idx]

            # Bounds
            lo_d = -a[k]
            max_cv = float(np.max(conv))
            hi_d = max(2.0 / n, 0.5 * max_cv / max(np.max(d), 1e-10) + 1.0 / n)
            hi_d = hi_d * 2.0  # be generous

            # Golden section
            a_gs = lo_d
            b_gs = hi_d
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)

            def obj_delta(delta):
                val = conv + 2.0 * delta * d
                val[2 * k] += delta * delta
                return float(np.max(val))

            fc = obj_delta(c_gs)
            fd = obj_delta(d_gs)

            for _ in range(40):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = obj_delta(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = obj_delta(d_gs)

            delta_opt = (a_gs + b_gs) / 2.0

            if obj_delta(delta_opt) >= obj_before - 1e-13:
                continue

            a[k] = max(0.0, a[k] + delta_opt)
            conv += 2.0 * delta_opt * d
            conv[2 * k] += delta_opt * delta_opt

            # Renormalize
            S_new = float(np.sum(a))
            if S_new < 1e-15:
                a[:] = 1.0 / n
                conv = compute_conv(a)
            else:
                a /= S_new
                conv /= (S_new * S_new)

        sweep += 1
        obj_after = float(np.max(conv))

        if obj_after < best_obj - 1e-12:
            best_obj = obj_after
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 3:
                if rng is not None:
                    scale = 0.01 * (1 + stagnant * 0.3)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    conv = compute_conv(a)
                    best_obj = float(np.max(conv))
                    stagnant = 0
                else:
                    break

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={2.0*n*obj_after:.8f}, t={elapsed:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation, renormalize."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 0:
        a_new /= s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations on the simplex."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = 1.0 / (np.sqrt(1 - (2*x - 1)**2 + 0.1))
        a = a / a.sum()
    elif mode == 4:
        # Two gaussian bumps
        c1 = n // 4
        c2 = 3 * n // 4
        w = n // 10
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.05, 1, n)
        a = x ** rng.uniform(-1.5, 1.5)
        a = a / a.sum()
    elif mode == 6:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    else:
        a = rng.uniform(0.05, 1, n)
        a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses multi-phase strategy:
    Phase 1: Nesterov proximal gradient (exploration)
    Phase 2: LP alternating with quadratic line search (exploitation)
    Phase 3: Coordinate descent (polish)
    Phase 4: Multi-n exploration
    Phase 5: Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf
    best_n = 800

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                best_n = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_n = 800
        best_seq = np.ones(best_n) / best_n
        best_score = compute_obj(best_seq)
        print(f"Default init (n={best_n}): obj={best_score:.8f}")

    def check_time():
        return time.monotonic() < deadline

    # === Phase 1: Nesterov Proximal Gradient (exploration) ===
    print(f"\n=== Phase 1: Nesterov Proximal Gradient ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 250.0)
    if p1_budget > 30:
        a_ng = nesterov_prox_gradient(best_seq, time_budget=p1_budget,
                                      tau_init=5.0, tau_end=300.0, verbose=True)
        obj_ng = compute_obj(a_ng)
        if obj_ng < best_score:
            best_score = obj_ng
            best_seq = a_ng.copy()
            print(f"  ** NG: obj={best_score:.8f} **")
        else:
            print(f"  NG: obj={obj_ng:.8f}")

    # === Phase 2: LP Alternating ===
    print(f"\n=== Phase 2: LP Alternating ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.30, 350.0)
    if p2_budget > 30:
        # Multiple rounds with decreasing perturbation
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(p2_budget / (3 - rnd), remaining * 0.7)
            if alloc < 20:
                break
            a_lp = lp_alternating(best_seq, time_budget=alloc, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  ** LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
                if rnd < 2:
                    scale = 0.015 * (rnd + 1)
                    best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                    best_seq = np.maximum(best_seq, 0)
                    best_seq = best_seq / best_seq.sum()

    # === Phase 3: Coordinate Descent ===
    print(f"\n=== Phase 3: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.25, 250.0)
    if p3_budget > 30:
        a_cd = coord_descent(best_seq, time_budget=p3_budget, verbose=True, rng=rng)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  ** CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Generate candidate n values, spread around current best
    n_candidates = []
    for delta in [-200, -100, -50, 50, 100, 200, 300, 400]:
        n_try = n_current + delta
        if 300 <= n_try <= 2000:
            n_candidates.append(n_try)
    # Add some absolute values
    for n_abs in [400, 500, 600, 700, 900, 1000, 1200, 1500, 2000]:
        if 300 <= n_abs <= 2000 and abs(n_abs - n_current) > 30:
            n_candidates.append(n_abs)

    n_candidates = sorted(set(n_candidates))
    # Limit to top 6 candidates by distance from current (prefer nearby)
    n_candidates = sorted(n_candidates, key=lambda x: abs(x - n_current))[:6]
    print(f"  Trying n values: {n_candidates}")

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick NG
        alloc1 = min(remaining * 0.25, 60.0)
        if alloc1 > 20:
            a_try = nesterov_prox_gradient(a_try, time_budget=alloc1,
                                           tau_init=10.0, tau_end=200.0, verbose=False)

        # LP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.35, 80.0)
        if alloc2 > 15:
            a_try = lp_alternating(a_try, time_budget=alloc2, verbose=False)

        # CD
        remaining = deadline - time.monotonic()
        alloc3 = min(remaining * 0.25, 50.0)
        if alloc3 > 15:
            a_try = coord_descent(a_try, time_budget=alloc3, verbose=False, rng=rng)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            best_n = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Final Intensive Polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP
    if remaining > 30:
        alloc = min(remaining * 0.30, 80.0)
        a_f1 = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  ** Final LP: obj={best_score:.8f} **")

    # CD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.30, 60.0)
        a_f2 = coord_descent(best_seq, time_budget=alloc, verbose=True, rng=rng)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  ** Final CD: obj={best_score:.8f} **")

    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.40, 50.0)
        a_f3 = lp_alternating(best_seq, time_budget=alloc, verbose=False)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  ** Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
