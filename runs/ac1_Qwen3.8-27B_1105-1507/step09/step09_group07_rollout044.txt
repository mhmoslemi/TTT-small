import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a[:n]
    return A


def solve_lp_fast(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    c[-1] = 1.0

    A_ub[:, :n] = A
    A_ub[:, n] = -1.0

    A_eq = np.zeros((1, n + 1))

    start = time.monotonic()

    # Polyak step size
    best_h = float(np.max(compute_conv(a)))
    lr_base = 0.1

    avg_a = a.copy()
    count = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        current_val = float(conv[k_star])

        # Subgradient of max(conv(a,a)) w.r.t. a[i]
        # d/d a[i] of conv(a,a)[k] = a[k-i] (if 0 <= k-i < n)
        subgrad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                subgrad[i] = a[j]

        # Also consider that the sum constraint means we should project
        # Simple projected subgradient: a_new = a - lr * subgrad, then project to simplex
        lr = lr_base / (np.linalg.norm(subgrad) + 1e-10)
        lr = min(lr, 0.5)

        a_trial = a - lr * subgrad
        a_trial = np.maximum(a_trial, 0)
        s_trial = float(np.sum(a_trial))
        if s_trial > 1e-10:
            a_trial = a_trial / s_trial

        obj_trial = compute_obj(a_trial)
        count += 1
        avg_a += a_trial / count

        if obj_trial < best_obj - 1e-10:
            best_obj = obj_trial
            best_a = a_trial.copy()
            best_h = current_val
            a = a_trial.copy()
        elif count % 500 == 0:
            # Try the averaged solution
            avg_a = np.maximum(avg_a, 0)
            s_avg = float(np.sum(avg_a))
            if s_avg > 1e-10:
                avg_a = avg_a / s_avg
                obj_avg = compute_obj(avg_a)
                if obj_avg < best_obj:
                    best_obj = obj_avg
                    best_a = avg_a.copy()
                    a = avg_a.copy()
                    if verbose:
                        print(f"          Subgrad (avg): obj={best_obj:.8f}")

        # Periodic restart
        if count % 2000 == 0:
            a = best_a + 0.05 * np.random.default_rng(count).normal(0, 1, n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            avg_a = a.copy()
            count = 0

        if verbose and count % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"          Subgrad it{count}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_a


def continuous_init(n, mode=0):
    """
    Generate initializations inspired by continuous relaxations.
    Functions on [0,1] with integral 1, known to have flat autocorrelation.
    """
    x = np.linspace(0, 1, n)
    if mode == 0:
        # Uniform
        a = np.ones(n)
    elif mode == 1:
        # Cosine window (zero at endpoints)
        a = np.sin(np.pi * x) ** 2
        a = np.maximum(a, 1e-6)
    elif mode == 2:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = np.maximum(a, 1e-6)
    elif mode == 3:
        # Sinc-like (bandlimited)
        a = np.sinc(3 * (x - 0.5)) ** 2
        a = np.maximum(a, 1e-6)
    elif mode == 4:
        # Two peaks
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    elif mode == 5:
        # Power law
        a = x ** 0.5 + (1 - x) ** 0.5 + 0.1
    elif mode == 6:
        # Modulated
        a = 1 + 0.3 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif mode == 7:
        # Three peaks
        a = (np.exp(-30 * (x - 0.2) ** 2) + np.exp(-30 * (x - 0.5) ** 2) +
             np.exp(-30 * (x - 0.8) ** 2))
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 0:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating with dual-guided perturbation, n-sweep,
    continuous-inspired initializations, and subgradient polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = continuous_init(800, mode=0)
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"Initial: n={n_current}, obj={best_score:.8f}, target=1.5027")

    # === Phase 1: Quick LP polish on parent (8%) ===
    print(f"\n=== Phase 1: Quick LP Polish ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 80.0)
    if p1_budget > 15:
        a_p1 = lp_alternate_v2(best_seq, time_budget=p1_budget, verbose=False)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_p1:.8f} (no improvement)")

    # === Phase 2: n-sweep with continuous inits (22%) ===
    print(f"\n=== Phase 2: n-Sweep ===")
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.22

    n_options = sorted(set([
        600, 700, 750, 800, 850, 900, 950, 1000, 1050, 1100,
        1150, 1200, 1300, 1400, 1500, 1600, 1800, 2000,
    ]))
    # Add values around current best
    for delta in [-100, -50, -20, 0, 20, 50, 100, 200, 300]:
        n_try = n_current + delta
        if 500 <= n_try <= 2500:
            n_options.append(n_try)
    n_options = sorted(set(n for n in n_options if 500 <= n <= 2500))

    per_n_budget = p2_budget / max(1, len(n_options))
    print(f"  Trying {len(n_options)} n values, ~{per_n_budget:.1f}s each")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_n_budget * 1.5, remaining * 0.12)
        if alloc < 10:
            alloc = 10

        # Use resampled parent AND a continuous init
        a_resample = resample_sequence(best_seq, n_try)
        a_cont = continuous_init(n_try, mode=n_try % 8)
        # Blend
        a_init = 0.6 * a_resample + 0.4 * a_cont
        a_init = np.maximum(a_init, 0)
        a_init = a_init / a_init.sum()

        a_opt = lp_alternate_v2(a_init, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)
        marker = " **" if obj_opt < best_score else ""
        if marker or n_try in [800, 1000, 1200, 1500]:
            print(f"  n={n_try}: obj={obj_opt:.8f}{marker}")

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try

    # === Phase 3: Intensive LP on best n (35%) ===
    print(f"\n=== Phase 3: Intensive LP (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.35, 600.0)

    n_rounds = 5
    per_round = p3_budget / n_rounds
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        alloc = min(per_round * 1.3, remaining * 0.22)
        if alloc < 20:
            break

        if rnd == 0:
            a_start = best_seq.copy()
        elif rnd == 1:
            a_start = resample_sequence(best_seq, n_current + 10)
            a_start = resample_sequence(a_start, n_current)
        elif rnd == 2:
            a_cont = continuous_init(n_current, mode=rnd)
            a_start = 0.7 * best_seq + 0.3 * a_cont
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        elif rnd == 3:
            rng_r = np.random.default_rng(rnd * 7919 + n_current)
            a_start = best_seq * np.exp(rng_r.normal(0, 0.05, n_current))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        else:
            rng_r = np.random.default_rng(99999 + n_current)
            a_start = best_seq * np.exp(rng_r.normal(0, 0.1, n_current))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()

        a_lp = lp_alternate_v2(a_start, time_budget=alloc, verbose=(rnd == 0))
        obj_lp = compute_obj(a_lp)
        marker = " **" if obj_lp < best_score else ""
        print(f"  LP round {rnd}: obj={obj_lp:.8f}{marker}")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()

    # === Phase 4: Subgradient + diverse restarts (15%) ===
    print(f"\n=== Phase 4: Subgradient + Restarts ===")
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.15

    # Subgradient on current best
    alloc_sg = min(p4_budget * 0.4, 60.0)
    if alloc_sg > 10:
        a_sg = subgradient_descent(best_seq, time_budget=alloc_sg, verbose=False)
        obj_sg = compute_obj(a_sg)
        marker = " **" if obj_sg < best_score else ""
        print(f"  Subgradient: obj={obj_sg:.8f}{marker}")
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()

    # Diverse restarts
    remaining = deadline - time.monotonic()
    n_restarts = 4
    per_restart = min(p4_budget * 0.6 / n_restarts, 30.0)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_restart, remaining * 0.15)
        if alloc < 10:
            break

        mode = (r * 2 + 3) % 8
        a_r = continuous_init(n_current, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        a_r = lp_alternate_v2(a_r, time_budget=alloc, verbose=False)
        obj_r = compute_obj(a_r)
        marker = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}{marker}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()

    # === Phase 5: Fine n adjustment (10%) ===
    print(f"\n=== Phase 5: Fine n Adjustment ===")
    remaining = deadline - time.monotonic()
    p5_budget = remaining * 0.10

    fine_ns = sorted(set([n_current - 5, n_current - 3, n_current - 1,
                          n_current + 1, n_current + 3, n_current + 5]))
    fine_ns = [n for n in fine_ns if n >= 100]
    per_fine = p5_budget / max(1, len(fine_ns))

    for n_fine in fine_ns:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(per_fine, remaining * 0.15)
        if alloc < 8:
            break
        a_f = resample_sequence(best_seq, n_fine)
        a_f = lp_alternate_v2(a_f, time_budget=alloc, verbose=False)
        obj_f = compute_obj(a_f)
        marker = " **" if obj_f < best_score else ""
        print(f"  Fine n={n_fine}: obj={obj_f:.8f}{marker}")
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()

    # === Phase 6: Final polish (10%) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    alloc = min(remaining * 0.5, 120.0)
    if alloc > 20:
        a_final = lp_alternate_v2(best_seq, time_budget=alloc, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = lp_alternate_v2(best_seq, time_budget=min(remaining * 0.5, 40.0), verbose=False)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: obj={obj_last:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"=== Target: 1.5027, Achieved: {best_score:.8f}, Gap: {best_score - 1.5027:.8f} ===")

    return result
