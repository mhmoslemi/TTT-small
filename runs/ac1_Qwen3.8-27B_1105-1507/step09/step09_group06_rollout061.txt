Uses iterative LP (alternating optimization), spectral equalization,

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * np.max(conv)

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        r = min(n, 2*n-1-j)
        A[j:j+r, j] = a[:r]
    return A

def solve_lp_for_g(a_norm):
    """Solve min M s.t. conv(a_norm, g)_k <= M, sum g = 1, g >= 0"""
    n = len(a_norm)
    A_mat = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

        except:
            continue
    return None, None

def spectral_equalize(a, beta=0.1, iters=5):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    N = 1
    while N < 2*n:
        N <<= 1
    for _ in range(iters):
        A = fft(a, N)
        power = np.abs(A)**2
        # Reduce peak power
        max_power = np.max(power)
        if max_power < 1e-15:
            break
        # Apply inverse power filter
        A_new = A * (np.abs(A)**(-2*beta))
        a_new = np.real(ifft(A_new, N))[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new
    return a

def peak_targeted_reduction(a, top_k=5, scale=0.02):
    """Reduce values that contribute most to the peak convolution lags."""
    n = len(a)
    conv = compute_conv(a)
    peak_lags = np.argsort(conv)[::-1][:top_k]
    
    a_new = a.copy()
    for lag in peak_lags:
        lag = int(lag)
        # a[i] * a[lag-i] contributes to conv[lag]
        # Reduce a[i] where a[i]*a[lag-i] is large
        for i in range(max(0, lag-n+1), min(n, lag+1)):

            if 0 <= j < n:

                a_new[i] -= scale * contribution
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()

    return a_new

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

    return A_mat

def solve_lp_cvxpy(a_norm):
    """Fallback LP solver using cvxpy."""
    import cvxpy as cp
    n = len(a_norm)
    A_mat = build_conv_matrix(a_norm)

def blend_linear(a, g, t):
    """Linear blend: (1-t)*a + t*g, normalized."""
    a_new = (1.0 - t) * a + t * g
    a_new = np.maximum(a_new, 0)
    s = float(a_new.sum())
    if s < 1e-10:
        return None
    return a_new / s


def blend_geometric(a, g, alpha):
    """Geometric blend: a^alpha * g^(1-alpha), normalized. alpha close to 1 means closer to a."""
    eps = 1e-30
    a_new = np.power(np.maximum(a, eps), alpha) * np.power(np.maximum(g, eps), 1.0 - alpha)
    a_new = np.maximum(a_new, 0)
    s = float(a_new.sum())
    if s < 1e-10:
        return None
    return a_new / s


def peak_targeted_perturb(a, conv, scale=0.02, n_peaks=5, seed=0):
    """Perturb a by reducing values that contribute most to peak convolution lags."""
    n = len(a)
    rng = np.random.default_rng(seed)
    peak_lags = np.argsort(conv)[::-1][:n_peaks]

    a_new = a.copy()
    for lag in peak_lags:
        lag = int(lag)
        i_start = max(0, lag - n + 1)
        i_end = min(n - 1, lag)
        for i in range(i_start, i_end + 1):
            j = lag - i
            if 0 <= j < n:
                contrib = a[i] * a[j]
                a_new[i] -= scale * contrib * rng.uniform(0.5, 1.5)

    a_new = np.maximum(a_new, 0)
    s = float(a_new.sum())
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_perturb(a, scale=0.03, seed=0):
    """Multiplicative random perturbation."""
    rng = np.random.default_rng(seed)
    noise = rng.normal(0, 1, len(a)) * scale
    a_new = a * np.exp(noise)
    a_new = np.maximum(a_new, 0)
    s = float(a_new.sum())
    if s < 1e-10:
        return a.copy()
    return a_new / s

    3. Repeat until time runs out or convergence

    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    consecutive_no_improve = 0
    prev_a = a.copy()

    while time.monotonic() - start < time_budget:
        a_norm = a / a.sum()
        g, M = solve_lp_fast(a_norm)
        if g is None:
            g, M = solve_lp_cvxpy(a_norm)
        if g is None:
            break

        candidates = []

        if use_geometric:

                a_try = blend_geometric(a, g, alpha)
                if a_try is not None:
                    candidates.append(a_try)
        else:
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                a_try = blend_linear(a, g, t)
                if a_try is not None:
                    candidates.append(a_try)

        # Also try momentum-based update
        momentum = 0.3 * (a - prev_a)
        a_mom = a + momentum
        a_mom = np.maximum(a_mom, 0)
        s_mom = a_mom.sum()
        if s_mom > 1e-10:
            a_mom = a_mom / s_mom
            candidates.append(a_mom)

        best_local = best_a.copy()
        best_local_obj = best_obj

        for a_try in candidates:
            obj = compute_obj(a_try)
            if obj < best_local_obj - 1e-13:
                best_local_obj = obj
                best_local = a_try.copy()

        if best_local_obj < best_obj - 1e-13:
            best_obj = best_local_obj
            best_a = best_local.copy()
            prev_a = a.copy()
            a = best_local.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            prev_a = a.copy()

                conv = compute_conv(a)
                scale = 0.01 + 0.005 * min(consecutive_no_improve // 5, 10)
                a = peak_targeted_perturb(a, conv, scale=scale, n_peaks=5,
                                          seed=seed + it)
                a = 0.85 * a + 0.15 * best_a
                a = np.maximum(a, 0)

            print(f"    LP it{it}: obj={best_obj:.8f}, M={M:.6f}, t={elapsed:.0f}s")

def coordinate_descent_fast(a, time_budget=50.0, seed=0):
    """Fast coordinate descent: try adjusting pairs of coordinates."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        # Pick random pairs to adjust
        for _ in range(50):
            if time.monotonic() - start >= time_budget:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            old_i = a[i]
            old_j = a[j]

            # Try transferring mass from i to j and vice versa
            for delta in [0.001, 0.005, 0.01, 0.02, 0.05]:
                # Transfer from i to j
                if old_i > delta:
                    a_try = a.copy()
                    a_try[i] = old_i - delta
                    a_try[j] = old_j + delta
                    s = a_try.sum()
                    if s > 0:
                        a_try = a_try / s
                        obj = compute_obj(a_try)
                        if obj < best_obj - 1e-13:
                            best_obj = obj
                            best_a = a_try.copy()
                            a = a_try.copy()
                            old_i = a[i]

                # Transfer from j to i
                if old_j > delta:
                    a_try = a.copy()
                    a_try[j] = old_j - delta
                    a_try[i] = old_i + delta
                    s = a_try.sum()
                    if s > 0:
                        a_try = a_try / s
                        obj = compute_obj(a_try)
                        if obj < best_obj - 1e-13:
                            best_obj = obj
                            best_a = a_try.copy()
                            a = a_try.copy()
                            old_j = a[j]

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        c1, c2 = n // 4, 3 * n // 4
        w = n // 8
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Semicircle (Wigner semicircle-like)
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x ** 2, 0)) + 1e-10
        a = a / a.sum()
    elif mode == 4:
        # Smooth random (low-pass filtered)
        a = rng.uniform(0, 1, n)
        k = max(3, n // 30)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        # Three bumps
        centers = [n // 6, n // 2, 5 * n // 6]
        w = n // 10
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 8:
        # Alternating high-low (smoothed)
        a = np.ones(n)
        a[::2] = rng.uniform(0.5, 1.5)
        a[1::2] = rng.uniform(0.1, 0.5)
        k = max(3, n // 20)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

    print(f"\n=== Phase 1: LP Iteration (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.2, 250.0)
    if p1_budget > 30:
        a_opt = lp_iteration(best_seq, time_budget=p1_budget,
                             verbose=True, seed=seed, use_geometric=True)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Phase1 (geometric): obj={best_score:.8f} **")
        else:
            print(f"  Phase1 (geometric): obj={obj:.8f}")

        # Also try linear blending
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_opt2 = lp_iteration(best_seq, time_budget=min(remaining * 0.15, 100.0),
                                  verbose=False, seed=seed + 100, use_geometric=False)
            obj2 = compute_obj(a_opt2)
            if obj2 < best_score:
                best_score = obj2
                best_seq = a_opt2.copy()
                print(f"  Phase1 (linear): obj={best_score:.8f} **")
            else:
                print(f"  Phase1 (linear): obj={obj2:.8f}")

    # === Phase 2: Diverse restarts at current n ===
    print(f"\n=== Phase 2: Diverse Restarts (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    n_restart_budget = min(remaining * 0.3, 400.0)
    if n_restart_budget > 60:
        n_restarts = 8
        per_restart = n_restart_budget / n_restarts

        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            mode = r % 9
            a_r = diverse_init(n_current, rng, mode=mode)
            # Blend with best to keep it in the right basin
            blend_frac = 0.3 + 0.05 * r
            a_r = (1 - blend_frac) * a_r + blend_frac * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

            alloc = min(per_restart * 0.8, remaining * 0.25)
            if alloc > 20:
                a_r = lp_iteration(a_r, time_budget=alloc,
                                   verbose=False, seed=seed + 1000 + r,
                                   use_geometric=(r % 2 == 0))
                obj_r = compute_obj(a_r)
                if obj_r < best_score:
                    best_score = obj_r
                    best_seq = a_r.copy()
                    print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
                else:
                    if r % 2 == 0:
                        print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 3: Multi-n exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Choose n values to explore
    n_options = set()
    n_options.add(max(400, n_current - 200))
    n_options.add(max(400, n_current - 100))
    n_options.add(n_current + 100)
    n_options.add(n_current + 200)
    n_options.add(n_current + 400)
    n_options.add(n_current + 600)
    n_options.add(1000)
    n_options.add(1200)
    n_options.add(1500)
    n_options.add(2000)

    # Filter to reasonable range
    n_options = sorted([n for n in n_options if 400 <= n <= 2500 and n != n_current])
    n_options = n_options[:6]  # Limit to 6 different lengths

    n_explore_total = min(remaining * 0.3, 450.0)
    per_n = n_explore_total / max(1, len(n_options))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        alloc = min(per_n * 0.7, remaining * 0.2)
        if alloc > 20:
            a_try = lp_iteration(a_try, time_budget=alloc,
                                 verbose=False, seed=seed + 5000 + n_try,
                                 use_geometric=True)

        remaining = deadline - time.monotonic()
        if remaining > 25:
            # Quick polish
            a_try = lp_iteration(a_try, time_budget=min(remaining * 0.08, 40.0),
                                 verbose=False, seed=seed + 6000 + n_try,
                                 use_geometric=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Intensive polish on best solution ===
    print(f"\n=== Phase 4: Intensive Polish (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()

    if remaining > 60:
        # LP iteration with geometric blending
        alloc = min(remaining * 0.3, 150.0)
        a_p = lp_iteration(best_seq, time_budget=alloc,
                           verbose=True, seed=seed + 7000, use_geometric=True)
        obj_p = compute_obj(a_p)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Polish (geo): obj={best_score:.8f} **")
        else:
            print(f"  Polish (geo): obj={obj_p:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 40:
        # Try a few more perturbations + LP
        for p in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            conv = compute_conv(best_seq)
            a_pert = peak_targeted_perturb(best_seq, conv, scale=0.03,
                                           n_peaks=8, seed=seed + 8000 + p)
            a_pert = 0.7 * a_pert + 0.3 * best_seq
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()

            alloc = min(remaining * 0.2, 60.0)
            if alloc > 15:
                a_pert = lp_iteration(a_pert, time_budget=alloc,
                                      verbose=False, seed=seed + 9000 + p,
                                      use_geometric=True)
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  Perturb {p}: obj={best_score:.8f} **")

    # === Phase 5: Coordinate descent final polish ===
    print(f"\n=== Phase 5: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 80.0)
        a_cd = coordinate_descent_fast(best_seq, time_budget=alloc, seed=seed + 10000)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 6: Final LP burst ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 50.0)
        a_final = lp_iteration(best_seq, time_budget=alloc,
                               verbose=False, seed=seed + 11000, use_geometric=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
