"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses aggressive LP-based alternating optimization with warm-starting, geometric
step-size search, active-set reduction, multi-n exploration, and KKT-guided
perturbation for escaping local minima.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build dense matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a[k-j] for valid k-j in [0, n-1]
        lo = max(0, j)
        hi = min(2 * n - 2, j + n - 1)
        for k in range(lo, hi + 1):
            A[k, j] = a[k - j]
    return A


def build_conv_matrix_fast(a):
    """Build dense matrix A (2n-1 x n) such that A @ g = conv(a, g) using vectorized ops."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = max(0, j)
        hi = min(2 * n - 1, j + n)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_lp_for_fixed_a(a, warm_g=None, use_top_frac=1.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    # Build constraint matrix: A[k,j] = a_norm[k-j] if valid
    A = build_conv_matrix_fast(a_norm)
    
    # For large n, use only top fraction of constraints
    if use_top_frac < 1.0:
        conv_aa = compute_conv(a_norm)
        n_constraints = int(len(conv_aa) * use_top_frac)
        # Keep the top n_constraints rows by current conv value
        top_idx = np.argsort(conv_aa)[::-1][:n_constraints]
        A = A[top_idx, :]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    # Warm start
    if warm_g is not None:
        g.value = warm_g
        M.value = float(np.max(A @ warm_g)) + 0.01
    
    constraints = [A @ g <= M, cp.sum(g) == 1]
    objective = cp.Minimize(M)
    prob = cp.Problem(objective, constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(np.max(A @ g_val))
                    return g_val, M_val
        except Exception:
            continue
    
    return None, None

def coord_descent(a, time_budget=60.0, rng=None, verbose=False):

    Coordinate descent with golden section search on 1D convex subproblems.
    Uses incremental convolution updates.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S
    conv = compute_conv(a)
    
    best_a = a.copy()
    best_obj = float(np.max(conv))
    gr = (np.sqrt(5) - 1) / 2.0
    start = time.monotonic()
    stagnant = 0
    
    for sweep in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        obj_before = float(np.max(conv))
        
        # Cyclic order
        order = np.arange(n)
        
        any_update = False
        for k in order:
            # Direction: changing a[k] by delta changes conv by 2*delta*d + delta^2 at 2k
            d = np.zeros(2 * n - 1)
            d[k:k + n] = a
            
            lo_delta = -a[k]
            # Estimate upper bound
            max_c = float(np.max(conv))
            upper_est = np.sqrt(max(0.0, max_c - conv[2 * k] + a[k] ** 2 + 0.1 * max_c)) - a[k]
            hi_delta = max(upper_est, 3.0 / n)
            hi_delta = max(hi_delta, lo_delta + 2.0 / n)
            
            # Golden section search (30 iterations for speed)
            a_gs = lo_delta
            b_gs = hi_delta
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)
            
            def obj_at_delta(delta):
                val = conv + 2.0 * delta * d
                val[2 * k] += delta * delta
                return float(np.max(val))
            
            fc = obj_at_delta(c_gs)
            fd = obj_at_delta(d_gs)
            
            for _ in range(30):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = obj_at_delta(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = obj_at_delta(d_gs)
            
            delta_opt = (a_gs + b_gs) / 2.0
            new_obj_est = obj_at_delta(delta_opt)
            
            if new_obj_est < obj_before - 1e-14:
                a[k] += delta_opt
                a[k] = max(0.0, a[k])
                conv += 2.0 * delta_opt * d
                conv[2 * k] += delta_opt * delta_opt
                
                # Re-normalize
                S_new = float(np.sum(a))
                if S_new < 1e-10:
                    a[:] = 1.0 / n
                    conv = compute_conv(a)
                    break
                else:
                    a /= S_new
                    conv /= (S_new * S_new)
                any_update = True
        
        obj_after = float(np.max(conv))

            best_obj = obj_after
            best_a = a.copy()
            stagnant = 0

            stagnant += 1

            elapsed = time.monotonic() - start
            print(f"      CD sweep {sweep}: obj={2.0*n*best_obj:.10f}, elapsed={elapsed:.0f}s")
        
        if stagnant > 3:
            if rng is not None:
                scale = 0.02 * (1 + stagnant * 0.3)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv = compute_conv(a)
                best_obj = float(np.max(conv))
                stagnant = 0
            else:
                break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        c1 = n // 4
        c2 = 3 * n // 4
        w = max(1, n // 10)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-8)
        a = a / a.sum()
    elif mode == 4:
        # Half-sine
        x = np.linspace(0, np.pi, n)
        a = np.sin(x)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Cosine window
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 7:
        # Three peaks
        centers = [n//5, n//2, 4*n//5]
        a = np.zeros(n)
        for c in centers:
            w = max(1, n // 15)
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 8:
        # Random smooth (low-pass filtered noise)
        a = rng.uniform(0, 1, n)
        # Apply simple moving average
        k = max(3, n // 50)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def kkt_guided_step(a, rng, n_perturb=20):
    """
    KKT-inspired perturbation: identify coordinates where the KKT condition
    is most violated and perturb them preferentially.
    """
    n = len(a)
    a = a / a.sum()
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    
    # Active set: indices where conv is near max
    active = np.where(conv >= max_val * 0.99)[0]
    
    # Compute "shadow gradient": v[i] = 2 * sum_{k in active} (1/len(active)) * a[k-i]
    v = np.zeros(n)
    p = 1.0 / len(active)
    for k in active:
        for j in range(n):
            ki = k - j
            if 0 <= ki < n:
                v[j] += p * 2.0 * a[ki]
    
    # KKT violation for positive coordinates
    pos_mask = a > 1e-6
    if np.any(pos_mask):
        lambda_val = np.median(v[pos_mask])
        violation = np.abs(v - lambda_val)
    else:
        violation = np.zeros(n)
    
    # Perturb the most violated coordinates
    top_violated = np.argsort(violation)[::-1][:min(n_perturb, n)]
    scale = 0.03
    a_new = a.copy()
    a_new[top_violated] *= np.exp(rng.normal(0, scale, len(top_violated)))
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new /= s
    else:
        a_new = a
    return a_new


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Aggressive LP polishing + coordinate descent + multi-n + KKT perturbation.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception:
        pass
    
    n_target = len(best_seq) if best_seq is not None else 800
    
    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.10f}")
    
    # === Phase 1: Quick Coordinate Descent ===
    print(f"\n=== Phase 1: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 80.0)
    if p1_budget > 10:
        a_cd = coord_descent(best_seq, time_budget=p1_budget, rng=rng, verbose=False)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase1 CD: obj={best_score:.10f} **")
        else:
            print(f"  Phase1 CD: obj={obj_cd:.10f}")
    
    # === Phase 2: Aggressive LP Polishing (main workhorse) ===
    print(f"\n=== Phase 2: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.45, 600.0)
    if p2_budget > 50:
        # Split into multiple rounds with perturbation between
        n_rounds = 3
        for r in range(n_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(p2_budget / (n_rounds - r), remaining * 0.8)
            if alloc < 20:
                break
            
            a_lp, obj_lp = lp_polish_aggressive(best_seq, time_budget=alloc, rng=rng, verbose=True)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {r}: obj={best_score:.10f} **")
            else:
                print(f"  LP round {r}: obj={obj_lp:.10f}")
            
            # Perturb between rounds
            if r < n_rounds - 1:
                best_seq = kkt_guided_step(best_seq, rng, n_perturb=max(10, len(best_seq) // 20))
                obj_check = compute_obj(best_seq)
                if obj_check > best_score * 1.01:
                    best_seq = best_seq * np.exp(rng.normal(0, 0.01, len(best_seq)))
                    best_seq = np.maximum(best_seq, 0)
                    best_seq = best_seq / best_seq.sum()
    
    # === Phase 3: Multi-n Exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    
    n_options = [n_current + 200, n_current + 400, n_current + 600, n_current + 1000]
    n_options = [n for n in n_options if 500 <= n <= 3000]
    # Also try slightly larger
    n_options = list(set(n_options + [1000, 1200, 1500, 2000]))
    n_options = sorted([n for n in n_options if abs(n - n_current) >= 100])[:4]
    
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 120:
            break
        
        print(f"  Trying n={n_try}...")
        a_try = resample_sequence(best_seq, n_try)
        
        # Quick CD
        alloc1 = min(remaining * 0.2, 40.0)
        if alloc1 > 10:
            a_try = coord_descent(a_try, time_budget=alloc1, rng=rng)
        
        # LP polish
        remaining2 = deadline - time.monotonic()
        alloc2 = min(remaining2 * 0.3, 100.0)
        if alloc2 > 30:
            a_try, _ = lp_polish_aggressive(a_try, time_budget=alloc2, rng=rng)
        
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")
    
    # === Phase 4: Final Intensive Polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()
    
    # CD
    if remaining > 40:
        alloc = min(remaining * 0.2, 60.0)
        a_f = coord_descent(best_seq, time_budget=alloc, rng=rng)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final CD: obj={best_score:.10f} **")
    
    # LP (most important final step)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 200.0)
        a_f, obj_f = lp_polish_aggressive(best_seq, time_budget=alloc, rng=rng)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.10f} **")
    
    # One more CD + LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 30.0)
        a_f = coord_descent(best_seq, time_budget=alloc, rng=rng)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
    
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 80.0)
        a_f, obj_f = lp_polish_aggressive(best_seq, time_budget=alloc, rng=rng)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP2: obj={best_score:.10f} **")
    
    # === Phase 5: Try a few diverse restarts if time remains ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 5: Diverse Restarts (remaining={remaining:.0f}s) ===")
        n_best = len(best_seq)
        for r in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            mode = (r * 3 + 2) % 9
            a_r = diverse_init(n_best, rng, mode=mode)
            blend = 0.4 + 0.1 * r
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
            
            alloc1 = min(remaining * 0.3, 30.0)
            if alloc1 > 10:
                a_r = coord_descent(a_r, time_budget=alloc1, rng=rng)
            
            remaining2 = deadline - time.monotonic()
            alloc2 = min(remaining2 * 0.3, 60.0)
            if alloc2 > 20:
                a_r, _ = lp_polish_aggressive(a_r, time_budget=alloc2, rng=rng)
            
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.10f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.10f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    
    return result
