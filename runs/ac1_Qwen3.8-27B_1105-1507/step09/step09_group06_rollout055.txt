from scipy.fft import rfft, irfft
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""

    if n == 0:

        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def top_k_indices(conv, k=10):
    """Return indices of top k values in conv."""
    return np.argsort(conv)[::-1][:k]


def build_conv_matrix_rows(a, row_indices):
    """Build sparse rows of convolution matrix for given row indices.
    Returns matrix A such that A @ g = [conv(a,g)[idx] for idx in row_indices]."""
    n = len(a)
    rows = np.zeros((len(row_indices), n))
    for r, k in enumerate(row_indices):
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if 0 <= j < n:
                rows[r, i] += a[j]
    return rows


def solve_top_k_lp(a, k=10):
    """
    Solve LP: min M s.t. (A_a[top_k]) @ g <= M, sum(g)=1, g>=0.
    Only uses the top-k autocorrelation constraints for speed and focus.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv = compute_conv(a_norm)
    top_k_idx = top_k_indices(conv, k)
    A_rows = build_conv_matrix_rows(a_norm, top_k_idx)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_rows @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_full_lp(a):
    """Solve the full LP with all 2n-1 constraints."""
    n = len(a)
    a_norm = a / np.sum(a)
    A_mat = build_conv_matrix_rows(a_norm, np.arange(2 * n - 1))

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def golden_section_line_search(a, direction, obj_func, max_iter=30):
    """
    Golden section search along path (1-t)*a + t*direction for t in [0, 1].
    Returns optimal t.
    """
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ~0.618

    lo, hi = 0.0, 1.0
    c = lo + (1 - gr) * (hi - lo)
    d = lo + gr * (hi - lo)

    a_norm = a / a.sum()
    d_norm = direction / direction.sum() if direction.sum() > 0 else a / a.sum()

    fc = obj_func((1 - c) * a_norm + c * d_norm)
    fd = obj_func((1 - d) * a_norm + d * d_norm)

    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + (1 - gr) * (hi - lo)
            fc = obj_func((1 - c) * a_norm + c * d_norm)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_func((1 - d) * a_norm + d * d_norm)

    return (lo + hi) / 2


def smoothed_gd(a, time_budget=150.0, tau_init=20.0, tau_end=800.0, seed=0):
    """
    Nesterov accelerated smoothed gradient descent with exponential tau annealing.
    Uses mirror descent step (multiplicative) for better conditioning.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))

        # Exponential annealing for tau
        tau = tau_init * np.exp((tau_end / tau_init - 1) * progress)

        # Periodic reset to escape local minima
        if it > 0 and it % 8000 == 0:
            tau = tau_init * 1.5

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        # Gradient via FFT
        from scipy.fft import fft, ifft
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_fft = fft(y, N)
        g_full = ifft(P * np.conj(A_fft)).real
        grad = 2.0 * g_full[:n]

        # Nesterov update
        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        # Adaptive learning rate
        lr *= 0.9998
        lr = max(1e-8, min(lr, 10.0 / np.sqrt(n)))

        # Reset if stuck
        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)
            best_it = it
            # Small perturbation
            noise = rng.normal(0, 0.01, n)
            a = a * np.exp(noise)
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a


def mirror_descent_step(a, grad, lr):
    """Multiplicative (mirror) descent step with simplex projection."""
    a_new = a * np.exp(-lr * grad)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lp_polish_adaptive(a, time_budget=400.0, k=10, seed=0, verbose=False):
    """
    LP polishing with adaptive golden-section line search.
    Uses top-K LP for speed and focus.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    consecutive_fails = 0
    last_good_t = 0.3

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 5:
            break

        # Solve top-K LP
        g_lp, M_lp = solve_top_k_lp(a, k=k)
        if g_lp is None:
            # Try full LP as fallback
            g_lp, M_lp = solve_full_lp(a)
            if g_lp is None:
                break

        # Adaptive line search along LP direction
        # Use a truncated search range based on last good t
        t_max = min(1.0, max(0.5, last_good_t * 2))
        t_opt = golden_section_line_search(a, g_lp, compute_obj, max_iter=20)

        # Evaluate at optimal t
        a_try = (1 - t_opt) * a + t_opt * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            t_opt = 0.1
            a_try = (1 - t_opt) * a + t_opt * g_lp
            a_try = a_try / a_try.sum()

        obj_try = compute_obj(a_try)

        it += 1
        if obj_try < best_obj - 1e-11:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()
            last_good_t = t_opt
            consecutive_fails = 0
            if verbose and it % 10 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, M={M_lp:.6f}")
        else:
            consecutive_fails += 1
            # Structured perturbation exploiting asymmetry
            conv = compute_conv(a)
            # Create a perturbation that breaks symmetry
            x = np.arange(n)
            mid = n / 2
            asym_bias = np.tanh((x - mid) / (n * 0.3))
            scale = 0.01 + 0.003 * min(consecutive_fails, 8)
            noise = rng.normal(0, 1, n) * scale
            a_pert = a * np.exp(noise + 0.5 * scale * asym_bias)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            # Blend with best
            blend = min(0.3, 0.05 * consecutive_fails)
            a = (1 - blend) * a_pert + blend * best_a
            a = np.maximum(a, 0)
            a = a / a.sum()

            if consecutive_fails > 30:
                # Try a larger jump
                a = 0.5 * a + 0.5 * g_lp
                a = np.maximum(a, 0)
                a = a / a.sum()
                obj_check = compute_obj(a)
                if obj_check < best_obj:
                    best_obj = obj_check
                    best_a = a.copy()
                    consecutive_fails = 0
                else:
                    a = best_a.copy()
                    if consecutive_fails > 50:
                        break

    return best_a


def coordinate_descent_2d(a, time_budget=200.0, seed=0, verbose=False):
    """
    2D coordinate descent: optimize pairs of coordinates simultaneously.
    Uses exact 2D minimization via grid search + refinement.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    while time.monotonic() - start < time_budget:
        # Random pair
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue

        # Optimize a[i] and a[j] keeping sum constant
        # Let a[i] = old_i + d, a[j] = old_j - d (keep sum fixed)
        old_i, old_j = a[i], a[j]
        total = old_i + old_j

        # Grid search over d
        best_d = 0.0
        best_local_obj = best_obj
        num_samples = 50
        d_range = total * 0.9
        d_vals = np.linspace(-d_range, d_range, num_samples)

        for d in d_vals:
            new_i = max(0, old_i + d)
            new_j = max(0, old_j - d)
            a[i], a[j] = new_i, new_j
            s_curr = float(np.sum(a))
            if s_curr < 1e-10:
                a[i], a[j] = old_i, old_j
                continue
            a_norm = a / s_curr
            obj_local = compute_obj(a_norm)
            if obj_local < best_local_obj:
                best_local_obj = obj_local
                best_d = d

        # Fine refinement around best_d
        if best_d != 0:
            fine_range = d_range / num_samples
            d_fine = np.linspace(best_d - fine_range, best_d + fine_range, 20)
            for d in d_fine:
                new_i = max(0, old_i + d)
                new_j = max(0, old_j - d)
                a[i], a[j] = new_i, new_j
                s_curr = float(np.sum(a))
                if s_curr < 1e-10:
                    a[i], a[j] = old_i, old_j
                    continue
                a_norm = a / s_curr
                obj_local = compute_obj(a_norm)
                if obj_local < best_local_obj:
                    best_local_obj = obj_local
                    best_d = d

        # Apply best
        a[i] = max(0, old_i + best_d)
        a[j] = max(0, old_j - best_d)
        s_curr = float(np.sum(a))
        if s_curr > 1e-10:
            a = a / s_curr
            if best_local_obj < best_obj:
                best_obj = best_local_obj
                best_a = a.copy()

        it += 1
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD2D it{it}: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def kkt_polish(a, time_budget=100.0, seed=0, verbose=False):
    """
    KKT-condition-based polish: exploit the active set structure.
    At the optimum, the gradient of the active constraints (weighted by dual vars)
    should be constant on the support of a.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        # Active set: constraints within 1% of max
        active_mask = conv >= max_conv * 0.99
        active_idx = np.where(active_mask)[0]

        if len(active_idx) == 0:
            break

        # Compute gradient of each active constraint
        # grad_b[k][i] = a[k-i] + a[i-k] (accounting for symmetry)
        # More precisely: d(b[k])/da[i] = a[k-i] if 0<=k-i<n, plus a[i-k] if 0<=i-k<n
        # But since b[k] = sum_j a[j]*a[k-j], db[k]/da[i] = a[k-i] + a[i-k] (if both valid)
        # Actually: b[k] = sum_j a[j]*a[k-j], so db[k]/da[i] = sum_j [delta_{j,i}*a[k-j] + a[j]*delta_{k-j,i}]
        # = a[k-i] + a[k-i] ... no wait.
        # b[k] = sum_j a[j] * a[k-j]
        # db[k]/da[i] = sum_j [delta(j,i) * a[k-j] + a[j] * delta(k-j,i)]
        # = a[k-i] (from first term, j=i) + a[k-i] (from second term, k-j=i => j=k-i)
        # Wait, that's only if both i and k-i are valid indices.
        # Let me be more careful:
        # db[k]/da[i] = a[k-i] (if 0 <= k-i < n) + a[k-i] (if 0 <= k-i < n, from the other term)
        # Actually no. Let's index properly:
        # b[k] = sum_{j=0}^{n-1} a[j] * a[k-j] (where a[x] = 0 if x not in [0,n-1])
        # db[k]/da[i] = a[k-i] + a[k-i] = 2*a[k-i] if 0 <= k-i < n
        # Wait, that's because a appears twice: once as a[j] with j=i, and once as a[k-j] with k-j=i, i.e., j=k-i.
        # So db[k]/da[i] = a[k-i] (from j=i) + a[k-i] (from j=k-i) = 2*a[k-i] (if valid)
        # Unless i = k-i, i.e., k = 2i, in which case it's a[i]^2 and derivative is 2*a[i].
        # So in all cases: db[k]/da[i] = 2*a[k-i] if 0 <= k-i < n, else 0.

        # Build gradient matrix for active constraints
        K = len(active_idx)
        G = np.zeros((K, n))
        for r, k in enumerate(active_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                G[r, i] = 2.0 * a[k - i]

        # KKT condition: G^T @ mu should be constant on support of a
        # We want to find mu >= 0 such that G^T @ mu is as constant as possible on supp(a)
        # This is: min_mu ||G^T mu - c*1||^2 on supp(a), s.t. mu >= 0
        # where c is a free variable.
        # Equivalently: min_{mu, c} ||(G^T mu)_supp - c*1||^2 s.t. mu >= 0
        # This is a QP in (mu, c).

        support = a > 1e-10
        if not np.any(support):
            break
        G_supp = G[:, support]
        n_supp = int(np.sum(support))

        # Solve: min ||G_supp^T mu - c*1||^2 s.t. mu >= 0, c free
        # Using cvxpy
        mu_var = cp.Variable(K, nonneg=True)
        c_var = cp.Variable()
        objective = cp.Minimize(cp.sum_squares(G_supp.T @ mu_var - c_var))
        constraints = []
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CLARABEL)
            if mu_var.value is not None:
                mu_val = np.maximum(mu_val if hasattr(mu_val, 'value') else mu_var.value, 0)
                # The "residual" tells us how far from KKT we are
                residual = G_supp.T @ mu_val - c_var.value
                resid_norm = float(np.linalg.norm(residual)) / max(1, n_supp)

                # Use mu to get a correction direction
                # The direction that reduces the max while respecting KKT:
                # Move a in the direction that equalizes the active constraints
                direction = -G @ mu_val  # Negative gradient weighted by dual vars
                # Project to tangent space of simplex (zero sum)
                direction -= np.mean(direction)
                # Only update positive components
                direction = direction * (a > 1e-10)
                direction -= np.sum(direction)  # Ensure zero sum
                direction = direction * a  # Scale by current values (mirror-like)

                # Take a small step
                step_size = 0.01
                a_try = a + step_size * direction
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        if verbose:
                            print(f"    KKT it{it}: obj={best_obj:.8f}, resid={resid_norm:.6f}")
        except Exception:
            pass

        it += 1
        if it > 200:
            break

    return best_a


def diverse_init_asymmetric(n, rng, mode=0):
    """Generate diverse asymmetric initializations (key insight: asymmetry helps)."""
    if mode == 0:
        # Power law (asymmetric)
        p = rng.uniform(0.5, 3.0)
        a = np.arange(1, n + 1) ** p
        a = a / a.sum()
    elif mode == 1:
        # Exponential decay (asymmetric)
        rate = rng.uniform(0.5, 3.0)
        a = np.exp(-rate * np.arange(n) / n)
        a = a / a.sum()
    elif mode == 2:
        # Two bumps (asymmetric positions)
        c1 = rng.integers(0, n // 3)
        c2 = rng.integers(2 * n // 3, n)
        w1 = max(1, n // 10)
        w2 = max(1, n // 15)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w1 ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w2 ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Sinusoidal modulation (asymmetric carrier)
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.sin(2 * np.pi * freq * x + phase)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Random smooth (low-pass filtered)
        a = rng.uniform(0, 1, n)
        k = max(2, n // 30)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 5:
        # Heavy-tailed (chi-squared like)
        a = rng.chisquare(1, n)
        a = a / a.sum()
    elif mode == 6:
        # Anti-symmetric bias
        x = np.linspace(-1, 1, n)
        a = 1 + rng.uniform(0.2, 0.8) * x
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses focused LP polishing with top-K constraints,
    adaptive line search, mirror descent, KKT polish, and 2D coordinate descent.
    Key insight: asymmetric sequences can beat the symmetric (uniform) bound of 2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Phase 0: Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    n = len(best_seq)

    # === Phase 1: Quick smoothed GD (120s) ===
    print(f"\n=== Phase 1: Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.1, 120.0)
    if p1_budget > 30:
        a_gd = smoothed_gd(best_seq, time_budget=p1_budget,
                           tau_init=20.0, tau_end=800.0, seed=seed)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase1 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 GD: obj={obj_gd:.8f} (no improvement)")

    # === Phase 2: LP polishing with adaptive line search (450s) ===
    print(f"\n=== Phase 2: LP Polish (Top-K, adaptive LS) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 450.0)
    if p2_budget > 30:
        a_lp = lp_polish_adaptive(best_seq, time_budget=p2_budget, k=10,
                                  seed=seed, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f}")

    # === Phase 3: 2D Coordinate Descent (200s) ===
    print(f"\n=== Phase 3: 2D Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 200.0)
    if p3_budget > 30:
        a_cd = coordinate_descent_2d(best_seq, time_budget=p3_budget,
                                     seed=seed + 100, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase3 CD2D: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 CD2D: obj={obj_cd:.8f}")

    # === Phase 4: KKT-based polish (100s) ===
    print(f"\n=== Phase 4: KKT Polish ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.1, 100.0)
    if p4_budget > 20:
        a_kkt = kkt_polish(best_seq, time_budget=p4_budget,
                           seed=seed + 200, verbose=True)
        obj_kkt = compute_obj(a_kkt)
        if obj_kkt < best_score:
            best_score = obj_kkt
            best_seq = a_kkt.copy()
            print(f"  Phase4 KKT: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 KKT: obj={obj_kkt:.8f}")

    # === Phase 5: Second round of LP with different K (150s) ===
    print(f"\n=== Phase 5: LP Polish (K=5, focused) ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.2, 150.0)
    if p5_budget > 30:
        a_lp2 = lp_polish_adaptive(best_seq, time_budget=p5_budget, k=5,
                                   seed=seed + 300, verbose=False)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase5 LP(K=5): obj={best_score:.8f} **")
        else:
            print(f"  Phase5 LP(K=5): obj={obj_lp2:.8f}")

    # === Phase 6: Try alternative n values (150s) ===
    print(f"\n=== Phase 6: Alternative n ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    n_candidates = [n - 100, n + 100, 700, 900, 1000]
    n_candidates = [x for x in n_candidates if 400 <= x <= 1500 and abs(x - n_current) >= 50]
    n_candidates = n_candidates[:2]  # Try at most 2

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        # Resample to new length
        x_old = np.linspace(0, 1, n_current)
        x_new = np.linspace(0, 1, n_try)
        a_try = np.interp(x_new, x_old, best_seq)
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # Quick LP polish
        alloc = min(remaining * 0.3, 80.0)
        if alloc > 20:
            a_try = lp_polish_adaptive(a_try, time_budget=alloc, k=8,
                                       seed=seed + n_try, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final intensive polish (remaining) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Final LP with small K for precision
    if remaining > 20:
        alloc = min(remaining * 0.4, 80.0)
        a_final = lp_polish_adaptive(best_seq, time_budget=alloc, k=3,
                                     seed=seed + 500, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP(K=3): obj={best_score:.8f} **")

    # Final GD with high tau
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 50.0)
        a_sgd = smoothed_gd(best_seq, time_budget=alloc,
                            tau_init=100.0, tau_end=2000.0, seed=seed + 600)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Final CD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 30.0)
        a_cd_final = coordinate_descent_2d(best_seq, time_budget=alloc,
                                           seed=seed + 700, verbose=False)
        obj_cd_final = compute_obj(a_cd_final)
        if obj_cd_final < best_score:
            best_score = obj_cd_final
            best_seq = a_cd_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
