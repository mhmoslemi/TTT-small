def solve_lp(a):

    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

def compute_conv(a):

    n = len(a)

def frank_wolfe_step(a, tau=50.0):
    """
    Frank-Wolfe: compute subgradient, then move towards the vertex
    (coordinate) with the smallest gradient component.
    Returns new point on simplex.
    """
    n = len(a)
    g = subgradient_direction(a, tau)
    # Find coordinate with minimum gradient
    j_star = int(np.argmin(g))
    # Return the vertex e_{j_star}
    # (Frank-Wolfe linear oracle minimizes g^T v over simplex = pick argmin g_i)
    return j_star


def mirror_descent_update(a, g, eta):
    """Multiplicative weight update: a_i <- a_i * exp(-eta * g_i), then normalize."""
    a_new = a * np.exp(-eta * g)
    a_new = np.maximum(a_new, 1e-30)
    s = float(np.sum(a_new))
    if s < 1e-15:
        return a.copy()
    return a_new / s


def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu * rho > (cssv - 1.0)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_val = idx[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    it = 0
    while time.monotonic() - start < time_budget and it < 2000:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Identify active set (top peaks)
        K = min(2 * n - 1, max(50, n // 2))
        active = np.argsort(conv)[::-1][:K]

        # Solve restricted LP
        g_active = solve_lp_active_set(a, active.tolist())
        if g_active is None:
            # Fallback to full LP
            g_lp, M_lp = solve_lp_full(a)
            if g_lp is None:
                break
            g_do = g_lp
        else:
            g_do = g_active
            M_lp = float(np.max(compute_conv_cross(a, g_do)))

        # Try multiple blending factors
        best_t_obj = best_obj
        best_t_a = best_a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_do
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-12:
                continue
            a_try = a_try / s
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_a = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = best_t_a.copy()
        else:
            # Perturb to escape
            noise = np.random.default_rng(it).normal(0, 0.02, n)
            a_pert = a * np.exp(noise)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            a = 0.7 * a + 0.3 * a_pert
            a = np.maximum(a, 0)
            a = a / a.sum()

        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = (1 + np.cos(2 * np.pi * x)) / 2
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
    elif mode == 3:
        # Triangular (peak at center)
        x = np.arange(n)
        center = n / 2
        a = 1.0 - np.abs(x - center) / center
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
    elif mode == 4:
        # Gaussian (center)
        x = np.arange(n)
        center = n / 2
        sigma = n / 4
        a = np.exp(-((x - center) ** 2) / (2 * sigma ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Smooth random
        a = rng.uniform(0, 1, n)
        k = max(2, n // 30)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
    elif mode == 6:
        # Sine wave
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.sin(2 * np.pi * x)
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
    elif mode == 7:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** 0.5
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses LP fixed-point with Anderson acceleration,
    hybrid subgradient-LP, multi-length exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0  # Safety margin

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Phase 0: Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj_normalized(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
            else:
                raise ValueError("Sum too small")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj_normalized(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Initial LP-Anderson iteration ===
    print(f"\n=== Phase 1: LP-Anderson (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.3, 300.0)
    if p1_budget > 30:
        a1 = lp_anderson_iteration(best_seq, time_budget=p1_budget, verbose=True)
        obj1 = compute_obj_normalized(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1 LP-A: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 LP-A: obj={obj1:.8f}")

    # === Phase 2: Hybrid subgradient + LP ===
    print(f"\n=== Phase 2: Hybrid Subgrad-LP ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 250.0)
    if p2_budget > 30:
        a2 = hybrid_subgrad_lp(best_seq, time_budget=p2_budget, verbose=True)
        obj2 = compute_obj_normalized(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 Hybrid: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 Hybrid: obj={obj2:.8f}")

    # === Phase 3: Multi-scale LP polish ===
    print(f"\n=== Phase 3: Multi-scale LP Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 200.0)
    if p3_budget > 30:
        a3 = multi_scale_lp_polish(best_seq, time_budget=p3_budget, verbose=True)
        obj3 = compute_obj_normalized(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3 Multi-LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 Multi-LP: obj={obj3:.8f}")

    # === Phase 4: Second LP-Anderson round (from improved point) ===
    print(f"\n=== Phase 4: LP-Anderson Round 2 ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.2, 200.0)
    if p4_budget > 30:
        a4 = lp_anderson_iteration(best_seq, time_budget=p4_budget, verbose=True)
        obj4 = compute_obj_normalized(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4 LP-A2: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 LP-A2: obj={obj4:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Try several n values
    n_options = []
    for delta in [-200, -100, 100, 200, 300, 500]:
        n_try = n_current + delta
        if 300 <= n_try <= 3000 and n_try != n_current:
            n_options.append(n_try)
    # Also try some absolute values
    for n_abs in [500, 600, 1000, 1200, 1500, 2000]:
        if 300 <= n_abs <= 3000 and n_abs != n_current and n_abs not in n_options:
            n_options.append(n_abs)

    # Sort by proximity to current (try close ones first)
    n_options.sort(key=lambda x: abs(x - n_current))
    n_options = n_options[:5]  # Limit to 5 attempts

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        # Quick LP-Anderson on new length
        alloc = min(remaining * 0.25, 120.0)
        if alloc > 20:
            a_try = lp_anderson_iteration(a_try, time_budget=alloc, verbose=False)

        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc2 = min(remaining * 0.1, 40.0)
            if alloc2 > 10:
                a_try = multi_scale_lp_polish(a_try, time_budget=alloc2, verbose=False)

        obj_try = compute_obj_normalized(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = (r * 2 + 1) % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with best
        blend = 0.3
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.25, 100.0)
        if alloc > 20:
            a_r = lp_anderson_iteration(a_r, time_budget=alloc, verbose=False)
            obj_r = compute_obj_normalized(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        alloc = min(remaining * 0.5, 120.0)
        a_final = multi_scale_lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj_normalized(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final polish: obj={best_score:.8f} **")
        else:
            print(f"  Final polish: obj={obj_final:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
        best_score = compute_obj_normalized(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
