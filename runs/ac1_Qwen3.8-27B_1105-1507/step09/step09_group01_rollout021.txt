"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses combined strategy: diverse analytical initializations, Nesterov smoothed
gradient descent, active-set LP polishing with constraint sweeping, direct peak
shaving, multi-n exploration, and final intensive polishing.
Key differences from previous: active-set LP for speed, peak shaving heuristic,
frequency-domain informed initialization, and adaptive time allocation.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    mu = v.copy()
    rho = 0
    tot = 0.0
    for j in range(n - 1, -1, -1):
        mu_j = mu[j]
        tot += mu_j
        if mu_j * (j + 1) > tot - 1.0:
            rho = j + 1
            break
    if rho == 0:
        return np.ones(n) / n
    theta = (tot - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv, max_conv


def nesterov_smoothed_descent(a, time_budget=300.0, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget * 0.85, 1.0))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        if verbose and it % 10000 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv, max_conv = smoothed_gradient_fft(y, tau)

        if prev_g is not None and it % 2 == 0 and it > 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15:
                bb_lr = ss / max(abs(sy), 1e-15)
                if bb_lr < 50.0 / np.sqrt(n):
                    lr = 0.3 * lr + 0.7 * min(max(bb_lr, 1e-8), 20.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.99995)

        if it - best_it > 50000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = max(0, j - (n - 1))
        hi = min(n - 1, j)
        for i in range(lo, hi + 1):
            A[j, i] = a[j - i]
    return A


def solve_lp_minpeak(a, M_target=None, max_constraints=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Optionally limit to top 'max_constraints' binding constraints for speed.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))

    if max_constraints is not None and max_constraints < 2 * n - 1:
        conv_self = compute_conv(a_norm)
        top_idx = np.argsort(conv_self)[::-1][:max_constraints]
        A = np.zeros((len(top_idx), n))
        for col_i, k in enumerate(top_idx):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                A[col_i, i] = a_norm[k - i]
    else:
        A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue

    return None, None


def golden_section_search(f, a_lo, a_hi, tol=0.01, max_iter=35):
    """Golden section search for minimum of f on [a_lo, a_hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if abs(a_hi - a_lo) < tol:
            break
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = f(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = f(d)
    return (a_lo + a_hi) / 2.0


def lp_polish(a, time_budget=200.0, verbose=False, seed=42, use_active_set=True):
    """
    LP-based polishing with golden-section line search and adaptive strategies.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    no_improve = 0
    last_t = 0.4

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed

        # Adaptive: use active set (fewer constraints) when time is tight
        if use_active_set:
            if remaining < 30:
                max_c = max(80, n // 2)
            elif remaining < 60:
                max_c = max(120, int(n * 0.75))
            else:
                max_c = 2 * n - 1  # Full problem
        else:
            max_c = None

        g_lp, M_lp = solve_lp_minpeak(a, max_constraints=max_c)
        if g_lp is None:
            break

        # Golden section for blend factor
        def blend_obj(t):
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return 1e10
            a_try = a_try / s_try
            return compute_obj(a_try)

        t_opt = golden_section_search(blend_obj, 0.02, 1.0, tol=0.02, max_iter=30)

        a_try = (1.0 - t_opt) * best_a + t_opt * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            break
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)

        iter_count += 1

        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a.copy()
            last_t = t_opt
            no_improve = 0
            if verbose and iter_count % 3 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={t_opt:.3f}, el={elapsed:.0f}s")
        else:
            no_improve += 1
            if no_improve == 2:
                t_small = max(0.02, last_t * 0.3)
                a_try2 = (1.0 - t_small) * best_a + t_small * g_lp
                a_try2 = np.maximum(a_try2, 0)
                a_try2 = a_try2 / a_try2.sum()
                obj_try2 = compute_obj(a_try2)
                if obj_try2 < best_obj - 1e-14:
                    best_obj = obj_try2
                    best_a = a_try2.copy()
                    a = best_a.copy()
                    no_improve = 0
            elif no_improve >= 5:
                scale = 0.005 * min(no_improve, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                if iter_count > 50 and remaining < 15:
                    break

        # Periodic constraint tightening
        if iter_count % 6 == 3 and M_lp is not None and remaining > 20:
            for tight_frac in [0.93, 0.96, 0.99]:
                M_tight = M_lp * tight_frac
                g_tight, _ = solve_lp_minpeak(a, M_target=M_tight, max_constraints=max_c)
                if g_tight is not None:
                    a_try3 = 0.5 * best_a + 0.5 * g_tight
                    a_try3 = np.maximum(a_try3, 0)
                    a_try3 = a_try3 / a_try3.sum()
                    obj_try3 = compute_obj(a_try3)
                    if obj_try3 < best_obj - 1e-14:
                        best_obj = obj_try3
                        best_a = a_try3.copy()
                        a = best_a.copy()
                        no_improve = 0
                        break

    return best_a


def peak_shaving(a, time_budget=100.0, verbose=False):
    """
    Direct peak shaving: identify top peaks in convolution and adjust
    contributing coefficients to reduce them.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        mean_val = float(np.mean(conv))
        threshold = best_obj / (2.0 * n)

        # Find top K peaks above threshold
        K = max(3, n // 20)
        sorted_idx = np.argsort(conv)[::-1]
        top_idx = sorted_idx[:K]
        top_vals = conv[top_idx]

        # For each peak, compute gradient and adjust
        total_grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            # Gradient of conv[k] w.r.t. a[j]: a[k-j] for valid j
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                contribution = a[k - i_end:i_start + 1][::-1]
                # Weight by how much this peak exceeds target
                weight = max(0.0, conv[k] - threshold)
                total_grad += weight * contribution

        # Normalize gradient
        grad_norm = float(np.linalg.norm(total_grad))
        if grad_norm < 1e-15:
            break

        # Adaptive step size
        lr = 0.1 * max_val / (grad_norm + 1e-15)
        lr = min(lr, 0.5)

        a_new = a - lr * total_grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            it += 1
            if verbose and it % 20 == 0:
                print(f"    PS it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
        else:
            # Backtrack
            lr *= 0.5
            a_new = a - lr * total_grad
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                # Try smaller step
                lr *= 0.3
                a_new = a - lr * total_grad
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                else:
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def generate_raised_cosine(n):
    """Raised cosine initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = a / a.sum()
    return a


def generate_gaussian(n, sigma_frac=0.3):
    """Gaussian initialization."""
    x = np.linspace(0, 1, n)
    center = 0.5
    sigma = sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = a / a.sum()
    return a


def generate_polynomial(n, order=2):
    """Polynomial (parabola-like) initialization."""
    x = np.linspace(-1, 1, n)
    a = 1.0 + 0.5 * (1.0 - x ** 2) ** order
    a = a / a.sum()
    return a


def generate_bell_pair(n, rng):
    """Two-bump initialization."""
    c1 = rng.uniform(0.15, 0.35)
    c2 = 1.0 - c1
    w = rng.uniform(0.08, 0.2)
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - c1) / w) ** 2) + np.exp(-((x - c2) / w) ** 2)
    a = a / a.sum()
    return a


def generate_smooth_random(n, rng, n_folds=5):
    """Smooth random initialization via low-frequency sine mixture."""
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for _ in range(n_folds):
        freq = rng.integers(1, 6)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4)
        a += amp * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: diverse analytical init + Nesterov + active-set LP polishing
    + peak shaving + multi-n exploration + final intensive polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = generate_raised_cosine(n_target)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Diverse initialization + quick Nesterov ===
    print(f"\n=== Phase 1: Diverse Init + Nesterov (n={n_target}) ===")
    remaining = deadline - time.monotonic()
    p1_total = min(remaining * 0.13, 150.0)
    if p1_total > 30:
        inits = [
            ("raised_cos", generate_raised_cosine(n_target)),
            ("gaussian", generate_gaussian(n_target, 0.25)),
            ("polynomial", generate_polynomial(n_target, 2)),
            ("bell_pair", generate_bell_pair(n_target, rng)),
            ("smooth_rand", generate_smooth_random(n_target, rng, 6)),
        ]
        # Include parent if we have one
        if best_seq is not None:
            inits.append(("parent", best_seq.copy()))

        per_init_budget = p1_total / len(inits)
        for name, a_init in inits:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(per_init_budget, remaining * 0.8)
            if alloc < 5:
                continue

            a_opt = nesterov_smoothed_descent(a_init, time_budget=alloc,
                                              tau_init=8.0, tau_end=400.0)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  {name} (n={n_target}): obj={best_score:.8f} **")
            else:
                print(f"  {name} (n={n_target}): obj={obj_opt:.8f}")

    # === Phase 2: Intensive LP Polishing ===
    print(f"\n=== Phase 2: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 420.0)
    if p2_budget > 30:
        a_lp = lp_polish(best_seq, time_budget=p2_budget, verbose=True, seed=42,
                         use_active_set=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f}")

    # === Phase 3: Peak Shaving ===
    print(f"\n=== Phase 3: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 100.0)
    if p3_budget > 15:
        a_ps = peak_shaving(best_seq, time_budget=p3_budget, verbose=True)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  Phase3 PS: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 PS: obj={obj_ps:.8f}")

        # Quick LP after peak shaving
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.3, 80.0)
            a_lp2 = lp_polish(best_seq, time_budget=alloc, seed=77)
            obj_lp2 = compute_obj(a_lp2)
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  Phase3 post-LP: obj={best_score:.8f} **")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 200),
        max(400, n_current - 100),
        n_current + 100,
        n_current + 200,
        n_current + 400,
        600, 800, 1000, 1200, 1500,
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2500 and abs(n - n_current) > 50]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)

        # Quick Nesterov
        alloc1 = min(remaining * 0.2, 35.0)
        if alloc1 > 10:
            a_try = nesterov_smoothed_descent(a_try, time_budget=alloc1,
                                              tau_init=10.0, tau_end=300.0)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.25, 80.0)
        if alloc2 > 15:
            a_try = lp_polish(a_try, time_budget=alloc2, seed=n_try % 100)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Perturbation + Re-LP ===
    print(f"\n=== Phase 5: Perturbation + Re-LP ===")
    remaining = deadline - time.monotonic()

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        scale = 0.008 + 0.006 * r
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc = min(remaining * 0.2, 80.0)
        if alloc > 15:
            a_pert = lp_polish(a_pert, time_budget=alloc, seed=200 + r)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Perturb {r} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Perturb {r} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 6: Final intensive LP + peak shaving ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Split remaining between LP and peak shaving
        lp_alloc = min(remaining * 0.6, 200.0)
        ps_alloc = min(remaining * 0.2, 60.0)

        if lp_alloc > 15:
            a_final = lp_polish(best_seq, time_budget=lp_alloc, verbose=True, seed=999,
                                use_active_set=False)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        # Peak shave
        remaining = deadline - time.monotonic()
        if remaining > 10:
            ps_alloc = min(remaining * 0.4, 50.0)
            if ps_alloc > 5:
                a_ps2 = peak_shaving(best_seq, time_budget=ps_alloc)
                obj_ps2 = compute_obj(a_ps2)
                if obj_ps2 < best_score:
                    best_score = obj_ps2
                    best_seq = a_ps2.copy()
                    print(f"  Final PS: obj={best_score:.8f} **")

        # One more short LP
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.5, 40.0)
            a_final2 = lp_polish(best_seq, time_budget=alloc, seed=1234)
            obj_final2 = compute_obj(a_final2)
            if obj_final2 < best_score:
                best_score = obj_final2
                best_seq = a_final2.copy()
                print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
