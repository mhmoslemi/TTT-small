def anderson_step(history_a, history_F, m=5):
    """
    Given history of iterates and their images, compute Anderson accelerated step.
    history_a: list of arrays a_0, ..., a_k
    history_F: list of arrays F(a_0), ..., F(a_k)
    m: memory (use last m+1 points)
    Returns: new iterate as convex combination of F values.
    """
    k = min(len(history_a) - 1, m)
    if k < 1:
        # Not enough history, just use latest
        return history_F[-1].copy()
    
    # Use last k+1 points
    a_hist = np.array(history_a[-(k+1):])  # (k+1, n)
    F_hist = np.array(history_F[-(k+1):])  # (k+1, n)
    
    # Residuals: r_i = F(a_i) - a_i
    R = F_hist - a_hist  # (k+1, n)
    
    # Solve: min ||sum theta_i r_i||^2 s.t. sum theta = 1, theta >= 0 (optional)
    # This is: min theta^T (R^T R) theta s.t. 1^T theta = 1
    # Using the formula: theta = (1 + G^{-1} 1) / (1 + 1^T G^{-1} 1)
    # where G = R^T R (but we use R^T R / (n-1) for numerical stability)
    
    G = R @ R.T  # (k+1, k+1)
    # Add small regularization
    G += 1e-10 * np.eye(k+1)
    
    try:
        # Solve the system
        # theta = argmin ||R theta||^2 s.t. sum(theta) = 1
        # KKT: [2G  1; 1^T 0] [theta; nu] = [0; 1]
        kkt = np.zeros((k+2, k+2))
        kkt[:k+1, :k+1] = 2 * G
        kkt[:k+1, k+1] = 1
        kkt[k+1, :k+1] = 1
        rhs = np.zeros(k+2)
        rhs[k+1] = 1
        
        sol = np.linalg.solve(kkt, rhs)
        theta = sol[:k+1]
        
        # If theta has negative entries, project to simplex
        if np.any(theta < -1e-6):
            # Fallback: uniform
            theta = np.ones(k+1) / (k+1)
        
        # Compute new point
        new_a = theta @ F_hist
        return new_a
    except:
        return history_F[-1].copy()

def anderson_accelerate(history_a, history_F, m=6):
    """
    Anderson acceleration: given history of (a_i, F(a_i)), compute next iterate.
    Returns convex combination of F values.
    """
    k = min(len(history_a) - 1, m)
    if k < 1:
        return history_F[-1].copy()

    a_hist = np.array(history_a[-(k + 1):])
    F_hist = np.array(history_F[-(k + 1):])
    R = F_hist - a_hist  # residuals

    # Solve: min ||R^T theta||^2 s.t. sum(theta) = 1
    G = R @ R.T
    G += 1e-12 * np.eye(k + 1)

    try:
        # KKT system
        size = k + 2
        KKT = np.zeros((size, size))
        KKT[:k + 1, :k + 1] = 2 * G
        KKT[:k + 1, k + 1] = 1.0
        KKT[k + 1, :k + 1] = 1.0
        rhs = np.zeros(size)
        rhs[k + 1] = 1.0

        sol = np.linalg.solve(KKT, rhs)
        theta = sol[:k + 1]

        # Check for numerical issues
        if not np.all(np.isfinite(theta)) or np.any(theta < -0.5):
            # Project to simplex
            theta = np.maximum(theta, 0)
            tsum = theta.sum()
            if tsum < 1e-10:
                theta = np.ones(k + 1) / (k + 1)
            else:
                theta = theta / tsum

        new_a = theta @ F_hist
        if not np.all(np.isfinite(new_a)):
            return history_F[-1].copy()
        return new_a
    except Exception:
        return history_F[-1].copy()

def fourier_perturbation(a, rng, strength=0.03):
    """Add low-frequency Fourier mode perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 4
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.0) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def smoothed_gradient_fft(a, tau):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*conv_k(a))) using FFT.
    Returns (gradient, conv).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1

    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def smoothed_gd_polish(a, time_budget, tau_init=50.0, tau_end=2000.0, verbose=True):
    """

    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Cosine schedule for tau
        cos_val = 0.5 * (1 + np.cos(np.pi * progress))
        tau = tau_init + (tau_end - tau_init) * (1 - cos_val)

        if it % 2000 == 0 and verbose:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6f}")

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)
        g_norm = float(np.linalg.norm(g))

        # Adaptive LR
        if g_norm > 1e-15:
            adaptive_lr = min(0.5 / (g_norm + 1e-10), 3.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adaptive_lr

        # BB step
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = min(max(ss / abs(sy), 1e-8), 5.0 / np.sqrt(n))
                lr = 0.6 * lr + 0.4 * bb_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9998)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.3
            best_it = it

    return best_a


def mirror_symmetrize(a):
    """Make sequence mirror-symmetric."""
    n = len(a)
    return (a + a[::-1]) / 2.0


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Arcsine-like
        x = np.linspace(0.01, 0.99, n)
        a = 1.0 / np.sqrt(x * (1 - x))
        a = a / a.sum()
    elif mode == 3:
        # Two Gaussian bumps
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 10)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 4:
        # Cosine taper
        x = np.linspace(0, np.pi, n)
        a = 0.3 + 0.7 * np.abs(np.sin(x))
        a = a / a.sum()
    elif mode == 5:
        # Edge-concentrated
        x = np.linspace(0, 1, n)
        a = x**2 + (1 - x)**2 + 0.1
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def active_set_lp_polish(a, time_budget, K=100, verbose=True):
    """
    LP polish using only top-K active constraints for speed.
    Iteratively adds constraints that are violated.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Identify initial active set
    conv = compute_conv(a)
    active = np.argsort(conv)[-K:]

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break

        g_lp, M_lp = solve_lp_subset(a, active)
        if g_lp is None:
            # Fall back to full LP
            g_lp, M_lp = solve_lp(a)
            if g_lp is None:
                break

        # Blend
        t = 0.3
        a_new = (1 - t) * a + t * g_lp
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

        # Update active set
        conv_new = compute_conv(a)
        # Check if any previously-inactive constraint is violated
        full_ag = compute_conv_cross(a, g_lp)
        M_current = float(np.max(full_ag))
        violated = np.where(full_ag > M_current * 0.999)[0]
        if len(violated) > 0:
            active = np.unique(np.concatenate([active, violated]))
            if len(active) > 2 * K:
                # Keep only most relevant
                active = np.argsort(full_ag)[-K * 2:]
        else:
            # No violations, update active set from current conv
            active = np.argsort(conv_new)[-K:]

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Anderson-accelerated LP + smoothed GD + multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                n_current = best_seq.size
                print(f"Parent (n={n_current}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(n_current) / n_current
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")

    # === Phase 1: Anderson-accelerated LP (main workhorse) ===
    print(f"\n=== Phase 1: Anderson-Accelerated LP ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.55, 780.0)

    # Multiple rounds of LP-Anderson with escalating perturbation
    n_lp_rounds = 4
    for rnd in range(n_lp_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(p1_budget / n_lp_rounds, remaining * 0.6)
        if alloc < 20:
            break

        if rnd > 0:
            # Perturb to escape potential local min
            strength = 0.01 * (rnd + 1)
            best_seq = fourier_perturbation(best_seq, rng, strength=strength)
            print(f"  Round {rnd}: perturbed with strength={strength:.3f}")

        a_lp = lp_anderson_iteration(best_seq, time_budget=alloc, m=6,
                                     tol=1e-7, verbose=True, seed=100 + rnd)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** LP round {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: Active-set LP polish ===
    print(f"\n=== Phase 2: Active-Set LP Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 200.0)
    if p2_budget > 20:
        a_as = active_set_lp_polish(best_seq, time_budget=p2_budget, K=150, verbose=True)
        obj_as = compute_obj(a_as)
        if obj_as < best_score:
            best_score = obj_as
            best_seq = a_as.copy()
            print(f"  ** Active-set: obj={best_score:.8f} **")
        else:
            print(f"  Active-set: obj={obj_as:.8f}")

    # === Phase 3: Smoothed GD polish ===
    print(f"\n=== Phase 3: Smoothed GD Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 250.0)
    if p3_budget > 30:
        a_gd = smoothed_gd_polish(best_seq, time_budget=p3_budget,
                                  tau_init=100.0, tau_end=3000.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  ** GD: obj={best_score:.8f} **")
        else:
            print(f"  GD: obj={obj_gd:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        n_best, n_best + 100, n_best + 200, n_best + 400,
        1000, 1200, 1400, 1600
    ]))
    n_options = [n for n in n_options if 600 <= n <= 2000 and n != n_best]
    n_options = n_options[:4]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick LP-Anderson
        alloc1 = min(remaining * 0.3, 80.0)
        if alloc1 > 15:
            a_try = lp_anderson_iteration(a_try, time_budget=alloc1, m=4,
                                          tol=1e-6, verbose=False, seed=n_try)

        # Quick GD
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 40.0)
        if alloc2 > 10:
            a_try = smoothed_gd_polish(a_try, time_budget=alloc2,
                                       tau_init=200.0, tau_end=1000.0, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = 2 + r
        a_r = diverse_init(n_current, rng, mode=mode)
        blend = 0.3 + 0.15 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.25, 80.0)
        if alloc > 20:
            half = alloc * 0.6
            a_r = lp_anderson_iteration(a_r, time_budget=half, m=4,
                                        tol=1e-6, verbose=False, seed=500 + r)
            remaining = deadline - time.monotonic()
            gd_alloc = min(remaining * 0.1, 30.0)
            if gd_alloc > 10:
                a_r = smoothed_gd_polish(a_r, time_budget=gd_alloc,
                                         tau_init=100.0, tau_end=800.0, verbose=False)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  ** Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Final LP round
    if remaining > 40:
        alloc = min(remaining * 0.3, 100.0)
        a_final = lp_anderson_iteration(best_seq, time_budget=alloc, m=8,
                                        tol=1e-9, verbose=True, seed=9999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # Final GD with very high tau
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 80.0)
        a_gd = smoothed_gd_polish(best_seq, time_budget=alloc,
                                  tau_init=500.0, tau_end=5000.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  ** Final GD: obj={best_score:.8f} **")
        else:
            print(f"  Final GD: obj={obj_gd:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
