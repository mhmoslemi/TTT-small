Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based smoothed gradient descent with Nesterov acceleration as the main engine,
combined with LP polishing and coordinate descent for fine-tuning.
Key innovation: O(n log n) per gradient step via FFT-computed smoothed max gradient,
with adaptive temperature annealing and LP-informed subgradient weighting.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

# Try to import cvxpy for LP polishing

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv)), conv


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_max_and_grad(a, tau):
    """
    Compute the smoothed max of conv(a,a) and its gradient.
    F_tau(a) = (1/tau) * log(sum_k exp(tau * (a*a)[k]))
    grad_i = 2 * sum_k p_k * a_{k-i}
    Returns (F_tau, grad, conv, p)
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    
    # Compute softmax weights (numerically stable)
    exp_vals = np.exp(tau * (conv - max_c))
    p = exp_vals / float(np.sum(exp_vals))
    
    F_tau = (1.0 / tau) * (np.log(float(np.sum(exp_vals))) + tau * max_c)
    
    # Gradient via cross-correlation
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    grad = 2.0 * g_full[:n]
    
    return F_tau, grad, conv, p


def nesterov_smoothed_descent(a, time_budget=200.0, tau_init=5.0, tau_end=500.0, 
                                verbose=True, seed=42):
    """
    Nesterov accelerated gradient descent on smoothed objective over simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_conv = compute_conv(a)
    best_obj = float(np.max(best_conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Parameters
    beta = 0.9  # momentum
    lr = 1.0 / np.sqrt(n)  # learning rate
    v = np.zeros(n)
    prev_a = a.copy()
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        # Anneal tau
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)
        
        # Nesterov extrapolation
        y = a + beta * (a - prev_a)
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        # Compute gradient
        _, grad, conv, p = smoothed_max_and_grad(y, tau)
        
        # Gradient step + projection
        a_new = y - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            # Reset
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        # Evaluate
        obj_new = float(np.max(compute_conv(a_new)))
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        # Adaptive learning rate (backtracking)
        if obj_new > best_obj * 1.01 and it > 100:
            lr *= 0.9
        elif it % 1000 == 0 and it > 1000:
            lr = min(lr * 1.05, 5.0 / np.sqrt(n))
        
        prev_a = a.copy()
        a = a_new
        it += 1
        
        # Restart if stuck
        if it - best_it > 20000:
            a = best_a.copy()
            prev_a = best_a.copy()
            lr = 1.0 / np.sqrt(n)
            beta = 0.8
            best_it = it
        
        if verbose and it % 5000 == 0:
            print(f"      NSD it={it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")
    
    return best_a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based smoothed gradient descent with Nesterov acceleration as the primary engine,
LP polishing with dual-informed gradient weighting, fast coordinate descent, and multi-n exploration.
Key innovations: (1) efficient rfft-based gradient computation, (2) LP dual weights as softmax
temperatures for the smoothed max, (3) adaptive multiplicative-weight vs projected-gradient switching.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

try:
    import cvxpy as cp
    HAVE_CVXPY = True
except ImportError:
    HAVE_CVXPY = False


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_smoothed_grad(a, tau, N_cache=None):
    """
    Compute smoothed max of conv(a,a) and its gradient using FFT.
    Returns (F_tau, grad, conv, p, N_used).
    """
    n = len(a)
    if N_cache is None or N_cache < 3 * n:
        N = 1
        while N < 3 * n:
            N <<= 1
    else:
        N = N_cache

    fa = rfft(a, N)
    conv_full = irfft(fa * fa, N)
    conv = conv_full[:2 * n - 1]

    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    F_tau = max_conv + np.log(sum_exp) / tau

    p_pad = np.zeros(N)
    p_pad[:2 * n - 1] = p
    fp = rfft(p_pad)
    grad_spec = fp * np.conj(fa)
    grad_full = irfft(grad_spec)
    grad = 2.0 * grad_full[:n]

    return F_tau, grad, conv, p, N


def smoothed_gradient_descent(a, time_budget=200.0, tau_init=3.0, tau_end=800.0,
                              verbose=True, seed=42, use_multiplicative=False):
    """
    Nesterov accelerated smoothed gradient descent on the simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_conv = compute_conv(a)
    best_obj = float(np.max(best_conv))
    start = time.monotonic()

    N = 1
    while N < 3 * n:
        N <<= 1

    lr = 2.0 / np.sqrt(n)
    prev_a = a.copy()
    it = 0
    best_it = 0
    stagnation = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(1.0, time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        # Nesterov extrapolation
        y = a + 0.9 * (a - prev_a)
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        _, grad, conv, p, _ = fast_smoothed_grad(y, tau, N)

        if use_multiplicative:
            # Multiplicative weight update
            step = lr * grad
            step = np.maximum(step, 1e-8)
            a_new = y * np.exp(-step)
            a_new = np.maximum(a_new, 1e-15)
            s_new = float(np.sum(a_new))
            a_new = a_new / s_new
        else:
            # Projected gradient
            a_new = y - lr * grad
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
            else:
                a_new = a_new / s_new

        obj_new = float(np.max(compute_conv(a_new)))
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stagnation = 0
        else:
            stagnation += 1

        # Adaptive LR
        if stagnation > 5000:
            lr *= 0.95
            if lr < 0.01 / np.sqrt(n):
                # Perturb and restart
                rng = np.random.default_rng(it + seed)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                prev_a = a.copy()
                lr = 2.0 / np.sqrt(n)
                best_it = it
                stagnation = 0

        prev_a = a.copy()
        a = a_new
        it += 1

        if verbose and it % 10000 == 0:
            print(f"        SGD it={it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.4f}, t={elapsed:.0f}s")

    return best_a


def fast_coord_descent(a, time_budget=100.0, verbose=True, seed=42):
    """
    Fast coordinate descent with efficient 1D minimization.
    Uses the structure: changing a[k] by delta changes conv by 2*delta*d + delta^2*I(2k).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
    a = a / S
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5) - 1) / 2.0
    sweep = 0

    while time.monotonic() - start < time_budget:
        order = rng.permutation(n) if sweep % 3 == 2 else np.arange(n)
        obj_before = float(np.max(conv))

        for k in order:
            d = np.zeros(2 * n - 1)
            d[k:k + n] = a

            lo = -a[k]
            hi = 2.0 / n + a[k]

            # Golden section on max(conv + 2*delta*d, with extra delta^2 at 2k)
            a_gs = lo
            b_gs = hi
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)

            def eval_delta(delta):
                val = conv + 2.0 * delta * d
                val[2 * k] += delta * delta
                return float(np.max(val))

            fc = eval_delta(c_gs)
            fd = eval_delta(d_gs)

            for _ in range(35):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = eval_delta(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = eval_delta(d_gs)

            delta_opt = (a_gs + b_gs) / 2.0

            if delta_opt != 0.0:
                new_val = eval_delta(delta_opt)
                if new_val < obj_before - 1e-14:
                    a[k] = max(0.0, a[k] + delta_opt)
                    conv += 2.0 * delta_opt * d
                    conv[2 * k] += delta_opt * delta_opt
                    S_new = float(np.sum(a))
                    if S_new < 1e-10:
                        break
                    a /= S_new
                    conv /= (S_new * S_new)

        sweep += 1
        obj_after = float(np.max(conv))
        if obj_after < best_obj - 1e-12:
            best_obj = obj_after
            best_a = a.copy()
        else:
            if sweep > 5:
                break
            # Perturb
            scale = 0.01 * min(sweep, 5)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            conv = compute_conv(a)
            best_obj = float(np.max(conv))

        if verbose and sweep % 5 == 0:
            print(f"        CD sweep={sweep}: obj={2*n*best_obj:.8f}, t={time.monotonic()-start:.0f}s")

    return best_a


def build_conv_matrix(a):
    """Build (2n-1 x n) matrix for conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLP, cp.GLPK, cp.ECOS, cp.SCS, cp.CLARABEL]
    if hasattr(cp, 'MOSEK'):
        solvers.insert(0, cp.MOSEK)
    if hasattr(cp, 'PDLP'):
        solvers.insert(1, cp.PDLP)

    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def lp_polish(a, time_budget=100.0, verbose=True, seed=42):
    """LP-based polishing with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    last_best_it = 0

    while time.monotonic() - start < time_budget:
        g_lp = solve_lp(a)
        if g_lp is None:
            if it > 5:
                break
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            it += 1
            continue

        # Try multiple blending parameters
        candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        best_t = 0
        best_t_obj = best_obj
        best_t_a = best_a.copy()

        for t in candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_t_a = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_t_a
            a = best_t_a
            last_best_it = it
            if verbose and it % 2 == 0:
                print(f"        LP it={it}: obj={best_obj:.8f}, t={best_t:.2f}, elapsed={time.monotonic()-start:.0f}s")
        else:
            scale = 0.015 + 0.005 * min(it - last_best_it, 5)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            if it - last_best_it > 15:
                break

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)

    elif mode == 3:
        freqs = rng.integers(1, 6, size=3)

        a = 1.0

        a = np.maximum(a, 0.01)

    elif mode == 4:

        a = np.sin(x) ** rng.uniform(0.5, 3.0)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 5:
        c = rng.integers(n // 4, 3 * n // 4)
        w = rng.uniform(n // 10, n // 4)
        a = np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 6:
        # Two bumps
        c1, c2 = n // 4, 3 * n // 4
        w = n // 10
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))

    elif mode == 7:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1, 3)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase4 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 CD: obj={obj_cd:.8f}")

    # === Phase 5: Multi-n exploration (10% time) ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(300, n_current - 200), max(300, n_current - 100),
        n_current + 100, n_current + 200, n_current + 400,
        500, 600, 700, 800, 900, 1000, 1200, 1400
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000 and abs(n - n_current) > 50]

    tried = 0
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60 or tried >= 4:
            break
        tried += 1

        a_try = resample_sequence(best_seq, n_try)

        alloc1 = min(remaining * 0.3, 60.0)
        if alloc1 > 15:
            a_try = smoothed_gradient_descent(a_try, time_budget=alloc1,
                                              tau_init=3.0, tau_end=300.0, verbose=False)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.1, 30.0)
        if alloc2 > 10:
            a_try = fast_coord_descent(a_try, time_budget=alloc2, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts (5% time) ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        mode = (r * 3 + 2) % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.4 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.25, 60.0)
        if alloc > 10:
            a_r = smoothed_gradient_descent(a_r, time_budget=alloc,
                                            tau_init=3.0, tau_end=400.0, verbose=False)
            # Quick CD
            remaining2 = deadline - time.monotonic()
            alloc2 = min(remaining2 * 0.1, 20.0)
            if alloc2 > 5:
                a_r = fast_coord_descent(a_r, time_budget=alloc2, verbose=False)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        alloc = min(remaining * 0.3, 80.0)
        a_f1 = smoothed_gradient_descent(best_seq, time_budget=alloc,
                                         tau_init=20.0, tau_end=3000.0, verbose=False)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20 and HAVE_CVXPY:
        alloc = min(remaining * 0.3, 60.0)
        a_f2 = lp_polish(best_seq, time_budget=alloc, verbose=False)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 50.0)
        a_f3 = fast_coord_descent(best_seq, time_budget=alloc, verbose=False)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
