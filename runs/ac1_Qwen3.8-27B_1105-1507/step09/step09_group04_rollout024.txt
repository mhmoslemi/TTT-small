Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Aggressive LP-based polishing with active-set acceleration, 
coordinate descent refinement, multi-length exploration, and diverse restarts.
Key innovations: (1) reduced-constraint LP for speed, (2) adaptive blending 
schedule, (3) structured initializations based on window functions.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time
import math

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[l, j] = a[j] if 0 <= l - j <= n-1, i.e., j <= l <= j+n-1
        lo = j
        hi = min(2 * n - 1, j + n)
        A[lo:hi, j] = a[:hi - lo]
    return A

def solve_lp(a, verbose=False):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val
        except Exception:
            continue
    return None

def coord_descent(a, budget=60.0, rng=None, verbose=False):
    """Coordinate descent with incremental convolution updates."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
    a = a / S
    conv = compute_conv(a)
    
    best_obj = float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    
    stagnant = 0
    sweep = 0
    while time.monotonic() - start < budget:
        sweep += 1
        if rng is not None and sweep % 3 == 2:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        
        for k in order:
            d = np.zeros(2 * n - 1)
            d[k:k+n] = a
            
            lo = -a[k]
            hi = max(1.0/n, 1.0)
            
            # Golden section search
            gr = (np.sqrt(5) - 1) / 2.0
            a_gs, b_gs = lo, hi
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)
            
            def obj_delta(delta):
                val = conv + 2.0 * delta * d
                val[2*k] += delta * delta
                return float(np.max(val))
            
            fc = obj_delta(c_gs)
            fd = obj_delta(d_gs)
            
            for _ in range(30):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = obj_delta(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = obj_delta(d_gs)
            
            delta = (a_gs + b_gs) / 2.0
            new_obj = obj_delta(delta)
            if new_obj < best_obj - 1e-14:
                a[k] = max(0.0, a[k] + delta)
                conv += 2.0 * delta * d
                conv[2*k] += delta * delta
                S = float(np.sum(a))
                if S > 1e-10:
                    a /= S
                    conv /= (S * S)
                best_obj = float(np.max(conv))
                stagnant = 0
            else:
                stagnant += 1
        
        if stagnant > 3 * n:
            if rng is not None:
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv = compute_conv(a)
                best_obj = float(np.max(conv))
                stagnant = 0
            else:
                break
        
        if verbose and sweep % 5 == 0:
            print(f"      CD sweep {sweep}: obj={2*n*best_obj:.8f}, t={time.monotonic()-start:.1f}s")
    
    return best_a

def lp_polish(a, budget=200.0, rng=None, verbose=False):
    """LP-based polishing with adaptive blending."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    iter_count = 0
    last_t = 1.0
    
    while time.monotonic() - start < budget:
        g = solve_lp(a)
        if g is None:
            break
        
        iter_count += 1
        # Try multiple blending factors
        candidates = [0.3, 0.5, 0.7, 1.0, max(0.1, last_t * 0.8), min(1.0, last_t * 1.2)]
        best_t_obj = best_obj
        best_t_val = last_t
        best_a_local = best_a.copy()
        
        for t in candidates:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            if verbose:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={last_t:.2f}")
        else:
            # Perturb to escape
            if rng is not None:
                scale = 0.03 * min(1 + iter_count * 0.1, 3.0)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
            if iter_count > 30:
                break
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        x = np.linspace(0, np.pi, n)
        a = np.sin(x)
        a = np.maximum(a, 0.001)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.001)
    elif mode == 3:
        a = np.zeros(n)
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 10)
        a += np.exp(-((np.arange(n) - c1)**2) / (2*w**2))
        a += np.exp(-((np.arange(n) - c2)**2) / (2*w**2))
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + (n * x * (1-x))**2)
    elif mode == 5:
        a = rng.uniform(0, 1, n)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    return a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Aggressive LP-based polishing with adaptive blending, coordinate descent
refinement, multi-length exploration, and diverse structured restarts.
Key innovations: (1) Intensive LP phase with annealed blending schedule,
(2) efficient sparse-aware coordinate descent with incremental updates,
(3) structured initializations exploiting window functions and symmetry.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        l_lo = j
        l_hi = min(2 * n - 1, j + n)
        A[l_lo:l_hi, j] = a[:l_hi - l_lo]
    return A


def solve_lp(a, time_limit=10.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > time_limit:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0).astype(np.float64)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def coord_descent(a, budget=60.0, rng=None, verbose=False):
    """Coordinate descent with incremental convolution updates and golden section search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
    a = a / S
    conv = compute_conv(a)

    best_obj = float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()

    stagnant = 0
    sweep = 0
    gr = (np.sqrt(5) - 1) / 2.0

    while time.monotonic() - start < budget:
        sweep += 1
        if rng is not None and sweep % 4 == 3:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        obj_before_sweep = float(np.max(conv))

        for k in order:
            # Direction vector for coordinate k
            d = np.zeros(2 * n - 1)
            d[k:k + n] = a

            # Bounds
            lo = -a[k]
            hi = max(2.0 / n, 0.5 / np.sqrt(n))

            # Golden section search
            a_gs, b_gs = lo, hi
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)

            # Compute objective at candidate points
            base = conv
            fc = float(np.max(base + 2.0 * c_gs * d))
            fc = max(fc, base[2 * k] + 2.0 * c_gs * a[k] + c_gs * c_gs)
            fd = float(np.max(base + 2.0 * d_gs * d))
            fd = max(fd, base[2 * k] + 2.0 * d_gs * a[k] + d_gs * d_gs)

            for _ in range(25):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    tmp = float(np.max(base + 2.0 * c_gs * d))
                    tmp = max(tmp, base[2 * k] + 2.0 * c_gs * a[k] + c_gs * c_gs)
                    fc = tmp
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    tmp = float(np.max(base + 2.0 * d_gs * d))
                    tmp = max(tmp, base[2 * k] + 2.0 * d_gs * a[k] + d_gs * d_gs)
                    fd = tmp

            delta = (a_gs + b_gs) / 2.0

            # Evaluate improvement
            new_val = float(np.max(base + 2.0 * delta * d))
            new_val = max(new_val, base[2 * k] + 2.0 * delta * a[k] + delta * delta)

            if new_val < best_obj - 1e-14:
                a[k] = max(0.0, a[k] + delta)
                conv += 2.0 * delta * d
                conv[2 * k] += delta * delta
                S = float(np.sum(a))
                if S > 1e-10:
                    a /= S
                    conv /= (S * S)
                best_obj = float(np.max(conv))
                stagnant = 0
            else:
                stagnant += 1

        # Check for stagnation
        if stagnant > 2 * n:
            if rng is not None:
                scale = 0.03 + 0.01 * min(sweep, 20)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                S = a.sum()
                if S > 1e-10:
                    a = a / S
                    conv = compute_conv(a)
                    best_obj = float(np.max(conv))
                stagnant = 0
            else:
                break

        if verbose and sweep % 10 == 0:
            print(f"      CD sweep {sweep}: obj={2 * n * best_obj:.8f}, t={time.monotonic() - start:.1f}s")

    return best_a


def lp_polish(a, budget=200.0, rng=None, verbose=False):
    """LP-based polishing with adaptive annealed blending."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    iter_count = 0
    last_t = 1.0

    while time.monotonic() - start < budget:
        g = solve_lp(a)
        if g is None:
            break

        iter_count += 1

        # Annealed blending: start aggressive, become conservative
        progress = min(1.0, iter_count / 50.0)
        base_t = 1.0 - 0.7 * progress  # from 1.0 down to 0.3

        candidates = [
            base_t,
            min(1.0, base_t * 1.3),
            max(0.05, base_t * 0.7),
            1.0,
            0.5,
            max(0.1, last_t * 0.9),
            min(1.0, last_t * 1.1),
        ]
        candidates = sorted(set(np.round(candidates, 2)))

        best_t_obj = best_obj
        best_t_val = last_t
        best_a_local = best_a.copy()

        for t in candidates:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            if verbose and (iter_count <= 5 or iter_count % 5 == 0):
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={last_t:.3f}, "
                      f"lp_peak={float(np.max(compute_conv(a))):.6e}")
        else:
            # Perturb to escape local minimum
            if rng is not None:
                scale = 0.02 + 0.005 * min(iter_count, 30)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
            if iter_count > 60:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations based on structured window functions."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        # Half-sine window
        x = np.linspace(0, np.pi, n)
        a = np.sin(x)
        a = np.maximum(a, 0.001)
    elif mode == 2:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.001)
    elif mode == 3:
        # Twin Gaussians
        a = np.zeros(n)
        c1, c2 = n // 4, 3 * n // 4
        w = max(2, n // 8)
        idx = np.arange(n)
        a += np.exp(-((idx - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((idx - c2) ** 2) / (2 * w ** 2))
    elif mode == 4:
        # Lorentzian / Cauchy-like
        x = np.linspace(-1, 1, n)
        a = 1.0 / (1.0 + 4 * n * x * x)
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        alpha = rng.uniform(-1.5, 1.5)
        a = x ** alpha
    elif mode == 6:
        # Flat-top with tapered edges
        a = np.ones(n)
        taper = max(3, n // 10)
        for i in range(min(taper, n)):
            a[i] *= 0.5 + 0.5 * np.sin(0.5 * np.pi * i / taper)
            a[n - 1 - i] *= 0.5 + 0.5 * np.sin(0.5 * np.pi * i / taper)
    elif mode == 7:
        # Bilinear (tent)
        mid = n // 2
        a = 1.0 - np.abs(np.arange(n) - mid) / (mid + 1)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    return a


def pair_coord_descent(a, budget=60.0, rng=None, verbose=False):
    """Coordinate descent that updates symmetric pairs (k, n-1-k) together."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
    a = a / S
    conv = compute_conv(a)

    best_obj = float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    gr = (np.sqrt(5) - 1) / 2.0
    sweep = 0

    # Only update first half (pairs with second half by symmetry)
    half = (n + 1) // 2

    while time.monotonic() - start < budget:
        sweep += 1
        for k in range(half):
            k2 = n - 1 - k
            if k == k2:
                # Single coordinate in the middle
                d = np.zeros(2 * n - 1)
                d[k:k + n] = a
                lo = -a[k]
                hi = 1.0 / n
                a_gs, b_gs = lo, hi
                c_gs = b_gs - gr * (b_gs - a_gs)
                d_gs = a_gs + gr * (b_gs - a_gs)

                def obj_single(delta):
                    val = conv + 2.0 * delta * d
                    val[2 * k] += delta * delta
                    return float(np.max(val))

                fc = obj_single(c_gs)
                fd = obj_single(d_gs)
                for _ in range(20):
                    if fc < fd:
                        b_gs = d_gs
                        d_gs = c_gs
                        fd = fc
                        c_gs = b_gs - gr * (b_gs - a_gs)
                        fc = obj_single(c_gs)
                    else:
                        a_gs = c_gs
                        c_gs = d_gs
                        fc = fd
                        d_gs = a_gs + gr * (b_gs - a_gs)
                        fd = obj_single(d_gs)

                delta = (a_gs + b_gs) / 2.0
                new_val = obj_single(delta)
                if new_val < best_obj - 1e-14:
                    a[k] = max(0.0, a[k] + delta)
                    conv += 2.0 * delta * d
                    conv[2 * k] += delta * delta
                    S = float(np.sum(a))
                    if S > 1e-10:
                        a /= S
                        conv /= (S * S)
                    best_obj = float(np.max(conv))
            else:
                # Update pair (k, k2) with shared delta
                d = np.zeros(2 * n - 1)
                d[k:k + n] = a
                d[k2:k2 + n] = a

                lo = -a[k]
                hi = 1.0 / n
                a_gs, b_gs = lo, hi
                c_gs = b_gs - gr * (b_gs - a_gs)
                d_gs = a_gs + gr * (b_gs - a_gs)

                def obj_pair(delta):
                    val = conv + 2.0 * delta * d
                    val[2 * k] += delta * delta
                    if k != k2:
                        val[2 * k2] += delta * delta
                    return float(np.max(val))

                fc = obj_pair(c_gs)
                fd = obj_pair(d_gs)
                for _ in range(20):
                    if fc < fd:
                        b_gs = d_gs
                        d_gs = c_gs
                        fd = fc
                        c_gs = b_gs - gr * (b_gs - a_gs)
                        fc = obj_pair(c_gs)
                    else:
                        a_gs = c_gs
                        c_gs = d_gs
                        fc = fd
                        d_gs = a_gs + gr * (b_gs - a_gs)
                        fd = obj_pair(d_gs)

                delta = (a_gs + b_gs) / 2.0
                new_val = obj_pair(delta)
                if new_val < best_obj - 1e-14:
                    a[k] = max(0.0, a[k] + delta)
                    a[k2] = max(0.0, a[k2] + delta)
                    conv += 2.0 * delta * d
                    conv[2 * k] += delta * delta
                    if k != k2:
                        conv[2 * k2] += delta * delta
                    S = float(np.sum(a))
                    if S > 1e-10:
                        a /= S
                        conv /= (S * S)
                    best_obj = float(np.max(conv))

        if verbose and sweep % 5 == 0:
            print(f"      PCD sweep {sweep}: obj={2 * n * best_obj:.8f}, t={time.monotonic() - start:.1f}s")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search orchestrator.
    Phases: LP polish -> CD -> LP -> multi-n -> restarts -> final polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Intensive LP Polishing (main workhorse) ===
    print(f"\n=== Phase 1: LP Polishing (intensive) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 500.0)
    if p1_budget > 30:
        a_lp = lp_polish(best_seq, budget=p1_budget, rng=rng, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 LP: obj={obj_lp:.8f}")

    # === Phase 2: Coordinate Descent ===
    print(f"\n=== Phase 2: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.12, 100.0)
    if p2_budget > 20:
        a_cd = coord_descent(best_seq, budget=p2_budget, rng=rng, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase2 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 CD: obj={obj_cd:.8f}")

    # === Phase 3: Second LP round ===
    print(f"\n=== Phase 3: LP Round 2 ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.20, 200.0)
    if p3_budget > 30:
        a_lp2 = lp_polish(best_seq, budget=p3_budget, rng=rng, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase3 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 LP: obj={obj_lp2:.8f}")

    # === Phase 4: Pair coordinate descent (exploit symmetry) ===
    print(f"\n=== Phase 4: Pair CD ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.08, 60.0)
    if p4_budget > 15:
        a_pcd = pair_coord_descent(best_seq, budget=p4_budget, rng=rng, verbose=True)
        obj_pcd = compute_obj(a_pcd)
        if obj_pcd < best_score:
            best_score = obj_pcd
            best_seq = a_pcd.copy()
            print(f"  Phase4 PCD: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 PCD: obj={obj_pcd:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    n_options = sorted(set([
        max(400, n_current - 150), max(400, n_current - 50),
        n_current + 100, n_current + 200,
        600, 1000, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500 and abs(n - n_current) >= 50]

    for n_try in n_options[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        alloc_lp = min(remaining * 0.25, 50.0)
        if alloc_lp > 15:
            a_try = lp_polish(a_try, budget=alloc_lp, rng=rng)

        alloc_cd = min(remaining * 0.1, 25.0)
        if alloc_cd > 10:
            a_try = coord_descent(a_try, budget=alloc_cd, rng=rng)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 70:
            break

        mode = (r + 1) % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = a_r.sum()
        if s_r > 0:
            a_r = a_r / s_r

        alloc_lp = min(remaining * 0.2, 40.0)
        if alloc_lp > 12:
            a_r = lp_polish(a_r, budget=alloc_lp, rng=rng)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 40:
        alloc = min(remaining * 0.35, 80.0)
        a_final = lp_polish(best_seq, budget=alloc, rng=rng, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.35, 40.0)
        a_cd_f = coord_descent(best_seq, budget=alloc, rng=rng)
        obj_cd_f = compute_obj(a_cd_f)
        if obj_cd_f < best_score:
            best_score = obj_cd_f
            best_seq = a_cd_f.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 20.0)
        a_pcd_f = pair_coord_descent(best_seq, budget=alloc, rng=rng)
        obj_pcd_f = compute_obj(a_pcd_f)
        if obj_pcd_f < best_score:
            best_score = obj_pcd_f
            best_seq = a_pcd_f.copy()
            print(f"  Final PCD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
