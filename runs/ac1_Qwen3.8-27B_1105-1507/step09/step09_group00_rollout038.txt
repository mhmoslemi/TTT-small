"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses weighted LP polishing with peak targeting, Newton correction exploiting
anti-diagonal Hessian structure, differential evolution on Fourier parameters,
and multi-length exploration.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize, differential_evolution
import cvxpy as cp
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_weighted(a, weights=None, verbose=False):
    """
    Solve weighted LP: min sum(w_k * M_k) s.t. conv(a,g)[k] <= M_k for all k,
    sum(g)=1, g>=0.
    If weights is None, uniform weights are used (standard min-max LP).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    if weights is None:
        # Standard min-max LP
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
    else:
        # Weighted LP: minimize weighted sum of per-position bounds
        g = cp.Variable(n, nonneg=True)
        M_vec = cp.Variable(2 * n - 1)
        constraints = [A @ g <= M_vec, cp.sum(g) == 1]
        w = np.array(weights)
        w = w / w.sum()
        prob = cp.Problem(cp.Minimize(cp.sum(w * M_vec)), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    if weights is None:
                        return g_val / s, float(M.value)
                    else:
                        return g_val / s, float(M_vec.value)
        except Exception:
            continue
    return None, None

def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of Q_k(a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    N = 1
    while N < 3 * n:
        N <<= 1
    
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv

def fourier_perturbation(a, rng, strength=0.05):
    """Add low-frequency Fourier modes as structured perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 3
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

def peak_targeted_lp_polish(a, time_budget=100.0, verbose=False, seed=12345):
    """
    Enhanced LP polishing with peak targeting:
    - Identify top peaks in conv(a,a)
    - Use weighted LP to bias reduction towards those peaks
    - Backtracking line search
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a

    best_obj = compute_obj(a)

    rng = np.random.default_rng(seed)
    iter_count = 0
    stall_count = 0

        if remaining < 5:
            break

        max_c = float(np.max(conv_a))

        peak_vals = conv_a[peak_indices]

        weights = np.ones(len(conv_a))

            weights[idx] = 1.0 + 5.0 * (conv_a[idx] / max_c) ** 2

        g_lp, _ = solve_lp_weighted(a, weights=weights)
        if g_lp is None:

            g_lp, _ = solve_lp_weighted(a, weights=None)
            if g_lp is None:

        # Backtracking line search
        best_t_obj = best_obj
        best_a_local = best_a.copy()

def newton_correction(a, time_budget=30.0, verbose=False):
    """
    Newton-like correction exploiting the anti-diagonal Hessian structure.
    For the peak at position k*, the Hessian is a permutation matrix,
    so we can compute the Newton step analytically.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # The peak c[k*] = sum_j a[j]*a[k*-j] (valid indices)
        # Gradient: dc[k*]/da[i] = a[k*-i] + a[i-k*] (valid terms)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] += a[j]
            j2 = i - k_star
            if 0 <= j2 < n:
                grad[i] += a[j2]
        
        # Hessian: H[i,j] = 1 if j = k*-i (valid), plus symmetry
        # Newton step: d = -H^{-1} grad / (grad^T H^{-1} grad) * step_size
        # H^{-1} is the complementary permutation
        # H maps i -> k*-i, so H^{-1} maps i -> k*-i as well (it's an involution)
        # Actually H[i,j] = 1 if i+j = k* (for valid range)
        # So (H x)[i] = x[k*-i] (valid)
        # H is a permutation matrix, so H^{-1} = H^T
        
        # For the quadratic form: a^T H a = sum_i a[i]*a[k*-i]
        # Newton direction: d = -H^{-1} grad = -(grad permuted by H)
        # i.e., d[i] = -grad[k*-i] (valid)
        
        # But we need to project onto sum(d)=0 and d >= -a
        # Simple approach: just try d = -eps * grad (steepest descent on the peak)
        # with line search
        
        # Better: use the structure. The peak c[k*] = sum_{valid j} a[j]*a[k*-j]
        # To minimize this, we want a[j] and a[k*-j] to be small simultaneously
        # Newton step for the quadratic: d = -a + mean(a)*1 (projection onto sum=0)
        # This makes the peak = sum (mean(a))^2 = n * (1/n)^2 = 1/n
        
        # But this might increase other peaks. So use a damped version.
        for eps in [0.5, 0.3, 0.2, 0.1, 0.05, 0.01]:
            # d[i] = -a[i] + mean(a) makes the peak flat
            # But this is too aggressive. Use: d[i] = -eps*(a[i] - mean(a))
            d = -eps * (a - a.mean())
            a_try = a + d
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-15:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                break
        
        if verbose:
            print(f"    Newton: obj={best_obj:.8f}, k*={k_star}")
    
    return best_a

def fourier_param_search(a_init, time_budget=120.0, n_modes=12, verbose=False):
    """
    Differential evolution on Fourier parameters for global exploration.
    Parameterize a[j] = (1/n) * max(0, 1 + sum_k alpha_k * cos(2*pi*k*j/n + phi_k))

    """

    # Extract real/imag for first n_modes
    params_init = []
    for k in range(1, n_modes + 1):
        params_init.append(abs(A[k]) / n)  # amplitude
        params_init.append(np.angle(A[k]))  # phase
    
    def fourier_obj(params):
        alphas = params[0::2][:n_modes]
        phases = params[1::2][:n_modes]
        j = np.arange(n)
        a_f = np.ones(n) / n
        for k in range(1, n_modes + 1):

        a_f = np.maximum(a_f, 0)
        s = float(np.sum(a_f))
        if s < 0.01:
            return 100.0
        a_f = a_f / s
        conv = compute_conv(a_f)
        return 2.0 * n * float(np.max(conv))
    
    bounds = [(0, 0.5), (0, 2*np.pi)] * n_modes
    
    try:
        result = differential_evolution(

        )
        if result.success or result.fun < fourier_obj(params_init):
            best_params = result.x
            a_fourier = np.ones(n) / n
            alphas = best_params[0::2][:n_modes]
            phases = best_params[1::2][:n_modes]
            j = np.arange(n)
            for k in range(1, n_modes + 1):
                a_fourier += alphas[k-1] * np.cos(2 * np.pi * k * j / n + phases[k-1])
            a_fourier = np.maximum(a_fourier, 0)
            s = float(np.sum(a_fourier))

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 6, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_standard(a):
    """Solve standard min-max LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_weighted(a, peak_frac=0.5, weight_boost=3.0):
    """
    Weighted LP: min M s.t. conv(a,g)[k] <= w_k * M, where w_k < 1 for
    peak positions (tighter constraint) and w_k > 1 for others (looser).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv_self = compute_conv(a_norm)
    max_c = float(np.max(conv_self))
    
    # Create weights: peaks get tighter bounds (w < 1)
    w = np.ones(len(conv_self))
    threshold = max_c * peak_frac
    for k in range(len(conv_self)):
        if conv_self[k] > threshold:
            w[k] = 1.0 / weight_boost
    
    A = build_conv_matrix(a_norm)
    # Effective constraint: (A @ g)[k] <= w[k] * M, i.e., (A/w @ g)[k] <= M
    A_w = A / w[:, None]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_w @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_proximal(a, mu=0.01):
    """
    Proximal LP (QP): min M + (1/(2*mu)) * ||g - a||^2
    s.t. conv(a,g) <= M, sum(g)=1, g>=0
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    objective = cp.Minimize(M + (1.0 / (2.0 * mu)) * cp.sum_squares(g - a_norm))
    prob = cp.Problem(objective, constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)

        except Exception:
            continue
    return None, None


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv positions."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    N = 1
    while N < 3 * n:
        N <<= 1
    
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv

def fourier_perturbation(a, rng, strength=0.05):
    """Add low-frequency Fourier modes as structured perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 4
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

def newton_peak_reduction(a, time_budget=30.0, verbose=False):
    """
    Reduce the dominant peak using Newton-like steps.
    For peak at position k*, c[k*] = sum_j a[j]*a[k*-j].
    The gradient is g[i] = a[k*-i] + a[i-k*] (valid terms).
    We take damped steps in the negative gradient direction.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_peaks = np.argsort(conv)[::-1][:5]
        
        # Combined gradient from top peaks
        grad = np.zeros(n)
        for k_star in top_peaks:
            weight = conv[k_star] / max(conv[top_peaks[0]], 1e-15)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    grad[i] += weight * a[j]
                j2 = i - k_star
                if 0 <= j2 < n:
                    grad[i] += weight * a[j2]
        
        # Project gradient to sum-zero subspace
        grad = grad - grad.mean()
        
        # Damped descent with line search
        improved = False
        for eps in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_try = a - eps * grad
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-15:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        
        if not improved and verbose:
            break
    
    return best_a


def mirror_symmetrize(a):
    """Make sequence mirror-symmetric."""
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    return a_sym


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        a_newton = newton_peak_reduction(best_seq, time_budget=p3_budget, verbose=True)
        obj_newton = compute_obj(a_newton)
        if obj_newton < best_score - 1e-15:
            best_score = obj_newton
            best_seq = a_newton.copy()
            print(f"  Newton: obj={best_score:.8f} **")
        else:
            print(f"  Newton: obj={obj_newton:.8f}")
    
    # === Phase 4: LP after Newton ===
    print(f"\n=== Phase 4: LP After Newton ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.18, 150.0)
        a_lp = peak_targeted_lp_polish(best_seq, time_budget=alloc, verbose=True, seed=5000)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score - 1e-15:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase4 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 LP: obj={obj_lp:.8f}")
    
    # === Phase 5: Mirror + Polish ===
    print(f"\n=== Phase 5: Mirror Symmetry ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        a_mirror = mirror_symmetrize(best_seq)
        a_mirror = a_mirror / a_mirror.sum()
        alloc = min(remaining * 0.1, 80.0)
        a_mirror = peak_targeted_lp_polish(a_mirror, time_budget=alloc, verbose=False, seed=6000)
        obj_mirror = compute_obj(a_mirror)
        if obj_mirror < best_score - 1e-15:
            best_score = obj_mirror
            best_seq = a_mirror.copy()
            print(f"  Mirror: obj={best_score:.8f} **")
        else:
            print(f"  Mirror: obj={obj_mirror:.8f}")
    
    # === Phase 6: Fourier DE Global Search ===
    print(f"\n=== Phase 6: Fourier DE ===")
    remaining = deadline - time.monotonic()
    if remaining > 120:
        alloc = min(remaining * 0.15, 150.0)
        a_fourier = fourier_param_search(best_seq, time_budget=alloc, n_modes=10, verbose=True)
        obj_fourier = compute_obj(a_fourier)
        if obj_fourier < best_score - 1e-15:
            best_score = obj_fourier
            best_seq = a_fourier.copy()
            print(f"  Fourier DE: obj={best_score:.8f} **")
            # Polish
            remaining = deadline - time.monotonic()
            if remaining > 50:
                pol_budget = min(remaining * 0.2, 80.0)
                a_pol = peak_targeted_lp_polish(best_seq, time_budget=pol_budget,
                                                verbose=True, seed=7000)
                obj_pol = compute_obj(a_pol)
                if obj_pol < best_score:
                    best_score = obj_pol
                    best_seq = a_pol.copy()
                    print(f"  Fourier+LP: obj={best_score:.8f} **")
        else:
            print(f"  Fourier DE: no improvement")
    
    # === Phase 7: Multi-n Exploration ===
    print(f"\n=== Phase 7: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    n_options = sorted(set([
        max(600, n_best - 100), n_best + 50, n_best + 100,
        n_best + 200, 750, 850, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 600 <= n <= 1400 and abs(n - n_best) > 50]
    n_options = n_options[:5]
    
    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        
        a_try = resample_sequence(best_seq, n_try)
        alloc1 = min(remaining * 0.12, 40.0)
        if alloc1 > 20:
            a_try = adaptive_smoothed_descent(a_try, time_budget=alloc1,
                                              tau_init=10.0, tau_end=400.0)
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.1, 50.0)
        if alloc2 > 15:
            a_try = peak_targeted_lp_polish(a_try, time_budget=alloc2, seed=n_try)
        
        obj_try = compute_obj(a_try)
        if obj_try < best_score - 1e-15:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")
    
    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    
    # High-tau GD
    if remaining > 40:
        alloc = min(remaining * 0.15, 60.0)
        a_sgd = adaptive_smoothed_descent(best_seq, time_budget=alloc,
                                          tau_init=50.0, tau_end=1500.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score - 1e-15:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Final GD: obj={best_score:.8f} **")
    
    # Multiple LP passes with perturbation
    remaining = deadline - time.monotonic()
    for lp_rnd in range(5):
        if remaining < 20:
            break
        alloc = min(remaining * 0.2, 100.0)
        a_lp = peak_targeted_lp_polish(best_seq, time_budget=alloc, verbose=False,
                                       seed=9000 + lp_rnd * 137)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score - 1e-15:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP {lp_rnd}: obj={best_score:.8f} **")
        else:
            if lp_rnd < 4:
                best_seq = fourier_perturbation(best_seq, rng, strength=0.006 * (lp_rnd + 1))
            else:
                break
    
    # Newton + LP
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_n = newton_peak_reduction(best_seq, time_budget=min(remaining * 0.15, 25.0))
        obj_n = compute_obj(a_n)
        if obj_n < best_score - 1e-15:
            best_score = obj_n
            best_seq = a_n.copy()
            print(f"  Final Newton: obj={best_score:.8f} **")
    
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.5, 60.0)
        a_lp = peak_targeted_lp_polish(best_seq, time_budget=alloc, verbose=False, seed=15000)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score - 1e-15:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Ultra-final LP: obj={best_score:.8f} **")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
