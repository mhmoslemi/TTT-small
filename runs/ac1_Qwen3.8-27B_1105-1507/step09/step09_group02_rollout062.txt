"""
Optimizes step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses QR-code initialization, LP with golden-section line search,
Nesterov smoothed gradient, and simulated annealing.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

# Helper functions at top level

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # Row k has A[k,j] = a[k-j] when 0 <= k-j < n
        # So for column j, rows k where 0 <= k-j < n, i.e., j <= k <= j+n-1
        # But k ranges from 0 to 2n-2
        k_start = j  # k-j >= 0 => k >= j
        k_end = min(2*n-1, j+n)  # k-j < n => k < j+n
        A[k_start:k_end, j] = a[:k_end-k_start]
    return A

def solve_lp(a, solver=None):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]
    if solver:
        solvers = [solver] + [s for s in solvers if s != solver]
    
    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv, float(M.value)
        except:
            continue
    return None, None

def golden_section_line_search(a, g, tol=1e-6, max_iter=30):
    """Find optimal t in [0,1] minimizing compute_obj((1-t)*a + t*g)"""
    a_norm = a / float(np.sum(a))
    g_norm = g / float(np.sum(g))
    
    # First check endpoints and midpoint
    f0 = compute_obj(a_norm)
    f1 = compute_obj(g_norm)
    fm = compute_obj(0.5 * a_norm + 0.5 * g_norm)
    
    if f0 <= fm and f0 <= f1:
        return 0.0, f0
    if f1 <= fm and f1 <= f0:
        return 1.0, f1

    fd = compute_obj((1-d)*a_norm + d*g_norm)

            d = c

def qr_code_sequence(p):
    """Generate quadratic residue sequence of length p (prime, p ≡ 3 mod 4)."""
    # Check p is prime and p ≡ 3 mod 4
    if p < 3 or p % 4 != 3:
        return None
    # Simple primality test
    for i in range(2, int(np.sqrt(p)) + 1):
        if p % i == 0:
            return None
    
    a = np.zeros(p)
    for i in range(1, p):
        if pow(i, (p-1)//2, p) == 1:  # i is quadratic residue
            a[i] = 1
    
    return a  # a[0] = 0, a[i] = 1 if i is QR

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))

        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)

    prev_g = None

            ss = float(np.dot(s_vec, s_vec))
            sy2 = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy2) > 1e-12:

                lr = 0.5*lr + 0.5*np.clip(bb, 1e-8, 5.0/np.sqrt(n))
        
        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / sn
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        prev_a = a.copy()
        prev_g = grad.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)
        
        if it - (prev_best_it if 'prev_best_it' in dir() else 0) > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5

**Alternative: Direct convex optimization via reformulation**

The problem is: min max_k (a*a)[k] s.t. a >= 0, sum(a) = 1.

This is a quadratic program with the objective being the maximum of quadratics. It's non-convex.

However, we can use the following trick: introduce a variable p and minimize p subject to (a*a)[k] <= p for all k. This is still non-convex.

But we can use a sequence of convex relaxations:
- Start with a feasible point a_0
- At each iteration, linearize the quadratic constraints at the current point
- Solve the resulting LP
- Update

This is essentially what the cutting plane method does, and it's related to the LP approach I'm already using.

**Another alternative: Use the fact that the problem is invariant under permutation**

Actually, it's not invariant under arbitrary permutation, but it is invariant under reversal: if a is optimal, so is a_reversed.

**Yet another idea: Use the KKT conditions to verify optimality**

At the optimum:
- For all k where (a*a)[k] = M: the subgradient condition must hold
- For all k where (a*a)[k] < M: no condition
- For all i where a_i > 0: the gradient w.r.t. a_i must equal the Lagrange multiplier for the sum constraint
- For all i where a_i = 0: the gradient w.r.t. a_i must be >= the Lagrange multiplier

This can help us verify how close we are to optimal.

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: QR-code initialization, LP with golden-section line search,
Nesterov smoothed gradient descent, and simulated annealing for diversity.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.fft import fft, ifft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    rows = np.arange(2 * n - 1)[:, None]
    cols = np.arange(n)[None, :]
    idx = rows - cols
    mask = (idx >= 0) & (idx < n)
    A = np.zeros((2 * n - 1, n))
    A[mask] = a[idx[mask]]
    return A


def solve_lp(a):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]
    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv, float(M.value)
        except Exception:
            continue
    return None, None


def golden_section_lsearch(a, g):
    a_norm = a / float(np.sum(a))
    g_norm = g / float(np.sum(g))
    
    f0 = compute_obj(a_norm)
    f1 = compute_obj(g_norm)
    
    if f1 >= f0:
        # Check if any interior point is better
        fmid = compute_obj(0.5 * a_norm + 0.5 * g_norm)
        if fmid >= f0:
            return 0.0, f0
        # Search in [0, 0.5]
        lo, hi = 0.0, 0.5
    else:
        fmid = compute_obj(0.5 * a_norm + 0.5 * g_norm)
        if fmid >= f1:
            return 1.0, f1
        # Search in [0.5, 1]
        lo, hi = 0.5, 1.0
    
    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = compute_obj((1 - c) * a_norm + c * g_norm)

def qr_sequence(p):
    if p < 3 or p % 4 != 3:
        return None
    for i in range(2, int(np.sqrt(p)) + 1):
        if p % i == 0:
            return None
    a = np.zeros(p)
    for i in range(1, p):
        if pow(i, (p - 1) // 2, p) == 1:
            a[i] = 1.0
    return a


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_smoothed(a, time_budget, tau_init=10.0, tau_end=500.0):
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    best_it = 0
    start = time.time()
    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        progress = min(1.0, elapsed / max(time_budget, 1.0))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)
        
        y = a + beta * v
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy < 1e-10:
            y = best_a.copy()
        else:
            y = y / sy
        
        conv = compute_conv(y)
        max_c = float(np.max(conv))
        exp_v = np.exp(tau * (conv - max_c))
        p = exp_v / float(np.sum(exp_v))
        
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        Y = fft(y, N)
        g_full = ifft(P * np.conj(Y)).real
        grad = 2.0 * g_full[:n]
        
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = grad - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy2 = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy2) > 1e-12:
                bb = ss / abs(sy2)
                lr = 0.5 * lr + 0.5 * np.clip(bb, 1e-8, 5.0 / np.sqrt(n))
        
        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / sn
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        prev_a = a.copy()
        prev_g = grad.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)
        
        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
    
    return best_a


def simulate_annealing(a, time_budget, rng):

        T = 0.05 * np.exp(-3.0 * elapsed / max(time_budget, 1.0))
        if T < 1e-8:
            T = 1e-8
        
        scale = 0.01 + 0.04 * (T / 0.05)

            a = a_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
    
    return best_a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: QR-code and structured initialization, LP with golden-section line search,
Nesterov accelerated smoothed gradient descent, simulated annealing for diversity,
and adaptive multi-length exploration. Key improvements: better initialization diversity,
proper line search for LP steps, and aggressive perturbation to escape local minima.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    rows = np.arange(2 * n - 1)[:, None]
    cols = np.arange(n)[None, :]
    idx = rows - cols
    mask = (idx >= 0) & (idx < n)
    A = np.zeros((2 * n - 1, n))
    A[mask] = a[idx[mask]]
    return A


def solve_lp(a):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    solvers = [cp.MOSEK, cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]
    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv, float(M.value)
        except Exception:
            continue
    return None, None

    idx = points.index(best_t)
    lo = points[max(0, idx - 1)]
    hi = points[min(len(points) - 1, idx + 1)]
    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = compute_obj(np.maximum((1 - c) * a_norm + c * g_norm, 0))
    fd = compute_obj(np.maximum((1 - d) * a_norm + d * g_norm, 0))
    for _ in range(12):
        if hi - lo < 1e-4:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = compute_obj(np.maximum((1 - c) * a_norm + c * g_norm, 0))
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = compute_obj(np.maximum((1 - d) * a_norm + d * g, 0) if False else np.maximum((1 - d) * a_norm + d * g_norm, 0))
    t = (lo + hi) / 2
    a_try = (1 - t) * a_norm + t * g_norm
    a_try = np.maximum(a_try, 0)
    st = float(np.sum(a_try))
    if st > 1e-10:
        f_t = compute_obj(a_try / st)
    else:
        f_t = np.inf
    if f_t < best_f:
        return t, f_t
    return best_t, best_f


def qr_sequence(p):
    if p < 3 or p % 4 != 3:
        return None
    for i in range(2, int(np.sqrt(p)) + 1):
        if p % i == 0:
            return None
    a = np.zeros(p)
    for i in range(1, p):
        if pow(i, (p - 1) // 2, p) == 1:
            a[i] = 1.0
    return a


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_smoothed(a, time_budget, tau_init=10.0, tau_end=500.0):
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    best_it = 0
    start = time.time()
    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        progress = min(1.0, elapsed / max(time_budget, 1.0))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)
        y = a + beta * v
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy < 1e-10:
            y = best_a.copy()
            sy = 1.0
        else:
            y = y / sy
        conv = compute_conv(y)
        max_c = float(np.max(conv))
        exp_v = np.exp(tau * (conv - max_c))
        p = exp_v / float(np.sum(exp_v))
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        Y = fft(y, N)
        g_full = ifft(P * np.conj(Y)).real
        grad = 2.0 * g_full[:n]
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = grad - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy2 = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy2) > 1e-12:
                bb = ss / abs(sy2)
                lr = 0.5 * lr + 0.5 * float(np.clip(bb, 1e-8, 5.0 / np.sqrt(n)))
        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / sn
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        prev_a = a.copy()
        prev_g = grad.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)
        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
    return best_a


def simulate_annealing(a, time_budget, rng):
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    cur_obj = best_obj
    start = time.time()
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = 0.03 * np.exp(-4.0 * elapsed / max(time_budget, 1.0))
        if T < 1e-10:
            T = 1e-10
        scale = 0.005 + 0.05 * (T / 0.03)
        a_try = a * np.exp(rng.normal(0, scale, n))
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj_try = compute_obj(a_try)
        delta = obj_try - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_try
            cur_obj = obj_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
    return best_a


def lp_polish(a, time_budget, rng, use_lsearch=True, verbose=False, deadline=None):
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.time()
    it = 0
    stall = 0
    while True:
        if time.time() - start >= time_budget:
            break
        if deadline is not None and time.time() >= deadline:
            break
        if verbose and it % 20 == 0:
            elapsed = time.time() - start
            print(f"    LP it{it}: obj={best_obj:.10f}, t={elapsed:.0f}s")
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break
        if use_lsearch:
            t_best, f_best = lsearch_blend(best_a, g_lp)
        else:
            t_best = 0.5
            a_try = (1 - t_best) * best_a + t_best * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            f_best = compute_obj(a_try / st) if st > 1e-10 else np.inf
        if f_best < best_obj - 1e-13:
            a_new = (1 - t_best) * best_a + t_best * g_lp
            a_new = np.maximum(a_new, 0)
            st = float(np.sum(a_new))
            if st > 1e-10:
                a_new = a_new / st
            best_obj = f_best
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall % 3 == 0:
                scale = 0.008 + 0.004 * min(stall, 20)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
            if stall > 30:
                a = best_a * np.exp(rng.normal(0, 0.06, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 10
            if stall > 60:
                break
        it += 1
    return best_a


def diverse_init(n, rng, mode=0):
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.3, 1.0, n)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(0, np.pi, n)
        a = np.abs(np.cos(x)) + 0.05
        return a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = 1.0 / (np.sqrt(1 - (2*x - 1)**2 + 0.01))
        return a / a.sum()
    elif mode == 4:
        a = np.ones(n)
        a[::2] = 0.4
        return a / a.sum()
    elif mode == 5:
        freqs = rng.integers(1, 8, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.05)
        return a / a.sum()
    elif mode == 6:
        a = np.ones(n)
        a[:n//4] = 1.5
        a[3*n//4:] = 1.5
        return a / a.sum()
    elif mode == 7:
        centers = rng.integers(0, n, size=3)
        widths = rng.integers(n//20, n//5, size=3)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c)**2) / (2 * w**2))
        a += 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1500):
    start_time = time.time()
    deadline = start_time + budget_s - 25.0
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Phase 1: Diverse initialization
    print("=== Phase 1: Initialization ===")
    candidates = []

    # Parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.all(np.isfinite(parent_arr)):
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                candidates.append((obj, parent.copy()))
                print(f"  Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    # QR codes
    for p in [787, 779, 773, 761, 751, 743, 823, 829, 839, 853, 859, 863, 881, 883, 887]:
        qr = qr_sequence(p)
        if qr is not None:
            qr = qr / qr.sum()
            obj = compute_obj(qr)
            candidates.append((obj, qr.copy()))
    obj_qr = min(c[0] for c in candidates if len(c[1]) in [787,779,773,761,751,743,823,829,839,853,859,863,881,883,887]) if any(len(c[1]) < 1000 for c in candidates) else np.inf
    print(f"  Best QR: obj={obj_qr:.8f}")

    # Structured inits at various lengths
    for n in [500, 600, 700, 800, 900, 1000, 1100, 1200]:
        for mode in range(8):
            a = diverse_init(n, rng, mode)
            obj = compute_obj(a)
            candidates.append((obj, a))

    # Sort and pick top candidates
    candidates.sort(key=lambda x: x[0])
    top5 = candidates[:5]
    print("  Top 5 inits:")
    for obj, seq in top5:
        print(f"    n={len(seq)}, obj={obj:.8f}")

    best_score = top5[0][0]
    best_seq = top5[0][1].copy()
    n_best = len(best_seq)

    # Quick polish on best
    remaining = deadline - time.time()
    if remaining > 30:
        polished = lp_polish(best_seq, min(30, remaining * 0.03), rng, use_lsearch=False)
        obj_pol = compute_obj(polished)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = polished
            print(f"  After quick polish: obj={best_score:.8f}")

    # Also try polishing the 2nd and 3rd best if time
    for i in [1, 2]:
        if i >= len(candidates):
            break
        remaining = deadline - time.time()
        if remaining < 60:
            break
        a_try = candidates[i][1].copy()
        polished = lp_polish(a_try, min(30, remaining * 0.04), rng, use_lsearch=False)
        obj_pol = compute_obj(polished)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = polished
            n_best = len(best_seq)
            print(f"  Candidate {i+1} improved: n={n_best}, obj={best_score:.8f}")

    print(f"  Best after init: n={n_best}, obj={best_score:.8f}")

    # Phase 2: Main LP with line search
    print(f"\n=== Phase 2: Main LP with Line Search ===")
    remaining = deadline - time.time()
    lp_time = remaining * 0.42
    if lp_time > 60:
        result = lp_polish(best_seq, lp_time, rng, use_lsearch=True, verbose=True, deadline=deadline)
        obj_lp = compute_obj(result)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = result.copy()
            print(f"  After main LP: obj={best_score:.8f}")
        else:
            print(f"  Main LP final: obj={obj_lp:.8f}")

    # Phase 3: Nesterov smoothed GD
    print(f"\n=== Phase 3: Nesterov GD ===")
    remaining = deadline - time.time()
    gd_time = min(remaining * 0.18, 200.0)
    if gd_time > 20:
        a_gd = nesterov_smoothed(best_seq, gd_time, tau_init=5.0, tau_end=400.0)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  After GD: obj={best_score:.8f} **")
        else:
            print(f"  GD: obj={obj_gd:.8f}")
        # Try with different tau range
        remaining = deadline - time.time()
        if remaining > 40:
            a_gd2 = nesterov_smoothed(best_seq, min(remaining*0.15, 80.0), tau_init=50.0, tau_end=1000.0)
            obj_gd2 = compute_obj(a_gd2)
            if obj_gd2 < best_score:
                best_score = obj_gd2
                best_seq = a_gd2.copy()
                print(f"  After GD2 (high tau): obj={best_score:.8f} **")

    # Phase 4: Simulated Annealing
    print(f"\n=== Phase 4: Simulated Annealing ===")
    remaining = deadline - time.time()
    sa_time = min(remaining * 0.1, 100.0)
    if sa_time > 15:
        a_sa = simulate_annealing(best_seq, sa_time, rng)
        obj_sa = compute_obj(a_sa)
        if obj_sa < best_score:
            best_score = obj_sa
            best_seq = a_sa.copy()
            print(f"  After SA: obj={best_score:.8f} **")
        else:
            print(f"  SA: obj={obj_sa:.8f}")

    # Phase 5: Final intensive polish
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.time()

    # LP round
    if remaining > 50:
        alloc = min(remaining * 0.3, 250.0)
        a_lp = lp_polish(best_seq, alloc, rng, use_lsearch=True, verbose=False, deadline=deadline)

            best_seq = a_lp.copy()

        n_opts = [x for x in n_opts if 400 <= x <= 1600 and abs(x - n_curr) >= 30]
        for n_try in n_opts[:3]:
            remaining = deadline - time.time()
            if remaining < 40:
                break
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            alloc = min(remaining * 0.15, 50.0)
            a_try = lp_polish(a_try, alloc, rng, use_lsearch=False)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Multi-n n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  Multi-n n={n_try}: obj={obj_try:.8f}")

    # Final GD
    remaining = deadline - time.time()

        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Very last LP
    remaining = deadline - time.time()
    if remaining > 20:
        alloc = min(remaining * 0.5, 60.0)
        a_last = lp_polish(best_seq, alloc, rng, use_lsearch=True)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # Validate
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.time() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
