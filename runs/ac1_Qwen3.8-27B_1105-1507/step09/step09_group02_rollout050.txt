Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: LP polishing with golden-section blending, dual-aware perturbation,
multi-length exploration, and coordinate descent final polish.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, get_dual=False):
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    if get_dual:
                        # Try to get dual values
                        dual = None
                        for con in constraints:
                            if con.dual_value is not None:
                                dual = con.dual_value
                                break
                        return g_val, float(M.value), dual
                    return g_val, float(M.value)
        except Exception:
            continue
    return None, None

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP polishing with golden-section blending, multi-peak LP variants,
coordinate descent, and multi-length exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_full(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def build_conv_matrix_subset(a, indices):
    n = len(a)
    A = np.zeros((len(indices), n))
    for idx_col, k in enumerate(indices):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        j_start = k - i_end
        j_end = k - i_start
        A[idx_col, i_start:i_end + 1] = a[j_start:j_end + 1]
    return A


def solve_lp_full(a):
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_full(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                sv = float(np.sum(gv))
                if sv > 1e-10:
                    return gv / sv
        except Exception:
            continue
    return None


def solve_lp_topk(a, K):
    """LP that only constrains the top-K convolution peaks."""
    n = len(a)
    a_norm = a / np.sum(a)
    conv = compute_conv(a_norm)
    top_idx = np.argsort(conv)[::-1][:K]
    A = build_conv_matrix_subset(a_norm, top_idx)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                sv = float(np.sum(gv))
                if sv > 1e-10:
                    return gv / sv
        except Exception:
            continue
    return None


def golden_blend(a, g, iters=30):
    """Golden section search for optimal blend parameter."""
    a = np.maximum(a, 0).astype(np.float64)
    g = np.maximum(g, 0).astype(np.float64)
    sa = a.sum()
    sg = g.sum()
    if sa < 1e-10 or sg < 1e-10:
        return a.copy()
    a = a / sa
    g = g / sg
    
    obj0 = compute_obj(a)
    obj1 = compute_obj(g)
    
    if obj1 <= obj0:
        return g.copy()
    if obj0 == np.inf:
        return g.copy()
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    
    t1 = lo + (1.0 - gr) * (hi - lo)
    t2 = lo + gr * (hi - lo)
    
    b1 = (1 - t1) * a + t1 * g
    b1 = np.maximum(b1, 0)
    if b1.sum() < 1e-10:
        return a.copy()
    b1 = b1 / b1.sum()
    o1 = compute_obj(b1)
    
    b2 = (1 - t2) * a + t2 * g
    b2 = np.maximum(b2, 0)
    if b2.sum() < 1e-10:
        return a.copy()
    b2 = b2 / b2.sum()
    o2 = compute_obj(b2)
    
    best_t = 0.0
    best_obj = obj0
    best_b = a.copy()
    
    if o1 < best_obj:
        best_t, best_obj, best_b = t1, o1, b1.copy()
    if o2 < best_obj:
        best_t, best_obj, best_b = t2, o2, b2.copy()
    
    for _ in range(iters):
        if o1 < o2:
            hi = t2
            t2 = t1
            b2, o2 = b1, o1
            t1 = lo + (1.0 - gr) * (hi - lo)
            b1 = (1 - t1) * a + t1 * g
            b1 = np.maximum(b1, 0)
            b1 = b1 / b1.sum()
            o1 = compute_obj(b1)
        else:
            lo = t1
            t1 = t2
            b1, o1 = b2, o2
            t2 = lo + gr * (hi - lo)
            b2 = (1 - t2) * a + t2 * g
            b2 = np.maximum(b2, 0)
            b2 = b2 / b2.sum()
            o2 = compute_obj(b2)
        
        if o1 < best_obj:
            best_t, best_obj, best_b = t1, o1, b1.copy()
        if o2 < best_obj:
            best_t, best_obj, best_b = t2, o2, b2.copy()
    
    return best_b


def lp_polish(a, time_budget, verbose=False, use_topk=False, K=50):
    """LP polishing with golden section blending."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - start < time_budget - 5:
        if use_topk:
            g_lp = solve_lp_topk(a, K)
        else:
            g_lp = solve_lp_full(a)
        
        if g_lp is None:
            break
        
        blended = golden_blend(a, g_lp, iters=30)
        obj_b = compute_obj(blended)
        
        if obj_b < best_obj - 1e-14:
            best_obj = obj_b
            best_a = blended.copy()
            a = best_a.copy()
            stall = 0
            it += 1
            if verbose and it % 5 == 0:
                elapsed = time.monotonic() - start
                print(f"      LP it{it}: obj={best_obj:.8f}, t={elapsed:.0f}s/{time_budget:.0f}s")
        else:
            stall += 1
            if stall >= 8:
                scale = 0.005 * min(stall, 20)
                pert = np.exp(rng.normal(0, scale, n))
                a = best_a * pert
                a = np.maximum(a, 0)
                a = a / a.sum()
                if stall >= 25:
                    if verbose:
                        print(f"      LP: break after {stall} stalls, obj={best_obj:.8f}")
                    break
    
    return best_a


def coord_descent(a, time_budget, verbose=False):
    """Coordinate descent optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for rnd in range(100):
        if time.monotonic() - start > time_budget - 5:
            break
        
        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))
        improved = False
        
        i_start = max(0, peak_idx - (n - 1))
        i_end = min(n - 1, peak_idx)
        
        # Try reducing each contributor to the peak
        for i in range(i_start, i_end + 1):
            if time.monotonic() - start > time_budget - 5:
                break
            j = peak_idx - i
            for frac in [0.005, 0.01, 0.02, 0.05]:
                trial = a.copy()
                trial[i] *= (1.0 - frac)
                s_t = trial.sum()
                if s_t < 1e-10:
                    continue
                trial = trial / s_t
                obj_t = compute_obj(trial)
                if obj_t < best_obj - 1e-14:
                    best_obj = obj_t
                    best_a = trial.copy()
                    a = trial.copy()
                    improved = True
                    break
        
        if not improved:
            if verbose:
                print(f"      CD rnd {rnd}: no improvement, obj={best_obj:.8f}")
            break
    
    return best_a


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def diverse_init(n, rng, mode):
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * 3 * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + 5.0 * np.exp(-((x - 0.3) ** 2) / 0.01)) + 0.5
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 4:
        # Anti-peak: high at boundaries, low in middle (inverse triangle)
        x = np.linspace(0, 1, n)
        a = 1.0 - 0.5 * np.abs(2 * x - 1)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 5:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
        a = np.maximum(a, 0.001)
        return a / a.sum()
    else:
        return np.ones(n) / n
