"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based best-response iteration with FFT gradient polishing,
frequency-domain diversification, and multi-length exploration.
Key improvements over previous: faster LP via structured matrices,
adaptive tau annealing, dual-guided updates, and block-cyclic exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # conv(a,g)[k] = sum_i a[i]*g[k-i], so for fixed g index j:
        # g[j] contributes to conv[k] when k-j is a valid index of a
        # i.e., conv[k][j] = a[k-j] if 0 <= k-j < n
        start = j
        end = min(2 * n - 1, n + j)
        length = end - start
        if length > 0:
            A[start:end, j] = a[:length]
    return A


def solve_lp_minpeak(a, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (optimal_g, optimal_M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None


def solve_lp_multiobjective(a, verbose=False):
    """
    Solve a modified LP that also penalizes L2 norm:
    min M + lambda * ||conv(a,g)||_2^2  (approximated by min M + lambda * sum(conv^2))
    Since sum(conv^2) is quadratic in g, we approximate with min M + lambda * trace(g^T A^T A g)
    For simplicity, use a second LP: min sum of squared peaks approximated by top-k.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    # Add L2 penalty on g to encourage spread
    prob = cp.Problem(cp.Minimize(M + 0.01 * cp.sum_squares(g)), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS]:
        try:

                s = float(np.sum(g_val))
                if s > 1e-10:

        except Exception:

def smoothed_gradient_fft(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT.
    Returns gradient of the smoothed objective (before normalization).
    """
    n = len(a)
    conv = compute_conv(a)

    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    # Gradient of smoothed max wrt a:
    # d/da_j [sum_k p_k * conv[k]] = sum_k p_k * d conv[k] / da_j
    # conv[k] = sum_i a[i]*a[k-i], so d conv[k]/da_j = a[k-j] (if valid) + a[j-k] (if valid)
    # This is 2 * cross_correlation(p, a) shifted
    N = 1
    while N < 3 * n:
        N <<= 1

    P = fft(p, N)
    A = fft(a, N)
    # The gradient is 2 * ifft(P * conj(A))[:n] (for the quadratic form)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]

    return g, conv

    lr = base_lr
    v = np.zeros(n)

    best_it = 0
    grad_norms = []

        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(time_budget, 1e-10))

        # Cosine anneal tau
        if frac < 0.05:
            tau = tau_init
        else:
            cos_val = 0.5 * (1 + np.cos(np.pi * (frac - 0.05) / 0.95))
            tau = tau_init + (tau_end - tau_init) * (1 - cos_val)

        if it % 5000 == 0 and verbose:
            print(f"    ASD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute gradient at y

        g_norm = float(np.linalg.norm(g))
        grad_norms.append(g_norm)
        if len(grad_norms) > 300:
            grad_norms = grad_norms[-300:]

        # Adaptive learning rate
        avg_gnorm = float(np.mean(grad_norms))
        if avg_gnorm > 1e-15:
            adaptive_lr = 0.8 / (avg_gnorm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.2 * base_lr), 4.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adaptive_lr

        # Update

        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-9, lr * 0.9998)

        # Reset if stuck
        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it

        best_t_obj = best_obj
        best_t_val = last_t
        best_a_local = best_a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        iter_count += 1
        if best_t_obj < best_obj - 1e-15:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall = 0
            if verbose and iter_count % 3 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t_val:.3f}, M={M_lp:.6f}")
        else:
            stall += 1
            if stall > 3:
                # Perturb to escape
                noise = rng.normal(0, 0.02 * stall, n)
                a = a * (1 + noise)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                stall = 0
                if iter_count > 30:
                    break

    return best_a


def fourier_perturbation(a, rng, strength=0.05, n_modes=4):
    """Add structured Fourier mode perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, n_modes + 1):
        freq = 2 * np.pi * k / n
        amp = strength * rng.uniform(0.3, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def spectral_init(n, rng):
    """Initialize using random spectral phases with low-pass amplitude."""
    size = 1
    while size < 2 * n:
        size <<= 1
    freqs = np.fft.rfftfreq(size)
    mag = np.exp(-freqs * n * 0.3)
    phases = rng.uniform(0, 2 * np.pi, size=size // 2 + 1)
    F = mag * np.exp(1j * phases)
    F[0] = abs(F[0])
    a = np.fft.irfft(F, size)[:n]
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        # Random Fourier
        freqs = rng.integers(1, 8, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Cosine bump
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        # Two Gaussians
        c1 = rng.integers(0, n // 3)
        c2 = rng.integers(2 * n // 3, n)
        w = max(1, n // 12)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 8:
        # Triangular
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 4 * np.abs(x - 0.5), 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def block_cyclic_init(n, rng):
    """Create a block-cyclic pattern: repeat a short pattern."""
    block_len = rng.integers(2, max(3, n // 50))
    if n % block_len != 0:
        block_len = max(2, n // rng.integers(n // block_len, n // block_len + 1))
        if n % block_len != 0:
            return diverse_init(n, rng, mode=1)
    block = rng.uniform(0.1, 1.0, block_len)
    a = np.tile(block, n // block_len)
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP iteration + FFT gradient + multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        n_current = 800
        best_seq = np.ones(n_current) / n_current
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Quick FFT GD warmup ===
    print(f"\n=== Phase 1: FFT Gradient Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.1, 100.0)
    if p1_budget > 20:
        a = adaptive_smoothed_descent(best_seq, time_budget=p1_budget,
                                      tau_init=5.0, tau_end=300.0, verbose=True)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj:.8f} (no improvement)")

    # === Phase 2: LP Iteration (main workhorse) ===
    print(f"\n=== Phase 2: LP Best-Response Iteration ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 450.0)
    if p2_budget > 30:
        for rnd in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(p2_budget / (8 - rnd), remaining * 0.5)
            if alloc < 10:
                break
            a = lp_iterate(best_seq, time_budget=alloc, verbose=(rnd == 0),
                           seed=1000 + rnd * 7)
            obj = compute_obj(a)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj:.8f}")
                if rnd < 7:
                    strength = 0.01 + 0.005 * rnd
                    best_seq = fourier_perturbation(best_seq, rng, strength=strength)

    # === Phase 3: Intensive GD after LP ===
    print(f"\n=== Phase 3: Post-LP Gradient Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    if p3_budget > 30:
        a = adaptive_smoothed_descent(best_seq, time_budget=p3_budget,
                                      tau_init=20.0, tau_end=800.0, verbose=True)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj:.8f}")

    # === Phase 4: More LP rounds ===
    print(f"\n=== Phase 4: More LP Rounds ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.2, 200.0)
    if p4_budget > 20:
        for rnd in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(p4_budget / (4 - rnd), remaining * 0.6)
            if alloc < 10:
                break
            a = lp_iterate(best_seq, time_budget=alloc, verbose=False,
                           seed=5000 + rnd * 13)
            obj = compute_obj(a)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  LP2 round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP2 round {rnd}: obj={obj:.8f}")
                if rnd < 3:
                    best_seq = fourier_perturbation(best_seq, rng, strength=0.008)

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        n_best - 100, n_best - 50, n_best + 50, n_best + 100,
        n_best + 200, n_best + 400, n_best - 200,
        500, 600, 700, 800, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_best) > 20]
    n_options = n_options[:5]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick GD
        alloc1 = min(remaining * 0.25, 40.0)
        if alloc1 > 15:
            a_try = adaptive_smoothed_descent(a_try, time_budget=alloc1,
                                              tau_init=10.0, tau_end=300.0)

        # Short LP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.2, 30.0)
        if alloc2 > 10:
            a_try = lp_iterate(a_try, time_budget=alloc2, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        if r == 0:
            a_r = spectral_init(n_current, rng)
        elif r == 1:
            a_r = block_cyclic_init(n_current, rng)
        else:
            a_r = diverse_init(n_current, rng, mode=2 + r)
            blend = 0.3 + 0.1 * r
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.2, 80.0)
        if alloc > 25:
            half = alloc * 0.5
            a_r = adaptive_smoothed_descent(a_r, time_budget=half,
                                            tau_init=10.0, tau_end=400.0)
            remaining = deadline - time.monotonic()
            lp_alloc = min(remaining * 0.15, 35.0)
            if lp_alloc > 10:
                a_r = lp_iterate(a_r, time_budget=lp_alloc, seed=7000 + r)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau GD
    if remaining > 30:
        alloc = min(remaining * 0.3, 80.0)
        a = adaptive_smoothed_descent(best_seq, time_budget=alloc,
                                      tau_init=50.0, tau_end=1000.0, verbose=True)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Final LP passes
    remaining = deadline - time.monotonic()
    for lp_rnd in range(3):
        if remaining <= 15:
            break
        alloc = min(remaining * 0.35, 60.0)
        a = lp_iterate(best_seq, time_budget=alloc, verbose=False, seed=9000 + lp_rnd * 100)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  Final LP {lp_rnd}: obj={best_score:.8f} **")
        else:
            if lp_rnd < 2:
                best_seq = fourier_perturbation(best_seq, rng, strength=0.005)

    # One more GD if time
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.4, 50.0)
        a = adaptive_smoothed_descent(best_seq, time_budget=alloc,
                                      tau_init=100.0, tau_end=1500.0)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  Ultra-final GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
