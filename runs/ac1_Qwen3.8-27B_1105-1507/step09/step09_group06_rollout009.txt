"""
Optimizes step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: LP-based optimization with dual-guided perturbations,
multi-scale exploration, and KKT-based refinement.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_cols(a):
    """
    Return matrix A (2n-1 x n) where A[k,i] = a[k-i] if 0<=k-i<n else 0.
    So that A @ g = convolve(a, g).
    """
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        # A[k,i] = a[k-i] for k-i in [0, n-1], i.e., k in [i, i+n-1]
        lo = i
        hi = min(2 * n - 1, i + n - 1)
        A[lo:hi+1, i] = a[:hi+1-lo]
    return A


def solve_lp(a, verbose=False):
    """
    Solve LP: min M s.t. conv(a, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt, lambda_opt) or (None, None, None).
    lambda_opt is the dual variable (distribution over lags).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_cols(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    lambda_var = cp.Variable(2 * n - 1, nonneg=True)

    # Primal
    constraints_p = [A @ g <= M, cp.sum(g) == 1]
    prob_p = cp.Problem(cp.Minimize(M), constraints_p)

    g_val = None
    M_val = None
    lam_val = None

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob_p.solve(solver=solver)
            if g.value is not None and prob_p.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                break
        except Exception:
            continue

    if g_val is None:
        return None, None, None

    # Try to get dual variables (may not always be available)
    # The dual constraint A^T @ lambda <= M * 1, sum(lambda) = 1, lambda >= 0
    # We can try to extract from the solver
    try:
        for con in prob_p.constraints:
            if con.value is not None:
                # Clarabel provides dual values
                dual_vals = con.value
                if dual_vals is not None and len(dual_vals) == 2 * n - 1:
                    lam_val = np.maximum(dual_vals, 0)
                    lam_sum = float(np.sum(lam_val))
                    if lam_sum > 1e-10:
                        lam_val = lam_val / lam_sum
                    break
    except Exception:
        pass

    # If we couldn't get dual from solver, approximate it
    if lam_val is None:
        # The dual lambda is concentrated on the active constraints
        # Active constraints are those where (A@g)[k] ≈ M
        conv_ag = A @ g_val
        M_actual = float(np.max(conv_ag))
        # lambda proportional to "activity"
        threshold = M_actual * 0.99
        lam_val = (conv_ag >= threshold).astype(float)
        if lam_val.sum() > 0:
            lam_val = lam_val / lam_val.sum()
        else:
            lam_val = np.ones(2 * n - 1) / (2 * n - 1)

    return g_val, M_val, lam_val

    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    consecutive_fails = 0
    last_t = 0.3
    iter_count = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp, lam_lp = solve_lp(a)
        if g_lp is None:
            break

        # Line search for best blend
        ts = np.concatenate([
            np.linspace(0.01, 1.0, 20),
            [max(0.01, last_t * 0.5), min(1.0, last_t * 1.5),
             max(0.01, last_t * 0.7), min(1.0, last_t * 1.3)]
        ])
        ts = np.unique(np.clip(ts, 0.01, 1.0))

        best_t_obj = best_obj
        best_t_val = last_t
        best_a_local = best_a.copy()

        for t in ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        iter_count += 1

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a.copy()
            last_t = best_t_val
            consecutive_fails = 0
            if verbose and iter_count % 10 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t_val:.3f}, M={M_lp:.6f}, t={elapsed:.0f}s")
        else:
            consecutive_fails += 1
            # Dual-guided perturbation
            if lam_lp is not None:
                # Identify active lags
                active = lam_lp > 1e-4
                n_active = active.sum()

                # Create perturbation that "pushes" against active lags
                # and "pulls" towards inactive ones
                A = build_conv_matrix_cols(a)
                # Gradient of max at active lags
                # For each active lag k, the "direction" to reduce it is -A[k,:]
                # But we want to stay on the simplex
                push = np.zeros(n)
                for k in range(2 * n - 1):
                    if active[k]:
                        push -= A[k, :] * lam_lp[k]

                # Also add a small random component for exploration
                noise = rng.normal(0, 0.01, n)
                a_new = a + 0.05 * (push / (np.abs(push).sum() + 1e-10)) + noise
                a_new = np.maximum(a_new, 0)
                s_new = float(np.sum(a_new))
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    # Blend with best to avoid too-large jumps
                    blend = min(0.5, 0.1 * consecutive_fails)
                    a = (1 - blend) * a_new + blend * best_a
                    a = np.maximum(a, 0)
                    a = a / a.sum()

            if consecutive_fails > 50:
                # Large restart: random perturbation
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                consecutive_fails = 0

    return best_a


def gradient_descent_max(a, time_budget=100.0, tau=100.0, verbose=False, seed=0):
    """
    Subgradient descent on the (nearly) exact max function.
    Uses the smoothed approximation with high tau for quasi-exact gradients.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    beta = 0.9
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        # Compute smoothed gradient
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        # Softmax weights
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        # Gradient: d/d a[i] of sum_k p_k * (a*a)[k]
        # = sum_k p_k * (a[k-i] + a[i-k]) (for valid indices)
        # Using FFT: let P = FFT(p), A = FFT(a), then grad = 2 * IFFT(P * conj(A))[:n]
        N = 1
        while N < 3 * n:
            N <<= 1
        Pf = fft(p, N)
        Af = fft(a, N)
        g_full = ifft(Pf * np.conj(Af)).real
        grad = 2.0 * g_full[:n]

        # Nesterov
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Recompute gradient at y
        conv_y = compute_conv(y)
        max_y = float(np.max(conv_y))
        exp_y = np.exp(tau * (conv_y - max_y))
        p_y = exp_y / float(np.sum(exp_y))
        Pf2 = fft(p_y, N)
        Af2 = fft(y, N)
        g_full2 = ifft(Pf2 * np.conj(Af2)).real
        grad = 2.0 * g_full2[:n]

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr *= 0.9995

        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)
            best_it = it

    return best_a


def structured_perturbation(a, seed=0):
    """
    Create a perturbation based on structural properties:
    - Alternating perturbation (high-low pattern)
    - Sinusoidal perturbation
    - Random smooth perturbation
    """
    n = len(a)
    rng = np.random.default_rng(seed)
    mode = rng.integers(0, 4)

    if mode == 0:
        # Alternating
        pattern = np.ones(n)
        pattern[::2] = 1 + rng.uniform(0.02, 0.08)
        pattern[1::2] = 1 - rng.uniform(0.02, 0.08)
    elif mode == 1:
        # Sinusoidal
        freq = rng.uniform(0.5, 5)
        amplitude = rng.uniform(0.02, 0.08)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n) / n
        pattern = 1 + amplitude * np.sin(2 * np.pi * freq * x + phase)
    elif mode == 2:
        # Random smooth (low-pass filtered noise)
        noise = rng.normal(0, 1, n)
        # Simple boxcar filter
        k = max(2, n // 20)
        kernel = np.ones(k) / k
        noise = np.convolve(noise, kernel, mode='same')
        amplitude = rng.uniform(0.02, 0.06)
        pattern = 1 + amplitude * noise / (np.std(noise) + 1e-10)
    else:
        # Two-hump: emphasize one half
        split = rng.integers(n // 3, 2 * n // 3)
        boost = rng.uniform(0.03, 0.1)
        pattern = np.ones(n)
        pattern[:split] *= (1 + boost)
        pattern[split:] *= (1 - boost)

    a_new = a * pattern
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def multi_length_explore(base_seq, n_options, time_per_n, seed=0):
    """
    Try optimizing at different lengths by resampling and re-optimizing.
    """
    best = base_seq.copy()
    best_obj = compute_obj(base_seq)
    start = time.monotonic()

    for n_try in n_options:
        if time.monotonic() - start > time_per_n * len(n_options):
            break

        # Resample
        n_orig = len(base_seq)
        if n_orig == 1:
            a_try = np.full(n_try, base_seq[0])
        else:
            x_old = np.linspace(0, 1, n_orig)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, base_seq)
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s

        # Optimize at this length
        alloc = min(time_per_n, 120.0)
        a_opt = lp_optimize(a_try, time_budget=alloc, verbose=False, seed=seed + n_try)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_obj:
            best_obj = obj_opt
            best = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    return best, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    
    Strategy:
    1. Start from parent sequence
    2. LP-based optimization with dual-guided perturbations
    3. Gradient descent with high-tau smoothing for refinement
    4. Structured perturbations to escape local minima
    5. Multi-length exploration
    6. Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Initial LP optimization ===
    print(f"\n=== Phase 1: Initial LP (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 30:
        a_opt = lp_optimize(best_seq, time_budget=p1_budget, verbose=True, seed=seed)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase1 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 LP: obj={obj_opt:.8f}")

    # === Phase 2: Gradient descent refinement ===
    print(f"\n=== Phase 2: GD Refinement ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 150.0)
    if p2_budget > 20:
        a_gd = gradient_descent_max(best_seq, time_budget=p2_budget, tau=200.0,
                                     verbose=False, seed=seed + 100)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase2 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 GD: obj={obj_gd:.8f}")

    # === Phase 3: LP + perturbation cycles ===
    print(f"\n=== Phase 3: LP + Perturbation Cycles ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.3, 400.0)
    if p3_budget > 30:
        # Multiple rounds of: LP optimize, then perturb, then LP optimize
        n_rounds = 3
        per_round = p3_budget / n_rounds
        for r in range(n_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            # LP optimize
            alloc = per_round * 0.7
            a_opt = lp_optimize(best_seq, time_budget=alloc, verbose=True,
                                 seed=seed + 200 + r * 10)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Phase3 R{r} LP: obj={best_score:.8f} **")

            # Structured perturbation
            a_pert = structured_perturbation(best_seq, seed=seed + 500 + r)
            blend = 0.3 + 0.1 * r
            a_pert = (1 - blend) * a_pert + blend * best_seq
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()

            # Quick LP on perturbed
            remaining = deadline - time.monotonic()
            alloc2 = min(per_round * 0.3, 40.0)
            if remaining > alloc2:
                a_opt2 = lp_optimize(a_pert, time_budget=alloc2, verbose=False,
                                      seed=seed + 300 + r * 10)
                obj_opt2 = compute_obj(a_opt2)
                if obj_opt2 < best_score:
                    best_score = obj_opt2
                    best_seq = a_opt2.copy()
                    print(f"  Phase3 R{r} LP (perturbed): obj={best_score:.8f} **")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Try a few nearby and larger lengths
    n_opts = []
    for delta in [-100, -50, 50, 100, 200, 300, 400]:
        n_try = n_best + delta
        if 400 <= n_try <= 2500:
            n_opts.append(n_try)
    # Also try some standard sizes
    for n_std in [1000, 1200, 1500, 2000]:
        if n_std not in n_opts and 400 <= n_std <= 2500:
            n_opts.append(n_std)
    n_opts = sorted(set(n_opts))[:5]  # Limit to 5

    if n_opts and remaining > 60:
        multi_time = remaining * 0.2
        best_seq, best_score2 = multi_length_explore(best_seq, n_opts, multi_time / len(n_opts),
                                                      seed=seed + 1000)
        if best_score2 < best_score:
            best_score = best_score2
            print(f"  Phase4 multi-length: obj={best_score:.8f} **")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    n_restarts = 4
    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        # Different initialization strategies
        if r == 0:
            # Random
            a_r = rng.uniform(0, 1, n_final)
        elif r == 1:
            # Two-hump
            a_r = np.zeros(n_final)
            c1 = n_final // 4
            c2 = 3 * n_final // 4
            w = n_final // 10
            a_r += np.exp(-((np.arange(n_final) - c1) ** 2) / (2 * w ** 2))
            a_r += np.exp(-((np.arange(n_final) - c2) ** 2) / (2 * w ** 2))
        elif r == 2:
            # Alternating emphasis
            a_r = np.ones(n_final)
            a_r[::2] *= 1.3
            a_r[1::2] *= 0.7
        else:
            # Sinusoidal
            x = np.linspace(0, 2 * np.pi, n_final)
            a_r = 1 + 0.3 * np.sin(x) + 0.1 * np.sin(3 * x)

        a_r = np.maximum(a_r, 0)
        s = float(np.sum(a_r))
        if s < 1e-10:
            continue
        a_r = a_r / s

        # Blend with best
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        # Optimize
        alloc = min(remaining * 0.2, 80.0)
        if alloc > 20:
            a_opt = lp_optimize(a_r, time_budget=alloc, verbose=False, seed=seed + 2000 + r)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_opt:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 40:
        alloc = min(remaining * 0.5, 150.0)
        a_final = lp_optimize(best_seq, time_budget=alloc, verbose=True, seed=seed + 3000)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # GD polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 80.0)
        a_gd = gradient_descent_max(best_seq, time_budget=alloc, tau=500.0,
                                     verbose=False, seed=seed + 4000)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Final GD: obj={best_score:.8f} **")
        else:
            print(f"  Final GD: obj={obj_gd:.8f}")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.5, 60.0)
        a_final2 = lp_optimize(best_seq, time_budget=alloc, verbose=False, seed=seed + 5000)
        obj_final2 = compute_obj(a_final2)
