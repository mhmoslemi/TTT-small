import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, M_target=None):
    """Solve LP: min M s.t. conv(a_norm,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue
    return None, None


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv

    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:

        else:
            y = y / s_y
        g, conv = smoothed_gradient_fft(y, tau)
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)
        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            best_it = it
    return best_a


def lp_polish_anderson(a, time_budget=300.0, verbose=False, seed=42, anderson_m=3):
    """
    LP-based polishing with Anderson acceleration for the fixed-point iteration.
    Maintains a memory of past iterates for accelerated convergence.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    it_count = 0
    # Anderson memory
    history_x = []
    history_r = []
    t_history = []
    last_t = 0.4

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        # Solve LP
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        it_count += 1
        # Anderson acceleration: compute candidate using memory
        candidate = g_lp.copy()
        if len(history_x) >= anderson_m:
            # Build Anderson combination
            X = np.array(history_x[-anderson_m:])  # (m, n)
            R = X - np.array(history_r[-anderson_m:])  # residuals
            # Find weights w minimizing ||R @ w||^2 s.t. sum(w)=1
            m = R.shape[0]
            G = R.T @ R  # (m, m)
            ones = np.ones(m)
            try:
                # Solve KKT system
                KKT = np.zeros((m + 1, m + 1))
                KKT[:m, :m] = G
                KKT[:m, m] = ones
                KKT[m, :m] = ones
                rhs = np.zeros(m + 1)
                w = np.linalg.solve(KKT, rhs)
                w = w[:m]
                w = np.maximum(w, 0)
                ws = w.sum()
                if ws > 1e-10:
                    w = w / ws
                    candidate = np.sum(X * w[:, None], axis=0)
            except Exception:
                pass
        # Try candidate and also simple blends
        best_blend_obj = best_obj
        best_blend_a = None
        for t in [1.0, 0.6, 0.4, 0.2, 0.1]:
            a_try = (1.0 - t) * best_a + t * candidate
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_blend_a = a_try.copy()
                last_t = t
        if best_blend_a is not None and best_blend_obj < best_obj - 1e-14:
            best_obj = best_blend_obj
            best_a = best_blend_a.copy()
            a = best_a.copy()
            no_improve = 0
            if verbose and it_count % 10 == 0:
                print(f"    LP-And it{it_count}: obj={best_obj:.8f}, t={last_t:.2f}, el={elapsed:.0f}s")
        else:
            no_improve += 1
            if no_improve == 3:
                # Try Anderson extrapolation
                if len(history_x) >= 2:
                    X = np.array(history_x[-min(4, len(history_x)):])
                    candidate_ext = X[-1] + 0.5 * (X[-1] - X[-2])
                    a_try = (1.0 - 0.5) * best_a + 0.5 * np.maximum(candidate_ext, 0)
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try > 1e-10:
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = best_a.copy()
                            no_improve = 0
            elif no_improve >= 5:
                # Perturb and continue
                scale = 0.005 + 0.003 * min(no_improve, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                if it_count > 80:
                    break
                # Reset Anderson memory
                history_x = []
                history_r = []
        # Update Anderson memory
        history_x.append(a.copy())
        r = a - g_lp
        history_r.append(r.copy())
        if len(history_x) > anderson_m + 2:
            history_x = history_x[-(anderson_m + 2):]
            history_r = history_r[-(anderson_m + 2):]
    return best_a


def peak_splitting_search(a, time_budget=80.0, verbose=False, seed=42):
    """
    Direct peak-targeted local search. Identifies peak bins and tries
    targeted mass redistribution to reduce them.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_k = int(np.argmax(conv))
        max_val = conv[max_k]
        # Find top contributing pairs to the peak bin
        contributions = np.zeros(n)
        for i in range(n):
            j = max_k - i
            if 0 <= j < n:
                contributions[i] = a[i] * a[j]
        top_idx = np.argsort(contributions)[::-1][:min(20, n)]
        # For each top contributor, try moving mass to a random position
        best_move = None
        best_move_obj = best_obj
        for src in top_idx[:10]:
            for _ in range(5):
                dst = rng.integers(0, n)
                if dst == src:
                    continue
                delta = a[src] * 0.1
                if delta < 1e-12:
                    continue
                a_try = a.copy()
                a_try[src] -= delta
                a_try[dst] += delta
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_move_obj:
                    best_move_obj = obj_try
                    best_move = a_try.copy()
        if best_move is not None:
            best_obj = best_move_obj
            best_a = best_move.copy()
            a = best_a.copy()
            stall = 0
            if verbose and it % 50 == 0:
                print(f"    PeakSplit it{it}: obj={best_obj:.8f}, el={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall > 30:
                # Perturb
                scale = 0.005 * min(stall / 10.0, 3.0)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if it > 2000:
                    break
        it += 1
    return best_a


def subgradient_refine(a, time_budget=60.0, K=8, verbose=False):
    """Subgradient method with top-K peaks, momentum, and adaptive step."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 2.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        weights = np.exp((peak_vals - max_pv) * 5.0 / max_pv)
        weights /= weights.sum()
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]
        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-10:
                    break
            else:
                lr *= 0.997
        it += 1
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a / a.sum()
    elif mode == 4:
        # Hamming-like window
        x = np.linspace(0, 1, n)
        a = 0.54 + 0.46 * np.cos(2 * np.pi * x)
        a = a / a.sum()
    elif mode == 5:
        # Flat-top window
        x = np.linspace(-1, 1, n)
        a = 0.21557895 - 0.41663158 * np.cos(np.pi * x) + 0.277263158 * np.cos(2 * np.pi * x) \
            - 0.083578947 * np.cos(3 * np.pi * x) + 0.006947368 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0)
        if a.sum() < 1e-10:
            a = np.ones(n)
        a = a / a.sum()
    elif mode == 6:
        # Bell at center
        x = np.arange(n)
        a = np.exp(-((x - n / 2.0) ** 2) / (2 * (n / 3.0) ** 2))
        a = a / a.sum()
    elif mode == 7:
        # Two bells
        x = np.arange(n)
        a = np.exp(-((x - n * 0.25) ** 2) / (2 * (n / 10.0) ** 2)) + \
            np.exp(-((x - n * 0.75) ** 2) / (2 * (n / 10.0) ** 2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a
