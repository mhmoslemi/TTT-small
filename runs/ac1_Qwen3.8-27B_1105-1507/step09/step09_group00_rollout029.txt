from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def solve_conv_lp(a):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0. Returns (g_opt, M_opt)."""
    n = len(a)
    # Build convolution matrix: (a*g)[k] = sum_j A[k,j] * g_j
    # A[k,j] = a[k-j] if 0 <= k-j <= n-1, else 0
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # For column j, A[k,j] = a[k-j] for k = j, j+1, ..., j+n-1
        for i in range(n):
            k = j + i
            A[k, j] = a[i]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except:
            continue
    return None, None

def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of (a*a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    N = 1
    while N < 3*n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based alternating minimization as the primary workhorse, combined with
FFT-accelerated smoothed gradient polishing and adaptive blending schedules.
The LP at each step finds the optimal "best response" sequence to pair with the
current iterate, and adaptive convex combinations drive convergence to the
minimum peak convolution.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_conv_lp(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of (a*a)_k using FFT.
    Returns gradient vector of shape (n,).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g


def nesterov_gd_polish(a, time_budget, tau_init=30.0, tau_max=2000.0):
    """
    Nesterov accelerated gradient descent on the smoothed objective.
    Projects onto the simplex {x >= 0, sum(x) = 1} after each step.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    tau = tau_init
    it = 0

    while time.monotonic() - start < time_budget:
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        # Adaptive learning rate
        if g_norm > 1e-15:
            adaptive_lr = 0.5 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.1 * base_lr), 3.0 * base_lr)
            lr = 0.8 * lr + 0.2 * adaptive_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr *= 0.9998
        tau = min(tau * 1.02, tau_max)
        it += 1

        # Reset if stuck for too long
        if it % 5000 == 0 and compute_obj(a) > best_obj + 1e-6:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5

    return best_a


def fourier_perturb(a, rng, strength=0.03):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 5
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lp_alternate(a, time_budget, seed=42, verbose=True):
    """
    Main LP-based alternating minimization.
    At each step, solve the LP to find the best response g to current a,
    then blend toward g with an adaptive factor.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    t_blend = 0.5
    stall_count = 0
    lp_times = []

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        t0 = time.monotonic()
        g_lp, M_lp = solve_conv_lp(a)
        lp_sol_time = time.monotonic() - t0
        lp_times.append(lp_sol_time)

        if g_lp is None:
            break

        iter_count += 1

        # Try multiple blending factors (adaptive set based on progress)
        progress = iter_count / max(1, iter_count + 100)
        candidates_t = [
            t_blend,
            t_blend * 0.5,
            t_blend * 1.5 if t_blend * 1.5 <= 1.0 else 1.0,
            0.1, 0.2, 0.3, 0.5, 0.7, 1.0,
        ]
        # As we progress, favor smaller t
        if iter_count > 200:
            candidates_t.extend([0.05, 0.02, 0.01])

        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t_blend

        for t in candidates_t:
            t = min(max(t, 0.001), 1.0)
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            t_blend = max(0.01, best_local_t * 0.99)
            stall_count = 0
            if verbose and iter_count % 100 == 0:
                print(f"    LP alt {iter_count}: obj={best_obj:.10f}, t={t_blend:.4f}, "
                      f"M_lp={M_lp:.6f}, lp_time={lp_sol_time:.3f}s")
        else:
            stall_count += 1
            # Perturbation to escape
            if stall_count >= 3:
                strength = 0.01 + 0.005 * min(stall_count, 10)
                a = fourier_perturb(best_a, rng, strength=strength)
                t_blend = max(0.1, min(t_blend * 1.2, 0.8))
                stall_count = 0
            elif stall_count >= 1:
                # Small random nudge
                a = best_a + 0.005 * rng.standard_normal(n) * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()

    if verbose:
        print(f"    LP alternating: {iter_count} iterations, best_obj={best_obj:.10f}")

    return best_a


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
