import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def line_search_lp(a, g, max_iter=30):
    """Ternary search for best t in [0,1] along a -> g"""
    def f(t):
        a_t = (1 - t) * a + t * g
        a_t = np.maximum(a_t, 0)
        s = float(np.sum(a_t))
        if s < 1e-10:
            return np.inf
        a_t = a_t / s
        return compute_obj(a_t)

    f0 = f(0.0)
    f1 = f(1.0)
    fmid = f(0.5)
    
    if f1 < f0 and f1 < fmid:
        # Best is near t=1, search in [0.3, 1.0]
        lo, hi = 0.3, 1.0
    elif fmid < f0 and fmid < f1:
        lo, hi = 0.0, 1.0
    else:

    for _ in range(max_iter):

        f1 = f(m1)

        if f1 < f2:

        else:

        if hi - lo < 1e-6:

    a_best = np.maximum(a_best, 0)
    s = float(np.sum(a_best))
    if s < 1e-10:
        return a, f0
    a_best = a_best / s
    return a_best, f(t_best)

def smoothed_grad(a, tau):
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_v = np.exp(tau * (conv - max_c))
    p = exp_v / float(np.sum(exp_v))
    N = 1

        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real

def nesterov(a, time_budget, tau_start=20.0, tau_end=2000.0):

    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.5)
        
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:

        else:

        g = g - float(np.mean(g))

        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        lr *= 0.9999
    
    return best_a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses high-tau Nesterov smoothed gradient descent, LP polishing with golden-section
line search, dual-variable-guided perturbations, and multi-n exploration with
resampling. Key improvements over prior approach: proper line search in LP phase,
more aggressive tau schedule, dual-guided escaping, and tighter time management.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        length = min(n, 2 * n - 1 - j)
        A[j:j + length, j] = a[:length]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal, y_dual) or (None, None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c1 = A @ g <= M
    c2 = cp.sum(g) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])

    y_dual = None
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    try:
                        if c1.dual_value is not None and np.max(np.abs(c1.dual_value)) > 0:
                            y_dual = np.abs(c1.dual_value)
                    except Exception:
                        pass
                    return g_val, M_val, y_dual
        except Exception:
            continue
    return None, None, None


def golden_section_line_search(a, g, max_iter=25):
    """
    Find best t in [0,1] minimizing compute_obj((1-t)*a + t*g) via golden section.
    Returns (best_a, best_obj).
    """
    def f(t):
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0)
        s = float(np.sum(a_t))
        if s < 1e-12:
            return np.inf
        a_t = a_t / s
        return compute_obj(a_t)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)

    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
        if hi - lo < 1e-7:
            break

    t_best = (lo + hi) / 2.0
    a_best = (1.0 - t_best) * a + t_best * g
    a_best = np.maximum(a_best, 0)
    s = float(np.sum(a_best))
    if s < 1e-12:
        return a.copy(), f(0.0)
    a_best = a_best / s
    return a_best, f(t_best)


def smoothed_grad(a, tau):
    """Compute gradient of smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_v = np.exp(tau * (conv - max_c))
    p = exp_v / float(np.sum(exp_v))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def nesterov_smoothed_descent(a, time_budget=100.0, tau_init=50.0, tau_end=5000.0):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_grad(y, tau)
        g = g - float(np.mean(g))

        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.2 * lr + 0.8 * min(max(bb_lr, 1e-8), 20.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9998)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 5.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a

def perturb_random(a, rng, scale=0.05):

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        # Multiple rounds of LP polish with perturbations between rounds
        n_rounds = 8
        round_budget = p2_budget / n_rounds

        for rnd in range(n_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(round_budget, remaining * 0.6)
            if alloc < 20:
                break

            if rnd > 0:
                # Perturb to escape local minimum
                if rnd % 3 == 0:
                    a_cur = perturb_random(a_cur, rng, scale=0.03 + 0.01 * rnd)
                else:
                    a_cur = a_cur * np.exp(rng.normal(0, 0.02, len(a_cur)))
                    a_cur = np.maximum(a_cur, 0)
                    a_cur = a_cur / a_cur.sum()

            a_new, obj_new, iters = lp_polish(a_cur, max_iter=100,
                                              time_limit=alloc, verbose=(rnd == 0))
            if obj_new < best_score - 1e-12:
                best_score = obj_new
                best_seq = a_new.copy()
                a_cur = a_new.copy()
                print(f"  Round {rnd}: obj={best_score:.10f} ({iters} LP iters) **")
            else:
                print(f"  Round {rnd}: obj={obj_new:.10f} ({iters} LP iters)")
                a_cur = a_new.copy()

    # === Phase 3: Multi-n Exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        n_current - 100, n_current - 50, n_current + 50,
        n_current + 100, n_current + 200, n_current + 400,
        600, 700, 750, 850, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1600 and n != n_current]
    n_options.sort(key=lambda x: abs(x - n_current))

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        # Quick GD
        alloc1 = min(remaining * 0.05, 40.0)
        if alloc1 > 15:
            a_try = nesterov_smoothed_descent(a_try, time_budget=alloc1,
                                              tau_init=100.0, tau_end=3000.0)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.08, 100.0)
        if alloc2 > 20:
            a_try, obj_try, _ = lp_polish(a_try, max_iter=50, time_limit=alloc2)
        else:
            obj_try = compute_obj(a_try)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 4: Final Intensive Polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau Nesterov
    if remaining > 60:
        alloc = min(remaining * 0.15, 100.0)
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                          tau_init=200.0, tau_end=10000.0)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Final GD: obj={best_score:.10f} **")
        else:
            print(f"  Final GD: obj={obj_sgd:.10f}")

    # Multiple LP polish rounds
    for lp_rnd in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        alloc = min(remaining * 0.2, 100.0)
        if alloc < 15:
            break

        a_cur = best_seq.copy()
        if lp_rnd > 0 and lp_rnd % 2 == 1:
            a_cur = perturb_random(a_cur, rng, scale=0.02)

        a_lp, obj_lp, iters = lp_polish(a_cur, max_iter=100, time_limit=alloc)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP {lp_rnd}: obj={best_score:.10f} ({iters} iters) **")
        else:
            print(f"  Final LP {lp_rnd}: obj={obj_lp:.10f} ({iters} iters)")

    # === Final: One more n微调 ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_now = len(best_seq)
        for n_adj in [n_now - 20, n_now + 20, n_now + 50]:
            if n_adj < 300 or n_adj > 2000:
                continue
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            a_adj = resample_sequence(best_seq, n_adj)
            a_adj = a_adj / a_adj.sum()
            alloc = min(remaining * 0.3, 50.0)
            if alloc > 15:
                a_adj, obj_adj, _ = lp_polish(a_adj, max_iter=30, time_limit=alloc)
            else:
                obj_adj = compute_obj(a_adj)
            if obj_adj < best_score:
                best_score = obj_adj
                best_seq = a_adj.copy()
                print(f"  Fine-tune n={n_adj}: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
