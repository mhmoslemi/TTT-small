def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        lr = max(1e-8, lr * 0.9999)

        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def golden_section_search(f, a_lo, a_hi, tol=0.01, max_iter=40):
    """Golden section search for minimum of f on [a_lo, a_hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = f(c)
    fd = f(d)

    for _ in range(max_iter):
        if abs(a_hi - a_lo) < tol:
            break
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = f(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = f(d)

    return (a_lo + a_hi) / 2.0


def peak_pressure_descent(a, time_budget=60.0, K=10, verbose=False):
    """
    Peak-pressure method: compute pressure on each element from top peaks,
    move mass from high-pressure to low-pressure regions.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.0 / np.sqrt(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        if max_conv < 1e-15:
            break

        # Get top K peaks with soft weighting
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        weights = np.exp((peak_vals - peak_vals[0]) * 8.0 / max(peak_vals[0], 1e-15))
        weights /= weights.sum()

        # Compute pressure on each element
        pressure = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                pressure[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        # Normalize pressure to have mean 1
        mean_p = np.mean(pressure)
        if mean_p > 1e-15:
            pressure = pressure / mean_p

        # Move mass: reduce high-pressure, increase low-pressure
        update = (1.0 / np.maximum(pressure, 0.1))
        a_new = a * update
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        # Blend for stability
        t_blend = min(0.3, lr)
        a_try = (1.0 - t_blend) * a + t_blend * a_new
        a_try = np.maximum(a_try, 0)
        a_try = a_try / a_try.sum()

        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            stall = 0
            lr = min(lr * 1.05, 5.0 / np.sqrt(n))
        else:
            stall += 1
            lr *= 0.95
            if stall > 200:
                break

        it += 1

    return best_a


def lp_polish_aggressive(a, time_budget=200.0, verbose=False, seed=42):
    """
    Intensive LP polishing with aggressive constraint tightening,
    full LP replacement, and adaptive restarts.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    no_improve = 0
    last_M = None

    # Aggressive constraint tightening schedule
    tight_fracs = [0.85, 0.88, 0.91, 0.94, 0.96, 0.98]

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start

        # Solve LP with current a
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        if last_M is not None and abs(M_lp - last_M) < 1e-10:
            pass  # LP converged, try tightening
        last_M = M_lp

        # Try full replacement first (t=1)
        a_full = g_lp.copy()
        obj_full = compute_obj(a_full)

        if obj_full < best_obj - 1e-13:
            best_obj = obj_full
            best_a = a_full.copy()
            a = a_full.copy()
            iter_count += 1
            no_improve = 0
            if verbose and iter_count % 3 == 0:
                print(f"    LP-full it{iter_count}: obj={best_obj:.8f}, M={M_lp:.6f}, el={elapsed:.0f}s")
            continue

        # Golden section search for optimal blend
        def blend_obj(t):
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return 1e10
            a_try = a_try / s_try
            return compute_obj(a_try)

        t_opt = golden_section_search(blend_obj, 0.05, 0.95, tol=0.02, max_iter=30)

        a_try = (1.0 - t_opt) * best_a + t_opt * g_lp
        a_try = np.maximum(a_try, 0)
        a_try = a_try / a_try.sum()
        obj_try = compute_obj(a_try)

        iter_count += 1

        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()
            no_improve = 0
            if verbose and iter_count % 5 == 0:
                print(f"    LP-blend it{iter_count}: obj={best_obj:.8f}, t={t_opt:.3f}, M={M_lp:.6f}, el={elapsed:.0f}s")
        else:
            no_improve += 1
            if no_improve in [2, 3]:
                # Try aggressive constraint tightening
                for frac in tight_fracs:
                    M_tight = M_lp * frac
                    g_tight, M_ach = solve_lp(a, M_target=M_tight)
                    if g_tight is not None:
                        # Try different blend factors
                        for t_try in [1.0, 0.7, 0.4, 0.2]:
                            a_t = (1.0 - t_try) * best_a + t_try * g_tight
                            a_t = np.maximum(a_t, 0)
                            a_t = a_t / a_t.sum()
                            obj_t = compute_obj(a_t)
                            if obj_t < best_obj - 1e-13:
                                best_obj = obj_t
                                best_a = a_t.copy()
                                a = a_t.copy()
                                no_improve = 0
                                if verbose:
                                    print(f"    LP-tight it{iter_count}: obj={best_obj:.8f}, frac={frac}, t={t_try}")
                                break
                        if no_improve == 0:
                            break
            elif no_improve >= 5:
                # Perturb and continue
                scale = 0.005 + 0.003 * min(no_improve, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                if iter_count > 80 and time.monotonic() - start > time_budget * 0.7:
                    break

    return best_a


def ping_pong_lp(a, b, time_budget=100.0, verbose=False):
    """
    Ping-pong LP: alternate between finding best response to a (LP with kernel a)
    and best response to b (LP with kernel b). This creates an alternating
    minimization that might find better fixed points.
    """
    if b is None:
        b = a.copy()

    best_a = a.copy()
    best_b = b.copy()
    best_obj_a = compute_obj(a)
    best_obj_b = compute_obj(b)
    start = time.monotonic()

    a_curr = a.copy()
    b_curr = b.copy()

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        # Step 1: Find best g minimizing max conv(a_curr, g)
        g_new, M1 = solve_lp(a_curr)
        if g_new is not None:
            b_curr = 0.5 * b_curr + 0.5 * g_new
            b_curr = b_curr / b_curr.sum()
            obj_b = compute_obj(b_curr)
            if obj_b < best_obj_b - 1e-13:
                best_obj_b = obj_b
                best_b = b_curr.copy()
                stall = 0
            it += 1

        # Step 2: Find best h minimizing max conv(b_curr, h)
        h_new, M2 = solve_lp(b_curr)
        if h_new is not None:
            a_curr = 0.5 * a_curr + 0.5 * h_new
            a_curr = a_curr / a_curr.sum()
            obj_a = compute_obj(a_curr)
            if obj_a < best_obj_a - 1e-13:
                best_obj_a = obj_a
                best_a = a_curr.copy()
                stall = 0
                if verbose and it % 5 == 0:
                    print(f"    PP it{it}: obj_a={best_obj_a:.8f}, obj_b={best_obj_b:.8f}")
            it += 1

        # Track overall best
        if stall > 20:
            # Try full replacement
            if g_new is not None and compute_obj(g_new) < best_obj_a:
                best_a = g_new.copy()
                best_obj_a = compute_obj(g_new)
                a_curr = best_a.copy()
            stall = 0
            it += 1

        if it - stall * 10 > 100 and stall > 5:
            break

    if best_obj_a < best_obj_b:
        return best_a
    else:
        return best_b


def subgradient_refine(a, time_budget=100.0, K=5, verbose=False):
    """Subgradient method with top-K peaks, momentum, and adaptive step."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]

        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        weights = np.exp((peak_vals - max_pv) * 5.0 / max_pv)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 300:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-10:
                    break
            else:
                lr *= 0.995

        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        c1 = rng.integers(0, n // 3)
        c2 = rng.integers(2 * n // 3, n)
        w = max(1, n // 10)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 8:
        # Triangular with flat top
        mid = n // 2
        half = mid
        a = np.zeros(n)
        flat_width = max(1, n // 6)
        for i in range(n):
            d = abs(i - mid)
            if d <= flat_width:
                a[i] = 1.0
            elif d < half:
                a[i] = max(0, 1.0 - (d - flat_width) / max(1, half - flat_width))
            else:
                a[i] = 0.0
        a = a / a.sum()
    elif mode == 9:
        # Raised cosine squared
        x = np.linspace(0, 1, n)
        a = (np.cos(np.pi * x)) ** 4
        a = np.maximum(a, 1e-8)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

    # === Phase 9: Final intensive LP ===
    print(f"\n=== Phase 9: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.55, 180.0)
        a_final = lp_polish_aggressive(best_seq, time_budget=alloc, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # Quick final subgradient + peak pressure
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.25, 30.0)
        a_sg = subgradient_refine(best_seq, time_budget=alloc, K=8)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc = min(remaining * 0.3, 20.0)
            a_pp = peak_pressure_descent(best_seq, time_budget=alloc, K=20)
            obj_pp = compute_obj(a_pp)
            if obj_pp < best_score:
                best_score = obj_pp
                best_seq = a_pp.copy()
                print(f"  Final PP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
