1. **LP Alternating with direct assignment**: When the LP solution `g` gives `max(conv(a,g)) < max(conv(a,a))`, try setting `a = g` directly (not just mixing). This can make faster progress.

2. **Two-sided LP**: After getting `g` from the first LP, solve another LP with `g` fixed: `min M s.t. conv(g, h) <= M, sum(h)=1, h>=0`. This gives a "double-sided" improvement.

3. **Peak-targeted perturbation**: After LP converges, for the specific lags achieving the max, try to reduce them by transferring mass from positions that contribute to those lags to positions that don't.

4. **Continuous optimization with L-BFGS**: Use a smooth approximation of the max (log-sum-exp) and optimize with L-BFGS, using the LP solutions as warm starts.

5. **Symmetry search**: Try both symmetric and asymmetric initializations.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    # Try cvxpy first
    for solver in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    # Fallback to scipy linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None

def peak_shaving(a, time_budget, verbose=False):
    """
    Fine-grained peak shaving: identify peak lags, find contributors,
    and make small targeted transfers to reduce peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    for rnd in range(500):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        
        # Find lags near the maximum
        threshold = max_val * 0.95
        peak_lags = np.where(conv >= threshold)[0]
        
        if len(peak_lags) == 0:
            break
        
        improved = False
        
        for k in peak_lags[:5]:  # Focus on top 5 peak lags
            k = int(k)
            
            # Find the main contributors to this lag
            # conv[k] = sum_i a[i]*a[k-i] for valid i
            contribs = []
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contribs.append((a[i] * a[j], i, j))
            
            contribs.sort(reverse=True)
            
            # Try to reduce the top contributors by moving mass to "quiet" positions
            for contrib, i, j in contribs[:8]:
                if contrib < 1e-8:
                    continue
                # Find quiet positions (low a values that don't contribute to peak lags)
                quiet_score = np.zeros(n)
                for p in peak_lags[:3]:
                    p = int(p)
                    q = p - i
                    if 0 <= q < n:
                        quiet_score[i] -= a[q]
                
                # Target: position with low quiet_score (least active in peaks)
                targets = np.argsort(quiet_score)[:5]
                
                for target in targets:
                    if target == i or target == j:
                        continue
                    
                    # Try small transfers
                    for frac in [0.002, 0.005, 0.01, 0.02]:
                        delta = frac * a[i]
                        if delta <= 0 or delta >= a[i] * 0.9:
                            continue
                        
                        a[i] -= delta
                        a[target] += delta
                        
                        new_conv = compute_conv(a)
                        new_max = float(np.max(new_conv))
                        new_obj = 2.0 * n * new_max
                        
                        if new_obj < best_obj - 1e-12:
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[i] += delta
                            a[target] -= delta
                            break
                    
                    if improved:
                        break
            
            if improved:
                break
        
        if not improved and rnd > 50:
            break

        if verbose and rnd % 50 == 0:
            print(f"          Peak-shave rnd{rnd}: obj={best_obj:.8f}")

    return best_a


def lsesq_optimize(a, time_budget, verbose=False):
    """
    Optimize using log-sum-exp approximation of the max, with L-BFGS.
    This handles the non-smoothness of the max function.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Use log-sum-exp as smooth approximation of max
    def smooth_obj(params, beta=100):
        """Compute smooth approximation of 2n*max(conv)."""
        a_local = np.exp(params)
        a_local = a_local / a_local.sum()
        a_local = np.maximum(a_local, 1e-15)
        conv = compute_conv(a_local)
        # Log-sum-exp: log(sum(exp(beta*conv))) / beta ≈ max(conv) for large beta
        m = conv.max()
        lse = m + np.log(np.sum(np.exp(beta * (conv - m)))) / beta
        return 2.0 * n * lse

    def smooth_grad(params, beta=100):
        """Numerical gradient of smooth objective."""
        eps = 1e-7
        grad = np.zeros(n)
        f0 = smooth_obj(params, beta)
        for i in range(n):
            params[i] += eps
            f1 = smooth_obj(params, beta)
            params[i] -= eps
            grad[i] = (f1 - f0) / eps
        return grad

    # Optimize in log-space for automatically positive values
    log_a = np.log(a)
    
    beta_schedule = [20, 50, 100, 200, 500, 1000]
    
    for beta in beta_schedule:
        if time.monotonic() - start > time_budget:
            break
        beta_t = beta
        
        def obj(params):
            a_local = np.exp(params)
            a_local = np.maximum(a_local, 1e-15)
            a_local = a_local / a_local.sum()
            conv = compute_conv(a_local)
            m = conv.max()
            lse = m + np.log(np.sum(np.exp(beta_t * (conv - m)))) / beta_t
            return 2.0 * n * lse

        # Use scipy minimize
        try:
            res = minimize(obj, log_a, method='L-BFGS-B', 
                          options={'maxiter': 200, 'gtol': 1e-8, 'iprint': -1})
            if res.success or res.fun < obj(log_a):
                a_new = np.exp(res.x)
                a_new = np.maximum(a_new, 1e-15)
                a_new = a_new / a_new.sum()
                real_obj = compute_obj(a_new)
                if real_obj < best_obj - 1e-10:
                    best_obj = real_obj
                    best_a = a_new.copy()
                    log_a = res.x.copy()
                    if verbose:
                        print(f"          LSE beta={beta}: obj={best_obj:.8f}")
                else:
                    log_a = res.x.copy()
        except Exception:
            pass

    return best_a


def coordinate_descent_fine(a, time_budget, verbose=False):
    """
    Fine coordinate descent: for each coordinate, find the optimal value
    by 1D search (keeping sum constraint by adjusting one other coordinate).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    for sweep in range(100):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        # Randomize the order
        rng = np.random.default_rng(sweep * 1000 + n)
        order = rng.permutation(n)

        for m in order[:min(n, 300)]:  # Limit per sweep for speed
            if time.monotonic() - start > time_budget:
                break

            # Choose a partner to adjust (to maintain sum)
            # Pick the coordinate with largest value (other than m)
            others = [i for i in range(n) if i != m]
            partner = others[np.argmax(a[others])]
            
            if a[partner] < 1e-10:
                continue

            # Binary search for optimal a[m]
            # a[partner] adjusts to maintain sum: a[partner] += (old_a[m] - new_a[m])
            lo = 0.0
            hi = a[m] + a[partner]  # max we can set a[m] to
            
            best_m_val = a[m]
            best_m_obj = best_obj
            
            # Golden section search
            gr = (np.sqrt(5) - 1) / 2
            a1 = hi - gr * (hi - lo)
            a2 = lo + gr * (hi - lo)
            
            def eval_m(val):
                a[m] = val
                a[partner] = a[partner] + (best_m_val - val)
                if a[partner] < 0:
                    return np.inf
                conv = compute_conv(a)
                obj = 2.0 * n * float(np.max(conv))
                return obj
            
            f1 = eval_m(a1)
            f2 = eval_m(a2)
            
            for _ in range(20):
                if time.monotonic() - start > time_budget:
                    break
                if f1 < f2:
                    hi = a2
                    a2 = a1
                    f2 = f1
                    a1 = hi - gr * (hi - lo)
                    f1 = eval_m(a1)
                else:
                    lo = a1
                    a1 = a2
                    f1 = f2
                    a2 = lo + gr * (hi - lo)
                    f2 = eval_m(a2)
            
            # Evaluate at the best point
            m_opt = (lo + hi) / 2
            f_opt = eval_m(m_opt)
            
            if f_opt < best_m_obj - 1e-12:
                best_m_obj = f_opt
                best_m_val = m_opt
                best_obj = f_opt
                best_a = a.copy()
                improved = True
            else:
                # Restore
                a[m] = best_m_val
                a[partner] = a[partner] + (best_m_val - a[m])
                a[m] = best_m_val

        if not improved and sweep > 10:
            break

        if verbose and sweep % 10 == 0:
            print(f"          CoordDescent sweep{sweep}: obj={best_obj:.8f}")

    return best_a


def ping_pong_lp(a, time_budget, verbose=False):
    """
    Two-sided LP: alternate between optimizing left and right sides.
    This can escape local minima that one-sided LP cannot.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(100):
        if time.monotonic() - start > time_budget:
            break

        # Forward: fix left=a, optimize right
        g1, _ = solve_lp(a)
        if g1 is None:
            break
        
        # Backward: fix right=g1, optimize left (i.e., conv(h, g1) <= M)
        # This is the same LP but with g1 as the fixed side
        g2, _ = solve_lp(g1)
        if g2 is None:
            break

        # Try various combinations
        candidates = [g1, g2, (g1 + g2) / 2, (0.3 * g1 + 0.7 * g2), (0.7 * g1 + 0.3 * g2)]
        
        best_cand_obj = np.inf
        best_cand = None
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s_c = float(np.sum(cand))
            if s_c < 1e-10:
                continue
            cand = cand / s_c
            obj_c = compute_obj(cand)
            if obj_c < best_cand_obj:
                best_cand_obj = obj_c
                best_cand = cand

        if best_cand is not None and best_cand_obj < best_obj - 1e-10:
            best_obj = best_cand_obj
            best_a = best_cand.copy()
            a = best_cand.copy()
            if verbose and it % 10 == 0:
                print(f"          PingPong it{it}: obj={best_obj:.8f}")
        else:
            # Try mixing with current
            if best_cand is not None:
                mixed = 0.5 * a + 0.5 * best_cand
                mixed = np.maximum(mixed, 0)
                mixed = mixed / mixed.sum()
                obj_m = compute_obj(mixed)
                if obj_m < best_obj - 1e-10:
                    best_obj = obj_m
                    best_a = mixed.copy()
                    a = mixed.copy()
                else:
                    break

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode % 10 == 0:
        a = np.ones(n) / n
    elif mode % 10 == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode % 10 == 2:
        # Linear ramp (asymmetric)
        x = np.linspace(0, 1, n)
        a = 0.5 + x
        a = a / a.sum()
    elif mode % 10 == 3:
        # Exponential decay
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * x)
        a = a / a.sum()
    elif mode % 10 == 4:
        # Two bumps
        x = np.linspace(0, 1, n)
        a = np.exp(-10 * (x - 0.3)**2) + 0.7 * np.exp(-10 * (x - 0.7)**2)
        a = a / a.sum()
    elif mode % 10 == 5:
        # Cosine window
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 6:
        # Asymmetric cosine
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * (x - 0.3))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 7:
        # Random smooth
        coeffs = rng.normal(0, 1, 5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for i, c in enumerate(coeffs):
            a += c * np.sin(2 * np.pi * (i + 1) * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 8:
        # Triangle (asymmetric)
        x = np.linspace(0, 1, n)
        peak = 0.6
        a = np.where(x < peak, x / peak, (1 - x) / (1 - peak))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n) * 0.3)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

Improved hybrid optimization for minimizing peak autocorrelation of non-negative step functions.
Key improvements: direct LP assignment, peak-targeted coordinate descent, L-BFGS with log-sum-exp,

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv, float(M.value)

            continue

        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))

        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))

            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local.copy()
            last_good_t = best_t_val
            stall = 0
            if verbose and iteration % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"            LP-mix it{iteration}: obj={best_obj:.8f}, t={last_good_t:.3f}, el={elapsed:.0f}s")
        else:
            stall += 2

        if stall > patience:
            rng = np.random.default_rng((iteration * 7919 + n) % (2**31))
            strategy = stall % 4
            if strategy == 0:
                scale = 0.03 + 0.01 * min(stall // patience, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
            elif strategy == 1:
                x = np.linspace(0, 1, n)
                freq = int(rng.integers(2, 12))
                phase = rng.uniform(0, 2 * np.pi)
                a = best_a * (1.0 + 0.08 * np.sin(2 * np.pi * freq * x + phase))
            elif strategy == 2:
                block_size = max(1, n // 8)
                a = best_a.copy()
                for _ in range(3):
                    si = int(rng.integers(0, n - block_size))
                    a[si:si + block_size] *= rng.uniform(0.6, 1.4)
            else:
                # Swap-based perturbation
                a = best_a.copy()
                for _ in range(5):
                    i1 = int(rng.integers(n))
                    i2 = int(rng.integers(n))
                    swap_amt = 0.05 * (a[i1] + a[i2]) / 2
                    if a[i1] >= swap_amt:
                        a[i1] -= swap_amt
                        a[i2] += swap_amt

            a = np.maximum(a, 0)
            sv = float(np.sum(a))
            if sv > 1e-10:
                a = a / sv
            else:
                a = best_a.copy()

            obj_chk = compute_obj(a)
            if obj_chk < best_obj:
                best_obj = obj_chk
                best_a = a.copy()
            stall = 0
            last_good_t = 0.5

    return best_a


def peak_shaving(a, time_budget, verbose=False):
    """
    Fine-grained peak shaving: identify peak lags, find contributors,
    make small targeted transfers to reduce peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    for rnd in range(300):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        threshold = max_val * 0.92
        peak_lags = np.where(conv >= threshold)[0]

        if len(peak_lags) == 0:
            break

        improved = False
        peak_lags_list = [int(x) for x in peak_lags[:5]]

        # Precompute quiet scores
        quiet_score = np.zeros(n)
        for k in peak_lags_list[:3]:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    quiet_score[i] += a[i] * a[j]

        for k in peak_lags_list:
            if time.monotonic() - start > time_budget:
                break

            contribs = []
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contribs.append((a[i] * a[j], i, j))
            contribs.sort(reverse=True)

            for contrib, i, j in contribs[:6]:
                if contrib < 1e-10:
                    continue
                src = i if a[i] >= a[j] else j
                if a[src] < 1e-10:
                    continue

                # Find quiet targets
                targets = np.argsort(quiet_score)[:8]
                for target in targets:
                    target = int(target)
                    if target == src or target == j or target == i:
                        continue

                    for frac in [0.003, 0.007, 0.015, 0.03]:
                        delta = frac * a[src]
                        if delta <= 0 or delta >= a[src] * 0.8:
                            continue

                        a[src] -= delta
                        a[target] += delta

                        new_obj = 2.0 * n * float(np.max(compute_conv(a)))
                        if new_obj < best_obj - 1e-13:
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[src] += delta
                            a[target] -= delta
                            break

                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved and rnd > 30:
            # Try a small random nudge to escape
            if rnd % 10 == 0:
                rng = np.random.default_rng(rnd * 31 + n)
                idx = rng.choice(n, size=3, replace=False)
                for idx_i in idx[:2]:
                    delta = 0.005 * a[idx_i]
                    if delta > 0:
                        a[idx_i] -= delta
                        a[int(idx[2])] += delta
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        chk = 2.0 * n * float(np.max(compute_conv(a)))
                        if chk < best_obj:
                            best_obj = chk
                            best_a = a.copy()
                            improved = True
                        else:
                            a = best_a.copy()

        if verbose and rnd % 40 == 0:
            print(f"            PeakShave rnd{rnd}: obj={best_obj:.8f}")

    return best_a


def lsesq_optimize(a, time_budget, verbose=False):
    """
    Optimize using log-sum-exp smooth approximation with L-BFGS in log-space.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def smooth_max_conv(params, beta):
        a_local = np.exp(params - np.max(params))
        a_local = np.maximum(a_local, 1e-30)
        a_local = a_local / a_local.sum()
        conv = compute_conv(a_local)
        m = conv.max()
        lse = m + np.log(np.sum(np.exp(beta * (conv - m)))) / beta
        return lse

    log_a = np.log(a)
    beta_schedule = [30, 80, 200, 500, 1000, 2000]

    for beta in beta_schedule:
        if time.monotonic() - start > time_budget:
            break

        def obj_func(params):
            return 2.0 * n * smooth_max_conv(params, beta)

        try:
            res = minimize(obj_func, log_a, method='L-BFGS-B',
                          options={'maxiter': 100, 'gtol': 1e-9, 'iprint': -1})
            a_new = np.exp(res.x - np.max(res.x))
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            real_obj = compute_obj(a_new)
            log_a = res.x.copy()
            if real_obj < best_obj - 1e-10:
                best_obj = real_obj
                best_a = a_new.copy()
                if verbose:
                    print(f"            LSE b={beta}: obj={best_obj:.8f} **")
            elif verbose:
                print(f"            LSE b={beta}: obj={real_obj:.8f}")
        except Exception:
            pass

    return best_a


def coordinate_descent_fine(a, time_budget, verbose=False):
    """
    Fine coordinate descent with golden section search on pairs of coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(200):
        if time.monotonic() - start > time_budget:
            break

        rng = np.random.default_rng(sweep * 1009 + n)
        order = rng.permutation(n)
        improved_this_sweep = False

        for m in order[:min(n, 400)]:
            if time.monotonic() - start > time_budget:
                break

            # Pick partner: coordinate with significant mass, different from m
            mask = np.ones(n, dtype=bool)
            mask[m] = False
            candidates = np.where(mask & (a > 1e-8))[0]
            if len(candidates) == 0:
                break
            partner = candidates[np.argmax(a[candidates])]

            orig_m = a[m]
            orig_p = a[partner]
            hi = orig_m + orig_p
            lo = 0.0

            # Golden section
            a1 = hi - gr * (hi - lo)
            a2 = lo + gr * (hi - lo)

            def eval_at(val):
                if val < -1e-15 or val > hi + 1e-15:
                    return np.inf
                val = max(0.0, min(val, hi))
                a[m] = val
                a[partner] = orig_p + orig_m - val
                if a[partner] < -1e-15:
                    a[m] = orig_m
                    a[partner] = orig_p
                    return np.inf
                return 2.0 * n * float(np.max(compute_conv(a)))

            f1 = eval_at(a1)
            f2 = eval_at(a2)

            for _ in range(15):
                if time.monotonic() - start > time_budget:
                    break
                if f1 < f2:
                    hi = a2
                    a2 = a1
                    f2 = f1
                    a1 = hi - gr * (hi - lo)
                    f1 = eval_at(a1)
                else:
                    lo = a1
                    a1 = a2
                    f1 = f2
                    a2 = lo + gr * (hi - lo)
                    f2 = eval_at(a2)

            m_opt = (lo + hi) / 2.0
            f_opt = eval_at(m_opt)

            if f_opt < best_obj - 1e-13:
                best_obj = f_opt
                best_a = a.copy()
                improved_this_sweep = True
            else:
                a[m] = orig_m
                a[partner] = orig_p

        if not improved_this_sweep and sweep > 20:
            break

        if verbose and sweep % 20 == 0:
            print(f"            CoordDescent sweep{sweep}: obj={best_obj:.8f}")

    return best_a


def ping_pong_lp(a, time_budget, verbose=False):
    """
    Two-sided LP: alternate fixing left and right sides of convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(80):
        if time.monotonic() - start > time_budget:
            break

        g1, _ = solve_lp(a)
        if g1 is None:
            break
        g2, _ = solve_lp(g1)
        if g2 is None:
            break

        candidates = [g1, g2]
        # Also try averages
        candidates.append((g1 + g2) / 2.0)
        candidates.append(0.3 * g1 + 0.7 * g2)
        candidates.append(0.7 * g1 + 0.3 * g2)

        best_cand_obj = np.inf
        best_cand = None
        for cand in candidates:
            cand = np.maximum(cand, 0)
            sv = float(np.sum(cand))
            if sv < 1e-10:
                continue
            cand = cand / sv
            obj_c = compute_obj(cand)
            if obj_c < best_cand_obj:
                best_cand_obj = obj_c
                best_cand = cand.copy()

        if best_cand is not None:
            if best_cand_obj < best_obj - 1e-10:
                best_obj = best_cand_obj
                best_a = best_cand
                a = best_cand.copy()
                if verbose and it % 10 == 0:
                    print(f"            PingPong it{it}: obj={best_obj:.8f}")
            else:
                # Try mixing
                mixed = 0.4 * a + 0.6 * best_cand
                mixed = np.maximum(mixed, 0)
                sv = float(np.sum(mixed))
                if sv > 1e-10:
                    mixed = mixed / sv
                    obj_m = compute_obj(mixed)
                    if obj_m < best_obj - 1e-10:
                        best_obj = obj_m
                        best_a = mixed.copy()
                        a = mixed.copy()
                    else:
                        break
                else:
                    break
        else:
            break

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode % 10 == 0:
        a = np.ones(n) / n
    elif mode % 10 == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode % 10 == 2:
        x = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * x
        a = a / a.sum()
    elif mode % 10 == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4.0 * x)
        a = a / a.sum()
    elif mode % 10 == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (x - 0.35)**2) + 0.6 * np.exp(-12.0 * (x - 0.75)**2)
        a = a / a.sum()
    elif mode % 10 == 5:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 6:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.4 * np.cos(2.0 * np.pi * (x - 0.25))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 7:
        coeffs = rng.normal(0, 1, 5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for i in range(5):
            a += coeffs[i] * np.sin(2.0 * np.pi * (i + 1) * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 8:
        x = np.linspace(0, 1, n)
        peak = 0.55
        a = np.where(x < peak, x / peak, (1.0 - x) / (1.0 - peak + 1e-10))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n) * 0.4)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase hybrid of LP alternating, peak shaving,
    L-BFGS, coordinate descent, ping-pong, n-sweep, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"\nInitial: n={n_current}, obj={best_score:.8f}, target=1.5027")

    # === Phase 1: Quick LP polish (8%) ===
    print("\n=== Phase 1: Quick LP Polish ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 80.0)
    if p1_budget > 15:
        a_p1 = lp_alternate_improved(best_seq, time_budget=p1_budget, verbose=True)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Phase1 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 LP: obj={obj_p1:.8f}")

    # === Phase 2: Peak shaving (6%) ===
    print("\n=== Phase 2: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.06, 50.0)
    if p2_budget > 10:
        a_ps = peak_shaving(best_seq, time_budget=p2_budget, verbose=True)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  PeakShave: obj={best_score:.8f} **")
        else:
            print(f"  PeakShave: obj={obj_ps:.8f}")

    # === Phase 3: n-sweep (15%) ===
    print("\n=== Phase 3: n-Sweep ===")
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.15
    n_options = sorted(set([
        max(500, n_current - 80),
        max(500, n_current - 40),
        max(500, n_current - 20),
        n_current,
        n_current + 20,
        n_current + 40,
        n_current + 80,
        n_current + 120,
        700, 720, 740, 760, 780, 800, 820, 840, 860, 880, 900, 920, 950, 1000,
    ]))
    n_options = [x for x in n_options if 500 <= x <= 1200]

    per_n_budget = p3_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_n_budget, remaining * 0.1)
        if alloc < 8:
            alloc = max(8, remaining * 0.04)

        a_try = resample_sequence(best_seq, n_try)
        a_opt = lp_alternate_improved(a_try, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)
        marker = " **" if obj_opt < best_score else ""
        print(f"  n={n_try}: obj={obj_opt:.8f}{marker}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try

    # === Phase 4: Intensive LP (30%) ===
    print(f"\n=== Phase 4: Intensive LP (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.30, 420.0)

    n_rounds = 5
    per_round = p4_budget / n_rounds
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        alloc = min(per_round, remaining * 0.2)
        if alloc < 20:
            break

        if rnd == 0:
            a_start = best_seq.copy()
        elif rnd == 1:
            rng_r = np.random.default_rng(11111 + rnd)
            a_start = best_seq * np.exp(rng_r.normal(0, 0.015, n_current))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        elif rnd == 2:
            x = np.linspace(0, 1, n_current)
            a_start = best_seq * (1.0 + 0.04 * np.sin(2.0 * np.pi * 4.0 * x + 0.5))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        elif rnd == 3:
            rng_r = np.random.default_rng(22222 + rnd)
            a_start = best_seq * np.exp(rng_r.normal(0, 0.04, n_current))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        else:
            a_start = best_seq.copy()

        if rnd == 4:
            a_lp = ping_pong_lp(a_start, time_budget=alloc, verbose=True)
        else:
            a_lp =
