import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def solve_reduced_lp(a, max_constraints=200):
    """
    Solve reduced LP: min M s.t. (a*g)[m] <= M for active m, sum(g)=1, g>=0.
    Only uses the top max_constraints constraints.

    # conv(a, g)[m] = sum_i a[i] * g[m-i]
    # This is a Toeplitz-like system.
    
    # For the reduced LP, pick the top constraints
    # First, solve with all constraints but use a fast solver
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = max(0, j - (n - 1))
        hi = min(2 * n - 1, j + (n - 1)) + 1
        A[lo:hi, j] = a_norm[max(0, j - (n - 1)):min(n, hi - j + (n - 1) - (n - 1))]
        # Let me redo this more carefully
        # conv(a,g)[m] = sum_{i+j=m} a[i]*g[j] = sum_j a[m-j]*g[j]
        # So A[m,j] = a[m-j] if 0<=m-j<n, else 0
        pass

    # Redo: A[m, j] = a[m-j] if 0 <= m-j <= n-1, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        m_lo = j
        m_hi = min(j + n - 1, 2 * n - 2)
        A[m_lo:m_hi + 1, j] = a_norm[0:n - (m_hi - m_lo)]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

def spectral_perturb(a, rng, alpha=0.3):
    """
    Perturb the sequence by flattening its power spectrum.
    This encourages a more "random-like" structure.
    """
    n = len(a)
    N = 1
    while N < 2 * n:
        N <<= 1

    A = fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)

    # Compute target magnitude: compress dynamic range
    mag_min = np.max(mag[mag > 1e-10]) * 0.1
    mag_target = mag ** (1 - alpha) * (mag_min + 1e-10) ** alpha
    # Preserve DC component
    mag_target[0] = mag[0]
    mag_target[N // 2] = mag[N // 2]

    # Blend
    mag_new = (1 - alpha) * mag + alpha * mag_target
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, N))[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    print(f"\n=== Phase 4: LP/CD Alternate (budget: 15% of {time_left():.0f}s) ===")
    p4_budget = min(time_left() * 0.15, 150.0)
    if p4_budget > 40:
        for it in range(4):
            if time_left() < 40:
                break
            # LP step
            alloc_lp = p4_budget / (4 - it) * 0.4
            alloc_lp = min(alloc_lp, time_left() * 0.4)
            if alloc_lp > 10:
                a_lp = lp_polish_fast(best_seq, alloc_lp, rng)
                obj_lp = compute_obj(a_lp)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()

            # CD step
            alloc_cd = p4_budget / (4 - it) * 0.6
            alloc_cd = min(alloc_cd, time_left() * 0.5)
            if alloc_cd > 10:
                a_cd = coord_descent_fast(best_seq, alloc_cd, time.monotonic(), rng)
                obj_cd = compute_obj(a_cd)
                if obj_cd < best_score:
                    best_score = obj_cd
                    best_seq = a_cd.copy()

        print(f"  Alternate done: obj={best_score:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n (budget: 10% of {time_left():.0f}s) ===")
    p5_budget = time_left() * 0.10
    n_current = len(best_seq)

    n_options = []
    for dn in [-200, -100, -50, 50, 100, 200, 300]:
        nt = n_current + dn
        if 400 <= nt <= 1500:
            n_options.append(nt)
    n_options = sorted(set(n_options))

    for n_try in n_options[:4]:
        if time_left() < 60:
            break
        a_try = resample_seq(best_seq, n_try)
        a_try = a_try / a_try.sum()

        alloc1 = min(time_left() * 0.5, 60.0)
        if alloc1 > 15:
            a_try = coord_descent_fast(a_try, alloc1, time.monotonic(), rng)

        alloc2 = min(time_left() * 0.2, 30.0)
        if alloc2 > 10:
            a_try = lp_polish_fast(a_try, alloc2, rng)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts (budget: remaining) ===")
    n_best = len(best_seq)

    for r in range(5):
        if time_left() < 80:
            break
        mode = r % 10
        a_r = diverse_init(n_best, rng, mode)
        # Blend with best
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc1 = min(time_left() * 0.5, 80.0)
        if alloc1 > 20:
            a_r = coord_descent_fast(a_r, alloc1, time.monotonic(), rng)

        alloc2 = min(time_left() * 0.2, 30.0)
        if alloc2 > 10:
            a_r = lp_polish_fast(a_r, alloc2, rng)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    if time_left() > 30:
        alloc = min(time_left() * 0.4, 80.0)
        a_f = coord_descent_fast(best_seq, alloc, time.monotonic(), rng, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    if time_left() > 20:
        alloc = min(time_left() * 0.4, 50.0)
        a_f = lp_polish_fast(best_seq, alloc, rng, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    if time_left() > 15:
        alloc = min(time_left() * 0.6, 50.0)
        a_f = coord_descent_fast(best_seq, alloc, time.monotonic(), rng)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
