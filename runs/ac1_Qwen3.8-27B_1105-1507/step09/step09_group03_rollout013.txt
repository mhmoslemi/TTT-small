def compute_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)

        return np.array([])
    size = 1
    while size < 2 * n - 1:

    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    s = float(np.sum(a))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Build dense matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)

    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = a_norm[:end - j]
    return A

    Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).

        except Exception:
            continue
    return None, None

from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a_norm):
    """Build dense matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = a_norm[:end - j]
    return A


def solve_lp_minmax(a):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    if n > 1200:
        return None, None
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(M.value)
        except Exception:
            continue
    return None, None

def subgradient_fft(a, tol_frac=0.02):
    """
    Compute subgradient of max(conv(a,a)) using FFT-based cross-correlation.
    a should be normalized (sum=1).
    Returns (subgradient, conv, active_set).
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    tol = tol_frac * max_val
    active = np.where(conv >= max_val - tol)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    
    # Build indicator of active set
    S = np.zeros(2 * n - 1)
    S[active] = 1.0
    
    # sg[j] = (2/|active|) * sum_i a[i] * S[i+j]
    # = (2/|active|) * (reverse(a) * S)[n-1+j]
    a_rev = a[::-1]
    size = 1
    while size < len(a_rev) + len(S) - 1:
        size <<= 1
    A_fft = rfft(a_rev, size)
    S_fft = rfft(S, size)
    conv_result = irfft(A_fft * S_fft, size)
    
    sg = (2.0 / len(active)) * conv_result[n - 1:n - 1 + n]
    
    return sg, conv, active


def lp_subgradient_optimize(a, time_budget, verbose=False, seed=42):
    """
    Main optimization: alternates LP direction finding with FFT subgradient steps.
    Uses momentum and adaptive step size.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    it = 0
    stall = 0
    momentum_vec = np.zeros(n)
    momentum = 0.3
    lr_sg = 2.0 / np.sqrt(n)
    lp_freq = max(1, n // 200)  # do LP every lp_freq iterations
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed >= time_budget:
            break
        
        should_do_lp = (it % lp_freq == 0) and (n <= 900)
        
        if should_do_lp:
            g_lp, M_lp = solve_lp_minmax(a)
            if g_lp is not None:
                # Compute new momentum
                direction = g_lp - a
                momentum_vec = momentum * momentum_vec + direction
                momentum = min(0.7, momentum + 0.01)
                
                # Try several step sizes
                best_t = 0.0
                best_f = compute_obj(a)
                for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0]:
                    a_try = a + t * momentum_vec
                    a_try = project_simplex(a_try)
                    f_try = compute_obj(a_try)
                    if f_try < best_f:
                        best_f = f_try
                        best_t = t
                
                if best_t > 0 and best_f < best_obj - 1e-13:
                    a = a + best_t * momentum_vec
                    a = project_simplex(a)
                    obj = compute_obj(a)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()
                    stall = 0
                else:
                    # Try just the LP direction without momentum
                    for t in [0.1, 0.3, 0.5, 0.8]:
                        a_try = (1 - t) * a + t * g_lp
                        a_try = np.maximum(a_try, 0)
                        s_try = float(np.sum(a_try))
                        if s_try > 1e-10:
                            a_try = a_try / s_try
                            f_try = compute_obj(a_try)
                            if f_try < best_obj - 1e-13:
                                a = a_try
                                obj = compute_obj(a)
                                if obj < best_obj:
                                    best_obj = obj
                                    best_a = a.copy()
                                stall = 0
                                break
                    else:
                        stall += 1
                        momentum_vec *= 0.5
            else:
                stall += 1
        else:
            # Subgradient step
            sg, conv, active = subgradient_fft(a, tol_frac=0.02)
            
            # Adaptive lr
            a_try = project_simplex(a - lr_sg * sg)
            obj_try = compute_obj(a_try)
            
            if obj_try < best_obj - 1e-14:
                a = a_try
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                stall = 0
                lr_sg = min(lr_sg * 1.01, 5.0 / np.sqrt(n))
            else:
                # Backtrack
                lr_sg *= 0.7
                stall += 1
        
        it += 1
        
        # Perturbation when stuck
        if stall > 40:
            perturbation = 0.03 + 0.01 * (stall // 40)
            a = best_a * np.exp(rng.normal(0, min(perturbation, 0.15), n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            else:
                a = best_a.copy()
            stall = 0
            momentum_vec = np.zeros(n)
            momentum = 0.3
            lr_sg = 2.0 / np.sqrt(n)
            
            if it > 1000:
                break
        
        if verbose and it % 30 == 0:
            print(f"    LSG it{it}: obj={best_obj:.8f}, stall={stall}, lr={lr_sg:.4f}")
    
    return best_a
