"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses improved LP polishing with constraint reduction, adaptive line search,
multi-scale exploration, and enhanced gradient methods.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient_fft(a, tau):
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def build_conv_matrix_reduced(a, top_frac=0.6):
    """Build reduced convolution matrix keeping only top constraints."""
    n = len(a)
    conv = compute_conv(a)
    # Only keep top top_frac fraction of convolution coefficients
    n_keep = max(n, int(len(conv) * top_frac))
    # Get indices of the largest convolution coefficients
    top_idx = np.argsort(conv)[::-1][:n_keep]
    top_idx = np.sort(top_idx)
    
    A = np.zeros((len(top_idx), n))
    for j in range(n):
        # (a * e_j)[k] = a[k-j] if 0 <= k-j <= n-1
        # For each kept index k, A[k,j] = a[k-j] if valid
        k_vals = top_idx
        valid = (k_vals >= j) & (k_vals - j < n)
        A[valid, j] = a[k_vals[valid] - j]
    return A, top_idx


def solve_lp_reduced(a, top_frac=0.6, verbose=False):
    """Solve LP with reduced constraints."""
    n = len(a)
    a_norm = a / np.sum(a)
    A, top_idx = build_conv_matrix_reduced(a_norm, top_frac)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def fourier_perturbation(a, rng, strength=0.05):
    n = len(a)
    pert = np.zeros(n)
    n_modes = 5
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 8, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    else:
        a = np.ones(n) / n

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses reduced-constraint LP polishing with adaptive line search, multi-scale n exploration,
improved Nesterov accelerated smoothed gradient descent, and diversified restarts.
Key improvements over baseline: constraint-adaptive LP, smarter line search,
longer sequence exploration, and better stall recovery.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def build_conv_matrix(a, top_frac=1.0):
    """Build convolution matrix, optionally keeping only top constraints."""
    n = len(a)
    conv = compute_conv(a)
    n_total = len(conv)
    
    if top_frac >= 1.0:
        top_idx = np.arange(n_total)
    else:
        n_keep = max(n, int(n_total * top_frac))
        top_idx = np.argsort(conv)[::-1][:n_keep]
        top_idx = np.sort(top_idx)
    
    A = np.zeros((len(top_idx), n))
    for j in range(n):
        k_vals = top_idx
        valid = (k_vals >= j) & (k_vals - j < n)
        if np.any(valid):
            A[valid, j] = a[k_vals[valid] - j]
    return A, top_idx


def solve_lp(a, top_frac=0.6, verbose=False):
    """Solve LP: min M s.t. conv(a,x)_k <= M, sum(x)=1, x>=0. Uses top_frac constraints."""
    n = len(a)
    a_norm = a / np.sum(a)
    A, _ = build_conv_matrix(a_norm, top_frac)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

            best_obj = obj_try
            best_t = t
            best_a = a_try.copy()
    
    # Fine search around best_t
    if best_t > 0:
        for scale in [0.7, 0.8, 0.9, 1.1, 1.2, 1.3]:
            if evals >= max_evals:
                break
            t_fine = min(1.0, best_t * scale)
            a_try = (1.0 - t_fine) * a + t_fine * g_lp
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            evals += 1
            if obj_try < best_obj:
                best_obj = obj_try
                best_t = t_fine
                best_a = a_try.copy()
    
    return best_a, best_obj, best_t


def lp_step(a, top_frac=0.6):
    """Single LP iteration: solve LP, line search blend."""
    g_lp, M_lp = solve_lp(a, top_frac=top_frac)
    if g_lp is None:
        return a, compute_obj(a), 0.0
    a_new, obj, t = line_search_blend(a, g_lp)
    return a_new, obj, t


def adaptive_smoothed_descent(a, time_budget=200.0, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent with adaptive cosine annealing."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    base_lr = 3.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    grad_norms = []
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        
        # Cosine annealing for tau
        cos_val = 0.5 * (1 + np.cos(np.pi * progress))
        tau = tau_init + (tau_end - tau_init) * (1 - cos_val)
        
        if it % 5000 == 0 and verbose:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        g, conv = smoothed_gradient_fft(y, tau)
        g_norm = float(np.linalg.norm(g))
        grad_norms.append(g_norm)
        if len(grad_norms) > 300:
            grad_norms = grad_norms[-300:]
        
        # Adaptive learning rate
        avg_gnorm = float(np.mean(grad_norms))
        if avg_gnorm > 1e-15:
            adaptive_lr = 1.0 / (avg_gnorm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.05 * base_lr), 5.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adaptive_lr
        
        # Barzilai-Borwein step size
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                bb_lr = min(max(bb_lr, 1e-8), 3.0 / np.sqrt(n))
                lr = 0.6 * lr + 0.4 * bb_lr
        
        # Update
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-9, lr * 0.99995)
        
        # Reset if stuck
        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it
    
    return best_a


def fourier_perturbation(a, rng, strength=0.05):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 5
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 8, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def lp_polish_loop(a, time_budget, top_frac_start=0.5, top_frac_end=0.9, max_stall=8, seed=42, verbose=False):
    """LP polishing loop with adaptive constraint count and stall recovery."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        it += 1
        # Interpolate top_frac
        progress = min(1.0, (time.monotonic() - start) / max(time_budget, 1e-10))
        top_frac = top_frac_start + (top_frac_end - top_frac_start) * progress
        
        a_new, obj_new, t = lp_step(best_a, top_frac=top_frac)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            if verbose and (it % 3 == 0 or obj_new < best_obj + 1e-8):
                print(f"      LP it{it}: obj={best_obj:.8f}, t={t:.3f}, frac={top_frac:.2f}")
        else:
            stall += 1
            if stall == 2:
                # Try full LP
                g_full, _ = solve_lp(best_a, top_frac=1.0)
                if g_full is not None:
                    a_try, obj_try, t_try = line_search_blend(best_a, g_full)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        stall = 0
                        if verbose:
                            print(f"      LP full it{it}: obj={best_obj:.8f} **")
                    else:
                        best_a = fourier_perturbation(best_a, rng, strength=0.015 * stall)
            elif stall >= max_stall:
                if it > 20:
                    if verbose:
                        print(f"      LP stalled at it{it}, breaking")
                    break
                else:
                    best_a = fourier_perturbation(best_a, rng, strength=min(0.08, 0.015 * stall))
    
    return best_a
