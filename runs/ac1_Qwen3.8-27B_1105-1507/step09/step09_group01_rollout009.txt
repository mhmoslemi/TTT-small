import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_palindromic(a):
    """Symmetrize a to be palindromic."""
    n = len(a)
    half = (a[: (n + 1) // 2].copy() + a[::-1][: (n + 1) // 2].copy()) / 2.0
    result = np.zeros(n)
    result[: (n + 1) // 2] = half
    result[(n + 1) // 2:] = half[: (n - 1) // 2][::-1]
    return np.maximum(result, 0)

    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    if solver_order is None:
        solver_order = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]

    for solver in solver_order:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    dual_y = None
                    return g_val / s, float(M.value), dual_y
            elif prob.status == 'infeasible':
                return None, None, None
        except Exception:
            continue
    return None, None, None


def solve_lp_palindromic(a, M_target=None):
    """
    Solve LP with palindromic constraint on g.
    a is assumed to be palindromic.
    Variables: g[0], ..., g[m] where m = (n-1)//2, and g[n-1-j] = g[j].
    """
    n = len(a)
    a_norm = a / np.sum(a)
    half_len = (n + 1) // 2

    # Build convolution for palindromic g
    # conv(a, g)[k] = sum_j a[j] * g[k-j]
    # Since both a and g are palindromic, conv is palindromic with center at n-1
    # We need constraints for k = 0, ..., n-1

    # For palindromic g, the free variables are g[0..half_len-1]
    # g[j] for j >= half_len is determined: g[j] = g[n-1-j]

    # Build matrix A_p such that A_p @ g_half = conv(a, g)[:n]
    # conv(a,g)[k] = sum_j a[j] * g[k-j]
    # g[k-j] = g_half[min(k-j, n-1-(k-j))] if 0 <= k-j < n

    A_p = np.zeros((n, half_len))
    for k in range(n):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                # g[idx] = g_half[min(idx, n-1-idx)]
                phi = min(idx, n - 1 - idx)
                if phi < half_len:
                    A_p[k, phi] += a_norm[j]

    g_half = cp.Variable(half_len, nonneg=True)
    M = cp.Variable()

    # sum(g) = 1: sum of all g entries = 2 * sum(g_half[:half_len]) - (middle if n odd)
    if n % 2 == 1:
        # g has a middle element at position (n-1)//2 = half_len - 1
        # sum(g) = 2 * sum(g_half[:half_len-1]) + g_half[half_len-1] = 1
        sum_constraint = 2 * cp.sum(g_half[:half_len - 1]) + g_half[half_len - 1] == 1
    else:
        # sum(g) = 2 * sum(g_half) = 1
        sum_constraint = 2 * cp.sum(g_half) == 1

    constraints = [A_p @ g_half <= M, sum_constraint]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g_half.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gh = np.maximum(g_half.value, 0)
                # Expand to full palindromic vector
                g_full = np.zeros(n)
                g_full[:half_len] = gh
                g_full[half_len:] = gh[:n - half_len][::-1]
                s = float(np.sum(g_full))
                if s > 1e-10:
                    return g_full / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue
    return None, None


def solve_lp_with_dual(a, M_target=None):
    """Solve LP and attempt to extract dual information."""
    n = len(a)
    a_norm = a / np.sum(a)

    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    conv_constraints = [A @ g <= M]
    constraints = conv_constraints + [cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Try to get dual values
                    dual_y = None
                    try:
                        for cc in conv_constraints:
                            if cc.dual_value is not None:
                                dual_y = np.array(cc.dual_value)
                                break
                    except Exception:
                        pass
                    return g_val, float(M.value), dual_y
        except Exception:
            continue
    return None, None, None


def golden_section_search(f, a_lo, a_hi, tol=0.01, max_iter=40):
    """Golden section search for minimum of f on [a_lo, a_hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if abs(a_hi - a_lo) < tol:
            break
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = f(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = f(d)
    return (a_lo + a_hi) / 2.0


def lp_polish_aggressive(a, time_budget=200.0, verbose=False, seed=42, allow_full_replace=True):
    """
    LP polishing with aggressive full-replacement and adaptive blending.
    Key improvement: try t=1 first, fall back to golden section if needed.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    no_improve = 0
    M_prev = None

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed

        # Solve LP
        g_lp, M_lp, _ = solve_lp_standard(best_a)
        if g_lp is None:
            break

        # Check if full replacement helps
        best_t = 0.5
        best_blend_obj = compute_obj(best_a)

        if allow_full_replace:
            obj_full = compute_obj(g_lp)
            if obj_full < best_obj - 1e-12:
                # Full replacement is better
                best_obj = obj_full
                best_a = g_lp.copy()
                iter_count += 1
                no_improve = 0
                if verbose:
                    print(f"    LP-full it{iter_count}: obj={best_obj:.8f}, M={M_lp:.6f}, el={elapsed:.0f}s")
                continue

        # Golden section for blending
        def blend_obj(t):
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return 1e10
            return compute_obj(a_try / s_try)

        t_opt = golden_section_search(blend_obj, 0.05, 0.95, tol=0.02, max_iter=25)
        a_try = (1.0 - t_opt) * best_a + t_opt * g_lp
        a_try = np.maximum(a_try, 0)
        a_try = a_try / a_try.sum()
        obj_try = compute_obj(a_try)
        iter_count += 1

        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            no_improve = 0
            if verbose and iter_count % 3 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={t_opt:.3f}, M={M_lp:.6f}, el={elapsed:.0f}s")
        else:
            no_improve += 1
            if no_improve == 2:
                # Try tighter constraint
                if M_lp is not None:
                    M_tight = M_lp * 0.95
                    g_tight, _ = solve_lp_standard(best_a, M_target=M_tight)
                    if g_tight is not None:
                        obj_tight = compute_obj(g_tight)
                        if obj_tight < best_obj - 1e-13:
                            best_obj = obj_tight
                            best_a = g_tight.copy()
                            no_improve = 0
                            if verbose:
                                print(f"    LP-tight it{iter_count}: obj={best_obj:.8f}")
            elif no_improve >= 5:
                # Perturb and restart
                scale = 0.01 * min(no_improve, 10)
                best_a = best_a * np.exp(rng.normal(0, scale, n))
                best_a = np.maximum(best_a, 0)
                best_a = best_a / best_a.sum()
                best_obj = compute_obj(best_a)
                no_improve = 0
                if verbose:
                    print(f"    Perturb it{iter_count}: scale={scale:.4f}, obj={best_obj:.8f}")
                if iter_count > 80 and remaining < 30:
                    break

        # Early termination check
        if remaining < 10:
            break

    return best_a


def lp_polish_palindromic(a, time_budget=200.0, verbose=False, seed=42):

        if remaining < 10:
            break

    return best_a


def peak_shave_gradient(a, time_budget=100.0, K=8, verbose=False):
    """
    Peak shaving: iteratively reduce the top peaks of conv(a,a)
    using gradient information.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 5.0 / np.sqrt(n)
    momentum = 0.5
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        top_idx = np.argsort(conv)[::-1][:K]

        # Weight peaks by their excess over a target
        target = max_val * 0.9
        excess = np.maximum(conv[top_idx] - target, 0)
        if np.max(excess) < 1e-12:
            break
        weights = excess / np.sum(excess)

        # Compute gradient: grad_j of (a*a)[k] = a[k-j] (for valid j)
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-10:
                    break
            else:
                lr *= 0.997

        it += 1
        if verbose and it % 500 == 0:
            print(f"    PeakShave it{it}: obj={best_obj:.8f}, lr={lr:.6f}")

    return best_a


def smoothed_gradient_descent(a, time_budget=60.0, tau_init=5.0, tau_end=200.0, verbose=False):
    """Nesterov accelerated GD on smoothed max objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(y, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)

        if it % 10000 == 0 and verbose:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 3:
        a = np.zeros(n)
        for c in [n // 3, 2 * n // 3]:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * (n // 8) ** 2))
        a = a / a.sum()
    elif mode == 4:
        # Triangle-like with elevated center
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2 * x - 1)
        a = a / a.sum()
    elif mode == 5:
        # Cosine bump
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    else:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    return a

                                    seed=100, allow_full_replace=True)

        if remaining < 80:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)

        # Quick GD
        alloc1 = min(remaining * 0.1, 30.0)
        if alloc1 > 10:
            a_try = smoothed_gradient_descent(a_try, time_budget=alloc1,
                                              tau_init=10.0, tau_end=150.0)

        # Palindromic LP (quick)
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 60.0)
        if alloc2 > 15:
            a_try_pal = lp_polish_palindromic(a_try, time_budget=alloc2, seed=n_try)
            obj_pal_try = compute_obj(a_try_pal)
            if obj_pal_try < compute_obj(a_try):
                a_try = a_try_pal

        # Standard LP
        remaining = deadline - time.monotonic()
        alloc3 = min(remaining * 0.2, 80.0)
        if alloc3 > 15:
            a_try = lp_polish_aggressive(a_try, time_budget=alloc3, seed=n_try + 500,
                                         allow_full_replace=True)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Perturbation + Re-optimization ===
    print(f"\n=== Phase 6: Perturbation + Re-optimization ===")
    remaining = deadline - time.monotonic()

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        scales = [0.008, 0.015, 0.025, 0.04, 0.06, 0.1]
        scale = scales[r]
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        # Try both palindromic and standard
        alloc = min(remaining * 0.2, 80.0)
        if alloc > 15:
            # Standard LP
            a_p1 = lp_polish_aggressive(a_pert, time_budget=alloc * 0.6,
                                        seed=2000 + r, allow_full_replace=True)
            obj_p1 = compute_obj(a_p1)

            # Palindromic LP
            a_p2 = lp_polish_palindromic(a_pert, time_budget=alloc * 0.4,
                                         seed=3000 + r)
            obj_p2 = compute_obj(a_p2)

            if obj_p1 < best_score:
                best_score = obj_p1
                best_seq = a_p1.copy()
                print(f"  Perturb {r} (scale={scale:.4f}) LP: obj={best_score:.8f} **")
            elif obj_p2 < best_score:
                best_score = obj_p2
                best_seq = a_p2.copy()
                print(f"  Perturb {r} (scale={scale:.4f}) PalLP: obj={best_score:.8f} **")
            else:
                print(f"  Perturb {r} (scale={scale:.4f}): best={min(obj_p1, obj_p2):.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Alternate between palindromic and standard LP
        alloc_total = min(remaining * 0.7, 200.0)
        num_rounds = 3
        per_round = alloc_total / num_rounds

        for rd in range(num_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            if rd % 2 == 0:
                a_try = lp_polish_aggressive(best_seq, time_budget=per_round,
                                             verbose=True, seed=5000 + rd)
            else:
                a_try = lp_polish_palindromic(best_seq, time_budget=per_round,
                                              verbose=True, seed=6000 + rd)

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Final round {rd}: obj={best_score:.8f} **")

        # Quick peak shave at the end
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_ps = peak_shave_gradient(best_seq, time_budget=min(remaining * 0.3, 30.0), K=12)
            obj_ps = compute_obj(a_ps)
            if obj_ps < best_score:
                best_score = obj_ps
                best_seq = a_ps.copy()
                print(f"  Final PeakShave: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
