    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64)
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu * rho > (cssv - 1.0)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_val = idx[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient_fft(a, tau, active_set=None):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT.
    Optionally restricts smoothing to active_set positions.
    Returns (gradient, convolution values).
    """
    n = len(a)
    conv = compute_conv(a)
    if active_set is not None and len(active_set) < len(conv):
        # Only smooth over active set
        vals = conv[active_set]
        max_conv = float(np.max(vals))
        exp_vals = np.exp(tau * (vals - max_conv))
        p = np.zeros(len(conv))
        p[active_set] = exp_vals / float(np.sum(exp_vals))
    else:
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def build_conv_matrix(a_norm):
    """Build matrix A_mat (2n-1 x n) such that A_mat @ g = convolve(a_norm, g)."""
    n = len(a_norm)
    num_rows = 2 * n - 1
    A_mat = np.zeros((num_rows, n))
    for j in range(n):
        start = j
        end = j + n
        A_mat[start:end, j] = a_norm
    return A_mat


def solve_lp_fast(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses scipy HiGHS solver directly for speed.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))

    A_mat = build_conv_matrix(a_norm)
    num_constraints = 2 * n - 1

    # x = [M, g_0, ..., g_{n-1}]
    # minimize M
    # s.t. A_mat @ g - M <= 0  =>  [-1, A_mat] @ x <= 0
    #      sum(g) = 1           =>  [0, 1, ..., 1] @ x = 1
    #      M free, g >= 0

    A_ub = np.column_stack([np.full(num_constraints, -1.0), A_mat])
    b_ub = np.zeros(num_constraints)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0])

    c = np.zeros(n + 1)
    c[0] = 1.0

    bounds = [(None, None)] + [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            g_val = np.maximum(res.x[1:], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                g_val = g_val / s
                M_val = float(res.x[0])
                return g_val, M_val
    except Exception:
        pass
    return None, None


def solve_lp_with_active_set(a, active_set):
    """Solve LP restricted to active_set constraints."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    num_constraints = len(active_set)

    A_mat = build_conv_matrix(a_norm)
    A_active = A_mat[active_set, :]

    A_ub = np.column_stack([np.full(num_constraints, -1.0), A_active])
    b_ub = np.zeros(num_constraints)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0])

    c = np.zeros(n + 1)
    c[0] = 1.0

    bounds = [(None, None)] + [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            g_val = np.maximum(res.x[1:], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                g_val = g_val / s
                M_val = float(res.x[0])
                # Verify all constraints satisfied
                full_conv = compute_conv(a_norm)
                if float(np.max(np.abs(full_conv * g_val - np.sum(a_norm * g_val)))) < 1.0:
                    # Check full constraint
                    conv_check = np.zeros(2 * n - 1)
                    for k in range(2 * n - 1):
                        conv_check[k] = sum(a_norm[k - i] * g_val[i]
                                           for i in range(n) if 0 <= k - i < n)
                    if float(np.max(conv_check)) <= M_val + 1e-6:
                        return g_val, M_val
    except Exception:
        pass
    return None, None


def nesterov_gd(a, time_budget=100.0, tau_init=5.0, tau_rate=1.03,
                 tau_max=2000.0, verbose=False, seed=0):
    """
    Nesterov accelerated smoothed gradient descent with exponential tau.
    Includes periodic resets and active-set tracking.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    rng = np.random.default_rng(seed)
    tau = tau_init

    while time.monotonic() - start < time_budget:
        # Exponential tau increase
        tau = min(tau * tau_rate, tau_max)

        if it % 500 == 0 and verbose:
            elapsed = time.monotonic() - start
            print(f"      NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Determine active set: top positions in convolution
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        # Active set: positions within 1/tau of max
        threshold = max_conv - 3.0 / tau
        active = np.where(conv >= threshold)[0]

        g, _ = smoothed_gradient_fft(y, tau, active_set=active)

        # Barzilai-Borwein step size
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1

        # Decay learning rate
        lr = max(1e-9, lr * 0.9998)

        # Reset if stuck
        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)
            tau = tau_init
            best_it = it

    return best_a


def mirror_descent_step(a, grad, eta):
    """Multiplicative weights (mirror descent) step on the simplex."""
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lp_line_search(a, g_lp, num_trials=20):
    """
    Line search along direction (1-t)*a + t*g_lp.
    Uses golden section search for the optimal t.
    """
    best_t = 0.0
    best_obj = compute_obj(a)

    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    a_lo, a_hi = 0.0, 1.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    f_c = compute_obj((1 - c) * a + c * g_lp)
    f_d = compute_obj((1 - d) * a + d * g_lp)

    for _ in range(num_trials):
        if f_c < f_d:
            a_hi = d
            d = c
            f_d = f_c
            c = a_hi - gr * (a_hi - a_lo)
            f_c = compute_obj((1 - c) * a + c * g_lp)
        else:
            a_lo = c
            c = d
            f_c = f_d
            d = a_lo + gr * (a_hi - a_lo)
            f_d = compute_obj((1 - d) * a + d * g_lp)

    best_t = (a_lo + a_hi) / 2
    best_a = (1 - best_t) * a + best_t * g_lp
    best_a = np.maximum(best_a, 0)
    s = float(np.sum(best_a))
    if s > 1e-10:
        best_a = best_a / s
    best_obj = compute_obj(best_a)

    return best_a, best_obj, best_t


def peak_aware_perturbation(a, conv, scale=0.05, seed=0):
    """Perturb with noise weighted by autocorrelation peak contributions."""
    n = len(a)
    rng = np.random.default_rng(seed)
    top_k = np.argsort(conv)[::-1][:min(20, len(conv))]

    contrib = np.zeros(n)
    for k in top_k:
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        js = np.arange(i_start, i_end + 1)
        ks = k - js
        mask = (ks >= 0) & (ks < n)
        contrib[js[mask]] += a[js[mask]] * a[ks[mask]]

    max_c = contrib.max() if contrib.max() > 0 else 1.0
    contrib = contrib / max_c

    noise = rng.normal(0, 1, n) * scale * (0.5 + contrib)
    a_new = a * np.exp(noise)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def gp_perturbation(a, seed=0, scale=0.03):
    """Gaussian process perturbation with smooth kernel."""
    n = len(a)
    rng = np.random.default_rng(seed)
    # Generate smooth perturbation via random Fourier features
    num_freqs = max(5, n // 50)
    freqs = rng.uniform(0, 5, num_freqs)
    phases = rng.uniform(0, 2 * np.pi, num_freqs)
    x = np.linspace(0, 1, n)

    smooth_noise = np.zeros(n)
    for f, p in zip(freqs, phases):
        smooth_noise += np.sin(2 * np.pi * f * x + p) / np.sqrt(num_freqs)

    a_new = a * np.exp(scale * smooth_noise)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def combined_optimizer(a, time_budget=300.0, verbose=False, seed=0):
    """
    Combined optimizer alternating between:
    - Nesterov smoothed GD (fast, many steps)
    - LP solve + line search (slow, global direction)
    - Mirror descent steps (geometry-aware)
    - Periodic perturbation (escape local minima)
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    gd_phase_time = 20.0
    lp_count = 0
    pert_count = 0
    consecutive_lp_fails = 0
    last_improve = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break

        # GD phase
        gd_alloc = min(gd_phase_time, remaining * 0.3)
        if gd_alloc > 10:
            tau_init = 5.0 + 2.0 * (lp_count // 10)
            a_sgd = nesterov_gd(a, time_budget=gd_alloc, tau_init=tau_init,
                                tau_rate=1.02, tau_max=1000.0,
                                verbose=False, seed=seed + lp_count)
            obj_sgd = compute_obj(a_sgd)
            if obj_sgd < best_obj:
                best_obj = obj_sgd
                best_a = a_sgd.copy()
                a = a_sgd.copy()
                last_improve = time.monotonic() - start

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break

        # LP phase with line search
        try:
            g_lp, M_lp = solve_lp_fast(a)
            if g_lp is not None:
                lp_count += 1
                # Line search
                a_ls, obj_ls, t_ls = lp_line_search(a, g_lp, num_trials=15)
                if obj_ls < best_obj:
                    best_obj = obj_ls
                    best_a = a_ls.copy()
                    a = a_ls.copy()
                    last_improve = time.monotonic() - start
                    consecutive_lp_fails = 0
                    if verbose and lp_count % 5 == 0:
                        print(f"      LP#{lp_count}: obj={best_obj:.8f}, t={t_ls:.3f}, M={M_lp:.6f}")
                else:
                    # Try smaller steps
                    for t_try in [0.05, 0.1, 0.15, 0.2]:
                        a_try = (1 - t_try) * a + t_try * g_lp
                        a_try = np.maximum(a_try, 0)
                        s_try = float(np.sum(a_try))
                        if s_try > 1e-10:
                            a_try = a_try / s_try
                            obj_try = compute_obj(a_try)
                            if obj_try < best_obj:
                                best_obj = obj_try
                                best_a = a_try.copy()
                                a = a_try.copy()
                                last_improve = time.monotonic() - start
                                break
                    else:
                        consecutive_lp_fails += 1
            else:
                consecutive_lp_fails += 1
        except Exception as e:
            consecutive_lp_fails += 1

        # Perturbation if stuck
        if consecutive_lp_fails >= 3:
            pert_count += 1
            conv = compute_conv(a)
            if pert_count % 2 == 0:
                a = peak_aware_perturbation(a, conv, scale=0.03 + 0.01 * min(consecutive_lp_fails, 5),
                                            seed=seed + pert_count)
            else:
                a = gp_perturbation(a, seed=seed + pert_count * 7, scale=0.02)
            # Blend back towards best
            blend = min(0.5, 0.1 * consecutive_lp_fails)
            a = (1 - blend) * a + blend * best_a
            a = np.maximum(a, 0)
            a = a / a.sum()
            consecutive_lp_fails = 0

        # Periodic mirror descent step
        if lp_count % 5 == 0 and remaining > 20:
            conv = compute_conv(a)
            max_conv = float(np.max(conv))
            # Simple subgradient: indicator of max position
            grad = np.zeros(n)
            max_pos = np.argmax(conv)
            for i in range(n):
                j = max_pos - i
                if 0 <= j < n:
                    grad[i] = a[j]
            eta_md = 0.1 / np.sqrt(n)
            a_md = mirror_descent_step(a, grad, eta_md)
            obj_md = compute_obj(a_md)
            if obj_md < best_obj:
                best_obj = obj_md
                best_a = a_md.copy()
                a = a_md.copy()
                last_improve = time.monotonic() - start

        if verbose and (time.monotonic() - start) % 30 < gd_alloc:
            elapsed = time.monotonic() - start
            print(f"      Combined: obj={best_obj:.8f}, LPs={lp_count}, t={elapsed:.0f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations with structural priors."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        # Gaussian bump
        sigma = rng.uniform(0.1, 0.3)
        center = rng.uniform(0.2, 0.8)
        a = np.exp(-((x - center) ** 2) / (2 * sigma ** 2))
    elif mode == 3:
        # Cosine taper (Hann)
        a = np.sin(np.pi * x) ** 2
    elif mode == 4:
        # Fejer (triangle squared)
        a = np.maximum(0, 1 - np.abs(x - 0.5) * 2) ** 2
    elif mode == 5:
        # Multi-frequency
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
    elif mode == 6:
        # Power law
        alpha = rng.uniform(0.5, 2.0)
        a = (x + 0.01) ** alpha
    elif mode == 7:
        # Two bumps
        c1, c2 = 0.25, 0.75
        w = max(1, n // 16)
        a = np.exp(-((np.arange(n) - c1 * n) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2 * n) ** 2) / (2 * w ** 2))
    elif mode == 8:
        # Smooth random (low-pass)
        a = rng.uniform(0, 1, n)
        k = max(3, n // 30)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
    elif mode == 9:
        # Raised cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif mode == 10:
        # Super-Gaussian
        p = rng.uniform(2, 6)
        a = np.abs(x - 0.5) ** (-p) + 0.1
    elif mode == 11:
        # Sinc-like
        f = rng.uniform(1, 3)
        a = np.sinc(f * (x - 0.5)) ** 2 + 0.01
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 0:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def subgradient_averaging(a, time_budget=100.0, num_iters=5000):
    """
    Subgradient method with Polyak averaging for the non-smooth objective.
    Periodically projects onto simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    a_avg = a.copy()
    step_t0 = 1.0
    it = 0

    while it < num_iters and (time.monotonic() - start) < time_budget:
        conv = compute_conv(a)
        max_pos = int(np.argmax(conv))
        max_val = conv[max_pos]

        # Subgradient of max_k (a*a)_k at the max position
        grad = np.zeros(n)
        for i in range(n):
            j = max_pos - i
            if 0 <= j < n:
                grad[i] += a[j]
                if i != j:
                    grad[j] += a[i]  # Symmetric contribution
        # Actually, d/d a_i of (a*a)[max_pos] = a[max_pos - i] + (i != max_pos-i ? a[i - max_pos] : 0)
        # Since (a*a)[k] = sum_i a[i]*a[k-i], d/d a_i = a[k-i] + (if k-i != i then a[i-k+... ])
        # More carefully: (a*a)[k] = sum_i a[i]*a[k-i]
        # d/d a_i = a[k-i] + (if i != k-i, i.e., 2i != k) a[k-2i+... ]
        # Wait: (a*a)[k] = sum_j a[j]*a[k-j]
        # d/d a_i = a[k-i] + (if i = k-j for some j, i.e., j = k-i) a[k-j'] where j' = k-i
        # = a[k-i] + a[k-i] if i != k-i, or just a[k-i] if i = k-i
        # Actually no. Let me be more careful.
        # (a*a)[k] = sum_j a[j] * a[k-j]
        # d/da_i [sum_j a[j]*a[k-j]] = a[k-i] (from j=i term) + sum_j delta(i, k-j)*a[j]
        # = a[k-i] + a[k-i] if k-i = i (i.e., k = 2i), else a[k-i]
        # Wait no: the second term is: d/da_i [a[k-j]] for each j where k-j = i, i.e., j = k-i
        # So it's a[k-i] (from the first factor when j=i) + a[k-i] (from the second factor when j=k-i)
        # But if i = k-i, these are the same term, so it's just a[k-i]
        # If i != k-i, it's 2*a[k-i]... no wait.
        # (a*a)[k] = sum_j a[j] * a[k-j]
        # The terms involving a[i] are: j=i gives a[i]*a[k-i], and j=k-i gives a[k-i]*a[i]
        # So if i != k-i: d/da_i = a[k-i] + a[k-i] = 2*a[k-i]
        # If i == k-i (k=2i): d/da_i = a[k-i] (only one term, a[i]*a[i], derivative is 2*a[i])
        # Actually for k=2i: (a*a)[k] contains a[i]*a[i] = a[i]^2, derivative is 2*a[i]
        # For i != k-i: derivative is a[k-i] + a[i]... no.
        # Let me just do it directly:
        grad = np.zeros(n)
        for i in range(n):
            j = max_pos - i
            if 0 <= j < n:
                grad[i] += a[j]
            j2 = i - max_pos
            if 0 <= j2 < n and j2 != j:
                grad[i] += a[j2]
        # Hmm, this is getting confusing. Let me just use the FFT-based gradient.
        # Actually, the simplest correct formula:
        # (a*a)[k] = sum_{i=0}^{n-1} a[i] * a[k-i] (with a[m]=0 for m outside [0,n-1])
        # d/da[i] (a*a)[k] = a[k-i] + (if k-i != i) a[i-k+... ]
        # No: d/da[i] sum_j a[j]*a[k-j]
        # = a[k-i] (from j=i: d a[i]/da[i] * a[k-i])
        # + sum_j a[j] * d(a[k-j])/da[i]
        # d(a[k-j])/da[i] = 1 if k-j = i, i.e., j = k-i
        # So second term = a[k-i] (from j=k-i)
        # Total: if i != k-i: a[k-i] + a[k-i] = 2*a[k-i]
        #         if i == k-i: a[k-i] (but the j=i and j=k-i terms are the same, so it's a[i] derivative of a[i]^2 = 2*a[i])
        # In both cases it's 2*a[k-i] when 0 <= k-i < n, unless...
        # Actually when i = k-i, the sum has the term a[i]*a[i] = a[i]^2, and d/da[i] = 2*a[i] = 2*a[k-i].
        # When i != k-i, there are two terms: a[i]*a[k-i] and a[k-i]*a[i], each contributing a[k-i] and a[i] respectively.
        # Wait no: from j=i: a[i]*a[k-i], derivative w.r.t. a[i] is a[k-i].
        # From j=k-i: a[k-i]*a[i], derivative w.r.t. a[i] is a[k-i].
        # So total is 2*a[k-i].
        # When i = k-i (i.e., k=2i): from j=i: a[i]*a[i], derivative is 2*a[i] = 2*a[k-i].
        # So in all cases: d/da[i] (a*a)[k] = 2*a[k-i] if 0 <= k-i < n, else 0.

        grad = np.zeros(n)
        for i in range(n):
            j = max_pos - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        # Step size
        step = step_t0 / np.sqrt(it + 1)
        a_new = a - step * grad
        a_new = project_simplex(a_new)

        # Running average
        a_avg = (it / (it + 1)) * a_avg + (1 / (it + 1)) * a_new

        obj_new = compute_obj(a_new)
        obj_avg = compute_obj(a_avg)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        if obj_avg < best_obj:
            best_obj = obj_avg
            best_a = a_avg.copy()

        a = a_new
        it += 1

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining fast LP, Nesterov GD, mirror descent,
    subgradient averaging, and diverse structural restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Phase 0: Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Initial Nesterov GD (quick) ===
    print(f"\n=== Phase 1: Initial GD ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.1, 90.0)
    if p1_budget > 20:
        a1 = nesterov_gd(best_seq, time_budget=p1_budget, tau_init=5.0,
                         tau_rate=1.03, tau_max=500.0, verbose=True, seed=seed)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj1:.8f}")

    # === Phase 2: Combined optimizer (main workhorse) ===
    print(f"\n=== Phase 2: Combined Optimizer ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.40, 500.0)
    if p2_budget > 30:
        a2 = combined_optimizer(best_seq, time_budget=p2_budget, verbose=True, seed=seed)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2: obj={best_score:.8f} **")
        else:
            print(f"  Phase2: obj={obj2:.8f}")

    # === Phase 3: Intensive LP polishing ===
    print(f"\n=== Phase 3: LP Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.20, 250.0)
    if p3_budget > 30:
        a3 = combined_optimizer(best_seq, time_budget=p3_budget, verbose=True, seed=seed + 500)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj3:.8f}")

        # Try from perturbed start
        remaining = deadline - time.monotonic()
        if remaining > 40:
            conv = compute_conv(best_seq)
            a_pert = peak_aware_perturbation(best_seq, conv, scale=0.04, seed=seed + 600)
            a_pert = 0.7 * a_pert + 0.3 * best_seq
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            a3b = combined_optimizer(a_pert, time_budget=min(remaining * 0.3, 80.0),
                                     verbose=False, seed=seed + 700)
            obj3b = compute_obj(a3b)
            if obj3b < best_score:
                best_score = obj3b
                best_seq = a3b.copy()
                print(f"  Phase3b (perturbed): obj={best_score:.8f} **")
            else:
                print(f"  Phase3b: obj={obj3b:.8f}")

    # === Phase 4: Subgradient averaging ===
    print(f"\n=== Phase 4: Subgradient Averaging ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.10, 100.0)
    if p4_budget > 20:
        a4 = subgradient_averaging(best_seq, time_budget=p4_budget, num_iters=10000)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4: obj={best_score:.8f} **")
        else:
            print(f"  Phase4: obj={obj4:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(500, n_current - 150),
        max(500, n_current - 50),
        n_current + 100,
        n_current + 200,
        n_current + 400,
        max(600, n_current + 600),
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2500 and abs(n - n_current) >= 80]
    n_options = n_options[:4]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        alloc = min(remaining * 0.25, 120.0)
        if alloc > 30:
            a_try = combined_optimizer(a_try, time_budget=alloc, verbose=False, seed=seed + n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse structural restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    restart_modes = [3, 4, 9, 1, 5, 11]  # Hann, Fejer, raised cosine, random, multi-freq, sinc
    for r, mode in enumerate(restart_modes):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.15 + 0.05 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.25, 100.0)
        if alloc > 20:
            a_r = combined_optimizer(a_r, time_budget=alloc, verbose=False, seed=seed + 2000 + r * 10)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau GD
    if remaining > 25:
        alloc = min(remaining * 0.3, 60.0)
        a_sgd = nesterov_gd(best_seq, time_budget=alloc, tau_init=50.0,
                            tau_rate=1.05, tau_max=3000.0, verbose=False, seed=seed + 5000)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Final GD: obj={best_score:.8f} **")
        else:
            print(f"  Final GD: obj={obj_sgd:.8f}")

    # Final combined
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.4, 80.0)
        a_final = combined_optimizer(best_seq, time_budget=alloc, verbose=False, seed=seed + 6000)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final Combined: obj={best_score:.8f} **")
        else:
            print(f"  Final Combined: obj={obj_final:.8f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
