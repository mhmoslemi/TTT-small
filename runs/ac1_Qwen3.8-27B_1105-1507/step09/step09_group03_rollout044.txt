"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP direction finding, Nesterov-accelerated smoothed gradient descent,
multiplicative weight updates, peak-targeted coordinate descent, and multi-scale
exploration. Key innovations: adaptive constraint reduction for LP, entropy-based
mirror descent, and hierarchical n exploration with warm starts.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def smoothed_max_grad(a, tau):
    """
    Compute gradient of (1/tau) * log(sum(exp(tau * conv(a,a)[k]))) via FFT.
    Returns (gradient, smoothed_value, conv).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    # Shift for numerical stability
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    smoothed_val = max_conv + np.log(float(np.sum(exp_vals))) / tau
    
    # Gradient: d/da of sum_k p_k * conv(a,a)[k] = 2 * sum_k p_k * conv(a, e_k)
    # = 2 * conv_p where p is weighted
    # Using FFT: ifft(fft(p) * conj(fft(a))) gives cross-correlation
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, smoothed_val, conv, p


def lp_direction(a, active_indices=None, solver_list=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for k in active_indices (or all),
    sum(g) = 1, g >= 0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    if active_indices is None:
        active_indices = np.arange(2 * n - 1)
    
    A = build_conv_matrix(a_norm)
    A_active = A[active_indices, :]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_active @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def golden_section_search(a, direction, n_evals=8):
    """Golden section line search for a + t * (direction - a) on simplex."""
    phi = (np.sqrt(5) - 1) / 2
    lo = 0.0
    hi = 1.0
    f_mid = np.inf
    t_mid = 0.5
    
    for _ in range(n_evals):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)
        
        a1 = (1 - t1) * a + t1 * direction
        a1 = np.maximum(a1, 0)
        s1 = float(np.sum(a1))
        f1 = compute_obj(a1) if s1 > 1e-10 else np.inf
        
        a2 = (1 - t2) * a + t2 * direction
        a2 = np.maximum(a2, 0)
        s2 = float(np.sum(a2))
        f2 = compute_obj(a2) if s2 > 1e-10 else np.inf
        
        if f1 < f2:
            hi, f_mid, t_mid = t2, f1, t1
        else:
            lo, f_mid, t_mid = t1, f2, t2
    
    t_best = (lo + hi) / 2
    a_best = (1 - t_best) * a + t_best * direction
    a_best = np.maximum(a_best, 0)
    s_best = float(np.sum(a_best))
    if s_best > 1e-10:
        a_best = a_best / s_best
    return a_best, f_mid


def lp_polish(a, time_budget=100.0, verbose=False):
    """LP-based optimization with active set reduction and line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        # Find active constraints
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        # Use constraints where conv > 0.5 * max (reduces constraint count)
        threshold = 0.5 * max_conv
        active = np.where(conv >= threshold)[0]
        
        # Solve LP
        g_lp, M_lp = lp_direction(a, active_indices=active)
        if g_lp is None:
            # Try with all constraints
            g_lp, M_lp = lp_direction(a)
            if g_lp is None:
                break
        
        # Line search
        a_new, f_new = golden_section_search(a, g_lp, n_evals=6)
        
        if f_new < best_obj - 1e-13:
            best_obj = f_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, M={M_lp:.6f}, |S|={len(active)}")
        else:
            stall += 1
            # Perturb
            scale = 0.02 + 0.01 * min(stall, 10)
            a = best_a * np.exp(rng.normal(0, min(scale, 0.1), n))
            a = np.maximum(a, 0)
            sa = float(np.sum(a))
            if sa > 1e-10:
                a = a / sa
            else:
                a = best_a.copy()
            if stall > 25:
                break
        
        it += 1
    
    return best_a


def nesterov_smoothed(a, time_budget=200.0, tau_init=20.0, tau_max=500.0, verbose=False):
    """Nesterov-accelerated gradient descent on log-sum-exp smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.95
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_max / tau_init) ** (progress ** 0.6)
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy < 1e-10:
            y = best_a.copy()
        else:
            y = y / sy
        
        # Compute smoothed gradient
        g, smoothed_val, conv, p = smoothed_max_grad(y, tau)
        
        # Gradient step
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / sn
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        lr = max(1e-9, lr * 0.9999)
        
        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it
        
        if verbose and it % 1000 == 0:
            print(f"    NS it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.4f}")
    
    return best_a


def multiplicative_weight_descent(a, time_budget=100.0, eta_init=1.0, verbose=False):
    """
    Multiplicative weight / mirror descent with entropy regularizer.
    a_i <- a_i * exp(-eta * g_i) / sum_j a_j * exp(-eta * g_j)
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    # Avoid zeros (multiply by small epsilon)
    a = a + 1e-10
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = eta_init
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_k = int(np.argmax(conv))
        
        # Subgradient of max conv at a: 2 * D_{max_k} * a
        # D_k is the matrix for conv_k
        # conv_k(a) = sum_i a[i] * a[i+k-n+1] (for valid i)
        # d conv_k / da[j] = a[j+k-n+1] + a[j-(k-n+1)] (for valid indices)
        # = a[(j+k) mod (2n-1)] type thing... 
        # Actually: conv_k(a) = sum_i a[i] * a[i+k-(n-1)] 
        # Let's just use the FFT-based gradient
        
        g, _, _, _ = smoothed_max_grad(a, tau=50.0)
        
        # Multiplicative update
        # a_i <- a_i * exp(-eta * g_i)
        log_new = np.log(a + 1e-30) - eta * g
        # Clip for numerical stability
        log_new = np.clip(log_new, -50, 50)
        a_new = np.exp(log_new)
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        eta *= 0.999
        
        if it - best_it > 3000:
            a = best_a.copy()
            eta = eta_init * 0.5
            best_it = it
        
        if verbose and it % 500 == 0:
            print(f"    MWD it{it}: obj={best_obj:.8f}, eta={eta:.4f}")
    
    return best_a


def peak_cgshave(a, time_budget=60.0, verbose=False):
    """
    Peak shaving: for the dominant peak, find mass transfers that reduce it.
    Uses efficient contribution analysis.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))
        
        # Find contributing pairs for position k_max
        # conv[k] = sum_i a[i] * a[k-i] where 0<=i<n, 0<=k-i<n
        i_min = max(0, k_max - (n - 1))
        i_max = min(n - 1, k_max)
        
        if i_min > i_max:
            break
        
        # Compute contributions
        indices = np.arange(i_min, i_max + 1)
        partners = k_max - indices
        contribs = a[indices] * a[partners]
        
        # Sort by contribution (descending)
        order = np.argsort(-contribs)
        
        improved = False
        # Try moving mass from top contributors
        for idx in order[:5]:
            src = int(indices[idx])
            dst_part = int(partners[idx])
            
            if src == dst_part:
                continue
            
            # Try moving a small amount from src to a random target
            for delta_frac in [0.03, 0.01, 0.005]:
                delta = delta_frac * a[src]
                if delta >= a[src] * 0.5:
                    continue
                
                # Target: try to move to a position that doesn't contribute to the peak
                targets = [
                    rng.integers(0, n),
                    (src + n // 3) % n,
                    (src + 2 * n // 3) % n,
                    n - 1 - src
                ]
                
                for tgt in targets:
                    tgt = int(tgt) % n
                    if tgt == src:
                        continue
                    
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[tgt] += delta
                    if a_try[src] < 0:
                        continue
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
        
        if verbose and it % 10 == 0:
            print(f"    PCD it{it}: obj={best_obj:.8f}, k_max={k_max}")
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(0.01, 0.99, n)
        a = 1.0 / np.sqrt(x * (1 - x))
        a = a / a.sum()
    elif mode == 4:
        # Sine power
        x = np.linspace(0, 1, n)
        a = np.abs(np.sin(np.pi * x)) ** 3
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 5:
        # Cosine bell
        x = np.linspace(-1, 1, n)
        a = (1 + x) ** 2
        a = a / a.sum()
    else:
        # Random smooth
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.sin(2 * np.pi * rng.uniform(1, 4) * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    return a
