def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)

        return np.ones(n) / n
    rho = rho[-1]

    return np.maximum(v - theta, 0.0)

In practice, we often use gamma-relaxation: x_{k+1} = (1-gamma) * x_k + gamma * (Anderson extrapolation)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP fixed-point iteration with extrapolation, frequency-domain IAM iteration,
FISTA polishing with adaptive smoothing, and multi-n exploration.
Key innovations over previous approach: Anderson-accelerated LP steps, IAM frequency

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build A (2n-1 x n) where A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for l in range(n):
        max_i = min(n, 2 * n - 1 - l)
        A[l:l + max_i, l] = a[:max_i]
    return A


def solve_lp(a):
    """Solve: min M s.t. A(a)@g <= M, sum(g)=1, g>=0. Returns (g_opt, M_opt) or (None, None)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)_k."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    size = 1
    while size < 3 * n:
        size <<= 1
    A_fft = fft(a, size)
    P_fft = fft(p, size)
    grad = 2.0 * ifft(P_fft * np.conj(A_fft)).real[:n]
    return grad, conv, p


def fista_refine(a, time_budget, tau_init=50.0, tau_end=1000.0):
    """FISTA with log-sum-exp smoothing, adaptive step, and restarts."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    beta = 0.85
    v = np.zeros(n)
    lr = 3.0 / np.sqrt(n)
    it = 0
    best_it = 0
    patience = 8000

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init * (tau_end / tau_init) ** progress

        # Nesterov momentum
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        grad, conv, p = smoothed_gradient(y, tau)

        # Adaptive learning rate
        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-15:
            ideal_lr = 1.0 / g_norm
            lr = 0.8 * lr + 0.2 * min(ideal_lr, 5.0 / np.sqrt(n))

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
            a = a_new
            it += 1
            continue
        a_new = a_new / s_new

        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        # Slow decay
        lr = max(1e-9, lr * 0.99995)

        # Restart if stuck
        if it - best_it > patience:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n)
            beta = min(0.95, beta * 1.05)
            best_it = it

    return best_a


def iam_iteration(a, time_budget, alpha=0.7, n_iters_max=50000):
    """
    Iterative Amplitude Modification (IAM) in frequency domain.
    Alternates between:
      - Time domain: enforce non-negativity and sum=1
      - Frequency domain: compress |FFT(a)| dynamic range to flatten autocorrelation
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    # Use a larger size to avoid circular effects
    N = size * 2

    it = 0
    stall_count = 0

    while time.monotonic() - start < time_budget and it < n_iters_max:
        # Frequency domain step: compress |F|
        F = fft(a, N)
        mag = np.abs(F)
        phase = np.angle(F)

        # Compute current autocorrelation peak
        # Target: make |F| as flat as possible
        # Use power compression: mag_new = mag^alpha (alpha < 1 compresses)
        mag_min = float(np.min(mag[mag > 1e-10])) if np.any(mag > 1e-10) else 1e-10
        mag_max = float(np.max(mag))
        if mag_max > mag_min * 1e-10:
            # Normalize then apply alpha
            mag_norm = mag / mag_max
            mag_compressed = np.power(np.maximum(mag_norm, 1e-10), alpha)
            mag_new = mag_compressed * mag_max
        else:
            mag_new = mag

        # Reconstruct with new magnitude, same phase
        F_new = mag_new * np.exp(1j * phase)
        F_new[0] = abs(F_new[0])  # Ensure DC is real positive

        a_freq = ifft(F_new, N).real[:n]

        # Time domain step: enforce non-negativity and sum=1
        a_time = np.maximum(a_freq, 0)
        s_time = float(np.sum(a_time))
        if s_time < 1e-10:
            # Reset
            a_time = best_a.copy()
        else:
            a_time = a_time / s_time

        # Check improvement
        obj_new = compute_obj_normalized(a_time)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_time.copy()
            a = a_time
            stall_count = 0
        else:
            # Blend to maintain progress
            blend = 0.3
            a_blend = (1 - blend) * best_a + blend * a_time
            a_blend = np.maximum(a_blend, 0)
            s_blend = float(np.sum(a_blend))
            if s_blend > 1e-10:
                a_blend = a_blend / s_blend
                obj_blend = compute_obj_normalized(a_blend)
                if obj_blend < best_obj:
                    best_obj = obj_blend
                    best_a = a_blend.copy()
                a = a_blend
            else:
                a = best_a.copy()
            stall_count += 1

        # Adapt alpha: increase compression when stalled
        if stall_count > 500:
            alpha = max(0.3, alpha - 0.01)
            stall_count = 0

        it += 1

    return best_a

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:

        alphas = [0.5, 0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.3, 1.5]

            # Build G matrix (m x n)
            G_mat = np.array(G_hist)  # (m, n)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 0.01))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        a = np.ones(n) * 0.5
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        a = np.exp(-np.abs(np.arange(n) - n / 2) / (n / 6))
        a = a / a.sum()
    elif mode == 5:
        # Bathtub shape (inverse of what we want in autocorrelation)
        x = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * x * (1 - x) * 4
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            best_seq = a_lp.copy()

    # === Phase 2: FISTA Polishing ===
    print(f"\n=== Phase 2: FISTA Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 200.0)
    if p2_budget > 20:
        a_fista = fista_refine(best_seq, time_budget=p2_budget, tau_init=30.0, tau_end=800.0)
        obj_fista = compute_obj_normalized(a_fista)
        if obj_fista < best_score:
            best_score = obj_fista
            best_seq = a_fista.copy()
            print(f"  FISTA: obj={best_score:.8f} **")
        else:
            print(f"  FISTA: obj={obj_fista:.8f}")

    # === Phase 3: IAM Frequency-Domain Iteration (novel) ===
    print(f"\n=== Phase 3: IAM Frequency-Domain Iteration ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    if p3_budget > 20:
        a_iam = iam_iteration(best_seq, time_budget=p3_budget, alpha=0.75)
        obj_iam = compute_obj_normalized(a_iam)
        if obj_iam < best_score:
            best_score = obj_iam
            best_seq = a_iam.copy()
            print(f"  IAM: obj={best_score:.8f} **")
        else:
            print(f"  IAM: obj={obj_iam:.8f}")

        # Try IAM from a perturbed start
        remaining = deadline - time.monotonic()
        if remaining > 40:
            a_pert = best_seq + rng.normal(0, 0.01, len(best_seq))
            a_pert = np.maximum(a_pert, 0)
            s_p = float(np.sum(a_pert))
            if s_p > 1e-10:
                a_pert = a_pert / s_p
            alloc = min(remaining * 0.3, 60.0)
            if alloc > 15:
                a_iam2 = iam_iteration(a_pert, time_budget=alloc, alpha=0.6)
                obj_iam2 = compute_obj_normalized(a_iam2)
                if obj_iam2 < best_score:
                    best_score = obj_iam2
                    best_seq = a_iam2.copy()
                    print(f"  IAM2: obj={best_score:.8f} **")
                else:
                    print(f"  IAM2: obj={obj_iam2:.8f}")

    # === Phase 4: More LP Iteration (second round) ===
    print(f"\n=== Phase 4: Second LP Round ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.2, 200.0)
    if p4_budget > 20:
        a_lp2 = lp_iteration_with_extrapolation(best_seq, time_budget=p4_budget, verbose=True)
        obj_lp2 = compute_obj_normalized(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  LP round 2: obj={best_score:.8f} **")
        else:
            print(f"  LP round 2: obj={obj_lp2:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = [n_best - 100, n_best + 100, n_best + 200, n_best + 400,
                 1000, 1200, 1400, 1600]
    n_options = sorted(set([n for n in n_options if 600 <= n <= 2000 and abs(n - n_best) > 50]))
    n_options = n_options[:5]
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick LP
        alloc1 = min(remaining * 0.25, 80.0)
        if alloc1 > 20:
            a_try = lp_iteration_with_extrapolation(a_try, time_budget=alloc1)

        # Quick FISTA
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.1, 40.0)
        if alloc2 > 10:
            a_try = fista_refine(a_try, time_budget=alloc2, tau_init=50.0, tau_end=500.0)

        obj_try = compute_obj_normalized(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = (r * 3 + 1) % 6
        a_r = diverse_init(n_cur, rng, mode=mode)
        # Blend with best
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        alloc = min(remaining * 0.2, 80.0)
        if alloc > 20:
            # Short LP
            half = alloc * 0.6
            a_r = lp_iteration_with_extrapolation(a_r, time_budget=half)
            remaining = deadline - time.monotonic()
            alloc2 = min(remaining * 0.1, 30.0)
            if alloc2 > 10:
                a_r = fista_refine(a_r, time_budget=alloc2, tau_init=50.0, tau_end=400.0)

            obj_r = compute_obj_normalized(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        alloc = min(remaining * 0.3, 100.0)
        a_final = lp_iteration_with_extrapolation(best_seq, time_budget=alloc)
        obj_final = compute_obj_normalized(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 80.0)
        a_f = fista_refine(best_seq, time_budget=alloc, tau_init=100.0, tau_end=1500.0)
        obj_f = compute_obj_normalized(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final FISTA: obj={best_score:.8f} **")
        else:
            print(f"  Final FISTA: obj={obj_f:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_normalized(best_seq)

    # Scale to valid range and validate
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
