But (g*g)[k] is quadratic in g, so this is no longer an LP. We could linearize it by also fixing g (i.e., use the previous g as a linearization point), leading to an alternating scheme.

**Improvement: Use the KKT conditions**

At the optimum of the original problem, there exist Lagrange multipliers λ[k] >= 0 for the constraints c[k] <= M such that:
- c[k] = M for active constraints (λ[k] > 0)
- The gradient of L = M + sum_k λ[k](c[k] - M) + μ(sum(a) - 1) - ν^T a is zero

sum_k λ[k] * a[k-i] + μ = ν[i] for i where a[i] > 0.

This is a system of equations that characterizes the optimum. We could try to solve it directly.

But this is a fixed-point equation (a appears on both sides), so we'd need to iterate. This is essentially what the LP approach does (the LP solution g satisfies the KKT conditions for the subproblem, and the blending is the outer iteration).

def quadratic_line_search(a, g, obj_a, tol=0.005, max_iter=30):
    """
    Line search along a + t*(g - a) using a quadratic model.
    Returns optimal t in [0, 1].
    """
    # Evaluate at a few points to build quadratic model
    t_evals = [0.0, 0.25, 0.5, 0.75, 1.0]
    f_vals = []
    for t in t_evals:
        a_try = a + t * (g - a)
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            f_vals.append(1e10)
        else:
            f_vals.append(compute_obj(a_try / s))

    # Fit quadratic: f(t) ≈ c0 + c1*t + c2*t^2
    t_arr = np.array(t_evals)
    f_arr = np.array(f_vals)
    # Use least squares for quadratic fit
    A_fit = np.vstack([np.ones_like(t_arr), t_arr, t_arr ** 2]).T
    try:
        coeffs, _, _, _ = np.linalg.lstsq(A_fit, f_arr, rcond=None)
    except Exception:
        coeffs = [f_arr[0], f_arr[1] - f_arr[0], 0]

    c0, c1, c2 = coeffs
    if abs(c2) < 1e-15:
        if c1 < 0:
            t_opt = 1.0
        else:
            t_opt = 0.0
    else:
        t_opt = -c1 / (2 * c2)
        t_opt = max(0.01, min(1.0, t_opt))

    # Verify with golden section in a neighborhood
    t_lo = max(0.01, t_opt - 0.15)
    t_hi = min(1.0, t_opt + 0.15)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_pt = t_hi - gr * (t_hi - t_lo)
    d_pt = t_lo + gr * (t_hi - t_lo)

    def f_of_t(t):
        a_try = a + t * (g - a)
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            return 1e10
        return compute_obj(a_try / s)

    fc = f_of_t(c_pt)
    fd = f_of_t(d_pt)
    for _ in range(20):
        if abs(t_hi - t_lo) < tol:
            break
        if fc < fd:
            t_hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = t_hi - gr * (t_hi - t_lo)
            fc = f_of_t(c_pt)
        else:
            t_lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = t_lo + gr * (t_hi - t_lo)
            fd = f_of_t(d_pt)

    return (t_lo + t_hi) / 2.0


def compute_adaptive_weights(a, peak_frac=0.5, boost=2.0):
    """Compute adaptive weights for LP constraints, boosting near current peaks."""
    conv = compute_conv(a / np.sum(a))
    max_conv = np.max(conv)
    # Weight: high near peaks, lower elsewhere
    # w[k] = 1 + boost * exp(-alpha * (1 - conv[k]/max_conv))
    alpha = 3.0
    ratios = conv / max_conv
    weights = 1.0 + boost * np.exp(-alpha * (1.0 - ratios))

    weights = np.maximum(weights, 0.5)

def peeling_refine(a, time_budget=60.0, verbose=False):
    """
    Local refinement by peeling: iteratively reduce the peak lag
    by transferring mass from contributing coordinates to non-contributing ones.
    """
    n = len(a)
    a = np.maximum(a.copy(), 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]

        if peak_val < 1e-15:
            break

        # Find contributing pairs for the peak lag
        # conv[peak_idx] = sum_j a[j] * a[peak_idx - j] for valid j
        contrib = []
        for j in range(max(0, peak_idx - (n - 1)), min(n, peak_idx + 1)):
            other = peak_idx - j
            if 0 <= other < n:
                contrib.append((j, other, a[j] * a[other]))

        if not contrib:
            break

        # Sort by contribution (largest first)
        contrib.sort(key=lambda x: -x[2])

        # Find a "safe" position to receive mass (one with low contribution to all top lags)
        top_lags = np.argsort(conv)[::-1][:5]
        risk = np.zeros(n)
        for k in top_lags:
            k = int(k)
            for j in range(max(0, k - (n - 1)), min(n, k + 1)):
                other = k - j
                if 0 <= other < n:
                    risk[j] += conv[k] * max(a[j], a[other])

        # Transfer mass from the most contributing coordinate to the safest one
        j_src, j_other, _ = contrib[0]
        # Choose source: the larger of the pair
        if a[j_src] >= a[j_other]:
            src = j_src
        else:
            src = j_other

        # Find safe destination
        safe_mask = risk < np.mean(risk)
        safe_positions = np.where(safe_mask)[0]
        safe_positions = safe_positions[safe_positions != src]
        if len(safe_positions) == 0:
            safe_positions = np.array([0])
        dst = safe_positions[rng.integers(0, len(safe_positions))]

        # Transfer amount
        eps = min(a[src] * 0.05, a[src] * 0.001 * (1 + it * 0.1))
        if eps < 1e-12:
            break

        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            it += 1
        else:
            # Try smaller epsilon
            eps2 = eps * 0.3
            a_new2 = a.copy()
            a_new2[src] -= eps2
            a_new2[dst] += eps2
            a_new2 = np.maximum(a_new2, 0)
            a_new2 = a_new2 / a_new2.sum()
            obj_new2 = compute_obj(a_new2)
            if obj_new2 < best_obj - 1e-14:
                best_obj = obj_new2
                best_a = a_new2.copy()
                a = a_new2
                it += 1
            else:
                # Try different destination
                if len(safe_positions) > 1:
                    dst2 = safe_positions[rng.integers(0, len(safe_positions))]
                    if dst2 == src:
                        continue
                    a_new3 = a.copy()
                    a_new3[src] -= eps
                    a_new3[dst2] += eps
                    a_new3 = np.maximum(a_new3, 0)
                    a_new3 = a_new3 / a_new3.sum()
                    obj_new3 = compute_obj(a_new3)
                    if obj_new3 < best_obj - 1e-14:
                        best_obj = obj_new3
                        best_a = a_new3.copy()
                        a = a_new3
                        it += 1
                    else:
                        break

    return best_a


def lp_polish_main(a, time_budget=300.0, verbose=True, seed=42, use_weights=True, use_self_bound=True):
    """
    Main LP-based polishing loop with adaptive weights, self-correlation bound,
    and quadratic line search.
    """
    n = len(a)
    a = np.maximum(a.copy(), 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    g_prev = None
    iter_count = 0
    no_improve = 0
    lp_times = []
    stall_threshold = 10

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed

        # Solve LP
        t_lp_start = time.monotonic()
        if use_weights:
            weights = compute_adaptive_weights(a, peak_frac=0.5, boost=1.5)
        else:
            weights = None

        g_lp, M_lp = solve_lp_improved(
            a, weights=weights, with_self_bound=use_self_bound,
            M_target=None, g_prev=g_prev
        )
        t_lp = time.monotonic() - t_lp_start
        lp_times.append(t_lp)

        if g_lp is None:
            # Try without self-bound
            g_lp, M_lp = solve_lp_improved(a, weights=None, with_self_bound=False, g_prev=None)
            if g_lp is None:
                break
            t_lp = time.monotonic() - t_lp_start

        g_prev = g_lp

        # Line search for blend factor
        t_opt = quadratic_line_search(best_a, g_lp, best_obj, tol=0.008, max_iter=25)

        a_try = (1.0 - t_opt) * best_a + t_opt * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            break
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)

        iter_count += 1

        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a.copy()
            no_improve = 0
            if verbose and iter_count % 10 == 0:
                avg_lp_t = np.mean(lp_times[-20:]) if len(lp_times) >= 20 else t_lp
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={t_opt:.3f}, "
                      f"M={M_lp:.6f}, lp_t={avg_lp_t:.3f}s, el={elapsed:.0f}s")
        else:
            no_improve += 1
            if no_improve == 3:
                # Try smaller step
                for t_small in [t_opt * 0.3, t_opt * 0.1, 0.05, 0.02]:
                    a_try2 = (1.0 - t_small) * best_a + t_small * g_lp
                    a_try2 = np.maximum(a_try2, 0)
                    s2 = float(np.sum(a_try2))
                    if s2 < 1e-10:
                        continue
                    a_try2 = a_try2 / s2
                    obj_try2 = compute_obj(a_try2)
                    if obj_try2 < best_obj - 1e-14:
                        best_obj = obj_try2
                        best_a = a_try2.copy()
                        a = best_a.copy()
                        no_improve = 0
                        if verbose:
                            print(f"    LP small it{iter_count}: obj={best_obj:.8f}, t={t_small:.4f}")
                        break
            elif no_improve >= stall_threshold:
                # Perturb and continue
                scale = 0.005 * min(no_improve - 8, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                g_prev = None
                if verbose:
                    print(f"    LP perturb it{iter_count}: scale={scale:.4f}")
                if iter_count > 200:
                    break

        # Check if we should stop (LP taking too long)
        if len(lp_times) >= 50:
            recent_avg = np.mean(lp_times[-10:])
            if recent_avg > 3.0:
                if verbose:
                    print(f"    LP slowing: avg={recent_avg:.1f}s, stopping")
                break

    return best_a


def nesterov_smoothed_descent(a, time_budget=200.0, tau_init=10.0, tau_end=400.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(y, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        # Barzilai-Borwein step size
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)

        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations with various kernel structures."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Bimodal
        c1, c2 = 0.3, 0.7
        w = 0.1 + 0.05 * rng.random()
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Semicircle (Wigner semicircle)
        a = np.sqrt(np.maximum(1 - x ** 2, 0))
        a = a / a.sum()
    elif mode == 4:
        # Fourier-based
        freqs = rng.integers(1, 8, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        alpha = rng.uniform(-1, 3)
        a = x ** alpha + 0.01
        a = a / a.sum()
    elif mode == 6:
        # Raised cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a / a.sum()
    elif mode == 7:
        # Triangular
        a = 1 - np.abs(2 * x - 1)
        a = a / a.sum()
    elif mode == 8:
        # Product of two cosines
        f1 = rng.integers(1, 4)
        f2 = rng.integers(4, 8)
        a = (1 + np.cos(2 * np.pi * f1 * x)) * (1 + np.cos(2 * np.pi * f2 * x))
        a = a / a.sum()
    elif mode == 9:
        # Log-normal
        a = np.exp(rng.normal(0, 0.5, n))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate():
    """
    Main search function: combines improved LP polishing, smoothed descent,
    peeling refinement, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + 1450.0  # Leave 50s buffer

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(2024)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Quick smoothed GD warmup ===
    print(f"\n=== Phase 1: Smoothed GD Warmup ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 90.0)
    if p1_budget > 20:
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=p1_budget,
                                          tau_init=5.0, tau_end=300.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase1 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 GD: obj={obj_sgd:.8f}")

    # === Phase 2: Main LP Polishing (improved) ===
    print(f"\n=== Phase 2: Improved LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 500.0)
    if p2_budget > 30:
        a_lp = lp_polish_main(best_seq, time_budget=p2_budget, verbose=True,
                              seed=42, use_weights=True, use_self_bound=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f}")

    # === Phase 3: Peeling refinement ===
    print(f"\n=== Phase 3: Peeling Refinement ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.05, 50.0)
    if p3_budget > 10:
        a_peel = peeling_refine(best_seq, time_budget=p3_budget, verbose=True)
        obj_peel = compute_obj(a_peel)
        if obj_peel < best_score:
            best_score = obj_peel
            best_seq = a_peel.copy()
            print(f"  Phase3 Peel: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 Peel: obj={obj_peel:.8f}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 300), max(400, n_current - 200), max(400, n_current - 100),
        n_current + 100, n_current + 200, n_current + 300, n_current + 500,
        600, 800, 1000, 1200, 1500, 1800
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2500 and abs(n - n_current) > 50]
    # Sort by proximity to current
    n_options.sort(key=lambda x: abs(x - n_current))

    for n_try in n_options[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")

        # Try both resampled and diverse init
        candidates = [resample_sequence(best_seq, n_try)]
        candidates.append(diverse_init(n_try, rng, mode=int(rng.integers(0, 10))))

        for a_try in candidates[:2]:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            # Quick GD warmup
            alloc1 = min(remaining * 0.15, 30.0)
            if alloc1 > 8:
                a_try = nesterov_smoothed_descent(a_try, time_budget=alloc1,
                                                  tau_init=10.0, tau_end=200.0)

            # LP polish
            remaining = deadline - time.monotonic()
            alloc2 = min(remaining * 0.25, 80.0)
            if alloc2 > 15:
                a_try = lp_polish_main(a_try, time_budget=alloc2, seed=n_try, verbose=False)

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_current = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: LP without self-bound (sometimes helps) ===
    print(f"\n=== Phase 5: LP without Self-Bound ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.12, 120.0)
    if p5_budget > 20:
        a_lp2 = lp_polish_main(best_seq, time_budget=p5_budget, verbose=False,
                               seed=777, use_weights=True, use_self_bound=False)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase5 LP-nobound: obj={best_score:.8f} **")
        else:
            print(f"  Phase5 LP-nobound: obj={obj_lp2:.8f}")

    # === Phase 6: Perturbation + Re-LP ===
    print(f"\n=== Phase 6: Perturbation + Re-LP ===")
    remaining = deadline - time.monotonic()

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        scales = [0.005, 0.01, 0.015, 0.02, 0.03, 0.04]
        scale = scales[r]
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc = min(remaining * 0.3, 100.0)
        if alloc > 20:
            a_pert = lp_polish_main(a_pert, time_budget=alloc, seed=200 + r, verbose=False,
                                    use_weights=True, use_self_bound=True)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Perturb {r} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Perturb {r} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 7: Final intensive LP ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.6, 180.0)
        a_final = lp_polish_main(best_seq, time_budget=alloc, verbose=True, seed=999,
                                 use_weights=True, use_self_bound=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # Final peeling
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 40.0)
        a_pf = peeling_refine(best_seq, time_budget=alloc)
        obj_pf = compute_obj(a_pf)
        if obj_pf < best_score:
            best_score = obj_pf
            best_seq = a_pf.copy()
            print(f"  Final Peel: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
