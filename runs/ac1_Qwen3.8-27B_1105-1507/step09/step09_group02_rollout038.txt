"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: LP with golden-section line search + QP refinement + smoothed GD + adaptive peak shaving.
Key improvements: proper line search for LP blending, weighted QP for smoothing,

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # Row k has A[k, j] = a[k-j] if 0 <= k-j <= n-1
        k_start = max(0, j)  # k-j >= 0 => k >= j
        k_end = min(2 * n - 2, j + n - 1)  # k-j <= n-1 => k <= j + n - 1
        # Actually: A[k, j] = a[k-j] when 0 <= k-j <= n-1, i.e., j <= k <= j+n-1
        # And k ranges from 0 to 2n-2
        k_lo = max(0, j)
        k_hi = min(2 * n - 2, j + n - 1)
        ks = np.arange(k_lo, k_hi + 1)
        A[ks, j] = a[ks - j]
    return A


def solve_lp_minpeak(a, solver=None):
    """
    Solve LP: min M s.t. A@g <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    if solver:
        solvers = [solver] + [s for s in solvers if s != solver]
    
    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g, a_conv, ag_conv, gg_conv, num_samples=30):
    """
    Find optimal t in [0,1] minimizing phi(t) = max_k h_k(t)
    where h_k(t) = (1-t)^2 * a_conv[k] + 2*t*(1-t) * ag_conv[k] + t^2 * gg_conv[k]
    Uses dense sampling + parabolic refinement.
    """
    n = len(a)
    
    # Dense sampling
    ts = np.linspace(0, 1, num_samples)
    phis = np.zeros(num_samples)
    for i, t in enumerate(ts):
        h = (1-t)**2 * a_conv + 2*t*(1-t) * ag_conv + t**2 * gg_conv
        phis[i] = float(np.max(h))
    
    # Find best sample
    best_idx = int(np.argmin(phis))
    best_t = ts[best_idx]
    best_val = phis[best_idx]
    
    # Parabolic refinement around best
    if best_idx > 0 and best_idx < num_samples - 1:
        y0, y1, y2 = phis[best_idx-1], phis[best_idx], phis[best_idx+1]
        x0, x1, x2 = ts[best_idx-1], ts[best_idx], ts[best_idx+1]
        denom = (y0 - 2*y1 + y2)
        if abs(denom) > 1e-15:
            tp = x1 + 0.5 * (x0-x1)**2 * (y1-y2) - 0.5 * (x1-x2)**2 * (y1-y0)
            tp = tp / (0.5 * (x0-x1)**2 / (x0-x1) ... )
            # Let me use the standard parabolic interpolation formula
            # tp = x1 - 0.5 * ((x1-x0)^2*(y1-y2) - (x1-x2)^2*(y1-y0)) / ((x1-x0)*(y1-y2) - (x1-x2)*(y1-y0))
            denom2 = (x1-x0)*(y1-y2) - (x1-x2)*(y1-y0)
            if abs(denom2) > 1e-15:
                tp = x1 - 0.5 * ((x1-x0)**2*(y1-y2) - (x1-x2)**2*(y1-y0)) / denom2
                if 0 <= tp <= 1:

                    val = float(np.max(h))
                    if val < best_val:
                        best_t = tp
                        best_val = val
    
    return best_t, best_val

    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_lo = max(0, j)
        k_hi = min(2 * n - 2, j + n - 1)
        ks = np.arange(k_lo, k_hi + 1)
        A[ks, j] = a[ks - j]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv, float(M.value)
        except Exception:
            continue
    return None, None


def solve_qp_smooth(a, tau_weight=0.0):
    """
    Solve QP: min ||W^{1/2} A g||^2 + mu*||g-a||^2 s.t. sum(g)=1, g>=0.
    If tau_weight=0, unweighted: min ||A g||^2.
    Returns g_optimal or None.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    m = 2 * n - 1

    if tau_weight > 0:
        conv_a = compute_conv(a_norm)
        max_c = float(np.max(conv_a))
        w = np.exp(tau_weight * (conv_a - max_c) / (max_c + 1e-15))
        w = np.sqrt(w)
        AW = w[:, None] * A
    else:
        AW = A

    mu = 1.0
    g = cp.Variable(n, nonneg=True)
    # Objective: ||AW @ g||^2 + mu * ||g - a_norm||^2
    obj = cp.sum_squares(AW @ g) + mu * cp.sum_squares(g - a_norm)
    constraints = [cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(obj), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv
        except Exception:
            continue
    return None


def line_search_blend(a_conv, ag_conv, gg_conv):
    """
    Find optimal t in [0,1] minimizing max_k [(1-t)^2*a_conv[k] + 2t(1-t)*ag_conv[k] + t^2*gg_conv[k]].
    """
    def phi(t):
        h = (1.0 - t) ** 2 * a_conv + 2.0 * t * (1.0 - t) * ag_conv + t ** 2 * gg_conv
        return float(np.max(h))

    try:
        result = minimize_scalar(phi, bounds=(0.0, 1.0), method='bounded',
                                 options={'xatol': 1e-8, 'maxiter': 60})
        return float(result.x), float(result.fun)
    except Exception:
        # Fallback: dense sampling
        ts = np.linspace(0, 1, 50)
        phis = np.array([phi(t) for t in ts])
        best_idx = int(np.argmin(phis))
        return float(ts[best_idx]), float(phis[best_idx])


def lp_polish_with_linesearch(a, time_budget, verbose=False):
    """
    LP polishing with proper line search.
    Each iteration: solve LP, compute line search, update.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    rng = np.random.default_rng(123)
    consecutive_fails = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Compute convolution terms for line search
        a_conv = compute_conv(a)
        ag_conv = compute_conv_cross(a, g_lp)
        gg_conv = compute_conv(g_lp)

        # Line search
        t_opt, phi_opt = line_search_blend(a_conv, ag_conv, gg_conv)

        if t_opt < 1e-10:
            consecutive_fails += 1
        else:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                consecutive_fails += 1
            else:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)

                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    consecutive_fails = 0
                    if verbose and it % 20 == 0:
                        elapsed = time.monotonic() - start
                        print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, M={M_lp:.6f}, elapsed={elapsed:.0f}s")
                else:
                    consecutive_fails += 1
                    if consecutive_fails > 5:
                        # Small perturbation
                        scale = 0.01 + 0.003 * min(consecutive_fails, 30)
                        a = best_a * np.exp(rng.normal(0, scale, n))
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        if verbose:
                            print(f"    LP it{it}: perturbed (scale={scale:.4f})")
                        consecutive_fails = 0

        it += 1

        if consecutive_fails > 50:
            break

    return best_a


def qp_refine(a, time_budget, verbose=False):
    """
    QP-based refinement: alternate between weighted QP and line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    tau_schedule = [0.0, 5.0, 10.0, 20.0, 5.0, 10.0]

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5.0:
            break

        tau = tau_schedule[it % len(tau_schedule)]
        g_qp = solve_qp_smooth(a, tau_weight=tau)
        if g_qp is None:
            break

        # Line search
        a_conv = compute_conv(a)
        ag_conv = compute_conv_cross(a, g_qp)
        gg_conv = compute_conv(g_qp)

        t_opt, phi_opt = line_search_blend(a_conv, ag_conv, gg_conv)

        if t_opt > 1e-10:
            a_new = (1.0 - t_opt) * a + t_opt * g_qp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    if verbose and it % 5 == 0:
                        elapsed = time.monotonic() - start
                        print(f"    QP it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={t_opt:.4f}, elapsed={elapsed:.0f}s")

        it += 1

    return best_a


def smoothed_grad(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a).
    Returns (gradient, conv).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    total = float(np.sum(exp_vals))
    if total < 1e-300:
        p = np.ones(len(conv)) / len(conv)
    else:
        p = exp_vals / total

    # grad_i = 2 * sum_k p[k] * (d conv[k] / d a[i])
    # d conv[k] / d a[i] = a[k-i] + a[i-k] (with boundary handling)
    # For linear convolution: conv[k] = sum_j a[j]*a[k-j]
    # d conv[k]/d a[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
    n_conv = len(conv)
    g = np.zeros(n)
    for i in range(n):
        # Term 1: a[k-i] for k where 0 <= k-i < n, i.e., i <= k <= i+n-1
        k1_lo = i
        k1_hi = min(n_conv - 1, i + n - 1)
        if k1_lo <= k1_hi:
            g[i] += float(np.sum(p[k1_lo:k1_hi + 1] * np.flip(a[:k1_hi + 1 - k1_lo])))
        # Term 2: a[i-k] for k where 0 <= i-k < n, i.e., i-n+1 <= k <= i
        k2_lo = max(0, i - n + 1)
        k2_hi = i
        if k2_lo <= k2_hi:
            g[i] += float(np.sum(p[k2_lo:k2_hi + 1] * np.flip(a[k2_hi + 1 - k2_lo:])))

    return 2.0 * g, conv


def nesterov_smoothed_descent(a, time_budget, tau_init=5.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.7
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        y = a + beta * v
        y = project_simplex(y)

        g, _ = smoothed_grad(y, tau)

        v = beta * v - lr * g
        a_new = y + v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)

        if it - best_it > 10000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it

        if verbose and it % 2000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, elapsed={elapsed:.0f}s")

    return best_a


def peak_shave(a, time_budget, verbose=False):
    """Peak shaving using subgradient of top peaks with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.5 / np.sqrt(n)
    it = 0
    stall = 0
    K = min(15, 2 * n - 1)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        if max_c < 1e-15:
            break

        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        # Soft weighting
        temp = 0.05 * max_c
        weights = np.exp((peak_vals - peak_vals[0]) / temp)
        weights /= float(np.sum(weights))

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            # d conv[k]/d a[i] = a[k-i] + a[i-k] (boundary handled)
            for i in range(n):
                val = 0.0
                if 0 <= k - i < n:
                    val += a[k - i]
                if 0 <= i - k < n:
                    val += a[i - k]
                g[i] += w * val

        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break

        a_new = a - lr * g / g_norm
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.05, 5.0 / np.sqrt(n))
        else:
            stall += 1
            lr *= 0.7
            a = best_a
            if stall > 50:
                lr = 0.5 / np.sqrt(n)
                stall = 0
            if lr < 1e-10:
                break

        it += 1
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    PS it{it}: obj={best_obj:.8f}, lr={lr:.4f}, elapsed={elapsed:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.ones(new_n) / new_n
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def coordinate_descent(a, time_budget, verbose=False):
    """Simple coordinate descent: cycle through coordinates, optimize each."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    # Precompute: for each coordinate i, the "rest" contribution to each conv[k]
    # conv[k] = a[i]^2 (if k=2i) + a[i]*(rest_k) + rest2_k
    # Actually let's just do direct optimization per coordinate

    while time.monotonic() - start < time_budget:
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break

            # Optimize a[i] with rest fixed
            rest = a.copy()
            rest[i] = 0
            s_rest = float(np.sum(rest))
            if s_rest < 1e-10:
                continue

            # conv[k] as function of x = a[i]:
            # conv[k] = x^2 * delta(k, 2i... no)
            # Actually conv[k] = sum_j rest[j]*a[k-j] + a[i]*a[k-i] + a[i]*rest[k-i] + a[i]*a[k-i]*delta
            # This is getting complex. Let me just try a few values.
            x_old = a[i]
            candidates = [0.0, x_old * 0.5, x_old * 0.8, x_old * 1.2, x_old * 1.5,
                          x_old + 0.001, x_old - 0.001]
            candidates = [max(0.0, c) for c in candidates]

            best_x = x_old
            best_local = np.inf
            for x_try in candidates:
                a_try = a.copy()
                a_try[i] = x_try
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_local:
                    best_local = obj_try
                    best_x = x_try

            if best_x != x_old:
                a[i] = best_x
                a = a / float(np.sum(a))
                obj_now = compute_obj(a)
                if obj_now < best_obj - 1e-14:
                    best_obj = obj_now
                    best_a = a.copy()
                    improved = True

        it += 1
        if not improved and it > 1:
            break
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD it{it}: obj={best_obj:.8f}, elapsed={elapsed:.0f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        x = np.arange(n)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-0.999, 0.999, n)
        a = 1.0 / np.sqrt(1 - x ** 2)
        a = a / a.sum()
    elif mode == 4:
        # Fourier-based
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1.5, 1.5)
        a = a / a.sum()
    elif mode == 6:
        # Triangle
        a = np.abs(1.0 - 2.0 * np.arange(n) / n)
        a = a / a.sum()
    elif mode == 7:
        # Cosine window
        x = np.arange(n)
        a = 0.5 * (1 + np.cos(np.pi * x / (n - 1)))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses LP with line search as the primary optimizer,
    complemented by QP smoothing, smoothed GD, peak shaving, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0  # 30s safety buffer
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    def remaining_time():
        return deadline - time.monotonic()

    # === Phase 1: Quick multi-n exploration ===
    print(f"\n=== Phase 1: Multi-n quick scan ===")
    if remaining_time() > 100:
        n_options = sorted(set([n_target, n_target - 100, n_target + 100, n_target + 200,
                                 700, 900, 1000, 1100, 1200]))
        n_options = [n for n in n_options if 500 <= n <= 1400 and abs(n - n_target) >= 50]

        per_n_time = min(12.0, remaining_time() * 0.7 / max(1, len(n_options)))
        for n_try in n_options[:6]:
            if remaining_time() < 30:
                break
            a_try = resample_sequence(best_seq, n_try)
            a_try = lp_polish_with_linesearch(a_try, time_budget=per_n_time)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_target = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 2: Intensive LP with line search (main workhorse) ===
    print(f"\n=== Phase 2: Intensive LP with line search (n={len(best_seq)}) ===")
    lp_budget = min(remaining_time() * 0.55, 600.0)
    if lp_budget > 30:
        # Run in multiple rounds with perturbation between
        num_rounds = 3
        per_round = lp_budget / num_rounds
        for rnd in range(num_rounds):
            if remaining_time() < 30:
                break
            alloc = min(per_round, remaining_time() * 0.5)
            if alloc < 15:
                break
            a_lp = lp_polish_with_linesearch(best_seq, time_budget=alloc, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd+1}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd+1}: obj={obj_lp:.8f}")
            # Perturb for next round
            if rnd < num_rounds - 1 and remaining_time() > 30:
                scale = 0.012 * (rnd + 1)
                best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                best_seq = np.maximum(best_seq, 0)
                best_seq = best_seq / best_seq.sum()

    # === Phase 3: QP refinement ===
    print(f"\n=== Phase 3: QP refinement ===")
    qp_budget = min(remaining_time() * 0.15, 180.0)
    if qp_budget > 20:
        a_qp = qp_refine(best_seq, time_budget=qp_budget, verbose=True)
        obj_qp = compute_obj(a_qp)
        if obj_qp < best_score:
            best_score = obj_qp
            best_seq = a_qp.copy()
            print(f"  QP: obj={best_score:.8f} **")
        else:
            print(f"  QP: obj={obj_qp:.8f}")

    # === Phase 4: Smoothed GD ===
    print(f"\n=== Phase 4: Smoothed Gradient Descent ===")
    gd_budget = min(remaining_time() * 0.15, 150.0)
    if gd_budget > 20:
        a_gd = nesterov_smoothed_descent(best_seq, time_budget=gd_budget,
                                         tau_init=5.0, tau_end=400.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  GD: obj={best_score:.8f} **")
        else:
            print(f"  GD: obj={obj_gd:.8f}")

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving ===")
    ps_budget = min(remaining_time() * 0.12, 120.0)
    if ps_budget > 15:
        a_ps = peak_shave(best_seq, time_budget=ps_budget, verbose=True)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  PeakShave: obj={best_score:.8f} **")
        else:
            print(f"  PeakShave: obj={obj_ps:.8f}")

    # === Phase 6: Final LP polish ===
    print(f"\n=== Phase 6: Final LP polish ===")
    final_budget = min(remaining_time() * 0.3, 150.0)
    if final_budget > 15:
        a_final = lp_polish_with_linesearch(best_seq, time_budget=final_budget, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # === Phase 7: Try one alternate n ===
    print(f"\n=== Phase 7: Alternate n attempt ===")
    if remaining_time() > 60:
        n_alt = len(best_seq) + 200
        if n_alt <= 1500:
            a_alt = resample_sequence(best_seq, n_alt)
            a_alt = lp_polish_with_linesearch(a_alt, time_budget=min(remaining_time() * 0.3, 80.0))
            obj_alt = compute_obj(a_alt)
            if obj_alt < best_score:
                best_score = obj_alt
                best_seq = a_alt.copy()
                print(f"  ** n={n_alt}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_alt}: obj={obj_alt:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
