where `T_k a` is the vector with `(T_k a)[i] = a[k-i] + a[i-k]` (with bounds).

We can then do a subgradient step in the direction that opposes the subgradient (i.e., the most negative direction in the subdifferential).

**Even simpler: projected subgradient method**

At each step:
1. Compute conv(a, a) and find the argmax set S
2. Pick k* (e.g., the one with maximum value, or randomly from S)
3. Compute gradient: `grad[i] = a[k*-i] + a[i-k*]` (with bounds)
4. Update: `a_new = project_simplex(a - lr * grad)`
5. Use decreasing step size

This is very simple and might find a different local minimum than the LP method.

**Combined approach:**
- Alternately use LP-based updates and subgradient updates
- This might help escape from LP fixed points

Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated LP-based alternating optimization with proper line search,
subgradient escape mechanisms, multi-length exploration, and structured restarts.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def solve_lp_minpeak(a, solver=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal, dual_values) or (None, None, None).
    """
    n = len(a)
    if n == 0:
        return None, None, None
    
    # Build constraint matrix: A @ g = conv(a, g), A is (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # For column j: A[k, j] = a[k-j] for 0 <= k-j < n, i.e., j <= k < j+n
        k_start = j
        k_end = min(2 * n - 1, j + n)
        a_slice = a[:2 * n - 1 - j]  # a[0], a[1], ..., a[2n-2-j]
        A[k_start:k_end, j] = a_slice
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    # Use the dual to get information about binding constraints
    A_g_le_M = (A @ g <= M)
    constraints = [A_g_le_M, cp.sum(g) == 1]
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    if solver:
        solvers = [solver] + solvers
    
    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    g_val = g_val / s_val
                    # Get dual values
                    dual = A_g_le_M.dual_value
                    if dual is None:
                        dual = np.zeros(2 * n - 1)
                    return g_val, float(M.value), np.asarray(dual)
        except Exception:
            continue
    return None, None, None

def subgradient_step(a, lr, top_k=1):
    """
    One step of subgradient descent on F(a) = max_k conv(a,a)[k].
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Get top_k indices of the convolution
    top_indices = np.argsort(conv)[::-1][:top_k]
    
    grad = np.zeros(n)
    for k in top_indices:
        k = int(k)
        for i in range(n):
            j1 = k - i
            j2 = i - k
            if 0 <= j1 < n:
                grad[i] += a[j1]
            if 0 <= j2 < n:
                grad[i] += a[j2]
    
    grad /= top_k
    # The actual gradient of (a*a)[k] is 2*(...), but we use 1x for stability
    grad = 2.0 * grad
    
    a_new = a - lr * grad
    a_new = project_simplex(a_new)
    return a_new, conv

def diverse_init(n, rng, mode=0):

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
