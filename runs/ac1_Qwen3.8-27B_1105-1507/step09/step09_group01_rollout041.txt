```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Primary method: coordinate descent with exact golden-section line search per
coordinate, exploiting the rank-1 structure of the convolution update.
Supplemented by LP-based polishing (cvxpy) and multi-n exploration.
The coordinate descent achieves O(n^2) per sweep with small constants,
enabling thousands of sweeps within the time budget.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def coord_descent(a, n_sweeps=None, time_budget=60.0, rng=None, verbose=False):
    """
    Coordinate descent with exact golden-section line search.
    Maintains C = conv(a,a) and S = sum(a) incrementally.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(a)
    a = a.copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-15:
        return a
    a = a / S
    S = 1.0
    
    C = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(C))
    best_a = a.copy()
    
    start = time.monotonic()
    sweep = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        if n_sweeps is not None and sweep >= n_sweeps:
            break
        
        perm = rng.permutation(n)
        S = 1.0  # Should always be 1 after normalization, but track for safety
        
        for i in perm:
            old = a[i]
            # Compute v_i = 2*a shifted by i
            v_offset = 2.0 * a  # will be placed at positions i..i+n-1
            
            # Spike position
            spike_pos = 2 * i
            has_spike = (spike_pos < 2 * n - 1)
            
            # Determine search range
            if old < 1e-10:
                lo, hi = 0.0, min(1.0, 0.1 / n)
            else:
                lo, hi = 0.0, min(1.0, 4.0 * old)
            
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            
            # f(t) = 2n * max(C + (t-old)*v + (t-old)^2*spike) / (S + (t-old))^2
            # Since S=1: = 2n * max(C + (t-old)*v + (t-old)^2*spike) / (1 + (t-old))^2
            
            fc = _eval_cd(C, v_offset, i, old, c, spike_pos, has_spike, n, S)
            fd = _eval_cd(C, v_offset, i, old, d, spike_pos, has_spike, n, S)
            
            for _ in range(25):
                if hi - lo < 1e-13:
                    break
                if fc < fd:
                    hi = d; d = c; fd = fc
                    c = hi - gr * (hi - lo)
                    fc = _eval_cd(C, v_offset, i, old, c, spike_pos, has_spike, n, S)
                else:
                    lo = c; c = d; fc = fd
                    d = lo + gr * (hi - lo)
                    fd = _eval_cd(C, v_offset, i, old, d, spike_pos, has_spike, n, S)
            
            t_opt = (lo + hi) * 0.5
            t_opt = max(0.0, min(t_opt, 1.0))
            
            d_val = t_opt - old
            if abs(d_val) > 1e-15:
                a[i] = t_opt
                # Update C incrementally
                C[i:i+n] += d_val * v_offset
                if has_spike:
                    C[spike_pos] += d_val * d_val
                S += d_val
        
        # End of sweep
        obj = 2.0 * n * float(np.max(C)) / (S * S) if S > 1e-15 else 1e10
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                # Perturb
                a = best_a * np.exp(rng.normal(0, 0.005, n))
                a = np.maximum(a, 0)
                S = float(np.sum(a))
                a = a / S
                S = 1.0
                C = compute_conv(a)
                best_obj = 2.0 * n * float(np.max(C))
                no_improve = 0
        
        if verbose and sweep % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")
        
        sweep += 1
    
    return best_a


def _eval_cd(C, v_offset, i, old_t, t, spike_pos, has_spike, n, S):
    """Evaluate objective at value t for coordinate i."""
    d = t - old_t
    # Compute max of C + d*v_offset[i:i+n] + d^2*spike
    # v_offset is 2*a, placed at positions i..i+n-1
    new_C = C.copy()
    new_C[i:i+n] += d * v_offset
    if has_spike:
        new_C[spike_pos] += d * d
    new_S = S + d
    if new_S < 1e-15:
        return 1e10
    return 2.0 * n * float(np.max(new_C)) / (new_S * new_S)
```

```python
def _eval_cd_fast(C, a, i, old_t, t, n, S):
    """Fast evaluation: only operates on the window [i, i+n-1]."""
    d = t - old_t
    if abs(d) < 1e-15:
        new_S = S
        max_c = float(np.max(C))
    else:
        window = C[i:i+n] + d * 2.0 * a
        window[i] += d * d  # spike
        max_in = float(np.max(window))
        max_out = 0.0
        if i > 0:
            max_out = max(max_out, float(np.max(C[:i])))
        if i + n < 2 * n - 1:
            max_out = max(max_out, float(np.max(C[i+n:])))
        max_c = max(max_in, max_out)
        new_S = S + d
    
    if new_S < 1e-15:
        return 1e10
    return 2.0 * n * max_c / (new_S * new_S)
```

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Primary method: coordinate descent with exact golden-section line search per
coordinate, exploiting the rank-1 structure of the convolution update for O(n)
per evaluation. Supplemented by LP-based polishing (cvxpy) and multi-n exploration.
The coordinate descent achieves O(n^2) per sweep with very small constants,
enabling thousands of sweeps within the time budget.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def _eval_cd_fast(C, a, i, old_t, t, n, S):
    """Fast objective evaluation for coordinate descent at coordinate i."""
    d = t - old_t
    new_S = S + d
    if new_S < 1e-15:
        return 1e10
    if abs(d) < 1e-15:
        max_c = float(np.max(C))
    else:
        window = C[i:i + n] + d * 2.0 * a
        window[i] += d * d
        max_in = float(np.max(window))
        max_out = -1e18
        if i > 0:
            max_out = max(max_out, float(np.max(C[:i])))
        if i + n < 2 * n - 1:
            max_out = max(max_out, float(np.max(C[i + n:])))
        max_c = max(max_in, max_out)
    return 2.0 * n * max_c / (new_S * new_S)


def coord_descent(a, time_budget=60.0, rng=None, verbose=False):
    """
    Coordinate descent with exact golden-section line search.
    Maintains C = conv(a,a) incrementally for O(n) per evaluation.
    """
    if rng is None:
        rng = np.random.default_rng(0)

    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-15:
        return a
    a = a / S
    S = 1.0

    C = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(C))
    best_a = a.copy()

    start = time.monotonic()
    sweep = 0
    no_improve = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        perm = rng.permutation(n)
        S = 1.0

        for i in perm:
            old = a[i]

            # Determine search range
            if old < 1e-10:
                lo, hi = 0.0, max(2.0 / n, 1e-6)
            else:
                lo, hi = 0.0, min(1.0, max(5.0 / n, 4.0 * old))

            # Golden section search
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = _eval_cd_fast(C, a, i, old, c, n, S)
            fd = _eval_cd_fast(C, a, i, old, d, n, S)

            for _ in range(22):
                if hi - lo < 1e-14:
                    break
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = _eval_cd_fast(C, a, i, old, c, n, S)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = _eval_cd_fast(C, a, i, old, d, n, S)

            t_opt = (lo + hi) * 0.5
            t_opt = max(0.0, t_opt)

            d_val = t_opt - old
            if abs(d_val) > 1e-15:
                a[i] = t_opt
                C[i:i + n] += d_val * 2.0 * a
                spike = 2 * i
                if spike < 2 * n - 1:
                    C[spike] += d_val * d_val
                S += d_val

        # End of sweep
        if S < 1e-15:
            a = best_a.copy()
            S = 1.0
            C = compute_conv(a)
            continue

        obj = 2.0 * n * float(np.max(C)) / (S * S)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                scale = 0.003 + 0.002 * no_improve
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                S = float(np.sum(a))
                a = a / S
                S = 1.0
                C = compute_conv(a)
                best_obj = 2.0 * n * float(np.max(C))
                no_improve = 0
                if no_improve > 20:
                    break

        if verbose and sweep % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

        sweep += 1

    return best_a


def build_conv_matrix(a_norm):
    """Build matrix A such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp_minpeak(a, M_target=None, verbose=False):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue

    return None, None


def lp_polish(a, time_budget=200.0, verbose=False, seed=42):
    """LP-based polishing with golden-section blending."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Golden section for blend factor
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.01, 1.0
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)

        def blend_eval(t):
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return 1e10
            return compute_obj(a_try / s_try)

        fc = blend_eval(c)
        fd = blend_eval(d)
        for _ in range(30):
            if hi - lo < 0.01:
                break
            if fc < fd:
                hi = d; d = c; fd = fc
                c = hi - gr * (hi - lo)
                fc = blend_eval(c)
            else:
                lo = c; c = d; fc = fd
                d = lo + gr * (hi - lo)
                fd = blend_eval(d)

        t_opt = (lo + hi) * 0.5
        a_try = (1.0 - t_opt) * best_a + t_opt * g_lp
        a_try = np.maximum(a_try, 0)
        a_try = a_try / a_try.sum()
        obj_try = compute_obj(a_try)

        iter_count += 1
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a.copy()
            no_improve = 0
            if verbose and iter_count % 5 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={t_opt:.3f}, el={time.monotonic()-start:.0f}s")
        else:
            no_improve += 1
            if no_improve >= 4:
                scale = 0.01 * min(no_improve, 6)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                if iter_count > 50:
                    break

    return best_a
