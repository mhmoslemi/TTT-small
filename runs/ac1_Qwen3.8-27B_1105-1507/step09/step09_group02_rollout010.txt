"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses multiplicative weights for coarse search, LP alternating for polishing,
and FFT-based gradient methods for fine-tuning. Explores multiple sequence lengths.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        lo = 0
        hi = min(n, m - j)
        A[j:j+hi, j] = a[:hi]
    return A


def solve_lp_minpeak(a, solver=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if solver is None:
        solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    else:
        solvers = [solver]
    
    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv, float(M.value)
        except Exception:
            continue
    return None, None


def multiplicative_weights(a, time_budget, verbose=False):
    """
    Multiplicative weights / exponential weights method for minimizing max conv.
    Fast iteration, good for coarse optimization.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 1.0 / np.sqrt(n)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_k = int(np.argmax(conv))
        
        # Subgradient: d/d a_i of (a*a)[max_k] = a[max_k - i] (reversed)
        # For the max, the subgradient is the gradient at the argmax
        subgrad = np.zeros(n)
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        if i_start <= i_end:
            subgrad[i_start:i_end + 1] = a[max_k - i_end : max_k - i_start + 1][::-1]
        
        # Multiplicative update
        a_new = a * np.exp(-eta * subgrad)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = max(eta * 0.999, 1.0 / np.sqrt(n) * 0.1)
        else:
            stall += 1
            eta *= 0.95
            if stall > 100:
                # Restart with perturbation
                rng = np.random.default_rng(it)
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                eta = 1.0 / np.sqrt(n)
                stall = 0
            if eta < 1e-10:
                break

        if verbose and it % 5000 == 0:
            print(f"    MW it{it}: obj={best_obj:.8f}, eta={eta:.6f}")
        it += 1

    return best_a


def lp_alternating(a, time_budget, verbose=False):

    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    rng = np.random.default_rng(123)
    consecutive_no_improve = 0
    alpha = 1.0  # full step initially

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:

        # Try several step sizes
        step_candidates = [0.3, 0.5, 0.7, 0.9, 1.0]
        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_alpha = alpha

        for t in step_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_alpha = t

        it += 1
        if best_local_obj < best_obj - 1e-13:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a
            alpha = best_local_alpha
            consecutive_no_improve = 0
            if verbose and it % 3 == 0:
                print(f"    LP-Alt it{it}: obj={best_obj:.8f}, alpha={alpha:.2f}, M={M_lp:.6f}")
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > 10:
                # Perturb to escape
                scale = 0.02 * min(consecutive_no_improve / 10, 3)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                consecutive_no_improve = 0
            if consecutive_no_improve > 50:

def gradient_polish(a, time_budget, verbose=False):
    """
    Gradient-based polishing using subgradient of max conv.

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: LP alternating minimization with adaptive step sizes, combined with
multiplicative weights for coarse search, gradient polishing, and multi-length
exploration. Uses CLARABEL/MOSEK for fast LP solves.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build sparse matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        lo = 0
        hi = min(n, m - j)
        A[j:j+hi, j] = a[:hi]
    return A


def solve_lp_minpeak(a, preferred_solver=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if preferred_solver is not None:
        solvers = [preferred_solver] + [cp.CLARABEL, cp.MOSEK, cp.ECOS]
    else:
        solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]

    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv, float(M.value)
        except Exception:
            continue
    return None, None

    """

    if s < 1e-10:

    a = a / s

            obj_try = compute_obj(a_try)

                best_local_a = a_try.copy()
                best_local_alpha = t

            a = best_a
            last_alpha = best_local_alpha

    Good for coarse optimization and escaping local minima.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    eta = 2.0 / np.sqrt(n)
    it = 0
    stall = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Use top-K peaks for a smoother subgradient
        K = min(10, 2 * n - 1)
        top_idx = np.argsort(conv)[::-1][:K]
        max_val = conv[top_idx[0]]
        weights = np.exp(5.0 * (conv[top_idx] - max_val) / (max_val + 1e-15))
        weights /= weights.sum()

        subgrad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:

        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

                a = a / a.sum()
                eta = 2.0 / np.sqrt(n)
                stall = 0

                break

        if verbose and it % 2000 == 0:

def frequency_domain_init(n, rng):
    """
    Initialize in frequency domain: generate a flat-ish power spectrum,
    then take inverse FFT to get time-domain sequence.
    """
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    # Generate a "flat" power spectrum with some randomness
    # The power spectrum determines the autocorrelation
    H = np.ones(size)
    # Add some frequency-dependent modulation
    freqs = np.fft.fftfreq(size)
    bandwidth = rng.uniform(0.3, 0.8)
    H *= 1.0 + rng.uniform(0, 0.3) * np.cos(2 * np.pi * rng.uniform(1, 5) * freqs / (2 * bandwidth))
    H = np.maximum(H, 0.1)

    # Take inverse FFT
    a_complex = np.fft.ifft(H)
    a = np.real(a_complex[:n])
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        # U-shaped
        x = np.linspace(-1, 1, n)
        a = 1.0 + x ** 2
        return a / a.sum()
    elif mode == 3:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 4:
        # Frequency domain init
        return frequency_domain_init(n, rng)
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1, 1)
        return a / a.sum()
    elif mode == 6:
        # Sine modulation
        x = np.linspace(0, 1, n)
        f = rng.integers(2, 8)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 7:
        # Symmetric with peak at center
        x = np.linspace(-1, 1, n)
        a = np.exp(-((x * rng.uniform(2, 5)) ** 2))
        return a / a.sum()
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
