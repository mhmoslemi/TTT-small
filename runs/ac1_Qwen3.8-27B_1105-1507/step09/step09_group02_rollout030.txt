    else:
        constraints = [A @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_min_weighted_obj(a):
    """
    Alternative LP: min sum_k w_k * conv(a,g)[k] s.t. sum(g)=1, g>=0.
    Where w_k emphasizes current peaks.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    conv_a = compute_conv(a_norm)
    max_conv = float(np.max(conv_a))
    # Soft-max weights
    tau = 0.1 * max_conv
    w = np.exp(conv_a / tau)
    w = w / w.sum() * (2 * n - 1)

    b = A.T @ w
    g = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(-cp.sum(b * g)), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def blend_linear(a, g, t):
    """Linear blend: (1-t)*a + t*g, normalized."""
    a_new = (1.0 - t) * a + t * g
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def blend_geometric(a, g, t):
    """Geometric blend: a^(1-t) * g^t, normalized."""
    eps = 1e-12
    a_pos = np.maximum(a, eps)
    g_pos = np.maximum(g, eps)
    a_new = a_pos ** (1.0 - t) * g_pos ** t
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lp_iteration_core(a, time_budget, use_anderson=True, verbose=False, seed=42):
    """
    Core LP iteration with adaptive blending and Anderson acceleration.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)
    stall = 0

    # Anderson acceleration history
    history_a = []
    history_g = []
    anderson_window = 5

    # Track recent LP solutions for averaging
    recent_gs = []

    while time.monotonic() - start < time_budget:
        g, M_lp = solve_lp(a)
        if g is None:
            break

        it += 1
        recent_gs.append(g.copy())
        if len(recent_gs) > 10:
            recent_gs.pop(0)

        # Try multiple blending strategies
        candidates = []

        # Linear blending
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            candidates.append(blend_linear(a, g, t))

        # Geometric blending
        for t in [0.2, 0.4, 0.6]:
            candidates.append(blend_geometric(a, g, t))

        # Anderson acceleration: extrapolate using history
        if use_anderson and len(recent_gs) >= 2:
            # Simple extrapolation: average of recent LP solutions
            g_avg = np.mean(recent_gs[-5:], axis=0)
            g_avg = g_avg / g_avg.sum()
            for t in [0.3, 0.5, 0.7]:
                candidates.append(blend_linear(a, g_avg, t))

            # Extrapolate: a + beta*(g - a_prev)
            if len(history_a) >= 1:
                a_prev = history_a[-1]
                for beta in [0.5, 1.0, 1.5]:
                    a_ext = a + beta * (g - a_prev)
                    a_ext = np.maximum(a_ext, 0)
                    s_ext = a_ext.sum()
                    if s_ext > 1e-10:
                        candidates.append(a_ext / s_ext)

        # Evaluate candidates
        best_local_obj = best_obj
        best_local_a = best_a.copy()
        for a_try in candidates:
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()

        if best_local_obj < best_obj - 1e-11:
            # Record for Anderson
            history_a.append(a.copy())
            history_g.append(g.copy())
            if len(history_a) > anderson_window:
                history_a.pop(0)
                history_g.pop(0)

            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a
            stall = 0
            if verbose and it % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it{it}: obj={best_obj:.9f}, M_lp={M_lp:.6f}, t={elapsed:.0f}s/{time_budget:.0f}s")
        else:
            stall += 1
            if stall % 3 == 0:
                # Perturb: use current LP solution as perturbation direction
                scale = min(0.08, 0.01 * (stall // 3 + 1))
                pert = g - a
                pert = pert / (np.linalg.norm(pert) + 1e-10)
                a = a + scale * pert
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()

            if stall % 7 == 0:
                # Random perturbation
                scale = min(0.1, 0.02 * (stall // 7 + 1))
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-10:
                    a = a / s

            if stall > 50:
                if verbose:
                    print(f"    LP: stuck at it{it}, obj={best_obj:.9f}, restarting perturbation")
                # Big restart perturbation
                a = best_a * np.exp(rng.normal(0, 0.15, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                history_a = []
                history_g = []
                recent_gs = []

            if stall > 100:
                break

    return best_a


def weighted_lp_iteration(a, time_budget, verbose=False, seed=43):
    """LP iteration with peak-weighted LP to target heavy peaks."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        # Compute current peaks for weighting
        conv_a = compute_conv(a)
        max_conv = float(np.max(conv_a))
        # Exponential weights emphasizing top peaks
        tau = 0.15 * max_conv
        w = np.exp(conv_a / tau)
        w = w / w.sum()

        g, M_lp = solve_lp(a, weighted=True, weights=w)
        if g is None:
            # Fall back to unweighted
            g, M_lp = solve_lp(a)
            if g is None:
                break

        it += 1

        # Blend
        best_local_obj = best_obj
        best_local_a = best_a.copy()
        for t in [0.1, 0.2, 0.4, 0.6, 0.8]:
            a_try = blend_linear(a, g, t)
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()

        if best_local_obj < best_obj - 1e-11:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a
            stall = 0
            if verbose and it % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    WLP it{it}: obj={best_obj:.9f}, t={elapsed:.0f}s")
        else:
            stall += 1
            if stall > 5:
                scale = min(0.06, 0.01 * stall / 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
            if stall > 40:
                break

    return best_a


def smoothed_gradient_descent(a, time_budget, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov smoothed gradient descent on the max-conv objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    beta = 0.9
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.95))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        # Compute smoothed gradient using log-sum-exp
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_c))
        p = exp_vals / float(np.sum(exp_vals))

        # Gradient of sum_k p_k * conv(a,a)[k] w.r.t. a
        # d conv(a,a)[k] / da[i] = 2*a[k-i] (if valid)
        from scipy.fft import fft, ifft
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_fft = fft(a, N)
        g_full = ifft(P * np.conj(A_fft)).real
        g = 2.0 * g_full[:n]

        # Nesterov step
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr *= 0.9998

        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it

        if verbose and it % 2000 == 0:
            print(f"    SGD it{it}: obj={best_obj:.9f}, tau={tau:.1f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-8))
        a = a / a.sum()
    elif mode == 4:
        # Cosine window
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 5:
        # Triangular
        a = 1.0 - 2.0 * np.abs(np.arange(n) / n - 0.5)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP-based mirror descent with Anderson acceleration,
    geometric blending, peak-weighted LP, multi-n exploration, and
    smoothed gradient descent for final polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent if available
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.9f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.9f}")

    n_current = len(best_seq)

    # === Phase 1: Core LP iteration (40% of time) ===
    phase1_budget = min((deadline - time.monotonic()) * 0.40, 550.0)
    print(f"\n=== Phase 1: Core LP (budget={phase1_budget:.0f}s) ===")
    if phase1_budget > 60:
        a_lp = lp_iteration_core(best_seq, time_budget=phase1_budget,
                                 use_anderson=True, verbose=True, seed=42)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1 LP: obj={best_score:.9f} **")
        else:
            print(f"  Phase1 LP: obj={obj_lp:.9f}")

    # === Phase 2: Weighted LP iteration (20% of remaining) ===
    remaining = deadline - time.monotonic()
    phase2_budget = min(remaining * 0.25, 300.0)
    print(f"\n=== Phase 2: Weighted LP (budget={phase2_budget:.0f}s) ===")
    if phase2_budget > 40:
        a_wlp = weighted_lp_iteration(best_seq, time_budget=phase2_budget,
                                      verbose=True, seed=43)
        obj_wlp = compute_obj(a_wlp)
        if obj_wlp < best_score:
            best_score = obj_wlp
            best_seq = a_wlp.copy()
            n_current = len(best_seq)
            print(f"  Phase2 WLP: obj={best_score:.9f} **")
        else:
            print(f"  Phase2 WLP: obj={obj_wlp:.9f}")

    # === Phase 3: Multi-n exploration (20% of remaining) ===
    remaining = deadline - time.monotonic()
    phase3_budget = min(remaining * 0.25, 250.0)
    print(f"\n=== Phase 3: Multi-n exploration (budget={phase3_budget:.0f}s) ===")
    if phase3_budget > 60:
        n_options = sorted(set([
            max(400, n_current - 200),
            max(500, n_current - 100),
            n_current + 100,
            n_current + 200,
            n_current + 400,
            1000, 1200
        ]))
        n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_current) >= 100]
        n_options = n_options[:5]

        per_n_budget = phase3_budget / max(1, len(n_options))
        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(per_n_budget, remaining * 0.4)
            if alloc < 30:
                break

            a_try = resample_sequence(best_seq, n_try)
            a_try = lp_iteration_core(a_try, time_budget=alloc,
                                      use_anderson=True, verbose=False, seed=seed + n_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_current = n_try
                print(f"  ** n={n_try}: obj={best_score:.9f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.9f}")

    # === Phase 4: Smoothed gradient descent (15% of remaining) ===
    remaining = deadline - time.monotonic()
    phase4_budget = min(remaining * 0.2, 200.0)
    print(f"\n=== Phase 4: Smoothed GD (budget={phase4_budget:.0f}s) ===")
    if phase4_budget > 30:
        a_sgd = smoothed_gradient_descent(best_seq, time_budget=phase4_budget,
                                          tau_init=20.0, tau_end=600.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase4 SGD: obj={best_score:.9f} **")
        else:
            print(f"  Phase4 SGD: obj={obj_sgd:.9f}")

    # === Phase 5: Final LP polish (remaining) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 5: Final LP Polish (budget={remaining:.0f}s) ===")
    if remaining > 30:
        # Second core LP run with different seed
        alloc = min(remaining * 0.5, 200.0)
        a_lp2 = lp_iteration_core(best_seq, time_budget=alloc,
                                  use_anderson=True, verbose=True, seed=99)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Final LP: obj={best_score:.9f} **")
        else:
            print(f"  Final LP: obj={obj_lp2:.9f}")

    # Final quick GD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 100.0)
        a_final = smoothed_gradient_descent(best_seq, time_budget=alloc,
                                            tau_init=100.0, tau_end=2000.0)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final GD: obj={best_score:.9f} **")

    # Validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.9f}, time={elapsed:.1f}s ===")

    return result
