"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: LP power iteration with adaptive blending, frequency-domain flattening,
and smoothed gradient polishing. Key innovations: direct LP fixed-point iteration
(faster convergence than conservative blending), IAPA-style frequency flattening,
and aggressive multi-n exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.sparse import csr_matrix
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_sparse(a):
    """Build sparse convolution matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(2 * n - 1):
            # conv(a, g)[k] = sum_i a[i] * g[k-i]
            # For column j (g[j]), the contribution to row k is a[k-j] if 0 <= k-j < n
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix_sparse(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.CBC, cp.SCS, cp.ECOS, cp.PDLP, cp.SCIP, cp.GLPK]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def freq_flatten(a, alpha=0.7, num_iter=5, verbose=False):
    """
    IAPA-style frequency domain flattening.
    Iteratively modifies the amplitude spectrum to be flatter.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(num_iter):
        # Compute spectrum
        F = fft(a, 2 * n)
        amp = np.abs(F)
        phase = np.angle(F)
        
        # Flatten: new_amp = amp^alpha (compresses dynamic range)
        max_amp = np.max(amp)
        if max_amp < 1e-15:
            break
        amp_norm = amp / max_amp
        new_amp = amp_norm ** alpha
        
        # Reconstruct
        F_new = new_amp * np.exp(1j * phase)
        a_new = np.real(ifft(F_new, 2 * n))[:n]
        
        # Alternative: use 1/(1+amp^2) which is more aggressive flattening
        # new_amp2 = 1.0 / (1.0 + (amp / max_amp) ** 2)
        
        # Project to simplex
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose:
                print(f"      FreqFlatten it{it}: obj={best_obj:.8f}")
        else:
            # Try with lower alpha
            break
    
    return best_a


def smoothed_grad_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def nesterov_smoothed_descent(a, time_budget, tau_init=50.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-6))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        g = smoothed_grad_fft(a, tau)

        # Barzilai-Borwein step size
        if prev_g is not None and it % 2 == 0:
            s_vec = a - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 5.0 / np.sqrt(n))

        a_new = a - lr * g
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)

        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it

        if verbose and it % 5000 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}")

    return best_a


def lp_power_iteration(a, time_budget, verbose=False):
    """
    LP power iteration: repeatedly solve LP and update a.
    Uses adaptive blending with line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    iter_count = 0
    rng = np.random.default_rng(123)
    consecutive_fails = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            if verbose:
                print(f"    LP iter{iter_count}: solver failed")
            break

        # Adaptive blending: try multiple t values
        # As we progress, use more aggressive blending
        if progress < 0.2:
            t_candidates = [0.5, 0.7, 0.9, 1.0]
        elif progress < 0.5:
            t_candidates = [0.3, 0.5, 0.7, 1.0]
        else:
            t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]

        best_t_obj = best_obj
        best_t_val = 0.3
        best_a_local = best_a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        iter_count += 1
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            consecutive_fails = 0
            if verbose and iter_count % 3 == 0:
                print(f"    LP iter{iter_count}: obj={best_obj:.8f}, t={best_t_val:.2f}, M={M_lp:.6f}")
        else:
            consecutive_fails += 1
            # Perturb to escape
            if consecutive_fails <= 5:
                scale = 0.02 * consecutive_fails
            elif consecutive_fails <= 15:
                scale = 0.05
            else:
                scale = 0.1
            
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s < 1e-10:
                break
            a = a / s
            consecutive_fails = 0
            
            # Also try frequency flattening
            a_ff = freq_flatten(best_a, alpha=0.5, num_iter=3)
            obj_ff = compute_obj(a_ff)
            if obj_ff < best_obj:
                best_obj = obj_ff
                best_a = a_ff.copy()
                a = a_ff.copy()
            elif consecutive_fails > 10:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Hann window
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
        a = a / a.sum()
    elif mode == 3:
        # Triangular window
        a = 1 - np.abs(np.arange(n) - n//2) / (n//2 + 1)
        a = a / a.sum()
    elif mode == 4:
        # Spectral factor: start from random phase, take amplitude
        F = fft(rng.normal(0, 1, n) + 1j * rng.normal(0, 1, n), 2 * n)
        a = np.real(ifft(np.abs(F) ** 0.5 * np.exp(1j * np.angle(F)), 2 * n))[:n]
        a = np.maximum(a, 0)
        if a.sum() < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / a.sum()
    elif mode == 5:
        # Bell-shaped
        x = np.arange(n) / n
        a = np.exp(-4 * (x - 0.5) ** 2)
        a = a / a.sum()
    elif mode == 6:
        # Multi-peak
        a = np.zeros(n)
        for center in [0.25, 0.5, 0.75]:
            a += np.exp(-((np.arange(n)/n - center) ** 2) / 0.005)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Aggressive LP power iteration with adaptive blending and frequency-domain
flattening (IAPA), combined with smoothed gradient polishing and multi-n exploration.
Key innovations: direct LP fixed-point iteration, spectral amplitude compression,
and aggressive perturbation for escaping local minima.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.sparse import csr_matrix
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.CBC, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def freq_flatten(a, alpha=0.7, num_iter=10, verbose=False):
    """IAPA-style frequency domain flattening."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(num_iter):
        F = fft(a, N)
        amp = np.abs(F)
        phase = np.angle(F)
        
        max_amp = np.max(amp)
        if max_amp < 1e-15:
            break
        amp_norm = amp / max_amp
        new_amp = amp_norm ** alpha
        
        F_new = new_amp * np.exp(1j * phase)
        a_new = np.real(ifft(F_new, N))[:n]
        
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose:
                print(f"      FreqFlatten it{it}: obj={best_obj:.8f}, alpha={alpha}")
        else:
            break
    
    return best_a


def smoothed_grad_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)

    exp_vals = np.exp(tau * (conv - max_conv))

        N <<= 1
