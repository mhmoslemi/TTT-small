Multi-strategy optimizer for step function auto-convolution minimization.
Uses LP relaxation for direction, entropic mirror descent for smooth exploration,
peak deflation for escaping local minima, and adaptive annealing for final polish.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_cross(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CVXOPT, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def smoothed_grad(a, tau):
    """Compute gradient of log-sum-exp smoothed max of auto-conv."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    N = 1
    while N < 2 * (2 * n - 1):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv, p

    s = float(np.sum(a))

        return a

    best_obj = compute_obj(a)

    beta = 0.85

    v = np.zeros(n)
    it = 0
    best_it = 0
    stall_count = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)
        
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        g, conv, p = smoothed_grad(y, tau)
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))

            v = np.zeros(n)

            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * (0.5 ** min(stall_count // 10000, 5))
            stall_count = 0
        
        if verbose and it % 5000 == 0:
            print(f"    Nesterov it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")

def entropic_mirror_descent(a, time_budget, tau=50.0, verbose=False):
    """Mirror descent with entropic regularization on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = 1.0 / np.sqrt(n)
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        g, conv, p = smoothed_grad(a, tau)
        
        # Mirror descent: a_new[i] proportional to a[i] * exp(-eta * g[i])
        log_a = np.log(np.maximum(a, 1e-15))
        log_a_new = log_a - eta * g
        # Shift for numerical stability
        log_a_new = log_a_new - np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Decrease tau slowly
            tau = max(5.0, tau * 0.999)
        else:
            stall += 1
            eta *= 0.8
            if eta < 1e-10:
                eta = 1.0 / np.sqrt(n)
                tau = max(5.0, tau * 0.9)
        
        it += 1
        if stall > 200:
            break
        
        if verbose and it % 1000 == 0:

def peak_deflation(a, time_budget, n_peaks=5, verbose=False):
    """Peak deflation: iteratively reduce dominant peaks by redistributing mass."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    it = 0
    stall = 0

        conv = compute_conv(a)
        # Find top peaks
        peak_indices = np.argsort(conv)[::-1][:n_peaks]
        max_k = peak_indices[0]
        
        # Compute pressure: how much each position contributes to the peaks
        pressure = np.zeros(n)
        for k in peak_indices:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += a[i] * a[j]
                    pressure[j] += a[i] * a[j]
        
        # Find low-pressure positions (good targets for receiving mass)
        low_pressure = np.argsort(pressure)[:n // 4]
        high_pressure = np.argsort(pressure)[::-1][:n // 4]
        
        # Try moving mass from high to low pressure
        improved = False
        src_idx = rng.choice(high_pressure)
        if a[src_idx] < 1e-8:
            stall += 1
            if stall > 100:
                break
            continue
        
        dst_idx = rng.choice(low_pressure)
        if src_idx == dst_idx:
            continue
        
        # Try several step sizes
        for frac in [0.1, 0.05, 0.02, 0.01]:
            delta = frac * a[src_idx]
            if delta > a[src_idx] * 0.9:
                continue
            a_try = a.copy()
            a_try[src_idx] -= delta
            a_try[dst_idx] += delta
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:

            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            if stall > 200:
                break
        
        it += 1

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])

    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):

        a = np.ones(n) / n

    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)

            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 4, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    else:

        a = 0.5 + 0.3 * np.cos(2 * x) + 0.2 * np.sin(3 * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    return a

Multi-strategy optimizer for minimizing 2*n*max(auto_conv(a,a))/(sum(a))^2.

def compute_conv(a):
    """Full linear auto-convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_cross(a):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g <= M_var, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers = [cp.CVXOPT, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M_var.value)
        except Exception:
            continue
    return None, None


def smoothed_grad(a, tau):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*conv(a,a))) w.r.t. a.
    Also returns the softmax weights p and the convolution.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 2 * (2 * n - 1):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv, p


def nesterov_smoothed(a, time_budget, tau_init=20.0, tau_end=500.0, oscillate=False, verbose=False):
    """
    Nesterov accelerated gradient descent on smoothed max of auto-conv.
    Optionally oscillates tau to escape local minima.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    lr = 2.5 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget * 0.85, 1e-6))

        if oscillate:
            # Oscillate tau to escape local minima
            base_tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)
            tau = base_tau * (1.0 + 0.5 * np.sin(0.02 * it))
            tau = max(5.0, tau)
        else:
            tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv, p = smoothed_grad(y, tau)

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall_count = 0
        else:
            stall_count += 1

        a = a_new
        it += 1
        lr = max(1e-9, lr * 0.99995)

        # Adaptive restart on long stall
        if stall_count > 8000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * (0.5 ** min(stall_count // 15000, 4))
            stall_count = 0

        if verbose and it % 5000 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")

    return best_a


def entropic_mirror_descent(a, time_budget, tau=40.0, verbose=False):
    """
    Mirror descent with entropic regularization on the simplex.
    Uses the geometry of the simplex for potentially better exploration.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = 1.5 / np.sqrt(n)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        g, conv, p = smoothed_grad(a, tau)

        # Mirror descent update (exponential family)
        log_a = np.log(np.maximum(a, 1e-15))
        log_a_new = log_a - eta * g
        log_a_new = log_a_new - np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            tau = max(5.0, tau * 0.9995)
        else:
            stall += 1
            eta *= 0.85
            if eta < 1e-10:
                eta = 1.5 / np.sqrt(n)
                tau = max(5.0, tau * 0.95)
                # Perturb
                rng = np.random.default_rng(it)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / a.sum()

        it += 1
        if stall > 300:
            break

        if verbose and it % 1000 == 0:
            print(f"    EMD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, eta={eta:.6f}")

    return best_a


def peak_deflation(a, time_budget, n_peaks=5, verbose=False):
    """
    Peak deflation: identify dominant peaks and redistribute mass away from
    positions that contribute most to those peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        # Compute pressure vector: contribution of each position to the peaks
        pressure = np.zeros(n)
        for k in peak_indices:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += a[i] * a[j]

        # Sort positions by pressure
        sorted_idx = np.argsort(pressure)
        low_pressure = sorted_idx[:max(1, n // 5)]
        high_pressure = sorted_idx[::-1][:max(1, n // 5)]

        # Try moving mass from high to low pressure
        improved = False
        for _ in range(10):
            src_idx = int(rng.choice(high_pressure))
            if a[src_idx] < 1e-8:
                continue
            dst_idx = int(rng.choice(low_pressure))
            if src_idx == dst_idx:
                continue

            for frac in [0.08, 0.04, 0.02, 0.01, 0.005]:
                delta = frac * a[src_idx]
                if delta >= a[src_idx]:
                    continue
                a_try = a.copy()
                a_try[src_idx] -= delta
                a_try[dst_idx] += delta
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            if improved:
                break

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 150:
                break

        if verbose and it % 500 == 0:
            print(f"    PD it{it}: obj={best_obj:.8f}")

    return best_a


def simulated_annealing_polish(a, time_budget, verbose=False):
    """Simulated annealing for final fine-tuning."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(54321)

    T = 0.03
    T_cool = 0.9995
    it = 0

    while time.monotonic() - start < time_budget:
        batch = 50
        for _ in range(batch):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            frac = rng.uniform(0.01, 0.08)
            delta = frac * a[i]
            if delta >= a[i]:
                continue
            a_try = a.copy()
            a_try[i] -= delta
            a_try[j] += delta
            a_try = a_try / a_try.sum()

            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

            cur_obj = compute_obj(a)
            if obj_try < cur_obj or rng.random() < np.exp(-(obj_try - cur_obj) / max(T, 1e-10)):
                a = a_try

        T *= T_cool
        it += batch
        if verbose and it % 500 == 0:
            print(f"    SA it{it}: best={best_obj:.8f}, T={T:.6f}")

    return best_a


def lp_polish(a, time_budget, verbose=False):
    """LP-based polishing using cross-convolution direction with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_cross(a)
        if g_lp is None:
            break

        # Multi-point line search
        best_t = 0.0
        best_t_obj = best_obj
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t > 0 and best_t_obj < best_obj - 1e-14:
            a = (1 - best_t) * a + best_t * g_lp
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_obj = best_t_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break

        it += 1
        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_score:.8f}, M_lp={M_lp:.6f}, t={best_t:.3f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.25 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Wavy pattern
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * rng.uniform(2, 6) * x) + 0.2 * np.sin(2 * np.pi * rng.uniform(5, 12) * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.3 * np.cos(2 * x) + 0.15 * np.sin(4 * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    return a


def coarse_to_fine(a_coarse, n_fine, time_budget, verbose=False):
    """
    Multi-scale: use coarse solution to initialize fine-scale optimization.
    Upsample and then optimize.
    """
    a_fine = resample_sequence(a_coarse, n_fine)
    a_fine = a_fine / a_fine.sum()
    a_fine = nesterov_smoothed(a_fine, time_budget=time_budget,
                                tau_init=10.0, tau_end=200.0)
    return a_fine


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization:
    1. LP-based direction finding (fixed solver)
    2. Nesterov smoothed GD with oscillating tau
    3. Entropic mirror descent (new geometry)
    4. Peak deflation (new escape mechanism)
    5. Multi-length exploration
    6. Diverse restarts with full pipeline
    7. Final intensive polish (SA + PD + Nesterov)
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent available: {e}")

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: LP-based polishing ===
    print(f"\n=== Phase 1: LP Polish ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.18, 200.0)
    if p1_budget > 30:
        a_res = lp_polish(best_seq, time_budget=p1_budget, verbose=True)
        obj_res = compute_obj(a_res)
        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            print(f"  Phase 1 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 LP: obj={obj_res:.8f}")

    # === Phase 2: Nesterov smoothed GD (oscillating tau) ===
    print(f"\n=== Phase 2: Nesterov GD (oscillating) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 300.0)
    if p2_budget > 30:
        a_res = nesterov_smoothed(best_seq, time_budget=p2_budget,
                                   tau_init=15.0, tau_end=400.0,
                                   oscillate=True, verbose=True)
        obj_res = compute_obj(a_res)
        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            print(f"  Phase 2 Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 Nesterov: obj={obj_res:.8f}")

    # === Phase 3: Entropic mirror descent ===
    print(f"\n=== Phase 3: Entropic Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 120.0)
    if p3_budget > 20:
        a_res = entropic_mirror_descent(best_seq, time_budget=p3_budget, tau=35.0, verbose=True)
        obj_res = compute_obj(a_res)
        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            print(f"  Phase 3 EMD: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3 EMD: obj={obj_res:.8f}")

    # === Phase 4: Peak deflation ===
    print(f"\n=== Phase 4: Peak Deflation ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.12, 120.0)
    if p4_budget > 15:
        a_res = peak_deflation(best_seq, time_budget=p4_budget, n_peaks=5, verbose=True)
        obj_res = compute_obj(a_res)
        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            print(f"  Phase 4 PD: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4 PD: obj={obj_res:.8f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    n_options = [200, 300, 400, 500, 600, 700, 900, 1000, 1200, 1500]
    n_options = [n for n in n_options if abs(n - n_best) > 80]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        alloc = min(remaining * 0.2, 70.0)
        if alloc > 15:
            a_try = nesterov_smoothed(a_try, time_budget=alloc,
                                       tau_init=10.0, tau_end=200.0)
            a_try = peak_deflation(a_try, time_budget=min(remaining * 0.05, 20.0), n_peaks=3)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break

        mode = r % 7
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.35 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.25, 120.0)
        if alloc > 30:
            # Run a mini-pipeline
            t1 = alloc * 0.5
            t2 = alloc * 0.3
            t3 = alloc * 0.2
            a_r = nesterov_smoothed(a_r, time_budget=t1, tau_init=15.0, tau_end=300.0)
            remaining_inner = deadline - time.monotonic()
            if remaining_inner > 20:
                a_r = peak_deflation(a_r, time_budget=min(t2, remaining_inner * 0.5), n_peaks=4)
            remaining_inner = deadline - time.monotonic()
            if remaining_inner > 15:
                a_r = entropic_mirror_descent(a_r, time_budget=min(t3, remaining_inner * 0.6), tau=30.0)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Nesterov with high tau
    if remaining > 50:
        alloc = min(remaining * 0.25, 100.0)
        a_res = nesterov_smoothed(best_seq, time_budget=alloc,
                                   tau_init=50.0, tau_end=800.0,
                                   oscillate=False, verbose=True)
        obj_res = compute_obj(a_res)
        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            print(f"  Final Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Final Nesterov: obj={obj_res:.8f}")

    # Peak deflation
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 50.0)
        a_res = peak_deflation(best_seq, time_budget=alloc, n_peaks=3)
        obj_res = compute_obj(a_res)
        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            print(f"  Final PD: obj={best_score:.8f} **")
        else:
            print(f"  Final PD: obj={obj_res:.8f}")

    # Simulated annealing
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.25, 60.0)
        a_res = simulated_annealing_polish(best_seq, time_budget=alloc, verbose=True)
        obj_res = compute_obj(a_res)
        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            print(f"  Final SA: obj={best_score:.8f} **")
        else:
            print(f"  Final SA: obj={obj_res:.8f}")

    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 40.0)
        a_res = lp_polish(best_seq, time_budget=alloc)
        obj_res = compute_obj(a_res)
        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_res:.8f}")

    # Final Nesterov sweep
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 40.0)
        a_res = nesterov_smoothed(best_seq, time_budget=alloc,
                                   tau_init=100.0, tau_end=1000.0)
        obj_res = compute_obj(a_res)
        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            print(f"  Final Sweep: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
