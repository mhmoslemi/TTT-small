"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated smoothed gradient descent with adaptive tau,
LP polishing, subgradient descent, and multi-scale exploration.
Key improvements: fixed LP solver bug, better NAGD implementation,

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*c[k])) where c = conv(a,a).
    This equals sum_k p[k] * dc[k]/da where p[k] = exp(tau*c[k])/sum(exp(tau*c)).
    dc[k]/da[m] = 2*a[k-m] for valid k-m.
    So grad[m] = 2 * sum_k p[k] * a[k-m] = 2 * (cross-correlation of p and a).
    """
    n = len(a)
    c = compute_conv(a)
    max_c = float(np.max(c))
    exp_c = np.exp(tau * (c - max_c))
    p_sum = float(np.sum(exp_c))
    if p_sum < 1e-300:
        p = np.ones(len(c)) / len(c)
    else:
        p = exp_c / p_sum

    N = 1
    while N < 4 * n:
        N <<= 1

    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    grad = 2.0 * cross[:n]
    return grad, c


def nesterov_smoothed_descent(a, time_budget, tau_init=10.0, tau_max=2000.0,
                               beta=0.9, lr_init=None, verbose=False, seed=0):

    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    if lr_init is None:
        lr = 5.0 / np.sqrt(n)
    else:
        lr = lr_init

    v = np.zeros(n)
    it = 0
    best_it = 0
    stall_count = 0
    total_iters = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_max / tau_init) ** (progress ** 0.6)

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            v = np.zeros(n)
        else:
            y = y / s_y

        # Compute gradient
        grad, conv = smoothed_gradient(y, tau)

        # Adaptive learning rate (Barzilai-Borwein-inspired)

            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-18 and abs(sy) > 1e-14:
                bb_lr = ss / abs(sy)
                bb_lr = np.clip(bb_lr, 1e-8, 20.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb_lr

        nesterov_smoothed_descent._prev_y = y.copy()
        nesterov_smoothed_descent._prev_grad = grad.copy()

        # Gradient step with Nesterov momentum
        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        # Evaluate

        total_iters += 1

        if obj_new < best_obj - 1e-14:
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a_new.copy()
                best_it = total_iters
                stall_count = 0
            else:
                best_obj = obj_new
                best_a = a_new.copy()
                best_it = total_iters
                stall_count = 0
            a = a_new
        else:
            a = a_new
            stall_count += 1
            if stall_count > 500:
                # Adaptive perturbation to escape
                scale = 0.01 + 0.02 * min(stall_count / 500.0, 1.0)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = float(np.sum(a))
                if s_a > 1e-10:
                    a = a / s_a
                else:
                    a = best_a.copy()
                v = np.zeros(n)
                lr = 5.0 / np.sqrt(n)
                stall_count = 0

        # Decay learning rate slowly
        lr = max(1e-6, lr * 0.9999)

                a = a_sg
                v = np.zeros(n)

    nesterov_smoothed_descent._prev_y = None
    nesterov_smoothed_descent._prev_grad = None
    return best_a

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a, active_indices=None):
    """
    Solve LP to find optimal direction.
    If active_indices is None: min M s.t. A@g <= M, sum(g)=1, g>=0
    If active_indices given: min sum(A[active,:] @ g) s.t. sum(g)=1, g>=0
    Returns normalized g or None.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)

    if active_indices is not None:
        active_A = A[active_indices, :]
        obj = cp.sum(active_A @ g)
        constraints = [cp.sum(g) == 1]
    else:
        M = cp.Variable()
        obj = M
        constraints = [A @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(obj), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    result = g_val / s
                    if active_indices is None:
                        return result, float(M.value)
                    else:
                        return result, None
        except Exception:
            continue
    if active_indices is None:
        return None, None
    return None, None


def lp_polish(a, time_budget=100.0, verbose=False):

    with golden-section line search for blending.

    best_obj = compute_obj(best_a)

    rng = np.random.default_rng(12345)

    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        # Stage 1: min max
        g1, M1 = solve_lp_direction(a)
        if g1 is None:
            break

        # Find active constraints
        conv_ag = build_conv_matrix(a) @ g1
        max_val = float(np.max(conv_ag))
        tol = 0.02 * max(max_val, 1e-10)
        active = np.where(conv_ag >= max_val - tol)[0]

        if len(active) == 0:
            active = np.array([int(np.argmax(conv_ag))])

        # Stage 2: min sum at active
        g2, _ = solve_lp_direction(a, active_indices=active)

        # Choose direction
        if g2 is not None:
            peak_g1 = float(np.max(build_conv_matrix(a) @ g1))
            peak_g2 = float(np.max(build_conv_matrix(a) @ g2))
            if peak_g2 < peak_g1:
                g_dir = g2
            else:
                g_dir = (g1 + g2) / 2.0

            g_dir = g1

        phi = (np.sqrt(5) - 1) / 2
        lo, hi = 0.0, 1.0
        t_best = 0.0
        obj_best = best_obj

            a2 = np.maximum(a2, 0)

            f2 = compute_obj(a2 / s2) if s2 > 1e-10 else np.inf

                hi = t2
            else:
                lo = t1

        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
        else:
            obj_new = np.inf
            a_new = best_a.copy()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 20:

                stall = 0
                if it > 40:
                    break

        it += 1
        if verbose and it % 2 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, M1={M1:.6f}, |S|={len(active)}, t={t_best:.4f}")

    return best_a


def subgradient_polish(a, time_budget=60.0, verbose=False):
    """
    Subgradient descent targeting the maximum peak directly.
    At each step, move mass away from the peak's contributing indices.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    lr = 2.0 / np.sqrt(n)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))
        max_val = float(conv[peak_idx])

        # Subgradient of max: for the active peak
        # dc[k]/da[m] = 2*a[k-m] for valid k-m
        sub_grad = np.zeros(n)
        for m in range(n):
            j = peak_idx - m
            if 0 <= j < n:
                sub_grad[m] = 2.0 * a[j]

        # Also include nearby peaks (within 10% of max)
        near_peaks = np.where(conv >= 0.9 * max_val)[0]
        for pk in near_peaks:
            if pk == peak_idx:
                continue
            for m in range(n):
                j = pk - m
                if 0 <= j < n:
                    sub_grad[m] += 0.5 * 2.0 * a[j]

        # Normalize subgradient
        sg_norm = float(np.linalg.norm(sub_grad))
        if sg_norm < 1e-10:
            break

        sub_grad = sub_grad / sg_norm

        # Step with decreasing learning rate
        a_new = a - lr * sub_grad
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 100:
                # Try multi-peak subgradient
                top_k = min(5, len(conv))
                top_indices = np.argsort(conv)[-top_k:]
                multi_grad = np.zeros(n)
                for pk in top_indices:
                    for m in range(n):
                        j = pk - m
                        if 0 <= j < n:
                            multi_grad[m] += 2.0 * a[j]
                mg_norm = float(np.linalg.norm(multi_grad))
                if mg_norm > 1e-10:
                    a_try = project_simplex(a - lr * multi_grad / mg_norm)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        stall = 0
                        continue
                break

        lr *= 0.998

        if verbose and stall % 200 == 0:
            print(f"    SG it{stall}: obj={best_obj:.8f}, peak={max_val:.8f}")

    return best_a


def peak_redistribute(a, time_budget=60.0, verbose=False):
    """
    Targeted redistribution: for the top peaks, find the best source-target
    pair to move a small amount of mass and reduce the peak.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(4242)

    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        peak_idx = int(np.argmax(conv))

        # Find contributing indices to the peak
        i_start = max(0, peak_idx - (n - 1))
        i_end = min(n - 1, peak_idx)

        # Top contributors
        contribs = []
        for i in range(i_start, i_end + 1):
            j = peak_idx - i
            if 0 <= j < n:
                contribs.append((a[i] * a[j], i, j))
        contribs.sort(reverse=True)

        # Try to reduce top peaks by moving mass
        improved = False
        deltas = [0.03, 0.02, 0.01, 0.005]

        for cval, src, dst in contribs[:5]:
            if src == dst:
                continue
            for delta_frac in deltas:
                delta = delta_frac * a[src]
                if delta < 1e-10 or delta >= a[src]:
                    continue

                # Try multiple targets
                n_targets = max(10, n // 50)
                targets = rng.integers(0, n, size=n_targets)
                # Also try structured targets
                structured = [
                    (src + n // 3) % n,
                    (src + 2 * n // 3) % n,
                    (src + n // 2) % n,
                    (n - 1 - src) % n,
                ]
                targets = np.concatenate([targets, np.array(structured)])

                for tgt in targets:
                    tgt = int(tgt) % n
                    if tgt == src:
                        continue
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[tgt] += delta
                    if a_try[src] < 0:
                        continue
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [max(1, n // 8), max(1, n // 8)]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        # Symmetric initialization
        half_n = n // 2
        a_half = rng.uniform(0.1, 1.0, half_n)
        a = np.concatenate([a_half, a_half[::-1]])
        if n % 2 == 1:
            a = np.concatenate([a_half, [a_half[0]], a_half[::-1]])
            a = a[:n]
        a = a / a.sum()
    elif mode == 7:
        # Low-pass filtered random
        a = rng.uniform(0, 1, n)
        # Simple moving average
        k = max(1, n // 20)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization combining NAGD, LP polishing, subgradient,
    peak redistribution, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent available: {e}")

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    def remaining_time():
        return max(0, deadline - time.monotonic())

    # === Phase 1: NAGD (primary optimizer) ===
    print(f"\n=== Phase 1: NAGD ===")
    p1_budget = min(remaining_time() * 0.3, 400.0)
    if p1_budget > 30:
        nesterov_smoothed_descent._prev_y = None
        nesterov_smoothed_descent._prev_grad = None
        a_nag = nesterov_smoothed_descent(best_seq, time_budget=p1_budget,
                                          tau_init=5.0, tau_max=500.0,
                                          beta=0.9, verbose=True, seed=seed)
        obj_nag = compute_obj(a_nag)
        if obj_nag < best_score:
            best_score = obj_nag
            best_seq = a_nag.copy()
            print(f"  NAGD: obj={best_score:.8f} **")
        else:
            print(f"  NAGD: obj={obj_nag:.8f}")

    # === Phase 2: LP Polishing ===
    print(f"\n=== Phase 2: LP Polish ===")
    p2_budget = min(remaining_time() * 0.2, 200.0)
    if p2_budget > 30:
        a_lp = lp_polish(best_seq, time_budget=p2_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 3: High-tau NAGD ===
    print(f"\n=== Phase 3: High-tau NAGD ===")
    p3_budget = min(remaining_time() * 0.2, 200.0)
    if p3_budget > 30:
        nesterov_smoothed_descent._prev_y = None
        nesterov_smoothed_descent._prev_grad = None
        a_ht = nesterov_smoothed_descent(best_seq, time_budget=p3_budget,
                                         tau_init=50.0, tau_max=3000.0,
                                         beta=0.85, verbose=True, seed=seed + 1)
        obj_ht = compute_obj(a_ht)
        if obj_ht < best_score:
            best_score = obj_ht
            best_seq = a_ht.copy()
            print(f"  HT-NAGD: obj={best_score:.8f} **")
        else:
            print(f"  HT-NAGD: obj={obj_ht:.8f}")

    # === Phase 4: Subgradient + Peak Redistribution ===
    print(f"\n=== Phase 4: Subgradient + Peak Redistribution ===")
    remaining = remaining_time()
    p4a_budget = min(remaining * 0.15, 80.0)
    p4b_budget = min(remaining * 0.1, 60.0)

    if p4a_budget > 15:
        a_sg = subgradient_polish(best_seq, time_budget=p4a_budget, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  SubGrad: obj={best_score:.8f} **")
        else:
            print(f"  SubGrad: obj={obj_sg:.8f}")

    if p4b_budget > 15:
        a_pr = peak_redistribute(best_seq, time_budget=p4b_budget, verbose=True)
        obj_pr = compute_obj(a_pr)
        if obj_pr < best_score:
            best_score = obj_pr
            best_seq = a_pr.copy()
            print(f"  PeakRedist: obj={best_score:.8f} **")
        else:
            print(f"  PeakRedist: obj={obj_pr:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_current = len(best_seq)

    n_options = sorted(set([
        max(200, n_current - 200),
        n_current + 100,
        n_current + 200,
        400, 600, 1000, 1200, 1500, 2000
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000 and abs(n - n_current) > 50]

    # Sort by proximity to current best first, then by diversity
    n_options.sort(key=lambda x: abs(x - n_current))

    for n_try in n_options[:5]:
        remaining = remaining_time()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)
        budget_here = min(remaining * 0.2, 80.0)
        if budget_here > 20:
            # Quick NAGD
            nesterov_smoothed_descent._prev_y = None
            nesterov_smoothed_descent._prev_grad = None
            a_try = nesterov_smoothed_descent(a_try, time_budget=budget_here,
                                              tau_init=10.0, tau_max=500.0,
                                              beta=0.9, seed=seed + n_try % 100)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    n_best = len(best_seq)

    for r in range(4):
        remaining = remaining_time()
        if remaining < 60:
            break

        mode = (r + 1) % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with current best
        blend = 0.4
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        budget_here = min(remaining * 0.2, 80.0)
        if budget_here > 20:
            half_budget = budget_here / 2
            nesterov_smoothed_descent._prev_y = None
            nesterov_smoothed_descent._prev_grad = None
            a_r = nesterov_smoothed_descent(a_r, time_budget=half_budget,
                                            tau_init=10.0, tau_max=300.0,
                                            beta=0.9, seed=seed + 100 + r)

            remaining = remaining_time()
            if remaining > 20:
                a_r = lp_polish(a_r, time_budget=min(remaining * 0.3, 40.0))

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = remaining_time()

    # NAGD with very high tau
    if remaining > 40:
        alloc = min(remaining * 0.3, 100.0)
        nesterov_smoothed_descent._prev_y = None
        nesterov_smoothed_descent._prev_grad = None
        a_final = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                            tau_init=100.0, tau_max=5000.0,
                                            beta=0.8, verbose=True, seed=seed + 200)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final NAGD: obj={best_score:.8f} **")
        else:
            print(f"  Final NAGD: obj={obj_final:.8f}")

    # LP polish
    remaining = remaining_time()
    if remaining > 30:
        alloc = min(remaining * 0.3, 60.0)
        a_lp2 = lp_polish(best_seq, time_budget=alloc)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_lp2:.8f}")

    # Subgradient
    remaining = remaining_time()
    if remaining > 20:
        alloc = min(remaining * 0.25, 40.0)
        a_sg2 = subgradient_polish(best_seq, time_budget=alloc)
        obj_sg2 = compute_obj(a_sg2)
        if obj_sg2 < best_score:
            best_score = obj_sg2
            best_seq = a_sg2.copy()
            print(f"  Final SG: obj={best_score:.8f} **")
        else:
            print(f"  Final SG: obj={obj_sg2:.8f}")

    # Peak redistribution
    remaining = remaining_time()
    if remaining > 15:
        alloc = min(remaining * 0.25, 30.0)
        a_pr2 = peak_redistribute(best_seq, time_budget=alloc)
        obj_pr2 = compute_obj(a_pr2)
        if obj_pr2 < best_score:
            best_score = obj_pr2
            best_seq = a_pr2.copy()
            print(f"  Final PR: obj={best_score:.8f} **")
        else:
            print(f"  Final PR: obj={obj_pr2:.8f}")

    # One more LP if time
    remaining = remaining_time()
    if remaining > 15:
        alloc = min(remaining * 0.5, 30.0)
        a_lp3 = lp_polish(best_seq, time_budget=alloc)
        obj_lp3 = compute_obj(a_lp3)
        if obj_lp3 < best_score:
            best_score = obj_lp3
            best_seq = a_lp3.copy()
            print(f"  Ultimate LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
