"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Goeman-style reduced-constraint LP with proper line search, 
FFT-accelerated multiplicative peak-shaving updates, 
and multi-length exploration with spectral initialization.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_reduced(a, active_set, tolerance=1e-8):
    """
    Solve LP with reduced constraints: min M s.t. (A@g)[k] <= M for k in active_set,
    sum(g)=1, g>=0.
    Returns (g, M, new_active_set) where new_active_set includes newly violated constraints.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    # Build only the active constraints
    A_sub = np.zeros((len(active_set), n))
    for idx, k in enumerate(active_set):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_sub[idx, j] = a_norm[i]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_sub @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Check which constraints are violated
                    A_full = build_conv_matrix(a_norm)
                    conv_g = A_full @ g_val
                    max_conv = float(np.max(conv_g))
                    violated = np.where(conv_g > max_conv - tolerance)[0]
                    return g_val, float(M.value), list(violated)
        except Exception:
            continue
    return None, None, None

    Goeman-style LP polishing with reduced constraints and proper line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf, 0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    no_improve = 0
    start = time.monotonic()
    it = 0
    
    # Initialize active set with top constraints
    conv_a = compute_conv(a)
    top_k = np.argsort(conv_a)[::-1][:min(20, 2*n-1)]
    active_set = set(top_k.tolist())
    
    while it < max_iter and (time.monotonic() - start) < time_limit:

        if g_lp is None:

            g_lp, _ = solve_lp_full(a)

def multiplicative_peak_shave(a, n_iter=50, tau=50.0, verbose=False):
    """
    Multiplicative update to reduce convolution peaks.
    a_j <- a_j * exp(-eta * r_j) where r_j is the weighted gradient.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    eta = 0.05
    
    for it in range(n_iter):
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Compute gradient: r_j = 2 * sum_k p_k * a_{k-j} (for valid k-j)
        # This is a cross-correlation
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_fft = fft(a, N)
        g_full = ifft(P * np.conj(A_fft)).real
        r = 2.0 * g_full[:n]
        
        # Multiplicative update
        a = a * np.exp(-eta * r)
        a = np.maximum(a, 1e-15)
        a = a / float(np.sum(a))
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Anneal
        eta *= 0.98
        tau *= 1.02
        
        if verbose and it % 10 == 0:
            print(f"    MPS it{it}: obj={best_obj:.8f}, eta={eta:.4f}, tau={tau:.1f}")
    
    return best_a, best_obj

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Goeman-style reduced-constraint LP with golden-section line search,
FFT-accelerated multiplicative peak-shaving, subgradient polishing,
and multi-length exploration with diverse initializations.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_reduced(a, active_set):
    """
    Solve LP with reduced constraints: min M s.t. (A@g)[k] <= M for k in active_set,
    sum(g)=1, g>=0.
    Returns (g, M, newly_violated_set).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))

    if len(active_set) == 0:
        active_set = list(range(2 * n - 1))

    A_sub = np.zeros((len(active_set), n))
    for idx, k in enumerate(active_set):
        k = int(k)
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_sub[idx, j] = a_norm[i]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_sub @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Check all constraints
                    A_full = build_conv_matrix(a_norm)
                    conv_g = A_full @ g_val
                    max_conv = float(np.max(conv_g))
                    tol = max_conv * 0.01 + 1e-10
                    violated = np.where(conv_g > max_conv - tol)[0]
                    return g_val, float(M.value), list(int(v) for v in violated)
        except Exception:
            continue
    return None, None, None


def solve_lp_full(a):
    """Solve the full LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def goeman_lp_polish(a, max_iter=200, time_limit=120.0, verbose=False):
    """
    Goeman-style LP polishing with reduced constraints and golden-section line search.
    Returns (best_a, best_obj, num_iterations).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf, 0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    no_improve = 0
    start = time.monotonic()
    it = 0

    # Initialize active set with top constraints
    conv_a = compute_conv(a)
    num_active = min(30, 2 * n - 1)
    top_k = np.argsort(conv_a)[::-1][:num_active]
    active_set = list(int(x) for x in top_k)

    gr_conj = (np.sqrt(5.0) - 1.0) / 2.0

    while it < max_iter and (time.monotonic() - start) < time_limit:
        # Solve reduced LP
        g_lp, M_lp, new_active = solve_lp_reduced(a, active_set)

        if g_lp is None:
            # Fallback to full LP
            g_lp, _ = solve_lp_full(a)
            if g_lp is None:
                break
            new_active = list(range(2 * n - 1))

        # Update active set (keep it bounded)
        if new_active:
            active_set = new_active[:50]  # Cap active set size
        else:
            active_set = active_set[:50]

        # Golden section line search: minimize f((1-t)*a + t*g_lp) for t in [0,1]
        def obj_at_t(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return np.inf
            a_try = a_try / s_try
            return compute_obj(a_try)

        lo, hi = 0.0, 1.0
        c = hi - gr_conj * (hi - lo)
        d = lo + gr_conj * (hi - lo)
        fc = obj_at_t(c)
        fd = obj_at_t(d)

        for _ in range(30):
            if time.monotonic() - start > time_limit:
                break
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = hi - gr_conj * (hi - lo)
                fc = obj_at_t(c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr_conj * (hi - lo)
                fd = obj_at_t(d)

        best_t = (lo + hi) / 2.0
        best_t_obj = obj_at_t(best_t)

        # Also evaluate t=1 and t=0.5 for robustness
        obj_t1 = obj_at_t(1.0)
        if obj_t1 < best_t_obj:
            best_t = 1.0
            best_t_obj = obj_t1

        it += 1
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            a_new = (1.0 - best_t) * a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
            else:
                a_new = a_new / s_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            if verbose and it % 10 == 0:
                print(f"      Goeman it{it}: obj={best_obj:.8f}, t={best_t:.3f}, |active|={len(active_set)}")
        else:
            no_improve += 1
            if no_improve >= 8:
                break

    return best_a, best_obj, it


def multiplicative_peak_shave(a, n_iter=100, tau_init=30.0, verbose=False):
    """
    Multiplicative update to reduce convolution peaks.
    Uses smoothed subgradient with adaptive tau.
    Returns (best_a, best_obj).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    tau = tau_init
    N = 1
    while N < 3 * n:
        N <<= 1

    for it in range(n_iter):
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        # Compute gradient via FFT cross-correlation
        P = fft(p, N)
        A_fft = fft(a, N)
        g_full = ifft(P * np.conj(A_fft)).real
        r = 2.0 * g_full[:n]

        # Adaptive step size
        max_r = float(np.max(np.abs(r)))
        if max_r < 1e-15:
            break
        eta = 2.0 / (tau * max_r + 1.0)

        # Multiplicative update
        a = a * np.exp(-eta * r)
        a = np.maximum(a, 1e-15)
        s = float(np.sum(a))
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Anneal tau and adapt eta
        tau *= 1.03

        if verbose and it % 20 == 0:
            print(f"    MPS it{it}: obj={best_obj:.8f}, tau={tau:.1f}, eta={eta:.4f}")

    return best_a, best_obj


def subgradient_polish(a, n_iter=200, verbose=False):
    """
    Projection subgradient method on the simplex for the max function.
    Returns (best_a, best_obj).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    step = 1.0 / np.sqrt(n)

    for it in range(n_iter):
        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Subgradient: average of active constraints
        active_tol = max_conv * 0.05 + 1e-10
        active = np.where(conv >= max_conv - active_tol)[0]
        p = np.zeros(len(conv))
        p[active] = 1.0 / len(active)

        # Gradient
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_fft = fft(a, N)
        g_full = ifft(P * np.conj(A_fft)).real
        g = 2.0 * g_full[:n]

        # Projected subgradient step on simplex
        a = a - step * g
        # Project onto simplex {x >= 0, sum(x) = 1}
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        step *= 0.995

        if verbose and it % 50 == 0:
            print(f"    SP it{it}: obj={best_obj:.8f}, step={step:.5f}, |active|={len(active)}")

    return best_a, best_obj


def nesterov_smoothed_descent(a, time_budget=100.0, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    N = 1
    while N < 3 * n:
        N <<= 1

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        P = fft(p, N)
        A = fft(y, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9997)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def perturb_sequence(a, rng, method=0, scale=0.03):
    """Apply a perturbation to escape local minima."""
    n = len(a)
    a = a.copy()
    if method == 0:
        a = a * np.exp(rng.normal(0, scale, n))
    elif method == 1:
        noise = rng.uniform(1 - scale, 1 + scale, n)
        a = a * noise
    elif method == 2:
        mask = rng.random(n) > scale * 3
        a = a * mask
    elif method == 3:
        shift = int(rng.integers(1, max(2, n // 20)))
        a = np.roll(a, shift)
    elif method == 4:
        a = (1 - scale) * a + scale / n
    elif method == 5:
        A = fft(a, 2 * n)
        A[1:-1] *= np.exp(1j * rng.normal(0, scale, len(A[1:-1])))
        a = ifft(A).real[:n]
        a = np.maximum(a, 0)
    elif method == 6:
        # Targeted perturbation: reduce values at peak-contributing positions
        conv = compute_conv(a / a.sum())
        peak_idx = int(np.argmax(conv))
        # Positions that contribute to the peak
        contrib = np.zeros(n)
        for j in range(n):
            i = peak_idx - j
            if 0 <= i < n:
                contrib[j] = a[i]
        contrib = contrib / max(float(np.max(contrib)), 1e-10)
        a = a * (1 - scale * contrib)
        a = np.maximum(a, 0)
    else:
        a = a * np.exp(rng.normal(0, scale, n))

    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if n == 2:
        x_old = np.array([0.0, 1.0])
        x_new = np.linspace(0, 1, new_n)
        a_new = np.interp(x_new, x_old, a)
        return np.maximum(a_new, 0)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two-gaussian bump
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        x = np.arange(n)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution (relates to Chebyshev polynomials)
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / np.sqrt(1 - x ** 2)
        a = a / a.sum()
    elif mode == 4:
        # Random Fourier
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for _ in range(5):
            f = int(rng.integers(1, 8))
            p = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1.5, 1.5)
        a = a / a.sum()
    elif mode == 6:
        # Tukey window
        x = np.linspace(0, 1, n)
        r = 0.3
        a = np.ones(n)
        left = x < (1 - r) / 2
        right = x > (1 + r) / 2
        a[left] = 0.5 + 0.5 * np.cos(np.pi * (2 * x[left] / (1 - r) - 1) / r)
        a[right] = 0.5 + 0.5 * np.cos(np.pi * (2 * x[right] / (1 - r) - 1 + 2) / r)
        a = a / a.sum()
    elif mode == 7:
        # Optimal for small n (try harmonic-like)
        x = np.arange(1, n + 1, dtype=float)
        a = 1.0 / x ** 0.5
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)

    if p1_budget > 20:
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=p1_budget,
                                          tau_init=5.0, tau_end=300.0, verbose=False)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase1 SGD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 SGD: obj={obj_sgd:.8f}")

    # === Phase 2: Intensive Goeman LP polishing ===
    print(f"\n=== Phase 2: Goeman LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.3, 350.0)
    if p2_budget > 30:
        # Multiple rounds with increasing aggressiveness
        for rnd in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(p2_budget / (4 - rnd), remaining * 0.4)
            if alloc < 15:
                break

            a_current = best_seq.copy()
            if rnd > 0:
                a_current = perturb_sequence(a_current, rng, method=rnd - 1, scale=0.015 + 0.005 * rnd)

            a_new, obj_new, iters = goeman_lp_polish(a_current, max_iter=200, time_limit=alloc,
                                                      verbose=(rnd == 0))
            if obj_new < best_score - 1e-13:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} ({iters} iters) **")
            else:
                print(f"  LP round {rnd}: obj={obj_new:.8f} ({iters} iters)")

    # === Phase 3: Multiplicative peak shaving ===
    print(f"\n=== Phase 3: Multiplicative Peak Shaving ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    if p3_budget > 15:
        # Run with different tau values
        for tau_try in [20.0, 50.0, 100.0]:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = p3_budget / 3.0
            a_new, obj_new = multiplicative_peak_shave(best_seq, n_iter=int(alloc * 4),
                                                       tau_init=tau_try, verbose=False)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  MPS tau={tau_try}: obj={best_score:.8f} **")
            else:
                print(f"  MPS tau={tau_try}: obj={obj_new:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = set()
    for delta in [-100, -50, 0, 50, 100, 200, 300, 500, 800]:
        n_try = n_current + delta
        if 400 <= n_try <= 2500:
            n_options.add(n_try)
    # Add some fixed values
    for nv in [500, 600, 700, 800, 900, 1000, 1200, 1500, 2000]:
        if 400 <= nv <= 2500:
            n_options.add(nv)
    n_options.discard(n_current)

    # Sort by proximity to current best n
    n_list = sorted(n_options, key=lambda x: abs(x - n_current))

    explored = 0
    for n_try in n_list:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        explored += 1
        if explored > 6:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick optimization
        alloc_sgd = min(remaining * 0.05, 30.0)
        alloc_lp = min(remaining * 0.08, 60.0)
        alloc_mps = min(remaining * 0.05, 30.0)

        if alloc_sgd > 10:
            a_try = nesterov_smoothed_descent(a_try, time_budget=alloc_sgd,
                                              tau_init=10.0, tau_end=200.0)

        remaining = deadline - time.monotonic()
        if alloc_lp > 15 and remaining > 30:
            a_try, obj_try, _ = goeman_lp_polish(a_try, max_iter=80, time_limit=alloc_lp)
        else:
            obj_try = compute_obj(a_try)

        remaining = deadline - time.monotonic()
        if remaining > 15 and obj_try < best_score + 0.01:
            a_try, obj_try = multiplicative_peak_shave(a_try, n_iter=int(alloc_mps * 3), tau_init=50.0)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Combined polish on best ===
    print(f"\n=== Phase 5: Combined Polish ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.25, 200.0)
    if p5_budget > 20:
        a_new, obj_new = combined_polish(best_seq, time_budget=p5_budget, verbose=True)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Combined: obj={best_score:.8f} **")
        else:
            print(f"  Combined: obj={obj_new:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        mode = r % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.1 + 0.05 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.15, 60.0)
        if alloc > 20:
            half = alloc * 0.5
            a_r, obj_r = multiplicative_peak_shave(a_r, n_iter=int(half * 4), tau_init=30.0)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_r, obj_r, _ = goeman_lp_polish(a_r, max_iter=40, time_limit=min(alloc * 0.4, remaining * 0.3))

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau multiplicative
    if remaining > 20:
        alloc = min(remaining * 0.2, 60.0)
        a_new, obj_new = multiplicative_peak_shave(best_seq, n_iter=int(alloc * 5), tau_init=80.0)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Final MPS: obj={best_score:.8f} **")

    # Final LP rounds
    for lp_rnd in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining * 0.2, 60.0)
        a_current = best_seq.copy()
        if lp_rnd > 0:
            a_current = perturb_sequence(a_current, rng, method=lp_rnd, scale=0.01)
        a_lp, obj_lp, _ = goeman_lp_polish(a_current, max_iter=100, time_limit=alloc)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP {lp_rnd}: obj={best_score:.8f} **")
        else:
            print(f"  Final LP {lp_rnd}: obj={obj_lp:.8f}")

    # Final subgradient
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_new, obj_new = subgradient_polish(best_seq, n_iter=int(remaining * 2))
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Final SP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
