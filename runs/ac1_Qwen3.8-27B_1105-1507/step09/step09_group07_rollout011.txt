from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: minimize M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Uses HiGHS solver directly via scipy.
    """
    n = len(a)
    if n == 0:
        return None, None
    a_s = float(np.sum(a))
    if a_s < 1e-10:
        return None, None
    a_norm = a / a_s
    
    A = build_conv_matrix(a_norm)
    
    # LP: min M
    # Variables: [g_0, ..., g_{n-1}, M]
    # c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: A@g - M <= 0, i.e., A@g + (-1)*M <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception as e:
        pass
    
    return None, None

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def local_hill_climb(a, time_budget):
    """
    Simple hill climbing: randomly perturb pairs of coordinates and accept if objective improves.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    step_size = 0.01
    no_improve = 0
    max_no_improve = 200
    
    while time.monotonic() - start < time_budget:
        if no_improve > max_no_improve:
            break
            
        # Pick two random coordinates
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Transfer mass from i to j
        delta = step_size * a[i]
        if delta <= 0:
            continue
            
        a[i] -= delta
        a[j] += delta
        
        if a[i] < 0:
            a[i] += delta
            a[j] -= delta
            continue
            
        s = float(np.sum(a))
        if s < 1e-10:
            a[i] += delta
            a[j] -= delta
            continue
        a = a / s
        
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            step_size = min(0.05, step_size * 1.1)
        else:
            a[i] += delta
            a[j] -= delta
            a = a / a.sum()
            no_improve += 1
            if no_improve % 50 == 0:
                step_size = max(0.001, step_size * 0.8)

    return best_a

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: minimize M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses HiGHS directly via scipy for maximum speed.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    if n == 0:
        return None, None
    a_s = float(np.sum(a))
    if a_s < 1e-10:
        return None, None
    a_norm = a / a_s

    A = build_conv_matrix(a_norm)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None

def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    # Use linear interpolation (cubic can produce negatives)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def hill_climb(a, time_budget, seed=42):
    """
    Hill climbing via pairwise mass transfer.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    step_size = 0.005
    no_improve = 0
    max_no_improve = 500

    while time.monotonic() - start < time_budget:
        if no_improve > max_no_improve:
            break

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        delta = step_size * max(a[i], a[j])
        if delta <= 0 or delta >= max(a[i], a[j]) * 0.9:
            continue

        src = i if a[i] >= a[j] else j
        tgt = j if src == i else i

        a[src] -= delta
        a[tgt] += delta

        s = float(np.sum(a))
        if s < 1e-10 or a[src] < -1e-15:
            a[src] += delta
            a[tgt] -= delta
            continue

        a = np.maximum(a, 0)
        a = a / a.sum()

        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            step_size = min(0.03, step_size * 1.05)
        else:
            a[src] += delta
            a[tgt] -= delta
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve += 1
            if no_improve % 100 == 0:
                step_size = max(0.001, step_size * 0.85)

    return best_a

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init n={n0}: obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"\nInitial: n={n_current}, obj={best_score:.8f}, target=1.5027")

    # === Phase 1: LP polish on parent (8%) ===
    print(f"\n=== Phase 1: LP Polish on Parent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 90.0)
    if p1_budget > 20:
        a_p1 = lp_alternate(best_seq, time_budget=p1_budget, verbose=True, verbose_interval=100)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_p1:.8f} (no improvement)")

    # === Phase 2: Systematic n-sweep (22%) ===
    print(f"\n=== Phase 2: n-Sweep ===")
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.22

    base_n = n_current
    n_options = set()
    for delta in range(-150, 300, 15):
        n_try = base_n + delta
        if 400 <= n_try <= 1600:
            n_options.add(n_try)
    for n_r in [500, 600, 650, 700, 750, 800, 850, 900, 950, 1000, 1024, 1100, 1200]:
        if 400 <= n_r <= 1600:
            n_options.add(n_r)
    n_options = sorted(n_options)

    per_n_budget = p2_budget / max(1, len(n_options))
    n_results = []

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = max(8, min(per_n_budget, remaining * 0.08))
        if alloc < 5:
            alloc = 5

        a_try = resample_sequence(best_seq, n_try)
        a_opt = lp_alternate(a_try, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)
        n_results.append((obj_opt, n_try, a_opt.copy()))

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try
            print(f"  n={n_try}: obj={obj_opt:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    n_results.sort(key=lambda x: x[0])
    if n_results:
        print(f"  Best from sweep: n={n_results[0][1]}, obj={n_results[0][0]:.8f}")

    # === Phase 3: Intensive LP on top candidates (35%) ===
    print(f"\n=== Phase 3: Intensive LP on Top Candidates ===")
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.35

    # Top n values from sweep + current best
    top_ns = list(set([best_seq.__len__()]))
    if n_results:
        for i in range(min(4, len(n_results))):
            top_ns.append(n_results[i][1])
    top_ns = list(set(top_ns))
    top_ns.sort(key=lambda x: abs(x - n_current))
    top_ns = top_ns[:4]

    per_n = p3_budget / max(1, len(top_ns))

    for n_use in top_ns:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc_n = max(20, min(per_n, remaining * 0.25))

        # Run multiple rounds with perturbation
        for rnd in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            rnd_alloc = max(10, min(alloc_n / 4, remaining * 0.2))

            if rnd == 0:
                a_start = resample_sequence(best_seq, n_use)
            else:
                rng_r = np.random.default_rng(rnd * 7919 + n_use + 1000)
                scale = 0.02 + 0.015 * rnd
                a_base = resample_sequence(best_seq, n_use)
                a_start = a_base * np.exp(rng_r.normal(0, scale, n_use))
                a_start = np.maximum(a_start, 0)
                a_start = a_start / a_start.sum()

            a_lp = lp_alternate(a_start, time_budget=rnd_alloc, verbose=True, verbose_interval=150)
            obj_lp = compute_obj(a_lp)

            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                n_current = n_use
                print(f"  n={n_use} rnd{rnd}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_use} rnd{rnd}: obj={obj_lp:.8f}")

    # === Phase 4: Diverse restarts (18%) ===
    print(f"\n=== Phase 4: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.18
    n_restarts = 6
    per_restart = p4_budget / max(1, n_restarts)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        alloc = max(15, min(per_restart, remaining * 0.12))
        if alloc < 10:
            break

        mode = r % 12
        n_use = n_current + (r % 3) * 20 - 20  # Slight n variation
        n_use = max(400, min(1600, n_use))

        a_r = diverse_init(n_use, rng, mode=mode)
        blend = 0.25 + 0.08 * (r % 4)
        a_best_resampled = resample_sequence(best_seq, n_use)
        a_r = (1 - blend) * a_r + blend * a_best_resampled
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        a_r = lp_alternate(a_r, time_budget=alloc, verbose=False)
        obj_r = compute_obj(a_r)
        marker = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (mode={mode}, n={n_use}): obj={obj_r:.8f}{marker}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            n_current = n_use

    # === Phase 5: Hill climbing + final polish (17%) ===
    print(f"\n=== Phase 5: Hill Climb + Final Polish ===")
    remaining = deadline - time.monotonic()
    p5_budget = remaining * 0.17

    # Hill climb on best
    alloc_hc = min(p5_budget * 0.3, 40.0)
    if alloc_hc > 10:
        a_hc = hill_climb(best_seq, time_budget=alloc_hc, seed=12345)
        obj_hc = compute_obj(a_hc)
        if obj_hc < best_score:
            best_score = obj_hc
            best_seq = a_hc.copy()
            print(f"  Hill climb: obj={best_score:.8f} **")
        else:
            print(f"  Hill climb: obj={obj_hc:.8f}")

    # Final LP polish
    remaining = deadline - time.monotonic()
    alloc_lp = min(remaining * 0.4, 50.0)
    if alloc_lp > 15:
        a_final = lp_alternate(best_seq, time_budget=alloc_lp, verbose=True, verbose_interval=100)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # One more hill climb after LP
    remaining = deadline - time.monotonic()
    alloc_hc2 = min(remaining * 0.25, 20.0)
    if alloc_hc2 > 8:
        a_hc2 = hill_climb(best_seq, time_budget=alloc_hc2, seed=54321)
        obj_hc2 = compute_obj(a_hc2)
        if obj_hc2 < best_score:
            best_score = obj_hc2
            best_seq = a_hc2.copy()
            print(f"  Final HC: obj={best_score:.8f} **")

    # Final tiny LP
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_last = lp_alternate(best_seq, time_budget=min(remaining * 0.6, 15.0), verbose=False)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"=== Target: 1.5027, Achieved: {best_score:.8f}, Gap: {best_score - 1.5027:.8f} ===")

    return result
