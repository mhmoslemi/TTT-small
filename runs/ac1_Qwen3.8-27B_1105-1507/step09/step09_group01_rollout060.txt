"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a hybrid approach: LP-based direction finding with quadratic line search,
multiplicative weight updates for positive variable handling, 2D coordinate
descent for exact pair updates, and adaptive multi-length exploration.
Key novelty: quadratic line search for LP directions, multiplicative updates,
and structured (non-Gaussian) perturbations for escape from local optima.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, M_target=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue
    return None, None


def quadratic_line_search(a, d, t_max=1.0, max_iter=60):
    """
    Quadratic line search: minimize max_k [(a+td)*(a+td)]_k over t in [0, t_max].
    Uses the quadratic approximation conv_k(t) = C_k + 2t*P_k + t^2*Q_k.
    """
    n = len(a)
    conv_a = compute_conv(a)
    conv_ad = compute_conv(a) * 0  # placeholder
    # Compute P_k = (a*d)_k and Q_k = (d*d)_k
    ad = np.convolve(a, d)
    dd = np.convolve(d, d)
    # f(t) = max_k [conv_a[k] + 2*t*ad[k] + t^2*dd[k]]
    # We want to minimize this over t in [0, t_max]
    
    lo = 0.0
    hi = t_max
    
    def f_of_t(t):
        vals = conv_a + 2.0 * t * ad + t * t * dd
        return float(np.max(vals))
    
    # Ternary search
    for _ in range(max_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = f_of_t(m1)
        f2 = f_of_t(m2)
        if f1 <= f2:
            hi = m2
        else:
            lo = m1
        if hi - lo < 1e-8:
            break
    
    t_opt = (lo + hi) / 2.0
    return t_opt, f_of_t(t_opt)


def golden_section_search(f, a_lo, a_hi, max_iter=40):
    """Golden section search for minimum of f on [a_lo, a_hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if abs(a_hi - a_lo) < 1e-7:
            break
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = f(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = f(d)
    return (a_lo + a_hi) / 2.0


def multiplicative_update(a, conv, K=10, lr=0.01, n_iter=200, verbose=False):
    """
    Multiplicative weight update: a_i <- a_i / (1 + lr * g_i) where g_i is
    the weighted subgradient from top-K peaks. Maintains positivity naturally.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(n_iter):
        if it % 20 == 0 or it == n_iter - 1:
            conv = compute_conv(a)
        
        # Find top K peaks
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        
        # Weighted subgradient
        weights = np.exp((peak_vals - max_pv) * 3.0)
        weights /= weights.sum()
        
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * 2.0 * a[j_start:j_end + 1][::-1]
        
        # Multiplicative update
        lr_adapt = lr * (1.0 + 0.5 * np.cos(np.pi * it / n_iter))
        a = a / (1.0 + lr_adapt * g)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        if it % 50 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
            else:
                # Revert if not improving
                a = best_a.copy()
        
    return best_a


def coord_descent_pairs(a, n_iter=500, pair_size=2, verbose=False, seed=42):

            pass  # Skip expensive prints in tight loop
        
        # Pick a random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            j = (j + 1) % n
        
        # 1D search over t: a_i + t, a_j - t
        # t range: [-a_i, a_j] to maintain non-negativity

        t_hi = a[j] * 0.99

            a_temp = a.copy()
            a_temp[i] += t
            a_temp[j] -= t
            if a_temp[i] < 0 or a_temp[j] < 0:
                return 1e10
            return compute_obj(a_temp)
        
        # Sample a few points first
        n_samples = 10
        t_candidates = np.linspace(t_lo, t_hi, n_samples)
        best_t = 0.0
        best_t_obj = obj_at_t(0.0)
        for tc in t_candidates:
            oc = obj_at_t(tc)
            if oc < best_t_obj:
                best_t_obj = oc
                best_t = tc
        
        # Refine around best_t
        t_lo2 = max(t_lo, best_t - (t_hi - t_lo) / n_samples)
        t_hi2 = min(t_hi, best_t + (t_hi - t_lo) / n_samples)
        t_opt = golden_section_search(obj_at_t, t_lo2, t_hi2, max_iter=15)
        
        a[i] += t_opt
        a[j] -= t_opt
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            continue
        a = a / s
        
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
    
    return best_a

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        if it % 5000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Smoothed gradient via FFT
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_ff = fft(y, N)
        g_full = ifft(P * np.conj(A_ff)).real
        g = 2.0 * g_full[:n]

        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)

        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def lp_polish_with_quad(a, time_budget=200.0, verbose=False, seed=42):
    """
    LP-based polishing with quadratic line search and structured perturbations.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start

        # Solve LP for direction
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Quadratic line search
        # Direction: d = g_lp - best_a (move from best_a toward g_lp)
        d = g_lp - best_a
        t_opt, t_val = quadratic_line_search(best_a, d, t_max=1.0, max_iter=50)
        
        # Also try golden section on full objective for comparison
        def blend_obj(t):
            a_try = best_a + t * d
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return 1e10
            a_try = a_try / s_try
            return compute_obj(a_try)
        
        t_gs = golden_section_search(blend_obj, 0.0, 1.0, max_iter=25)
        
        # Use the better of the two
        a_try_q = np.maximum(best_a + t_opt * d, 0)
        a_try_q = a_try_q / a_try_q.sum()
        obj_q = compute_obj(a_try_q)
        
        a_try_g = np.maximum(best_a + t_gs * d, 0)
        a_try_g = a_try_g / a_try_g.sum()
        obj_g = compute_obj(a_try_g)
        
        if obj_q <= obj_g:
            a_try = a_try_q
            obj_try = obj_q
        else:
            a_try = a_try_g
            obj_try = obj_g
        
        iter_count += 1

        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a.copy()
            no_improve = 0
            if verbose and iter_count % 5 == 0:
                print(f"    LP-Q it{iter_count}: obj={best_obj:.8f}, t_q={t_opt:.4f}, t_gs={t_gs:.4f}, el={elapsed:.0f}s")
        else:
            no_improve += 1
            if no_improve == 2:
                # Try smaller step
                t_small = max(0.01, t_opt * 0.3)
                a_try2 = np.maximum(best_a + t_small * d, 0)
                a_try2 = a_try2 / a_try2.sum()
                obj_try2 = compute_obj(a_try2)
                if obj_try2 < best_obj - 1e-14:
                    best_obj = obj_try2
                    best_a = a_try2.copy()
                    a = best_a.copy()
                    no_improve = 0
            elif no_improve >= 4:
                # Structured perturbation (not just Gaussian)
                pert_type = rng.integers(0, 4)
                if pert_type == 0:
                    # Sine perturbation
                    freq = rng.uniform(0.5, 3.0)
                    phase = rng.uniform(0, 2 * np.pi)
                    x = np.linspace(0, 1, n)
                    pert = 1.0 + 0.02 * np.sin(2 * np.pi * freq * x + phase)
                    a = best_a * pert
                elif pert_type == 1:
                    # Linear ramp perturbation
                    x = np.linspace(0, 1, n)
                    pert = 1.0 + 0.03 * (x - 0.5)
                    a = best_a * pert
                elif pert_type == 2:
                    # Quadratic perturbation (concentrate center or edges)
                    x = np.linspace(-1, 1, n)
                    sign = rng.choice([-1, 1])
                    pert = 1.0 + sign * 0.02 * x**2
                    a = best_a * pert
                else:
                    # Gaussian perturbation (fallback)
                    a = best_a * np.exp(rng.normal(0, 0.015, n))
                
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-10:
                    a = best_a.copy()
                else:
                    a = a / s
                no_improve = 0
                if iter_count > 80:
                    break

        # Periodic constraint tightening
        if iter_count % 6 == 3 and M_lp is not None:
            for tight_frac in [0.93, 0.96]:
                M_tight = M_lp * tight_frac
                g_tight, _ = solve_lp_minpeak(a, M_target=M_tight)
                if g_tight is not None:
                    d_t = g_tight - best_a
                    t_try, _ = quadratic_line_search(best_a, d_t, t_max=1.0, max_iter=30)
                    a_try3 = np.maximum(best_a + t_try * d_t, 0)
                    a_try3 = a_try3 / a_try3.sum()
                    obj_try3 = compute_obj(a_try3)
                    if obj_try3 < best_obj - 1e-14:
                        best_obj = obj_try3
                        best_a = a_try3.copy()
                        a = best_a.copy()
                        no_improve = 0
                        if verbose:
                            print(f"    LP-tight it{iter_count}: obj={best_obj:.8f}")
                        break

    return best_a


def subgradient_refine(a, time_budget=100.0, K=5, verbose=False):
    """Subgradient method with top-K peaks, momentum, and adaptive step."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]

        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        weights = np.exp((peak_vals - max_pv) * 5.0 / max_pv)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-10:
                    break
            else:
                lr *= 0.995

        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Skewed distribution
        x = np.linspace(0, 1, n)
        a = x ** rng.uniform(0.5, 3.0)
        a = a / a.sum()
    elif mode == 3:
        # Asymmetric bimodal
        c1 = n * rng.uniform(0.1, 0.3)
        c2 = n * rng.uniform(0.7, 0.9)
        w = n * 0.1
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += 0.7 * np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 4:
        # Cosine modulation (asymmetric)
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP polishing with quadratic line search,
    multiplicative updates, coordinate descent, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Nesterov smoothed GD warmup ===
    print(f"\n=== Phase 1: Smoothed GD Warmup ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.10, 100.0)
    if p1_budget > 20:
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=p1_budget,
                                          tau_init=5.0, tau_end=350.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase1 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 GD: obj={obj_sgd:.8f}")

    # === Phase 2: LP polish with quadratic line search (main workhorse) ===
    print(f"\n=== Phase 2: LP + Quadratic Line Search ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.30, 400.0)
    if p2_budget > 30:
        a_lp = lp_polish_with_quad(best_seq, time_budget=p2_budget, verbose=True, seed=42)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP-Q: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP-Q: obj={obj_lp:.8f}")

    # === Phase 3: Multiplicative updates ===
    print(f"\n=== Phase 3: Multiplicative Updates ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.08, 80.0)
    if p3_budget > 15:
        conv = compute_conv(best_seq)
        a_mu = multiplicative_update(best_seq, conv, K=10, lr=0.008,
                                     n_iter=int(p3_budget * 50), verbose=True)
        obj_mu = compute_obj(a_mu)
        if obj_mu < best_score:
            best_score = obj_mu
            best_seq = a_mu.copy()
            print(f"  Phase3 MU: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 MU: obj={obj_mu:.8f}")

    # === Phase 4: Coordinate descent pairs ===
    print(f"\n=== Phase 4: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.12, 120.0)
    if p4_budget > 15:
        n_iter_cd = int(p4_budget * 8)  # ~8 iterations per second
        a_cd = coord_descent_pairs(best_seq, n_iter=n_iter_cd, seed=123)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase4 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 CD: obj={obj_cd:.8f}")

    # === Phase 5: Subgradient refinement ===
    print(f"\n=== Phase 5: Subgradient Refine ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.07, 70.0)
    if p5_budget > 15:
        a_sg = subgradient_refine(best_seq, time_budget=p5_budget, K=8)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Phase5 SG: obj={best_score:.8f} **")
        else:
            print(f"  Phase5 SG: obj={obj_sg:.8f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        n_current - 100, n_current + 100,
        n_current + 200, n_current + 300,
        600, 700, 900, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_current) > 50]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)

        # Quick GD warmup
        alloc1 = min(remaining * 0.12, 30.0)
        if alloc1 > 10:
            a_try = nesterov_smoothed_descent(a_try, time_budget=alloc1,
                                              tau_init=10.0, tau_end=250.0)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.20, 80.0)
        if alloc2 > 15:
            a_try = lp_polish_with_quad(a_try, time_budget=alloc2, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Perturbation + re-LP (structured) ===
    print(f"\n=== Phase 7: Structured Perturbation + Re-LP ===")
    remaining = deadline - time.monotonic()

    for r in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        # Structured perturbation
        pert_type = r % 5
        x = np.linspace(0, 1, len(best_seq))
        if pert_type == 0:
            pert = 1.0 + 0.03 * np.sin(2 * np.pi * (1.0 + r * 0.3) * x)
        elif pert_type == 1:
            pert = 1.0 + 0.02 * (2 * x - 1) ** 2
        elif pert_type == 2:
            pert = 1.0 + 0.025 * np.cos(4 * np.pi * x + r)
        elif pert_type == 3:
            pert = np.exp(0.02 * rng.normal(0, 1, len(best_seq)))
        else:
            # Asymmetric ramp
            pert = 1.0 + 0.03 * rng.uniform(-1, 1) * (x - 0.5)

        a_pert = best_seq * pert
        a_pert = np.maximum(a_pert, 0)
        s_p = a_pert.sum()
        if s_p < 1e-10:
            continue
        a_pert = a_pert / s_p

        alloc = min(remaining * 0.20, 80.0)
        if alloc > 15:
            a_pert = lp_polish_with_quad(a_pert, time_budget=alloc, seed=500 + r)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Perturb {r} (type={pert_type}): obj={best_score:.8f} **")
            else:
                print(f"  Perturb {r} (type={pert_type}): obj={obj_pert:.8f}")

    # === Phase 8: Final intensive LP ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.6, 180.0)
        a_final = lp_polish_with_quad(best_seq, time_budget=alloc, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # Final subgradient
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 40.0)
        a_sg = subgradient_refine(best_seq, time_budget=alloc, K=10)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # Final multiplicative polish
    remaining = deadline - time.monotonic()
    if remaining > 10:
        conv = compute_conv(best_seq)
        a_mu = multiplicative_update(best_seq, conv, K=15, lr=0.005, n_iter=200)
        obj_mu = compute_obj(a_mu)
        if obj_mu < best_score:
            best_score = obj_mu
            best_seq = a_mu.copy()
            print(f"  Final MU: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
