"""

Uses enhanced multi-kernel LP polishing with geometric blending, peak-directed

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = 0
        k_end = min(2 * n - 2, j + n - 1)
        for k in range(k_start, k_end + 1):
            i = k - j
            if 0 <= i < n:
                A[k, j] = a[i]
    return A


def build_conv_matrix_fast(a):
    """Build sparse-friendly convolution matrix using Toeplitz structure."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for i in range(n):
            k = i + j
            rows.append(k)
            cols.append(j)
            vals.append(a[i])
    from scipy.sparse import coo_matrix
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A.tocsr()


def solve_lp_minpeak(a, M_target=None, sparse=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Optionally add M <= M_target constraint.
    """
    n = len(a)
    a_norm = a / np.sum(a) + 1e-15
    a_norm = a_norm / np.sum(a_norm)

    if sparse:
        A = build_conv_matrix_fast(a_norm)
    else:
        A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue

    return None, None


def golden_section_search(f, a_lo, a_hi, tol=0.01, max_iter=40):
    """Golden section search for minimum of f on [a_lo, a_hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if abs(a_hi - a_lo) < tol:
            break
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = f(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = f(d)
    return (a_lo + a_hi) / 2.0


def geometric_blend(a, g, t):
    """Geometric (multiplicative) blend: a^(1-t) * g^t, normalized."""
    a_safe = np.maximum(a, 1e-15)
    g_safe = np.maximum(g, 1e-15)
    result = a_safe ** (1.0 - t) * g_safe ** t
    s = float(np.sum(result))
    if s < 1e-15:
        return a.copy()
    return result / s


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max using FFT cross-correlation."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def nesterov_smoothed_descent(a, time_budget=200.0, tau_init=10.0, tau_end=400.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        if it % 5000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)

        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)

        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def peak_directed_subgradient(a, time_budget=100.0, K=8, verbose=False):
    """
    Subgradient method targeting top-K peaks with adaptive step size
    and momentum. New: uses peak gap to adapt step size.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    momentum = 0.75
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break

        # Adaptive weighting: emphasize peaks more if they're close together
        peak_spread = max_pv - peak_vals[-1]
        if peak_spread > 1e-15:
            temp = max_pv * 0.3
        else:
            temp = max_pv * 1.0
        weights = np.exp((peak_vals - max_pv) / max(temp, 1e-15))
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += 2.0 * w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 200 == 0:
                print(f"    SG it{it}: obj={best_obj:.8f}, lr={lr:.4f}")
        else:
            stall += 1
            if stall > 200:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-10:
                    break
            else:
                lr *= 0.997

        it += 1

    return best_a


def surgical_peak_reduction(a, n_iter=50, verbose=False):
    """
    Directly target the top peaks by reducing contributing elements
    and redistributing mass to low-contribution positions.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(n_iter):
        conv = compute_conv(a)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]

        # Find contributing indices
        i_start = max(0, peak_k - (n - 1))
        i_end = min(n - 1, peak_k)
        contrib_indices = list(range(i_start, i_end + 1))

        if len(contrib_indices) == 0:
            break

        contrib_vals = np.array([a[i] * a[peak_k - i] for i in contrib_indices])
        total_contrib = float(np.sum(contrib_vals))
        if total_contrib < 1e-15:
            break

        # Find the element with the largest individual contribution
        max_contrib_idx = int(np.argmax(contrib_vals))
        i_star = contrib_indices[max_contrib_idx]
        j_star = peak_k - i_star

        # Amount to reduce: reduce the peak by a fraction
        target_reduction = peak_val * 0.05
        current_contrib = a[i_star] * a[j_star]
        if current_contrib < 1e-15:
            break

        # Reduce a[i_star] (or a[j_star] if different) proportionally
        if i_star == j_star:
            # Reduce a[i_star]
            reduction = min(a[i_star] * 0.1, np.sqrt(target_reduction) * 0.5)
        else:
            reduction = min(a[i_star] * 0.05, a[j_star] * 0.05)
            if a[i_star] > a[j_star]:
                reduction = min(a[i_star] * 0.1, target_reduction / max(a[j_star], 1e-10))
            else:
                reduction = min(a[j_star] * 0.1, target_reduction / max(a[i_star], 1e-10))

        reduction = max(0, min(reduction, 0.05 * float(np.max(a))))
        if reduction < 1e-10:
            break

        if i_star != j_star:
            a[i_star] -= reduction
            a[j_star] -= reduction
            freed = 2.0 * reduction
        else:
            a[i_star] -= reduction
            freed = reduction

        # Redistribute to positions far from the peak
        # Positions that don't contribute to this peak
        safe_mask = np.ones(n, dtype=bool)
        for i in contrib_indices:
            j = peak_k - i
            safe_mask[i] = False
            if j != i:
                safe_mask[j] = False

        if np.sum(safe_mask) == 0:
            safe_mask[:] = True

        # Distribute proportional to current values (to maintain shape)
        safe_vals = a[safe_mask].copy()
        total_safe = float(np.sum(safe_vals))
        if total_safe > 1e-15:
            a[safe_mask] += freed * safe_vals / total_safe
        else:
            n_safe = int(np.sum(safe_mask))
            if n_safe > 0:
                a[safe_mask] += freed / n_safe

        a = np.maximum(a, 0)
        a = a / a.sum()

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        else:
            a = best_a.copy()
            if it > 10:
                break

    return best_a


def enhanced_lp_polish(a, time_budget=300.0, verbose=False, seed=42):
    """
    Enhanced LP polishing with:
    - Multi-kernel LP ensemble
    - Both linear and geometric blending
    - Adaptive constraint tightening
    - Perturbative restarts
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    no_improve = 0
    last_t = 0.5

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start

        # Multi-kernel LP: solve with current a and a smoothed version
        kernels = [best_a]
        # Add a Gaussian-smoothed version
        sigma = max(1.0, n / 50.0)
        x = np.arange(n)
        kernel_gauss = np.exp(-((x - n // 2) ** 2) / (2 * sigma ** 2))
        from scipy.ndimage import gaussian_filter1d
        smoothed = gaussian_filter1d(best_a, sigma=n / 30.0)
        smoothed = np.maximum(smoothed, 1e-10)
        smoothed = smoothed / smoothed.sum()
        kernels.append(smoothed)

        g_candidates = []
        for kernel in kernels:
            g_lp, M_lp = solve_lp_minpeak(kernel)
            if g_lp is not None:
                g_candidates.append((g_lp, M_lp))

        if not g_candidates:
            break

        # Try each g candidate with both linear and geometric blending
        best_blend_obj = best_obj
        best_blend_a = best_a.copy()
        best_blend_info = None

        for g_lp, M_lp in g_candidates:
            # Linear blend with golden section
            def linear_blend_obj(t):
                a_try = (1.0 - t) * best_a + t * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    return 1e10
                a_try = a_try / s_try
                return compute_obj(a_try)

            t_opt = golden_section_search(linear_blend_obj, 0.01, 1.0, tol=0.02, max_iter=30)
            a_try = (1.0 - t_opt) * best_a + t_opt * g_lp
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            obj_try = compute_obj(a_try)

            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_blend_a = a_try.copy()
                best_blend_info = ('linear', t_opt, M_lp)

            # Geometric blend (only for first kernel to save time)
            if g_candidates.index((g_lp, M_lp)) == 0:
                def geo_blend_obj(t):
                    a_try = geometric_blend(best_a, g_lp, t)
                    return compute_obj(a_try)

                t_geo = golden_section_search(geo_blend_obj, 0.01, 0.8, tol=0.03, max_iter=25)
                a_geo = geometric_blend(best_a, g_lp, t_geo)
                obj_geo = compute_obj(a_geo)

                if obj_geo < best_blend_obj:
                    best_blend_obj = obj_geo
                    best_blend_a = a_geo.copy()
                    best_blend_info = ('geo', t_geo, M_lp)

        iter_count += 1

        if best_blend_obj < best_obj - 1e-14:
            best_obj = best_blend_obj
            best_a = best_blend_a.copy()
            no_improve = 0
            if verbose and iter_count % 5 == 0:
                print(f"    ELP it{iter_count}: obj={best_obj:.8f}, info={best_blend_info}, el={elapsed:.0f}s")
        else:
            no_improve += 1
            if no_improve == 2:
                # Try smaller linear step
                if g_candidates:
                    g_lp = g_candidates[0][0]
                    t_small = max(0.01, 0.1)
                    a_try2 = (1.0 - t_small) * best_a + t_small * g_lp
                    a_try2 = np.maximum(a_try2, 0)
                    a_try2 = a_try2 / a_try2.sum()
                    obj_try2 = compute_obj(a_try2)
                    if obj_try2 < best_obj - 1e-14:
                        best_obj = obj_try2
                        best_a = a_try2.copy()
                        no_improve = 0
            elif no_improve >= 4:
                # Perturb and continue
                scale = 0.005 + 0.003 * min(no_improve, 10)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 1e-10)
                a_pert = a_pert / a_pert.sum()
                no_improve = 0
                if iter_count > 80:
                    break

        # Periodic constraint-tightening
        if iter_count % 6 == 3 and g_candidates:
            M_lp = g_candidates[0][1]
            for tight_frac in [0.93, 0.96, 0.99]:
                M_tight = M_lp * tight_frac
                g_tight, _ = solve_lp_minpeak(best_a, M_target=M_tight)
                if g_tight is not None:
                    for t_try in [0.3, 0.5, 0.7]:
                        a_try3 = (1.0 - t_try) * best_a + t_try * g_tight
                        a_try3 = np.maximum(a_try3, 0)
                        a_try3 = a_try3 / a_try3.sum()
                        obj_try3 = compute_obj(a_try3)
                        if obj_try3 < best_obj - 1e-14:
                            best_obj = obj_try3
                            best_a = a_try3.copy()
                            no_improve = 0
                            if verbose:
                                print(f"    ELP-tight it{iter_count}: obj={best_obj:.8f}, frac={tight_frac}, t={t_try}")
                            break
                    else:
                        continue
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    from scipy.interpolate import CubicSpline
    try:
        x_old = np.linspace(0, 1, n)
        cs = CubicSpline(x_old, a, bc_type='natural')
        x_new = np.linspace(0, 1, new_n)
        a_new = cs(x_new)
        a_new = np.maximum(a_new, 0)
    except Exception:
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, new_n)
        a_new = np.interp(x_new, x_old, a)
        a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # U-shaped (high at edges)
        x = np.linspace(0, 1, n)
        a = 1.0 / (0.1 + np.abs(x - 0.5) * 4 + 1e-6)
        a = a / a.sum()
    elif mode == 3:
        # Asymmetric: heavy on one side
        x = np.linspace(0, 1, n)
        a = np.exp(-3.0 * x)
        a = a / a.sum()
    elif mode == 4:
        # Two bumps
        c1 = n // 4
        c2 = 3 * n // 4
        w = n // 8
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Cosine bell (concentrated in middle)
        x = np.linspace(-1, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 6:
        # Sine-modulated
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * 3 * x + rng.uniform(0, 6.28))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 7:
        # Power law at edges
        x = np.linspace(0.01, 1, n)
        a = x ** (-0.5) + (1 - x) ** (-0.5)
        a = a / a.sum()
    elif mode == 8:
        # Random bimodal
        c1 = rng.integers(n // 4, n // 3)
        c2 = rng.integers(2 * n // 3, 3 * n // 4)
        w1 = max(1, n // 10)
        w2 = max(1, n // 10)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w1 ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w2 ** 2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def annealing_search(a_start, time_budget=200.0, verbose=False, seed=42):
    """
    Simulated annealing with LP polish at each step.
    Perturb, polish briefly, accept/reject based on temperature.
    """
    n = len(a_start)
    a_current = a_start.copy()
    s = float(np.sum(a_current))
    a_current = a_current / s

    best_a = a_current.copy()
    best_obj = compute_obj(a_current)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    T = 0.01
    T_min = 0.001
    cooling = 0.95

    it = 0
    while time.monotonic() - start < time_budget:
        # Perturb
        scale = 0.01 + 0.02 * T / 0.01
        a_pert = a_current * np.exp(rng.normal(0, scale, n))
        a_pert = np.maximum(a_pert, 1e-10)
        a_pert = a_pert / a_pert.sum()

        # Brief LP polish
        remaining = time_budget - (time.monotonic() - start)
        polish_time = min(15.0, remaining * 0.3)
        if polish_time > 5:
            a_pert = enhanced_lp_polish(a_pert, time_budget=polish_time, seed=it)

        obj_pert = compute_obj(a_pert)
        delta = obj_pert - compute_obj(a_current)

        # Accept/reject
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-6)):
            a_current = a_pert.copy()
            if obj_pert < best_obj:
                best_obj = obj_pert
                best_a = a_pert.copy()
                if verbose:
                    print(f"    SA it{it}: obj={best_obj:.8f}, T={T:.5f}")
        else:
            # Try geometric blend as alternative
            if a_pert is not None and len(a_pert) == n:
                a_geo = geometric_blend(a_current, a_pert, 0.5)
                obj_geo = compute_obj(a_geo)
                if obj_geo < best_obj:
                    best_obj = obj_geo
                    best_a = a_geo.copy()
                    a_current = a_geo.copy()

        T *= cooling
        it += 1

        if T < T_min:
            # Reheat occasionally
            if it % 20 == 0:
                T = T_min * 5
            else:
                break

    return best_a

            if obj_follow < best_score:
                best_score = obj_follow
                best_seq = a_follow.copy()
                print(f"  Phase2 follow-up: obj={best_score:.8f} **")

    # === Phase 3: Peak-directed Subgradient ===
    print(f"\n=== Phase 3: Peak-directed Subgradient ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 100.0)
    if p3_budget > 20:
        a_sg = peak_directed_subgradient(best_seq, time_budget=p3_budget, K=8, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Phase3 SG: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 SG: obj={obj_sg:.8f}")
        # Quick LP polish after subgradient
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_pol = enhanced_lp_polish(best_seq, time_budget=min(remaining * 0.15, 80.0), seed=88)
            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  Phase3 polish: obj={best_score:.8f} **")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 200), max(400, n_current - 100),
        n_current + 100, n_current + 200, n_current + 300, n_current + 500,
        500, 600, 700, 750, 850, 900, 1000, 1100, 1200, 1400
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500 and abs(n - n_current) > 50]

    # Sort by proximity to current n first, then try others
    n_options.sort(key=lambda x: abs(x - n_current))

    for n_try in n_options[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)

        # Quick GD warmup
        alloc1 = min(remaining * 0.15, 30.0)
        if alloc1 > 10:
            a_try = nesterov_smoothed_descent(a_try, time_budget=alloc1, tau_init=10.0, tau_end=200.0)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.2, 80.0)
        if alloc2 > 15:
            a_try = enhanced_lp_polish(a_try, time_budget=alloc2, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Annealing Search ===
    print(f"\n=== Phase 5: Annealing Search ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        alloc_sa = min(remaining * 0.3, 250.0)
        a_sa = annealing_search(best_seq, time_budget=alloc_sa, verbose=True, seed=123)
        obj_sa = compute_obj(a_sa)
        if obj_sa < best_score:
            best_score = obj_sa
            best_seq = a_sa.copy()
            print(f"  Phase5 SA: obj={best_score:.8f} **")
        else:
            print(f"  Phase5 SA: obj={obj_sa:.8f}")

    # === Phase 6: Diverse Initializations + Polish ===
    print(f"\n=== Phase 6: Diverse Inits ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for mode in [2, 3, 7, 4]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_div = diverse_init(n_best, rng, mode=mode)
        # Quick polish
        alloc = min(remaining * 0.2, 60.0)
        if alloc > 15:
            a_div = enhanced_lp_polish(a_div, time_budget=alloc, seed=200 + mode)

        obj_div = compute_obj(a_div)
        if obj_div < best_score:
            best_score = obj_div
            best_seq = a_div.copy()
            print(f"  Diverse mode{mode}: obj={best_score:.8f} **")
        else:
            print(f"  Diverse mode{mode}: obj={obj_div:.8f}")

        # Try blending diverse with best
        if best_seq is not None and obj_div < 2.0:
            a_blend = geometric_blend(best_seq, a_div, 0.3)
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_blend = enhanced_lp_polish(a_blend, time_budget=min(remaining * 0.1, 40.0), seed=300 + mode)
            obj_blend = compute_obj(a_blend)
            if obj_blend < best_score:
                best_score = obj_blend
                best_seq = a_blend.copy()
                print(f"  Blend mode{mode}: obj={best_score:.8f} **")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 150.0)
        a_final = enhanced_lp_polish(best_seq, time_budget=alloc, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final ELP: obj={best_score:.8f} **")
        else:
            print(f"  Final ELP: obj={obj_final:.8f}")

    # Final surgical + subgradient
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_s = surgical_peak_reduction(best_seq, n_iter=30)
        obj_s = compute_obj(a_s)
        if obj_s < best_score:
            best_score = obj_s
            best_seq = a_s.copy()
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_sg = peak_directed_subgradient(best_seq, time_budget=min(remaining * 0.4, 40.0), K=10)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
