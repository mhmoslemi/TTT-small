    """One step of subgradient descent on max(a*a)."""
    n = len(a)
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:K]
    # Weights proportional to peak values
    w = conv[top_idx].copy()
    w = w / w.sum()

    # This is the cross-correlation of 'a' with the peak indicator
    peak_vec = np.zeros(2 * n - 1)

            g = g / s

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):

    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v

    cssv = np.cumsum(u)

    theta = (cssv[rho] - 1.0) / (rho + 1.0)

def golden_section_search(f, a_lo, a_hi, tol=0.02, max_iter=30):
    """Golden section search for minimum of f on [a_lo, a_hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if abs(a_hi - a_lo) < tol:
            break
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = f(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = f(d)
    return (a_lo + a_hi) / 2.0


def subgradient_descent(a, time_budget=60.0, K=20, verbose=True):
    """Subgradient descent with adaptive K, momentum, and peak weighting."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.5 / np.sqrt(n)
    momentum = 0.85
    v = np.zeros(n)
    stall = 0
    current_K = K
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        if max_conv < 1e-15:
            break

        actual_K = min(current_K, n)
        top_idx = np.argsort(conv)[::-1][:actual_K]
        top_vals = conv[top_idx]

        w = np.exp((top_vals - top_vals[0]) * 8.0 / max(max_conv, 1e-15))
        w /= w.sum()

        peak_vec = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            peak_vec[int(k)] = w[i]

        size = 1
        while size < 3 * n:
            size <<= 1
        P = rfft(peak_vec, size)
        A = rfft(a, size)
        g_full = irfft(P * A, size)
        g = g_full[:n]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 150:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break
                current_K = min(current_K + 15, n // 2)
            else:
                lr *= 0.998

        it += 1
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    SG it{it}: obj={best_obj:.8f}, K={current_K}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a


def peak_knock(a, n_steps=50, eta_init=0.3):
    """Reduce the peak of a*a by iteratively reducing contributions to the top peak."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    eta = eta_init

    for step in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        if j_start > j_end:
            break

        contribution = a[k_star - j_end:k_star - j_start + 1][::-1]
        max_contrib = float(np.max(contribution))
        if max_contrib < 1e-15:
            break

        decay = np.exp(-eta * contribution / max_contrib)
        a[j_start:j_end + 1] *= decay

        s = float(np.sum(a))
        if s < 1e-10:
            break
        a = a / s

        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
        else:
            a = best_a.copy()
            eta *= 0.7
            if eta < 1e-4:
                break

    return best_a


def lp_polish(a, time_budget=600.0, verbose=True, seed=42):
    """
    LP-based polishing with adaptive line search, peak-knocking,
    constraint tightening, and perturbative restarts.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    no_improve = 0
    last_t = 0.5

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break

        a_norm = best_a / best_a.sum()
        g_lp, M_lp = solve_lp_scipy(a_norm)
        if g_lp is None:
            break

        # Multi-point line search, then golden section refinement
        candidates_t = [0.2, 0.4, 0.5, 0.6, 0.8, 1.0]
        best_t = 0
        best_blend_obj = best_obj
        best_a_try = None

        for t in candidates_t:
            a_try = (1 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_t = t
                best_a_try = a_try.copy()

        # If improvement found, refine with golden section
        if best_t > 0 and best_a_try is not None:
            lo = max(0.01, best_t * 0.5)
            hi = min(1.0, best_t * 1.5 + 0.1)
            if hi <= lo:
                hi = lo + 0.1

            def blend_obj_fn(t_val):
                a_tmp = (1 - t_val) * best_a + t_val * g_lp
                a_tmp = np.maximum(a_tmp, 0)
                s_tmp = float(np.sum(a_tmp))
                if s_tmp < 1e-10:
                    return 1e10
                a_tmp = a_tmp / s_tmp
                return compute_obj(a_tmp)

            t_gs = golden_section_search(blend_obj_fn, lo, hi, tol=0.02, max_iter=25)
            a_gs = (1 - t_gs) * best_a + t_gs * g_lp
            a_gs = np.maximum(a_gs, 0)
            s_gs = float(np.sum(a_gs))
            if s_gs > 1e-10:
                a_gs = a_gs / s_gs
                obj_gs = compute_obj(a_gs)
                if obj_gs < best_blend_obj:
                    best_blend_obj = obj_gs
                    best_t = t_gs
                    best_a_try = a_gs.copy()

        if best_a_try is not None and best_blend_obj < best_obj - 1e-14:
            # Apply peak-knock refinement
            a_knocked = peak_knock(best_a_try, n_steps=20, eta_init=0.2)
            obj_knocked = compute_obj(a_knocked)
            if obj_knocked < best_blend_obj:
                best_blend_obj = obj_knocked
                best_a_try = a_knocked

            best_obj = best_blend_obj
            best_a = best_a_try.copy()
            no_improve = 0
            last_t = best_t
        else:
            no_improve += 1
            if no_improve == 2:
                for t in [0.05, 0.1, 0.15]:
                    a_try = (1 - t) * best_a + t * g_lp
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        no_improve = 0
                        break
            elif no_improve == 4:
                # Try constraint-tightened LP
                if M_lp is not None:
                    for frac in [0.93, 0.96, 0.99]:
                        g_tight, _ = solve_lp_scipy(a_norm, M_target=M_lp * frac)
                        if g_tight is not None:
                            for t in [0.3, 0.5, 0.7]:
                                a_try = (1 - t) * best_a + t * g_tight
                                a_try = np.maximum(a_try, 0)
                                s_try = float(np.sum(a_try))
                                if s_try < 1e-10:
                                    continue
                                a_try = a_try / s_try
                                obj_try = compute_obj(a_try)
                                if obj_try < best_obj - 1e-14:
                                    best_obj = obj_try
                                    best_a = a_try.copy()
                                    no_improve = 0
                                    break
                            if no_improve == 0:
                                break
            elif no_improve >= 6:
                # Perturb and restart
                scale = 0.008 + 0.004 * min(no_improve - 6, 10)
                best_a = best_a * np.exp(rng.normal(0, scale, n))
                best_a = np.maximum(best_a, 0)
                best_a = best_a / best_a.sum()
                no_improve = 0
                if iter_count > 80:
                    break

        iter_count += 1
        if verbose and iter_count % 5 == 0:
            print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t:.3f}, M={M_lp:.6f}, el={elapsed:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        # U-shaped (high at ends)
        a = 1.0 / (np.sqrt(np.minimum(x + 1, n - x) + 0.1) + 0.5)
    elif mode == 3:
        # Cosine bump
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    elif mode == 4:
        # Double Gaussian
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 8)
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
    elif mode == 5:
        # Power law
        alpha = rng.uniform(0.5, 2.0)
        a = (x + 1) ** (-alpha)
    elif mode == 6:
        # Sine-modulated
        f = rng.uniform(1, 4)
        a = 1.0 + 0.4 * np.sin(2 * np.pi * f * x / n)
    elif mode == 7:
        # Symmetric arch
        center = n / 2.0
        a = 1.0 - 0.5 * (x / center - 1.0) ** 2
    elif mode == 8:
        # Exponential decay from center
        center = n / 2.0
        a = np.exp(-0.5 * np.abs(x - center) / (n * 0.1))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses hybrid LP + subgradient + peak-knocking
    with multi-length exploration and perturbation restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Quick subgradient warmup ===
    print(f"\n=== Phase 1: Subgradient Warmup ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.05, 50.0)
    if p1_budget > 10:
        a_sgd = subgradient_descent(best_seq, time_budget=p1_budget, K=20, verbose=False)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase1 SG: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 SG: obj={obj_sgd:.8f}")

    # Peak knock after warmup
    a_pk = peak_knock(best_seq, n_steps=30, eta_init=0.3)
    obj_pk = compute_obj(a_pk)
    if obj_pk < best_score:
        best_score = obj_pk
        best_seq = a_pk.copy()
        print(f"  Peak knock: obj={best_score:.8f} **")

    # === Phase 2: Main LP Polishing ===
    print(f"\n=== Phase 2: Main LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.40, 550.0)
    if p2_budget > 30:
        a_lp = lp_polish(best_seq, time_budget=p2_budget, verbose=True, seed=42)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_current = len(best_seq)
        n_options = sorted(set([
            max(400, n_current - 100),
            n_current + 50,
            n_current + 100,
            n_current + 200,
            n_current + 300,
            600, 800, 1000, 1200, 1400, 1600, 1800
        ]))
        n_options = [n for n in n_options if 400 <= n <= 2500 and abs(n - n_current) > 50]

        for n_try in n_options[:8]:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break

            print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
            a_try = resample_sequence(best_seq, n_try)

            alloc1 = min(remaining * 0.12, 25.0)
            if alloc1 > 8:
                a_try = subgradient_descent(a_try, time_budget=alloc1, K=15, verbose=False)

            alloc2 = min(remaining * 0.25, 80.0)
            if alloc2 > 15:
                a_try = lp_polish(a_try, time_budget=alloc2, verbose=False, seed=n_try)

            # Peak knock
            a_try = peak_knock(a_try, n_steps=15, eta_init=0.2)

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Diverse re-initialization + polish ===
    print(f"\n=== Phase 4: Diverse Restart ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_cur = len(best_seq)
        for mode in [2, 5, 7, 3]:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            a_init = diverse_init(n_cur, rng, mode)
            # Blend with current best
            a_blend = 0.5 * a_init + 0.5 * best_seq
            a_blend = a_blend / a_blend.sum()

            alloc = min(remaining * 0.2, 70.0)
            if alloc > 15:
                a_blend = lp_polish(a_blend, time_budget=alloc, verbose=False, seed=200 + mode)
                obj_blend = compute_obj(a_blend)
                if obj_blend < best_score:
                    best_score = obj_blend
                    best_seq = a_blend.copy()
                    print(f"  Mode {mode}: obj={best_score:.8f} **")
                else:
                    print(f"  Mode {mode}: obj={obj_blend:.8f}")

    # === Phase 5: Perturbation + re-polish ===
    print(f"\n=== Phase 5: Perturbation + Re-polish ===")
    remaining = deadline - time.monotonic()

    for r in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        scale = 0.005 + 0.003 * r
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc = min(remaining * 0.22, 80.0)
        if alloc > 15:
            a_pert = lp_polish(a_pert, time_budget=alloc, verbose=False, seed=500 + r)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Perturb {r} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Perturb {r}: obj={obj_pert:.8f}")

    # === Phase 6: Final intensive LP ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.55, 150.0)
        a_final = lp_polish(best_seq, time_budget=alloc, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # Final subgradient squeeze
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.4, 30.0)
        a_sg = subgradient_descent(best_seq, time_budget=alloc, K=30, verbose=False)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # Final peak knock
    a_pk = peak_knock(best_seq, n_steps=30, eta_init=0.15)
    obj_pk = compute_obj(a_pk)
    if obj_pk < best_score:
        best_score = obj_pk
        best_seq = a_pk.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
