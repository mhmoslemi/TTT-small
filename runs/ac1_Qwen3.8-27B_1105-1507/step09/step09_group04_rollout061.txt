where `d[m] = a[m-k]` (the direction).

This is the maximum of:
- For `m ≠ 2k`: `L_m(delta) = c[m] + 2*delta*d[m]` (linear)
- For `m = 2k`: `Q(delta) = c[2k] + 2*delta*d[2k] + delta^2` (quadratic)

The minimum of `max(L_0, ..., L_{2n-2}, Q)` can be found by:
- The max of linear functions is piecewise linear convex.
- Adding a quadratic makes it piecewise: for each linear function `L_m`, find where `L_m(delta) = Q(delta)`, giving candidates.
- The minimum is either at a "kink" point (where two linear functions cross) or where the derivative of the active function is zero.

Actually, for a simpler approach: since `f(delta)` is convex (max of convex functions), we can use bisection on the derivative. The subderivative at `delta` is `{2*d[m] + 2*delta*(m==2k) : m is active}`. We want to find where 0 is in the subdifferential.

More practically: the minimum of `max_m (c[m] + 2*delta*d[m])` (ignoring the quadratic term for a moment) is found by bisection: find `delta` where the slope changes sign. The slope is `2 * max(2*d[m] : m active at current delta)`.

For the full problem with the quadratic term, we can still use bisection since the function is convex and we can evaluate it and its subgradient in O(n).

    LP-based direction finding: minimize max_m conv(a, b)[m] s.t. b>=0, sum(b)=sum(a).
    Then blend with current solution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    # Build convolution matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]

    iter_count = 0
    last_t = 0.3
    no_improve = 0

    while time.monotonic() - start < time_budget:
        # Solve LP: min M s.t. A @ b <= M, b >= 0, sum(b) = 1
        b = cp.Variable(n, nonneg=True)
        M = cp.Parameter(nonneg=True)
        constraints = [A @ b <= M, cp.sum(b) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        solved = False
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                prob.solve(solver=solver)
                if b.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    b_val = np.maximum(b.value, 0)
                    s_b = float(np.sum(b_val))
                    if s_b > 1e-10:
                        b_val = b_val / s_b
                        solved = True
                        break
            except Exception:
                continue

        if not solved:
            break

        # Try different blending factors
        candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5,
                      max(0.05, last_t * 0.7), min(0.9, last_t * 1.3), last_t]
        best_t = last_t
        best_t_obj = best_obj
        best_a_local = best_a.copy()

        for t in candidates:
            a_try = (1.0 - t) * a + t * b_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()

        iter_count += 1
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t
            no_improve = 0
            if verbose and iter_count % 3 == 0:
                print(f"      LP it{iter_count}: obj={best_obj:.8f}, t={best_t:.3f}")
        else:
            no_improve += 1
            # Perturb
            scale = 0.01 * min(no_improve, 20)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            if no_improve > 30:
                break

    return best_a


def dual_subgradient(a, time_budget, verbose=False):
    """
    Dual subgradient method: maintain weights on convolution positions,
    use them to guide the primal update via projected subgradient descent.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Dual weights over 2n-1 convolution positions
    w = np.ones(2 * n - 1) / (2 * n - 1)
    lr = 2.0 / np.sqrt(n)
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        obj = float(np.max(conv))

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        # Weighted gradient: sum_m w[m] * grad(c_m)
        # grad(c_m)[j] = 2 * sum over valid pairs
        # For convolution c[m] = sum_j a[j]*a[m-j], grad wrt a[j] = 2*(a[m-j] + a[j-m]) (with bounds)
        # Efficient: gradient of a^T C_m a = 2 C_m a
        # C_m a is the cross-correlation of a with the "shifted" version

        # Actually, let's compute the weighted objective gradient directly
        # F_w(a) = sum_m w[m] * c[m] where c[m] = sum_j a[j]*a[m-j]
        # grad F_w = sum_m w[m] * 2 * (C_m a)
        # C_m a at position j = sum over valid k of a[k] where m-k=j, i.e., a[m-j] (with bounds)
        # So (C_m a)[j] = a[m-j] if 0<=m-j<n else 0, but we also need the symmetric part
        # Actually for the full quadratic form c[m] = a^T C_m a where (C_m)_{j,k} = 1 if m-j=k or m-k=j
        # This gives (C_m a)_j = a[m-j] + a[j-m] (with bounds), but since c[m] is symmetric,
        # grad c[m] at a[j] = 2*a[m-j] (for valid indices in the linear convolution)

        # Simpler: use FFT to compute the gradient of the weighted sum
        # sum_m w[m] * c[m] = sum_m w[m] * sum_j a[j]*a[m-j]
        # = sum_j a[j] * sum_m w[m]*a[m-j]
        # Let g[j] = sum_m w[m]*a[m-j] (this is a cross-correlation)
        # Then grad = 2 * g (with the appropriate bounds for linear vs circular)

        # For linear convolution, we need to be careful. Let's just compute directly.
        # grad_obj[j] = 2 * sum_{m: m-j in [0,n-1]} w[m] * a[m-j]
        # = 2 * sum_{k=0}^{n-1} w[j+k] * a[k]  (with bounds on w)
        # This is a convolution of w (reversed) with a, restricted to valid range.

        # Use FFT for efficiency
        N = 1
        while N < 3 * n:
            N <<= 1

        from scipy.fft import fft, ifft
        W = fft(w, N)
        A = fft(a, N)
        # Cross-correlation: sum_k w[j+k] * a[k] = ifft(W * conj(A))[j] ... need to be careful
        # Actually: (w * a)[j] = sum_k w[k] * a[j-k] (circular)
        # We want: g[j] = sum_k w[j+k] * a[k] for k=0..n-1 (with w indexed 0..2n-2)
        # This is the cross-correlation of w and a at lag -j (or correlation at lag j)
        # corr[j] = sum_k a[k] * conj(w[k-j]) ... let me just do it directly for clarity

        # Direct computation (O(n^2) but with early termination for bandwidth)
        # Actually for the weighted sum, the gradient at position j is:
        # 2 * sum_{m=max(0, n-1-j)}^{min(2n-2, n-1+j)} w[m] * a[m-j]
        # where m-j must be in [0, n-1]

        # Let's use the FFT approach properly
        # c = conv(a, a), and we want grad of sum_m w[m]*c[m]
        # By chain rule: d(sum_m w[m]*c[m])/d(a[j]) = 2 * sum_m w[m] * (dc[m]/d(a[j]))
        # dc[m]/d(a[j]) = a[m-j] if 0 <= m-j <= n-1, else 0
        # So grad[j] = 2 * sum_{m: 0<=m-j<=n-1} w[m] * a[m-j]
        # = 2 * sum_{k=0}^{n-1} w[j+k] * a[k]  (with w indexed 0..2n-2)
        # = 2 * (cross-correlation of w_rev and a)[j] where w_rev[m] = w[2n-2-m]

        # Use FFT: let u = w padded to length N, v = a padded to length N
        # cross_corr[j] = sum_k u[j+k] * v[k] = ifft(fft(u) * conj(fft(v)))[j + offset]
        # Actually this is getting complicated. Let me just do it with the original approach.

        # Simpler: compute the gradient using the fact that
        # sum_m w[m] * c[m] = sum_m w[m] * sum_j a[j] * a[m-j]
        # Let h = conv(w, a) (full convolution, length 2n-1 + n - 1 = 3n-2)
        # Then the expression = sum_j a[j] * h[j+n-1] (approximately, need to check indices)
        # Actually: conv(w, a)[m] = sum_k w[k] * a[m-k]
        # sum_m w[m] * sum_j a[j] * a[m-j] = sum_j a[j] * sum_m w[m] * a[m-j]
        # = sum_j a[j] * (conv(w_rev, a))[j] where w_rev[m] = w[2n-2-m]
        # Hmm, let me just verify: sum_m w[m] * a[m-j], the valid m range is max(0,j) to min(2n-2, j+n-1)
        # Let k = m-j: sum_{k=max(0,j)-(j)}^{min(2n-2,j+n-1)-j} w[k+j] * a[k]
        # = sum_{k=0}^{n-1} w[k+j] * a[k] (with w[k+j]=0 if out of range)
        # This is (conv(w, a))[j + (2n-2) - (n-1)] ... I'm getting confused.

        # Let me just compute it directly in a vectorized way:
        # grad[j] = 2 * dot(w[j:j+2n-1], a) for j=0..n-1 (with w padded)
        # Actually w has length 2n-1, and for grad[j]:
        # grad[j] = 2 * sum_{k=0}^{n-1} w[j+k] * a[k]  (w indexed 0 to 2n-2)
        # For j=0: sum_{k=0}^{n-1} w[k] * a[k]
        # For j=n-1: sum_{k=0}^{n-1} w[n-1+k] * a[k]

        # This can be computed as: for each j, it's the dot product of w[j:j+n] with a[0:n]
        # which is a sliding dot product = cross-correlation.

        # Using FFT: cross_corr[j] = sum_k w[j+k]*a[k]
        # = ifft(fft(w_padded) * conj(fft(a_padded)))[some_index]
        # Let w_pad = [w, zeros(n-1)] (length 3n-2)
        # a_pad = [zeros(n-1), a, zeros(n-1)] (length 3n-2)
        # Then ifft(fft(w_pad)*conj(fft(a_pad))) gives the cross-correlation...
        # This is getting too complicated. Let me just use a simple O(n*bandwidth) approach.

        # The bandwidth of w is typically small (concentrated near the max positions),
        # so we can compute the gradient efficiently.

        # For now, let's use a simpler approach: just compute the subgradient of the max
        # at the current point, which is the gradient of the active term(s).

        active = np.where(conv >= obj - 1e-10 * max(1, obj))[0]
        # Subgradient direction: average of gradients of active terms
        # grad c[m] at position j: dc[m]/da[j] = 2*a[m-j] if valid
        sg = np.zeros(n)
        for m in active:
            # c[m] = sum_j a[j]*a[m-j] (valid bounds)
            # dc[m]/da[j] = a[m-j] (from the j term) + a[j-m] (from the m-k=j term, i.e., k=m-j... wait)
            # c[m] = sum_{j} a[j] * a[m-j] where 0<=j<n and 0<=m-j<n
            # dc[m]/da[j] = a[m-j] (if 0<=m-j<n) + a[j-m] (if 0<=j-m<n) [from the symmetric term]
            # But actually in the linear convolution, c[m] = sum_{j=max(0,m-n+1)}^{min(n-1,m)} a[j]*a[m-j]
            # dc[m]/da[j] = a[m-j] if j is in the valid range, plus a[j-m] if j-m is valid
            # Hmm, for a symmetric quadratic form a^T C a, the gradient is 2Ca.
            # C_m has (C_m)_{j,k} = 1 if k = m-j (i.e., j+k=m)
            # So (C_m a)_j = a[m-j] if 0 <= m-j < n
            # And since C_m is symmetric, grad c[m] = 2 * C_m a, i.e., grad_j = 2*a[m-j] if valid.

            j_lo = max(0, m - n + 1)
            j_hi = min(n - 1, m)
            valid_j = np.arange(j_lo, j_hi + 1)
            sg[valid_j] += 2.0 * a[m - valid_j]

        if len(active) > 0:
            sg /= len(active)

        # Projected gradient step on simplex
        a_new = a - lr * sg
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1
        lr *= 0.999

        if it % 500 == 0 and verbose:
            elapsed = time.monotonic() - start
            print(f"      DSG it{it}: obj={2*n*best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def dpss_init(n, W=0.4, rng=None):
    """
    Initialize using the dominant eigenvector of the prolate matrix.
    The DPSS with time-bandwidth product nW has minimum sidelobe energy.
    """
    # Build the prolate matrix N (n x n) with N[j,k] = sinc((j-k)*2W)
    j = np.arange(n)
    idx = j[:, None] - j[None, :]
    # sinc(x) = sin(pi*x)/(pi*x), with sinc(0)=1
    N_mat = np.sinc(2.0 * W * idx)
    # N_mat[j,k] = sinc(2W*(j-k)) = sin(2W*pi*(j-k))/(2W*pi*(j-k))
    # Wait, numpy's sinc is sin(pi*x)/(pi*x), so np.sinc(2W*idx) = sin(2W*pi*idx)/(2W*pi*idx)
    # That's what we want for the DPSS.

    # Compute dominant eigenvector (power iteration for speed)
    v = np.ones(n) / np.sqrt(n)
    if rng is not None:
        v = rng.standard_normal(n)
        v = v / np.linalg.norm(v)

    for _ in range(200):
        v = N_mat @ v
        norm = np.linalg.norm(v)
        if norm < 1e-15:
            break
        v /= norm

    # Make non-negative and normalize
    v = np.maximum(v, 0.001)  # Small floor to avoid zeros
    v = v / v.sum()
    return v


def raised_cosine_init(n, alpha=1.0):
    """Raised cosine window initialization."""
    x = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * (x - n / 2) / (n * alpha)))
    a = np.maximum(a, 0.001)
    return a / a.sum()


def parabolic_init(n):
    """Parabolic (quadratic) profile."""
    x = np.linspace(0, 1, n)
    a = 1.0 - (2 * x - 1) ** 2
    a = np.maximum(a, 0.001)
    return a / a.sum()


def gaussian_init(n, sigma=0.3):
    """Gaussian profile."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 0.001)
    return a / a.sum()


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, np.maximum(a, 0))
    return a_new


def refine_block(a, block_size, time_budget, start_time, rng):
    """Optimize blocks of coordinates simultaneously using a local QP-like approach."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / s
    a = a / s
    conv = compute_conv(a)
    best_obj = float(np.max(conv))
    best_a = a.copy()

    num_blocks = n // block_size
    if num_blocks == 0:
        return a

    gr = (np.sqrt(5) - 1) / 2.0

    while time.monotonic() - start_time < time_budget:
        improved = False
        for b in range(num_blocks):
            start_idx = b * block_size
            end_idx = min(start_idx + block_size, n)
            idx = np.arange(start_idx, end_idx)
            block = a[idx].copy()
            b_sum = float(np.sum(block))

            # Coordinate descent within the block
            for k in idx:
                d = np.zeros(2 * n - 1)
                d[k:k + n] = a

                lo = -a[k]
                hi = max(a[k] * 0.5, 2.0 / n)
                hi = max(hi, lo + 1e-12)

                def f_del(delta):
                    v = conv + 2.0 * delta * d
                    v[2 * k] += delta * delta
                    return float(np.max(v))

                f0 = f_del(0)
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = f_del(c1)
                f2 = f_del(c2)

                for _ in range(40):
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = f_del(c1)
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = f_del(c2)

                delta = (lo + hi) / 2
                new_val = f_del(delta)
                if new_val < f0 - 1e-14:
                    a[k] = max(0.0, a[k] + delta)
                    conv += 2.0 * delta * d
                    conv[2 * k] += delta * delta
                    improved = True

        obj_now = float(np.max(conv))
        if obj_now < best_obj - 1e-13:
            best_obj = obj_now
            best_a = a.copy()

        if not improved:
            # Small perturbation
            if rng is not None:
                a = best_a * np.exp(rng.normal(0, 0.005, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv = compute_conv(a)
                best_obj = float(np.max(conv))
            else:
                break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: multi-phase optimization combining coordinate descent,
    LP polishing, dual subgradient, and diverse initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Parent load error: {e}")

    if best_seq is None:
        best_seq = dpss_init(n_target, W=0.35, rng=rng)
        best_score = compute_obj(best_seq)
        print(f"DPSS init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Initial Coordinate Descent ===
    print(f"\n=== Phase 1: Initial CD ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 30:
        a1 = coord_descent(best_seq, p1_budget, time.monotonic(), rng, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 CD: obj={obj1:.8f}")

    # === Phase 2: LP Polishing ===
    print(f"\n=== Phase 2: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.20, 250.0)
    if p2_budget > 30:
        a2 = lp_polish(best_seq, p2_budget, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj2:.8f}")

    # === Phase 3: Fine Coordinate Descent ===
    print(f"\n=== Phase 3: Fine CD ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.20, 250.0)
    if p3_budget > 20:
        a3 = coord_descent(best_seq, p3_budget, time.monotonic(), rng, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 CD: obj={obj3:.8f}")

    # === Phase 4: Dual Subgradient ===
    print(f"\n=== Phase 4: Dual Subgradient ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 150.0)
    if p4_budget > 30:
        a4 = dual_subgradient(best_seq, p4_budget, verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4 DSG: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 DSG: obj={obj4:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    n_options = []
    for n_try in [n_cur - 100, n_cur - 50, n_cur + 50, n_cur + 100, n_cur + 200, 600, 700, 900, 1000, 1200]:
        if 400 <= n_try <= 2000 and n_try not in n_options and abs(n_try - n_cur) > 20:
            n_options.append(n_try)
    n_options = sorted(n_options)[:4]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        # CD
        alloc1 = min(remaining * 0.4, 60.0)
        if alloc1 > 15:
            a_try = coord_descent(a_try, alloc1, time.monotonic(), rng)

        # LP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.3, 40.0)
        if alloc2 > 10:
            a_try = lp_polish(a_try, alloc2)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    inits = [
        ("dpss", lambda: dpss_init(n_best, W=0.3, rng=rng)),
        ("dpss2", lambda: dpss_init(n_best, W=0.5, rng=rng)),
        ("raised", lambda: raised_cosine_init(n_best, alpha=0.8)),
        ("parabolic", lambda: parabolic_init(n_best)),
        ("gaussian", lambda: gaussian_init(n_best, sigma=0.25)),
    ]

    for r, (name, fn) in enumerate(inits[:3]):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_r = fn()
        # Blend with best
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc1 = min(remaining * 0.4, 50.0)
        if alloc1 > 15:
            a_r = coord_descent(a_r, alloc1, time.monotonic(), rng)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.2, 30.0)
        if alloc2 > 10:
            a_r = lp_polish(a_r, alloc2)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {name}: obj={best_score:.8f} **")
        else:
            print(f"  Restart {name}: obj={obj_r:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # CD
    if remaining > 30:
        alloc = min(remaining * 0.30, 80.0)
        a_f = coord_descent(best_seq, alloc, time.monotonic(), rng, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # LP
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.30, 60.0)
        a_f = lp_polish(best_seq, alloc, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Another CD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.40, 50.0)
        a_f = coord_descent(best_seq, alloc, time.monotonic(), rng)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")

    # Last LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.50, 40.0)
        a_f = lp_polish(best_seq, alloc)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(100) / 100
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
