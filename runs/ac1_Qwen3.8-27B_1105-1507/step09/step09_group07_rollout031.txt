"""
Optimized step function search combining LP alternating, smoothed gradient descent,
dual-guided perturbation, and n-sweep. Targets score < 1.5027.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g). Shape (2n-1, n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, A=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt, dual_weights) or (None, None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    if A is None:
        A = build_conv_matrix(a_norm)

    # Use scipy linprog with HiGHS
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                # Get dual variables for the inequality constraints (A_ub)
                # In HiGHS, res.ineqlin.marginals gives duals for A_ub
                duals = np.zeros(2 * n - 1)
                if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                    if hasattr(res.ineqlin, 'marginals') and res.ineqlin.marginals is not None:
                        duals = np.abs(np.array(res.ineqlin.marginals))
                return g_val / s, float(res.x[n]), duals
    except Exception:
        pass

    return None, None, None

        iteration += 1
        if best_t_obj < best_obj - 1e-11:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local.copy()
            last_good_t = best_t_val
            stall = 0
            A_cache = None  # Invalidate cache since a changed
        else:
            stall += 1
            if stall > patience:
                # Dual-guided perturbation
                if duals is not None and np.max(duals) > 1e-10:
                    # Find the lags with highest dual weights (most constrained)
                    peak_lags = np.argsort(duals)[::-1][:5]
                    # Find positions that contribute most to these peak lags
                    contrib = np.zeros(n)
                    for k in peak_lags:
                        k = int(k)
                        for i in range(n):
                            j = k - i
                            if 0 <= j < n:
                                contrib[i] += duals[k] * a[j]
                        for i in range(n):
                            j = k - i
                            if 0 <= j < n:
                                contrib[j] += duals[k] * a[i]
                    
                    # Perturb: reduce "contributing" positions, increase "quiet" positions
                    quiet_mask = contrib < np.percentile(contrib, 30)
                    active_mask = contrib > np.percentile(contrib, 70)
                    
                    rng = np.random.default_rng(iteration * 7919 + n)
                    scale = 0.03 + 0.01 * (stall // patience)
                    noise = rng.normal(0, scale, n)
                    # Boost quiet positions, suppress active ones
                    noise[quiet_mask] += scale * 0.5
                    noise[active_mask] -= scale * 0.5
                    a = best_a * np.exp(noise)
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                    else:
                        a = best_a.copy()
                else:
                    # Random perturbation
                    rng = np.random.default_rng(iteration * 31337 + n)
                    scale = min(0.3, 0.03 + 0.01 * (stall // patience))
                    noise = rng.normal(0, scale, n)
                    a = best_a * np.exp(noise)
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                    else:
                        a = best_a.copy()

                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                last_good_t = 0.5
                A_cache = None

            else:
                A_cache = None

        if verbose and iteration % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"        LP-alt it{iteration}: obj={best_obj:.8f}, t={last_good_t:.3f}, "
                  f"stall={stall}, elapsed={elapsed:.1f}s")

    return best_a


def block_lp_optimize(a, time_budget, verbose=False):
    """
    Block coordinate LP: split variables into blocks, optimize each block via LP
    while others are fixed. This can escape local optima of full LP alternating.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    n_blocks = 4
    block_size = n // n_blocks
    block_starts = [i * block_size for i in range(n_blocks)]
    block_starts.append(n)

    for rnd in range(100):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        for b in range(n_blocks):
            if time.monotonic() - start > time_budget:
                break
            b_start = block_starts[b]
            b_end = block_starts[b + 1]
            b_len = b_end - b_start

            # The LP for the block: minimize M subject to conv(a, g) <= M, 
            # but g[i] = a[i] for i outside the block, and sum(g) = 1.
            # This is more complex. Instead, let's do a simplified version:
            # Fix all variables except the block, and find the best values for the block.
            
            # The convolution constraint: for each lag k,
            # sum_i a[i]*g[k-i] <= M
            # Split into block and fixed parts:
            # sum_{i in block} a[i]*g[k-i] + sum_{i not in block} a[i]*g[k-i] <= M
            # sum_{i in block} a[i]*g[k-i] <= M - fixed_part[k]
            
            # Let h = g[b_start:b_end] (the variable block)
            # The constraint becomes: A_block[h] <= M - fixed_part
            # where A_block is the (2n-1) x b_len submatrix
            
            # Also: sum(h) = 1 - sum(fixed)
            
            # This is still an LP with b_len+1 variables and 2n-1 constraints.
            # For b_len = 200, this is much faster than the full LP.
            
            fixed_sum = float(np.sum(a) - np.sum(a[b_start:b_end]))
            target_sum = 1.0 - fixed_sum
            if target_sum <= 1e-10:
                continue

            a_fixed = a.copy()

            # optimize the block variables. This is approximate but fast.

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode % 10 == 0:
        a = np.ones(n) / n
    elif mode % 10 == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode % 10 == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 3:
        # Anti-symmetric: large at ends, small in middle
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * np.cos(np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 4:
        # Bimodal
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.15) ** 2) + np.exp(-8 * (x - 0.85) ** 2)
        a = a / a.sum()
    elif mode % 10 == 5:
        # Tri-modal
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * (x - 0.1) ** 2) + np.exp(-6 * (x - 0.5) ** 2) + np.exp(-6 * (x - 0.9) ** 2)
        a = a / a.sum()
    elif mode % 10 == 6:
        freqs = rng.integers(2, 10, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 7:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(0.5, 2.0)
        a = a / a.sum()
    elif mode % 10 == 8:
        # Random sparse + dense
        a = rng.uniform(0, 1, n)
        mask = rng.random(n) < 0.3
        a[~mask] *= 0.1
        a = a / a.sum()
    else:
        # Anti-symmetric sine
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.4 * np.sin(np.pi * x) ** 2
        a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def antipodal_refinement(a, time_budget, verbose=False):
    """
    Exploit the anti-symmetric structure: minimize a[i]*a[n-1-i] for all i
    by iteratively pushing mass from "active" pairs to "quiet" positions.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        peak_val = float(np.max(conv))
        peak_lags = np.argsort(conv)[::-1][:3]

        improved = False
        for k_star in peak_lags:
            k_star = int(k_star)
            # Find the pairs contributing to this lag
            pairs = []
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    pairs.append((a[i] * a[j], i, j))
            pairs.sort(reverse=True)

            for prod, i, j in pairs[:5]:
                if prod < 1e-12:
                    break
                # Move mass from the larger element to a "quiet" position
                src = i if a[i] >= a[j] else j
                # Find quiet positions (low contribution to all peak lags)
                quiet_contrib = np.zeros(n)
                for k_p in peak_lags:
                    k_p = int(k_p)
                    for ii in range(n):
                        jj = k_p - ii
                        if 0 <= jj < n:
                            quiet_contrib[ii] += a[jj]
                            quiet_contrib[jj] += a[ii]

                quiet_positions = np.argsort(quiet_contrib)[:10]
                for target in quiet_positions:
                    if target == src:
                        continue
                    for frac in [0.01, 0.02, 0.05]:
                        delta = frac * a[src]
                        if delta <= 0 or delta >= a[src] * 0.9:
                            continue
                        a[src] -= delta
                        a[target] += delta
                        a = np.maximum(a, 0)
                        s = float(np.sum(a))
                        if s < 1e-10:
                            a[src] += delta
                            a[target] -= delta
                            break
                        a = a / s
                        obj = compute_obj(a)
                        if obj < best_obj - 1e-12:
                            best_obj = obj
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[src] += delta
                            a[target] -= delta
                            a = a / s
                            a = np.maximum(a, 0)
                            break
                    if improved:
                        break
            if improved:
                break

        if not improved:
            break

    return best_a
