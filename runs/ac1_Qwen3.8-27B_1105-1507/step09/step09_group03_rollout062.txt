    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_subgradient_descent(a, time_budget, verbose=False, seed=42):
    """Nesterov accelerated subgradient descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Nesterov parameters
    beta = 0.9  # momentum
    # Initial step size based on Lipschitz constant estimate
    # L ~ 2 * max ||a|| * n (rough bound)
    L_est = 2.0 * n * float(np.max(a))
    t = 1.0
    v = np.zeros(n)
    
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        # Compute subgradient at y
        conv = compute_conv(y)
        max_k = int(np.argmax(conv))
        
        # Occasionally use random active constraint
        if rng.random() < 0.1:
            top_indicep = np.argsort(conv)[::-1][:max(1, n // 10)]
            max_k = int(rng.choice(top_indicep))
        
        g = conv_subgradient_fast(y, max_k)
        
        # Polyak step size or diminishing
        # Use: step = alpha / sqrt(t)
        alpha = 1.0 / np.sqrt(n)
        step = alpha / np.sqrt(max(1, it))
        
        v = beta * v - step * g
        a_new = y + v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        
        if it % 5000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}")

        if it - best_it > 30000:

def mirror_descent_entropy(a, time_budget, verbose=False, seed=123):

    n = len(a)

def peak_targeted_moves(a, time_budget, verbose=False, seed=777):
    """Directly target peaks by moving mass from contributing positions."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_k = int(np.argmax(conv))
        
        # Find contributing pairs to this peak
        i_min = max(0, max_k - (n - 1))
        i_max = min(n - 1, max_k)
        
        # Contributions: a[i] * a[max_k - i] for i in [i_min, i_max]
        indices = np.arange(i_min, i_max + 1)
        partners = max_k - indices
        contribs = a[indices] * a[partners]
        
        # Sort by contribution
        order = np.argsort(contribs)[::-1]
        
        improved = False
        # Try reducing the top contributors
        for idx_order in order[:5]:
            i = indices[idx_order]
            j = partners[idx_order]
            
            if i == j:
                # Can only reduce (split into two)
                delta = 0.01 * a[i]
                if delta < a[i] * 0.5:
                    # Try moving half to a nearby position
                    for tgt in [max(0, i - 1), min(n-1, i + 1), rng.integers(0, n)]:
                        a_try = a.copy()
                        a_try[i] -= delta
                        a_try[tgt % n] += delta
                        a_try = np.maximum(a_try, 0)
                        s_try = a_try.sum()
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break
            else:
                # Move mass from i to reduce a[i]*a[j]
                for delta_frac in [0.05, 0.02, 0.01]:
                    delta = delta_frac * a[i]
                    if delta >= a[i]:
                        continue
                    for tgt in [rng.integers(0, n), (i + n//3) % n, (i + 2*n//3) % n]:
                        tgt = int(tgt)
                        a_try = a.copy()
                        a_try[i] -= delta
                        a_try[tgt] += delta
                        a_try = np.maximum(a_try, 0)
                        s_try = a_try.sum()
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break
        
        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break
        
        if verbose and it % 50 == 0:
            print(f"    PTC it{it}: obj={best_obj:.8f}")
    
    return best_a

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def lp_polish(a, time_budget=60.0, verbose=False):
    """LP-based polishing using cvxpy."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a

        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A_mat @ g <= M, cp.sum(g) == 1]

        solved = False
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)

                    g_val = np.maximum(g.value, 0)
                    s_g = float(np.sum(g_val))

                        g_val = g_val / s_g

                            a_try = (1 - t_blend) * a + t_blend * g_val
                            a_try = np.maximum(a_try, 0)

        it += 1
        
        if verbose and it % 5 == 0:

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):

        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 0.1))

    elif mode == 5:

        a = a / a.sum()

        a = a / a.sum()

Wait, I realize there might be an issue with my Nesterov subgradient descent. The step size and momentum need to be tuned more carefully. Let me reconsider.

For subgradient descent on the simplex:
- The function `f(a) = 2n * max_k conv(a,a)_k` is not Lipschitz on the simplex (it's a quadratic form).

import numpy as np

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def conv_subgradient_fast(a, k):
    """Gradient of (a*a)_k w.r.t. a: 2*a[k-m] for valid m."""
    n = len(a)
    grad = np.zeros(n)
    m_min = max(0, k - (n - 1))
    m_max = min(n - 1, k)
    if m_min > m_max:
        return grad
    indices = np.arange(m_min, m_max + 1)
    grad[indices] = 2.0 * a[k - indices]
    return grad


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_subgradient_descent(a, time_budget, verbose=False, seed=42, momentum_init=0.7,
                                  momentum_end=0.95, alpha_init=0.5):
    """Nesterov accelerated subgradient descent on the simplex with adaptive parameters."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    v = np.zeros(n)
    it = 0
    best_it = 0
    total_iter_est = 200000

    while time.monotonic() - start < time_budget:
        progress = min(1.0, it / total_iter_est)
        momentum = momentum_init + (momentum_end - momentum_init) * progress
        alpha = alpha_init / (1.0 + 0.01 * it)

        y = a + momentum * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        conv = compute_conv(y)
        max_k = int(np.argmax(conv))

        # Randomized subgradient: sometimes use a random near-max constraint
        if rng.random() < 0.2:
            num_top = max(1, n // 8)
            top_indices = np.argsort(conv)[::-1][:num_top]
            max_k = int(rng.choice(top_indices))

        g = conv_subgradient_fast(y, max_k)

        v = momentum * v - alpha * g
        a_new = project_simplex(y + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        if verbose and it % 20000 == 0:
            elapsed = time.monotonic() - start

            v = np.zeros(n)
            best_it = it
            # Perturb

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def lp_polish(a, time_budget=60.0, verbose=False):
    """LP-based polishing using cvxpy with two-stage approach."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(31415)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget and it < 80:
        a_norm = a / np.sum(a)
        A_mat = build_conv_matrix(a_norm)

        # Stage 1: min max
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A_mat @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        g_val = None
        M_val = None
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        g_val = g_val / s_g
                        M_val = float(M.value)
                        break
            except Exception:
                continue

        if g_val is None:
            break

        # Find active constraints
        conv_ag = A_mat @ g_val
        tol = 0.05 * max(float(np.max(conv_ag)), 1e-10)
        active = np.where(conv_ag >= float(np.max(conv_ag)) - tol)[0]

        if len(active) > 1:
            # Stage 2: min sum at active
            g2 = cp.Variable(n, nonneg=True)
            obj2 = cp.sum(A_mat[active, :] @ g2)
            constraints2 = [cp.sum(g2) == 1]
            prob2 = cp.Problem(cp.Minimize(obj2), constraints2)
            g2_val = None
            for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
                try:
                    prob2.solve(solver=solver)
                    if g2.value is not None and prob2.status in ('optimal', 'optimal_inaccurate'):
                        g2_val = np.maximum(g2.value, 0)
                        s_g2 = float(np.sum(g2_val))
                        if s_g2 > 1e-10:
                            g2_val = g2_val / s_g2
                            break
                except Exception:
                    continue

            if g2_val is not None:
                blended = (g_val + g2_val) / 2
            else:
                blended = g_val
        else:
            blended = g_val

        # Line search
        improved = False
        for t_blend in [0.5, 0.7, 1.0, 0.3, 0.8]:
            a_try = (1 - t_blend) * a + t_blend * blended
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 15:
                # Perturb
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
            if it > 60 and stall > 20:
                break

        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, M={M_val:.6f} if M_val else 'N/A'")

    return best_a


def simulated_annealing(a, time_budget, verbose=False, seed=271):
    """Simulated annealing on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    T_init = 0.01
    T_end = 0.0001
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        progress = min(1.0, it / 50000)
        T = T_init * (T_end / T_init) ** progress

        # Random perturbation: move mass between two random positions
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        delta = T * 0.5 * min(a[i], a[j]) * 2
        delta = min(delta, a[i] * 0.8)

        a_try = a.copy()
        a_try[i] -= delta
        a_try[j] += delta
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        obj_try = compute_obj(a_try)
        delta_obj = obj_try - best_obj

            a = a_try

                best_a = a_try.copy()
                best_it = it

        it += 1

        if verbose and it % 20000 == 0:
            print(f"    SA it{it}: obj={best_obj:.8f}, T={T:.6f}")

            a = best_a.copy()
            best_it = it

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def golden_blend(a1, a2, n_evals=8):
    """Find optimal blend of a1 and a2 via golden section search."""
    phi = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)
        a1_try = (1 - t1) * a1 + t1 * a2
        a1_try = np.maximum(a1_try, 0)
        s1 = a1_try.sum()
        f1 = compute_obj(a1_try / s1) if s1 > 1e-10 else np.inf
        a2_try = (1 - t2) * a1 + t2 * a2
        a2_try = np.maximum(a2_try, 0)
        s2 = a2_try.sum()
        f2 = compute_obj(a2_try / s2) if s2 > 1e-10 else np.inf
        if f1 < f2:
            hi = t2
        else:
            lo = t1
    t_best = (lo + hi) / 2
    a_best = (1 - t_best) * a1 + t_best * a2
    a_best = np.maximum(a_best, 0)
    s_best = a_best.sum()
    if s_best < 1e-10:
        return a1.copy()
    return a_best / s_best


def diverse_init(n, rng, mode=0):
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 3:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 0.05))
        a = a / a.sum()
    elif mode == 5:
        a = rng.exponential(1.0, n)
        a = a / a.sum()
    elif mode == 6:
        # Smoothed noise
        a = rng.normal(0, 1, n)
        # Smooth with Gaussian kernel
        kernel_size = max(3, n // 20)
        kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1) / (kernel_size // 2 + 1)) ** 2)
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 7:
        # Comb pattern
        a = np.zeros(n)
        tooth_width = max(1, n // 20)
        gap = max(1, n // 10)
        pos = 0
        while pos < n:
            a[pos:pos + tooth_width] = 1.0
            pos += tooth_width + gap
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Multi-n + diverse init exploration ===
    print(f"\n=== Phase 1: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Try different n values
    n_options = [400, 500, 600, 700, 800, 900, 1000, 1100, 1200, 1400]
    n_options = [n for n in n_options if abs(n - n_current) >= 100]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_try = resample_sequence(best_seq, n_try)
        # Quick optimization
        a_try = nesterov_subgradient_descent(a_try, time_budget=8.0, seed=n_try)
        obj_try = compute_obj(a_try)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # Try diverse initializations at current n
    for mode in [2, 4, 6, 7]:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        a_try = diverse_init(n_current, rng, mode=mode)
        a_try = nesterov_subgradient_descent(a_try, time_budget=5.0, seed=mode * 100)
        obj_try = compute_obj(a_try)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** mode={mode}: obj={best_score:.8f} **")
        else:
            print(f"  mode={mode}: obj={obj_try:.8f}")

    # === Phase 2: Main Nesterov Subgradient Descent ===
    print(f"\n=== Phase 2: Nesterov Subgradient Descent (Main) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 450.0)
    if p2_budget > 30:
        a_nsd = nesterov_subgradient_descent(best_seq, time_budget=p2_budget,
                                             verbose=True, seed=42,
                                             momentum_init=0.6, momentum_end=0.9,
                                             alpha_init=0.8)
        obj_nsd = compute_obj(a_nsd)
        if obj_nsd < best_score:
            best_score = obj_nsd
            best_seq = a_nsd.copy()
            print(f"  NSD: obj={best_score:.8f} **")
        else:
            print(f"  NSD: obj={obj_nsd:.8f}")

    # === Phase 3: Mirror Descent ===
    print(f"\n=== Phase 3: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 250.0)
    if p3_budget > 30:
        a_md = mirror_descent_entropy(best_seq, time_budget=p3_budget,
                                      verbose=True, seed=123)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            # Use golden blend
            a_blend = golden_blend(best_seq, a_md)
            obj_blend = compute_obj(a_blend)
            if obj_blend < best_score:
                best_score = obj_blend
                best_seq = a_blend.copy()
            else:
                best_score = obj_md
                best_seq = a_md.copy()
            print(f"  MD: obj={best_score:.8f} **")
        else:
            print(f"  MD: obj={obj_md:.8f}")

    # === Phase 4: LP Polish ===
    print(f"\n=== Phase 4: LP Polish ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.2, 200.0)
    if p4_budget > 30:
        a_lp = lp_polish(best_seq, time_budget=p4_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            a_blend = golden_blend(best_seq, a_lp)
            obj_blend = compute_obj(a_blend)
            if obj_blend < best_score:
                best_score = obj_blend
                best_seq = a_blend.copy()
            else:
                best_score = obj_lp
                best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 5: Peak-Targeted Moves ===
    print(f"\n=== Phase 5: Peak-Targeted Moves ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.15, 120.0)
    if p5_budget > 15:
        a_pt = peak_targeted_moves(best_seq, time_budget=p5_budget,
                                   verbose=True, seed=777)
        obj_pt = compute_obj(a_pt)
        if obj_pt < best_score:
            best_score = obj_pt
            best_seq = a_pt.copy()
            print(f"  PTM: obj={best_score:.8f} **")
        else:
            print(f"  PTM: obj={obj_pt:.8f}")

    # === Phase 6: Simulated Annealing ===
    print(f"\n=== Phase 6: Simulated Annealing ===")
    remaining = deadline - time.monotonic()
    p6_budget = min(remaining * 0.15, 120.0)
    if p6_budget > 15:
        a_sa = simulated_annealing(best_seq, time_budget=p6_budget,
                                   verbose=True, seed=271)
        obj_sa = compute_obj(a_sa)
        if obj_sa < best_score:
            best_score = obj_sa
            best_seq = a_sa.copy()
            print(f"  SA: obj={best_score:.8f} **")
        else:
            print(f"  SA: obj={obj_sa:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # NSD with different params
    if remaining > 30:
        alloc = min(remaining * 0.25, 80.0)
        a_f1 = nesterov_subgradient_descent(best_seq, time_budget=alloc,
                                            seed=999, momentum_init=0.8,
                                            momentum_end=0.95, alpha_init=0.3)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  Final NSD: obj={best_score:.8f} **")

    # LP polish
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.25, 60.0)
        a_f2 = lp_polish(best_seq, time_budget=alloc)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Peak moves
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.2, 40.0)
        a_f3 = peak_targeted_moves(best_seq, time_budget=alloc, seed=888)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Final PTM: obj={best_score:.8f} **")

    # One more NSD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 50.0)
        a_f4 = nesterov_subgradient_descent(best_seq, time_budget=alloc,
                                            seed=2024, momentum_init=0.5,
                                            momentum_end=0.85, alpha_init=1.0)
        obj_f4 = compute_obj(a_f4)
        if obj_f4 < best_score:
            best_score = obj_f4
            best_seq = a_f4.copy()
            print(f"  Final NSD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
