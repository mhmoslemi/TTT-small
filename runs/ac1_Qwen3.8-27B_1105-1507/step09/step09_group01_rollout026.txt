    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_grad_fft(a, tau):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*c[k])) where c=conv(a,a).
    Returns (gradient, conv) with gradient length n. Assumes sum(a)=1.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    # result[m] = sum_{j=0}^{n-1} p[m+j] * a[j]
    # = convolve(p, a_flipped)[m + n - 1]
    a_flipped = a[::-1]
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    P = fft(p, N)
    Af = fft(a_flipped, N)
    conv_pa = ifft(P * Af).real
    result = conv_pa[n - 1:2 * n - 1]

    grad = (2.0 / tau) * result
    return grad, conv


def nesterov_smoothed(a, time_budget=300.0, tau_start=5.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    prev_a = a.copy()
    lr = 1.0 / np.sqrt(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.5)

        # Nesterov extrapolation
        t_prev = t_k
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = t_prev / t_k

        y = a + beta * (a - prev_a)
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        grad, conv = smoothed_grad_fft(y, tau)

        # Adaptive step size based on tau and n
        alpha = min(2.0 * tau / (tau + np.sqrt(n)), 1.0 / np.sqrt(n) * 3.0)
        a_new = y - alpha * grad
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        a = a_new
        it += 1

        # Restart if stuck
        if it - best_it > 20000:
            a = best_a.copy()
            prev_a = a.copy()
            t_k = 1.0
            best_it = it

        if verbose and it % 5000 == 0:
            print(f"      NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

    return best_a


def away_fw_smoothed(a, time_budget=200.0, tau=50.0, verbose=False):
    """Away-step Frank-Wolfe with smoothed gradient and line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        grad, conv = smoothed_grad_fft(a, tau)

        # Stay step: move towards argmin grad
        i_min = int(np.argmin(grad))
        e_min = np.zeros(n)
        e_min[i_min] = 1.0

        # Away step: move away from argmax grad (if it has mass)
        i_max = int(np.argmax(grad))
        use_away = a[i_max] > 1e-8

        best_t = 0.0
        best_f = best_obj

        # Stay step line search
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * e_min
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            f_try = compute_obj(a_try)
            if f_try < best_f:
                best_f = f_try
                best_t = t
                best_dir = 'stay'

        # Away step line search
        if use_away:
            lam = a[i_max]
            # a_away(t) = a * (1 - lam*(1-t)*0 + t*lam/(1-lam)) - t*lam/(1-lam) * e_max
            # Simplified: a_away(t)[i_max] = a[i_max]*(1-t), others scale up
            for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
                a_try = a.copy()
                a_try[i_max] = a[i_max] * (1.0 - t)
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                f_try = compute_obj(a_try)
                if f_try < best_f:
                    best_f = f_try
                    best_t = t
                    best_dir = 'away'

        # Apply best step
        if best_f < best_obj - 1e-14:
            if best_dir == 'stay':
                a_new = (1.0 - best_t) * a + best_t * e_min
            else:
                a_new = a.copy()
                a_new[i_max] = a[i_max] * (1.0 - best_t)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            a_new = a_new / s_new
            best_obj = best_f
            best_a = a_new.copy()
            a = a_new
            it += 1

        # Adaptive tau: increase as we converge
        if it % 50 == 0:
            tau *= 1.3

        if verbose and it % 10 == 0:
            print(f"      AFW it{it}: obj={best_obj:.8f}, tau={tau:.1f}")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, M_target=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue

    return None, None


def lp_polish(a, time_budget=100.0, verbose=False, seed=42):
    """LP-based polishing with golden-section blend and constraint tightening."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Try several blend factors
        best_t = 0.0
        best_f = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            f_try = compute_obj(a_try)
            if f_try < best_f:
                best_f = f_try
                best_t = t

        if best_f < best_obj - 1e-14:
            a_try = (1.0 - best_t) * best_a + best_t * g_lp
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            best_obj = best_f
            best_a = a_try.copy()
            a = best_a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve == 2:
                # Perturb
                scale = 0.005 * min(no_improve, 6)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
            elif no_improve >= 5:
                break

        # Periodic constraint tightening
        if iter_count % 6 == 3 and M_lp is not None:
            for frac in [0.93, 0.96]:
                g_t, _ = solve_lp_minpeak(a, M_target=M_lp * frac)
                if g_t is not None:
                    for t in [0.3, 0.5, 0.7]:
                        a_try = (1.0 - t) * best_a + t * g_t
                        a_try = np.maximum(a_try, 0)
                        a_try = a_try / a_try.sum()
                        f_try = compute_obj(a_try)
                        if f_try < best_obj - 1e-14:
                            best_obj = f_try
                            best_a = a_try.copy()
                            a = best_a.copy()
                            no_improve = 0
                            break
                    if no_improve == 0:
                        break

        iter_count += 1
        if verbose and iter_count % 5 == 0:
            print(f"      LP it{iter_count}: obj={best_obj:.8f}, t={best_t:.2f}, el={time.monotonic()-start:.0f}s")

    return best_a


def subgradient_refine(a, time_budget=80.0, K=5, verbose=False):
    """Subgradient method targeting top-K peaks with momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.5 / np.sqrt(n)
    momentum = 0.6
    v = np.zeros(n)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        weights = np.exp((peak_vals - max_pv) * 8.0 / max_pv)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break
            else:
                lr *= 0.998

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x * x + 0.01))
        a = a / a.sum()
    elif mode == 3:
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    elif mode == 4:
        # Arcsine-like distribution (heavier at edges)
        x = np.linspace(0.01, 0.99, n)
        a = 1.0 / np.sqrt(x * (1 - x))
        a = a / a.sum()
    elif mode == 5:
        # Beta distribution (various shapes)
        alpha = rng.uniform(0.5, 3.0)
        beta = rng.uniform(0.5, 3.0)
        x = np.linspace(0.01, 0.99, n)
        a = x ** (alpha - 1) * (1 - x) ** (beta - 1)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Nesterov smoothed GD with tau continuation, away-step FW,
    multi-length exploration, and LP polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Nesterov smoothed GD with tau continuation ===
    print(f"\n=== Phase 1: Nesterov Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.20, 250.0)
    if p1_budget > 30:
        a_ns = nesterov_smoothed(best_seq, time_budget=p1_budget,
                                 tau_start=5.0, tau_end=400.0, verbose=True)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Phase1 NSD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 NSD: obj={obj_ns:.8f}")

    # === Phase 2: Away-step FW ===
    print(f"\n=== Phase 2: Away-Step FW ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 180.0)
    if p2_budget > 20:
        a_fw = away_fw_smoothed(best_seq, time_budget=p2_budget, tau=30.0, verbose=True)
        obj_fw = compute_obj(a_fw)
        if obj_fw < best_score:
            best_score = obj_fw
            best_seq = a_fw.copy()
            print(f"  Phase2 AFW: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 AFW: obj={obj_fw:.8f}")

    # === Phase 3: Subgradient refinement ===
    print(f"\n=== Phase 3: Subgradient Refine ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.08, 80.0)
    if p3_budget > 15:
        a_sg = subgradient_refine(best_seq, time_budget=p3_budget, K=5)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Phase3 SG: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 SG: obj={obj_sg:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Generate candidate lengths
    n_options = sorted(set([
        max(300, n_current - 300),
        max(300, n_current - 150),
        n_current + 150,
        n_current + 300,
        n_current + 500,
        500, 700, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000 and abs(n - n_current) > 100]
    n_options = n_options[:7]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)

        # Quick Nesterov warmup
        alloc1 = min(remaining * 0.12, 40.0)
        if alloc1 > 10:
            a_try = nesterov_smoothed(a_try, time_budget=alloc1,
                                      tau_start=10.0, tau_end=200.0)

        # FW polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.08, 30.0)
        if alloc2 > 10:
            a_try = away_fw_smoothed(a_try, time_budget=alloc2, tau=50.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        # Generate a diverse starting point
        n_r = len(best_seq)
        mode = r % 6
        a_div = diverse_init(n_r, rng, mode=mode)

        # Blend with current best
        blend = 0.3 + 0.1 * r
        a_div = blend * a_div + (1 - blend) * best_seq
        a_div = np.maximum(a_div, 0)
        a_div = a_div / a_div.sum()

        alloc = min(remaining * 0.15, 80.0)
        if alloc > 15:
            a_div = nesterov_smoothed(a_div, time_budget=alloc * 0.5,
                                      tau_start=10.0, tau_end=300.0)
            remaining = deadline - time.monotonic()
            alloc2 = min(remaining * 0.1, 40.0)
            if alloc2 > 10:
                a_div = away_fw_smoothed(a_div, time_budget=alloc2, tau=80.0)

            obj_div = compute_obj(a_div)
            if obj_div < best_score:
                best_score = obj_div
                best_seq = a_div.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_div:.8f}")

    # === Phase 6: Perturbation + Nesterov ===
    print(f"\n=== Phase 6: Perturbation + Re-optimize ===")
    remaining = deadline - time.monotonic()

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        scale = 0.005 + 0.005 * r
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc = min(remaining * 0.15, 60.0)
        if alloc > 15:
            a_pert = nesterov_smoothed(a_pert, time_budget=alloc,
                                       tau_start=20.0, tau_end=300.0)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Perturb {r} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Perturb {r} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 7: Final LP polish ===
    print(f"\n=== Phase 7: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.5, 150.0)
        a_final = lp_polish(best_seq, time_budget=alloc, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # === Phase 8: Final subgradient ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 40.0)
        a_sg = subgradient_refine(best_seq, time_budget=alloc, K=8)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
