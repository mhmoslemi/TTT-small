"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based active set refinement, smoothed gradient descent with Nesterov acceleration,
peak-targeted coordinate descent, and multi-length exploration.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, active_indices=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k (or active_indices), sum(g)=1, g>=0.
    Returns (g, M_value, dual_vars).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)

    if active_indices is not None and len(active_indices) < A.shape[0]:
        # Minimize M with only active constraints
        M = cp.Variable()
        active_A = A[active_indices, :]
        constraints = [active_A @ g <= M, cp.sum(g) == 1]
    else:
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    m_val = float(M.value)
                    # Get dual variables if available
                    dual = None
                    if hasattr(prob, 'value') and M.value is not None:
                        try:
                            # Try to get dual for the inequality constraints
                            if len(active_indices) < A.shape[0] and active_indices is not None:
                                dual = None  # Not easily accessible
                            else:
                                dual = None
                        except:
                            dual = None
                    return g_val, m_val, dual
        except Exception:
            continue
    return None, None, None


def solve_lp_minpeak_full(a):
    """Solve full LP: min M s.t. conv(a,g) <= M for all k, sum(g)=1, g>=0."""
    return solve_lp_minpeak(a, active_indices=None)


def golden_section_line_search(a, g, n_evals=8):
    """Find optimal t in [0,1] that minimizes compute_obj((1-t)*a + t*g)."""
    phi = (np.sqrt(5) - 1) / 2

    lo = 0.0
    hi = 1.0

    for _ in range(n_evals):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)

        a1 = (1 - t1) * a + t1 * g
        a1 = np.maximum(a1, 0)
        s1 = float(np.sum(a1))
        if s1 > 1e-10:
            a1 = a1 / s1
            f1 = compute_obj(a1)
        else:
            f1 = np.inf

        a2 = (1 - t2) * a + t2 * g
        a2 = np.maximum(a2, 0)
        s2 = float(np.sum(a2))
        if s2 > 1e-10:
            a2 = a2 / s2
            f2 = compute_obj(a2)
        else:
            f2 = np.inf

        if f1 < f2:
            hi = t2
        else:
            lo = t1

    t_best = (lo + hi) / 2
    a_best = (1 - t_best) * a + t_best * g
    a_best = np.maximum(a_best, 0)
    s_best = float(np.sum(a_best))
    if s_best > 1e-10:
        a_best = a_best / s_best
    return a_best

            break

        max_conv = float(np.max(conv_ag))

        active = np.where(conv_ag >= max_conv - tol)[0]

        # Add some margin constraints

        # Choose direction
        if g_reduced is not None:
            # Compare: how does each direction affect the full max?
            conv_full_1 = compute_conv(a * g_reduced)  # Not quite right

            A_full = build_conv_matrix(a)
            conv_g_red = A_full @ g_reduced
            max_g_red = float(np.max(conv_g_red))

            conv_g_lp = A_full @ g_lp
            max_g_lp = float(np.max(conv_g_lp))

            if max_g_red < max_g_lp:
                g_dir = g_reduced
            else:

def smoothed_gradient_fft(a, tau):
    """
    Compute gradient of smoothed max of conv(a,a) using FFT.
    Uses log-sum-exp smoothing.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv

    prev_a = a.copy()
    prev_g = None
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        if it % 3000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)

        # Barzilai-Borwein step size
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g

            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:

                lr = 0.6 * lr + 0.4 * min(max(bb_lr, 1e-8), 5.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))

            v = np.zeros(n)

        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)

        if it - best_it > 50000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5

def peak_targeted_optimization(a, time_budget=60.0, verbose=False):
    """
    Targeted optimization: for the dominant peak, find the best perturbation
    to reduce it while keeping other peaks controlled.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Get top peaks
        top_k = 5
        peak_indices = np.argsort(conv)[::-1][:top_k]
        max_val = float(conv[peak_indices[0]])

        improved = False
        for pk in peak_indices:
            # Find contributing pairs
            i_start = max(0, pk - (n - 1))
            i_end = min(n - 1, pk)

            if i_start > i_end:
                continue

            # Compute contributions
            contribs = []
            for i in range(i_start, i_end + 1):
                j = pk - i
                if 0 <= j < n:
                    contribs.append((a[i] * a[j], i, j))

            if not contribs:
                continue

            contribs.sort(reverse=True)

            # Try reducing top contributors
            for contrib_val, src, dst in contribs[:5]:
                if src == dst:
                    continue

                # Try several reduction amounts
                for frac in [0.1, 0.05, 0.02]:
                    delta = frac * a[src]
                    if delta > a[src] * 0.5:
                        continue

                    # Try redistributing to several targets
                    targets = list(set([
                        int(rng.integers(0, n)),
                        int(rng.integers(0, n)),
                        (src + n // 3) % n,
                        (src + 2 * n // 3) % n,
                        n - 1 - src
                    ]))

                    for tgt in targets:
                        tgt = int(tgt) % n
                        if tgt == src:
                            continue
                        a_try = a.copy()
                        a_try[src] -= delta
                        a_try[tgt] += delta
                        s_try = float(np.sum(a_try))
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 80:
                # Larger perturbation
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if it > 500:
                    break

        if verbose and it % 30 == 0:
            print(f"    PTO it{it}: obj={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two bump
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine (flat autocorrelation for infinite support)
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / (np.sqrt(1 - x**2))
        a = a / a.sum()
    elif mode == 4:
        # Sine mixture
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = 1.0
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        # Sawtooth
        x = np.linspace(0, 1, n)
        a = np.abs(2 * x - 1) + 0.1
        a = a / a.sum()
    elif mode == 8:
        # Alternating high-low
        a = np.ones(n)
        a[::2] = rng.uniform(0.1, 0.5)
        a[1::2] = rng.uniform(0.8, 1.5)
        a = a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)
    return a


def fourier_domain_optimization(a, time_budget=120.0, verbose=False):

    DFT. We use/project the real-space constraints.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    it = 0
    stall = 0
    lr = 0.5 / n

    while time.monotonic() - start < time_budget:
        # Compute autocorrelation via FFT
        N = 1
        while N < 2 * n:
            N <<= 1
        A = fft(a, N)
        autocorr = ifft(A * A).real[:2 * n - 1]
        max_ac = float(np.max(autocorr))

        # Gradient of smoothed max

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous version:
1. Fixed LP solver bug (s_solver -> solver)
2. Constraint tightening loop for LP

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, M_bound=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Optionally with M <= M_bound.
    Returns (g, M_value).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_bound is not None:
        constraints.append(M <= M_bound)

    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_active(a, active_indices):
    """
    Solve LP with only active constraints: min M s.t. conv(a,g)[k] <= M for k in active,
    sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    active_A = A[active_indices, :]
    constraints = [active_A @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget=150.0, verbose=False):
    """
    LP-based polishing with constraint tightening and active set refinement.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(54321)

    iter_count = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        # Solve full LP
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        # Find active set
        A = build_conv_matrix(a)
        conv_ag = A @ g_lp
        max_val = float(np.max(conv_ag))
        tol = max(0.01 * max_val, 1e-10)
        active = np.where(conv_ag >= max_val - tol)[0]

        if len(active) == 0:
            active = np.array([np.argmax(conv_ag)])

        # Solve with active constraints only (faster, potentially better direction)
        g_active, M_active = solve_lp_active(a, active)

        # Choose best direction
        if g_active is not None:
            # Check which direction gives lower full max
            A_full = build_conv_matrix(a)
            conv_lp = A_full @ g_lp
            conv_act = A_full @ g_active
            full_max_lp = float(np.max(conv_lp))
            full_max_act = float(np.max(conv_act))

            if full_max_act < full_max_lp:
                g_dir = g_active
            else:
                # Blend
                g_dir = 0.7 * g_lp + 0.3 * g_active
                g_dir = np.maximum(g_dir, 0)
                g_dir = g_dir / g_dir.sum()
        else:
            g_dir = g_lp

        # Line search
        best_t_obj = best_obj
        best_t_a = a.copy()
        for t in [0.5, 0.8, 1.0, 0.3, 0.6, 0.9]:
            a_try = (1 - t) * a + t * g_dir
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_a = a_try.copy()

        iter_count += 1
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = best_t_a
            stall = 0
            if verbose and iter_count % 5 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, M_lp={M_lp:.6f}, |S|={len(active)}")
        else:
            stall += 1
            # Perturb
            scale = min(0.02 + 0.003 * stall, 0.08)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s_a = float(np.sum(a))
            if s_a > 1e-10:
                a = a / s_a
            else:
                a = best_a.copy()
            if stall > 30:
                break

    return best_a


def smoothed_grad(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g


def nesterov_descent(a, time_budget=200.0, tau_init=15.0, tau_end=400.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)

        if verbose and it % 5000 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}")

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_grad(y, tau)

        # Adaptive step size (Barzilai-Borwein)
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                lr = 0.7 * lr + 0.3 * min(max(bb, 1e-9), 8.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-9, lr * 0.99995)

        if it - best_it > 40000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def fourier_opt(a, time_budget=100.0, verbose=False):
    """Fourier-domain gradient optimization."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(31415)

    it = 0
    stall = 0
    lr = 0.3 / n

    while time.monotonic() - start < time_budget:
        N = 1
        while N < 2 * n:
            N <<= 1
        A = fft(a, N)
        autocorr = ifft(A * A).real[:2 * n - 1]
        max_ac = float(np.max(autocorr))

        tau = 40.0
        exp_ac = np.exp(tau * (autocorr - max_ac))
        p = exp_ac / np.sum(exp_ac)

        P = fft(p, N)
        grad = 2.0 * ifft(P * np.conj(A)).real[:n]

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.03, 3.0 / n)
        else:
            stall += 1
            lr *= 0.7
            if stall > 80:
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                lr = 0.3 / n
                stall = 0
                if it > 800:
                    break

        it += 1
        if verbose and it % 300 == 0:
            print(f"    FDO it{it}: obj={best_obj:.8f}")

    return best_a


def peak_cd(a, time_budget=50.0, verbose=False):
    """Peak-targeted coordinate descent."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(27182)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))

        i_start = max(0, peak_idx - (n - 1))
        i_end = min(n - 1, peak_idx)

        if i_start > i_end:
            break

        contribs = []
        for i in range(i_start, i_end + 1):
            j = peak_idx - i
            if 0 <= j < n:
                contribs.append((a[i] * a[j], i, j))

        contribs.sort(reverse=True)
        improved = False

        for cv, src, dst in contribs[:4]:
            if src == dst:
                continue
            for frac in [0.08, 0.04, 0.02]:
                delta = frac * a[src]
                if delta >= a[src]:
                    continue
                for tgt in [(src + n // 4) % n, (src + 3 * n // 4) % n,
                            int(rng.integers(0, n)), n - 1 - src]:
                    tgt = int(tgt) % n
                    if tgt == src:
                        continue
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[tgt] += delta
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-15:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 60:
                break

        if verbose and it % 20 == 0:
            print(f"    PCD it{it}: obj={best_obj:.8f}")

    return best_a


def resample(a, new_n):
    """Resample via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two bumps
        a = np.zeros(n)
        for c, w in [(n // 4, n // 12), (3 * n // 4, n // 12)]:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(-0.95, 0.95, n)
        a = 1.0 / (np.sqrt(1 - x ** 2))
        a = a / a.sum()
    elif mode == 4:
        # Sine mixture
        x = np.linspace(0, 1, n)
        a = 1.0
        for f, p in [(2, 0), (3, 1), (5, 2)]:
            a += 0.35 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Geometric decay (asymmetric)
        a = np.power(0.97, np.arange(n))
        a += np.power(0.97, np.arange(n)[::-1]) * 0.5
        a = a / a.sum()
    elif mode == 6:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.3 + 0.7 * np.cos(0.5 * x) ** 2
        a = a / a.sum()
    elif mode == 7:
        # Alternating
        a = np.ones(n)
        a[::2] *= 0.3
        a[1::2] *= 1.8
        a = a / a.sum()
    elif mode == 8:
        # Triangular
        x = np.linspace(0, 1, n)
        a = 1 - 2 * np.abs(x - 0.5) + 0.1
        a = a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n) * 0.3)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization: LP active set refinement, Nesterov smoothed descent,
    Fourier-domain optimization, peak-targeted CD, multi-length exploration,
    and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent: {e}")

    if best_seq is None:
        n_init = 600
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Init (n={n_init}): obj={best_score:.8f}")

    # === Phase 1: LP Polish ===
    print("\n=== Phase 1: LP Polish ===")
    remaining = deadline - time.monotonic()
    p1 = min(remaining * 0.25, 300.0)
    if p1 > 30:
        a1 = lp_polish(best_seq, time_budget=p1, verbose=True)
        o1 = compute_obj(a1)
        if o1 < best_score:
            best_score = o1
            best_seq = a1.copy()
            print(f"  LP: {best_score:.8f} **")
        else:
            print(f"  LP: {o1:.8f}")

    # === Phase 2: Nesterov Descent ===
    print("\n=== Phase 2: Nesterov ===")
    remaining = deadline - time.monotonic()
    p2 = min(remaining * 0.2, 200.0)
    if p2 > 25:
        a2 = nesterov_descent(best_seq, time_budget=p2, tau_init=10.0, tau_end=300.0, verbose=True)
        o2 = compute_obj(a2)
        if o2 < best_score:
            best_score = o2
            best_seq = a2.copy()
            print(f"  NSD: {best_score:.8f} **")
        else:
            print(f"  NSD: {o2:.8f}")

    # === Phase 3: Fourier Optimization ===
    print("\n=== Phase 3: Fourier ===")
    remaining = deadline - time.monotonic()
    p3 = min(remaining * 0.12, 100.0)
    if p3 > 15:
        a3 = fourier_opt(best_seq, time_budget=p3, verbose=True)
        o3 = compute_obj(a3)
        if o3 < best_score:
            best_score = o3
            best_seq = a3.copy()
            print(f"  FDO: {best_score:.8f} **")
        else:
            print(f"  FDO: {o3:.8f}")

    # === Phase 4: Peak CD ===
    print("\n=== Phase 4: Peak CD ===")
    remaining = deadline - time.monotonic()
    p4 = min(remaining * 0.08, 50.0)
    if p4 > 10:
        a4 = peak_cd(best_seq, time_budget=p4, verbose=True)
        o4 = compute_obj(a4)
        if o4 < best_score:
            best_score = o4
            best_seq = a4.copy()
            print(f"  PCD: {best_score:.8f} **")
        else:
            print(f"  PCD: {o4:.8f}")

    # === Phase 5: Multi-Length Exploration ===
    print("\n=== Phase 5: Multi-Length ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    n_opts = sorted([200, 300, 400, 500, 600, 700, 800, 900, 1000, 1100, 1200],
                    key=lambda x: abs(x - n_cur))
    n_opts = [x for x in n_opts if abs(x - n_cur) >= 50 and 150 <= x <= 1400]

    for nn in n_opts[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break
        a_try = resample(best_seq, nn)
        # Quick LP
        alloc = min(remaining * 0.12, 45.0)
        if alloc > 12:
            a_try = lp_polish(a_try, time_budget=alloc)
        # Quick NSD
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc2 = min(remaining * 0.1, 30.0)
            a_try = nesterov_descent(a_try, time_budget=alloc2, tau_init=20.0, tau_end=250.0)

        o_try = compute_obj(a_try)
        if o_try < best_score:
            best_score = o_try
            best_seq = a_try.copy()
            n_cur = nn
            print(f"  ** n={nn}: {best_score:.8f} **")
        else:
            print(f"  n={nn}: {o_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print("\n=== Phase 6: Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break
        mode = r % 9
        a_r = diverse_init(n_best, rng, mode=mode)
        a_r = 0.45 * a_r + 0.55 * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r
        else:
            a_r = best_seq.copy()

        alloc = min(remaining * 0.18, 55.0)
        if alloc > 15:
            a_r = lp_polish(a_r, time_budget=alloc * 0.6)
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_r = nesterov_descent(a_r, time_budget=min(remaining * 0.15, 30.0),
                                       tau_init=25.0, tau_end=350.0)

        o_r = compute_obj(a_r)
        if o_r < best_score:
            best_score = o_r
            best_seq = a_r.copy()
            print(f"  RS{r}(m={mode}): {best_score:.8f} **")
        else:
            print(f"  RS{r}(m={mode}): {o_r:.8f}")

    # === Phase 7: Final Polish ===
    print("\n=== Phase 7: Final ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.35, 80.0)
        a_f = lp_polish(best_seq, time_budget=alloc, verbose=True)
        o_f = compute_obj(a_f)
        if o_f < best_score:
            best_score = o_f
            best_seq = a_f.copy()
            print(f"  FinLP: {best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 50.0)
        a_f = nesterov_descent(best_seq, time_budget=alloc, tau_init=60.0, tau_end=800.0)
        o_f = compute_obj(a_f)
        if o_f < best_score:
            best_score = o_f
            best_seq = a_f.copy()
            print(f"  FinNSD: {best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 12:
        alloc = min(remaining * 0.25, 25.0)
        a_f = peak_cd(best_seq, time_budget=alloc)
        o_f = compute_obj(a_f)
        if o_f < best_score:
            best_score = o_f
            best_seq = a_f.copy()
            print(f"  FinPCD: {best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
