import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a, subset=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    N = 2 * n - 1
    if subset is not None:
        A = build_conv_matrix(a_norm)
        A_sub = A[subset, :]
    else:
        A = build_conv_matrix(a_norm)
        A_sub = A
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_sub @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCIPS, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def golden_section_search(a, g, lo=0.01, hi=1.0, tol=0.01):
    """Find optimal blend factor t in (1-t)*a + t*g."""
    def f(t):
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            return np.inf
        return compute_obj(a_try / s)
    
    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    
    while hi - lo > tol:
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    
    t_opt = (lo + hi) / 2
    return f(t_opt), t_opt

def peak_aware_perturbation(a, rng, n_peaks=5, scale=0.03):
    """Perturb a in a direction that specifically targets the current peaks."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv = compute_conv(a_norm)
    
    # Get top peak positions
    top_k = np.argsort(conv)[::-1][:n_peaks]
    
    # Build perturbation direction
    delta = np.zeros(n)

            if 0 <= idx < n:

            idx2 = j + k

    s = float(np.sum(a_new))

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    a_n = a / s
    return 2.0 * n * np.max(compute_conv(a_n))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        r = min(n, 2*n-1-j)
        A[j:j+r, j] = a[:r]
    return A

def solve_lp(a, constraint_indices=None):
    n = len(a)
    a_n = a / np.sum(a)
    A = build_conv_matrix(a_n)
    if constraint_indices is not None:
        A = A[constraint_indices]
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    prob = cp.Problem(cp.Minimize(M), [A @ g <= M, cp.sum(g) == 1])
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                s = gv.sum()
                if s > 1e-10:
                    return gv/s, M.value
        except:
            pass
    return None, None

def blend_search(a, g):
    """Golden section search for optimal blend factor."""
    def f(t):
        x = (1-t)*a + t*g
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-10:
            return np.inf
        return compute_obj(x/s)
    
    gr = (np.sqrt(5)-1)/2
    lo, hi = 0.0, 1.0
    c = hi - gr*(hi-lo)
    d = lo + gr*(hi-lo)
    fc, fd = f(c), f(d)
    for _ in range(30):
        if fc < fd:
            hi, d, fd = d, c, fc
            c = hi - gr*(hi-lo)
            fc = f(c)
        else:
            lo, c, fc = c, d, fd
            d = lo + gr*(hi-lo)
            fd = f(d)
    t = (lo+hi)/2
    return f(t), t

def peak_subgrad_perturb(a, rng, n_peaks=8, scale=0.02):
    """Subgradient-based perturbation targeting the peaks."""
    n = len(a)
    a_n = a / a.sum()
    conv = compute_conv(a_n)
    peaks = np.argsort(conv)[::-1][:n_peaks]
    
    grad = np.zeros(n)
    for k in peaks:
        k = int(k)
        lo = max(0, k-n+1)
        hi = min(n-1, k)
        if lo <= hi:
            js = np.arange(lo, hi+1)
            grad[js] += 2.0 * a_n[k-js]
    
    # Center the gradient (to preserve sum)
    grad -= grad.mean()
    gmax = np.abs(grad).max()
    if gmax < 1e-15:
        return a_n + rng.normal(0, 0.01, n)
    grad /= gmax
    
    a_new = a_n - scale * grad
    a_new += scale * 0.1 * rng.normal(0, 1, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return np.ones(n)/n
    return a_new / s

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

def diverse_init(n, rng, mode=0):

    elif mode == 3:

    elif mode == 4:

def lp_polish(a, max_iter=50, time_limit=60.0, verbose=False):
    """LP polishing with golden section blend search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() - start > time_limit:
            break
        g, M = solve_lp(a)
        if g is None:
            break
        
        obj_new, t_opt = blend_search(a, g)
        
        if obj_new < best_obj - 1e-13:
            a_new = (1-t_opt)*best_a + t_opt*g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            best_a = a_new
            best_obj = obj_new
            no_improve = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.3f}, M={M:.6f}")
        else:
            no_improve += 1
            if no_improve >= 3:
                break
    
    return best_a, best_obj, it+1

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based polishing with golden section line search, peak-aware subgradient
perturbation for escaping local minima, multi-length exploration, and adaptive
time allocation across optimization phases.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, verbose=False):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, max_time=5)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def blend_search(a, g, n_evals=25):
    """Golden section search for optimal blend factor t in (1-t)*a + t*g."""
    def f(t):
        x = (1.0 - t) * a + t * g
        x = np.maximum(x, 0)
        s = float(np.sum(x))
        if s < 1e-10:
            return np.inf
        return compute_obj(x / s)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)

    for _ in range(n_evals):
        if hi - lo < 0.005:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)

    t_opt = (lo + hi) / 2.0
    return f(t_opt), t_opt


def peak_subgrad_perturb(a, rng, n_peaks=8, scale=0.02):
    """
    Subgradient-based perturbation targeting current peaks.
    Moves in the negative subgradient direction of max_k (a*a)[k].
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv = compute_conv(a_norm)
    peaks = np.argsort(conv)[::-1][:n_peaks]

    grad = np.zeros(n)
    for k in peaks:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            grad[js] += 2.0 * a_norm[k - js]

    # Center to preserve sum
    grad -= grad.mean()
    gmax = float(np.abs(grad).max())
    if gmax < 1e-15:
        a_new = a_norm + 0.01 * rng.normal(0, 1, n)
    else:
        grad /= gmax
        a_new = a_norm - scale * grad + scale * 0.1 * rng.normal(0, 1, n)

    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(n) / n
    return a_new / s


def random_perturb(a, rng, scale=0.03):
    """Simple random multiplicative perturbation."""
    a_new = a * np.exp(rng.normal(0, scale, len(a)))
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a_new / s


def structured_perturb(a, rng, scale=0.02):
    """Perturbation that shifts mass from peak-contributing elements."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv = compute_conv(a_norm)
    peak_k = int(np.argmax(conv))

    # Elements that contribute most to the peak
    lo = max(0, peak_k - n + 1)
    hi = min(n - 1, peak_k)
    if lo <= hi:
        contrib = a_norm[peak_k - np.arange(lo, hi + 1)]
        total = contrib.sum()
        if total > 1e-10:
            a_new = a_norm.copy()
            redist = scale * a_norm[lo:hi + 1]
            a_new[lo:hi + 1] -= redist
            # Redistribute to low-contribution elements
            mask = np.ones(n, dtype=bool)
            mask[lo:hi + 1] = False
            if mask.sum() > 0:
                a_new[mask] += redist.sum() / mask.sum() * mask[mask]
            a_new = np.maximum(a_new, 0)
            s = float(np.sum(a_new))
            if s > 1e-10:
                return a_new / s

    return random_perturb(a_norm, rng, scale)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    else:

def lp_polish(a, max_iter=50, time_limit=60.0, verbose=False):
    """
    LP polishing: repeatedly solve LP for optimal g given current a,
    then find optimal blend via line search. Returns (best_a, best_obj, iterations).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf, 0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    no_improve = 0
    start = time.monotonic()
    it = 0

    while it < max_iter:
        if time.monotonic() - start > time_limit:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        obj_new, t_opt = blend_search(a, g_lp)

        it += 1

        if obj_new < best_obj - 1e-13:
            a_new = (1.0 - t_opt) * best_a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            best_a = a_new.copy()
            best_obj = obj_new
            no_improve = 0
            if verbose and (it % 10 == 0 or it == 1):
                print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.3f}, M={M_lp:.6f}")
        else:
            no_improve += 1
            if no_improve >= 4:
                break

    return best_a, best_obj, it

    remaining = deadline - time.monotonic()

    # Multiple LP rounds with no perturbation (pure convergence)
    if remaining > 30:
        alloc = min(remaining * 0.4, 100.0)
        a_f, obj_f, it_f = lp_polish(best_seq, max_iter=80, time_limit=alloc, verbose=True)
        if obj_f < best_score:
            best_score, best_seq = obj_f, a_f.copy()
        print(f"  Final LP: obj={best_score:.8f}")

    # One more round with tiny perturbation + LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_p = peak_subgrad_perturb(best_seq, rng, n_peaks=10, scale=0.008)
        alloc = min(remaining * 0.3, 60.0)
        a_f2, obj_f2, _ = lp_polish(a_p, max_iter=40, time_limit=alloc)
        if obj_f2 < best_score:
            best_score, best_seq = obj_f2, a_f2.copy()
            print(f"  Post-perturb LP: obj={best_score:.8f} **")

    # Try one more n value near the best
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_final = len(best_seq) + 50
        a_rf = resample_sequence(best_seq, n_final)
        alloc = min(remaining * 0.3, 50.0)
        a_rf, obj_rf, _ = lp_polish(a_rf, max_iter=30, time_limit=alloc)
        if obj_rf < best_score:
            best_score, best_seq = obj_rf, a_rf.copy()
            print(f"  Final n={n_final}: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
