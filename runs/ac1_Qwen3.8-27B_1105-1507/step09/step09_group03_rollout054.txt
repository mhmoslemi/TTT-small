Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based smoothed gradient descent with adaptive temperature, active-set LP polishing,
peak-shaving heuristics, and multi-length exploration.
Key innovations: efficient FFT gradient computation, two-phase tau schedule,
active-set reduction for LP, and structured n exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fft_gradient_normalized(a, tau):

    n = len(a)

    p = exp_vals / float(np.sum(exp_vals))

    S1 = ifft(np.conj(A) * P, N).real[:n]

        if verbose and it % 1000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")
    
    return best_a

def solve_active_lp(a, active_set, verbose=False):
    """
    Solve LP: maximize sum(g) / minimize max on active set.
    Actually: min M s.t. (A@g)[k] <= M for k in active_set, sum(g)=1, g>=0.
    """
    import cvxpy as cp
    
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix for active set only
    # (A@g)[k] = sum_j a[k-j]*g[j] (valid j)
    active_indices = active_set
    m = len(active_indices)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    constraints = [cp.sum(g) == 1]
    
    # Add constraints for active indices
    for k in active_indices:
        # (A@g)[k] = sum_{j: 0<=k-j<n} a[k-j] * g[j]
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            a_slice = a[j_min:k+1]  # a[k-j] for j from j_min to j_max
            # Wait: a[k-j] for j = j_min...j_max
            # When j = j_min: a[k - j_min]
            # When j = j_max: a[k - j_max]
            # So the coefficients are a[k-j_max], a[k-j_max+1], ..., a[k-j_min]
            # = a[j_min], a[j_min+1], ..., a[k] ... no
            # a[k-j] where j ranges from j_min to j_max
            # = a[k-j_max], a[k-j_max+1], ..., a[k-j_min]
            # Since j_min = max(0, k-n+1) and j_max = min(n-1, k):
            # k - j_max = k - min(n-1, k) = max(0, k-n+1) = j_min
            # k - j_min = k - max(0, k-n+1) = min(k, n-1) = j_max
            # So the coefficients are a[j_min], a[j_min+1], ..., a[j_max]
            # But in reverse order: a[k-j] for j=j_min is a[k-j_min] = a[j_max]
            # and for j=j_max is a[k-j_max] = a[j_min]
            # So the coefficient vector (for g[j_min], ..., g[j_max]) is:
            # a[j_max], a[j_max-1], ..., a[j_min]
            coeffs = a[j_min:j_max+1][::-1]
            constraints.append(coeffs @ g[j_min:j_max+1] <= M)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def lp_polish(a, time_budget=100.0, verbose=False):
    """
    Active-set LP polishing:
    1. Find active set (peaks of convolution)
    2. Solve LP on active set
    3. Blend and repeat
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Find active set: positions where conv is close to max
        tol = 0.05 * max_conv
        active = np.where(conv >= max_conv - tol)[0]
        
        # Limit active set size for efficiency
        if len(active) > 200:
            # Keep the top 200
            active_convs = conv[active]
            top_idx = np.argsort(active_convs)[::-1][:200]
            active = active[top_idx]
        
        g, M = solve_active_lp(a, active)
        if g is None:
            break
        
        # Line search: find best blend
        best_t = 0
        best_obj_blend = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_blend = (1 - t) * a + t * g
            a_blend = np.maximum(a_blend, 0)
            s_b = float(np.sum(a_blend))
            if s_b < 1e-10:
                continue
            a_blend = a_blend / s_b
            obj_blend = compute_obj(a_blend)
            if obj_blend < best_obj_blend:
                best_obj_blend = obj_blend
                best_t = t
        
        if best_obj_blend < best_obj - 1e-13:
            best_obj = best_obj_blend
            a_new = (1 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, |active|={len(active)}, t={best_t:.2f}")
        else:
            stall += 1
            if stall > 10:
                # Perturb
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if it > 50:
                    break
        
        it += 1
    
    return best_a

def peak_shaving(a, time_budget=30.0, verbose=False):
    """
    Directly target the peaks: for each peak, reduce the contributing elements.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    
    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_k = int(np.argmax(conv))
        
        # Find top contributing pairs to this peak
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        
        contributions = []
        for i in range(i_start, i_end + 1):
            j = max_k - i
            if 0 <= j < n:
                contributions.append((a[i] * a[j], i, j))
        
        contributions.sort(reverse=True)
        
        improved = False
        # Try reducing the top contributors
        for contrib_val, i, j in contributions[:5]:
            if a[i] < 1e-8 or a[j] < 1e-8:
                continue
            
            # Try moving mass from the larger one to a random position
            if a[i] >= a[j]:
                src = i
            else:
                src = j
            
            delta = 0.01 * a[src]
            # Move to a position that's far from the peak
            # Good targets: positions that don't contribute to the current peak
            target_candidates = []
            # Positions far from contributing to peak at max_k
            for t in [0, n-1, n//2, rng.integers(0, n)]:
                t = int(t) % n
                # Check if (t, max_k - t) would contribute to peak
                if not (i_start <= t <= i_end and 0 <= max_k - t < n):
                    target_candidates.append(t)
            
            if not target_candidates:
                target_candidates = [rng.integers(0, n)]
            
            tgt = target_candidates[0]
            if tgt == src:
                continue
            
            a_try = a.copy()
            a_try[src] -= delta
            a_try[tgt] += delta
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        
        it += 1
        if not improved and it % 10 == 0:
            break
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - n*0.25)**2) / (2*max(1,n//6)**2))
        a += np.exp(-((np.arange(n) - n*0.75)**2) / (2*max(1,n//6)**2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(-1, 1, n, endpoint=False)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-8) + 1e-8)
        a = a / a.sum()
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 5:
        # Random Fourier
        freqs = rng.integers(1, 10, size=6)
        phases = rng.uniform(0, 2*np.pi, size=6)
        amps = rng.uniform(0.1, 0.5, size=6)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.sin(2*np.pi*f*x + p)
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    else:
        # Trapezoidal
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        a[:n//5] = np.linspace(0.3, 1, n//5)
        a[-n//5:] = np.linspace(1, 0.3, n//5)
        a = a / a.sum()
    return a
