def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a, top_k=None):

    n = len(a)

        for j in range(n):
            r = min(n, 2 * n - 1 - j)
            A[j:j + r, j] = a[:r]
        return A
    else:

        for j in range(n):
            r = min(n, 2 * n - 1 - j)
            A[j:j + r, j] = a[:r]
        return A

    # Actually: (w * a)[i] = sum_k w[k] * a[i-k] (convolution)

def pressure_update(a, eta=0.1, K=50, temperature=0.1):
    """Multiplicative pressure-based update."""
    n = len(a)
    pressure = compute_pressure(a, K=K, temperature=temperature)
    p_max = pressure.max()
    if p_max < 1e-15:
        return a.copy()
    pressure_norm = pressure / p_max
    a_new = a * (1.0 - eta * pressure_norm)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def solve_lp_minpeak(a, use_top_k=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def optimal_t_blend(a, g, n_points=50):
    """
    Find optimal t in [0, 1] to minimize max(conv((1-t)*a + t*g, (1-t)*a + t*g)).
    Uses the quadratic structure: each conv coefficient is a quadratic in t.
    """
    n = len(a)
    
    # Precompute conv(a,a), conv(a,g), conv(g,g)
    conv_aa = compute_conv(a)
    conv_ag = compute_conv(a, g) if hasattr(compute_conv, '__code__') else None
    # Need cross-convolution
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g, size)
    conv_ag = irfft(fa * fg, size)[:2 * n - 1]
    conv_gg = irfft(fg * fg, size)[:2 * n - 1]
    
    # For each t: conv[t] = (1-t)^2 * conv_aa + 2t(1-t) * conv_ag + t^2 * conv_gg
    # Find t that minimizes max over k
    
    best_t = 0.0
    best_val = float(np.max(conv_aa))
    
    # Coarse grid
    for i in range(n_points + 1):
        t = i / n_points
        conv_t = (1-t)**2 * conv_aa + 2*t*(1-t) * conv_ag + t**2 * conv_gg
        val = float(np.max(conv_t))
        if val < best_val:
            best_val = val
            best_t = t
    
    # Fine search around best_t
    if 0 < best_t < 1:
        lo = max(0, best_t - 1.0/n_points)
        hi = min(1, best_t + 1.0/n_points)
        for i in range(21):
            t = lo + (hi - lo) * i / 20
            conv_t = (1-t)**2 * conv_aa + 2*t*(1-t) * conv_ag + t**2 * conv_gg
            val = float(np.max(conv_t))
            if val < best_val:
                best_val = val
                best_t = t
    
    return best_t, best_val


def lp_polish_v2(a, time_budget, verbose=False):
    """
    Advanced LP polishing with quadratic-aware t-selection and pressure updates between LPs.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    iter_count = 0
    rng = np.random.default_rng(123)
    consecutive_fails = 0
    last_t = 0.3
    pressure_cooldown = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        
        # Solve LP
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        iter_count += 1
        
        # Find optimal t using quadratic-aware method
        t_opt, val_opt = optimal_t_blend(a, g_lp, n_points=40)
        
        # Also try a few additional t values
        candidates = sorted(set([
            t_opt,
            max(0.01, t_opt * 0.7),
            min(1.0, t_opt * 1.3),
            min(1.0, t_opt + 0.1),
            max(0.01, t_opt - 0.1),
            last_t,
            0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0
        ]))
        
        best_t_obj = best_obj
        best_t_val = last_t
        best_a_local = best_a.copy()
        
        for t in candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            consecutive_fails = 0
            if verbose and iter_count % 3 == 0:
                print(f"    LPv2 it{iter_count}: obj={best_obj:.8f}, t={best_t_val:.4f}, M_lp={M_lp:.6f}")
        else:
            consecutive_fails += 1
            # Apply pressure-based update to escape
            if consecutive_fails >= 2 and consecutive_fails % 3 == 0:
                eta = 0.05 + 0.03 * min(consecutive_fails, 10)
                a = pressure_update(best_a, eta=eta, temperature=0.05)
                obj_p = compute_obj(a)
                if obj_p < best_obj:
                    best_obj = obj_p
                    best_a = a.copy()
                    if verbose:
                        print(f"    LPv2 it{iter_count}: pressure improved to {best_obj:.8f}")
                else:
                    a = best_a.copy()
            
            # Random perturbation if stuck longer
            if consecutive_fails >= 5:
                scale = 0.02 + 0.005 * min(consecutive_fails, 15)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
            
            if consecutive_fails > 25:
                if verbose:
                    print(f"    LPv2: stuck after {consecutive_fails} iterations")
                break
        
        # Check time
        if time.monotonic() - start > time_budget - 10:
            break

    return best_a


def peak_pair_surgery(a, time_budget, verbose=False):
    """
    For the top peaks in conv(a,a), identify the pairs (i, k-i) that contribute most
    and try to redistribute mass to reduce those peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(77)
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Find top peaks (excluding index 0 which is a[0]^2 and 2n-1 which is a[n-1]^2)
        top_peaks = np.argsort(conv)[::-1][:20]
        
        improved = False
        for k_peak in top_peaks[:10]:
            k_peak = int(k_peak)
            if k_peak < 0 or k_peak >= 2 * n - 1:
                continue
            
            # Find contributing pairs
            i_min = max(0, k_peak - (n - 1))
            i_max = min(n - 1, k_peak)
            
            if i_min > i_max:
                continue
            
            # Contributions from each pair
            contributions = np.array([a[i] * a[k_peak - i] for i in range(i_min, i_max + 1)])
            total_contrib = contributions.sum()
            if total_contrib < 1e-15:
                continue
            
            # The peak value is conv[k_peak], we want to reduce it
            # Strategy: move mass from high-contribution pairs to "neutral" positions
            # A position i is "neutral" for peak k if a[k-i] is small
            
            # Find the position with highest contribution
            top_idx_rel = np.argmax(contributions)
            top_i = i_min + top_idx_rel
            top_j = k_peak - top_i
            
            # Try moving a small amount from top_i to a random low-contribution position
            # Find a target position that doesn't contribute much to any top peak
            contrib_vector = np.zeros(n)
            for pk in top_peaks[:5]:
                pk = int(pk)
                for i in range(max(0, pk - (n-1)), min(n-1, pk) + 1):
                    contrib_vector[i] += a[pk - i]
            
            # Normalize
            cv_max = contrib_vector.max()
            if cv_max > 1e-15:
                contrib_norm = contrib_vector / cv_max
            else:
                contrib_norm = np.zeros(n)
            
            # Target: positions with low contribution and high current value (to balance)
            # Or positions with low value and low contribution (to add mass)
            target_score = -contrib_norm + 0.5 * (a / (a.max() + 1e-15))
            target_score[top_i] = -1  # Don't target the source
            if top_j != top_i:
                target_score[top_j] = min(target_score[top_j], 0)
            
            # Pick a few candidate targets
            top_targets = np.argsort(target_score)[::-1][:5]
            
            for target in top_targets:
                target = int(target)
                if target == top_i:
                    continue
                
                # Try moving different amounts
                for frac in [0.02, 0.05, 0.1, 0.15]:
                    move_amt = frac * a[top_i]
                    if move_amt < 1e-15:
                        continue
                    a_try = a.copy()
                    a_try[top_i] -= move_amt
                    a_try[target] += move_amt
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                
                if improved:
                    break
        
        if not improved:
            # Try a small random perturbation
            perturbation = rng.normal(0, 0.01, n)
            a_try = a * np.exp(perturbation)
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
            else:
                it += 1
                if it > 50:
                    break

        if verbose and it % 10 == 0:
            print(f"    Surgery it{it}: obj={best_obj:.8f}")

    return best_a


def multi_start_optimize(n, time_budget, rng, verbose=False):
    """Try multiple random initializations and pick the best one after quick optimization."""
    best_a = None
    best_obj = np.inf
    
    initializations = []
    
    # Uniform
    initializations.append(np.ones(n) / n)
    
    # Random
    initializations.append(rng.dirichlet(np.ones(n)))
    
    # Gaussian-like
    x = np.linspace(0, 1, n)
    mu = rng.uniform(0.3, 0.7)
    sig = rng.uniform(0.05, 0.2)
    initializations.append(np.exp(-((x - mu)**2) / (2*sig**2)))
    initializations[-1] /= initializations[-1].sum()
    
    # Two peaks
    c1, c2 = sorted(rng.choice(n, 2, replace=False))
    initializations.append(np.exp(-((np.arange(n) - c1)**2) / (n*0.01)) + 
                          np.exp(-((np.arange(n) - c2)**2) / (n*0.01)))
    initializations[-1] = np.maximum(initializations[-1], 0.01)
    initializations[-1] /= initializations[-1].sum()
    
    # Arcsine-like
    x = np.linspace(-0.99, 0.99, n)
    initializations.append(1.0 / (np.sqrt(1 - x**2) + 0.1))
    initializations[-1] /= initializations[-1].sum()
    
    # Power law
    initializations.append(np.linspace(0.1, 1, n) ** rng.uniform(-1, 3))
    initializations[-1] /= initializations[-1].sum()
    
    per_init_budget = time_budget / len(initializations)
    
    for idx, init in enumerate(initializations):
        init = np.maximum(init, 0).astype(np.float64)
        init = init / init.sum()
        
        # Quick pressure-based optimization
        for step in range(20):
            eta = 0.1 / (1 + step * 0.5)
            init = pressure_update(init, eta=eta, temperature=0.1)
        
        # Quick LP polish
        if per_init_budget > 20:
            init = lp_polish_v2(init, time_budget=min(per_init_budget, 60))
        
        obj = compute_obj(init)
        if obj < best_obj:
            best_obj = obj
            best_a = init.copy()
            if verbose:
                print(f"    Multi-start {idx}: obj={obj:.8f}")
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fine_tune(a, time_budget, verbose=False):
    """Final fine-tuning with very small perturbations and coordinate descent."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)
    
    while time.monotonic() - start < time_budget:
        improved = False
        
        # Try small random perturbations
        for scale in [0.005, 0.002, 0.001]:
            a_try = a * np.exp(rng.normal(0, scale, n))
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-15:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        
        # Try coordinate descent: move mass between two random positions
        for _ in range(50):
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            for frac in [0.01, 0.03, 0.05]:
                move = frac * min(a[i], a[j])
                if move < 1e-15:
                    continue
                for direction in [(i, j), (j, i)]:
                    src, dst = direction
                    a_try = a.copy()
                    a_try[src] -= move
                    a_try[dst] += move
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-15:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
        
        if not improved:
            # Try a pressure update with very small eta
            a_try = pressure_update(a, eta=0.01, temperature=0.02)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-15:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
            else:
                # Increasing scale
                scale = 0.01
                a_try = a * np.exp(rng.normal(0, scale, n))
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                else:
                    a = best_a.copy()
                    # Break if we've been stuck
                    if obj_try >= best_obj and scale > 0.5:
                        break
        
        if verbose:
            elapsed = time.monotonic() - start
            if int(elapsed) % 30 == 0:
                print(f"    Fine-tune t={elapsed:.0f}s: obj={best_obj:.8f}")
    
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Allocates time across:
    - Phase 1 (15%): Multi-start to find good initial point
    - Phase 2 (45%): Intensive LP polishing
    - Phase 3 (15%): Peak pair surgery  
    - Phase 4 (15%): Multi-length exploration
    - Phase 5 (10%): Fine-tuning
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")
    
    # === Phase 1: Quick pressure-based improvement + multi-start (15%) ===
    print(f"\n=== Phase 1: Quick Optimization ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.15, 200.0)
    
    if p1_budget > 20:
        # First try pressure updates on the parent
        a_press = best_seq.copy()
        for step in range(30):
            eta = 0.15 / (1 + step * 0.3)
            a_press = pressure_update(a_press, eta=eta, temperature=0.08)
        obj_press = compute_obj(a_press)
        if obj_press < best_score:
            best_score = obj_press
            best_seq = a_press.copy()
            print(f"  Pressure: obj={best_score:.8f} **")
        else:
            print(f"  Pressure: obj={obj_press:.8f}")
        
        # Multi-start with shorter budget
        remaining = deadline - time.monotonic()
        ms_budget = min(remaining * 0.5, 80.0)
        if ms_budget > 30:
            a_ms, obj_ms = multi_start_optimize(n_target, ms_budget, rng, verbose=True)
            if a_ms is not None and obj_ms < best_score:
                best_score = obj_ms
                best_seq = a_ms.copy()
                print(f"  Multi-start: obj={best_score:.8f} **")
    
    # === Phase 2: Intensive LP Polishing (45%) ===
    print(f"\n=== Phase 2: Intensive LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.45, 600.0)
    
    if p2_budget > 30:
        # Multiple rounds with perturbation between
        num_rounds = 5
        per_round = p2_budget / num_rounds
        for rnd in range(num_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(per_round, remaining * 0.4)
            if alloc < 20:
                break
            
            a_lp = lp_polish_v2(best_seq, time_budget=alloc, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd+1}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd+1}: obj={obj_lp:.8f}")
                # Perturb for next round
                if rnd < num_rounds - 1:
                    scale = 0.01 + 0.005 * (rnd + 1)
                    best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                    best_seq = np.maximum(best_seq, 0)
                    best_seq = best_seq / best_seq.sum()
                    # Also try pressure update
                    a_p = pressure_update(best_seq, eta=0.08, temperature=0.05)
                    if compute_obj(a_p) < compute_obj(best_seq):
                        best_seq = a_p
    
    # === Phase 3: Peak Pair Surgery (15%) ===
    print(f"\n=== Phase 3: Peak Surgery ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    
    if p3_budget > 15:
        a_surg = peak_pair_surgery(best_seq, time_budget=p3_budget, verbose=True)
        obj_surg = compute_obj(a_surg)
        if obj_surg < best_score:
            best_score = obj_surg
            best_seq = a_surg.copy()
            print(f"  Surgery: obj={best_score:.8f} **")
        else:
            print(f"  Surgery: obj={obj_surg:.8f}")
        
        # LP polish after surgery
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.3, 60.0)
            a_lp = lp_polish_v2(best_seq, time_budget=alloc)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Post-surgery LP: obj={best_score:.8f} **")
    
    # === Phase 4: Multi-length Exploration (10%) ===
    print(f"\n=== Phase 4: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    
    # Try a few different lengths
    n_options = []
    for delta in [-200, -100, 100, 200, 300]:
        n_try = n_current + delta
        if 400 <= n_try <= 2000 and abs(n_try - n_current) > 50:
            n_options.append(n_try)
    
    # Also try some fixed "round" lengths
    for n_fixed in [600, 700, 900, 1000, 1200]:
        if n_fixed not in n_options and abs(n_fixed - n_current) > 50:
            n_options.append(n_fixed)
    
    n_options = sorted(set(n_options))[:5]
    
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        
        a_try = resample_sequence(best_seq, n_try)
        
        # Quick optimization
        alloc = min(remaining * 0.3, 80.0)
        if alloc > 15:
            # Pressure update
            for step in range(10):
                eta = 0.1 / (1 + step * 0.3)
                a_try = pressure_update(a_try, eta=eta, temperature=0.08)
            # Short LP
            if alloc > 30:
                a_try = lp_polish_v2(a_try, time_budget=min(alloc * 0.7, 40))
        
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")
    
    # === Phase 5: Fine-tuning (remaining) ===
    print(f"\n=== Phase 5: Fine-tuning ===")
    remaining = deadline - time.monotonic()
    
    if remaining > 20:
        # Final LP
        alloc = min(remaining * 0.3, 80.0)
        if alloc > 15:
            a_lp = lp_polish_v2(best_seq, time_budget=alloc, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: obj={best_score:.8f} **")
        
        # Fine tune
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.5, 100.0)
            a_ft = fine_tune(best_seq, time_budget=alloc, verbose=True)
            obj_ft = compute_obj(a_ft)
            if obj_ft < best_score:
                best_score = obj_ft
                best_seq = a_ft.copy()
                print(f"  Fine-tune: obj={best_score:.8f} **")
        
        # One last LP if time permits
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.4, 50.0)
            a_lp = lp_polish_v2(best_seq, time_budget=alloc)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Last LP: obj={best_score:.8f} **")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
