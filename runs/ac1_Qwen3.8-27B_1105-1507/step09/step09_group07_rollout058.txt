    for i in range(n):

            M2[i, j] = p[i - j]
    second_part = M2 @ a
    
    grad = first_part + second_part
    return grad

import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def compute_grad_smoothed(a, beta):
    """Compute gradient of (1/beta) * log(sum(exp(beta * conv[a,a])))"""
    n = len(a)
    conv = compute_conv(a)
    shifted = beta * (conv - conv.max())
    p = np.exp(shifted)
    p_sum = p.sum()
    p = p / p_sum

    # = sum_k p[k] * dconv[k]/da[i]
    # where p[k] = exp(beta*conv[k]) / sum_j exp(beta*conv[j])

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 1.0 / n  # initial learning rate
    total_time = time_budget
    it = 0
    max_iters = int(total_time * 10)  # estimate
    
    no_improve = 0
    plateau_patience = 30
    
    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        grad, p = compute_grad_smoothed(a, beta)
        
        # Projected gradient step
        a_new = project_simplex(a - lr * grad)
        
        # Backtracking line search
        obj_new = compute_obj(a_new)
        step = 0
        while obj_new > best_obj and step < 20:
            lr = lr * 0.5
            a_new = project_simplex(a - lr * grad)
            obj_new = compute_obj(a_new)
            step += 1
        
        if step == 0 and obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
            # Increase lr slightly
            lr = min(lr * 1.05, 1.0)
        else:
            no_improve += 1
            if no_improve > plateau_patience:
                # Try a different beta or restart
                a = best_a * (1 + 0.02 * np.random.default_rng(it).normal(0, 1, n))
                a = project_simplex(a)
                best_obj = compute_obj(a)
                best_a = a.copy()
                lr = 1.0 / n
                no_improve = 0
        
        it += 1
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    SmoothGrad it{it}: obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")
    
    return best_a

- When k-j = i, i.e., j = k-i: the term is a[k-i] * a[i], derivative w.r.t. a[i] is a[k-i]
- If j = i AND k-j = i (i.e., i = k-i, i.e., k = 2i): we've double-counted, so the derivative is 2*a[i]... wait no.

Let me redo: conv[k] = sum_j a[j] * a[k-j]

The partial derivative w.r.t. a[i]:
= sum_j (da[j]/da[i]) * a[k-j] + a[j] * da[k-j]/da[i]
= a[k-i] + a[i-k]

Wait, that's summing over all j, but da[j]/da[i] = delta_{j,i} and da[k-j]/da[i] = delta_{k-j,i} = delta_{j,k-i}.

So: d conv[k] / d a[i] = (sum_j delta_{j,i} * a[k-j]) + (sum_j a[j] * delta_{j,k-i})
= a[k-i] + a[k-(k-i)]
= a[k-i] + a[i]

Wait, that's not right either. Let me be very explicit.

conv[k] = sum_{j=0}^{n-1} a[j] * a[k-j]

d conv[k] / d a[i] = sum_{j=0}^{n-1} [ (d a[j]/d a[i]) * a[k-j] + a[j] * (d a[k-j]/d a[i]) ]

= sum_{j=0}^{n-1} [ delta_{j,i} * a[k-j] + a[j] * delta_{k-j,i} ]

= a[k-i] + a[i-k]

Wait, the second sum: delta_{k-j,i} means k-j = i, i.e., j = k-i. So the second term is a[k-i] (the value of a[j] when j = k-i).

So d conv[k] / d a[i] = a[k-i] + a[k-i] = 2*a[k-i]?

No wait. The second sum is: sum_j a[j] * delta_{k-j,i} = a[j] evaluated at j = k-i = a[k-i].

So d conv[k] / d a[i] = a[k-i] + a[k-i] = 2*a[k-i]??

That can't be right. Let me check with the example:
a = [a0, a1, a2], conv[1] = a0*a1 + a1*a0 = 2*a0*a1

d conv[1] / d a0 = 2*a1. Using formula: a[k-i] + a[k-i] = a[1-0] + a[1-0] = a1 + a1 = 2*a1. ✓

conv[2] = a0*a2 + a1*a1 + a2*a0 = 2*a0*a2 + a1^2
d conv[2] / d a1 = 2*a1. Using formula: a[2-1] + a[2-1] = a1 + a1 = 2*a1. ✓
d conv[2] / d a0 = 2*a2. Using formula: a[2-0] + a[2-0] = a2 + a2 = 2*a2. ✓

OK so the formula is: d conv[k] / d a[i] = 2*a[k-i] (if 0 <= k-i < n), else 0?

Wait, but what about boundary effects? Let's check:
a = [a0, a1, a2], conv[4] = a2*a2 = a2^2
d conv[4] / d a2 = 2*a2. Formula: a[4-2] + a[4-2] = a[2] + a[2] = 2*a2. ✓ (4-2=2, which is in [0,3))

conv[0] = a0*a0 = a0^2
d conv[0] / d a0 = 2*a0. Formula: a[0-0] + a[0-0] = a[0] + a[0] = 2*a0. ✓

So it seems like d conv[k] / d a[i] = 2*a[k-i] if 0 <= k-i < n, else 0.

Wait, but what about when k-i is out of bounds? E.g., k=0, i=1: k-i = -1, out of bounds.
d conv[0] / d a1: conv[0] = a0^2, doesn't involve a1. So derivative = 0. ✓ (k-i = -1, out of bounds)

k=4, i=0: k-i = 4, out of bounds (n=3, so valid range is [0,3)).
d conv[4] / d a0: conv[4] = a2^2, doesn't involve a0. Derivative = 0. ✓

So the formula is simply: d conv[k] / d a[i] = 2*a[k-i] if 0 <= k-i < n, else 0.

This is MUCH simpler than I thought! The gradient is:
grad[i] = sum_k p[k] * 2*a[k-i] * [0 <= k-i < n]
= 2 * sum_{k=i}^{i+n-1} p[k] * a[k-i]
= 2 * sum_{j=0}^{n-1} p[i+j] * a[j]

So grad[i] = 2 * (cross-correlation of p and a)[i] where:
(cross_corr)[i] = sum_{j=0}^{n-1} p[i+j] * a[j]

This is just one cross-correlation, not two! I was overcomplicating it.

Let me verify with the example:
a = [a0, a1, a2], p = [p0, p1, p2, p3, p4]

grad[0] = 2 * (p0*a0 + p1*a1 + p2*a2)
grad[1] = 2 * (p1*a0 + p2*a1 + p3*a2)
grad[2] = 2 * (p2*a0 + p3*a1 + p4*a2)

Check: d log-sum-exp / d a0 = sum_k p[k] * 2*a[k-0] (if valid)
= 2*(p0*a0 + p1*a1 + p2*a2) ✓

Great! So the gradient computation is simply:
grad = 2 * cross_corr(p, a) (first n elements)

where cross_corr(p, a)[i] = sum_{j=0}^{n-1} p[i+j] * a[j]

Using FFT:
N >= n + (2n-1) - 1 = 3n-2
cross_full = ifft(fft(a, N) * conj(fft(p, N)))
grad = 2 * cross_full[:n]

This is very clean and efficient!

"""
Hybrid optimization: smooth gradient descent + LP polishing + coordinate descent.
Uses FFT-based gradient computation for the log-sum-exp smoothed max autocorrelation.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    s = float(np.sum(v))
    if s <= 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = rho_candidates[valid][-1] - 1
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def compute_grad_smoothed(a, beta):
    """Compute gradient of (1/beta)*log(sum(exp(beta*conv))) and the softmax weights."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = conv.max()
    shifted = beta * (conv - max_conv)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    
    # grad[i] = 2 * sum_j a[j] * p[i+j] (cross-correlation)
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)
    cross = np.fft.irfft(fa * np.conj(fp), N)
    grad = 2.0 * cross[:n]
    
    return grad, p, max_conv

def build_conv_matrix_normalized(a):
    """Build convolution matrix A such that A@g gives conv(a, g) for normalized a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_normalized(a_norm)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.PDLP]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None

def smooth_gradient_descent(a, time_budget, beta_start=15, beta_end=400, verbose=False):
    """Smooth gradient descent with adaptive beta and backtracking."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 2.0 / n
    no_improve = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / time_budget)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        grad, p, max_conv = compute_grad_smoothed(a, beta)
        
        # Take gradient step
        for attempt in range(30):
            a_new = project_simplex(a - lr * grad)
            obj_new = compute_obj(a_new)
            if obj_new <= best_obj:
                break
            lr *= 0.5
            if lr < 1e-10:
                a_new = a.copy()
                obj_new = best_obj
                break
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
            # Adaptive learning rate: increase if we took full step
            if attempt == 0:
                lr = min(lr * 1.1, 10.0 / n)
        else:
            no_improve += 1
            if no_improve > 50:
                # Perturb to escape
                rng = np.random.default_rng(hash((it, n, beta)))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = project_simplex(a)
                best_obj = compute_obj(a)
                best_a = a.copy()
                lr = 2.0 / n
                no_improve = 0
                if verbose:
                    print(f"    SGD perturbed at it{it}, obj={best_obj:.8f}")
        
        it += 1
        if verbose and it % 100 == 0:
            print(f"    SGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj

def lp_alternate(a, time_budget, max_iter=200, verbose=False):
    """LP alternating with smart step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n, np.inf
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break
        
        # Compute M_gg = max conv(g_lp, g_lp)
        conv_gg = compute_conv(g_lp)
        M_gg = float(np.max(conv_gg))
        M_cur = best_obj / (2.0 * n)  # current max conv (since obj = 2n * max_conv for normalized)
        
        # Smart step size: t = (M_cur - M_lp) / (M_cur - 2*M_lp + M_gg)
        denom = M_cur - 2 * M_lp + M_gg
        if denom > 1e-10:
            t_opt = (M_cur - M_lp) / denom
            t_opt = np.clip(t_opt, 0.01, 1.0)
        else:
            t_opt = 0.5
        
        # Try a few step sizes around t_opt
        candidate_ts = [t_opt]
        if t_opt > 0.1:
            candidate_ts.append(t_opt * 0.5)
        candidate_ts.append(min(1.0, t_opt * 1.5))
        candidate_ts.append(0.5)
        candidate_ts.append(1.0)
        candidate_ts = sorted(set(np.clip(candidate_ts, 0.01, 1.0)))
        
        best_t_obj = np.inf
        best_t_a = None
        for t in candidate_ts:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_a = a_try.copy()
        
        if best_t_a is not None and best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = best_t_a.copy()
        else:
            # No improvement, try smaller steps or break
            improved = False
            for t in [0.1, 0.05, 0.02]:
                a_try = (1 - t) * a + t * g_lp
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break
            if not improved:
                # Check if we've converged (a is close to g_lp)
                similarity = 1.0 - np.linalg.norm(a - g_lp) / (np.linalg.norm(a) + np.linalg.norm(g_lp) + 1e-15)
                if similarity > 0.999:
                    break
                # Perturb
                rng = np.random.default_rng(it * 2654435761)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = project_simplex(a)
                best_obj = compute_obj(a)
                best_a = a.copy()
        
        if verbose and (it + 1) % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP-alt it{it+1}: obj={best_obj:.8f}, M_lp={M_lp:.6f}, t={t_opt:.3f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
