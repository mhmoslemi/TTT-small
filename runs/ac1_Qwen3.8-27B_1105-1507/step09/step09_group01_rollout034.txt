"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS-B with log-sum-exp smoothing in softmax space for fast convergence,
LP-based active-set polishing with warm starts for precision, greedy peak-shaving
coordinate descent, and multi-n exploration. Key improvements: direct quasi-Newton
optimization, adaptive active-set LP, structured peak-shaving moves, and diverse
initializations with symmetry exploitation.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    return e / np.sum(e)

    x is the log-parameter, a = softmax(x). Returns (obj, grad_x)."""

    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    obj = n * (float(max_conv) + np.log(np.sum(exp_vals)) / tau)

    # Gradient of smoothed max w.r.t. a: 2 * cross_correlation(p, a)
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_a = 2.0 * ifft(P * np.conj(A)).real[:n]

    # Chain rule through softmax: grad_x[j] = a[j] * (g_a[j] - sum(a * g_a))
    mean_g = float(np.dot(a, g_a))
    grad_x = a * (g_a - mean_g)
    return obj, grad_x


def lse_obj_and_grad_simple(a, tau):
    """Compute log-sum-exp smoothed objective and gradient w.r.t. a directly.
    a should sum to 1 and be non-negative."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    obj = n * (float(max_conv) + np.log(np.sum(exp_vals)) / tau)

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_a = 2.0 * ifft(P * np.conj(A)).real[:n]
    return obj, g_a


def lbfgs_softmax(a_init, tau_schedule, max_iter_iter, time_budget, verbose=False):
    """Run L-BFGS-B with softmax parameterization and increasing tau schedule."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy()
    a = a / a.sum()
    x = np.log(a)  # initial log-parameter

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    total_iters = 0

    for tau in tau_schedule:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Allocate iterations proportional to remaining time
        est_time_per_iter = 0.005 * n / 800.0  # rough estimate
        max_iter = max(50, min(max_iter_iter, int(remaining / (est_time_per_iter * 2))))

        def func_and_grad(x):
            obj, grad = lse_obj_and_grad(x, tau)
            return obj, grad

        res = minimize(func_and_grad, x, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-14,
                                'maxfun': max_iter * 3})

        total_iters += res.nit
        a_new = softmax(res.x)
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s > 1e-10:
            a_new = a_new / s
        obj_new = compute_obj(a_new)

        if verbose:
            print(f"    L-BFGS tau={tau}: obj={obj_new:.8f}, iters={res.nit}, time={time.monotonic()-start:.1f}s")

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            x = res.x.copy()
        else:
            # Keep previous x for next tau
            x = np.log(np.maximum(a, 1e-15))

        a = best_a.copy()

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_end = min(n - 1, 2 * n - 2 - j)
        for k in range(j, j + n):
            A[k, j] = a[k - j] if k - j < n else 0.0
    return A


def build_conv_matrix_fast(a):
    """Build convolution matrix using stride tricks for speed."""
    n = len(a)
    rows, cols = [], []
    for j in range(n):
        a_slice = a[:n]
        for i in range(n):
            k = i + j
            if k < 2 * n - 1:
                rows.append(k)
                cols.append(j)
                # We'll fill the matrix
    # Use a more efficient approach
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # a[j] contributes to conv[k] for k = j, j+1, ..., j+n-1 with value a[k-j]
        # But conv[k] = sum_i a[i]*a[k-i], so a[j] * a[k-j] contributes to conv[k]
        # When we compute A @ g, (A@g)[k] = sum_j A[k,j]*g[j]
        # We want A[k,j]*g[j] = a[j]*g[j] * a[k-j] (when k-j is valid)
        # So A[k,j] = a[k-j] if 0 <= k-j < n, else 0
        max_k = min(2 * n - 2, j + n - 1)
        for k in range(j, max_k + 1):
            A[k, j] = a[k - j]
    return A


def build_conv_matrix_v2(a):
    """Efficient construction of convolution matrix using numpy."""
    n = len(a)
    # A[k, j] = a[k-j] if 0 <= k-j < n else 0
    # k ranges 0..2n-2, j ranges 0..n-1
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    mask = (diff >= 0) & (diff < n)
    A = np.where(mask, a[diff * mask].reshape(-1, 1) if False else 0, 0)
    # Simpler: just do it with broadcasting
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def solve_lp_minpeak(a, prev_g=None, n_constraints=None):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0.
    Optionally limit to top n_constraints most important constraints.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_v2(a_norm)

    # If we want to reduce constraints, keep only the top ones
    if prev_g is not None and n_constraints is not None and n_constraints < (2 * n - 1):
        conv_prev = compute_conv(a_norm) * np.ones(1)  # placeholder
        # Actually, use the previous g to identify active constraints
        agg = A @ prev_g
        max_agg = float(np.max(agg))
        # Keep constraints where agg[k] > 0.7 * max_agg
        threshold = 0.7 * max_agg
        active = np.where(agg > threshold)[0]
        if len(active) < 10:
            # Make sure we keep enough
            top_idx = np.argsort(agg)[::-1][:max(20, n_constraints)]
            active = top_idx
        A = A[active, :]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue

    return None, None


def lp_polish_active(a, time_budget=300.0, verbose=False, seed=42):
    """LP-based polishing with active set identification and adaptive steps."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    prev_g = None
    iter_count = 0
    no_improve = 0
    last_t = 0.3

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start

        # Determine constraint subset size (adaptive)
        frac_active = max(0.1, 0.5 - 0.01 * iter_count)
        n_cons = max(20, int(frac_active * (2 * n - 1)))

        g_lp, M_lp = solve_lp_minpeak(a, prev_g=prev_g, n_constraints=n_cons)
        if g_lp is None:
            break

        prev_g = g_lp.copy()
        iter_count += 1

        # Line search for optimal blend
        best_t = last_t
        best_blend_obj = compute_obj(best_a)

        for t in [last_t, last_t * 0.5, last_t * 1.5, 0.1, 0.3, 0.5, 0.7, 1.0]:
            t = min(1.0, max(0.01, t))
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_t = t

        a_new = (1.0 - best_t) * best_a + best_t * g_lp
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = best_a.copy()
            last_t = best_t
            no_improve = 0
            if verbose and iter_count % 10 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t:.3f}, M={M_lp:.6f}, el={elapsed:.0f}s")
        else:
            no_improve += 1
            if no_improve == 3:
                # Perturb
                scale = 0.01 * min(no_improve, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                prev_g = None
                no_improve = 0
                if iter_count > 100:
                    break
            elif no_improve >= 6:
                break

    return best_a


def greedy_peak_shave(a, time_budget=100.0, verbose=False):
    """Greedy peak-shaving: move mass from positions contributing to peak
    to positions that don't contribute to any peak."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)

    step_size = 0.05 / n
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]

        if peak_val < 1e-12:
            break

        # Find positions that contribute most to the peak
        # conv[peak_idx] = sum_j a[j] * a[peak_idx - j]
        contrib = np.zeros(n)
        for j in range(n):
            k_j = peak_idx - j
            if 0 <= k_j < n:
                contrib[j] = a[j] * a[k_j]

        # Also check second peak
        conv_sorted = np.sort(conv)[::-1]
        if len(conv_sorted) > 5:
            # Also consider positions contributing to top-5 peaks
            top_peaks = np.argsort(conv)[::-1][:5]
            for pk in top_peaks[1:]:
                pk = int(pk)
                for j in range(n):
                    k_j = pk - j
                    if 0 <= k_j < n:
                        contrib[j] += 0.5 * a[j] * a[k_j]

        # Move mass from high-contrib positions to low-contrib positions
        sorted_idx = np.argsort(contrib)[::-1]
        top_from = sorted_idx[:max(5, n // 20)]
        bottom_to = sorted_idx[-max(5, n // 20):]

        # Transfer a small amount
        amounts = {}
        from_mass = a[top_from].min() if len(top_from) > 0 else 0
        amount = min(step_size, from_mass * 0.1)

        if amount < 1e-15:
            break

        # Spread the transfer
        n_from = len(top_from)
        n_to = len(bottom_to)
        per_from = amount / n_from
        per_to = amount / n_to

        for j in top_from:
            a[j] = max(0, a[j] - per_from)
        for j in bottom_to:
            a[j] = a[j] + per_to

        # Re-normalize
        s = float(np.sum(a))
        if s < 1e-10:
            break
        a = a / s

        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
        else:
            a = best_a.copy()
            # Try a different random direction
            idx_from = rng.integers(0, n, size=10)
            idx_to = rng.integers(0, n, size=10)
            # Make sure from and to are different
            for i in range(10):
                if idx_to[i] in idx_from:
                    idx_to[i] = (idx_to[i] + 7) % n
            amt = 0.01 / n
            for j in idx_from:
                a[j] = max(0, a[j] - amt)
            for j in idx_to:
                a[j] += amt
            a = a / a.sum()
            obj_try = compute_obj(a)
            if obj_try < best_obj:
                best_a = a.copy()
                best_obj = obj_try

        it += 1
        if it % 1000 == 0 and verbose:
            print(f"    PeakShave it{it}: obj={best_obj:.8f}, el={time.monotonic()-start:.1f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Raised cosine
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 3:
        # Semicircle
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(0, 1 - x**2))
        a = a / a.sum()
    elif mode == 4:
        # Two gaussian bumps
        c1, c2 = n // 3, 2 * n // 3
        w = n // 8
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Flat middle, tapered ends
        a = np.ones(n)
        taper = n // 10
        for i in range(taper):
            a[i] = 0.5 + 0.5 * i / taper
            a[n - 1 - i] = 0.5 + 0.5 * i / taper
        a = a / a.sum()
    elif mode == 6:
        # sinc-like (main lobe of sinc)
        x = np.linspace(-2, 2, n)
        a = np.sinc(x / (2 * np.pi)) ** 2
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def nesterov_subgradient(a, time_budget=100.0, K=8, verbose=False):
    """Nesterov-accelerated subgradient method with top-K peaks."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 3.0 / np.sqrt(n)
    momentum = 0.8
    v = np.zeros(n)
    stall = 0
    it = 0
    y = a.copy()

    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break

        weights = np.exp((peak_vals - max_pv) * 10.0 / max_pv)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        y = a - lr * v
        y = project_simplex(y)

        obj_new = compute_obj(y)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = y.copy()
            a = y
            stall = 0
        else:
            a = project_simplex(a - lr * v)
            stall += 1
            if stall > 200:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break
            else:
                lr *= 0.998

        it += 1

    return best_a


def coordinate_descent_polish(a, time_budget=60.0, verbose=False):
    """Pairwise coordinate descent: transfer mass between pairs to reduce peak."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_k = int(np.argmax(conv))

        # Find best pair to transfer
        # For each j, the "cost" of having mass at j is a[peak_k - j] (if valid)
        cost = np.zeros(n)
        for j in range(n):
            k_j = peak_k - j
            if 0 <= k_j < n:
                cost[j] = a[k_j]

        # We want to move mass from high-cost to low-cost positions
        # But also to not increase other peaks
        # Simplified: move from top-cost to bottom-cost
        sorted_idx = np.argsort(cost)[::-1]
        top_idx = sorted_idx[:max(3, n // 50)]
        bot_idx = sorted_idx[-max(3, n // 50):]

        # Try several transfer amounts
        best_delta = 0
        best_trial_obj = best_obj

        for frac in [0.01, 0.02, 0.05, 0.1]:
            delta = frac * a[top_idx[0]]
            if delta < 1e-15:
                continue
            a_trial = a.copy()
            nf = len(top_idx)
            nt = len(bot_idx)
            for j in top_idx:
                a_trial[j] = max(0, a_trial[j] - delta / nf)
            for j in bot_idx:
                a_trial[j] += delta / nt
            s_t = float(np.sum(a_trial))
            if s_t < 1e-10:
                continue
            a_trial = a_trial / s_t
            obj_t = compute_obj(a_trial)
            if obj_t < best_trial_obj:
                best_trial_obj = obj_t
                best_delta = delta
                best_trial_a = a_trial.copy()

        if best_delta > 0:
            a = best_trial_a
            if best_trial_obj < best_obj:
                best_obj = best_trial_obj
                best_a = a.copy()
                no_improve = 0
            it += 1
        else:
            no_improve += 1
            if no_improve > 50:
                # Random perturbation
                a = best_a * np.exp(rng.normal(0, 0.005, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                if it > 2000:
                    break

        if verbose and it % 500 == 0:
            print(f"    CoordDes it{it}: obj={best_obj:.8f}, el={time.monotonic()-start:.1f}s")

    return best_a
