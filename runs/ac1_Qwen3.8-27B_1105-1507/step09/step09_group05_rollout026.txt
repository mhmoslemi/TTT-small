"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses sparse LP polishing with double-LP alternation, genetic algorithm outer loop,
and diverse perturbation strategies for global exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.sparse import csr_matrix
from scipy.sparse import diags
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_sparse_conv_matrix(a):
    """Build sparse matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_minpeak_sparse(a, prev_g=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_sparse_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    if prev_g is not None:
        g.value = prev_g
    
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish_with_double(a, max_iter=50, time_limit=60.0, verbose=False):
    """
    Double-LP polishing: alternate between solving for g given a and a' given g.
    This is faster convergence than single-LP alternation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf, 0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    no_improve = 0
    start = time.monotonic()
    it = 0
    prev_g = None

    while it < max_iter and (time.monotonic() - start) < time_limit:
        # Step 1: Solve LP for g given a
        g_lp, M_lp = solve_lp_minpeak_sparse(a, prev_g)
        if g_lp is None:
            break
        prev_g = g_lp

        # Step 2: Solve LP for a' given g_lp
        a_prime, M_prime = solve_lp_minpeak_sparse(g_lp, prev_g)
        if a_prime is None:
            a_prime = g_lp  # fallback

        # Step 3: Try blending a with both g_lp and a_prime
        candidates = []
        for t in [0.3, 0.5, 0.7, 1.0]:
            c1 = (1 - t) * a + t * g_lp
            c1 = np.maximum(c1, 0)
            if c1.sum() > 1e-10:
                candidates.append(c1 / c1.sum())
            c2 = (1 - t) * a + t * a_prime
            c2 = np.maximum(c2, 0)
            if c2.sum() > 1e-10:
                candidates.append(c2 / c2.sum())
            c3 = (1 - t) * a + 0.5 * t * g_lp + 0.5 * t * a_prime
            c3 = np.maximum(c3, 0)
            if c3.sum() > 1e-10:
                candidates.append(c3 / c3.sum())

        best_candidate = None
        best_candidate_obj = best_obj

        for c in candidates:
            obj_c = compute_obj(c)
            if obj_c < best_candidate_obj:
                best_candidate_obj = obj_c
                best_candidate = c.copy()

        it += 1
        if best_candidate is not None and best_candidate_obj < best_obj - 1e-13:
            best_obj = best_candidate_obj
            best_a = best_candidate.copy()
            a = best_a.copy()
            no_improve = 0
            if verbose and it % 5 == 0:
                print(f"      DLP it{it}: obj={best_obj:.8f}, M_lp={M_lp:.6f}, M_prime={M_prime:.6f}")
        else:
            no_improve += 1
            if no_improve >= 4:
                break

    return best_a, best_obj, it


def perturb_sequence(a, rng, method=0, scale=0.05):
    """Apply various perturbations to escape local minima."""
    n = len(a)
    a = a.copy()
    if method == 0:
        a = a * np.exp(rng.normal(0, scale, n))
    elif method == 1:
        noise = rng.uniform(1 - scale, 1 + scale, n)
        a = a * noise
    elif method == 2:
        mask = rng.random(n) > scale * 3
        a = a * mask
    elif method == 3:
        shift = int(rng.integers(1, max(2, n // 8)))
        a = np.roll(a, shift)
    elif method == 4:
        a = (1 - scale) * a + scale / n
    elif method == 5:
        A = fft(a, 2 * n)
        A[1:-1] *= np.exp(1j * rng.normal(0, scale * 2, len(A[1:-1])))
        a = ifft(A).real[:n]
        a = np.maximum(a, 0)
    elif method == 6:
        # Anti-peak perturbation: reduce elements that contribute to peak
        conv = compute_conv(a / max(a.sum(), 1e-10))
        peak_idx = np.argmax(conv)
        deltas = np.zeros(n)
        for j in range(n):
            idx = peak_idx - j
            if 0 <= idx < n:
                deltas[j] = a[idx]
        max_d = np.max(deltas)
        if max_d > 1e-10:
            a = a - scale * deltas / max_d
        a = np.maximum(a, 0)
    elif method == 7:
        # Swap two random blocks
        block_len = max(1, n // 20)
        i = int(rng.integers(0, n - block_len))
        j = int(rng.integers(0, n - block_len))
        if abs(i - j) > block_len:
            a[i:i+block_len], a[j:j+block_len] = a[j:j+block_len].copy(), a[i:i+block_len].copy()
    elif method == 8:
        # Frequency domain scaling
        A = fft(a, 2 * n)
        mask = rng.random(2 * n) > 0.5
        A[mask] *= (1 + rng.normal(0, scale, mask.sum()))
        a = ifft(A).real[:n]
        a = np.maximum(a, 0)
    else:
        a = a * np.exp(rng.normal(0, scale, n))

    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 6, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        # Chirp-like
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.sin(2 * np.pi * rng.uniform(2, 8) * x + rng.uniform(0, 6.28))
        a = np.maximum(a, 0.1)
        a = a / a.sum()
    elif mode == 8:
        # Exponential decay + noise
        x = np.arange(n)
        a = np.exp(-x / (n * rng.uniform(0.3, 1.0)))
        a += rng.uniform(0, 0.1, n)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)


def nesterov_smoothed_descent(a, time_budget=100.0, tau_init=10.0, tau_end=500.0):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))

        p = exp_vals / float(np.sum(exp_vals))

            N <<= 1
        P = fft(p, N)
        A_fft = fft(y, N)
        g_full = ifft(P * np.conj(A_fft)).real
        g = 2.0 * g_full[:n]

        if prev_g is not None and it % 3 == 0:

            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:

            v = np.zeros(n)

            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9997)

            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses sparse double-LP polishing with genetic algorithm outer loop,
adaptive perturbations, and multi-length exploration.
Key innovations: double-LP alternation (solve for both a and g), sparse matrices,
GA-based global exploration with LP elitism, and structured anti-peak perturbations.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_sparse_conv_matrix(a):
    """Build sparse matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    row_starts = np.maximum(0, np.arange(2 * n - 1) - n + 1)
    row_ends = np.minimum(n - 1, np.arange(2 * n - 1))
    row_lens = row_ends - row_starts + 1
    total_nz = int(np.sum(row_lens))
    rows_arr = np.zeros(total_nz, dtype=np.int32)
    cols_arr = np.zeros(total_nz, dtype=np.int32)
    vals_arr = np.zeros(total_nz, dtype=np.float64)
    pos = 0
    for k in range(2 * n - 1):
        start = int(row_starts[k])
        end = int(row_ends[k])
        length = end - start + 1
        rows_arr[pos:pos + length] = k
        cols_arr[pos:pos + length] = np.arange(start, end + 1)
        vals_arr[pos:pos + length] = a[k - np.arange(start, end + 1)]
        pos += length
    return csr_matrix((vals_arr, (rows_arr, cols_arr)), shape=(2 * n - 1, n))

    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_sparse_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    if prev_g is not None:
        g.value = prev_g

    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish_with_double(a, max_iter=40, time_limit=60.0, verbose=False):
    """
    Double-LP polishing: alternate between solving for g given a and a' given g.
    Provides faster convergence than single-LP alternation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf, 0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    no_improve = 0
    start = time.monotonic()
    it = 0
    prev_g = None

    while it < max_iter and (time.monotonic() - start) < time_limit:
        g_lp, M_lp = solve_lp_minpeak(a, prev_g)
        if g_lp is None:
            break
        prev_g = g_lp.copy()

        a_prime, M_prime = solve_lp_minpeak(g_lp, prev_g)
        if a_prime is None:
            a_prime = g_lp.copy()

        candidates = []
        for t in [0.3, 0.5, 0.7, 1.0]:
            c1 = (1.0 - t) * a + t * g_lp
            c1 = np.maximum(c1, 0)
            if c1.sum() > 1e-10:
                candidates.append(c1 / c1.sum())
            c2 = (1.0 - t) * a + t * a_prime
            c2 = np.maximum(c2, 0)
            if c2.sum() > 1e-10:
                candidates.append(c2 / c2.sum())
            c3 = (1.0 - t) * a + 0.5 * t * g_lp + 0.5 * t * a_prime
            c3 = np.maximum(c3, 0)
            if c3.sum() > 1e-10:
                candidates.append(c3 / c3.sum())

        best_candidate = None
        best_candidate_obj = best_obj

        for c in candidates:
            obj_c = compute_obj(c)
            if obj_c < best_candidate_obj:
                best_candidate_obj = obj_c
                best_candidate = c.copy()

        it += 1
        if best_candidate is not None and best_candidate_obj < best_obj - 1e-13:
            best_obj = best_candidate_obj
            best_a = best_candidate.copy()
            a = best_a.copy()
            no_improve = 0
            if verbose and it % 5 == 0:
                print(f"      DLP it{it}: obj={best_obj:.8f}, M_lp={M_lp:.6f}, M_p={M_prime:.6f}")
        else:
            no_improve += 1
            if no_improve >= 5:
                break

    return best_a, best_obj, it


def perturb_sequence(a, rng, method=0, scale=0.05):
    """Apply various perturbations to escape local minima."""
    n = len(a)
    a = a.copy()
    if method == 0:
        a = a * np.exp(rng.normal(0, scale, n))
    elif method == 1:
        noise = rng.uniform(1 - scale, 1 + scale, n)
        a = a * noise
    elif method == 2:
        mask = rng.random(n) > scale * 3
        a = a * mask
    elif method == 3:
        shift = int(rng.integers(1, max(2, n // 8)))
        a = np.roll(a, shift)
    elif method == 4:
        a = (1 - scale) * a + scale / n
    elif method == 5:
        A = fft(a, 2 * n)
        A[1:-1] *= np.exp(1j * rng.normal(0, scale * 2, len(A[1:-1])))
        a = ifft(A).real[:n]
        a = np.maximum(a, 0)
    elif method == 6:
        conv = compute_conv(a / max(a.sum(), 1e-10))
        peak_idx = int(np.argmax(conv))
        deltas = np.zeros(n)
        for j in range(n):
            idx = peak_idx - j
            if 0 <= idx < n:
                deltas[j] = a[idx]
        max_d = np.max(deltas)
        if max_d > 1e-10:
            a = a - scale * deltas / max_d
        a = np.maximum(a, 0)
    elif method == 7:
        block_len = max(1, n // 20)
        i = int(rng.integers(0, n - block_len))
        j = int(rng.integers(0, n - block_len))
        if abs(i - j) > block_len:
            a[i:i + block_len], a[j:j + block_len] = a[j:j + block_len].copy(), a[i:i + block_len].copy()
    elif method == 8:
        A = fft(a, 2 * n)
        mask = rng.random(2 * n) > 0.5
        A[mask] *= (1 + rng.normal(0, scale, int(mask.sum())))
        a = ifft(A).real[:n]
        a = np.maximum(a, 0)
    elif method == 9:
        # Multi-scale anti-peak: reduce contributions to top 3 peak lags
        conv = compute_conv(a / max(a.sum(), 1e-10))
        top_k = np.argsort(conv)[::-1][:3]
        deltas = np.zeros(n)
        for k in top_k:
            k = int(k)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    deltas[j] += a[idx]
        max_d = np.max(deltas)
        if max_d > 1e-10:
            a = a - 0.5 * scale * deltas / max_d
        a = np.maximum(a, 0)
    elif method == 10:
        # Random amplitude modulation
        periods = rng.uniform(0.5, 3.0, 3)
        x = np.arange(n) / n
        mod = np.ones(n)
        for p in periods:
            mod += 0.3 * scale * np.sin(2 * np.pi * p * x + rng.uniform(0, 6.28))
        a = a * np.maximum(mod, 0.1)
    else:
        a = a * np.exp(rng.normal(0, scale, n))

    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 6, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.sin(2 * np.pi * rng.uniform(2, 8) * x + rng.uniform(0, 6.28))
        a = np.maximum(a, 0.1)
        a = a / a.sum()
    elif mode == 8:
        x = np.arange(n)
        a = np.exp(-x / (n * rng.uniform(0.3, 1.0)))
        a += rng.uniform(0, 0.1, n)
        a = a / a.sum()
    elif mode == 9:
        # Two-hump with random positions
        c1 = int(rng.integers(0, n // 2))
        c2 = int(rng.integers(n // 2, n))
        w = max(1, n // 10)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)


def nesterov_smoothed_descent(a, time_budget=100.0, tau_init=10.0, tau_end=500.0):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget * 0.9, 1e-10))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_fft = fft(y, N)
        g_full = ifft(P * np.conj(A_fft)).real
        g = 2.0 * g_full[:n]

        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.3 * lr + 0.7 * min(max(bb_lr, 1e-8), 15.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9997)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Double-LP polishing + GA outer loop with diverse perturbations.
    Returns list of non-negative floats minimizing the evaluation objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Quick gradient descent warm-up ===
    print(f"\n=== Phase 1: Quick GD Warm-up ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.06, 60.0)
    if p1_budget > 15:
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=p1_budget,
                                          tau_init=5.0, tau_end=300.0)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  GD: obj={best_score:.8f} **")
        else:
            print(f"  GD: obj={obj_sgd:.8f}")

    # === Phase 2: Intensive Double-LP Polishing ===
    print(f"\n=== Phase 2: Intensive Double-LP Polishing ===")
    a_current = best_seq.copy()

    num_rnds = 10
    remaining = deadline - time.monotonic()
    per_rnd = min(remaining * 0.30 / num_rnds, 50.0)

    for rnd in range(num_rnds):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(per_rnd, remaining * 0.35)
        if alloc < 8:
            break

        if rnd > 0:
            method = int(rng.integers(0, 11))
            scale = 0.015 + 0.01 * (rnd % 5)
            a_current = perturb_sequence(a_current, rng, method=method, scale=scale)

        a_new, obj_new, iters = lp_polish_with_double(a_current, max_iter=35,
                                                       time_limit=alloc, verbose=(rnd == 0))
        if obj_new < best_score - 1e-13:
            best_score = obj_new
            best_seq = a_new.copy()
            a_current = a_new.copy()
            print(f"  DLP r{rnd}: obj={best_score:.8f} ({iters} it) **")
        else:
            print(f"  DLP r{rnd}: obj={obj_new:.8f} ({iters} it)")
            a_current = a_new.copy()

    # === Phase 3: GA with LP Elitism ===
    print(f"\n=== Phase 3: GA + LP Elitism ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Initialize population
    pop = [best_seq.copy()]
    pop_obj = [best_score]

    for r in range(1, 12):
        mode = (r * 3 + 1) % 10
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.15 + 0.08 * (r % 4)
        a_r = (1.0 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        if a_r.sum() > 1e-10:
            a_r = a_r / a_r.sum()
        obj_r = compute_obj(a_r)
        pop.append(a_r)
        pop_obj.append(obj_r)

    # GA loop
    ga_gen = 0
    ga_end = start_time + float(budget_s) * 0.68

    while time.monotonic() < ga_end:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        ga_gen += 1

        # Sort population
        indices = np.argsort(pop_obj)
        pop = [pop[i] for i in indices]
        pop_obj = [pop_obj[i] for i in indices]

        # LP polish top 2-3
        num_polish = min(3, len(pop))
        for idx in range(num_polish):
            remaining = deadline - time.monotonic()
            if remaining < 12:
                break
            alloc = min(20.0, remaining * 0.08)
            if alloc < 6:
                break
            a_pol, obj_pol, _ = lp_polish_with_double(pop[idx], max_iter=15,
                                                       time_limit=alloc)
            if obj_pol < pop_obj[idx]:
                pop[idx] = a_pol
                pop_obj[idx] = obj_pol
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()

        # Generate new candidates
        new_candidates = []

        # Mutations of best
        for m in range(3):
            a_mut = perturb_sequence(pop[0], rng, method=int(rng.integers(0, 11)),
                                     scale=0.025 + 0.01 * m)
            new_candidates.append(a_mut)

        # Crossovers
        for c in range(2):
            i1 = int(rng.integers(0, min(4, len(pop))))
            i2 = (i1 + int(rng.integers(1, 4))) % min(4, len(pop))
            mask = rng.random(n_best) > 0.5
            a_cross = np.where(mask, pop[i1], pop[i2])
            a_cross = np.maximum(a_cross, 0)
            if a_cross.sum() > 1e-10:
                a_cross = a_cross / a_cross.sum()
                new_candidates.append(a_cross)

        # Diverse restart blended with best
        a_div = diverse_init(n_best, rng, mode=int(rng.integers(0, 10)))
        blend = rng.uniform(0.25, 0.45)
        a_div = (1.0 - blend) * a_div + blend * pop[0]
        a_div = np.maximum(a_div, 0)
        if a_div.sum() > 1e-10:
            a_div = a_div / a_div.sum()
            new_candidates.append(a_div)

        # Evaluate and add
        for a_nc in new_candidates:
            obj_nc = compute_obj(a_nc)
            pop.append(a_nc)
            pop_obj.append(obj_nc)

        # Keep top 12
        indices = np.argsort(pop_obj)
        pop = [pop[i] for i in indices[:12]]
        pop_obj = [pop_obj[i] for i in indices[:12]]

        if ga_gen % 3 == 0:
            print(f"  GA g{ga_gen}: pop_best={pop_obj[0]:.8f}, global={best_score:.8f}")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = [n_current - 100, n_current + 100, n_current + 200, n_current + 400]
    n_options = [n for n in n_options if 500 <= n <= 1600 and n != n_current]

    for n_try in n_options[:3]:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.12, 40.0)
        if alloc > 8:
            a_try, obj_try, _ = lp_polish_with_double(a_try, max_iter=20, time_limit=alloc)
        else:
            obj_try = compute_obj(a_try)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Final Intensive Polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    for final_rnd in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 12:
            break
        alloc = min(remaining * 0.18, 45.0)
        if alloc < 8:
            break

        a_final = best_seq.copy()
        if final_rnd > 0:
            a_final = perturb_sequence(a_final, rng, method=final_rnd % 6, scale=0.012)

        a_pol, obj_pol, iters = lp_polish_with_double(a_final, max_iter=40,
                                                       time_limit=alloc,
                                                       verbose=(final_rnd == 0))
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  Final {final_rnd}: obj={best_score:.8f} ({iters} it) **")
        else:
            print(f"  Final {final_rnd}: obj={obj_pol:.8f} ({iters} it)")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
