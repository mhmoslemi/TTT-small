"""

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_objective(x, tau, n):
    """Smoothed objective: 2n * LSE(tau * conv(a,a)) / (sum a)^2 where a = softmax(x) normalized."""
    a = np.exp(x - np.max(x))
    s = float(np.sum(a))
    if s < 1e-30:
        return 1e10, np.zeros(n)
    a = a / s
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    lse = max_c + np.log(float(np.sum(exp_vals))) / tau
    obj = 2.0 * n * lse
    return obj, a


def smoothed_grad(x, tau, n):
    """Gradient of smoothed objective with respect to x (log-parameterization)."""
    a = np.exp(x - np.max(x))
    s = float(np.sum(a))
    if s < 1e-30:
        return np.zeros(n)
    a = a / s
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    p = exp_vals / float(np.sum(exp_vals))

    # Gradient of smoothed max w.r.t. a: g_a[i] = 2 * sum_k p[k] * a[k-i]
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g_a = 2.0 * g_full[:n]

    # Now convert to gradient w.r.t. x via chain rule
    # a = exp(x)/sum(exp(x)), so da/dx is the centered version
    # grad_x[i] = sum_j (grad_a[j] * da[j]/dx[i])
    # da[j]/dx[i] = a[j]*(delta[j,i] - a[i])
    # grad_x[i] = a[i]*(grad_a[i] - sum_j a[j]*grad_a[j])

    dot_a_g = float(np.dot(a, g_a))
    g_x = a * (g_a - dot_a_g)

    # Also need to account for the 1/(sum a)^2 factor and its derivative
    # Actually since we normalized a to sum=1, the objective is just 2n*LSE
    # and the gradient through the normalization is already handled by the softmax chain rule
    return g_x * 2.0 * n

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def entropic_updates(a, time_budget, K=20, verbose=False):
    """
    Entropic (multiplicative) weight updates for peak reduction.
    a_new[i] = a[i] * exp(-eta * g[i]) / Z
    """

    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]

        # Soft weights on top peaks
        w = np.exp((peak_vals - peak_vals[0]) * 5.0 / max(max_c, 1e-15))
        w = w / w.sum()

        # Subgradient
        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k = int(k)
            for i in range(n):
                if 0 <= k - i < n:
                    g[i] += wi * a[k - i]

        # Entropic update
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                eta *= 0.7
                if eta < 1e-8:
                    break
            a = best_a

    return best_a


def coordinate_descent(a, time_budget, verbose=False):
    """
    Coordinate descent: for each coordinate, find optimal value via golden section search.
    Maintains sum(a) = 1 by adjusting a partner coordinate.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99)

    # Order coordinates by index for systematic sweep
    order = np.arange(n)

    while time.monotonic() - start < time_budget:
        improved = False
        # Randomize order each pass
        rng.shuffle(order)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            # Choose partner j (different from i) to maintain sum=1
            j = (i + 1) % n
            if j == i:
                j = (i + 2) % n

            # Current values
            ai = a[i]
            aj = a[j]
            total = 1.0

            # Golden section search for new ai (aj = total - ai' - (1-ai-aj) = adjust)
            # Actually: fix sum of all other coords. Let c = 1 - ai - aj (sum of rest)
            # New: ai' + aj' = ai + aj =: s_ij
            # So ai' in [0, s_ij], aj' = s_ij - ai'

            s_ij = ai + aj
            lo, hi = 0.0, s_ij
            golden = (np.sqrt(5.0) - 1.0) / 2.0

            # Evaluate at current point
            obj_cur = best_obj

            for _ in range(20):
                x1 = hi - golden * (hi - lo)
                x2 = lo + golden * (hi - lo)

                a[i] = x1
                a[j] = s_ij - x1
                obj1 = compute_obj(a)

                a[i] = x2
                a[j] = s_ij - x2
                obj2 = compute_obj(a)

                if obj1 < obj2:
                    hi = x2
                else:
                    lo = x1

            a[i] = (lo + hi) / 2.0
            a[j] = s_ij - a[i]

            obj_new = compute_obj(a)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a.copy()
                improved = True

        if not improved and time.monotonic() - start < time_budget:
            # No improvement in full pass, break or perturb
            a = best_a * np.exp(rng.normal(0, 0.005, n))
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 + 0.5 * np.cos(np.pi * x)
        a = a / a.sum()
    elif mode == 3:
        # Two-hump
        centers = [n * 0.3, n * 0.7]
        widths = [n * 0.15, n * 0.15]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 4:
        # Sine-modulated
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.2 * np.sin(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: L-BFGS + LP polishing (ternary t) + entropic updates + multi-n + coord descent.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    # === Phase 1: L-BFGS Smoothed Optimization (250s) ===
    print(f"\n=== Phase 1: L-BFGS Smoothed (budget 250s) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(250.0, remaining * 0.3)
    if p1_budget > 30:
        a_lbfgs = lbfgs_smoothed(best_seq, time_budget=p1_budget,
                                  tau_init=15.0, tau_end=1500.0, verbose=True)
        obj_lbfgs = compute_obj(a_lbfgs)
        if obj_lbfgs < best_score:
            best_score = obj_lbfgs
            best_seq = a_lbfgs.copy()
            print(f"  Phase1 L-BFGS: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 L-BFGS: obj={obj_lbfgs:.8f} (no improvement)")

    # === Phase 2: LP Polishing with ternary t (550s) ===
    print(f"\n=== Phase 2: LP Polishing (budget 550s) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(550.0, remaining * 0.5)
    if p2_budget > 50:
        # Multiple rounds with perturbation
        num_rounds = 3
        per_round = p2_budget / num_rounds
        for rnd in range(num_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(per_round, remaining * 0.6)
            if alloc < 25:
                break
            a_lp = lp_polish_improved(best_seq, time_budget=alloc, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd+1}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd+1}: obj={obj_lp:.8f}")
                if rnd < num_rounds - 1:
                    # Structured perturbation
                    scale = 0.01 * (rnd + 1)
                    best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                    best_seq = np.maximum(best_seq, 1e-10)
                    best_seq = best_seq / best_seq.sum()

    # === Phase 3: Entropic Updates (150s) ===
    print(f"\n=== Phase 3: Entropic Updates (budget 150s) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(150.0, remaining * 0.2)
    if p3_budget > 20:
        a_ent = entropic_updates(best_seq, time_budget=p3_budget, K=30, verbose=True)
        obj_ent = compute_obj(a_ent)
        if obj_ent < best_score:
            best_score = obj_ent
            best_seq = a_ent.copy()
            print(f"  Phase3 Entropic: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 Entropic: obj={obj_ent:.8f}")

    # === Phase 4: Multi-n Exploration (300s) ===
    print(f"\n=== Phase 4: Multi-n Exploration (budget 300s) ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = []
    for delta in [-200, -100, 100, 200, 300, 400, 500]:
        n_try = n_current + delta
        if 400 <= n_try <= 1600 and n_try not in n_options:
            n_options.append(n_try)
    # Also try some absolute values
    for n_abs in [600, 700, 900, 1000, 1200, 1400]:
        if n_abs not in n_options and abs(n_abs - n_current) > 50:
            n_options.append(n_abs)
    n_options = sorted(n_options)

    # Limit to top 5 most promising
    n_options = n_options[:5]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick LP polish
        alloc = min(remaining * 0.3, 60.0)
        if alloc > 15:
            a_try = lp_polish_improved(a_try, time_budget=alloc)

        # Quick L-BFGS
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.2, 30.0)
        if alloc2 > 10:
            a_try = lbfgs_smoothed(a_try, time_budget=alloc2, tau_init=20.0, tau_end=500.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Coordinate Descent (150s) ===
    print(f"\n=== Phase 5: Coordinate Descent (budget 150s) ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(150.0, remaining * 0.3)
    if p5_budget > 20:
        a_cd = coordinate_descent(best_seq, time_budget=p5_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase5 CoordDesc: obj={best_score:.8f} **")
        else:
            print(f"  Phase5 CoordDesc: obj={obj_cd:.8f}")

    # === Phase 6: Final Intensive Polish (remaining) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP
    if remaining > 30:
        alloc = min(remaining * 0.35, 120.0)
        a_lp = lp_polish_improved(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_lp:.8f}")

    # L-BFGS with high tau
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 80.0)
        a_lbfgs = lbfgs_smoothed(best_seq, time_budget=alloc, tau_init=100.0, tau_end=5000.0)
        obj_lbfgs = compute_obj(a_lbfgs)
        if obj_lbfgs < best_score:
            best_score = obj_lbfgs
            best_seq = a_lbfgs.copy()
            print(f"  Final L-BFGS: obj={best_score:.8f} **")

    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 100.0)
        a_lp = lp_polish_improved(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final coord descent
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 60.0)
        a_cd = coordinate_descent(best_seq, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CoordDesc: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
