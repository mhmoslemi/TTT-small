import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def peak_directed_perturb(a, scale=0.05):
    """Perturb a by reducing elements that contribute most to peaks."""
    n = len(a)
    a_norm = a / a.sum()
    conv = compute_conv(a_norm)
    max_conv = np.max(conv)
    
    # Find top peaks
    peak_threshold = max_conv * 0.95
    peak_indices = np.where(conv >= peak_threshold)[0]
    
    # For each peak, compute contribution of each element
    delta = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                delta[j] += a_norm[idx]
    
    # Reduce elements with high contribution to peaks
    max_delta = np.max(np.abs(delta))
    if max_delta > 1e-10:
        a_new = a - scale * delta / max_delta
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s > 1e-10:
            return a_new / s
    return a_norm

            no_improve = 0
            if verbose and it % 5 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={t_best:.2f}, M={M_lp:.6f}")
        else:
            no_improve += 1
            if no_improve >= 4:
                break

    return best_a, best_obj, it


def anneal_perturb(a, rng, temperature):
    """Multiplicative noise perturbation with given temperature."""
    n = len(a)
    noise = rng.normal(0, temperature, n)
    a_new = a * np.exp(noise)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    return a_new / s


def peak_directed_perturb(a, rng, scale=0.04):
    """Perturb by targeting elements that contribute most to the top peaks."""
    n = len(a)
    a_norm = a / a.sum()
    conv = compute_conv(a_norm)

    top_k = np.argsort(conv)[::-1][:4]

    delta = np.zeros(n)
    for k_idx in top_k:
        k_idx = int(k_idx)
        for j in range(n):
            idx = k_idx - j
            if 0 <= idx < n:
                delta[j] += a_norm[idx]

    max_d = np.max(delta)
    if max_d > 1e-10:
        a_new = a - scale * delta / max_d
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s > 1e-10:
            return a_new / s
    return a_norm


def spread_perturb(a, rng, scale=0.03):
    """Perturb by spreading mass from peaks to valleys of the sequence."""
    n = len(a)
    a_norm = a / a.sum()
    mean_val = 1.0 / n
    # Push values toward the mean
    a_new = (1.0 - scale) * a_norm + scale * mean_val
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a_norm
    return a_new / s


def freq_perturb(a, rng, scale=0.03):
    """Perturb in frequency domain."""
    n = len(a)
    A = fft(a, 2 * n)
    phases = np.exp(1j * rng.normal(0, scale, len(A)))
    A *= phases
    a_new = ifft(A).real[:n]
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a / a.sum()
    return a_new / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def symmetric_fold(a):
    """Enforce symmetry a[j] = a[n-1-j]."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    s = np.sum(a_sym)
    if s > 1e-10:
        return a_sym / s
    return a.copy()


def diverse_init(n, rng, mode):
    """Generate diverse initializations based on mode."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (x - 0.3) ** 2) + np.exp(-2.0 * (x - 0.7) ** 2)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 0.01))
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) + 0.2 * np.sin(7 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        a = rng.dirichlet(np.ones(n) * 0.5)
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(2 * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)


def nesterov_descent(a, time_budget=100.0, tau_init=10.0, tau_end=400.0):
    """Nesterov smoothed gradient descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    lr = 2.5 / np.sqrt(n)
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.95))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_fft = fft(y, N)
        g_full = ifft(P * np.conj(A_fft)).real
        g = 2.0 * g_full[:n]

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9998)

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP-based alternating optimization with adaptive perturbations,
    multi-n exploration, diverse restarts, and final intensive polishing.
    Returns list of non-negative floats minimizing the evaluation objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # --- Load parent ---
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Initial LP polish (20% time) ===
    print("\n=== Phase 1: Initial LP Polish ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.18, 200.0)
    if p1_budget > 30:
        a1, obj1, it1 = lp_polish(best_seq, max_iter=300, time_limit=p1_budget, verbose=True)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1: obj={best_score:.8f} ({it1} iters) **")
        else:
            print(f"  Phase1: obj={obj1:.8f} ({it1} iters)")

    # === Phase 2: Smoothed gradient descent warm start (10% time) ===
    print("\n=== Phase 2: Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.08, 80.0)
    if p2_budget > 15:
        a2 = nesterov_descent(best_seq, time_budget=p2_budget, tau_init=5.0, tau_end=400.0)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 GD: obj={obj2:.8f}")
        # Quick LP after GD
        remaining = deadline - time.monotonic()
        if remaining > 40:
            a2b, obj2b, _ = lp_polish(best_seq, max_iter=50, time_limit=min(remaining * 0.15, 60.0))
            if obj2b < best_score:
                best_score = obj2b
                best_seq = a2b.copy()
                print(f"  Phase2 GD+LP: obj={best_score:.8f} **")

    # === Phase 3: Main search loop with adaptive perturbation (35% time) ===
    print("\n=== Phase 3: Adaptive Search ===")
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.35
    p3_end = time.monotonic() + p3_budget

    temperature = 0.015
    a_current = best_seq.copy()
    it = 0
    stall_count = 0

    while time.monotonic() < p3_end:
        it += 1
        if it % 20 == 0:
            print(f"  SA it{it}: best={best_score:.8f}, T={temperature:.4f}")

        # Choose perturbation method based on stall count
        if stall_count > 5:
            # More aggressive escape
            method = it % 4
            if method == 0:
                a_current = peak_directed_perturb(a_current, rng, scale=0.05)
            elif method == 1:
                a_current = anneal_perturb(a_current, rng, temperature * 2)
            elif method == 2:
                a_current = freq_perturb(a_current, rng, scale=0.05)
            else:
                a_current = spread_perturb(a_current, rng, scale=0.06)
            stall_count = 0
        else:
            # Gentle perturbation
            a_current = anneal_perturb(a_current, rng, temperature)

        # LP polish (short burst)
        alloc = min(25.0, (p3_end - time.monotonic()) * 0.15)
        if alloc < 10:
            break
        a_new, obj_new, _ = lp_polish(a_current, max_iter=25, time_limit=alloc)

        if obj_new < best_score - 1e-12:
            best_score = obj_new
            best_seq = a_new.copy()
            a_current = a_new.copy()
            temperature = max(0.003, temperature * 0.85)
            stall_count = 0
        else:
            temperature = min(0.12, temperature * 1.08)
            a_current = a_new.copy()
            stall_count += 1

    # === Phase 4: Multi-n exploration (20% time) ===
    print("\n=== Phase 4: n-Sweep ===")
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.22
    p4_end = time.monotonic() + p4_budget

    n_best = len(best_seq)
    n_options = sorted(set([
        max(200, n_best - 150), max(200, n_best - 75),
        n_best + 50, n_best + 100, n_best + 150, n_best + 300, n_best + 500,
        500, 600, 700, 800, 900, 1000, 1100, 1200, 1300, 1500, 1800
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000 and n != n_best]
    n_options.sort(key=lambda x: abs(x - n_best))

    for n_try in n_options[:10]:
        if time.monotonic() > p4_end:
            break
        alloc = min(50.0, (p4_end - time.monotonic()) * 0.2)
        if alloc < 15:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick GD then LP
        half1 = alloc * 0.3
        half2 = alloc * 0.7
        if half1 > 8:
            a_try = nesterov_descent(a_try, time_budget=half1, tau_init=10.0, tau_end=300.0)
        a_try, obj_try, _ = lp_polish(a_try, max_iter=40, time_limit=half2)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts (10% time) ===
    print("\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p5_budget = remaining * 0.12
    p5_end = time.monotonic() + p5_budget
    n_best = len(best_seq)

    for r in range(6):
        if time.monotonic() > p5_end:
            break
        alloc = min(40.0, (p5_end - time.monotonic()) * 0.25)
        if alloc < 12:
            break

        mode = r % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.1 + 0.06 * r
        a_r = (1.0 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        # GD + LP
        gd_alloc = alloc * 0.4
        lp_alloc = alloc * 0.6
        if gd_alloc > 8:
            a_r = nesterov_descent(a_r, time_budget=gd_alloc, tau_init=10.0, tau_end=250.0)
        a_r, obj_r, _ = lp_polish(a_r, max_iter=30, time_limit=lp_alloc)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Symmetry attempt (5% time) ===
    print("\n=== Phase 6: Symmetry ===")
    if time.monotonic() < deadline - 40:
        a_sym = symmetric_fold(best_seq)
        alloc = min(40.0, (deadline - time.monotonic()) * 0.15)
        if alloc > 12:
            a_sym_new, obj_sym, _ = lp_polish(a_sym, max_iter=50, time_limit=alloc)
            if obj_sym < best_score:
                best_score = obj_sym
                best_seq = a_sym_new.copy()
                print(f"  Symmetric: obj={best_score:.8f} **")
            else:
                print(f"  Symmetric: obj={obj_sym:.8f}")

    # === Phase 7: Final intensive polish (remaining time) ===
    print("\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Multiple rounds of LP with micro-perturbations
    for rnd in range(5):
        if time.monotonic() > deadline - 10:
            break
        alloc = min(remaining * 0.25, 100.0)
        if alloc < 15:
            break

        a_current = best_seq.copy()
        if rnd > 0:
            a_current = anneal_perturb(a_current, rng, 0.008)

        a_new, obj_new, iters = lp_polish(a_current, max_iter=200, time_limit=alloc,
                                          verbose=(rnd == 0))
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Final {rnd}: obj={best_score:.8f} ({iters} iters) **")
        else:
            print(f"  Final {rnd}: obj={obj_new:.8f} ({iters} iters)")

    # Last attempt: high-tau GD on the best
    if time.monotonic() < deadline - 20:
        alloc = min(30.0, deadline - time.monotonic() - 15)
        if alloc > 10:
            a_final = nesterov_descent(best_seq, time_budget=alloc, tau_init=50.0, tau_end=800.0)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final GD: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
