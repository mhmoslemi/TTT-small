def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:

    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):

def simplex_project(v):

    n = len(v)

    cssv = np.cumsum(u)

    theta = (cssv[rho] - 1) / (rho + 1.0)

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    out = irfft(fa * fa, size)
    return out[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * np.max(conv)

def build_conv_matrix(a):
    n = len(a)
    idx = np.arange(2*n-1)[:, None] - np.arange(n)[None, :]
    mask = (idx >= 0) & (idx < n)
    A = np.where(mask, a[idx], 0.0)
    return A

def solve_lp(a):
    n = len(a)
    a_n = a / np.sum(a)
    A = build_conv_matrix(a_n)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = np.sum(g)
        if s > 1e-10:
            return g / s, float(res.x[n])
    return None, None

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def line_search(a, g, f0, n_evals=15):
    """Find best t in [0,1] minimizing obj((1-t)*a + t*g)."""
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = compute_obj((1-c)*a + c*g)
    fd = compute_obj((1-d)*a + d*g)
    for _ in range(n_evals - 2):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = compute_obj((1-c)*a + c*g)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = compute_obj((1-d)*a + d*g)
    t_star = (lo + hi) / 2
    f_star = compute_obj((1-t_star)*a + t_star*g)
    return t_star, f_star

def lp_alternate(a, time_budget, verbose=False):

        return a

        if g is None:

        iteration += 1
        if f_star < best_obj - 1e-12:
            best_obj = f_star
            best_a = (1 - t_star) * a + t_star * g
            best_a = np.maximum(best_a, 0)
            best_a = best_a / best_a.sum()
            a = best_a.copy()
            stall = 0
            
            if verbose and iteration % 10 == 0:
                elapsed = time.monotonic() - start
                print(f"  LP-alt it{iteration}: obj={best_obj:.8f}, t*={t_star:.3f}, M_lp={M_lp:.6f}, time={elapsed:.1f}s")
        else:

            if stall > 20:
                # Perturb and restart
                rng = np.random.default_rng(iteration * 1000 + n)
                noise = rng.normal(0, 0.05, n)
                a = a * np.exp(noise)
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                stall = 0
            
            # Try t=1 directly
            a_try = g.copy()
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                stall = 0
    
    return best_a

For the log-sum-exp gradient:
f_beta(a) = (1/beta) * log(sum_k exp(beta * conv_k))

grad[i] = sum_k p_k * a[k-i] (where p_k = exp(beta*conv_k) / sum_j exp(beta*conv_j))

I can compute this as:
- p = softmax(beta * conv) (length 2n-1)
- For each i: grad[i] = sum_{j=0}^{n-1} a[j] * p[i+j] (where p[i+j] = 0 if i+j >= 2n-1)

This is a "truncated" cross-correlation. I can compute it efficiently.

Let me use the following approach:
- Pad a to length 2n-1: a_ext = [a, zeros(n-1)]
- The cross-correlation R[i] = sum_j a_ext[j] * p[i+j] for valid indices
- This is R = ifft(conj(fft(a_ext)) * fft(p)) ... but I need to be careful.

Actually, the correlation (a ⋆ b)[n] = sum_k a[k] * b[k+n].
In FFT: FFT(a ⋆ b) = conj(FFT(a)) * FFT(b)

So: a ⋆ p = ifft(conj(fft(a_ext, N)) * fft(p, N))

where N is the FFT size (>= 2*(2n-1)-1 = 4n-3 for linear correlation).

Let me just use the direct method for simplicity:

"""
Optimization for minimizing peak autocorrelation of non-negative step functions.
Uses improved LP alternating with line search, gradient-based optimization,
n-sweep, and adaptive perturbation restarts.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A such that (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    idx = np.arange(2*n-1)[:, None] - np.arange(n)[None, :]
    mask = (idx >= 0) & (idx < n)
    A = np.where(mask, a[idx], 0.0)
    return A

def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / np.sum(a)
    A = build_conv_matrix(a_n)
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, float(res.x[n])
    except Exception:
        pass
    return None, None

def simplex_project(v):
    """Project v onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = int(rho[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def line_search_obj(a, g, n_evals=20):
    """Golden section search for best t in [0,1] minimizing obj((1-t)*a + t*g)."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = compute_obj((1.0 - c) * a + c * g)
    fd = compute_obj((1.0 - d) * a + d * g)
    for _ in range(n_evals - 2):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = compute_obj((1.0 - c) * a + c * g)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = compute_obj((1.0 - d) * a + d * g)
    t_star = (lo + hi) / 2.0
    f_star = compute_obj((1.0 - t_star) * a + t_star * g)
    return t_star, f_star

def lp_alternate(a, time_budget, verbose=False, max_stall=30):

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    start = time.monotonic()

    it = 0

        if remaining < 5.0:
            break

        g, M_lp = solve_lp(a)
        lp_time = time.monotonic() - t0
        if g is None:

        it += 1
        # Line search
        t_star, f_star = line_search_obj(a, g, n_evals=15)
        
        if f_star < best_obj - 1e-13:
            best_obj = f_star
            new_a = (1.0 - t_star) * a + t_star * g
            new_a = np.maximum(new_a, 0)
            ns = float(np.sum(new_a))
            if ns > 1e-10:
                new_a = new_a / ns
            else:
                new_a = best_a.copy()
            best_a = new_a.copy()
            a = new_a.copy()
            stall = 0
        else:
            # Try t=1
            obj_1 = compute_obj(g)
            if obj_1 < best_obj - 1e-13:
                best_obj = obj_1
                best_a = g.copy()
                a = g.copy()
                stall = 0

                    scale = 0.03 + 0.02 * min(stall // max_stall, 5)

                    a = a * np.exp(noise)
                    a = np.maximum(a, 0)

                    else:

                        if obj_gp < best_obj:

def gradient_optimize(a, time_budget, beta_start=10.0, beta_end=500.0, verbose=False):
    """Projected gradient descent on log-sum-exp surrogate."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 0.05
    it = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break
        
        frac = min(1.0, it / 200.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Compute gradient
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        p = np.exp(beta * (conv - max_c))
        p /= p.sum()
        
        # grad[i] = sum_j a[j] * p[i+j]
        grad = np.zeros(n)
        for i in range(n):
            j_max = min(n - 1, 2 * n - 2 - i)
            if j_max >= 0:
                grad[i] = np.dot(a[:j_max+1], p[i:i+j_max+1])
        
        # Projected gradient step
        a_new = simplex_project(a - lr * grad)
        
        # Backtracking line search
        obj_new = compute_obj(a_new)
        step = lr
        while obj_new > best_obj + 1e-10 and step > 1e-6:
            step *= 0.5
            a_new = simplex_project(a - step * grad)
            obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
