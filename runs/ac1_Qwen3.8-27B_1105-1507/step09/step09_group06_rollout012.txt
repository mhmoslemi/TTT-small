import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time
import math

def compute_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2**int(math.ceil(math.log2(2*n-1)))
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Matrix A such that A @ g = convolve(a, g)."""
    n = len(a)
    m = 2*n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j+r, j] = a[:r]
    return A

def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A_mat = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_g = float(np.sum(g_val))
                if s_g > 1e-10:
                    return g_val / s_g, float(M.value)
        except Exception:
            continue
    return None, None

def line_search_blend(a, g, n_steps=20):
    """Find best t in [0,1] to blend a and g."""
    best_t = 0.0
    best_obj = compute_obj(a)

        obj = compute_obj(a_try)

def smoothed_gradient(a, tau):
    """Smoothed gradient of max(conv) via FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    p = exp_vals / float(np.sum(exp_vals))
    
    N = 2**int(math.ceil(math.log2(3*n)))
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]

def gd_step(a, tau, lr, momentum=0.9):
    """One GD step with momentum on simplex."""
    n = len(a)
    g = smoothed_gradient(a, tau)
    # Project gradient onto tangent space of simplex
    g_proj = g - np.mean(g)
    a_new = a - lr * g_proj
    a_new = np.maximum(a_new, 0)
    a_new = a_new / float(np.sum(a_new))
    return a_new

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses aggressive LP-based iterative refinement (solving min M s.t. conv(a,g)<=M, sum(g)=1, g>=0)
as the primary optimizer, combined with Nesterov accelerated gradient descent for smoothing,
adaptive line search along LP directions, constraint-aware perturbation for escape,
and multi-length exploration. The LP provides the exact optimal linearized direction
to reduce the convolution peak, making it the workhorse optimizer.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time
import math
import sys


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build dense matrix A (2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A_mat = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start
            M.value = float(np.max(compute_conv(a_norm) / float(np.sum(a_norm))))
        except Exception:
            pass

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0.0)
                s_g = float(np.sum(g_val))
                if s_g > 1e-10:
                    return g_val / s_g, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g, n_coarse=10, n_fine=10):
    """Two-stage line search: coarse then fine. Returns (best_t, best_obj)."""
    base_obj = compute_obj(a)
    best_t = 0.0
    best_obj = base_obj

    # Coarse
    for i in range(1, n_coarse + 1):
        t = i / n_coarse
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0.0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_t = t

    # Fine around best
    if best_t > 0:
        for i in range(1, n_fine):
            t = best_t * (1.0 - 0.5 / n_coarse) + (i / n_fine) * (best_t * 1.0 / n_coarse)
            t = max(0.001, min(1.0, t))
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_t = t

    return best_t, best_obj


def lp_iteration_loop(a, time_budget, seed=0, verbose=False, max_stall=60):
    """
    Intensive LP iteration: repeatedly solve LP, blend, re-solve.
    Uses adaptive perturbation to escape local minima.
    """
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    it = 0
    warm = None

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 2.0:
            break

        g_lp, M_lp = solve_lp(a, warm_start=warm)
        if g_lp is None:
            stall += 1
            if stall > 5:
                break
            continue

        warm = g_lp

        best_t, best_new_obj = line_search_blend(a, g_lp)

        if best_new_obj < best_obj - 1e-13 and best_t > 0:
            a = (1.0 - best_t) * a + best_t * g_lp
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            best_obj = best_new_obj
            best_a = a.copy()
            stall = 0
            if verbose and it % 20 == 0:
                print(f"    LP it{it}: obj={best_obj:.9f}, t={best_t:.3f}, M_lp={M_lp:.7f}", flush=True)
        else:
            stall += 1
            # Adaptive perturbation
            scale = 0.005 + 0.003 * min(stall, 20)
            conv = compute_conv(a)
            # Weight perturbation by peak contribution
            max_c = float(np.max(conv))
            peak_near = conv > max_c * 0.95
            # Simple noise

            a_pert = a * np.exp(scale * noise)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
            # Blend with best to not lose progress
            mix = 0.5 + 0.02 * min(stall, 10)
            a = mix * best_a + (1.0 - mix) * a_pert
            a = np.maximum(a, 0.0)
            a = a / a.sum()

            if stall >= max_stall:
                if verbose:
                    print(f"    LP stalled at it{it}, obj={best_obj:.9f}", flush=True)
                break

        it += 1

    return best_a, best_obj


def smoothed_gradient_fft(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    total = float(np.sum(exp_vals))
    if total < 1e-300:
        return np.zeros(n)
    p = exp_vals / total

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def nesterov_gd(a, time_budget, tau_init=10.0, tau_end=500.0, seed=0, verbose=False):
    """Nesterov accelerated gradient descent on simplex with cosine tau annealing."""
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    beta = 0.85
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init + (tau_end - tau_init) * (1.0 - np.cos(np.pi * progress)) / 2.0

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0.0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient_fft(y, tau)
        g_proj = g - np.mean(g)

        v = beta * v - lr * g_proj
        a_new = y + v
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9997)

        # Stuck reset
        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n)
            best_it = it

    return best_a, best_obj


def subgradient_descent(a, time_budget, seed=0, verbose=False):
    """
    Exact subgradient method: identify active constraints (conv entries at max),
    compute subgradient from those, project onto simplex tangent.
    """
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 2.0 / n
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))

        # Active set: entries within 1% of max
        active = np.where(conv >= max_c * 0.99)[0]

        # Subgradient: average of convolution vectors at active indices
        # The gradient of conv[k] w.r.t. a[i] is a[k-i] if valid
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]
        grad /= len(active)

        # Project onto tangent space
        grad_proj = grad - np.mean(grad)

        # Update
        a_new = a - lr * grad_proj
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()

        a = a_new
        lr *= 0.9995
        lr = max(lr, 1e-10)
        it += 1

        if it % 5000 == 0 and verbose:
            print(f"    SubGrad it{it}: obj={best_obj:.9f}, lr={lr:.2e}, n_active={len(active)}", flush=True)

    return best_a, best_obj


def cosine_taper(n):
    """Generate normalized cosine taper."""
    x = np.arange(n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode):
    """Generate diverse initial sequences."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        return cosine_taper(n)
    elif mode == 2:
        # Gaussian bump
        x = np.arange(n)
        a = np.exp(-((x - n / 2) ** 2) / (2 * (n / 4) ** 2))
        return a / a.sum()
    elif mode == 3:
        # Random
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 4:
        # Arc-like
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x ** 2, 0.001))
        return a / a.sum()
    elif mode == 5:
        # Two bumps
        x = np.arange(n)
        a = np.exp(-((x - n * 0.25) ** 2) / (2 * (n / 12) ** 2))
        a += np.exp(-((x - n * 0.75) ** 2) / (2 * (n / 12) ** 2))
        return a / a.sum()
    elif mode == 6:
        # Power taper
        x = np.linspace(0, 1, n)
        a = (1 - np.abs(2 * x - 1)) ** rng.uniform(0.5, 3.0)
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    elif mode == 7:
        # Smoothed random
        a = rng.uniform(0, 1, n)
        k = max(2, n // 40)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        return a / a.sum()
    else:
        # Trapezoidal
        x = np.arange(n)
        a = np.minimum(x, n - 1 - x).astype(float)
        a = np.maximum(a, 0.1)
        return a / a.sum()


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def peak_weighted_perturb(a, scale=0.03, seed=0):
    """Perturb with weight based on peak contribution to convolution."""
    n = len(a)
    rng = np.random.default_rng(seed)
    conv = compute_conv(a)
    max_c = float(np.max(conv))

    # Compute contribution of each position to the peaks
    contrib = np.zeros(n)
    top_indices = np.argsort(conv)[::-1][:20]
    for k in top_indices:
        k = int(k)
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            j = k - i
            if 0 <= j < n:
                contrib[i] += a[i] * a[j]

    max_contrib = contrib.max() if contrib.max() > 0 else 1.0
    contrib = contrib / max_contrib

    noise = rng.normal(0, 1, n) * scale * (0.3 + contrib)
    a_new = a * np.exp(noise)
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new

    print(f"\n=== Phase 8: Final Polish ===", flush=True)
    remaining = deadline - time.monotonic()

    # LP
    if remaining > 30:
        alloc = min(100.0, remaining * 0.4)
        a8, obj8 = lp_iteration_loop(best_seq, alloc, seed=seed + 5000, verbose=True, max_stall=40)
        if obj8 < best_score:
            best_score = obj8
            best_seq = a8
            print(f"  Final LP: obj={best_score:.9f} **", flush=True)

    # High-tau GD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(60.0, remaining * 0.4)
        a9, obj9 = nesterov_gd(best_seq, alloc, tau_init=100.0, tau_end=3000.0, seed=seed + 6000)
        if obj9 < best_score:
            best_score = obj9
            best_seq = a9
            print(f"  Final GD: obj={best_score:.9f} **", flush=True)

    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(40.0, remaining * 0.5)
        a10, obj10 = lp_iteration_loop(best_seq, alloc, seed=seed + 7000, max_stall=20)
        if obj10 < best_score:
            best_score = obj10
            best_seq = a10
            print(f"  Final LP2: obj={best_score:.9f} **", flush=True)

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = 2.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val != np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.9f}, time={elapsed:.1f}s ===", flush=True)
    return result
