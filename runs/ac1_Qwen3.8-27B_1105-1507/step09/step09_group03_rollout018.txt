```python
    """Project v onto {x >= 0, sum(x) = 1}."""
```

```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations vs previous approach:
1. Gerchberg-Saxton spectral equalization (frequency domain peak reduction)
2. FFT-based smoothed gradient with adaptive backtracking step size
3. Block coordinate moves (shift contiguous mass blocks for structural changes)
4. Aggressive multi-size exploration with resampling
5. Dual LP with active-set refinement
6. Simulated annealing with structural restarts
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_conv_normalized(a):
    """Compute conv of normalized a (sum=1)."""
    s = float(np.sum(a))
    if s < 1e-10:
        return np.zeros(2 * len(a) - 1)
    return compute_conv(a / s)


def smoothed_grad_conv(a, tau):
    """
    Compute gradient of smoothed max of conv(a,a) using FFT.
    Returns (grad, conv) where grad[i] = d/d(a[i]) log-sum-exp(tau * conv).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    p = exp_vals / float(np.sum(exp_vals))

    # grad[i] = 2 * sum_k p[k] * a[k-i] (cross-correlation of p and a)
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real[:n]
    grad = 2.0 * cross
    return grad, conv


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = int(rho[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def gs_spectral_equalize(a, n_iters=200, tau=1.0, verbose=False):
    """
    Gerchberg-Saxton style spectral equalization.
    Alternates between:
    - Frequency domain: flatten |FFT(a)|^2 (soft flattening with parameter tau)
    - Time domain: enforce non-negativity and normalize
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    N = 1
    while N < 2 * n:
        N <<= 1

    for it in range(n_iters):
        # Frequency domain step: soften the spectrum
        A = fft(a, N)
        mag = np.abs(A)
        # Target: flatten the power spectrum
        # Use soft flattening: new_mag = mag / (1 + tau * (mag^2 - mean_mag2)/mean_mag2)^alpha
        power = mag ** 2
        mean_power = float(np.mean(power))
        if mean_power < 1e-20:
            break
        # Softly push toward flat spectrum
        ratio = power / mean_power
        # Where power is high, reduce; where low, increase (softly)
        new_power = mean_power * np.power(ratio, 1.0 / (1.0 + tau))
        new_mag = np.sqrt(new_power)
        # Preserve phase
        phase = A / (mag + 1e-30)
        A_new = phase * new_mag
        a_new = ifft(A_new).real[:n]

        # Time domain step: enforce non-negativity
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

        if verbose and it % 50 == 0:
            print(f"    GS it{it}: obj={best_obj:.8f}, max_spec={np.max(new_power)/mean_power:.4f}")

    return best_a


def fft_gradient_descent(a, time_budget=100.0, tau_init=20.0, tau_end=500.0,
                         lr_init=None, verbose=False):
    """
    Smoothed gradient descent with adaptive backtracking step size.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    if lr_init is None:
        lr = 5.0 / np.sqrt(n)
    else:
        lr = lr_init

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    best_it = 0
    momentum = 0.7
    v = np.zeros(n)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)

        # Compute smoothed gradient
        grad, conv = smoothed_grad_conv(a, tau)

        # Momentum update
        v = momentum * v - lr * grad
        a_try = a + v
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            v = np.zeros(n)
            a_try = a.copy()
        else:
            a_try = a_try / s_try

        obj_try = compute_obj(a_try)

        # Backtracking if objective worsened
        if obj_try > best_obj + 1e-12:
            # Try with smaller step
            found = False
            for scale in [0.5, 0.25, 0.1, 0.05]:
                a_try2 = a + scale * v
                a_try2 = np.maximum(a_try2, 0)
                s_try2 = float(np.sum(a_try2))
                if s_try2 > 1e-10:
                    a_try2 = a_try2 / s_try2
                    obj_try2 = compute_obj(a_try2)
                    if obj_try2 <= best_obj + 1e-12:
                        a_try = a_try2
                        obj_try = obj_try2
                        v = scale * v
                        found = True
                        break
            if not found:
                lr *= 0.7
                v = np.zeros(n)
                a_try = a.copy()
                obj_try = best_obj

        a = a_try
        obj_cur = compute_obj(a)
        if obj_cur < best_obj - 1e-14:
            best_obj = obj_cur
            best_a = a.copy()
            best_it = it

        # Slowly increase learning rate early, decrease late
        if progress < 0.3:
            lr = min(lr * 1.002, 10.0 / np.sqrt(n))
        else:
            lr = max(lr * 0.999, 0.1 / np.sqrt(n))

        # Gradually reduce momentum
        momentum = 0.7 * (1 - progress * 0.5)

        it += 1
        if verbose and it % 200 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.4f}")

        # Restart if stuck
        if it - best_it > max(5000, 2 * n):
            a = best_a * np.exp(np.random.default_rng(it).normal(0, 0.02, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = 5.0 / np.sqrt(n)
            best_it = it

    return best_a


def solve_lp_direction(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.MOSEK, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget=100.0, verbose=False):
    """LP-based polishing with active-set refinement."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    it = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break

        # Find active constraints
        conv_ag = build_conv_matrix(a / np.sum(a)) @ g_lp
        max_val = float(np.max(conv_ag))
        tol = 0.05 * max(max_val, 1e-10)
        active = np.where(conv_ag >= max_val - tol)[0]

        # Solve refined LP with only active constraints (minimize sum at active)
        A = build_conv_matrix(a / np.sum(a))
        g2 = cp.Variable(n, nonneg=True)
        obj2 = cp.sum(A[active, :] @ g2)
        cons2 = [cp.sum(g2) == 1]
        prob2 = cp.Problem(cp.Minimize(obj2), cons2)

        g2_val = None
        for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob2.solve(solver=solver)
                if g2.value is not None and prob2.status in ('optimal', 'optimal_inaccurate'):
                    g2_val = np.maximum(g2.value, 0)
                    s2 = float(np.sum(g2_val))
                    if s2 > 1e-10:
                        g2_val = g2_val / s2
                    break
            except Exception:
                continue

        # Choose direction
        if g2_val is not None:
            peak_g2 = float(np.max(build_conv_matrix(a / np.sum(a)) @ g2_val))
            peak_g1 = max_val
            if peak_g2 < peak_g1:
                direction = g2_val
            else:
                direction = (g_lp + g2_val) / 2
        else:
            direction = g_lp

        # Try several step sizes
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
            a_new = (1 - t) * a + t * direction
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                break

        it += 1
        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, M={M_lp:.6f}, |S|={len(active)}")

        if not improved:
            # Perturb
            scale = 0.02 + 0.01 * min(it, 10)
            a = best_a * np.exp(rng.normal(0, min(scale, 0.1), n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            if it > 40:
                break

    return best_a


def block_move_search(a, time_budget=100.0, verbose=False):
    """
    Search for good block moves: shift mass from one contiguous region to another.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(55555)
    it = 0
    stall = 0

    # Block sizes to try
    block_sizes = [max(1, n // 20), max(1, n // 10), max(1, n // 5), max(1, n // 4)]

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_k = int(np.argmax(conv))

        # For the peak, identify which source blocks would help most
        # Moving mass from position i reduces contributions where a[k-i] is large
        # We want to move FROM positions where a[peak_k - i] is large TO positions where a[peak_k - j] is small

        # Compute "blame" for each position
        blame = np.zeros(n)
        for i in range(n):
            mirror = peak_k - i
            if 0 <= mirror < n:
                blame[i] = a[i] * a[mirror]
            else:
                blame[i] = 0

        # Top blame positions (sources), bottom blame positions (targets)
        top_sources = np.argsort(-blame)[:n // 4]
        bottom_targets = np.argsort(blame)[:n // 4]

        improved = False

        # Try block moves from top blame regions to bottom blame regions
        for bs in block_sizes[:2]:
            # Find best source block (contiguous region with high total blame)
            best_src_start = -1
            best_src_blame = -1
            for start in range(0, n - bs + 1, max(1, bs // 3)):
                block_blame = np.sum(blame[start:start + bs])
                if block_blame > best_src_blame:
                    best_src_blame = block_blame
                    best_src_start = start

            # Find best target block (contiguous region with low total blame)
            best_tgt_start = -1
            best_tgt_blame = np.inf
            for start in range(0, n - bs + 1, max(1, bs // 3)):
                # Avoid overlapping with source
                if start + bs <= best_src_start or start >= best_src_start + bs:
                    block_blame = np.sum(blame[start:start + bs])
                    if block_blame < best_tgt_blame:
                        best_tgt_blame = block_blame
                        best_tgt_start = start

            if best_src_start < 0 or best_tgt_start < 0:
                continue

            src = slice(best_src_start, best_src_start + bs)
            tgt = slice(best_tgt_start, best_tgt_start + bs)

            # Try different transfer fractions
            total_src = float(np.sum(a[src]))
            if total_src < 1e-10:
                continue

            for frac in [0.3, 0.2, 0.1]:
                transfer = frac * total_src / bs
                a_try = a.copy()
                a_try[src] -= transfer
                a_try[tgt] += transfer
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        # Also try random block moves
        if not improved and rng.random() < 0.5:
            bs = block_sizes[rng.integers(0, len(block_sizes))]
            src_start = int(rng.integers(0, max(1, n - bs)))
            tgt_start = int(rng.integers(0, max(1, n - bs)))
            if abs(src_start - tgt_start) > bs:
                for frac in [0.2, 0.1]:
                    a_try = a.copy()
                    transfer = frac * float(np.sum(a[src_start:src_start + bs])) / bs
                    a_try[src_start:src_start + bs] -= transfer
                    a_try[tgt_start:tgt_start + bs] += transfer
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try > 1e-10:
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break

        if verbose and it % 20 == 0:
            print(f"    BM it{it}: obj={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        # Comb-like: alternating high/low
        a = np.ones(n)
        a[0::2] = 2.0
        a[1::2] = 0.5
        a = a / a.sum()
    elif mode == 2:
        # Multi-frequency comb
        x = np.arange(n) / n
        a = 1.0 + 0.5 * np.cos(2 * np.pi * 3 * x) + 0.3 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.1)
        a = a / a.sum()
    elif mode == 3:
        # Random
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 4:
        # Bimodal
        centers = [n // 3, 2 * n // 3]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Spectral: try to make FFT flat
        target_power = np.ones(n)
        # Random phase
        phases = rng.uniform(0, 2 * np.pi, n)
        # Only use real frequencies for real signal
        A = np.zeros(2 * n, dtype=complex)
        for k in range(n):
            A[k] = np.exp(1j * phases[k])
            if k > 0 and k < n:
                A[2 * n - k] = np.exp(-1j * phases[k])
        a = ifft(A).real[:n]
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Sine-based with random frequency
        freq = rng.uniform(2, 10)
        x = np.arange(n) / n
        a = 1.0 + 0.4 * np.sin(2 * np.pi * freq * x)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def anneal_structural(a, time_budget=60.0, verbose=False):
    """
    Simulated annealing with structural moves (permutations of blocks, amplitude scaling).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(77777)
    T = 0.1
    T_min = 0.001
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        T = max(T_min, 0.1 * np.exp(-elapsed / (time_budget * 0.3)))

        move_type = rng.integers(0, 4)

        if move_type == 0:
            # Swap two random blocks
            bs = int(rng.integers(max(2, n // 30), max(3, n // 8)))
            i = int(rng.integers(0, n - bs))
            j = int(rng.integers(0, n - bs))
            if abs(i - j) < bs:
                continue
            a_try = a.copy()
            a_try[i:i + bs], a_try[j:j + bs] = a_try[j:j + bs].copy(), a_try[i:i + bs].copy()
            a_try = a_try / a_try.sum()

        elif move_type == 1:
            # Scale a random block
            bs = int(rng.integers(max(2, n // 30), max(3, n // 5)))
            i = int(rng.integers(0, n - bs))
            factor = rng.uniform(0.5, 2.0)
            a_try = a.copy()
            a_try[i:i + bs] *= factor
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try

        elif move_type == 2:
            # Smooth a random region (local averaging)
            bs = int(rng.integers(max(2, n // 20), max(3, n // 5)))
            i = int(rng.integers(0, n - bs))
            a_try = a.copy()
            region = a_try[i:i + bs].copy()
            kernel = np.ones(3) / 3
            smoothed = np.convolve(region, kernel, mode='same')
            a_try[i:i + bs] = smoothed
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try

        else:
            # Gaussian perturbation on random subset
            subset_size = int(rng.integers(n // 10, n // 3))
            idx = rng.choice(n, size=subset_size, replace=False)
            a_try = a.copy()
            a_try[idx] *= np.exp(rng.normal(0, 0.1, subset_size))
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try

        obj_try = compute_obj(a_try)
        delta = obj_try - best_obj

        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_try
            if obj_try < best_obj:
                best_a = a_try.copy()
                best_obj = obj_try

        it += 1
        if verbose and it % 100 == 0:
            print(f"    Anneal it{it}: obj={best_obj:.8f}, T={T:.4f}")

    return best_a


def golden_section_search(a, direction, n_evals=8):
    """Golden section search for optimal blend parameter t in [0,1]."""
    phi = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0

    for _ in range(n_evals):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)

        a1 = np.maximum((1 - t1) * a + t1 * direction, 0)
        s1 = float(np.sum(a1))
        f1 = compute_obj(a1 / s1) if s1 > 1e-10 else np.inf

        a2 = np.maximum((1 - t2) * a + t2 * direction, 0)
        s2 = float(np.sum(a2))
        f2 = compute_obj(a2 / s2) if s2 > 1e-10 else np.inf

        if f1 < f2:
            hi = t2
        else:
            lo = t1

    t_best = (lo + hi) / 2
    a_best = np.maximum((1 - t_best) * a + t_best * direction, 0)
    s_best = float(np.sum(a_best))
    if s_best > 1e-10:
        a_best = a_best / s_best
    return a_best


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization combining spectral equalization, FFT gradient descent,
    LP polishing, block moves, and structural annealing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent: {e}")

    if best_seq is None:
        best_seq = diverse_init(n_target, rng, mode=1)
        best_score = compute_obj(best_seq)
        print(f"Default comb init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Spectral Equalization (Gerchberg-Saxton) ===
    print(f"\n=== Phase 1: Spectral Equalization ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.12, 120.0)
    if p1_budget > 20:
        a_gs = gs_spectral_equalize(best_seq, n_iters=500, tau=2.0, verbose=True)
        obj_gs = compute_obj(a_gs)
        if obj_gs < best_score:
            best_score = obj_gs
            best_seq = a_gs.copy()
            print(f"  GS: obj={best_score:.8f} ** IMPROVED")
        else:
            print(f"  GS: obj={obj_gs:.8f} (no improvement)")

    # === Phase 2: FFT Gradient Descent ===
    print(f"\n=== Phase 2: FFT Gradient Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 200.0)
    if p2_budget > 30:
        a_gd = fft_gradient_descent(best_seq, time_budget=p2_budget,
                                    tau_init=20.0, tau_end=400.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  GD: obj={best_score:.8f} ** IMPROVED")
        else:
            print(f"  GD: obj={obj_gd:.8f}")

    # === Phase 3: LP Polishing ===
    print(f"\n=== Phase 3: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 180.0)
    if p3_budget > 20:
        a_lp = lp_polish(best_seq, time_budget=p3_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} ** IMPROVED")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 4: Block Move Search ===
    print(f"\n=== Phase 4: Block Moves ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 120.0)
    if p4_budget > 15:
        a_bm = block_move_search(best_seq, time_budget=p4_budget, verbose=True)
        obj_bm = compute_obj(a_bm)
        if obj_bm < best_score:
            best_score = obj_bm
            best_seq = a_bm.copy()
            print(f"  BM: obj={best_score:.8f} ** IMPROVED")
        else:
            print(f"  BM: obj={obj_bm:.8f}")

    # === Phase 5: Multi-Size Exploration ===
    print(f"\n=== Phase 5: Multi-Size Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Try a range of sizes
    size_candidates = []
    for delta in [-200, -100, -50, 50, 100, 200, 300]:
        n_try = n_current + delta
        if 300 <= n_try <= 2000:
            size_candidates.append(n_try)
    # Add some fixed sizes
    for n_fixed in [400, 500, 600, 700, 900, 1000, 1200, 1500]:
        if n_fixed not in size_candidates and abs(n_fixed - n_current) > 30:
            size_candidates.append(n_fixed)

    size_candidates = sorted(set(size_candidates))
    # Limit to top candidates
    size_candidates = size_candidates[:8]

    for n_try in size_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick optimization on new size
        alloc_gs = min(remaining * 0.15, 30.0)
        alloc_gd = min(remaining * 0.2, 50.0)

        if alloc_gs > 10:
            a_try = gs_spectral_equalize(a_try, n_iters=int(alloc_gs * 5), tau=3.0)
        remaining = deadline - time.monotonic()
        if remaining > 40 and alloc_gd > 15:
            a_try = fft_gradient_descent(a_try, time_budget=alloc_gd,
                                         tau_init=30.0, tau_end=300.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} ** IMPROVED")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 70:
            break

        mode = (r * 2 + 1) % 7
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with current best
        blend = 0.4
        a_r = np.maximum((1 - blend) * a_r + blend * best_seq, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.18, 80.0)
        if alloc > 20:
            half = alloc / 2
            a_r = gs_spectral_equalize(a_r, n_iters=int(half * 5), tau=2.5)
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_r = fft_gradient_descent(a_r, time_budget=min(remaining * 0.4, 40.0),
                                           tau_init=30.0, tau_end=300.0)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} ** IMPROVED")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Structural Annealing ===
    print(f"\n=== Phase 7: Structural Annealing ===")
    remaining = deadline - time.monotonic()
    p7_budget = min(remaining * 0.15, 80.0)
    if p7_budget > 15:
        a_an = anneal_structural(best_seq, time_budget=p7_budget, verbose=True)
        obj_an = compute_obj(a_an)
        if obj_an < best_score:
            best_score = obj_an
            best_seq = a_an.copy()
            print(f"  Anneal: obj={best_score:.8f} ** IMPROVED")
        else:
            print(f"  Anneal: obj={obj_an:.8f}")

    # === Phase 8: Final Intensive Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    if remaining > 40:
        alloc = min(remaining * 0.35, 100.0)
        a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} ** IMPROVED")

    # GD polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 60.0)
        a_gd = fft_gradient_descent(best_seq, time_budget=alloc,
                                    tau_init=50.0, tau_end=600.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Final GD: obj={best_score:.8f} ** IMPROVED")

    # Spectral equalize
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = int(min(remaining * 0.25, 50.0) * 8)
        a_gs = gs_spectral_equalize(best_seq, n_iters=max(alloc, 50), tau=4.0)
        obj_gs = compute_obj(a_gs)
        if obj_gs < best_score:
            best_score = obj_gs
            best_seq = a_gs.copy()
            print(f"  Final GS: obj={best_score:.8f} ** IMPROVED")

    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 40.0)
        a_lp = lp_polish(best_seq, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()

    # Final validation
    if best_seq is None:

        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
