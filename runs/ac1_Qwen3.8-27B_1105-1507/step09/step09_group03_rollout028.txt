"""
Optimizes step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Methods: Weighted LP, spectral flattening, pair-swap local search, multi-n search.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build A (2n-1 x n) where A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def solve_weighted_lp(a, weights, time_limit=5.0):
    """
    Solve LP: min sum_k weights[k] * conv(a,g)[k]
    s.t. sum(g) = 1, g >= 0.
    Returns optimal g.
    """
    n = len(a)
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    obj = cp.sum(weights @ (A @ g))
    constraints = [cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(obj), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None

def solve_minimax_lp(a, time_limit=5.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def spectral_flatten(a, n_iters=100, gamma=0.5):
    """
    DFT-based spectral flattening.
    Iteratively adjusts a to flatten its power spectrum.
    """
    n = len(a)
    a = a.copy()
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    for it in range(n_iters):
        # Compute current autocorrelation
        fa = fft(a, size)
        power = np.abs(fa) ** 2
        
        # Target: flat spectrum (but keep DC component)
        # Smoothed inverse
        power_clipped = np.maximum(power, 1e-10)
        target_power = 1.0 / power_clipped ** gamma
        
        # Preserve phase, adjust amplitude
        phase = fa / np.abs(fa + 1e-15)
        new_fa = np.sqrt(target_power) * phase
        
        # Back to time domain
        a_new = np.real(ifft(new_fa, size))[:n]
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s > 1e-10:
            a_new = a_new / s
        else:
            break
        a = a_new
    
    return a

def pair_swap_search(a, time_budget=60.0):
    """
    Local search via pair swaps: move mass from one position to another
    to reduce the peak autocorrelation.

def robust_lp_polish(a, time_budget=300.0, verbose=False):
    """
    Robust LP polishing with adaptive strategies:
    - Minimix LP for direction
    - Weighted LP for refinement
    - Golden section line search
    - Perturbation to escape local minima
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a

    best_a = a.copy()

        weights = weights / weights.sum()
        
        g_w = solve_weighted_lp(a, weights)
        
        # Choose direction
        if g_w is not None:
            # Compare: which gives lower peak when blended?
            a_blend1 = project_simplex(a + 0.3 * (g_lp - a))
            a_blend2 = project_simplex(a + 0.3 * (g_w - a))
            obj1 = compute_obj(a_blend1)
            obj2 = compute_obj(a_blend2)
            if obj2 < obj1:
                direction = g_w
            else:
                direction = g_lp
        else:
            direction = g_lp
        
        # Line search for best step size
        best_t = 0.0
        best_f = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * direction
            a_try = np.maximum(a_try, 0)

                continue

            f_try = compute_obj(a_try)
            if f_try < best_f:
                best_f = f_try
                best_t = t

            a_new = (1 - best_t) * a + best_t * direction
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()

            best_a = a_new.copy()
            a = a_new

    return best_a

def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # U-shaped (mass at edges)
        x = np.linspace(-1, 1, n)
        a = 1.0 - 0.5 * x ** 2
        a = a / a.sum()
    elif mode == 4:
        # Sine-based
        freq = rng.uniform(1, 3)
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    else:
        # Half-sine
        x = np.linspace(0, np.pi, n)
        a = np.sin(x)
        a = a / a.sum()
    return a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) Adaptive two-stage LP with dual-guided active set, (2) DFT-based spectral
flattening with temperature annealing, (3) Peak-targeted pair-swap local search,
(4) Multi-n exploration, (5) Diversified restarts with blending.
Key novel elements: dual-weighted LP direction, annealed spectral IAA, and
peak-decomposition guided local search.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_minimax_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0. Returns (g, M, dual_weights)."""
    n = len(a)
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    con = A @ g <= M
    constraints = [con, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Get dual values for the inequality constraint
                    dual = None
                    try:
                        dual = con.dual_value
                        if dual is not None:
                            dual = np.maximum(dual, 0)
                    except Exception:
                        dual = None
                    return g_val, float(M.value), dual
        except Exception:
            continue
    return None, None, None


def solve_weighted_lp(a, weights):
    """Solve LP: min sum_k weights[k]*conv(a,g)[k] s.t. sum(g)=1, g>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    # Compute weighted sum efficiently
    wA = weights @ A  # shape (n,)
    obj = cp.sum(wA * g)
    constraints = [cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(obj), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def solve_topk_lp(a, k_frac=0.3):
    """Solve LP minimizing sum of top-k conv values."""
    n = len(a)
    A = build_conv_matrix(a)
    conv_a = A @ a  # current conv(a,a)
    
    # Use top k fraction of positions by current value
    k = max(1, int(k_frac * (2 * n - 1)))
    top_idx = np.argsort(conv_a)[::-1][:k]
    
    g = cp.Variable(n, nonneg=True)
    top_A = A[top_idx, :]
    obj = cp.sum(top_A @ g)
    constraints = [cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(obj), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def golden_section_search(a, direction, n_evals=10):
    """Golden section search for optimal blend t in [0,1]."""
    phi = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    
    for _ in range(n_evals):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)
        
        a1 = (1 - t1) * a + t1 * direction
        a1 = np.maximum(a1, 0)
        s1 = float(np.sum(a1))
        f1 = compute_obj(a1 / s1) if s1 > 1e-10 else np.inf
        
        a2 = (1 - t2) * a + t2 * direction
        a2 = np.maximum(a2, 0)
        s2 = float(np.sum(a2))
        f2 = compute_obj(a2 / s2) if s2 > 1e-10 else np.inf
        
        if f1 < f2:
            hi = t2
        else:
            lo = t1
    
    t = (lo + hi) / 2
    a_best = (1 - t) * a + t * direction
    a_best = np.maximum(a_best, 0)
    s = float(np.sum(a_best))
    if s > 1e-10:
        a_best = a_best / s
    return a_best


def adaptive_lp_polish(a, time_budget=300.0, verbose=False):
    """
    Adaptive LP polishing with multiple direction strategies:
    - Minimimax LP
    - Dual-weighted LP  
    - Top-k LP
    Combined with golden section line search and adaptive perturbation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(31415)

    it = 0
    stall = 0
    strategy_cycle = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        # Choose strategy based on cycle
        strategy = it % 3  # 0: minimax, 1: weighted, 2: topk
        
        direction = None
        
        if strategy == 0:
            g_lp, M_lp, dual = solve_minimax_lp(a)
            if g_lp is not None:
                direction = g_lp
        elif strategy == 1:
            # Dual-weighted: weight by exponential of current conv values
            conv = compute_conv(a)
            max_c = float(np.max(conv))
            if max_c > 1e-15:
                tau = 8.0
                weights = np.exp(tau * (conv - max_c))
                weights = weights / (weights.sum() + 1e-15)
                direction = solve_weighted_lp(a, weights)
        else:
            direction = solve_topk_lp(a, k_frac=0.25)
        
        if direction is None:
            # Fallback to minimax
            direction, _, _ = solve_minimax_lp(a)
            if direction is None:
                break

        # Line search
        a_new = golden_section_search(a, direction, n_evals=8)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            # Adaptive perturbation
            noise_scale = 0.02 + 0.005 * min(stall, 20)
            a = best_a * np.exp(rng.normal(0, noise_scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            
            if stall > 60:
                break

        it += 1
        if verbose and it % 5 == 0:
            print(f"    ALP it{it}: obj={best_obj:.8f}, strat={strategy}, stall={stall}")

    return best_a


def spectral_iaa(a, time_budget=100.0, verbose=False):
    """
    Iterative Amplitude Adjustment (IAA) for spectral flattening.
    Alternates between time-domain projection and frequency-domain shaping.
    Uses temperature annealing for the flattening exponent.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    it = 0
    stall = 0
    rng = np.random.default_rng(27182)

    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        # Anneal gamma: start mild, become aggressive
        gamma = 0.3 + 0.7 * progress ** 2

        fa = fft(a, size)
        power = np.abs(fa) ** 2
        power_clipped = np.maximum(power, 1e-10)

        # Flattening: target power proportional to 1/power^gamma
        target_power = 1.0 / (power_clipped ** gamma)

        # Preserve phase
        magnitude = np.abs(fa)
        phase = np.where(magnitude > 1e-15, fa / magnitude, 1.0 + 0j)
        
        # Blended amplitude update
        new_magnitude = np.sqrt(target_power)
        # Normalize to keep energy reasonable
        new_magnitude = new_magnitude / (np.max(new_magnitude) + 1e-15) * (np.max(magnitude) + 1e-15)
        
        new_fa = new_magnitude * phase
        a_new = np.real(ifft(new_fa, size))[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = best_a.copy()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 20:
                # Perturb
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if time.monotonic() - start > time_budget * 0.8:
                    break
            else:
                a = a_new

        it += 1
        if verbose and it % 50 == 0:
            print(f"    SIAA it{it}: obj={best_obj:.8f}, gamma={gamma:.3f}")

    return best_a


def peak_decomposition_search(a, time_budget=80.0, verbose=False):
    """
    Local search that decomposes the peak into contributing pairs and
    systematically tries to reduce the largest contributions.
    """
    n = len(a)
    best_a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(best_a))
    if s < 1e-10:
        return a
    best_a = best_a / s
    best_obj = compute_obj(best_a)
    start = time.monotonic()
    rng = np.random.default_rng(54321)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(best_a)
        peak_k = int(np.argmax(conv))
        peak_val = float(conv[peak_k])

        # Get all contributing pairs for this peak
        i_start = max(0, peak_k - (n - 1))
        i_end = min(n - 1, peak_k)

        # Compute contributions
        contributions = []
        for i in range(i_start, i_end + 1):
            j = peak_k - i
            if 0 <= j < n:
                contributions.append((best_a[i] * best_a[j], i, j))

        contributions.sort(reverse=True)

        improved = False
        # Try to reduce top contributions by moving mass
        for _, i, j in contributions[:5]:
            if i == j:
                continue

            # Try moving mass from i to various targets
            for delta_frac in [0.02, 0.05, 0.1]:
                delta = delta_frac * best_a[i]
                if delta >= best_a[i] * 0.5:
                    continue

                # Target selection: away from peak contributors
                # Prefer targets that don't contribute to this peak
                targets = []
                for t in range(n):
                    # Check if (i, t) or (t, j) would contribute to peak
                    if t == i or t == j:
                        continue
                    if abs(peak_k - t) < n // 4:  # avoid nearby positions
                        continue
                    targets.append(t)

                if not targets:
                    targets = [int(rng.integers(0, n))]

                # Try a few random targets plus structured ones
                tried = set()
                for _ in range(3):
                    t = int(rng.choice(targets)) if targets else int(rng.integers(0, n))
                    t = t % n
                    if t in tried or t == i:
                        continue
                    tried.add(t)

                    a_try = best_a.copy()
                    a_try[i] -= delta
                    a_try[t] += delta
                    if a_try[i] < 0:
                        continue
                    a_try = a_try / a_try.sum()
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-15:
                        best_obj = obj_try
                        best_a = a_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 80:
                break
            # Periodic perturbation
            if stall % 10 == 0:
                scale = min(0.05, 0.01 * stall)
                best_a = best_a * np.exp(rng.normal(0, scale, n))
                best_a = np.maximum(best_a, 0)
                best_a = best_a / best_a.sum()

        it += 1
        if verbose and it % 20 == 0:
            print(f"    PDS it{it}: obj={best_obj:.8f}, peak={peak_val:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - 0.9 * x**2 + 0.1))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        # Two-sided exponential decay from center
        center = n // 2
        a = np.exp(-0.5 * np.abs(np.arange(n) - center) / (n * 0.15))
        a = a / a.sum()
    else:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization:
    1. Multi-n quick LP exploration
    2. Adaptive LP polishing (minimax + weighted + topk cycling)
    3. Spectral IAA with annealing
    4. Peak-decomposition local search
    5. Post-spectral LP
    6. Diversified restarts
    7. Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent: {e}")

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Multi-n Quick Exploration ===
    print(f"\n=== Phase 1: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()

    n_current = len(best_seq)
    n_options = []
    for delta in [-300, -200, -100, 100, 200, 300]:
        n_try = n_current + delta
        if 200 <= n_try <= 1500:
            n_options.append(n_try)
    for n_fixed in [400, 500, 600, 700, 900, 1000, 1100]:
        if n_fixed not in n_options and 200 <= n_fixed <= 1500:
            n_options.append(n_fixed)
    n_options = sorted(set(n_options))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        alloc = min(remaining * 0.06, 20.0)
        if alloc < 10:
            continue
        a_try = resample_sequence(best_seq, n_try)
        a_try = adaptive_lp_polish(a_try, time_budget=alloc)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 2: Intensive Adaptive LP ===
    print(f"\n=== Phase 2: Intensive LP (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 450.0)
    if p2_budget > 50:
        a_lp = adaptive_lp_polish(best_seq, time_budget=p2_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** LP: obj={best_score:.8f}")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 3: Spectral IAA ===
    print(f"\n=== Phase 3: Spectral IAA ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    if p3_budget > 30:
        a_spec = spectral_iaa(best_seq, time_budget=p3_budget, verbose=True)
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  ** Spectral: obj={best_score:.8f}")
        else:
            print(f"  Spectral: obj={obj_spec:.8f}")

    # === Phase 4: Peak Decomposition Search ===
    print(f"\n=== Phase 4: Peak Decomposition Search ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.1, 100.0)
    if p4_budget > 20:
        a_pds = peak_decomposition_search(best_seq, time_budget=p4_budget, verbose=True)
        obj_pds = compute_obj(a_pds)
        if obj_pds < best_score:
            best_score = obj_pds
            best_seq = a_pds.copy()
            print(f"  ** PDS: obj={best_score:.8f}")
        else:
            print(f"  PDS: obj={obj_pds:.8f}")

    # === Phase 5: Post-Processing LP ===
    print(f"\n=== Phase 5: Post-Processing LP ===")

            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 25:
        alloc = min(remaining * 0.35, 100.0)
        a_f = adaptive_lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  ** Final LP: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 50.0)
        a_s = spectral_iaa(best_seq, time_budget=alloc)
        obj_s = compute_obj(a_s)
        if obj_s < best_score:
            best_score = obj_s
            best_seq = a_s.copy()
            print(f"  ** Final Spectral: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.4, 40.0)
        a_p = peak_decomposition_search(best_seq, time_budget=alloc)
        obj_p = compute_obj(a_p)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  ** Final PDS: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
