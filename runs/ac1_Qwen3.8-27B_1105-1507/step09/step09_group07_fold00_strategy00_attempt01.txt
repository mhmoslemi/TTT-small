def compute_conv(a):
    if isinstance(a, list):
        a = np.array(a)
    n = len(a)
    if n == 0:
        return np.array([]).astype(np.float64)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a.astype(np.float64), size)
    conv_result = np.fft.irfft(fa * np.conj(fa), 2 * n - 1)
    return conv_result[:2*n-1].astype(np.float64)
