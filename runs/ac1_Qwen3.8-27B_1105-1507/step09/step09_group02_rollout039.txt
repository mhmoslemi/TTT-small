import numpy as np

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

    n = len(a)

    if s < 1e-10:

        P_new *= P_mean / np.mean(P_new)

        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        
        # Ensure Hermitian symmetry
        A_new[0] = np.abs(A_new[0])
        for k in range(1, (N_fft + 1) // 2):

def resample_sequence(a, new_n):

"""
Spectral flattening + dual-formulation LP polishing for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses frequency-domain phase retrieval
for fast initial improvement, then time-domain LP with both primal
and dual formulations for precise peak reduction. Exploits convolution
symmetry and explores multiple sequence lengths.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import fft, ifft, rfft, irfft


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_half(a):
    """Build the first-half convolution matrix (n rows x n cols) exploiting symmetry."""
    n = len(a)
    A = np.zeros((n, n))
    for j in range(n):
        r = min(j + 1, n)
        A[:r, j] = a[:r][::-1]
    for j in range(n):
        r = min(n - j, n)
        A[r:, j] = a[r:] if r < n else np.array([])
    # Actually let's just do it directly: conv(a,g)[k] = sum_i a[i]*g[k-i] for k=0..n-1
    A = np.zeros((n, n))
    for k in range(n):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] += a[j]
    return A


def build_conv_matrix_full(a):
    """Build full convolution matrix (2n-1 x n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(2 * n - 1):
            i = k - j
            if 0 <= i < n:
                A[k, j] = a[i]
    return A


def solve_lp_primal(a):
    """LP: min M s.t. conv(a,g)_k <= M, sum(g)=1, g>=0. Using half constraints by symmetry."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_half(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0).astype(np.float64)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_dual(a):
    """LP: max sum(g) s.t. conv(a,g)_k <= M_current, g>=0. (SOTW formulation)"""
    n = len(a)
    a_norm = a / np.sum(a)
    conv_current = compute_conv(a_norm)
    M_current = float(np.max(conv_current))
    A = build_conv_matrix_half(a_norm)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M_current]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0).astype(np.float64)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None

    n = len(a)

    stall = 0

        progress = min(1.0, elapsed / time_budget)

        P_max = np.max(P)

            strat = 0
            beta = 0.5

        else:

            P_new = (1.0 - alpha) * P + alpha * P_mean

        P_new = np.maximum(P_new, 1e-30)

        phase = np.angle(A)

        A_new[0] = np.abs(A_new[0])
        half = (N_fft + 1) // 2

            a_new /= s_new

        it += 1

        if verbose and it % 300 == 0:

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        it += 1

    p3_spec = min(remaining * 0.1, 100.0)
    if p3_spec > 20:
        a_spec = spectral_flatten(best_seq, time_budget=p3_spec, verbose=True, strategy='adaptive')
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Phase3 Spectral: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 Spectral: obj={obj_spec:.8f}")

    remaining = deadline - time.monotonic()
    p3_peak = min(remaining * 0.1, 80.0)
    if p3_peak > 15:
        a_pk = peak_direct_reduce(best_seq, time_budget=p3_peak, verbose=True)
        obj_pk = compute_obj(a_pk)
        if obj_pk < best_score:
            best_score = obj_pk
            best_seq = a_pk.copy()
            print(f"  Phase3 Peak: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 Peak: obj={obj_pk:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        400, 500, 600, 700,
        max(300, n_current - 200),
        max(300, n_current - 100),
        n_current + 100,
        n_current + 200,
        n_current + 400,
        900, 1000, 1200, 1500, 2000, 2500
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2500 and abs(n - n_current) > 50]

    tried_n = set()
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        if n_try in tried_n:
            continue
        tried_n.add(n_try)

        a_try = resample_sequence(best_seq, n_try)

        # Quick spectral
        alloc_s = min(remaining * 0.08, 25.0)
        if alloc_s > 10:
            a_try = spectral_flatten(a_try, time_budget=alloc_s, strategy='adaptive')

        # Quick LP
        remaining = deadline - time.monotonic()
        alloc_l = min(remaining * 0.12, 35.0)
        if alloc_l > 15:
            a_try = lp_polish_hybrid(a_try, time_budget=alloc_l)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 40:
        alloc = min(remaining * 0.35, 180.0)
        a_lp = lp_polish_hybrid(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_lp:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.25, 80.0)
        a_spec = spectral_flatten(best_seq, time_budget=alloc, strategy='adaptive')
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Final Spectral: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 60.0)
        a_lp = lp_polish_hybrid(best_seq, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 40.0)
        a_pk = peak_direct_reduce(best_seq, time_budget=alloc)
        obj_pk = compute_obj(a_pk)
        if obj_pk < best_score:
            best_score = obj_pk
            best_seq = a_pk.copy()
            print(f"  Final Peak: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
