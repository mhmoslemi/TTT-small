"""
LP-centric optimization for step function coefficients.
Focuses on iterative LP-based polishing with golden-section line search,
subgradient descent for initialization, and active-set-aware restarts.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def golden_section_search(a, g_lp, time_budget=5.0):
    """Golden section search for optimal blend factor t in [0, 1]."""
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi  # golden ratio conjugate ≈ 0.618
    
    lo, hi = 0.0, 1.0
    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    
    fc = compute_obj((1-c)*a + c*g_lp)
    fd = compute_obj((1-d)*a + d*g_lp)
    
    start = time.monotonic()
    for _ in range(20):
        if time.monotonic() - start > time_budget:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = compute_obj((1-c)*a + c*g_lp)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = compute_obj((1-d)*a + d*g_lp)
    
    t = min(c, d) if fc < fd else max(c, d)
    # Actually, take the better one
    if fc < fd:
        t = c
    else:
        t = d

    # Also try t slightly beyond [0,1] for overshoot
    for t_try in [t * 1.2, t * 1.5, min(1.0, t + 0.1)]:
        if 0 <= t_try <= 1.5:
            a_try = (1 - t_try) * a + t_try * g_lp
            if np.min(a_try) >= -1e-10:
                a_try = np.maximum(a_try, 0)
                s = float(np.sum(a_try))
                if s > 0.01:
                    a_try = a_try / s
                    obj_try = compute_obj(a_try)
                    if obj_try < compute_obj((1-t)*a + t*g_lp):
                        t = t_try
                        break
    
    return t

def subgradient_descent(a, time_budget=30.0, lr_init=0.01, verbose=False):
    """Subgradient descent on the simplex for the peak of conv(a,a)."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))
        
        # Subgradient of conv(a,a)[k_max] w.r.t. a
        # d/d a[i] of sum_j a[j]*a[k_max-j] = a[k_max-i] + a[i-k_max] (with 0 for out of range)
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_max - i
            j2 = i - k_max
            if 0 <= j1 < n:
                grad[i] += a[j1]
            if 0 <= j2 < n:
                grad[i] += a[j2]
        
        # Projected gradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        lr *= 0.999  # slow decay
        it += 1
        
        if it % 1000 == 0 and verbose:
            print(f"    SGD it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
    
    return best_a

def u_shaped_init(n, sharpness=2.0):
    """U-shaped initialization: high at edges, low in middle."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.1 + sharpness * (2*x - 1)**2)
    # Or use the arcsine distribution
    # a = 1.0 / np.sqrt(1 - (2*x-1)**2 + 0.01)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:  # Uniform
        a = np.ones(n)
    elif mode == 1:  # U-shaped (arcsine)
        a = 1.0 / np.sqrt(1 - (2*x-1)**2 + 0.01)
    elif mode == 2:  # U-shaped (power)
        a = (2*x - 1)**4 + 0.1
    elif mode == 3:  # Bimodal
        a = np.exp(-((np.arange(n) - n//4)**2) / (2*(n//8)**2)) + np.exp(-((np.arange(n) - 3*n//4)**2) / (2*(n//8)**2))
    elif mode == 4:  # Triangle
        a = 1 - 4*abs(x - 0.5)
        a = np.maximum(a, 0.01)
    elif mode == 5:  # Random
        a = rng.uniform(0.1, 1, n)
    elif mode == 6:  # Cosine bump
        a = 0.5 + 0.5*np.cos(2*np.pi*x)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def lp_polish_focus(a, time_budget=60.0, verbose=False, max_iters=50):
    """
    Focused LP polishing with golden-section line search.
    This is the main workhorse.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall_count = 0
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        # Solve LP
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Golden section search for blend factor
        t_opt = golden_section_search(a, g_lp, time_budget=min(3.0, remaining*0.1))
        
        if t_opt <= 0:
            stall_count += 1
        else:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                
                if obj_new < best_obj - 1e-15:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    stall_count = 0
                else:
                    stall_count += 1
            else:
                stall_count += 1

"""
LP-centric optimization for step function coefficients.
Uses iterative LP-based polishing with golden-section line search,
subgradient descent for warm-up, U-shaped initialization, and systematic n exploration.
Key: maximizes the effective use of each LP solve via optimal line search.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g_lp, time_budget=4.0):
    """

            t = 0

        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)

    d = lo + (1 - gr) * (hi - lo)

    fd = obj_at(d)

            break

            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)

    obj_best = min(fc, fd)

def subgradient_descent(a, time_budget=40.0, verbose=False):
    """Subgradient descent on the simplex for the peak of conv(a,a)."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.02
    it = 0
    momentum = np.zeros(n)
    beta = 0.5
    best_it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))

        # Subgradient: d/d a[i] of sum_j a[j]*a[k_max-j]
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_max - i
            j2 = i - k_max
            if 0 <= j1 < n:
                grad[i] += a[j1]
            if 0 <= j2 < n:
                grad[i] += a[j2]

        # Nesterov momentum
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
        else:
            a_new = a_new / s_new

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        lr *= 0.9995
        it += 1

            momentum = np.zeros(n)
            lr = 0.01
            best_it = it

        if it % 2000 == 0 and verbose:
            print(f"    SGD it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

    return best_a

    """U-shaped initialization: high at edges, low in middle."""

    return a / s

        print("\n=== Phase 2: Main LP Polishing ===")
        alloc = min(remaining * 0.45, 500.0)
        a_lp = lp_polish_intensive(best_seq, time_budget=alloc, verbose=True, max_iters=120)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Second LP round with perturbation ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print("\n=== Phase 3: LP Round 2 (with perturbation) ===")
        # Perturb before second round
        noise = rng.normal(0, 0.005, len(best_seq))
        a_pert = best_seq + noise
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc = min(remaining * 0.25, 250.0)
        a_lp2 = lp_polish_intensive(a_pert, time_budget=alloc, verbose=True, max_iters=80)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase3 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 LP: obj={obj_lp2:.8f} (no improvement)")

    # === Phase 4: n exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print("\n=== Phase 4: n Exploration ===")
        n_best = len(best_seq)

        n_options = sorted(set([
            n_best - 200, n_best - 100, n_best - 50,
            n_best + 50, n_best + 100, n_best + 200, n_best + 400,
            600, 700, 900, 1000, 1200, 1400, 1600
        ]))
        n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_best) >= 30]
        n_options = n_options[:5]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break

            a_try, obj_try = try_length(best_seq, best_score, n_try, remaining * 0.7, rng)
            if a_try is not None and obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 5: Diverse Restarts ===")
        n_current = len(best_seq)

        for r in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break

            mode = r % 7
            a_r = diverse_init(n_current, rng, mode=mode)
            # Blend with current best
            blend = 0.35
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

            # Quick SG
            if remaining > 40:
                a_r = subgradient_descent(a_r, time_budget=min(15.0, remaining * 0.15))

            # LP
            alloc = min(remaining * 0.3, 80.0)
            if alloc > 10:
                a_r = lp_polish_intensive(a_r, time_budget=alloc, max_iters=40)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 6: Final Polish ===")
        alloc = min(remaining - 10, 200.0)
        if alloc > 20:
            a_final = lp_polish_intensive(best_seq, time_budget=alloc, verbose=True, max_iters=100)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final: obj={best_score:.8f} **")
            else:
                print(f"  Final: obj={obj_final:.8f}")

    # Very final check: try a few more LP iterations if time remains
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_check = lp_polish_intensive(best_seq, time_budget=min(remaining - 5, 60.0), max_iters=30)
        obj_check = compute_obj(a_check)
        if obj_check < best_score:
            best_score = obj_check
            best_seq = a_check.copy()
            print(f"  Extra: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
