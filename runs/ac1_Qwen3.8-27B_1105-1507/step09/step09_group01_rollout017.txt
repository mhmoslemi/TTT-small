A better bound: (δ*δ)[k] = sum_{i+j=k} δ[i]*δ[j] <= (sum |δ|)^2 / (2n-1) by Cauchy-Schwarz... no, that's not right either.

Actually, (δ*δ)[k] <= ||δ||_2^2 * ||δ||_2^2 / (number of terms) ... no.

The simplest bound: (δ*δ)[k] <= (sum |δ[i]| * |δ[j]| for i+j=k) <= (max_i |δ[i]|) * (sum_j |δ[j]|) <= r * (sum |δ|) <= r^2 * n (by Cauchy-Schwarz, sum|δ| <= sqrt(n) * ||δ||_2 <= sqrt(n) * r).

So (δ*δ)[k] <= r * sqrt(n) * r = r^2 * sqrt(n).

We can add this as a margin:
C_old[k] + 2*(a*δ)[k] + r^2 * sqrt(n) <= M

This makes the linearized constraint conservative, ensuring the true objective doesn't increase.

Actually, for the purpose of this problem, I think the linearization without the margin will work well in practice, especially with a small trust region. Let me just use the linearization and rely on the line search to verify improvement.

"""
Trust Region SOCP + LP Hybrid Optimizer for minimum peak autocorrelation.
Uses iteratively linearized SOCP with line search, combined with LP direction
finding, multi-length exploration, and spectral warmup.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = conv(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_tr_sscp(a, r, M_target=None):
    """
    Solve trust region SOCP:
    min M
    s.t. C_old[k] + 2*(a*δ)[k] <= M for all k
         ||δ||_2 <= r
         sum(a+δ) = 1
         a+δ >= 0
    """
    n = len(a)
    a = a / a.sum()
    C_old = compute_conv(a)
    A_cross = build_conv_matrix(a)
    
    delta = cp.Variable(n)
    M = cp.Variable()
    a_new = a + delta
    
    constraints = [
        cp.sum(a_new) == 1,
        a_new >= 0,
        C_old + 2 * (A_cross @ delta) <= M,
    ]
    
    if r is not None:
        constraints.append(cp.norm(delta, 2) <= r)
    
    if M_target is not None:
        constraints.append(M <= M_target)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(delta.value).flatten(), float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_direction(a, M_target=None):
    """
    Solve LP: min M s.t. a*g[k] <= M, sum(g)=1, g>=0.
    """
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(delta.value).flatten(), float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_direction(a, M_target=None):
    """
    Solve LP: min M s.t. a*g[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_value) or (None, None).
    """
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def tr_sscp_polish(a, time_budget=300.0, r_init=0.3, verbose=False, seed=42):
    """Trust region SOCP polishing with adaptive line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    r = r_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        delta_dir, M_val = solve_tr_sscp(a, r)

        if delta_dir is None:
            g_lp, M_val = solve_lp_direction(a)
            if g_lp is None:
                break
            delta_dir = g_lp - a
        else:
            norm_d = np.linalg.norm(delta_dir)
            if norm_d > r:
                delta_dir = delta_dir * (r / norm_d)

        finite_mask = np.isfinite(delta_dir)
        if not finite_mask.all():
            delta_dir = np.where(finite_mask, delta_dir, 0.0)

        best_alpha = 0
        best_alpha_obj = best_obj
        alphas = [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0, 1.3, 1.7, 2.2]
        for t_alpha in alphas:
            a_try = a + t_alpha * delta_dir
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_alpha_obj - 1e-15:
                best_alpha_obj = obj_try
                best_alpha = t_alpha

        if best_alpha > 0 and best_alpha_obj < best_obj - 1e-15:
            a = a + best_alpha * delta_dir
            a = np.maximum(a, 0)
            s_n = a.sum()
            if s_n > 1e-10:
                a = a / s_n
            best_obj = best_alpha_obj
            best_a = a.copy()
            no_improve = 0
            r = min(r * 1.12, 1.8)
        else:
            no_improve += 1
            r *= 0.65
            if no_improve >= 5:
                if r < 0.03:
                    a = best_a * np.exp(rng.normal(0, 0.012, n))
                    a = np.maximum(a, 0)
                    s_p = a.sum()
                    if s_p > 1e-10:
                        a = a / s_p
                    r = r_init
                    no_improve = 0
                    if it > 80:
                        break
                else:
                    r = max(r * 0.5, 0.03)

        it += 1
        if verbose and it % 8 == 0:
            elapsed = time.monotonic() - start
            print(f"    TR it{it}: obj={best_obj:.8f}, r={r:.4f}, t={elapsed:.0f}s")

    return best_a


def lp_polish(a, time_budget=200.0, verbose=False, seed=42):
    """LP-based polishing with line search and adaptive perturbation."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break

        best_t = 0
        best_t_obj = best_obj
        for t_val in [0.03, 0.08, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0]:
            a_try = (1 - t_val) * best_a + t_val * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-15:
                best_t_obj = obj_try
                best_t = t_val

        if best_t > 0 and best_t_obj < best_obj - 1e-15:
            best_obj = best_t_obj
            best_a = (1 - best_t) * best_a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            s_ba = best_a.sum()
            if s_ba > 1e-10:
                best_a = best_a / s_ba
            a = best_a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                scale = 0.008 + 0.005 * min(no_improve, 8)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                no_improve = 0
                if it > 60:
                    break

        it += 1
        if verbose and it % 8 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def smoothed_gradient_descent(a, time_budget=100.0, tau_init=20.0, tau_end=300.0):
    """Smoothed gradient descent with increasing temperature."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 5.0 / np.sqrt(n)
    momentum = 0.8
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.8)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr = max(1e-10, lr * 0.9995)
        it += 1

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def subgradient_refine(a, time_budget=80.0, K=8):
    """Subgradient method with top-K peak weighting."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 3.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break

        weights = np.exp((peak_vals - max_pv) * 8.0 / max_pv)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break
            else:
                lr *= 0.997

        it += 1

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: TR-SOCP + LP hybrid with multi-length exploration.
    Uses trust-region SOCP as the primary optimizer, LP for direction finding,
    smoothed gradient for warmup, and multi-length search for broader coverage.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Quick smoothed GD warmup ===
    print(f"\n=== Phase 1: Smoothed GD Warmup ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 80.0)
    if p1_budget > 15:
        a_sgd = smoothed_gradient_descent(best_seq, time_budget=p1_budget,
                                          tau_init=15.0, tau_end=250.0)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase1 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 GD: obj={obj_sgd:.8f}")

    # === Phase 2: TR-SOCP Polish (primary) ===
    print(f"\n=== Phase 2: TR-SOCP Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 400.0)
    if p2_budget > 30:
        a_tr = tr_sscp_polish(best_seq, time_budget=p2_budget,
                              r_init=0.3, verbose=True, seed=42)
        obj_tr = compute_obj(a_tr)
        if obj_tr < best_score:
            best_score = obj_tr
            best_seq = a_tr.copy()
            print(f"  Phase2 TR: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 TR: obj={obj_tr:.8f}")

    # === Phase 3: LP Polish (complementary) ===
    print(f"\n=== Phase 3: LP Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.18, 180.0)
    if p3_budget > 20:
        a_lp = lp_polish(best_seq, time_budget=p3_budget,
                         verbose=True, seed=77)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase3 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 LP: obj={obj_lp:.8f}")

    # === Phase 4: Subgradient refinement ===
    print(f"\n=== Phase 4: Subgradient Refine ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.05, 50.0)
    if p4_budget > 10:
        a_sg = subgradient_refine(best_seq, time_budget=p4_budget, K=8)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Phase4 SG: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 SG: obj={obj_sg:.8f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 200), max(400, n_current - 100),
        n_current + 100, n_current + 200, n_current + 400,
        1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_current) > 80]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample(best_seq, n_try)

        alloc1 = min(remaining * 0.12, 30.0)
        if alloc1 > 8:
            a_try = smoothed_gradient_descent(a_try, time_budget=alloc1,
                                              tau_init=10.0, tau_end=200.0)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 80.0)
        if alloc2 > 15:
            a_try = tr_sscp_polish(a_try, time_budget=alloc2, r_init=0.25)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Perturbation + TR-SOCP ===
    print(f"\n=== Phase 6: Perturbation + TR-SOCP ===")
    remaining = deadline - time.monotonic()

    for r_idx in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        scale = 0.008 + 0.007 * r_idx
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert < 1e-10:
            break
        a_pert = a_pert / s_pert

        alloc = min(remaining * 0.25, 80.0)
        if alloc > 15:
            a_pert = tr_sscp_polish(a_pert, time_budget=alloc,
                                    r_init=0.2, seed=200 + r_idx)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Pert{r_idx} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Pert{r_idx} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 7: Final intensive TR-SOCP ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.5, 120.0)
        a_final = tr_sscp_polish(best_seq, time_budget=alloc,
                                 r_init=0.15, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final TR: obj={best_score:.8f} **")
        else:
            print(f"  Final TR: obj={obj_final:.8f}")

    # Quick final LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 40.0)
        a_final_lp = lp_polish(best_seq, time_budget=alloc, seed=2000)
        obj_flp = compute_obj(a_final_lp)
        if obj_flp < best_score:
            best_score = obj_flp
            best_seq = a_final_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
