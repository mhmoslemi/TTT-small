def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):

                s = float(np.sum(g_val))

    return None, None

"""

Uses LP-based polishing with analytical t-selection, flattening gradient heuristics,
frequency-domain perturbations, multi-n exploration, and diverse restarts.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_cross_conv(a, b):
    """Cross-convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = fft(a, size)
    fb = fft(b, size)
    return ifft(fa * fb).real[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def find_optimal_t(a, g, num_search=80):
    """Find optimal blend parameter t minimizing obj of (1-t)*a + t*g."""
    # Analytical estimate from quadratic upper bound
    conv_aa = compute_conv(a)
    conv_ag = compute_cross_conv(a, g)
    conv_gg = compute_conv(g)
    max_aa = float(np.max(conv_aa))
    max_ag = float(np.max(conv_ag))
    max_gg = float(np.max(conv_gg))

    # Quadratic: (1-t)^2*max_aa + 2t(1-t)*max_ag + t^2*max_gg
    Aq = max_aa - 2.0 * max_ag + max_gg
    Bq = 2.0 * (max_ag - max_aa)
    if abs(Aq) > 1e-15:
        t_guess = -Bq / (2.0 * Aq)

                         np.clip([t_guess - dt, t_guess, t_guess + dt], 0.01, 0.99)]))

            best_obj = obj
            best_t = t

        else:
            lo = m1

    return best_t, best_obj


def flattening_step(a, num_steps=50, eta_init=0.01, eta_decay=0.95):
    """
    Direct flattening: penalize convolution entries above the mean.
    Uses gradient of sum_k max(0, conv[k] - mean)^2 with respect to a.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    eta = eta_init

    for step in range(num_steps):
        conv = compute_conv(a)
        mean_conv = 1.0 / (2 * n - 1)  # since sum(a)=1, sum(conv)=1
        excess = np.maximum(conv - mean_conv, 0)
        excess_sq = np.sum(excess ** 2)

        # Gradient of sum_k excess[k]^2 w.r.t. a[i]
        # d(conv[k])/d(a[i]) = a[k-i] if 0<=k-i<n, else 0 (for i in the "left" part)
        # + a[i + (k-n+1)] ... it's the cross-correlation structure
        # Actually: conv[k] = sum_j a[j]*a[k-j], so d(conv[k])/d(a[i]) = a[k-i] + a[i + something]
        # More precisely: d(conv[k])/d(a[i]) = sum of a[j] where i+j=k, i.e., a[k-i]
        # but we need to account for both a's in the product
        # d/d(a[i]) of sum_j a[j]*a[k-j] = a[k-i] (from the first a) 
        #   if we fix the second... actually it's:
        # conv[k] includes term a[i]*a[k-i] (if valid) AND a[k-i]*a[i] (if k-i != i)
        # So d(conv[k])/d(a[i]) = 2*a[k-i] if i != k-i, else a[k-i] (i.e., a[i])
        # But for the excess penalty, only k where excess>0 matter
        
        # Compute gradient via FFT cross-correlation
        # grad[i] = sum_k 2*excess[k] * a[k-i] (for valid k-i)
        # This is 2 * cross_corr(excess, a) truncated to length n
        
        N = 1
        while N < 3 * n:
            N <<= 1
        E = fft(excess, N)
        A = fft(a, N)
        grad_full = ifft(E * np.conj(A)).real
        grad = 2.0 * grad_full[:n]
        
        # Also need to handle the symmetry: conv[k] also gets contribution from a[k-i] in the other factor
        # Actually the above handles it correctly because:
        # d(conv[k])/d(a[i]) = a[k-i] (treating a in first position) 
        # but conv is symmetric so we get it twice... 
        # Let me verify: conv[k] = sum_j a[j] a[k-j]
        # d/d(a[i]) = a[k-i] (assuming k-i is valid)
        # But also when j = k-i: d/d(a[i]) of a[i]*a[k-i] = a[k-i]
        # And when k-j = i, i.e., j = k-i: same term
        # So if i != k-i, the term a[i]*a[k-i] appears once, derivative is a[k-i]
        # If i = k-i (i.e., 2i=k), derivative is a[i]
        # So d(conv[k])/d(a[i]) = a[k-i] for all valid (k,i) pairs
        # The gradient is: grad[i] = sum_k 2*excess[k]*a[k-i] (factor 2 from the squared excess)
        # Wait, d(excess^2)/d(a[i]) = 2*excess[k] * d(conv[k])/d(a[i]) = 2*excess[k]*a[k-i]
        # And we sum over k. So grad[i] = 2 * sum_k excess[k]*a[k-i]
        # Which is 2 * (excess * a evaluated at shift i) = 2 * cross_corr(excess, a)[i + offset]
        # My FFT computation gives: ifft(FFT(excess) * conj(FFT(a)))[i] = sum_k excess[k] * a[k-i]
        # So grad = 2 * that. Good.

        # Project gradient onto tangent space of simplex (sum(a)=1)
        grad = grad - np.mean(grad)

        # Update
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Line search: reduce eta
            eta *= 0.7
            if eta < 1e-8:
                break

        eta *= eta_decay

    return best_a


def freq_perturb(a, rng, scale=0.05):
    """Frequency-domain perturbation: add random phase noise to FFT."""
    n = len(a)
    N = 2 * n
    A = fft(a, N)
    # Preserve DC component
    phase_noise = np.exp(1j * rng.normal(0, scale, N))
    phase_noise[0] = 1.0
    A_perturbed = A * phase_noise
    a_new = ifft(A_perturbed).real[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def amp_perturb(a, rng, scale=0.03):

    a_new = ifft(A_perturbed).real[:n]
    a_new = np.maximum(a_new, 0)

def shape_perturb(a, rng, scale=0.05):
    """Multiply by a smooth random shape."""
    n = len(a)
    # Generate smooth random multipliers via sinc interpolation
    num_knots = max(3, n // 20)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Gaussian bumps
        num_bumps = rng.integers(2, 6)
        centers = rng.uniform(0, n, num_bumps)
        widths = rng.uniform(n * 0.05, n * 0.2, num_bumps)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like distribution
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-8))
        a = a / a.sum()
    elif mode == 4:
        # Random Fourier features
        num_freqs = rng.integers(3, 8)
        freqs = rng.uniform(0.5, 4.0, num_freqs)
        phases = rng.uniform(0, 2 * np.pi, num_freqs)
        amplitudes = rng.uniform(0.1, 0.5, num_freqs)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p, amp in zip(freqs, phases, amplitudes):
            a += amp * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        alpha = rng.uniform(-3, 3)
        a = x ** alpha
        a = a / a.sum()
    elif mode == 6:
        # Cosine bell
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        # Two-sided exponential
        center = n // 2
        decay = rng.uniform(0.02, 0.1)
        x = np.arange(n)
        a = np.exp(-decay * np.abs(x - center))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def lp_polish(a, max_iter=60, time_limit=100.0, verbose=False):
    """
    LP polishing: repeatedly solve LP for optimal g given a, find optimal t, blend.
    Returns (best_a, best_obj, num_iterations).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf, 0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    no_improve = 0
    start = time.monotonic()
    it = 0

    while it < max_iter and (time.monotonic() - start) < time_limit:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Find optimal t
        t_opt, obj_at_t = find_optimal_t(a, g_lp, num_search=60)
        it += 1

        if obj_at_t < best_obj - 1e-13:
            best_obj = obj_at_t
            a_blend = (1.0 - t_opt) * a + t_opt * g_lp
            a_blend = np.maximum(a_blend, 0)
            s_blend = float(np.sum(a_blend))
            if s_blend > 1e-10:
                best_a = a_blend / s_blend
                a = best_a.copy()
            no_improve = 0
            if verbose and it % 3 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, M={M_lp:.6f}")
        else:
            no_improve += 1
            if no_improve >= 4:
                break

    return best_a, best_obj, it

        lp_time = min(remaining * 0.5, 40.0)
        if lp_time > 10:
            a_lp, obj_lp, _ = lp_polish(a, max_iter=30, time_limit=lp_time)
            if obj_lp < best_obj - 1e-13:
                best_obj = obj_lp
                best_a = a_lp.copy()
                a = a_lp.copy()
                if verbose:
                    print(f"      Hybrid LP: obj={best_obj:.8f} **")
            else:
                a = a_lp  # still use the result as starting point

        # Flattening step
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        flat_time = min(remaining * 0.3, 20.0)
        num_flat_steps = max(10, int(flat_time * 2))
        a_flat = flattening_step(a, num_steps=num_flat_steps, eta_init=0.005)
        obj_flat = compute_obj(a_flat)
        if obj_flat < best_obj - 1e-13:
            best_obj = obj_flat
            best_a = a_flat.copy()
            a = a_flat.copy()
            if verbose:
                print(f"      Hybrid Flat: obj={best_obj:.8f} **")
        else:
            a = a_flat

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP polishing with analytical t + flattening + perturbations + multi-n.
    Returns list of non-negative floats minimizing the evaluation objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Intensive LP polishing from parent ===
    print(f"\n=== Phase 1: LP Polish (from parent) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.22, 280.0)
    if p1_budget > 30:
        a_p1, obj_p1, iters = lp_polish(best_seq, max_iter=80, time_limit=p1_budget, verbose=True)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Phase 1: obj={best_score:.8f} ({iters} iters) **")
        else:
            print(f"  Phase 1: obj={obj_p1:.8f} ({iters} iters)")

    # === Phase 2: Hybrid LP + Flattening ===
    print(f"\n=== Phase 2: Hybrid Optimization ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.20, 250.0)
    if p2_budget > 30:
        a_p2 = hybrid_optimize(best_seq, time_budget=p2_budget, verbose=True)
        obj_p2 = compute_obj(a_p2)
        if obj_p2 < best_score:
            best_score = obj_p2
            best_seq = a_p2.copy()
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj_p2:.8f}")

    # === Phase 3: Perturbation + LP recovery ===
    print(f"\n=== Phase 3: Perturb & Recover ===")
    remaining = deadline - time.monotonic()
    n_perturb_rounds = 8
    for r in range(n_perturb_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        # Choose perturbation method
        method = r % 4
        if method == 0:
            scale = 0.02 + 0.01 * r
            a_pert = freq_perturb(best_seq, rng, scale=scale)
        elif method == 1:
            scale = 0.02 + 0.01 * r
            a_pert = amp_perturb(best_seq, rng, scale=scale)
        elif method == 2:
            scale = 0.03 + 0.01 * r
            a_pert = shape_perturb(best_seq, rng, scale=scale)
        else:
            # Random local modification
            a_pert = best_seq.copy()
            num_modify = max(5, len(a_pert) // 20)
            idx = rng.choice(len(a_pert), num_modify, replace=False)
            a_pert[idx] *= rng.uniform(0.5, 1.5, num_modify)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()

        # Quick LP polish
        alloc = min(remaining * 0.3, 60.0)
        if alloc > 15:
            a_rec, obj_rec, _ = lp_polish(a_pert, max_iter=30, time_limit=alloc)
            if obj_rec < best_score:
                best_score = obj_rec
                best_seq = a_rec.copy()
                print(f"  Perturb {r} (method={method}): obj={best_score:.8f} **")
                # Additional polish on the improved sequence
                remaining = deadline - time.monotonic()
                if remaining > 30:
                    a_extra, obj_extra, _ = lp_polish(best_seq, max_iter=30, time_limit=min(remaining * 0.3, 40.0))
                    if obj_extra < best_score:
                        best_score = obj_extra
                        best_seq = a_extra.copy()
                        print(f"  Perturb {r} (extra): obj={best_score:.8f} **")
            else:
                print(f"  Perturb {r} (method={method}): obj={obj_rec:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Generate candidate n values
    n_candidates = set()
    for delta in [-100, -50, -25, 25, 50, 100, 200, 300, 500]:
        n_try = n_current + delta
        if 300 <= n_try <= 2000:
            n_candidates.add(n_try)
    # Add some fixed values
    for n_fixed in [500, 600, 800, 1000, 1200, 1500]:
        if 300 <= n_fixed <= 2000 and n_fixed != n_current:
            n_candidates.add(n_fixed)

    n_candidates = sorted(n_candidates, key=lambda x: abs(x - n_current))

    for n_try in n_candidates[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick hybrid optimization
        alloc = min(remaining * 0.15, 80.0)
        if alloc > 20:
            a_try = hybrid_optimize(a_try, time_budget=alloc)
            obj_try = compute_obj(a_try)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        mode = r % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with best to maintain quality
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.2, 80.0)
        if alloc > 25:
            a_r = hybrid_optimize(a_r, time_budget=alloc)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Coarse-to-fine optimization ===
    print(f"\n=== Phase 6: Coarse-to-Fine ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        # Try optimizing at coarser resolution then upsampling
        for n_coarse in [100, 200, 400]:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            a_coarse = resample_sequence(best_seq, n_coarse)
            alloc = min(remaining * 0.1, 40.0)
            if alloc > 15:
                a_coarse = hybrid_optimize(a_coarse, time_budget=alloc)
                # Upsample back
                n_target = len(best_seq)
                a_up = resample_sequence(a_coarse, n_target)
                # Fine polish
                remaining = deadline - time.monotonic()
                alloc2 = min(remaining * 0.08, 30.0)
                if alloc2 > 10:
                    a_up = hybrid_optimize(a_up, time_budget=alloc2)
                obj_up = compute_obj(a_up)
                if obj_up < best_score:
                    best_score = obj_up
                    best_seq = a_up.copy()
                    print(f"  Coarse-to-fine (n_coarse={n_coarse}): obj={best_score:.8f} **")
                else:
                    print(f"  Coarse-to-fine (n_coarse={n_coarse}): obj={obj_up:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Multiple rounds of LP with perturbation
    for lp_rnd in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        alloc = min(remaining * 0.2, 50.0)
        if alloc < 15:
            break

        a_cur = best_seq.copy()
        if lp_rnd > 0:
            # Small perturbation to escape
            r2 = np.random.default_rng(seed + 1000 + lp_rnd)
            if lp_rnd % 2 == 0:
                a_cur = freq_perturb(a_cur, r2, scale=0.015)
            else:
                a_cur = shape_perturb(a_cur, r2, scale=0.02)

        a_final, obj_final, _ = lp_polish(a_cur, max_iter=50, time_limit=alloc)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP {lp_rnd}: obj={best_score:.8f} **")
        else:
            print(f"  Final LP {lp_rnd}: obj={obj_final:.8f}")

    # Final flattening pass
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_flat_final = flattening_step(best_seq, num_steps=50, eta_init=0.003)
        obj_flat_final = compute_obj(a_flat_final)
        if obj_flat_final < best_score:
            best_score = obj_flat_final
            best_seq = a_flat_final.copy()
            print(f"  Final flatten: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
