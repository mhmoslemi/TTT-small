import numpy as np

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a, solver_list=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:
        solver_list = [cp.CBC, cp.GLPK, cp.CLARABEL, cp.SCS, cp.ECOS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def smoothed_max_grad(a, tau):
    """Compute gradient of smooth max of conv(a,a) using softmax weights."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:

    # (a*a)[k] = sum_i a[i] * a[k-i] (with appropriate bounds)
    # d(a*a)[k]/da[i] = a[k-i] + a[i-k] (with bounds)
    # So grad[i] = 2 * sum_k p[k] * (a[k-i] + a[i-k] with bounds)
    # This is 2 * (p * a)[i] + 2 * (p_rev * a)[i] where p is padded
    
    # Actually: (a*a)[k] = sum_j a[j] * a[k-j] for max(0,k-n+1) <= j <= min(k, n-1)
    # d/d a[i] (a*a)[k] = a[k-i] (if 0 <= k-i <= n-1) + a[i-k] (if 0 <= i-k <= n-1)
    # = a[k-i] * 1[0<=k-i<n] + a[i-k] * 1[0<=i-k<n]

    # = 2 * sum_k p[k] * a[k-i] * 1[0<=k<n and 0<=k-i<n]  ... this is getting complex

    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real

    return grad, conv

def chirp_perturbation(a, rng, strength=0.03):
    """Add a chirp-like perturbation in the frequency domain."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    A = rfft(a, size)
    # Add structured perturbation to phases
    freqs = np.arange(len(A))
    phase_shift = strength * np.sin(np.pi * freqs / len(A))
    A_new = A * np.exp(1j * phase_shift)
    A_new[0] = abs(A_new[0])
    a_new = irfft(A_new, size)[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

def fourier_perturbation(a, rng, strength=0.02, n_modes=5):
    """Add random low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)

        freq = k * 2 * np.pi / n

    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

def entropic_mirror_step(a, g, tau, lr):
    """Mirror descent step with entropic regularization."""
    # a_new[i] = a[i] * exp(-lr * g[i] / tau) / Z
    shifted = a - (lr / tau) * g
    # Shift for numerical stability

    a_new = np.exp(shifted)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, solver_list=None, verbose=False):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if solver_list is None:
        solver_list = [cp.CBC, cp.GLPK, cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]

    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_max_grad(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    grad = 2.0 * g_full[:n]
    return grad, conv

def resample(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):

    elif mode == 5:
        a = np.exp(-np.abs(np.arange(n) - n / 2) / (n / 5))
        return a / a.sum()
    elif mode == 6:
        # Three Gaussian bumps
        centers = [n // 5, n // 2, 4 * n // 5]
        w = max(1, n // 12)
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 7:
        # Power-law decay from both ends
        x = np.arange(n)
        d = np.minimum(x, n - 1 - x).astype(float)
        a = np.exp(-d / (n / 6))
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: aggressive LP iterations, smoothed GD, frequency perturbations,
    subspace optimization, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Quick LP to establish baseline ===
    print("\n=== Phase 1: Quick LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(remaining * 0.12, 150.0)
        a = lp_polish(best_seq, time_budget=alloc, verbose=True, seed=10)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            n_current = len(best_seq)
            print(f"  P1: obj={best_score:.8f} **")
        else:
            print(f"  P1: obj={obj:.8f}")

    # === Phase 2: Smoothed GD to refine ===
    print("\n=== Phase 2: Smoothed GD ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.12, 160.0)
        a = gradient_descent(best_seq, time_budget=alloc, tau_start=5.0, tau_end=400.0, verbose=True)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  P2: obj={best_score:.8f} **")
        else:
            print(f"  P2: obj={obj:.8f}")

    # === Phase 3: Intensive LP (main workhorse) ===
    print("\n=== Phase 3: Intensive LP ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        alloc = min(remaining * 0.25, 350.0)
        a = lp_polish(best_seq, time_budget=alloc, verbose=True, seed=20)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  P3: obj={best_score:.8f} **")
        else:
            print(f"  P3: obj={obj:.8f}")

    # === Phase 4: Perturbation + Refinement cycles ===
    print("\n=== Phase 4: Perturbation Cycles ===")
    remaining = deadline - time.monotonic()
    perturbations = [
        ("chirp", lambda a: chirp_perturbation(a, rng, strength=0.025)),
        ("mask", lambda a: mask_perturbation(a, rng, strength=0.05)),
        ("fourier", lambda a: fourier_perturbation(a, rng, strength=0.02, n_modes=6)),
        ("chirp2", lambda a: chirp_perturbation(a, rng, strength=0.05, n_modes=3)),
    ]
    for pidx, (pname, pfunc) in enumerate(perturbations):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        a = pfunc(best_seq)
        # GD then LP
        alloc_gd = min(remaining * 0.12, 60.0)
        if alloc_gd > 15:
            a = gradient_descent(a, time_budget=alloc_gd, tau_start=10.0, tau_end=300.0)
        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.12, 60.0)
        if alloc_lp > 15:
            a = lp_polish(a, time_budget=alloc_lp, seed=100 + pidx * 10)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  P4 {pname}: obj={best_score:.8f} **")
        else:
            print(f"  P4 {pname}: obj={obj:.8f}")

    # === Phase 5: Subspace LP ===
    print("\n=== Phase 5: Subspace LP ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.08, 60.0)
        a = subspace_lp(best_seq, time_budget=alloc, n_blocks=4, seed=500)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  P5: obj={best_score:.8f} **")
        else:
            print(f"  P5: obj={obj:.8f}")

    # === Phase 6: Multi-n exploration ===
    print("\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        n_best - 50, n_best + 50, n_best + 100, n_best + 150, n_best + 250,
        600, 650, 700, 750, 850, 900, 950, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500 and abs(n - n_best) >= 25]
    n_options = n_options[:5]

    print(f"  Trying n: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        a_try = resample(best_seq, n_try)
        # Quick GD
        alloc = min(remaining * 0.15, 50.0)
        a_try = gradient_descent(a_try, time_budget=alloc, tau_start=10.0, tau_end=300.0)
        # Short LP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.12, 40.0)
        if alloc2 > 12:
            a_try = lp_polish(a_try, time_budget=alloc2, seed=n_try)
        obj = compute_obj(a_try)
        if obj < best_score:
            best_score = obj
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj:.8f}")

    # === Phase 7: Diverse restarts ===
    print("\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 55:
            break
        a_r = diverse_init(n_cur, rng, mode=r + 1)
        # Blend with best
        blend = 0.25 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.18, 80.0)
        a_r = gradient_descent(a_r, time_budget=alloc * 0.5, tau_start=10.0, tau_end=300.0)
        remaining = deadline - time.monotonic()
        if remaining > 25:
            a_r = lp_polish(a_r, time_budget=min(remaining * 0.1, 35.0), seed=700 + r * 10)
        obj = compute_obj(a_r)
        if obj < best_score:
            best_score = obj
            best_seq = a_r.copy()
            print(f"  R{r} (mode {r+1}): obj={best_score:.8f} **")
        else:
            print(f"  R{r} (mode {r+1}): obj={obj:.8f}")

    # === Phase 8: Final intensive polish ===
    print("\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau GD
    if remaining > 35:
        alloc = min(remaining * 0.2, 100.0)
        a = gradient_descent(best_seq, time_budget=alloc, tau_start=50.0, tau_end=1200.0)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  Final GD: obj={best_score:.8f} **")
        else:
            print(f"  Final GD: obj={obj:.8f}")

    # LP polish
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.25, 100.0)
        a = lp_polish(best_seq, time_budget=alloc, verbose=True, seed=900)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj:.8f}")

    # One more short LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 50.0)
        a = lp_polish(best_seq, time_budget=alloc, seed=999)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  Ultra-final LP: obj={best_score:.8f} **")

    # Validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
