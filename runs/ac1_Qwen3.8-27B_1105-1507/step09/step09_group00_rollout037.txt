```python
def solve_lp_direct(a, constraint_indices, time_limit=5.0):
    """
    Solve: min M s.t. (conv(a, x))[k] <= M for k in constraint_indices, sum(x)=1, x>=0.
    Returns (x_opt, M_opt).
    """
    n = len(a)
    K = len(constraint_indices)
    
    # Build the constraint matrix
    # For each k in constraint_indices: sum_j a[max(0,k-n+1):min(n,k+1)] offset*x <= M
    # i.e., (A@x)[k] - M <= 0
    
    # The matrix for the inequality constraints: [B, -1] where B is K x n
    # B[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    B = np.zeros((K, n))
    for idx, k in enumerate(constraint_indices):
        for j in range(n):
            if 0 <= k - j < n:
                B[idx, j] = a[k - j]
    
    # Inequality constraints: B@x - M <= 0
    # Equality: sum(x) = 1
    # Variables: x_0,...,x_{n-1}, M (total n+1)
    
    c = np.zeros(n + 1)
    c[n] = 1  # minimize M
    
    A_ub = np.hstack([B, -np.ones((K, 1))])
    b_ub = np.zeros(K)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        x = res.x[:n]
        M = res.x[n]
        return x, M
    return None, None
```

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):

    n = max(len(a), len(b))

The LP step finds the best "response" g for the given a, and the update a ← (1-t)a + tg moves a toward this response. This is similar to the "mirror descent" or "follow the leader" approach.

The convergence rate depends on the geometry of the problem. For convex problems (which this is, since max of linear functions is convex and the feasible set is convex), we have sublinear convergence.

To speed up convergence:
- Use larger t (closer to 1)
- Use acceleration (Nesterov-style)
- Solve the LP more accurately (more constraints)

sum_i a[i]*a[k-i] is a quadratic form. Is it convex? The Hessian is the convolution matrix (symmetric part), which is PSD since it's a "Gram matrix" of the convolution structure. Actually, for the autocorrelation, the quadratic form x^T Q x where Q is the Toeplitz/circulant matrix is convex if Q is PSD.

The autocorrelation matrix (Toeplitz with first row a) is PSD if and only if a is the autocorrelation of some signal. But here a is the signal itself, so the matrix has entries Q[i,j] = a[|i-j|] (for the circulant version). This is PSD if the Fourier transform of a is non-negative, which is not guaranteed.

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses iterative LP with constraint generation (top-K peaks), adaptive blending with line search,
smoothed gradient descent for polish, and multi-length exploration.
Key ideas: fast LP solves via HiGHS with reduced constraint sets, adaptive K scheduling,
periodic perturbation to escape local optima, and spectral diversification.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    la, lb = len(a), len(b)
    size = 1
    while size < la + lb - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:la + lb - 1]


def objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 assuming a >= 0."""
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    n = len(a)
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_constraint_rows(a, k_indices):
    """Build B matrix (K x n) where B[i,j] = a[k_i - j] if valid."""
    n = len(a)
    K = len(k_indices)
    B = np.zeros((K, n))
    for i, k in enumerate(k_indices):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            B[i, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1][::-1]
    return B


def solve_lp(a, k_indices):
    """
    Solve: min M s.t. (conv(a,x))[k] <= M for k in k_indices, sum(x)=1, x>=0.
    Returns (x_opt, M_opt) or (None, None).
    """
    n = len(a)
    K = len(k_indices)
    if K == 0:
        return np.ones(n) / n, 0.0

    B = build_constraint_rows(a, k_indices)

    c = np.zeros(n + 1)
    c[n] = 1.0

    A_ub = np.hstack([B, -np.ones((K, 1))])
    b_ub = np.zeros(K)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            x = np.maximum(res.x[:n], 0)
            s = float(np.sum(x))
            if s > 1e-10:
                x = x / s
            return x, float(res.x[n])
    except Exception:
        pass
    return None, None


def line_search_blend(a, g, max_obj, n_trials=8):
    """Find best blending factor t in [0,1] for a_new = (1-t)*a + t*g."""
    best_t = 0.0
    best_obj = max_obj
    # Try geometrically spaced t values
    t_vals = [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05]
    for t in t_vals:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj_try = objective(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
    return best_t, best_obj


def smoothed_gradient(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    # Gradient of conv(a,a) with respect to a:
    # d/d a[j] conv[k] = a[k-j] + a[j-k] (when valid)
    # So grad[j] = sum_k p[k] * (a[k-j] + a[j-k])
    N = 1
    while N < 3 * n:
        N <<= 1
    from scipy.fft import fft, ifft
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def gd_polish(a, time_budget=60.0, tau_init=50.0, tau_end=2000.0):
    """Nesterov accelerated smoothed gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = objective(a)
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 2

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, _ = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))
        if g_norm > 1e-15:
            lr = 1.0 / (g_norm + 1e-10)
            lr = min(max(lr, 0.5 / np.sqrt(n)), 2.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = objective(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr = max(1e-10, lr * 0.999)
        it += 1

        if it % 20000 == 0:
            a = best_a.copy()
            v = np.zeros(n)

    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 4):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def lp_main_loop(a, time_budget, k_start=30, k_end=400, verbose=True):
    """
    Main LP iteration loop with adaptive constraint count.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = objective(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        # Adaptive K
        progress = min(1.0, (time.monotonic() - start) / max(time_budget, 1e-10))
        K = int(k_start + (k_end - k_start) * progress)
        K = min(K, 2 * n - 1)

        # Select top-K peaks of conv(a,a)
        conv_aa = compute_conv(a)
        k_indices = np.argsort(conv_aa)[::-1][:K]
        k_indices = list(k_indices)

        # Solve LP
        g, M = solve_lp(a, k_indices)
        if g is None:
            break

        # Check constraint violations (optional, for larger K)
        if K < 2 * n - 1 and it % 10 == 0:
            conv_ag = compute_conv_cross(a, g)
            max_viol = max(float(conv_ag[k]) - M for k in range(2 * n - 1) if k not in set(k_indices))
            if max_viol > 0.1 * M + 1e-6:
                # Add most violated constraint
                worst_k = int(np.argmax([conv_ag[k] if k not in set(k_indices) else -np.inf for k in range(2 * n - 1)]))
                k_indices.append(worst_k)
                g, M = solve_lp(a, k_indices)
                if g is None:
                    break

        # Line search blend
        best_t, obj_try = line_search_blend(a, g, best_obj)
        it += 1

        if best_t > 0 and obj_try < best_obj - 1e-10:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            a_new = a_new / s_new
            best_obj = obj_try
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 200 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it{it}: obj={best_obj:.8f}, K={K}, t={best_t:.3f}, t={elapsed:.0f}s")
        else:
            stall += 1
            if stall > 30:
                # Perturb to escape
                a = fourier_perturb(best_a, rng, strength=0.01 + 0.005 * min(stall, 20))
                stall = 0
                if it > 1000:
                    break

    return best_a
