import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    s2 = s * s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / s2


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu * rho > (cssv - 1.0)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_val = idx[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def build_conv_matrix(a_norm):
    """Build matrix A_mat (2n-1 x n) such that A_mat @ g = convolve(a_norm, g)."""
    n = len(a_norm)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A_mat[j:j + r, j] = a_norm[:r]
    return A_mat


def solve_lp_minpeak(a_norm, M_ub=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    If M_ub is given, add constraint M <= M_ub.
    """
    n = len(a_norm)
    A_mat = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    if M_ub is not None:
        constraints.append(M <= M_ub)
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate', 'optimal_unbounded', 'optimal_inaccurate_unconstrained'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

        best_obj_t = obj_t

        tau_end = tau * 5
        
        while time.monotonic() - (start + tau_idx * per_tau_budget) < alloc:
            progress = min(1.0, it / 20000)
            tau_cur = tau_start + (tau_end - tau_start) * (1 - np.cos(np.pi * progress)) / 2.0
            
            y = a_t + beta * v
            y = np.maximum(y, 0)
            s_y = float(np.sum(y))
            if s_y < 1e-10:
                y = best_a_t.copy()
            else:
                y = y / s_y
            
            g, conv = smoothed_gradient_fft(y, tau_cur)
            
            if prev_g is not None and it % 2 == 0:
                s_vec = y - prev_a
                y_vec = g - prev_g
                ss = float(np.dot(s_vec, s_vec))
                sy = float(np.dot(s_vec, y_vec))
                if ss > 1e-15 and abs(sy) > 1e-12:
                    bb_lr = ss / abs(sy)
                    lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 20.0 / np.sqrt(n))
            
            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a_t.copy()
                v = np.zeros(n)
            else:
                a_new = a_new / s_new
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_t - 1e-14:
                best_obj_t = obj_new
                best_a_t = a_new.copy()
                best_it = it
            
            prev_a = a_t.copy()
            prev_g = g.copy()
            a_t = a_new
            it += 1
            lr = max(1e-8, lr * 0.9998)
            
            if it - best_it > 20000:
                a_t = best_a_t.copy()
                v = np.zeros(n)
                lr = 3.0 / np.sqrt(n)
                best_it = it
        
        if best_obj_t < best_obj:
            best_obj = best_obj_t
            best_a = best_a_t.copy()
            if verbose:
                print(f"  tau={tau}: obj={best_obj:.8f} **")
    
    return best_a


def peak_targeted_perturbation(a, conv, strength=0.1, seed=0):
    """
    Perturb the sequence to reduce the peak contributions.
    Identify positions that strongly contribute to the top peaks
    and reduce their mass, redistributing to "quiet" positions.
    """
    n = len(a)
    rng = np.random.default_rng(seed)
    a_new = a.copy()
    
    # Find top peak positions
    top_k = np.argsort(conv)[::-1][:5]
    
    # For each peak, find which positions contribute most
    contrib = np.zeros(n)
    for k in top_k:
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if 0 <= j < n:
                contrib[i] += a[i] * a[j] / (conv[k] + 1e-10)
    
    # Normalize contribution
    contrib = contrib / (contrib.max() + 1e-10)
    
    # Reduce mass at high-contribution positions, increase at low-contribution
    # Use a multiplicative perturbation
    noise = rng.normal(0, 1, n)
    # Stronger perturbation where contribution is high
    perturbation = 1.0 + strength * (noise - contrib * 2)
    a_new = a * perturbation
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    
    return a_new
