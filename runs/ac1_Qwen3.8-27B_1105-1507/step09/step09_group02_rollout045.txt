import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """FFT-based linear convolution."""
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

def compute_obj(a):
    """2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * np.max(conv)

def build_conv_matrix(a):
    """Matrix A such that A@g = conv(a,g)."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # conv(a,g)[k] = sum_j a[j]*g[k-j]
        # For fixed j, g[k-j] contributes to position k = j, j+1, ..., j+n-1
        # A[k,j] = a[k-j] for k-j in [0, n-1]
        for k in range(j, j+n):
            if k < 2*n-1:
                A[k, j] = a[k-j]
    return A

def solve_lp(a, solver=None):
    """Solve min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / np.sum(a)
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    if solver:
        solvers = [solver] + solvers
    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                if np.sum(g_val) > 1e-10:
                    return g_val / np.sum(g_val), float(M.value)
        except:
            continue
    return None, None

def golden_section_search(a, g, f):
    """Find optimal t in [0,1] for a_new = (1-t)*a + t*g using golden section."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    a_lo, a_hi = 0.0, 1.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = f((1-c)*a + c*g)
    fd = f((1-d)*a + d*g)
    for _ in range(15):  # 15 iterations gives precision ~ 1/phi^15 ≈ 0.0001
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = f((1-c)*a + c*g)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = f((1-d)*a + d*g)
    t = (a_lo + a_hi) / 2
    return t

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.

Key improvements over the previous 1.506325 result:
1. Golden section search for optimal LP blending parameter (replaces fixed candidates)
2. LP dual-guided subgradient: use binding constraint info to focus effort on active peaks
3. Coordinate descent local search: individually optimize each element
4. Spectral pre-conditioning: apply frequency-domain smoothing to break symmetries
5. Simulated annealing restarts: occasional large perturbations to escape local minima
6. Better time allocation with more emphasis on LP polishing
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal, dual_values) or (None, None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    peak_con = A @ g <= M
    constraints = [peak_con, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    dual = peak_con.dual_value
                    if dual is None:
                        dual = np.zeros(2 * n - 1)
                    return g_val / s, float(M.value), dual
        except Exception:
            continue
    return None, None, None


def golden_blend(a, g, f, max_iter=20):
    """
    Find optimal t in [0,1] minimizing f((1-t)*a + t*g) via golden section search.
    Returns (best_t, best_a, best_f).
    """
    gr = (np.sqrt(5) - 1) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)

    a_c = np.maximum((1 - c) * a + c * g, 0)
    a_c = a_c / np.sum(a_c)
    f_c = f(a_c)

    a_d = np.maximum((1 - d) * a + d * g, 0)
    a_d = a_d / np.sum(a_d)
    f_d = f(a_d)

    for _ in range(max_iter):
        if f_c < f_d:
            hi = d
            d = c
            f_d = f_c
            a_d = a_c
            c = hi - gr * (hi - lo)
            a_c = np.maximum((1 - c) * a + c * g, 0)
            a_c = a_c / np.sum(a_c)
            f_c = f(a_c)
        else:
            lo = c
            c = d
            f_c = f_d
            a_c = a_d
            d = lo + gr * (hi - lo)
            a_d = np.maximum((1 - d) * a + d * g, 0)
            a_d = a_d / np.sum(a_d)
            f_d = f(a_d)

    t = (lo + hi) / 2.0
    a_best = np.maximum((1 - t) * a + t * g, 0)
    a_best = a_best / np.sum(a_best)
    f_best = f(a_best)

    # Also check t=0 (no change)
    f_0 = f(a)
    if f_0 < f_best:
        return 0.0, a.copy(), f_0

    return t, a_best, f_best


def lp_polish_golden(a, time_budget, verbose=False):
    """
    LP polishing with golden section blending search.
    Multiple rounds with adaptive perturbation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()

    if s < 1e-10:
        return a

    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp, dual = solve_lp(a)
        if g_lp is None:
            break

        # Golden section search for optimal t
        t_best, a_new, obj_new = golden_blend(best_a, g_lp, compute_obj, max_iter=18)

        iter_count += 1
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = best_a.copy()
            stall = 0
            if verbose and iter_count % 3 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP-golden it{iter_count}: obj={best_obj:.8f}, t={t_best:.4f}, "
                      f"M_lp={M_lp:.6f}, time={elapsed:.1f}s")
        else:
            stall += 1
            # Perturb to escape
            if stall <= 5:
                scale = 0.005 * stall

                scale = 0.02
            else:
                scale = 0.05
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            if stall > 30:
                if verbose:
                    print(f"    LP-golden: stuck at {stall} iterations, breaking")
                break

    return best_a


def spectral_smooth(a, sigma):
    """Apply Gaussian smoothing in frequency domain."""
    n = len(a)
    N = 1
    while N < n:
        N <<= 1
    A = fft(a, N)
    freqs = np.fft.fftfreq(N, d=1.0 / N)
    gauss = np.exp(-2 * (np.pi * sigma * freqs) ** 2)
    A_smooth = A * gauss
    a_smooth = ifft(A_smooth).real[:n]
    a_smooth = np.maximum(a_smooth, 0)
    s = float(np.sum(a_smooth))
    if s > 1e-10:
        a_smooth = a_smooth / s
    return a_smooth


def nesterov_smoothed_descent(a, time_budget, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(1, time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        if it % 3000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(y, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        # Barzilai-Borwein step size adjustment
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9998)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def peak_shave_dual_guided(a, time_budget, dual_weights=None, verbose=False):
    """
    Peak shaving guided by LP dual information.
    If dual_weights provided, weight the subgradient by them.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    lr = 0.05
    momentum = 0.5
    v = np.zeros(n)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Weight peaks by their squared excess
        excess = np.maximum(conv - 0.9 * max_conv, 0)
        weights = excess ** 2
        if np.sum(weights) < 1e-20:
            weights = np.ones_like(conv)

        # If we have dual info, blend with it
        if dual_weights is not None and len(dual_weights) == len(conv):
            dw = np.maximum(dual_weights, 0)
            if np.sum(dw) > 1e-10:
                dw = dw / dw.max()
                weights = 0.5 * weights + 0.5 * dw

        weights = weights / (np.sum(weights) + 1e-20)

        # Compute subgradient
        g = np.zeros(n)
        # For efficiency, only process peaks with significant weight
        threshold = np.sum(weights) * 0.01
        active = weights > threshold
        for k in np.where(active)[0]:
            k = int(k)
            w = weights[k]
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        g_norm = np.linalg.norm(v)
        if g_norm < 1e-15:
            break

        a_new = a - lr * v / g_norm
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr *= 0.98
        else:
            lr *= 0.7
            a = best_a
            if lr < 1e-10:
                break

        it += 1
        if verbose and it % 200 == 0:
            print(f"    PeakShave it{it}: obj={best_obj:.8f}, lr={lr:.6f}")

    return best_a


def coordinate_descent(a, time_budget, verbose=False):
    """
    Coordinate descent: individually optimize each element.
    For each element i, try small perturbations and keep if improved.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Compute initial convolution for reference
    step = 1.0 / (n * 5)  # small step
    rng = np.random.default_rng(99)
    order = np.arange(n)

    pass_count = 0
    while time.monotonic() - start < time_budget:
        # Randomize order each pass
        if pass_count > 0:
            rng.shuffle(order)

        improved = 0
        for i in order:
            remaining = time_budget - (time.monotonic() - start)
            if remaining < 1:
                break

            # Try increasing and decreasing a[i]
            for delta in [step, -step, 3 * step, -3 * step]:
                a_try = best_a.copy()
                a_try[i] = max(0, a_try[i] + delta)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    improved += 1
                    break  # Move to next coordinate

        pass_count += 1
        if verbose and pass_count % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD pass{pass_count}: obj={best_obj:.8f}, improved={improved}, t={elapsed:.1f}s")

        if improved == 0:
            # Decrease step size
            step *= 0.7
            if step < 1e-10:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
