"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses PDCP-accelerated LP polishing with multiplicative-weights warmup, symmetrization,
Nesterov smoothed descent, adaptive multi-n search, and gradient-directed perturbative
restarts. Key innovations: game-theoretic (HTM) warmup for LP, PDCP solver priority,
symmetry exploitation, and backtracking line search with objective verification.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def symmetrize(a):
    """Symmetrize a sequence: (a + reverse(a)) / 2."""
    return (a + a[::-1].copy()) / 2.0


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        row_start = j
        row_end = min(2 * n - 1, j + n)
        A[row_start:row_end, j] = a[:row_end - row_start]
    return A


def solve_lp(a, solver_list=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_value) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(M), constraints)

    if solver_list is None:
        solver_list = [cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]

    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue
    return None, None

def nesterov_smoothed_descent(a, time_budget=300.0, tau_init=10.0, tau_end=400.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget * 0.9, 1e-6))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_fft = fft(y, N)
        g_full = ifft(P * np.conj(A_fft)).real
        g = 2.0 * g_full[:n]

        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)

        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def backtracking_line_search(best_a, g_lp, max_t=1.0, min_t=0.005, n_tries=15):
    """
    Backtracking line search: find best t in (0, max_t] such that
    obj((1-t)*best_a + t*g_lp) is minimized.
    Uses logarithmic spacing for robustness.
    """
    best_t = min_t
    best_val = np.inf

    # Logarithmic spacing
    ts = np.exp(np.linspace(np.log(min_t), np.log(max_t), n_tries))

    for t in ts:
        a_try = (1.0 - t) * best_a + t * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        val = compute_obj(a_try)
        if val < best_val:
            best_val = val
            best_t = t

    return best_t, best_val

            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a.copy()
            no_improve = 0
            if verbose and iter_count % 3 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={t_opt:.4f}, M={M_lp:.6f}, el={elapsed:.0f}s")
        else:
            no_improve += 1
            if no_improve == 2:
                # Try smaller step
                t_small = max(0.005, t_opt * 0.3)
                a_try2 = (1.0 - t_small) * best_a + t_small * g_lp
                a_try2 = np.maximum(a_try2, 0)
                a_try2 = a_try2 / a_try2.sum()
                obj_try2 = compute_obj(a_try2)
                if obj_try2 < best_obj - 1e-14:
                    best_obj = obj_try2
                    best_a = a_try2.copy()
                    a = best_a.copy()
                    no_improve = 0
                    if verbose:
                        print(f"    LP small it{iter_count}: obj={best_obj:.8f}, t={t_small:.4f}")
            elif no_improve >= 4:
                # Perturb and continue
                scale = 0.005 * min(no_improve, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                if iter_count > 80:
                    break

        # Periodic constraint-tightening LP
        if iter_count % 10 == 5 and M_lp is not None and remaining > 30:
            for tight_frac in [0.93, 0.96, 0.99]:
                M_tight = M_lp * tight_frac
                g_tight, M_ach = solve_lp_constrained(a, M_tight)
                if g_tight is not None:
                    t_try = 0.4
                    a_try3 = (1.0 - t_try) * best_a + t_try * g_tight
                    a_try3 = np.maximum(a_try3, 0)
                    a_try3 = a_try3 / a_try3.sum()
                    obj_try3 = compute_obj(a_try3)
                    if obj_try3 < best_obj - 1e-14:
                        best_obj = obj_try3
                        best_a = a_try3.copy()
                        a = a_try3.copy()
                        no_improve = 0
                        if verbose:
                            print(f"    LP-tight it{iter_count}: obj={best_obj:.8f}, frac={tight_frac}")
                        break

    return best_a


def solve_lp_constrained(a, M_max):
    """Solve LP with additional constraint M <= M_max."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1, M <= M_max]

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue
    return None, None


def hstm_iters(budget_s, n):
    """Estimate number of HTM iterations affordable in budget_s seconds."""
    # Each iteration is O(n) with FFT
    iters_per_sec = 500000.0 / n
    return int(max(500, budget_s * iters_per_sec * 0.8))


def subgradient_refine(a, time_budget=100.0, K=5, verbose=False):
    """Subgradient method with top-K peaks, momentum, and adaptive step."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]

        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        weights = np.exp((peak_vals - max_pv) * 5.0 / max_pv)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break
            else:
                lr *= 0.997

        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        # Hann window
        a = 0.5 * (1 - np.cos(2 * np.pi * x))
    elif mode == 3:
        # Hamming window
        a = 0.54 - 0.46 * np.cos(2 * np.pi * x)
    elif mode == 4:
        # Blackman window
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    elif mode == 5:
        # Blackman-Harris
        a = (0.35875 - 0.48829 * np.cos(2 * np.pi * x) + 0.14128 * np.cos(4 * np.pi * x)
             - 0.01168 * np.cos(6 * np.pi * x))
    elif mode == 6:
        # Flattop window
        a = (0.21557895 - 0.41663158 * np.cos(2 * np.pi * x) + 0.277263158 * np.cos(4 * np.pi * x)
             - 0.083578947 * np.cos(6 * np.pi * x) + 0.006947368 * np.cos(8 * np.pi * x))
    elif mode == 7:
        # Exponential decay
        a = np.exp(-3 * x)
    elif mode == 8:
        # Gaussian
        a = np.exp(-((x - 0.5) ** 2) / (2 * 0.15 ** 2))
    elif mode == 9:
        # Two-peak
        a = np.exp(-((x - 0.25) ** 2) / 0.02) + np.exp(-((x - 0.75) ** 2) / 0.02)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def propose_candidate():
    """
    Main search: PDCP-accelerated LP polishing with HTM warmup, symmetrization,
    Nesterov smoothed GD, adaptive multi-n exploration, and gradient-directed
    perturbative restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + 1475.0  # Leave 25s buffer

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(42)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = diverse_init(n_target, rng, mode=5)  # Blackman-Harris
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # Symmetrize initially
    best_seq = symmetrize(best_seq)
    best_seq = best_seq / best_seq.sum()
    best_score = compute_obj(best_seq)
    print(f"After symmetrization: obj={best_score:.8f}")

    # === Phase 1: Quick smoothed GD warmup ===
    print(f"\n=== Phase 1: Smoothed GD Warmup ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.1, 100.0)
    if p1_budget > 20:
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=p1_budget,
                                          tau_init=5.0, tau_end=250.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase1 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 GD: obj={obj_sgd:.8f}")

    # Symmetrize again
    best_seq = symmetrize(best_seq)
    best_seq = best_seq / best_seq.sum()
    obj_sym = compute_obj(best_seq)
    if obj_sym < best_score:
        best_score = obj_sym
        print(f"  After re-sym: obj={best_score:.8f} **")
    else:
        # Keep non-symmetric if better
        print(f"  Non-sym better: {compute_obj(best_seq * rng.uniform(0.99, 1.01, len(best_seq))):.8f}")

    # === Phase 2: Intensive LP Polishing with HTM ===
    print(f"\n=== Phase 2: LP Polishing (with HTM) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.4, 500.0)
    if p2_budget > 30:
        a_lp = lp_polish(best_seq, time_budget=p2_budget, verbose=True, seed=42, use_htm=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f}")

    # === Phase 3: Subgradient refinement ===
    print(f"\n=== Phase 3: Subgradient Refine ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.06, 60.0)
    if p3_budget > 15:
        a_sg = subgradient_refine(best_seq, time_budget=p3_budget, K=7)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Phase3 SG: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 SG: obj={obj_sg:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 150), max(400, n_current - 50),
        n_current + 50, n_current + 150, n_current + 300, n_current + 500,
        600, 800, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2500 and abs(n - n_current) > 40]

    for n_try in n_options[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)

        # Quick GD
        alloc1 = min(remaining * 0.1, 30.0)
        if alloc1 > 10:
            a_try = nesterov_smoothed_descent(a_try, time_budget=alloc1,
                                              tau_init=10.0, tau_end=200.0)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 80.0)
        if alloc2 > 15:
            a_try = lp_polish(a_try, time_budget=alloc2, seed=n_try, use_htm=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Gradient-directed perturbations ===
    print(f"\n=== Phase 5: Perturbation + Re-LP ===")
    remaining = deadline - time.monotonic()

    for r in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        # Use gradient direction for perturbation
        conv = compute_conv(best_seq / best_seq.sum())
        peak_idx = int(np.argmax(conv))
        n = len(best_seq)

        # Create a perturbation that reduces the peak
        scale = 0.008 + 0.005 * r
        noise = rng.normal(0, scale, n)
        # Boost positions that contribute to the peak
        i_start = max(0, peak_idx - (n - 1))
        i_end = min(n - 1, peak_idx)
        if i_start <= i_end:
            noise[i_start:i_end+1] -= 0.5 * scale * rng.uniform(0.5, 1.0)

        a_pert = best_seq * np.exp(noise)
        a_pert = np.maximum(a_pert, 0)
        s_pert = float(np.sum(a_pert))
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert

        alloc = min(remaining * 0.22, 80.0)
        if alloc > 15:
            a_pert = lp_polish(a_pert, time_budget=alloc, seed=200 + r, use_htm=False)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Perturb {r} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Perturb {r} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 6: Try different window shapes at current n ===
    print(f"\n=== Phase 6: Window Shape Exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for mode in [2, 3, 4, 5, 6]:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        a_win = diverse_init(n_cur, rng, mode=mode)
        a_win = symmetrize(a_win)
        a_win = a_win / a_win.sum()
        obj_win = compute_obj(a_win)

        # Quick LP polish
        alloc = min(remaining * 0.15, 50.0)
        if alloc > 10:
            a_win = lp_polish(a_win, time_budget=alloc, seed=mode * 100, use_htm=False)
            obj_win = compute_obj(a_win)

        if obj_win < best_score:
            best_score = obj_win
            best_seq = a_win.copy()
            print(f"  Window {mode}: obj={best_score:.8f} **")
        else:
            print(f"  Window {mode}: obj={obj_win:.8f}")

    # === Phase 7: Final intensive LP ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 150.0)
        a_final = lp_polish(best_seq, time_budget=alloc, verbose=True, seed=999, use_htm=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # Final subgradient
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 30.0)
        a_sg = subgradient_refine(best_seq, time_budget=alloc, K=10)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
