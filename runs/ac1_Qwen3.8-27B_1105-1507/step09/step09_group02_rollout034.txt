            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)

            # Precompute outside max (doesn't change with x')
            max_outside = 0.0
            if k_lo > 0:
                max_outside = max(max_outside, float(np.max(conv[:k_lo])))
            if k_hi < 2 * n - 2:
                max_outside = max(max_outside, float(np.max(conv[k_hi + 1:])))

            def obj_at(xp):
                """Evaluate objective at a[i]=xp, using precomputed conv."""
                new_sum = s + xp - x
                if new_sum < 1e-10:
                    return np.inf
                if k_lo > k_hi:
                    return 2.0 * n * float(np.max(conv)) / (new_sum * new_sum)
                # Max inside affected range
                # conv[k] + delta for k in [k_lo, k_hi]
                idx_lo = k_lo - i
                idx_hi = k_hi - i
                # Linear part: 2*(xp-x)*a[k-i] for all k in range
                affected = conv[k_lo:k_hi + 1] + 2.0 * (xp - x) * a[idx_lo:idx_hi + 1]
                # Fix k=2i: replace linear with quadratic
                if k_lo <= 2 * i <= k_hi:
                    pos = 2 * i - k_lo
                    affected[pos] -= 2.0 * (xp - x) * a[i]
                    affected[pos] += xp * xp - x * x
                max_inside = float(np.max(affected))
                max_conv_new = max(max_outside, max_inside)
                return 2.0 * n * max_conv_new / (new_sum * new_sum)

            # Determine search range
            mean_val = 1.0 / n
            upper = max(5.0 * max(x, 1e-6), 3.0 * mean_val, 0.1)
            lower = 0.0

            # Coarse grid to find good region
            grid = np.linspace(lower, upper, 20)
            grid_vals = [obj_at(gv) for gv in grid]
            best_idx = int(np.argmin(grid_vals))
            # Refine search around best grid point
            if best_idx == 0:
                lo, hi = 0.0, grid[1]
            elif best_idx == len(grid) - 1:
                lo, hi = grid[-2], upper
            else:
                lo, hi = grid[best_idx - 1], grid[best_idx + 1]

            # Golden section search
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = obj_at(c)
            fd = obj_at(d)
            for _ in range(25):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = obj_at(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = obj_at(d)

            x_opt = (lo + hi) / 2.0
            f_opt = obj_at(x_opt)

            # Compare with current and zero
            f_cur = obj_at(x)
            f_zero = obj_at(0.0)

            if f_opt <= f_cur and f_opt <= f_zero:
                x_new = x_opt
                f_new = f_opt
            elif f_zero < f_cur:
                x_new = 0.0
                f_new = f_zero
            else:
                continue  # No improvement

            if abs(x_new - x) < 1e-12:
                continue

            # Update a and conv
            delta = x_new - x
            old_ai = a[i]
            a[i] = x_new
            s += delta
            # Update conv
            if k_lo <= k_hi:
                idx_lo = k_lo - i
                idx_hi = k_hi - i
                # For k != 2i: delta_conv[k] = 2*delta*a[k-i] (a[k-i] unchanged)
                conv[k_lo:k_hi + 1] += 2.0 * delta * a[idx_lo:idx_hi + 1]
                # Fix k=2i: we added 2*delta*a[i]_new = 2*delta*x_new, but correct is x_new^2 - x^2
                if k_lo <= 2 * i <= k_hi:
                    conv[2 * i] -= 2.0 * delta * x_new
                    conv[2 * i] += x_new * x_new - x * x

            obj_new = 2.0 * n * float(np.max(conv)) / (s * s)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a.copy()
                any_improved = True

        if verbose and passes % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"  CD pass {passes}: obj={best_obj:.8f}, t={elapsed:.0f}/{time_budget:.0f}s")

        if not any_improved and passes > 3:
            if verbose:
                print(f"  CD converged at pass {passes}, obj={best_obj:.8f}")
            # Try a perturbation before giving up
            if time.monotonic() - start < time_budget * 0.8:
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                s = 1.0
                conv = compute_conv(a)
                passes = max(0, passes - 50)  # Reset convergence counter
            else:
                break

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget, verbose=False, seed=42):
    """LP polishing with adaptive blending and perturbation on stall."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a))) / (float(np.sum(a)) ** 2)
    start = time.monotonic()
    iter_count = 0
    last_t = 0.5
    rng = np.random.default_rng(seed)
    consecutive_fails = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Adaptive t candidates
        base_t = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        adaptive_t = [
            max(0.05, last_t * 0.5),
            max(0.05, last_t * 0.75),
            last_t,
            min(1.0, last_t * 1.3),
            min(1.0, last_t * 1.5),
        ]
        candidates = sorted(set([round(c, 3) for c in base_t + adaptive_t]))

        best_t_obj = best_obj
        best_t_val = last_t
        best_a_local = best_a.copy()

        for t in candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try))) / (float(np.sum(a_try)) ** 2)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        iter_count += 1
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            consecutive_fails = 0
            if verbose and iter_count % 10 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t_val:.3f}, M={M_lp:.6f}")
        else:
            consecutive_fails += 1
            scale = min(0.05, 0.01 + 0.003 * consecutive_fails)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            if consecutive_fails > 25:
                break

    return best_a


def block_descent(a, time_budget, block_size=5, verbose=False, seed=42):
    """
    Block coordinate descent: update groups of consecutive coordinates together.
    For each block, do sequential 1D optimization within the block (multiple sweeps).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    s = 1.0
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv)) / (s * s)
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    passes = 0

    while time.monotonic() - start < time_budget:
        passes += 1
        # Define blocks
        num_blocks = n // block_size
        remaining = n - num_blocks * block_size
        blocks = []
        for b in range(num_blocks):
            blocks.append(list(range(b * block_size, (b + 1) * block_size)))
        if remaining > 0:
            blocks.append(list(range(num_blocks * block_size, n)))

        rng.shuffle(blocks)
        any_improved = False

        for block in blocks:
            # Multiple sweeps within the block
            for sweep in range(3):
                for i in block:
                    x = a[i]
                    k_lo = max(0, i)
                    k_hi = min(2 * n - 2, i + n - 1)

                    max_outside = 0.0
                    if k_lo > 0:
                        max_outside = max(max_outside, float(np.max(conv[:k_lo])))
                    if k_hi < 2 * n - 2:
                        max_outside = max(max_outside, float(np.max(conv[k_hi + 1:])))

                    # Quick 1D optimization (fewer iterations for speed)
                    mean_val = 1.0 / n
                    upper = max(5.0 * max(x, 1e-6), 3.0 * mean_val, 0.05)

                    # Grid + golden section (15 iterations)
                    grid = np.linspace(0, upper, 10)
                    best_gv = x
                    best_gf = np.inf
                    for gv in grid:
                        new_sum = s + gv - x
                        if new_sum < 1e-10:
                            continue
                        idx_lo = k_lo - i
                        idx_hi = k_hi - i
                        aff = conv[k_lo:k_hi + 1] + 2.0 * (gv - x) * a[idx_lo:idx_hi + 1]
                        if k_lo <= 2 * i <= k_hi:
                            pos = 2 * i - k_lo
                            aff[pos] -= 2.0 * (gv - x) * a[i]
                            aff[pos] += gv * gv - x * x
                        mc = max(max_outside, float(np.max(aff)))
                        f = 2.0 * n * mc / (new_sum * new_sum)
                        if f < best_gf:
                            best_gf = f
                            best_gv = gv

                    # Refine with golden section around best
                    lo = max(0, best_gv - upper / 10)
                    hi = min(upper, best_gv + upper / 10)
                    if hi - lo < 1e-10:
                        x_new = best_gv
                    else:
                        c = hi - gr * (hi - lo)
                        d = lo + gr * (hi - lo)
                        # Evaluate
                        def eval_xp(xp):
                            new_sum = s + xp - x
                            if new_sum < 1e-10:
                                return np.inf
                            idx_lo = k_lo - i
                            idx_hi = k_hi - i
                            aff = conv[k_lo:k_hi + 1] + 2.0 * (xp - x) * a[idx_lo:idx_hi + 1]
                            if k_lo <= 2 * i <= k_hi:
                                pos = 2 * i - k_lo
                                aff[pos] -= 2.0 * (xp - x) * a[i]
                                aff[pos] += xp * xp - x * x
                            mc = max(max_outside, float(np.max(aff)))
                            return 2.0 * n * mc / (new_sum * new_sum)
                        fc = eval_xp(c)
                        fd = eval_xp(d)
                        for _ in range(15):
                            if fc < fd:
                                hi = d
                                d = c
                                fd = fc
                                c = hi - gr * (hi - lo)
                                fc = eval_xp(c)
                            else:
                                lo = c
                                c = d
                                fc = fd
                                d = lo + gr * (hi - lo)
                                fd = eval_xp(d)
                        x_new = (lo + hi) / 2.0

                    if abs(x_new - x) < 1e-12:
                        continue

                    delta = x_new - x
                    a[i] = x_new
                    s += delta
                    if k_lo <= k_hi:
                        idx_lo = k_lo - i
                        idx_hi = k_hi - i
                        conv[k_lo:k_hi + 1] += 2.0 * delta * a[idx_lo:idx_hi + 1]
                        if k_lo <= 2 * i <= k_hi:
                            conv[2 * i] -= 2.0 * delta * x_new
                            conv[2 * i] += x_new * x_new - x * x

                    obj_new = 2.0 * n * float(np.max(conv)) / (s * s)
                    if obj_new < best_obj - 1e-15:
                        best_obj = obj_new
                        best_a = a.copy()
                        any_improved = True

        if verbose and passes % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"  Block pass {passes}: obj={best_obj:.8f}, t={elapsed:.0f}s")

        if not any_improved:
            if time.monotonic() - start < time_budget * 0.7:
                a = best_a * np.exp(rng.normal(0, 0.015, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                s = 1.0
                conv = compute_conv(a)
            else:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = a / a.sum()
    elif mode == 3:
        # Two bumps
        x = np.arange(n) / n
        a = np.exp(-20 * (x - 0.3) ** 2) + np.exp(-20 * (x - 0.7) ** 2)
        a = a / a.sum()
    elif mode == 4:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1, 3)
        a = a / a.sum()
    elif mode == 5:
        # Comb pattern
        a = np.ones(n)
        a[::2] = 0.5
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Coordinate descent (primary) + LP polishing + block descent + multi-n.
    Time allocation: CD (35%), LP (25%), Block (15%), Multi-n (15%), Final (10%).
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Coordinate Descent (35%) ===
    print(f"\n=== Phase 1: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 500.0)
    if p1_budget > 30:
        a_cd = coordinate_descent(best_seq, time_budget=p1_budget, verbose=True, seed=seed)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase1 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 CD: obj={obj_cd:.8f}")

    # === Phase 2: LP Polishing (25%) ===
    print(f"\n=== Phase 2: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 350.0)
    if p2_budget > 50:
        num_rounds = 3
        per_round = p2_budget / num_rounds
        for rnd in range(num_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(per_round, remaining * 0.6)
            if alloc < 20:
                break
            a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True, seed=seed + rnd)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd+1}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd+1}: obj={obj_lp:.8f}")
                if rnd < num_rounds - 1:
                    best_seq = best_seq * np.exp(rng.normal(0, 0.012, len(best_seq)))
                    best_seq = np.maximum(best_seq, 0)
                    best_seq = best_seq / best_seq.sum()

    # === Phase 3: Block Descent (15%) ===
    print(f"\n=== Phase 3: Block Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 200.0)
    if p3_budget > 30:
        a_bd = block_descent(best_seq, time_budget=p3_budget, block_size=4, verbose=True, seed=seed)
        obj_bd = compute_obj(a_bd)
        if obj_bd < best_score:
            best_score = obj_bd
            best_seq = a_bd.copy()
            print(f"  Phase3 Block: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 Block: obj={obj_bd:.8f}")

    # === Phase 4: Multi-n Exploration (15%) ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    n_options = sorted(set([
        max(400, n_current - 150),
        n_current + 100,
        n_current + 200,
        n_current + 300,
        600, 700, 900, 1000, 1100
    ]))
    n_options = [nt for nt in n_options if 400 <= nt <= 1500 and abs(nt - n_current) > 50]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick CD polish
        alloc = min(remaining * 0.2, 60.0)
        if alloc > 15:
            a_try = coordinate_descent(a_try, time_budget=alloc, seed=seed + n_try)

        # Quick LP
        alloc = min(remaining * 0.15, 40.0)
        if alloc > 10:
            a_try = lp_polish(a_try, time_budget=alloc, seed=seed + n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Alternate diverse starts (10%) ===
    print(f"\n=== Phase 5: Diverse Starts ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        for mode in [2, 3, 1]:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            a_div = diverse_init(n_current, rng, mode)
            # Quick CD
            alloc = min(remaining * 0.25, 50.0)
            if alloc > 15:
                a_div = coordinate_descent(a_div, time_budget=alloc, seed=seed + 100 + mode)
            obj_div = compute_obj(a_div)
            if obj_div < best_score:
                best_score = obj_div
                best_seq = a_div.copy()
                print(f"  ** Diverse mode {mode}: obj={best_score:.8f} **")
            else:
                print(f"  Diverse mode {mode}: obj={obj_div:.8f}")

    # === Phase 6: Final Intensive Polish (remaining) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # CD
    if remaining > 30:
        alloc = min(remaining * 0.35, 100.0)
        a_cd = coordinate_descent(best_seq, time_budget=alloc, seed=seed + 500)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # LP
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.3, 80.0)
        a_lp = lp_polish(best_seq, time_budget=alloc, seed=seed + 600)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # CD again
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 60.0)
        a_cd = coordinate_descent(best_seq, time_budget=alloc, seed=seed + 700)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")

    # One last LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 40.0)
        a_lp = lp_polish(best_seq, time_budget=alloc, seed=seed + 800)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
