import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def solve_response_lp(a):
    """
    Solve: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0
    Returns g_optimal.
    """
    n = len(a)
    a = a / np.sum(a)  # normalize
    
    # Build A_ub matrix: (2n-1) x (n+1)
    # x = [g_0, ..., g_{n-1}, M]
    # Constraint k: sum_j A[k,j]*g_j - M <= 0
    # A_ub[k, j] = a[j] if max(0,k-n+1) <= j <= min(n-1,k), else 0
    # A_ub[k, n] = -1
    
    num_constraints = 2 * n - 1
    A_ub = np.zeros((num_constraints, n + 1))
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A_ub[k, j_start:j_end+1] = a[j_start:j_end+1]
        A_ub[k, n] = -1.0
    
    b_ub = np.zeros(num_constraints)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    # Bounds: g >= 0, M >= 0 (since all conv values are non-negative)
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        g = res.x[:n]
        g = np.maximum(g, 0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s
    return None

from scipy.optimize import linprog

    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)

    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

    Returns (g_optimal, M_optimal).

        n = len(a)

        return None, None

    cols = []

        gs = float(np.sum(g))
        if gs > 1e-15:
            return g / gs, float(res.x[n])
    
    return None, None

def proportioning_method(a, max_iters, time_budget, verbose=False):
    """
    Alternating LP: a -> best_response(a) -> best_response(g) -> ...
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Step 1: g = best_response(a)
        g, M1 = solve_lp_response(a)
        if g is None:
            break
        
        # Step 2: a_new = best_response(g)
        a_new, M2 = solve_lp_response(g)
        if a_new is None:
            break
        
        # Check improvement
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_a = a_new.copy()
            best_obj = obj_new
            a = a_new
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    Prop it{it}: obj={best_obj:.10f}, M1={M1:.8f}, M2={M2:.8f}, t={elapsed:.1f}s")
        else:
            # No improvement, try blending
            a_blend = 0.5 * a + 0.5 * a_new
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()
            obj_blend = compute_obj(a_blend)
            if obj_blend < best_obj - 1e-12:
                best_a = a_blend.copy()
                best_obj = obj_blend
                a = a_blend
            else:

                a = a * np.exp(np.random.default_rng(it).normal(0, 0.01, n))
                a = np.maximum(a, 0)
                a = a / a.sum()

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def smoothed_grad_step(a, tau, lr):
    """One step of smoothed gradient descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    # Gradient: g[i] = 2 * sum_k p[k] * a[k-i] (with appropriate bounds)
    # This is the convolution of p (reversed) with a
    p_rev = p[::-1]
    g = np.convolve(p_rev, a, mode='full')[:n]
    g = 2.0 * g
    
    # Mirror descent update (multiplicative)
    a_new = a * np.exp(-lr * g)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-15:
        return a
    a_new = a_new / s
    return a_new

def mirror_descent_polish(a, time_budget, tau=50.0, lr=0.01, verbose=False):
    """Polish with multiplicative mirror descent."""

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2 (scale-invariant)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


class ConvLPSolver:
    """Precomputes LP structure for the best-response problem."""

    def __init__(self, n):

            for j in range(j_start, j_end + 1):
                rows.append(k)
                cols.append(j)
            rows.append(k)
            cols.append(n)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)

        n = self.n

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 0:
        return np.ones(new_n) / new_n
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def mirror_descent_polish(a, time_budget, tau=80.0, lr_init=0.003, verbose=False):
    """
    Multiplicative mirror descent on the smoothed objective.
    F_tau(a) = (1/tau) * log(sum_k exp(tau * (a*a)[k]))
    Update: a[i] *= exp(-lr * grad[i]), then normalize.
    """
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    s = float(np.sum(a))
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        # Gradient of F_tau: grad[i] = 2 * sum_k p[k] * a[k-i] (cross-correlation)
        p_rev = p[::-1]
        g_full = np.convolve(p_rev, a, mode='full')
        g = 2.0 * g_full[:n]

        a_new = a * np.exp(-lr * g)
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            it += 1
            if verbose and it % 200 == 0:
                print(f"    MD it{it}: obj={best_obj:.10f}, lr={lr:.6f}, t={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            lr *= 0.95
            if stall > 200:
                break

    return best_a


def subgradient_step(a, K=10):
    """One subgradient step targeting the top-K peaks."""
    n = len(a)
    a = np.maximum(a, 0.0)
    a = a / a.sum()

    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:K]
    max_val = float(conv[top_idx[0]])
    weights = np.exp((conv[top_idx] - max_val) * 2.0 / (max_val + 1e-15))
    weights /= weights.sum()

    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

    g /= K
    return g


def subgradient_polish(a, time_budget, verbose=False):
    """Subgradient method with momentum and adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.5 / np.sqrt(n)
    momentum = 0.6
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        g = subgradient_step(a)
        v = momentum * v + g
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            it += 1
            if verbose and it % 100 == 0:
                print(f"    SG it{it}: obj={best_obj:.10f}, t={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 50:
                lr *= 0.8
                v *= 0.5
                stall = 0
                if lr < 1e-8:
                    break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Proportioning method (alternating LPs) + multi-n + polishing.
    Time allocation: Proportioning (45%), Multi-n (25%), Final polish (30%).
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0.0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_parent = parent.size
                print(f"Parent (n={n_parent}): obj={obj:.10f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if best_seq is None:
        best_seq = np.ones(n_parent) / n_parent
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_parent}): obj={best_score:.10f}")

    # === Phase 1: Intensive Proportioning (45%) ===
    print(f"\n=== Phase 1: Proportioning Method ===")
    n_curr = len(best_seq)
    solver = ConvLPSolver(n_curr)

    num_rounds = 6
    p1_budget = (deadline - time.monotonic()) * 0.45
    per_round = p1_budget / num_rounds

    for rnd in range(num_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(per_round, remaining)
        if alloc < 10:
            break

        a_res = proportioning_method(best_seq, solver, time_budget=alloc, verbose=(rnd == 0))
        obj_res = compute_obj(a_res)
        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            print(f"  Prop round {rnd+1}: obj={best_score:.10f} **")
        else:
            print(f"  Prop round {rnd+1}: obj={obj_res:.10f}")
            # Perturb for next round
            scale = 0.003 * (rnd + 1)
            best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
            best_seq = np.maximum(best_seq, 0.0)
            best_seq = best_seq / best_seq.sum()

    # === Phase 2: Multi-n Exploration (25%) ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.25
    n_base = len(best_seq)

    n_options = sorted(set([
        max(400, n_base - 200),
        max(400, n_base - 100),
        n_base + 100,
        n_base + 200,
        n_base + 400,
        1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_base) >= 50]

    tried = 0
    max_tries = 5
    per_n = p2_budget / max_tries

    for n_try in n_options:
        if tried >= max_tries:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(per_n, remaining * 0.6)
        if alloc < 10:
            tried += 1
            continue

        solver_try = ConvLPSolver(n_try)
        a_try = proportioning_method(a_try, solver_try, time_budget=alloc)
        obj_try = compute_obj(a_try)
        tried += 1

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 3: Final Polish (30%) ===
    print(f"\n=== Phase 3: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)
    solver_final = ConvLPSolver(n_final)

    # Proportioning
    if remaining > 30:
        alloc = remaining * 0.35
        a_res = proportioning_method(best_seq, solver_final, time_budget=alloc)
        obj_res = compute_obj(a_res)
        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            print(f"  Final Prop: obj={best_score:.10f} **")
        else:
            print(f"  Final Prop: obj={obj_res:.10f}")

    # Mirror descent
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = remaining * 0.25
        a_res = mirror_descent_polish(best_seq, time_budget=alloc, tau=100.0, lr_init=0.002)
        obj_res = compute_obj(a_res)
        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            print(f"  Final MD: obj={best_score:.10f} **")
        else:
            print(f"  Final MD: obj={obj_res:.10f}")

    # Subgradient
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = remaining * 0.2
        a_res = subgradient_polish(best_seq, time_budget=alloc)
        obj_res = compute_obj(a_res)
        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            print(f"  Final SG: obj={best_score:.10f} **")
        else:
            print(f"  Final SG: obj={obj_res:.10f}")

    # One last proportioning round
    remaining = deadline - time.monotonic()
    if remaining > 15:
        n_final = len(best_seq)
        solver_last = ConvLPSolver(n_final)
        alloc = remaining * 0.5
        a_res = proportioning_method(best_seq, solver_last, time_budget=alloc)
        obj_res = compute_obj(a_res)
        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            print(f"  Last Prop: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
