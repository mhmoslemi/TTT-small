We can bisect on delta using the sign of the subgradient. This converges in O(log(precision/range)) iterations, with each iteration being O(n). 3. **Newton-like method**: Use the second derivative (which is 2 if the argmax is at position 2k, else 0) to do a Newton step. But the function is not twice differentiable, so this is tricky.

4. **Reduced line search**: Just do 10-15 bisection iterations instead of 50. This gives ~10-15 bits of precision, which might be enough.

Let me go with option 4: reduce the line search to 20 iterations of bisection. This should be about 2.5x faster than the previous golden section with 50 iterations.

So one coordinate update is O(50n). With n=800, that's 40,000 operations per coordinate, and 800 coordinates per sweep, so 32M operations per sweep. **The LP finds the best "direction" g to move toward, given the current linearization at a. After moving a small step toward g, the linearization changes, so we should re-solve the LP.**

1. First, find the optimal t via a 1D line search (since we have the LP solution g).
2. Then, take that step and re-solve the LP.

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a multi-phase approach: fast coordinate descent, aggressive LP polishing
with adaptive step sizes, subgradient refinement, and final fine-tuning.
Key improvements over previous: smaller LP steps with re-linearization,
faster CD with bisection line search, and a final "surgery" phase.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_coord_descent(a, time_budget=100.0, start_time=None, rng=None, verbose=False):
    """
    Fast coordinate descent with bisection line search (20 iterations per coord).
    Normalizes a to sum=1, optimizes max(conv(a,a)).
    """
    if start_time is None:
        start_time = time.monotonic()
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S
    conv = compute_conv(a)
    best_obj = float(np.max(conv))
    best_a = a.copy()
    stagnant = 0
    sweep_count = 0

    while time.monotonic() - start_time < time_budget:
        obj_before = float(np.max(conv))
        if rng is not None and sweep_count % 3 == 2:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for k in order:
            if time.monotonic() - start_time > time_budget:
                break
            # Direction: d[m] = a[m-k] for m in [k, k+n-1]
            d = np.zeros(2 * n - 1)
            d[k:k + n] = a

            lo = -a[k]
            hi = max(4.0 / n, lo + 2.0 / n)

            # Bisection line search: 20 iterations
            def obj_val(delta):
                val = conv + 2.0 * delta * d
                val[2 * k] += delta * delta
                return float(np.max(val))

            lo_d = lo
            hi_d = hi
            for _ in range(20):
                mid = 0.5 * (lo_d + hi_d)
                # Subgradient sign test
                # f(delta) is convex in delta (max of linear + one quadratic)
                # Approximate: evaluate at mid and slightly above/below
                fm = obj_val(mid)
                fp = obj_val(min(mid + 0.01 * (hi_d - lo_d + 1e-10), hi_d))
                if fp > fm:
                    hi_d = mid
                else:
                    lo_d = mid

            delta_opt = 0.5 * (lo_d + hi_d)
            new_obj = obj_val(delta_opt)

            if new_obj >= obj_before - 1e-15:
                continue

            # Apply update
            a[k] += delta_opt
            a[k] = max(0.0, a[k])
            conv += 2.0 * delta_opt * d
            conv[2 * k] += delta_opt * delta_opt

            # Re-normalize
            S_new = float(np.sum(a))
            if S_new < 1e-10:
                a[:] = 1.0 / n
                conv = compute_conv(a)
            else:
                scale = 1.0 / S_new
                a *= scale
                conv *= scale * scale

        sweep_count += 1
        obj_after = float(np.max(conv))
        if obj_after < best_obj - 1e-13:
            best_obj = obj_after
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 4:
                if rng is not None:
                    scale = 0.01 + 0.005 * min(stagnant, 10)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    ss = float(np.sum(a))
                    if ss < 1e-10:
                        a = best_a.copy()
                        conv = compute_conv(a)
                    else:
                        a = a / ss
                        conv = compute_conv(a)
                    best_obj = float(np.max(conv))
                    stagnant = 0
                else:
                    break

        if verbose and sweep_count % 20 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    CD sweep {sweep_count}: obj={2.0*n*best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_for_direction(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish_aggressive(a, time_budget=200.0, verbose=False, rng=None):
    """
    Aggressive LP polishing with small adaptive steps and re-linearization.
    Each iteration: solve LP for direction, find optimal t via line search, update.
    """
    if rng is None:
        rng = np.random.default_rng(12345)
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stagnant = 0
    last_t = 0.3

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp_for_direction(a)
        if g_lp is None:
            break

        # 1D line search on t in [0, 1] to minimize obj((1-t)*a + t*g_lp)
        best_t = 0.0
        best_t_obj = best_obj
        best_a_local = best_a.copy()

        # Try a range of t values, focused around last_t
        t_candidates = []
        for base in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            t_candidates.append(base)
        t_candidates.append(max(0.01, last_t * 0.5))
        t_candidates.append(max(0.01, last_t * 0.8))
        t_candidates.append(min(1.0, last_t * 1.5))
        t_candidates.append(min(1.0, last_t * 2.0))
        t_candidates = sorted(set(t_candidates))

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = max(0.01, min(best_t, 1.0))
            stagnant = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={last_t:.3f}")
        else:
            stagnant += 1
            if stagnant > 15:
                # Perturb to escape
                scale = 0.01 + 0.005 * min(stagnant, 15)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                ss = float(np.sum(a))
                if ss < 1e-10:
                    a = best_a.copy()
                else:
                    a = a / ss
                stagnant = 0
                last_t = 0.3
            elif it > 100:
                break

    return best_a


def subgradient_polish(a, time_budget=100.0, verbose=False, rng=None):
    """
    Fast subgradient method with momentum on the max of self-convolution.
    Uses the subgradient of max_m conv(a,a)[m] for direct computation.
    """
    if rng is None:
        rng = np.random.default_rng(777)
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    beta = 0.9
    lr = 1.0 / np.sqrt(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient: s[j] = 2*a[k_star - j] if 0 <= k_star - j < n
        s_grad = np.zeros(n)
        j_min = max(0, k_star - (n - 1))
        j_max = min(n - 1, k_star)
        for j in range(j_min, j_max + 1):
            s_grad[j] = 2.0 * a[k_star - j]

        # Also consider the "reflected" part for k_star >= n
        if k_star >= n:
            j_min2 = max(0, k_star - (n - 1) - (n - 1))
            j_max2 = min(n - 1, k_star - n + n - 1)
            # Actually this is the same formula, just different range
            # Let me recompute: s[j] = 2*a[k_star-j] for valid j where 0<=k_star-j<n
            # This means j ranges from k_star-(n-1) to k_star, clipped to [0, n-1]

        a = a - lr * s_grad
        a = np.maximum(a, 0)
        s_a = float(np.sum(a))
        if s_a < 1e-10:
            a = best_a.copy()
            v = np.zeros(n)
        else:
            a = a / s_a

        # Momentum (simple heavy ball)
        v = beta * v - lr * s_grad / max(1.0, float(np.max(np.abs(s_grad))))
        a = a + 0.1 * v
        a = np.maximum(a, 0)
        s_a = float(np.sum(a))
        if s_a < 1e-10:
            a = best_a.copy()
            v = np.zeros(n)
        else:
            a = a / s_a

        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            best_it = it

        it += 1
        lr = lr * 0.9999  # Slow decay

        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 1.0 / np.sqrt(n)
            best_it = it

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    SG it{it}: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def smoothed_max_gradient(a, tau):
    """Compute gradient of smoothed max of conv(a,a) using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    p = exp_vals / float(np.sum(exp_vals))

    # Gradient of sum_k p[k] * conv(a,a)[k] w.r.t. a
    # = 2 * (p * a) convolved, restricted to first n
    # Actually: d/d(a[j]) sum_k p[k]*conv(a,a)[k] = 2 * sum_k p[k]*a[k-j] (valid terms)
    # = 2 * (p_full * a) full conv, first n elements
    # where p_full is p embedded in a (2n-1) array

    from scipy.fft import fft, ifft
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    grad = 2.0 * g_full[:n]

    return grad


def nesterov_smoothed(a, time_budget=100.0, tau_init=20.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient on smoothed max-conv objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        y = np.maximum(a + beta * v, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        grad = smoothed_max_gradient(y, tau)
        # Normalize gradient
        g_norm = float(np.linalg.norm(grad))
        if g_norm > 0:
            grad = grad / g_norm

        v = beta * v - lr * grad
        a_new = np.maximum(a + v, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
            lr *= 0.5
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr *= 0.99995

        if it - best_it > 10000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)
            best_it = it

        if verbose and it % 3000 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def final_surgery(a, time_budget=60.0, rng=None):
    """
    Final fine-tuning: try tiny perturbations on individual coordinates
    and small pairs to squeeze out last improvements.
    """
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)

    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Try single-coordinate perturbations
    scales = [1e-5, 5e-5, 1e-4, 5e-4, 1e-3]
    for scale in scales:
        if time.monotonic() - start > time_budget:
            break
        for _ in range(n // 2):
            if time.monotonic() - start > time_budget:
                break
            k = rng.integers(0, n)
            for delta_sign in [1, -1]:
                a_try = best_a.copy()
                a_try[k] += delta_sign * scale
                if a_try[k] < 0:
                    continue
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()

    # Try coordinate pairs (transfer mass)
    for _ in range(200):
        if time.monotonic() - start > time_budget:
            break
        k1 = rng.integers(0, n)
        k2 = rng.integers(0, n)
        if k1 == k2:
            continue
        for scale in [1e-5, 5e-5, 1e-4]:
            a_try = best_a.copy()
            a_try[k1] -= scale
            a_try[k2] += scale
            if a_try[k1] < 0 or a_try[k2] < 0:
                continue
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()

    return best_a


def symmetrize(a):
    """Enforce palindrome symmetry: a[i] = a[n-1-i]."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    return a_sym


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 0.01))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1.5, 1.5)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.3 + 0.7 * np.sin(x) ** 2
        a = a / a.sum()
    elif mode == 7:
        c1 = rng.integers(n // 6, n // 3)
        c2 = n - 1 - c1
        w = max(1, n // 10)
        a = np.full(n, 0.01)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 8:
        # Triangle-like
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * x - 1.0)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 9:
        # Cosine window raised
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x) ** 2
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Initial CD refinement
    Phase 2: Aggressive LP polishing (main workhorse)
    Phase 3: Subgradient + smoothed gradient refinement
    Phase 4: Fine CD
    Phase 5: Final LP
    Phase 6: Multi-length exploration
    Phase 7: Final surgery
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    # Phase 0: Load parent
    best_seq = None
    best_score = np.inf
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")

    n_cur = len(best_seq)

    # === Phase 1: Initial Coordinate Descent ===
    print(f"\n=== Phase 1: Initial CD ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.12, 120.0)
    if p1_budget > 20:
        a_cd = fast_coord_descent(best_seq, time_budget=p1_budget,
                                  start_time=time.monotonic(), rng=rng, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase1 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 CD: obj={obj_cd:.8f}")

    # === Phase 2: Aggressive LP Polishing ===
    print(f"\n=== Phase 2: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 400.0)
    if p2_budget > 30:
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = p2_budget / (3 - rnd) * 0.8
            alloc = min(alloc, remaining * 0.7)
            if alloc < 20:
                break
            a_lp = lp_polish_aggressive(best_seq, time_budget=alloc, verbose=True, rng=rng)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
                # Perturb before next round
                scale = 0.01 * (rnd + 1)
                best_seq = best_seq * np.exp(rng.normal(0, scale, n_cur))
                best_seq = np.maximum(best_seq, 0)
                best_seq = best_seq / best_seq.sum()

    # === Phase 3: Subgradient + Smoothed Gradient ===
    print(f"\n=== Phase 3: Gradient Methods ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 200.0)
    if p3_budget > 30:
        # Subgradient first (faster)
        sg_budget = p3_budget * 0.4
        if sg_budget > 20:
            a_sg = subgradient_polish(best_seq, time_budget=sg_budget, verbose=True, rng=rng)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  Subgrad: obj={best_score:.8f} **")
            else:
                print(f"  Subgrad: obj={obj_sg:.8f}")

        # Nesterov smoothed
        remaining = deadline - time.monotonic()
        nsd_budget = min(remaining * 0.15, 150.0)
        if nsd_budget > 20:
            a_nsd = nesterov_smoothed(best_seq, time_budget=nsd_budget,
                                      tau_init=30.0, tau_end=800.0, verbose=True)
            obj_nsd = compute_obj(a_nsd)
            if obj_nsd < best_score:
                best_score = obj_nsd
                best_seq = a_nsd.copy()
                print(f"  NSD: obj={best_score:.8f} **")
            else:
                print(f"  NSD: obj={obj_nsd:.8f}")

    # === Phase 4: Fine Coordinate Descent ===
    print(f"\n=== Phase 4: Fine CD ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 150.0)
    if p4_budget > 20:
        a_cd2 = fast_coord_descent(best_seq, time_budget=p4_budget,
                                   start_time=time.monotonic(), rng=rng, verbose=True)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  Fine CD: obj={best_score:.8f} **")
        else:
            print(f"  Fine CD: obj={obj_cd2:.8f}")

    # === Phase 5: Final LP ===
    print(f"\n=== Phase 5: Final LP ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.2, 180.0)
    if p5_budget > 20:
        a_lp_f = lp_polish_aggressive(best_seq, time_budget=p5_budget, verbose=True, rng=rng)
        obj_lp_f = compute_obj(a_lp_f)
        if obj_lp_f < best_score:
            best_score = obj_lp_f
            best_seq = a_lp_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_lp_f:.8f}")

    # === Phase 6: Multi-length Exploration ===
    print(f"\n=== Phase 6: Multi-length ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    n_options = sorted(set([
        max(400, n_best - 100), n_best + 100, n_best + 200,
        600, 900, 1100, 1400
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1600 and abs(n - n_best) >= 50]

    for n_try in n_options[:3]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick CD
        alloc1 = min(remaining * 0.3, 50.0)
        if alloc1 > 15:
            a_try = fast_coord_descent(a_try, time_budget=alloc1,
                                       start_time=time.monotonic(), rng=rng)

        # Quick LP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.2, 40.0)
        if alloc2 > 10:
            a_try = lp_polish_aggressive(a_try, time_budget=alloc2, rng=rng)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse restarts (if time) ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        mode = (3 + r * 2) % 10
        a_r = diverse_init(n_final, rng, mode=mode)
        blend = 0.4 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc1 = min(remaining * 0.3, 40.0)
        if alloc1 > 10:
            a_r = fast_coord_descent(a_r, time_budget=alloc1,
                                     start_time=time.monotonic(), rng=rng)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 30.0)
        if alloc2 > 10:
            a_r = lp_polish_aggressive(a_r, time_budget=alloc2, rng=rng)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Final Surgery ===
    print(f"\n=== Phase 8: Final Surgery ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_surg = final_surgery(best_seq, time_budget=min(remaining * 0.5, 60.0), rng=rng)
        obj_surg = compute_obj(a_surg)
        if obj_surg < best_score:
            best_score = obj_surg
            best_seq = a_surg.copy()
            print(f"  Surgery: obj={best_score:.8f} **")
        else:
            print(f"  Surgery: obj={obj_surg:.8f}")

    # One more CD round if time
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_final = fast_coord_descent(best_seq, time_budget=min(remaining * 0.6, 50.0),
                                     start_time=time.monotonic(), rng=rng)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
