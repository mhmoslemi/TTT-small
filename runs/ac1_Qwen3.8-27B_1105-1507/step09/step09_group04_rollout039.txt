"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive mirror descent (exponential gradient) with FFT-accelerated gradients,
active-set refinement, block coordinate descent, LP polishing, and frequency-domain
restarts. Key innovations: entropy-regularized mirror descent on the simplex with
Barzilai-Borwein step sizes, dynamic tau scheduling, and structured frequency-domain
perturbations for escaping local minima.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax_weights(conv, tau):
    """Compute softmax weights for log-sum-exp smoothing."""
    max_c = np.max(conv)
    exp_vals = np.exp(tau * (conv - max_c))
    return exp_vals / np.sum(exp_vals)


def gradient_smoothed(a, p):
    """
    Compute gradient of 2n * sum_k p_k * c_k w.r.t. a (with sum(a)=1).
    p is softmax weights (length 2n-1). Returns gradient of length n.
    """
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    p_rev = p[::-1]
    fa = fft(a, N)
    fp = fft(p_rev, N)
    conv_ap = ifft(fa * fp).real[:2 * n - 1]
    r = conv_ap[2 * n - 2::-1][:n]
    return 4.0 * n * r


def gradient_active_set(a, conv, threshold=1e-6):
    """
    Compute exact subgradient at the active set (positions near max).
    Returns (gradient, active_set_size).
    """
    n = len(a)
    max_c = float(np.max(conv))
    active = conv >= (max_c - threshold)
    num_active = int(np.sum(active))
    if num_active == 0:
        active = np.zeros(len(conv), dtype=bool)
        active[int(np.argmax(conv))] = True
        num_active = 1
    p = np.zeros(len(conv))
    p[active] = 1.0 / num_active
    return gradient_smoothed(a, p), num_active


def mirror_descent(a, time_budget, tau_start=5.0, tau_end=300.0,
                   use_active=False, verbose=True, rng=None):
    """
    Mirror descent (exponential gradient) on the simplex with adaptive tau and BB steps.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = float(np.max(conv))
    start = time.monotonic()
    lr = 3.0 / np.sqrt(n)

    it = 0
    prev_a = a.copy()
    prev_grad = None
    stagnant = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.6)

        if use_active:
            grad, num_act = gradient_active_set(a, conv, threshold=max_c * 1e-4)
        else:
            p = softmax_weights(conv, tau)
            grad = gradient_smoothed(a, p)

        # Barzilai-Borwein step size
        if prev_grad is not None and it % 3 == 0:
            s_vec = a - prev_a
            y_vec = grad - prev_grad
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / sy
                if 0 < bb_lr < 50.0 / np.sqrt(n):
                    lr = 0.4 * lr + 0.6 * bb_lr

        scale = lr * grad
        scale -= np.mean(scale)  # Center for numerical stability
        exp_grad = np.exp(-scale)
        a_new = a * exp_grad
        a_new = np.maximum(a_new, 1e-30)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        # Evaluate
        conv_new = compute_conv(a_new)
        obj_new = float(np.max(conv_new))
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stagnant = 0
        else:
            stagnant += 1

                    scale_p = 0.005 * min(1.0 + stagnant / 8000, 5.0)

                    conv = compute_conv(a)
                    best_obj = float(np.max(conv))

                a_new = a  # Revert

        conv = conv_new
        prev_a = a.copy()
        prev_grad = grad.copy()
        it += 1
        lr = max(1e-10, lr * 0.99995)

        if verbose and it % 10000 == 0:
            print(f"    MD it{it}: obj={2*n*best_obj:.8f}, tau={tau:.1f}, lr={lr:.4f}, t={elapsed:.0f}s")

def block_coord_descent(a, time_budget, block_size=4, verbose=True, rng=None):
    """
    Block coordinate descent: optimize blocks of 'block_size' coordinates jointly
    using golden section search on the block's contribution to the max convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = float(np.max(conv))
    start = time.monotonic()
    gr = (np.sqrt(5) - 1) / 2.0
    stagnant = 0

    while time.monotonic() - start < time_budget:
        if stagnant > 20:
            if rng is not None:
                scale = 0.01 * (1 + stagnant * 0.3)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                conv = compute_conv(a)
                best_obj = float(np.max(conv))
                stagnant = 0
            else:
                break

        n_blocks = n // block_size
        if n_blocks == 0:
            break
        if rng is not None:
            block_indices = rng.choice(n_blocks, size=min(n_blocks, 200), replace=False)
        else:
            block_indices = np.arange(n_blocks)

        for bi in block_indices:
            start_idx = bi * block_size
            end_idx = start_idx + block_size
            indices = np.arange(start_idx, end_idx)
            block = a[indices].copy()

            # Build the effect of the block on the convolution
            # For each position m in the convolution, the contribution from the block is:
            # sum_{i in block} a[m-i] * block[i] + block[i] * a[m-i] (cross terms)
            # + sum_{i,j in block, i+j=m-start_idx*2} block[i]*block[j] (self terms)
            # This is complex, so use a simpler approach:
            # Parameterize the block as block * exp(eta * direction), then 1D search on eta

            # Direction: reduce contribution to the max positions
            max_pos = int(np.argmax(conv))
            direction = np.zeros(block_size)
            for di, idx in enumerate(indices):
                # Contribution of a[idx] to conv[max_pos] is a[max_pos - idx] (if in range)
                contrib = 0.0
                k = max_pos - idx
                if 0 <= k < n:
                    contrib = a[k]
                k2 = max_pos - idx  # symmetric
                if 0 <= k2 < n:
                    contrib += a[k2]
                direction[di] = -contrib

            # Also consider self-contribution at 2*idx
            for di, idx in enumerate(indices):
                k = 2 * idx
                if 0 <= k < 2 * n - 1:
                    direction[di] -= a[idx] * 0.5

            # Normalize direction
            d_norm = np.linalg.norm(direction)
            if d_norm < 1e-15:
                direction = rng.standard_normal(block_size) if rng is not None else np.ones(block_size)
                d_norm = np.linalg.norm(direction)
            direction /= d_norm

            # Golden section search on eta: new_block = block * exp(eta * direction / d_norm)
            # Or simpler: new_block = max(0, block + eta * direction)
            lo = -0.5 * np.min(block / (np.maximum(np.abs(direction), 1e-10)))
            hi = 0.5 / n

            if lo >= hi:
                continue

            a_c = lo
            b_c = hi
            c_c = b_c - gr * (b_c - a_c)
            d_c = a_c + gr * (b_c - a_c)

            def eval_eta(eta):
                new_block = np.maximum(0.0, block + eta * direction)
                sb = float(np.sum(new_block))
                if sb < 1e-15:
                    return float(np.max(conv))  # No improvement possible
                # Scale factor: new_sum = s - sum(block) + sum(new_block)
                # For simplicity, just evaluate the max change
                # Approximate: the change in conv at position max_pos
                # Delta at max_pos ≈ 2 * sum_i direction[i] * a[max_pos - indices[i]] * eta
                # This is approximate; for exact, we'd need full conv
                # Use a cheaper estimate
                return 0.0

            # For efficiency, do a limited search
            best_eta = 0.0
            best_block_obj = float(np.max(conv))

            for trial in range(8):
                eta = lo + (hi - lo) * (trial + 1) / 9.0
                new_block = np.maximum(0.0, block + eta * direction)
                a_try = a.copy()
                a_try[indices] = new_block
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                conv_try = compute_conv(a_try)
                obj_try = float(np.max(conv_try))
                if obj_try < best_block_obj:
                    best_block_obj = obj_try
                    best_eta = eta

            if best_eta != 0.0:
                a[indices] = np.maximum(0.0, block + best_eta * direction)
                s_new = float(np.sum(a))
                a = a / s_new
                conv = compute_conv(a)
                if float(np.max(conv)) < best_obj - 1e-14:
                    best_obj = float(np.max(conv))
                    best_a = a.copy()
                    stagnant = 0
                else:
                    a = best_a.copy()
                    conv = compute_conv(a)
                stagnant += 1

    return best_a


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv, float(M.value)
        except Exception:
            continue
    return None, None

        ]))

        best_t_obj = best_obj
        best_t_val = last_t
        best_a_local = best_a.copy()

        for t in candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        iter_count += 1
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            if verbose and iter_count % 3 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t_val:.2f}")
        else:
            scale = 0.015 + 0.005 * min(iter_count, 15)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            if iter_count > 50:
                break

    return best_a


def lbgfs_smoothed(a, time_budget=60.0, verbose=True):
    """L-BFGS on the log-sum-exp smoothed objective with simplex projection."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    sv = float(np.sum(a))
    if sv < 1e-10:
        a = np.ones(n) / n
    a = a / sv

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau = 50.0

    def obj_smoothed(x):
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx < 1e-10:
            return 1e6
        x = x / sx
        conv = compute_conv(x)
        max_c = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_c))
        logsumexp = max_c + np.log(np.sum(exp_vals)) / tau
        return 2.0 * n * logsumexp

    def grad_smoothed(x):
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx < 1e-10:
            return np.zeros(n)
        x = x / sx
        conv = compute_conv(x)
        p = softmax_weights(conv, tau)
        g = gradient_smoothed(x, p)
        # Subtract mean for simplex (projection handles it approximately)
        return g - np.mean(g)

    x0 = a.copy()
    result = minimize(obj_smoothed, x0, jac=grad_smoothed, method='L-BFGS-B',
                      options={'maxiter': 2000, 'ftol': 1e-14, 'gtol': 1e-10,
                               'iprint': -1, 'maxfun': 5000})

    if result.x is not None:
        a_new = np.maximum(result.x, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    L-BFGS: obj={best_obj:.8f}")

    return best_a


def freq_domain_perturb(a, rng, strength=0.02):
    """Perturb in frequency domain to escape local minima."""
    n = len(a)
    N = 1
    while N < 2 * n:
        N <<= 1
    A = fft(a, N)
    # Add random phase perturbation at higher frequencies
    mask = np.ones(N)
    cutoff = n // 4
    mask[cutoff:N - cutoff] = 0.0
    # Perturb the magnitudes
    perturbation = A * mask
    perturbation += A * (1 - mask) * (1 + strength * rng.standard_normal(N) * 0.1)
    a_new = ifft(perturbation).real[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Multi-phase optimization combining mirror descent,
    block coordinate descent, L-BFGS, LP polishing, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_target = 800
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    n_cur = len(best_seq)

    # === Phase 1: Mirror Descent (main workhorse) ===
    print(f"\n=== Phase 1: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 400.0)
    if p1_budget > 30:
        a_md, _ = mirror_descent(best_seq, time_budget=p1_budget,
                                 tau_start=5.0, tau_end=200.0,
                                 verbose=True, rng=rng)
        obj_md = compute_obj(a_md)
        if obj_md < best_score - 1e-10:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Phase1 MD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 MD: obj={obj_md:.8f}")

    # === Phase 2: Active-set Mirror Descent ===
    print(f"\n=== Phase 2: Active-Set MD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.20, 250.0)
    if p2_budget > 30:
        a_md2, _ = mirror_descent(best_seq, time_budget=p2_budget,
                                  tau_start=50.0, tau_end=500.0,
                                  use_active=True, verbose=True, rng=rng)
        obj_md2 = compute_obj(a_md2)
        if obj_md2 < best_score - 1e-10:
            best_score = obj_md2
            best_seq = a_md2.copy()
            print(f"  Phase2 Active-MD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 Active-MD: obj={obj_md2:.8f}")

    # === Phase 3: LP Polishing ===
    print(f"\n=== Phase 3: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.20, 250.0)
    if p3_budget > 30:
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(p3_budget / (3 - rnd), remaining * 0.7)
            if alloc < 20:
                break
            a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True, rng=rng)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score - 1e-10:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
                if rnd < 2:
                    scale = 0.015 * (rnd + 1)
                    best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                    best_seq = np.maximum(best_seq, 1e-10)
                    best_seq = best_seq / best_seq.sum()

    # === Phase 4: Block Coordinate Descent ===
    print(f"\n=== Phase 4: Block CD ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 150.0)
    if p4_budget > 30:
        a_bcd = block_coord_descent(best_seq, time_budget=p4_budget,
                                    block_size=4, verbose=True, rng=rng)
        obj_bcd = compute_obj(a_bcd)
        if obj_bcd < best_score - 1e-10:
            best_score = obj_bcd
            best_seq = a_bcd.copy()
            print(f"  Phase4 BlockCD: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 BlockCD: obj={obj_bcd:.8f}")

    # === Phase 5: L-BFGS Polish ===
    print(f"\n=== Phase 5: L-BFGS ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.10, 100.0)
    if p5_budget > 20:
        a_lb = lbgfs_smoothed(best_seq, time_budget=p5_budget, verbose=True)
        obj_lb = compute_obj(a_lb)
        if obj_lb < best_score - 1e-10:
            best_score = obj_lb
            best_seq = a_lb.copy()
            print(f"  Phase5 L-BFGS: obj={best_score:.8f} **")
        else:
            print(f"  Phase5 L-BFGS: obj={obj_lb:.8f}")

    # === Phase 6: Multi-length Exploration ===
    print(f"\n=== Phase 6: Multi-length ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(400, n_best - 100), max(400, n_best - 30),
        n_best + 30, n_best + 100, n_best + 200,
        600, 700, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500 and abs(n - n_best) > 20]

    for n_try in n_options[:3]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick mirror descent
        alloc1 = min(remaining * 0.4, 60.0)
        if alloc1 > 15:
            a_try, _ = mirror_descent(a_try, time_budget=alloc1,
                                      tau_start=10.0, tau_end=150.0,
                                      verbose=False, rng=rng)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.25, 40.0)
        if alloc2 > 15:
            a_try = lp_polish(a_try, time_budget=alloc2, verbose=False, rng=rng)

        obj_try = compute_obj(a_try)
        if obj_try < best_score - 1e-10:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse Restarts with Frequency Perturbation ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        mode = (r + 2) % 8
        a_r = diverse_init(n_final, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 1e-10)
        a_r = a_r / a_r.sum()

        # Frequency perturbation
        a_r = freq_domain_perturb(a_r, rng, strength=0.03)

        # Mirror descent
        alloc1 = min(remaining * 0.35, 60.0)
        if alloc1 > 15:
            a_r, _ = mirror_descent(a_r, time_budget=alloc1,
                                    tau_start=10.0, tau_end=200.0,
                                    verbose=False, rng=rng)

        # LP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.2, 30.0)
        if alloc2 > 10:
            a_r = lp_polish(a_r, time_budget=alloc2, verbose=False, rng=rng)

        obj_r = compute_obj(a_r)
        if obj_r < best_score - 1e-10:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Final Intensive Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Mirror descent final
    if remaining > 40:
        alloc = min(remaining * 0.30, 100.0)
        a_f1, _ = mirror_descent(best_seq, time_budget=alloc,
                                 tau_start=20.0, tau_end=400.0,
                                 verbose=True, rng=rng)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score - 1e-10:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # LP final
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.30, 80.0)
        a_f2 = lp_polish(best_seq, time_budget=alloc, verbose=True, rng=rng)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score - 1e-10:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Another MD pass
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.4, 60.0)
        a_f3, _ = mirror_descent(best_seq, time_budget=alloc,
                                 tau_start=50.0, tau_end=600.0,
                                 use_active=True, verbose=False, rng=rng)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score - 1e-10:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Final MD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
