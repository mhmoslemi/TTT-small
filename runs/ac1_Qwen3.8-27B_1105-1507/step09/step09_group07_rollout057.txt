Optimized LP-alternating with subgradient complement, multi-start, and n-sweep
for minimizing peak self-convolution of non-negative step functions.
Target: score < 1.5027

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Efficient linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A such that A@g = conv(a, g). A is (2n-1) x n."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses scipy linprog with HiGHS. Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)

    # Objective: minimize M => c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Inequality: A@g - M <= 0 => [A, -1] @ [g, M] <= 0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Bounds: g >= 0, M free
    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    # Fallback: try with reduced constraints (top 50% peaks)
    try:
        conv_ref = compute_conv(a_norm)
        threshold = np.median(conv_ref)
        keep = np.where(conv_ref >= threshold)[0]
        if len(keep) < 50:
            keep = np.argsort(conv_ref)[::-1][:100]

        A_red = A[keep, :]
        A_ub_red = np.hstack([A_red, -np.ones((len(keep), 1))])
        b_ub_red = np.zeros(len(keep))

        res = linprog(c, A_ub=A_ub_red, b_ub=b_ub_red, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def line_search(a, g, best_obj, max_evals=20):
    """
    Golden section line search for t in (0, 1) to minimize obj((1-t)*a + t*g).
    Returns (best_t, best_a, best_obj).
    """
    a_try_0 = a.copy()
    a_try_1 = g.copy()

    # Golden section parameters
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0

    # Evaluate at initial points
    t1 = lo + (1.0 - gr) * (hi - lo)
    t2 = lo + gr * (hi - lo)

    a1 = (1.0 - t1) * a_try_0 + t1 * a_try_1
    a1 = np.maximum(a1, 0)
    s1 = np.sum(a1)
    if s1 > 1e-10:
        a1 = a1 / s1
        f1 = compute_obj(a1)
    else:
        f1 = np.inf

    a2 = (1.0 - t2) * a_try_0 + t2 * a_try_1
    a2 = np.maximum(a2, 0)
    s2 = np.sum(a2)
    if s2 > 1e-10:
        a2 = a2 / s2
        f2 = compute_obj(a2)
    else:
        f2 = np.inf

    evals = 2
    best_t = t1 if f1 < f2 else t2
    best_f = min(f1, f2)
    best_a = a1.copy() if f1 < f2 else a2.copy()

    for _ in range(max_evals - 2):
        if f1 < f2:
            hi = t2
            t2 = t1
            f2 = f1
            a2 = a1.copy()
            t1 = lo + (1.0 - gr) * (hi - lo)
            a1 = (1.0 - t1) * a_try_0 + t1 * a_try_1
            a1 = np.maximum(a1, 0)
            s1 = np.sum(a1)
            if s1 > 1e-10:
                a1 = a1 / s1
                f1 = compute_obj(a1)
            else:
                f1 = np.inf
        else:
            lo = t1
            t1 = t2
            f1 = f2
            a1 = a2.copy()
            t2 = lo + gr * (hi - lo)
            a2 = (1.0 - t2) * a_try_0 + t2 * a_try_1
            a2 = np.maximum(a2, 0)
            s2 = np.sum(a2)
            if s2 > 1e-10:
                a2 = a2 / s2
                f2 = compute_obj(a2)
            else:
                f2 = np.inf

        evals += 1
        if f1 < best_f:
            best_f = f1
            best_t = t1
            best_a = a1.copy()
        if f2 < best_f:
            best_f = f2
            best_t = t2
            best_a = a2.copy()

        if abs(hi - lo) < 1e-6:
            break

    return best_t, best_a, best_f


def lp_alternate_v2(a, time_budget, verbose=False):
    """
    Improved LP alternating with golden section line search and stall handling.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    stall = 0
    patience = 15
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        g_lp, M_lp = solve_lp_fast(a)
        if g_lp is None:
            break

        # Check if we've converged
        if np.linalg.norm(a - g_lp) < 1e-10:
            stall += 5
        else:
            # Line search
            t_best, a_new, obj_new = line_search(a, g_lp, best_obj)

            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
                stall = 0
            else:
                # Try t=1 directly
                obj_full = compute_obj(g_lp)
                if obj_full < best_obj - 1e-12:
                    best_obj = obj_full
                    best_a = g_lp.copy()
                    a = g_lp.copy()
                    stall = 0
                else:
                    stall += 1

        if stall >= patience:
            # Perturb to escape
            rng = np.random.default_rng(it * 1000003 + n)
            scale = 0.02 + 0.01 * (stall // patience)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            else:
                a = best_a.copy()
            stall = 0

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"      LP-alt v2 it{it}: best={best_obj:.8f}, stall={stall}, t={elapsed:.1f}s")

    return best_a


def subgradient_step(a, lr):
    """
    One subgradient step for minimizing max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    n = len(a)
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    # Subgradient of conv[k_star] w.r.t. a[i] = 2*a[k_star - i] (if in range)
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]

    # Project gradient onto the feasible set tangent: remove mean
    grad_centered = grad - np.mean(grad)

    # Update
    a_new = a - lr * grad_centered
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()

    return a_new


def subgradient_method(a, time_budget, verbose=False):
    """Subgradient method with adaptive learning rate."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.1
    it = 0

    while time.monotonic() - start < time_budget:
        a_new = subgradient_step(a, lr)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = min(0.5, lr * 1.01)
        else:
            lr = max(1e-5, lr * 0.95)
            if lr < 1e-4:
                # Try a random restart
                rng = np.random.default_rng(it + n)
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                lr = 0.1

        it += 1
        if verbose and it % 100 == 0:
            print(f"      Subgrad it{it}: best={best_obj:.8f}, lr={lr:.6f}")

    return best_a


def peak_shaving(a, time_budget, verbose=False):
    """
    Greedy peak shaving: identify top peaks, reduce them by moving mass
    from contributors to quiet positions.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0

    while time.monotonic() - start < time_budget:
        round_num += 1
        conv = compute_conv(a)
        # Top 5 peaks
        peak_lags = np.argsort(conv)[::-1][:5]

        improved = False
        for k_star in peak_lags:
            k_star = int(k_star)
            # Find top contributors to this peak
            contributors = []
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    contributors.append((a[i] * a[j], i, j))
            contributors.sort(reverse=True)

            # Find quiet positions (small a[i], not contributing to peaks)
            peak_contrib = np.zeros(n)
            for pk in peak_lags:
                pk = int(pk)
                for i in range(n):
                    j = pk - i
                    if 0 <= j < n:
                        peak_contrib[i] += a[i] * a[j]
            quiet_idx = np.argsort(peak_contrib)[:10]

            for contrib_val, i, j in contributors[:8]:
                if contrib_val < 1e-10:
                    break
                # Try moving mass from the larger of a[i], a[j] to a quiet position
                src = i if a[i] >= a[j] else j
                for tidx in quiet_idx:
                    if tidx == src:
                        continue
                    for frac in [0.005, 0.01, 0.02, 0.05]:
                        delta = frac * a[src]
                        if delta <= 0 or delta >= 0.99 * a[src]:
                            continue
                        a[src] -= delta
                        a[tidx] += delta
                        new_conv = compute_conv(a)
                        new_obj = 2.0 * n * np.max(new_conv)
                        if new_obj < best_obj - 1e-13:
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[src] += delta
                            a[tidx] -= delta
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            break

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode % 10 == 0:
        a = np.ones(n) / n
    elif mode % 10 == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode % 10 == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 3:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + np.exp(-3 * (x - 0.7) ** 2) + 0.05
        a = a / a.sum()
    elif mode % 10 == 5:
        x = np.linspace(0, 1, n)
        a = x ** 2 + (1 - x) ** 2 + 0.1
        a = a / a.sum()
    elif mode % 10 == 6:
        freqs = rng.integers(2, 9, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 7:
        a = rng.dirichlet(np.ones(n) * 0.5)
    elif mode % 10 == 8:
        # Triangular (peaked in middle)
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(x - 0.5) * 1.5
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        # Exponential decay from center
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.5) ** 2)
        a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def multi_lp_alternate(a, time_budget, n_restarts=3, verbose=False):
    """
    Run LP alternating multiple times with increasing perturbation.
    Keep the best result.
    """
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    per_restart = time_budget / max(1, n_restarts)

    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        alloc = min(per_restart, remaining * 0.45)

        if r == 0:
            a_start = best_a.copy()
        else:
            rng = np.random.default_rng(r * 7919 + n + 42)
            scale = 0.03 * (r + 1)
            a_start = best_a * np.exp(rng.normal(0, scale, n))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()

        a_result = lp_alternate_v2(a_start, time_budget=alloc, verbose=verbose)
        obj_result = compute_obj(a_result)

        if verbose:
            print(f"      Multi-LP restart {r}: obj={obj_result:.8f}")

        if obj_result < best_obj:
            best_obj = obj_result
            best_a = a_result.copy()

    return best_a
