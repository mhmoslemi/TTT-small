def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def make_palindromic(b):
    """Extend a half-sequence b to a full palindromic sequence."""
    m = len(b)
    n = 2 * m if m == len(b) else 2 * m + 1
    # Actually, let me parameterize by n
    # For n = 2m: b has m elements, a[j] = b[min(j, n-1-j)]
    # For n = 2m+1: b has m+1 elements, a[j] = b[min(j, n-1-j)]
    return b


def full_from_half(b, n):
    """Given half-sequence b of length ceil(n/2), return full palindromic sequence of length n."""
    a = np.zeros(n)
    for j in range(n):
        a[j] = b[min(j, n - 1 - j)]
    return a


def half_from_full(a):
    """Extract half-sequence from a (assumed palindromic)."""
    n = len(a)
    m = (n + 1) // 2
    return a[:m].copy()


def build_conv_matrix_pal(a, n):
    """
    Build matrix A (n x m) such that A @ b = conv(a, full_from_half(b, n))[:n]
    where b is the half-sequence of length m = ceil(n/2).
    By palindromic symmetry of a, conv(a, pal) is also palindromic, so we only
    need k = 0, ..., n-1.
    """
    m = (n + 1) // 2
    A = np.zeros((n, m))
    # conv(a, pal)[k] = sum_j a[j] * pal[k-j] where 0<=j<n, 0<=k-j<n
    # pal[t] = b[min(t, n-1-t)]
    for k in range(n):
        for j in range(n):
            t = k - j
            if 0 <= t < n:
                col = min(t, n - 1 - t)
                A[k, col] += a[j]
    return A


def solve_lp_pal(a, M_target=None, solver_list=None):
    """
    Solve LP with palindromic structure:
    minimize M
    subject to: conv(a, g_pal)[k] <= M for k=0,...,n-1
                sum(g_pal) = 1
                g_pal >= 0
    where g_pal = full_from_half(g_half, n)
    """
    n = len(a)
    m = (n + 1) // 2
    a_norm = a / np.sum(a)
    
    A = build_conv_matrix_pal(a_norm, n)
    
    # sum(g_pal) = sum_j full_from_half(g_half)[j] = 2*sum(g_half) - g_half[m-1] (if n odd)
    # For n = 2m: sum = 2 * sum(g_half[0:m])
    # For n = 2m+1: sum = 2 * sum(g_half[0:m-1]) + g_half[m-1]
    if n % 2 == 0:
        sum_weights = 2.0 * np.ones(m)
    else:
        sum_weights = np.ones(m + 1)  # Wait, m = (n+1)//2
        # For n = 2m-1 (odd), m_half = m = (n+1)//2
        # sum = 2*sum(g[0:m-1]) + g[m-1]
        sum_weights = 2.0 * np.ones(m)
        sum_weights[-1] = 1.0
    
    g = cp.Variable(m, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, sum_weights @ g == 1]
    if M_target is not None:
        constraints.append(M <= M_target)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                g_full = full_from_half(g_val, n)
                s = float(np.sum(g_full))
                if s > 1e-10:
                    g_full = g_full / s
                    # Make exactly palindromic
                    g_full = (g_full + g_full[::-1]) / 2
                    return g_full, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue
    
    return None, None


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def resample_pal(a, new_n):
    """Resample a palindromic sequence to new length, maintaining palindromic structure."""
    n = len(a)
    if n == new_n:
        return a.copy()
    # Resample the half
    half = a[:(n + 1) // 2]
    new_half_len = (new_n + 1) // 2
    x_old = np.linspace(0, 1, len(half))
    x_new = np.linspace(0, 1, new_half_len)
    new_half = np.interp(x_new, x_old, half)
    new_half = np.maximum(new_half, 0)
    new_a = full_from_half(new_half, new_n)
    new_a = new_a / new_a.sum()
    return new_a

def subgradient_pal(a, time_budget=100.0, K=5, verbose=False):
    """Subgradient method respecting palindromic symmetry."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    a = (a + a[::-1]) / 2
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 3.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        weights = np.exp((peak_vals - max_pv) * 5.0 / max_pv)
        weights /= weights.sum()
        
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]
        
        # Make gradient palindromic
        g = (g + g[::-1]) / 2
        
        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        a_new = (a_new + a_new[::-1]) / 2
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-10:
                    break
        
        it += 1
    
    return best_a
