Optimization via L-BFGS on log-sum-exp surrogate with softmax parameterization.
Uses FFT-based O(n log n) gradient computation and beta annealing.
Diversifies with multiple starting points and periodic restarts.
"""

import numpy as np
from scipy.optimize import minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def softmax(x):
    """Numerically stable softmax."""
    x = x - np.max(x)
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def logsumexp(x, beta):
    """Compute logsumexp(beta * x) / beta."""
    shifted = beta * (x - x.max())
    m = x.max()
    return m + np.log(np.exp(shifted).sum()) / beta

    # sum2[i] = sum_{j=0}^{i} w[i-j]*a[j]    = convolution(w, a)[i]
    # grad_a[i] = sum1[i] + sum2[i] - w[2i]*a[i]

    # Compute sum1 = cross-correlation of w and a
    # (w cross a)[i] = sum_j w[j+i] * a[j]
    # = conv(flip(a), w)[i + n - 1]
    size1 = 1
    while size1 < (2 * n - 1) + n - 1:
        size1 <<= 1
    a_flip = a[::-1]
    WA = np.fft.rfft(w, size1)
    AF = np.fft.rfft(a_flip, size1)
    conv1 = np.fft.irfft(WA * AF, size1)[:2 * n - 1 + n - 1]
    sum1 = conv1[n - 1:2 * n - 1]  # indices [n-1] to [2n-2]

    # Compute sum2 = convolution of w and a (first n elements)
    # (w * a)[i] = sum_j w[j] * a[i-j]

    while size2 < (2 * n - 1) + n - 1:
        size2 <<= 1
    WW = np.fft.rfft(w, size2)
    AA = np.fft.rfft(a, size2)
    conv2 = np.fft.irfft(WW * AA, size2)[:n]
    sum2 = conv2

    # Diagonal correction
    diag_idx = np.arange(n) * 2
    diag_mask = diag_idx < (2 * n - 1)
    diag_corr = np.zeros(n)
    diag_corr[diag_mask] = w[diag_idx[diag_mask]] * a[diag_mask]

    grad_a = sum1 + sum2 - diag_corr
    return grad_a, obj

    Objective and gradient w.r.t. x (softmax parameterization).
    a = softmax(x), minimize logsumexp(beta * conv(a,a)) / beta.

Optimization via L-BFGS on log-sum-exp surrogate with softmax parameterization.
Uses FFT-based O(n log n) gradient computation and beta annealing.

Key innovation: natural gradient through softmax gives constraint-free

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def softmax_func(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s

    plus the scalar objective value. Uses FFT for O(n log n).

    b = compute_conv(a)
    m = b.max()

        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def perturb_and_reoptimize(a_base, rng, time_budget, n_pert=4, verbose=False):
    """Perturb the best solution and re-optimize to escape local minima."""
    n = len(a_base)
    best_a = a_base.copy()
    best_obj = compute_obj(a_base)
    start = time.monotonic()
    per_pert = time_budget / max(1, n_pert)

    for p in range(n_pert):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break
        alloc = min(per_pert, remaining * 0.8)

        scale = 0.04 + 0.03 * p
        noise = rng.normal(0, scale, n)
        a_pert = a_base * np.exp(noise)
        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert = a_pert / s

        a_opt, obj_opt = beta_anneal_lbfgs(
            a_pert, beta_start=200.0, beta_end=20000.0,
            n_levels=8, time_budget=alloc, verbose=False)

        if obj_opt < best_obj - 1e-12:
            best_obj = obj_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"      Perturb {p}: obj={best_obj:.8f} **")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: L-BFGS with softmax parameterization and beta annealing.
    Combines intensive optimization, n-sweep, diverse starts, and
    perturb-reoptimize cycles to escape local minima.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"\nInitial: n={n_current}, obj={best_score:.8f}, target=1.5027")

    # === Phase 1: Intensive L-BFGS beta annealing (40%) ===
    print(f"\n=== Phase 1: L-BFGS Beta Annealing (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.40, 550.0)

    if p1_budget > 30:
        a_p1, obj_p1 = beta_anneal_lbfgs(
            best_seq, beta_start=5.0, beta_end=50000.0,
            n_levels=35, time_budget=p1_budget, verbose=True)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
        print(f"  Phase 1 done: obj={best_score:.8f}")

    # === Phase 2: n-sweep (15%) ===
    print(f"\n=== Phase 2: n-Sweep ===")
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.15

    n_options = sorted(set([
        max(500, n_current - 100),
        max(500, n_current - 50),
        n_current,
        n_current + 50,
        n_current + 100,
        n_current + 200,
        750, 850, 950, 1050, 1200
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1500]
    per_n = p2_budget / max(1, len(n_options))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_n, remaining * 0.25)
        if alloc < 15:
            continue

        a_try = resample_sequence(best_seq, n_try)
        a_opt, obj_opt = beta_anneal_lbfgs(
            a_try, beta_start=50.0, beta_end=20000.0,
            n_levels=12, time_budget=alloc, verbose=False)

        marker = " **" if obj_opt < best_score else ""
        print(f"  n={n_try}: obj={obj_opt:.8f}{marker}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try

    # === Phase 3: Diverse starting points (25%) ===
    print(f"\n=== Phase 3: Diverse Starts ===")
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.25
    n_use = len(best_seq)
    n_starts = 6
    per_start = p3_budget / max(1, n_starts)

    for s_idx in range(n_starts):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_start, remaining * 0.35)
        if alloc < 15:
            break

        mode = (s_idx * 3 + 1) % 10
        a_div = diverse_init(n_use, rng, mode=mode)
        blend = 0.25 + 0.1 * s_idx
        a_div = (1.0 - blend) * a_div + blend * best_seq
        a_div = np.maximum(a_div, 0)
        s_val = a_div.sum()
        if s_val > 1e-10:
            a_div = a_div / s_val

        a_opt, obj_opt = beta_anneal_lbfgs(
            a_div, beta_start=20.0, beta_end=20000.0,
            n_levels=12, time_budget=alloc, verbose=False)

        marker = " **" if obj_opt < best_score else ""
        print(f"  Start {s_idx} (mode={mode}): obj={obj_opt:.8f}{marker}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()

    # === Phase 4: Perturb and re-optimize (10%) ===
    print(f"\n=== Phase 4: Perturb & Re-optimize ===")
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.10

    if p4_budget > 20:
        a_pert, obj_pert = perturb_and_reoptimize(
            best_seq, rng, time_budget=p4_budget,
            n_pert=4, verbose=True)
        if obj_pert < best_score:
            best_score = obj_pert
            best_seq = a_pert.copy()
        print(f"  Phase 4 done: obj={best_score:.8f}")

    # === Phase 5: Final high-beta polish (10%) ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 20:
        alloc1 = min(remaining * 0.5, 80.0)
        a_fin = lbfgs_optimize(best_seq, beta=50000.0,
                               max_iter=500, time_limit=alloc1)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  High-beta polish: obj={best_score:.8f} **")
        else:
            print(f"  High-beta polish: obj={obj_fin:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc2 = min(remaining * 0.5, 50.0)
            a_fin2 = lbfgs_optimize(best_seq, beta=150000.0,
                                    max_iter=300, time_limit=alloc2)
            obj_fin2 = compute_obj(a_fin2)
            if obj_fin2 < best_score:
                best_score = obj_fin2
                best_seq = a_fin2.copy()
                print(f"  Ultra-high beta: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"=== Target: 1.5027, Achieved: {best_score:.8f}, Gap: {best_score - 1.5027:.8f} ===")

    return result
