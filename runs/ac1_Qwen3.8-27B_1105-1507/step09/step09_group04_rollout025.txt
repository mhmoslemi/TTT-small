"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses accelerated multiplicative weight updates (AMW) as the main optimizer,
with periodic LP polishing and coordinate descent for final refinement.
The AMW method computes the "pressure" at each coordinate via FFT-based
cross-correlation, allowing O(n log n) per iteration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_pressure(a, conv, top_k=10, threshold_frac=0.9):
    """
    Compute the pressure vector using FFT-based cross-correlation.
    pressure[j] = sum_k w[k] * a[k-j] where w[k] is the weight at convolution position k.
    
    Returns (pressure, max_conv).
    """
    n = len(a)
    max_conv = float(np.max(conv))
    if max_conv <= 0:
        return np.zeros(n), 0.0
    
    # Find peak positions
    threshold = max_conv * threshold_frac
    peak_mask = conv >= threshold
    peak_idx = np.where(peak_mask)[0]
    
    if len(peak_idx) == 0:
        return np.zeros(n), max_conv
    
    # Weights: proportional to (conv[k] - threshold) / max_conv, or just 1 for all peaks
    # Use normalized weights
    w = np.zeros(2 * n - 1)
    w[peak_idx] = 1.0
    w_sum = float(np.sum(w))
    w /= w_sum  # normalize weights to sum to 1
    
    # Compute pressure via cross-correlation
    # pressure[j] = sum_k w[k] * a[k-j] (valid k-j)
    # = conv(w, a_flipped)[n-1+j]
    a_flipped = a[::-1]
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    W = rfft(w, size)
    AF = rfft(a_flipped, size)
    corr = irfft(W * AF, size)
    
    # pressure[j] = corr[n-1+j] for j = 0, ..., n-1
    pressure = corr[n-1 : 2*n-1]
    pressure = np.maximum(pressure, 0)
    
    return pressure, max_conv


def amw_optimize(a, time_budget=100.0, eta_init=0.5, verbose=False, rng=None):
    """
    Accelerated Multiplicative Weight optimization.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    conv = compute_conv(a)
    best_a = a.copy()
    best_obj = float(np.max(conv))
    best_global = 2.0 * n * best_obj
    
    eta = eta_init
    start = time.monotonic()
    it = 0
    stagnant = 0
    no_improve_count = 0
    
    # Momentum
    v = np.zeros(n)
    beta = 0.7
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Adaptive eta: decrease over time, with restarts
        eta = eta_init * (1.0 - 0.5 * progress)
        if stagnant > 50:
            eta = eta_init * 0.1  # small eta when stuck
            if rng is not None:
                # Perturb
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                conv = compute_conv(a)
                stagnant = 0
                v = np.zeros(n)
        
        # Compute pressure
        pressure, max_conv = compute_pressure(a, conv, top_k=20, threshold_frac=0.85)
        
        if max_conv <= 0:
            break
        
        # Nesterov-like extrapolation
        y = a - beta * v
        y = np.maximum(y, 1e-15)
        s_y = float(np.sum(y))
        y = y / s_y
        
        # Compute pressure at y
        conv_y = compute_conv(y)
        pressure_y, _ = compute_pressure(y, conv_y, top_k=20, threshold_frac=0.85)
        
        # Multiplicative update: a_new = y * exp(-eta * pressure_y)
        # Add a small uniform component to prevent any coordinate from dying
        log_update = -eta * pressure_y
        # Scale so the update is not too extreme
        max_log = float(np.max(log_update)) - float(np.min(log_update))
        if max_log > 3.0:
            log_update = log_update * (3.0 / max_log)
        
        a_new = y * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(np.sum(a_new))
        
        # Update momentum
        v = a_new - a
        a = a_new
        
        # Update convolution
        if it % 5 == 0 or it < 10:
            conv = compute_conv(a)
            new_max = float(np.max(conv))
            new_global = 2.0 * n * new_max
        else:
            new_global = None  # skip evaluation for speed
        
        if new_global is not None:
            if new_global < best_global - 1e-12:
                best_global = new_global
                best_obj = new_max
                best_a = a.copy()
                stagnant = 0
                no_improve_count = 0
            else:
                stagnant += 1
                no_improve_count += 1
            
            if verbose and it % 10000 == 0:
                print(f"    AMW it{it}: obj={best_global:.8f}, eta={eta:.4f}, t={elapsed:.0f}s")
        else:
            no_improve_count += 1
        
        it += 1
        
        # Stagnation handling
        if no_improve_count > 50000:
            no_improve_count = 0
            a = best_a.copy()
            v = np.zeros(n)
            if rng is not None:
                a = a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
    
    # Final evaluation
    conv_best = compute_conv(best_a)
    best_global = 2.0 * n * float(np.max(conv_best))
    
    return best_a, best_global

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Primary method: Fast Multiplicative Weight (FMW) with peak tracking and adaptive
step sizes, achieving O(n*K) per iteration. Secondary: LP polishing for global
direction finding, and coordinate descent for final refinement. Multi-n exploration
with structured perturbations for escaping local minima.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.MOSEK]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def fmw_optimize(a, time_budget=100.0, eta_init=1.0, n_peaks=10,
                 threshold_frac=0.85, verbose=False, rng=None, label=""):
    """
    Fast Multiplicative Weight optimization.
    At each step: find peak positions in conv(a,a), compute pressure,
    apply multiplicative update to reduce contribution to peaks.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64).copy()
    a = a / float(np.sum(a))

    conv = compute_conv(a)
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))

    eta = eta_init
    start = time.monotonic()
    it = 0
    stagnant = 0
    eval_interval = max(1, n // 50)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        # Find peaks
        max_val = float(np.max(conv))
        if max_val <= 0:
            break
        threshold = max_val * threshold_frac
        peak_mask = conv >= threshold
        peak_indices = np.where(peak_mask)[0]

        if len(peak_indices) > n_peaks:
            # Keep only top n_peaks
            peak_vals = conv[peak_indices]
            top_idx = np.argsort(peak_vals)[::-1][:n_peaks]
            peak_indices = peak_indices[top_idx]

        # Compute pressure: pressure[j] = sum_k w_k * a[k-j] for valid (k,j)
        pressure = np.zeros(n)
        for k in peak_indices:
            w = (conv[k] - threshold) / (max_val - threshold + 1e-15)
            w = max(w, 0.1)
            # a[k-j] is valid when 0 <= k-j <= n-1, i.e., max(0, k-n+1) <= j <= min(k, n-1)
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            if j_lo <= j_hi:
                pressure[j_lo:j_hi + 1] += w * a[k - j_hi:k - j_lo + 1][::-1]

        # Normalize pressure for stable update
        max_p = float(np.max(pressure))
        if max_p > 0:
            pressure = pressure / max_p
        else:
            pressure = np.ones(n) / n

        # Adaptive eta: decay over time, reset when stuck
        eta_eff = eta * (1.0 - 0.3 * progress)
        if stagnant > 200:
            eta_eff = eta * 0.1
        elif stagnant > 50:
            eta_eff = eta * 0.3

        # Multiplicative update with clipping
        log_factor = -eta_eff * pressure
        # Clip to prevent extreme changes
        log_factor = np.clip(log_factor, -5.0, 2.0)
        a = a * np.exp(log_factor)
        a = np.maximum(a, 1e-15)
        a = a / float(np.sum(a))

        it += 1

        # Periodic evaluation
        if it % eval_interval == 0:
            conv = compute_conv(a)
            new_obj = 2.0 * n * float(np.max(conv))
            if new_obj < best_obj - 1e-12:
                best_obj = new_obj
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1

            if verbose and it % (eval_interval * 50) == 0:
                print(f"    FMW{label} it{it}: obj={best_obj:.8f}, eta={eta_eff:.4f}, t={elapsed:.0f}s")

        # Stagnation: perturb and reset
        if stagnant > 500:
            stagnant = 0
            a = best_a.copy()
            if rng is not None:
                scale = 0.03 + 0.01 * (it % 5)
                a = a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
            conv = compute_conv(a)

    # Final evaluation
    conv_best = compute_conv(best_a)
    best_obj = 2.0 * n * float(np.max(conv_best))
    return best_a, best_obj


def fmw_with_lp(a, time_budget=100.0, n_lp_max=3, rng=None, verbose=False):
    """FMW with periodic LP polishing for global direction."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64).copy()
    a = a / float(np.sum(a))

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lp_calls = 0
    fmw_chunk = max(30.0, (time_budget - n_lp_max * 20.0) / (n_lp_max + 1))

    for phase in range(n_lp_max + 1):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        if phase > 0 and lp_calls < n_lp_max:
            # LP step
            lp_budget = min(20.0, remaining * 0.3)
            if lp_budget > 5:
                g_lp, M_lp = solve_lp_minpeak(best_a)
                if g_lp is not None:
                    # Try multiple blend factors
                    best_t_obj = best_obj
                    best_t = 0
                    for t in [0.2, 0.3, 0.5, 0.7, 1.0]:
                        a_try = (1 - t) * best_a + t * g_lp
                        a_try = np.maximum(a_try, 1e-15)
                        a_try = a_try / a_try.sum()
                        obj_try = compute_obj(a_try)
                        if obj_try < best_t_obj:
                            best_t_obj = obj_try
                            best_t = t
                    if best_t > 0:
                        best_obj = best_t_obj
                        best_a = ((1 - best_t) * best_a + best_t * g_lp)
                        best_a = np.maximum(best_a, 1e-15)
                        best_a = best_a / best_a.sum()
                        lp_calls += 1
                        if verbose:
                            print(f"    LP polish {lp_calls}: obj={best_obj:.8f}")

        # FMW step
        fmw_budget = min(fmw_chunk, remaining * 0.8)
        if fmw_budget > 5:
            a_fmw, obj_fmw = fmw_optimize(best_a, time_budget=fmw_budget,
                                          eta_init=1.0, n_peaks=10,
                                          threshold_frac=0.85,
                                          verbose=verbose, rng=rng,
                                          label=f"_p{phase}")
            if obj_fmw < best_obj:
                best_obj = obj_fmw
                best_a = a_fmw.copy()
                if verbose:
                    print(f"    FMW phase {phase}: obj={best_obj:.8f}")

    return best_a, best_obj


def coord_descent_fast(a, time_budget=50.0, rng=None, verbose=False):
    """Fast coordinate descent with incremental convolution updates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = float(np.max(conv))
    gr = (np.sqrt(5) - 1) / 2.0
    start = time.monotonic()
    stagnant = 0

    while time.monotonic() - start < time_budget:
        obj_before = float(np.max(conv))
        order = np.arange(n) if rng is None else rng.permutation(n)

        for k in order:
            if time.monotonic() - start > time_budget:
                break
            # Direction
            d = np.zeros(2 * n - 1)
            d[k:k + n] = a

            # Golden section search for optimal delta
            lo_delta = -a[k]
            hi_delta = max(2.0 / n, lo_delta + 1.0 / n)

            def obj_at_delta(delta):
                val = conv + 2.0 * delta * d
                val[2 * k] += delta * delta
                return float(np.max(val))

            a_gs, b_gs = lo_delta, hi_delta
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)
            fc, fd = obj_at_delta(c_gs), obj_at_delta(d_gs)

            for _ in range(30):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = obj_at_delta(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = obj_at_delta(d_gs)

            delta_opt = (a_gs + b_gs) / 2.0
            new_est = obj_at_delta(delta_opt)
            if new_est < obj_before - 1e-14:
                a[k] = max(0.0, a[k] + delta_opt)
                conv += 2.0 * delta_opt * d
                conv[2 * k] += delta_opt * delta_opt
                S_new = float(np.sum(a))
                if S_new < 1e-10:
                    break
                a /= S_new
                conv /= (S_new * S_new)

        obj_after = float(np.max(conv))
        if obj_after < best_obj - 1e-12:
            best_obj = obj_after
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 3:
                if rng is not None:
                    scale = 0.02 * (1 + stagnant * 0.3)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    conv = compute_conv(a)
                    best_obj = float(np.max(conv))
                    stagnant = 0
                else:
                    break

    return best_a, 2.0 * n * float(np.max(compute_conv(best_a)))


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-15)


def structured_perturb(a, rng, mode=0):
    """Apply structured perturbation to escape local minima."""
    n = len(a)
    x = np.linspace(0, 1, n)
    if mode == 0:
        # Sinusoidal
        freq = rng.integers(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        pert = 0.1 * np.sin(2 * np.pi * freq * x + phase)
    elif mode == 1:
        # Cosine window reshape
        pert = 0.1 * np.cos(2 * np.pi * x)
    elif mode == 2:
        # Random Fourier
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        pert = np.zeros(n)
        for f, p in zip(freqs, phases):
            pert += 0.05 * np.sin(2 * np.pi * f * x + p)
    elif mode == 3:
        # Gaussian bumps
        n_bumps = rng.integers(1, 4)
        pert = np.zeros(n)
        for _ in range(n_bumps):
            c = rng.uniform(0, n)
            w = rng.uniform(n * 0.05, n * 0.2)
            pert += 0.05 * np.exp(-((np.arange(n) - c) ** 2) / (2 * w * w))
    elif mode == 4:
        # Anti-peak: boost where current conv is low
        conv = compute_conv(a / a.sum())
        min_val = np.min(conv)
        low_mask = conv < min_val + 0.3 * (np.max(conv) - min_val)
        low_idx = np.where(low_mask)[0]
        if len(low_idx) > 0:
            pert = np.zeros(n)
            for k in low_idx[:5]:
                j_lo = max(0, k - n + 1)
                j_hi = min(k, n - 1)
                if j_lo <= j_hi:
                    pert[j_lo:j_hi + 1] += 0.05
    else:
        pert = rng.normal(0, 0.05, n)

    a_new = a + pert * a
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    return a_new

        alloc = min(remaining * 0.5, 40.0)
        a_f2, obj_f2 = fmw_optimize(best_seq, time_budget=alloc, eta_init=0.3,
                                    n_peaks=3, threshold_frac=0.95,
                                    verbose=False, rng=rng, label="_v2")
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  V2 FMW: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
