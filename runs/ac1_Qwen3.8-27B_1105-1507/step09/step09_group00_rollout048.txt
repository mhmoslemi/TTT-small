```python
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * np.max(conv)

# Active-set LP
def solve_active_set_lp(a, max_iter=100, verbose=False):
    """
    Solve: min M s.t. (A@g)_k <= M, sum(g)=1, g>=0
    using active-set method.
    """
    n = len(a)
    a = a / np.sum(a)
    
    # Initial guess: use the smoothed solution
    g = a.copy()
    M = np.max(compute_conv(g))
    
    # Active set: start with top convolution positions
    conv = compute_conv(g)
    active_idx = np.argsort(conv)[::-1][:min(2*n-1, max(50, n//2))]
    
    for it in range(max_iter):
        # Build LP with active constraints
        # min M
        # s.t. (A@g)_k <= M for k in active_idx
        #      sum(g) = 1
        #      g >= 0
        
        # A matrix: active_idx x n
        A_active = np.zeros((len(active_idx), n))
        for i, k in enumerate(active_idx):
            # (A@g)_k = sum_j a[k-j] * g[j] for valid j
            # j ranges from max(0, k-n+1) to min(k, n-1)
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                A_active[i, j] = a[k - j]
        
        # LP: min [1, 0, ..., 0] · [M; g]
        # s.t. [I, -A_active] · [M; g] <= 1 (i.e., M - (A_active@g) >= 0, rearranged)
        #      Actually: (A_active@g) - M <= 0
        #      And: sum(g) = 1
        #      g >= 0, M free (but M >= 0 in practice)
        
        # For scipy linprog: min c^T x s.t. A_ub x <= b_ub, A_eq x = b_eq, bounds
        c = np.concatenate([[1], np.zeros(n)])  # Minimize M
        
        # Inequality: (A_active @ g) - M <= 0 for each active constraint
        A_ub = np.zeros((len(active_idx), 1 + n))
        A_ub[:, 0] = -1  # -M
        A_ub[:, 1:] = A_active  # + A_active @ g
        b_ub = np.zeros(len(active_idx))
        
        # Equality: sum(g) = 1
        A_eq = np.zeros((1, 1 + n))
        A_eq[0, 1:] = 1
        b_eq = [1.0]
        
        # Bounds: M free (use -inf to +inf), g >= 0
        bounds = [(None, None)] + [(0, None)] * n
        
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                         bounds=bounds, method='highs')
        
        if not result.success:
            break
        
        M_new = result.x[0]
        g_new = np.maximum(result.x[1:], 0)
        g_new = g_new / np.sum(g_new)
        
        # Check all constraints
        conv_new = compute_conv(g_new)
        violated = np.where(conv_new > M_new + 1e-10)[0]
        
        if len(violated) == 0:
            # Optimal!
            if verbose:
                print(f"  Active-set LP converged in {it} iterations, M={M_new:.10f}")
            return g_new, M_new
        
        # Add the most violated constraint
        most_violated = violated[np.argmax(conv_new[violated])]

        # Check convergence
        if abs(M_new - M) < 1e-12:
            break
        M = M_new
        g = g_new
```

Wait: d/d(a_i) sum_j a(j) a(k-j)
= sum_j [delta(j,i) * a(k-j) + a(j) * delta(k-j, i)]
= a(k-i) + a(k-i)  [since when j=i, first term is a(k-i); when k-j=i i.e. j=k-i, second term is a(k-i)]
= 2 * a(k-i) (if k-i is valid, i.e., 0 <= k-i < n)

So d/d(a_i) [(a*a)(k)] = 2 * a(k-i) if 0 <= k-i < n, else 0.

This means the gradient of the smoothed objective is:
grad_i = (1/Z) * sum_k exp(tau * conv_k) * 2 * a(k-i) * [0 <= k-i < n]

Where Z = sum_k exp(tau * conv_k).

This can be written as a cross-correlation:
grad = 2 * (p * a_reversed) [truncated]

Where p_k = exp(tau * conv_k) / Z, and a_reversed is the reversed sequence.

Actually, L-BFGS requires a smooth objective. The log-sum-exp is smooth, but the constraint sum(a)=1, a>=0 makes it a constrained problem. We can:
1. Use a reparameterization: a = exp(x) / sum(exp(x)) (softmax), which automatically satisfies a > 0, sum(a) = 1
2. Minimize the log-sum-exp of conv(softmax(x), softmax(x)) over x

This converts the constrained problem to an unconstrained one!

The softmax parameterization is nice because:
- It automatically handles the non-negativity constraint
- It automatically handles the sum-to-1 constraint
- The optimization is unconstrained

The downside is that the softmax can be ill-conditioned (very peaked or very flat), but in practice this shouldn't be a problem.

```python
def softmax_obj(x, tau):
    a = np.exp(x - x.max())
    a = a / a.sum()
    conv = compute_conv(a)
    max_conv = np.max(conv)
    return (1/tau) * np.log(np.sum(np.exp(tau * (conv - max_conv))) * np.exp(tau * max_conv))
    # Actually, for numerical stability, just return log-sum-exp:
    shifted = tau * conv
    shifted -= shifted.max()
    return np.log(np.sum(np.exp(shifted))) / tau

def softmax_grad(x, tau):
    a = np.exp(x - x.max())
    a = a / a.sum()
    conv = compute_conv(a)
    max_conv = np.max(conv)
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / exp_vals.sum()
    # Gradient w.r.t. a:
    # d/da_i sum_k p_k conv_k = sum_k p_k * 2*a(k-i) [valid]
    # = 2 * (p cross-correlate a)
    # Using FFT:
    N = 1
    while N < 3 * len(a):
        N <<= 1
    P = np.fft.fft(p, N)
    A = np.fft.fft(a, N)
    g_a = np.fft.ifft(P * np.conj(A)).real[:len(a)]
    g_a *= 2
    # Gradient w.r.t. x: dx -> da = a * (dx - mean(dx weighted by a))
    # da_i/dx_j = a_i * (delta_ij - a_j)
    # g_x = a * (g_a - sum(a * g_a))
    g_x = a * (g_a - np.sum(a * g_a))
    return g_x
```

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS with softmax parameterization and adaptive log-sum-exp smoothing,
combined with multi-n exploration and active-set final polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def logsumexp_smoothened_obj_and_grad(x, tau):
    """
    Objective and gradient for softmax parameterization.
    x: parameter vector (unconstrained)
    tau: smoothing parameter
    Returns: (objective, gradient)
    """
    n = len(x)
    # Softmax to get a
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum()
    
    # Compute convolution
    conv = compute_conv(a)
    
    # Log-sum-exp of conv with smoothing parameter tau
    tau_conv = tau * conv
    max_tc = np.max(tau_conv)
    exp_vals = np.exp(tau_conv - max_tc)
    sum_exp = np.sum(exp_vals)
    obj = (max_tc + np.log(sum_exp)) / tau  # = log(sum exp(tau*conv)) / tau
    
    # Gradient of log-sum-exp w.r.t. a
    p = exp_vals / sum_exp  # probability distribution over conv positions
    
    # d/da_i [sum_k p_k * conv_k] = sum_k p_k * 2*a(k-i) [if 0<=k-i<n]
    # This is a cross-correlation: 2 * cross_correlate(p, a)
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    # cross_correlate(p, a)[i] = sum_k p[k] * a[k-i] (circular, but we made N large enough)
    g_a = ifft(P * np.conj(A)).real[:n]
    g_a *= 2.0
    
    # Chain rule: dx -> da
    # a_i = exp(x_i) / sum(exp(x_j))
    # da_i/dx_j = a_i * (delta_ij - a_j)
    # g_x_i = sum_j (da_i/dx_j) * g_a_i ... no wait
    # g_x = J^T * g_a where J is the Jacobian of a w.r.t. x
    # J[i,j] = da_i/dx_j = a_i * (delta_ij - a_j)
    # (J^T * g_a)_i = sum_j J[j,i] * g_a_j = sum_j a_j * (delta_ji - a_i) * g_a_j
    # = a_i * g_a_i - a_i * sum_j a_j * g_a_j
    # = a_i * (g_a_i - sum(a * g_a))
    mean_ag = np.sum(a * g_a)
    g_x = a * (g_a - mean_ag)
    
    return obj, g_x


def run_lBFGS(a_init, tau_schedule, time_budget, verbose=True):
    """
    Run L-BFGS on the softmax parameterization with adaptive tau.
    tau_schedule: list of (tau, fraction_of_time) pairs
    """
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-15)
    a_init = a_init / a_init.sum()
    
    # Initialize x from a: x = log(a)
    x0 = np.log(a_init)
    
    best_obj = compute_obj(a_init)
    best_a = a_init.copy()
    
    start = time.monotonic()
    
    # Split time budget among tau values
    total_frac = sum(f for _, f in tau_schedule)
    current_frac = 0.0
    
    for tau, frac in tau_schedule:
        alloc_time = time_budget * frac / total_frac
        phase_start = time.monotonic()
        
        def fun(x):
            f, _ = logsumexp_smoothened_obj_and_grad(x, tau)
            return f
        
        def fun_grad(x):
            return logsumexp_smoothened_obj_and_grad(x, tau)
        
        # Run L-BFGS with time limit
        elapsed_in_phase = 0
        x = x0
        while time.monotonic() - phase_start < alloc_time:
            result = minimize(fun_grad, x, method='L-BFGS-B', 
                           options={'maxiter': 100, 'maxfun': 200,
                                   'maxls': 30, 'gtol': 1e-8})
            if result.success:
                x = result.x
            else:
                x = result.x  # Use best point found
                break
            
            # Check time
            if time.monotonic() - phase_start >= alloc_time:
                break
            
            # Update x0 for next iteration
            x0 = x.copy()
            
            # Evaluate actual objective
            a_curr = np.exp(x - x.max())
            a_curr = a_curr / a_curr.sum()
            obj_curr = compute_obj(a_curr)
            if obj_curr < best_obj:
                best_obj = obj_curr
                best_a = a_curr.copy()
            
            if verbose:
                print(f"      L-BFGS tau={tau}: obj={best_obj:.8f}, it={result.nit}")
        
        # Update x0 for next tau
        x0 = x.copy()
        
        if time.monotonic() - start >= time_budget:
            break
    
    return best_a, best_obj


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # U-shaped (arcsine-like)
        x = np.arange(n)
        a = 1.0 / (np.sqrt(np.maximum(x + 1, 0.1)) * np.sqrt(np.maximum(n - x, 0.1)))
        a = a / a.sum()
    elif mode == 3:
        # Bimodal
        c1 = n // 4
        c2 = 3 * n // 4
        w = n // 8
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        # Triangular
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 4 * np.abs(x - 0.5), 0.01)
        a = a / a.sum()
    elif mode == 7:
        # Inverse U (peaked in middle)
        x = np.arange(n)
        a = np.exp(-((x - n/2) ** 2) / (2 * (n/6) ** 2))
        a = a / a.sum()
    elif mode == 8:
        # Sinc-like
        x = np.arange(n) - n // 2
        a = np.sinc(x / (n / 10))
        a = np.abs(a)
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 9:
        # Random Fourier
        freqs = rng.integers(1, 8, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def smoothed_gradient_descent(a, time_budget, tau_init=10.0, tau_end=500.0, verbose=False):
    """
    Nesterov accelerated gradient descent with cosine annealing of tau.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Work in log-space for the softmax parameterization
    x = np.log(a)
    v = np.zeros(n)
    beta = 0.9
    base_lr = 2.0 / np.sqrt(n)
    
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        
        # Cosine anneal tau
        cos_val = 0.5 * (1 + np.cos(np.pi * progress))
        tau = tau_init + (tau_end - tau_init) * (1 - cos_val)
        
        if it % 5000 == 0 and verbose:
            print(f"    SGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")
        
        # Nesterov lookahead
        y = x + beta * v
        # Convert y to a for gradient computation
        y_shifted = y - np.max(y)
        a_y = np.exp(y_shifted)
        a_y = a_y / a_y.sum()
        
        # Compute gradient at a_y
        conv = compute_conv(a_y)
        tau_conv = tau * conv
        max_tc = np.max(tau_conv)
        exp_vals = np.exp(tau_conv - max_tc)
        p = exp_vals / exp_vals.sum()
        
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a_y, N)
        g_a = ifft(P * np.conj(A)).real[:n] * 2.0
        
        mean_ag = np.sum(a_y * g_a)
        g_x = a_y * (g_a - mean_ag)
        
        g_norm = np.linalg.norm(g_x)
        
        # Adaptive learning rate
        lr = base_lr / (1 + g_norm * base_lr * 10)
        
        # Update
        v = beta * v - lr * g_x
        x_new = y + v
        
        # Convert back to a
        x_new_shifted = x_new - np.max(x_new)
        a_new = np.exp(x_new_shifted)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        x = x_new
        it += 1
        
        # Reset if stuck
        if it - best_it > 20000:
            x = np.log(np.maximum(best_a, 1e-15))
            v = np.zeros(n)
            best_it = it
    
    return best_a, best_obj


def active_set_polish(a, time_budget=60.0, verbose=False):
    """
    Final polishing: identify the active set (max convolution positions)
    and do a local optimization targeting those positions.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Use very high tau for near-exact max
    tau = 5000.0
    base_lr = 1.0 / np.sqrt(n)
    lr = base_lr
    
    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = np.max(conv)
        
        # Get gradient targeting the max
        tau_conv = tau * conv
        max_tc = np.max(tau_conv)
        exp_vals = np.exp(tau_conv - max_tc)
        p = exp_vals / exp_vals.sum()
        
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_a = ifft(P * np.conj(A)).real[:n] * 2.0
        
        # Project gradient onto tangent space of simplex
        g_a = g_a - np.mean(g_a)
        
        # Step
        a_new = a - lr * g_a
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Decrease lr slowly
            lr *= 0.999
        else:
            lr *= 0.9
            if lr < 1e-12:
                break
        
        it += 1
    
    return best_a, best_obj
