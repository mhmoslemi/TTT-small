"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast LP polishing with scipy HiGHS, peak-directed perturbations,
subgradient refinement, and multi-length exploration. Key improvements over
previous: 10x faster LP (direct HiGHS vs cvxpy), enabling 100x more iterations;
peak-directed perturbation strategy; and adaptive step-size subgradient.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_fast(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses scipy HiGHS with sparse matrix. Returns (g, M) or (None, None).
    """

    a = a / float(np.sum(a))

    # Build sparse A_ub matrix: (2n-1) x (n+1)
    rows = []
    cols = []
    vals = []

    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)

    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    c = np.zeros(n + 1)
    c[n] = 1.0

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s, float(res.x[n])
    return None, None


def solve_lp_fast_precomputed(a, row_indices, col_indices_base, nnz_base):
    """
    Solve LP using precomputed sparsity pattern. Only updates values.
    a: current normalized sequence
    row_indices, col_indices_base: precomputed COO structure (without -1 entries)
    """
    n = len(a)
    a = a / float(np.sum(a))

    # Build values: for each non-zero entry, the value is a[col]
    vals = a[col_indices_base].copy()

    # Add -1 entries for M variable
    extra_rows = np.arange(2 * n - 1)
    extra_cols = np.full(2 * n - 1, n)
    extra_vals = np.full(2 * n - 1, -1.0)

    all_rows = np.concatenate([row_indices, extra_rows])
    all_cols = np.concatenate([col_indices_base, extra_cols])
    all_vals = np.concatenate([vals, extra_vals])

    A_ub = csr_matrix((all_vals, (all_rows, all_cols)), shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s, float(res.x[n])
    return None, None


def precompute_lp_structure(n):
    """Precompute the sparsity structure for the LP at given n."""
    row_indices = []
    col_indices = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            row_indices.append(k)
            col_indices.append(j)
    return np.array(row_indices, dtype=np.int32), np.array(col_indices, dtype=np.int32)


def lp_polish_fast(a, time_budget, row_idx, col_idx, verbose=False, t_init=0.3):
    """
    Fast LP-based polishing loop using precomputed structure.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    no_improve = 0
    t_current = t_init

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start

        g_lp, M_lp = solve_lp_fast_precomputed(a, row_idx, col_idx)
        if g_lp is None:
            no_improve += 1
            if no_improve > 5:
                break
            continue

        # Try a few blend factors
        best_t = 0
        best_blend_obj = best_obj

        for t in [t_current * 0.5, t_current, t_current * 1.5, 0.1, 0.5]:
            t = min(max(t, 0.01), 1.0)
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_t = t

        it += 1

        if best_blend_obj < best_obj - 1e-14:
            best_obj = best_blend_obj
            a_try = (1.0 - best_t) * best_a + best_t * g_lp
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            best_a = a_try.copy()
            a = best_a.copy()
            t_current = best_t
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, el={elapsed:.0f}s")
        else:
            no_improve += 1
            # Try smaller step
            t_small = max(0.005, t_current * 0.3)
            a_try = (1.0 - t_small) * best_a + t_small * g_lp
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = best_a.copy()
                t_current = t_small
                no_improve = 0
            else:
                if no_improve >= 8:
                    # Perturb
                    rng = np.random.default_rng(it)
                    scale = 0.005 + 0.003 * min(no_improve, 10)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    best_a = a.copy()
                    t_current = 0.3
                    no_improve = 0
                    if it > 200:
                        break

    return best_a


def subgradient_refine_fast(a, time_budget, K=5, verbose=False):
    """Subgradient method with top-K peaks, momentum, and adaptive step."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 3.0 / np.sqrt(n)
    momentum = 0.8
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]

        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break

        # Softmax weights over top peaks
        weights = np.exp((peak_vals - max_pv) * 10.0 / max_pv)
        weights /= weights.sum()

        # Compute subgradient
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * 2.0 * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.7
                v *= 0.5
                stall = 0
                if lr < 1e-10:
                    break
            else:
                lr *= 0.998

        it += 1

    return best_a


def peak_directed_perturb(a, rng, strength=0.05):
    """Perturb by reducing coordinates that contribute most to peaks."""
    n = len(a)
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:3]

    # Compute contribution of each coordinate to the top peaks
    contrib = np.zeros(n)
    for k in top_idx:
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            contrib[i_start:i_end + 1] += a[j_start:j_end + 1][::-1] * a[i_start:i_end + 1]

    contrib /= (np.max(contrib) + 1e-15)

    # Reduce high-contribution coordinates, add noise to low-contribution
    perturb = 1.0 - strength * contrib + strength * 0.5 * rng.standard_normal(n)
    a_new = a * np.exp(perturb)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def nesterov_smoothed_descent(a, time_budget=120.0, tau_init=5.0, tau_end=200.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 4.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        from scipy.fft import fft, ifft
        P = fft(p, N)
        A = fft(y, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 15.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.99995)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 4.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a
