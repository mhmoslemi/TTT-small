def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu * rho > (cssv - 1.0)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_val = idx[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_obj_and_grad(a, tau):
    """

    lse = max_conv + np.log(np.sum(exp_vals)) / tau

            lse, g = smoothed_obj_and_grad(a_try, current_tau)
            ga = np.dot(g, a_try)
            grad_x = a_try * (g - ga)
            return lse, grad_x
        
        x_max = np.max(best_a)  # This is wrong, best_a is already on simplex
        # Use current best as starting point
        x_init = np.log(np.maximum(best_a, 1e-15))
        
        try:
            result = minimize(
                obj_and_grad2, x_init, method='L-BFGS-B', jac=True,
                options={'maxiter': 5000, 'maxfun': 5000,
                         'ftol': 1e-15, 'gtol': 1e-12,
                         'maxls': 50, 'iprint': 0}
            )
            if result.x is not None:
                x_max = np.max(result.x)
                exp_x = np.exp(result.x - x_max)
                a_result = exp_x / np.sum(exp_x)
                a_result = np.maximum(a_result, 0)
                s_r = float(np.sum(a_result))
                if s_r > 0:
                    a_result = a_result / s_r
                    obj_result = compute_obj(a_result)
                    if obj_result < best_obj:
                        best_obj = obj_result
                        best_a = a_result.copy()
        except Exception:
            pass
        
        if current_tau >= 2000.0:
            break
    
    return best_a


def nesterov_gd(a, time_budget=60.0, tau_init=10.0, tau_end=500.0, seed=0):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.8
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init + (tau_end - tau_init) * (1.0 - np.cos(np.pi * progress)) / 2.0

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_obj_and_grad(y, tau)  # Returns (lse, grad), but we want the raw gradient
        
        # Actually let me just get the gradient of the smoothed max
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(y, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        # Barzilai-Borwein step size
        if prev_g is not None and it > 0 and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 20.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1

        lr = max(1e-10, lr * 0.99995)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 5.0 / np.sqrt(n)
            best_it = it

    return best_a
