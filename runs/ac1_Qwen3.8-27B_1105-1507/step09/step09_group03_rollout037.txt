def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            g[m_lo:m_hi+1] += 2.0 * a[idx]
    
    g /= len(active)
    return g, active

"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_maxconv(a, conv, rel_tol=1e-10):
    """
    Compute a subgradient of f(a) = max_k conv(a,a)_k.
    Returns (gradient, active_indices).
    grad Q_k(a)_m = 2 * a[k-m] for valid m.
    """
    n = len(a)
    max_val = float(np.max(conv))
    active = np.where(conv >= max_val - rel_tol * max(max_val, 1e-15))[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(conv))])
    g = np.zeros(n)
    for k in active:
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            idx = k - np.arange(m_lo, m_hi + 1)
            g[m_lo:m_hi + 1] += 2.0 * a[idx]
    g /= len(active)
    return g, active


def fista_exact(a_init, time_budget=300.0, verbose=False):
    """FISTA on f(a) = max_k conv(a,a)_k over simplex with exact subgradient."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_f = float(np.max(compute_conv(a)))
    start = time.monotonic()
    t_k = 1.0
    eta = 0.4
    y = a.copy()
    it = 0
    stall = 0
    best_it = 0
    while time.monotonic() - start < time_budget:
        if it > 0:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            beta = (t_k - 1.0) / t_new
        else:
            t_new = 1.0
            beta = 0.0
        conv_y = compute_conv(y)
        g, active = subgradient_maxconv(y, conv_y)
        z = y - eta * g
        x = project_simplex(z)
        conv_x = compute_conv(x)
        f_x = float(np.max(conv_x))
        if f_x < best_f - 1e-15:
            best_f = f_x
            best_a = x.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
        if stall > 40:
            eta *= 0.5
            if eta < 1e-5:
                eta = 0.1
                t_k = 1.0
            stall = 0
        if it % 200 == 0 and stall < 10:
            eta = min(eta * 1.02, 0.5)
        y = x + beta * (x - a)
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        a = x
        t_k = t_new
        it += 1
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    FISTAx it{it}: obj={2.0*n*best_f:.8f}, eta={eta:.4f}, |A|={len(active)}, t={elapsed:.1f}s")
        if it - best_it > 8000:
            a = best_a.copy()
            y = best_a.copy()
            t_k = 1.0
            eta = 0.4
            best_it = it
            rng = np.random.default_rng(it)
            a = best_a * np.exp(rng.normal(0, 0.005, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            y = a.copy()
    return best_a

def mirror_descent(a_init, time_budget=120.0, verbose=False):
    """
    Mirror descent (multiplicative weights) on max_k conv(a,a)_k over simplex.
    No sorting needed - O(n) per step.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_f = float(np.max(compute_conv(a)))
    start = time.monotonic()
    eta = 0.1
    it = 0
    stall = 0
    best_it = 0
    while time.monotonic() - start < time_budget:
        conv_a = compute_conv(a)
        max_val = float(np.max(conv_a))
        active = np.where(conv_a >= max_val - 1e-10)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv_a))])
        # Gradient of max at active peaks (average)
        g = np.zeros(n)
        for k in active:
            m_lo = max(0, int(k) - n + 1)
            m_hi = min(n - 1, int(k))
            if m_lo <= m_hi:
                idx = int(k) - np.arange(m_lo, m_hi + 1)
                g[m_lo:m_hi + 1] += 2.0 * a[idx]
        g /= len(active)
        # Multiplicative update
        log_new = np.log(a + 1e-30) - eta * g
        log_new -= np.max(log_new)  # numerical stability
        a_new = np.exp(log_new)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        conv_new = compute_conv(a_new)
        f_new = float(np.max(conv_new))
        if f_new < best_f - 1e-15:
            best_f = f_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
        if stall > 30:
            eta *= 0.8
            if eta < 1e-4:
                eta = 0.05
            stall = 0
        a = a_new
        it += 1
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD it{it}: obj={2.0*n*best_f:.8f}, eta={eta:.6f}, |A|={len(active)}, t={elapsed:.1f}s")
        if it - best_it > 20000:
            break
    return best_a


def lp_polish(a_init, time_budget=100.0, verbose=False):

                    g_val = np.maximum(g_var.value, 0)

        # Golden section line search for blending

            else:
                lo = t1

def peak_balance(a, time_budget=60.0, verbose=False):
    """
    Directly target top peaks: for each top peak, find best (source, target)
    mass transfer to reduce it.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(777)
    it = 0
    stall = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_k = 3
        peak_indices = np.argsort(conv)[::-1][:top_k]
        improved = False
        for pk in peak_indices:
            pk = int(pk)
            m_lo = max(0, pk - n + 1)
            m_hi = min(n - 1, pk)
            if m_lo > m_hi:
                continue
            # Find top contributors to this peak
            contribs = []
            for m in range(m_lo, m_hi + 1):
                j = pk - m
                c = a[m] * a[j]
                contribs.append((c, m, j))
            contribs.sort(reverse=True)
            for c_val, src, dst in contribs[:5]:
                if src == dst:
                    continue
                for delta_frac in [0.1, 0.05, 0.02, 0.01]:
                    delta = delta_frac * a[src]
                    if delta > a[src] * 0.9:
                        continue
                    # Try moving to random targets
                    for _ in range(4):
                        tgt = int(rng.integers(0, n))
                        if tgt == src:
                            continue
                        a_try = a.copy()
                        a_try[src] -= delta
                        a_try[tgt] += delta
                        st = float(np.sum(a_try))
                        if st < 1e-10:
                            continue
                        a_try = a_try / st
                        obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
        if verbose and it % 100 == 0:
            print(f"    PB it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):

    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-8))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        # Two-peak structure
        a = np.zeros(n)
        w = max(1, n // 10)
        a[:w] = 1.0
        a[n - w:] = 1.0
        a = a + 0.1 * np.random.default_rng(0).uniform(0, 1, n)
        a = a / a.sum()
    else:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    return a


def full_optimize(a_init, time_budget=300.0, verbose=False):
    """Run full optimization pipeline on a single sequence."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    start = time.monotonic()

    # Stage 1: Smoothed FISTA (warm-up, broad exploration)
    remaining = time_budget - (time.monotonic() - start)
    t1 = min(remaining * 0.3, 90.0)
    if t1 > 10:
        a1 = fista_smoothed(a, time_budget=t1, tau_init=3.0, tau_end=200.0, verbose=verbose)
        o1 = 2.0 * n * float(np.max(compute_conv(a1)))
        if o1 < best_obj:
            best_obj = o1
            best_a = a1.copy()
            a = a1
            if verbose:
                print(f"      SmoothFISTA: {best_obj:.8f}")

    # Stage 2: Exact FISTA (fine convergence)
    a = (1 - 0.1) * a + 0.1 * best_a
    a = np.maximum(a, 0)
    a = a / a.sum()
    remaining = time_budget - (time.monotonic() - start)
    t2 = min(remaining * 0.3, 100.0)
    if t2 > 10:
        a2 = fista_exact(a, time_budget=t2, verbose=verbose)
        o2 = 2.0 * n * float(np.max(compute_conv(a2)))
        if o2 < best_obj:
            best_obj = o2
            best_a = a2.copy()
            a = a2
            if verbose:
                print(f"      ExactFISTA: {best_obj:.8f}")

    # Stage 3: Mirror descent (different trajectory)
    a = (1 - 0.2) * a + 0.2 * best_a
    a = np.maximum(a, 0)
    a = a / a.sum()
    remaining = time_budget - (time.monotonic() - start)
    t3 = min(remaining * 0.15, 50.0)
    if t3 > 10:
        a3 = mirror_descent(a, time_budget=t3, verbose=verbose)
        o3 = 2.0 * n * float(np.max(compute_conv(a3)))
        if o3 < best_obj:
            best_obj = o3
            best_a = a3.copy()
            if verbose:
                print(f"      MirrorDesc: {best_obj:.8f}")

    # Stage 4: LP polish
    remaining = time_budget - (time.monotonic() - start)
    t4 = min(remaining * 0.2, 60.0)
    if t4 > 15:
        a4 = lp_polish(best_a, time_budget=t4, verbose=verbose)
        o4 = 2.0 * n * float(np.max(compute_conv(a4)))
        if o4 < best_obj:
            best_obj = o4
            best_a = a4.copy()
            if verbose:
                print(f"      LP: {best_obj:.8f}")

    # Stage 5: Peak balance
    remaining = time_budget - (time.monotonic() - start)
    t5 = min(remaining * 0.25, 40.0)
    if t5 > 10:
        a5 = peak_balance(best_a, time_budget=t5, verbose=verbose)
        o5 = 2.0 * n * float(np.max(compute_conv(a5)))
        if o5 < best_obj:
            best_obj = o5
            best_a = a5.copy()
            if verbose:
                print(f"      PeakBal: {best_obj:.8f}")

    # Stage 6: Final exact FISTA
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        a6 = fista_exact(best_a, time_budget=remaining, verbose=False)
        o6 = 2.0 * n * float(np.max(compute_conv(a6)))
        if o6 < best_obj:
            best_obj = o6
            best_a = a6.copy()
            if verbose:
                print(f"      FinalFISTA: {best_obj:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization for best step function sequence.
    Phases: parent loading, full optimization, multi-n sweep, diverse restarts, final polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception as e:
        print(f"No parent: {e}")

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Full optimization on parent ===
    print(f"\n=== Phase 1: Full Optimize (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 400.0)
    if p1_budget > 50:
        a_new = full_optimize(best_seq, time_budget=p1_budget, verbose=True)
        o_new = compute_obj(a_new)
        if o_new < best_score:
            best_score = o_new
            best_seq = a_new.copy()
            print(f"  Phase 1 result: {best_score:.8f}")
        else:
            print(f"  Phase 1: no improvement ({o_new:.8f} vs {best_score:.8f})")

    # === Phase 2: Multi-n exploration ===
    print(f"\n=== Phase 2: Multi-n ===")
    n_current = len(best_seq)
    n_options = sorted(set([
        max(200, n_current - 200),
        n_current,
        n_current + 100,
        n_current + 200,
        n_current + 400,
        500, 700, 900, 1100, 1300, 1600
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        if n_try == n_current:
            continue
        # Skip if too close to current
        if abs(n_try - n_current) < 100:
            continue
        a_try = resample_sequence(best_seq, n_try)
        # Quick optimization
        alloc = min(remaining * 0.15, 80.0)
        if alloc > 20:
            a_try = full_optimize(a_try, time_budget=alloc, verbose=False)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 3: Diverse restarts ===
    print(f"\n=== Phase 3: Diverse Restarts ===")
    n_best = len(best_seq)
    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break
        mode = r % 7
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with current best
        blend = 0.4
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()
        alloc = min(remaining * 0.15, 90.0)
        if alloc > 30:
            a_r = full_optimize(a_r, time_budget=alloc, verbose=False)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 4: Try different n values from scratch ===
    print(f"\n=== Phase 4: Fresh n ===")
    for n_fresh in [600, 1000, 1200, 1500]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        if abs(n_fresh - len(best_seq)) < 100:
            continue
        a_f = np.ones(n_fresh) / n_fresh
        alloc = min(remaining * 0.12, 70.0)
        if alloc > 20:
            a_f = full_optimize(a_f, time_budget=alloc, verbose=False)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Fresh n={n_fresh}: obj={best_score:.8f} **")
        else:
            print(f"  Fresh n={n_fresh}: obj={obj_f:.8f}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    if remaining > 30:
        alloc = min(remaining * 0.3, 80.0)
        a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True)
        o_lp = compute_obj(a_lp)
        if o_lp < best_score:
            best_score = o_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: {best_score:.8f} **")

    # Exact FISTA
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 80.0)
        a_fi = fista_exact(best_seq, time_budget=alloc, verbose=True)
        o_fi = compute_obj(a_fi)
        if o_fi < best_score:
            best_score = o_fi
            best_seq = a_fi.copy()
            print(f"  Final FISTA: {best_score:.8f} **")

    # Peak balance
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 40.0)
        a_pb = peak_balance(best_seq, time_budget=alloc, verbose=True)
        o_pb = compute_obj(a_pb)
        if o_pb < best_score:
            best_score = o_pb
            best_seq = a_pb.copy()
            print(f"  Final PB: {best_score:.8f} **")

    # One more smoothed FISTA with high tau
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 50.0)
        a_sf = fista_smoothed(best_seq, time_budget=alloc, tau_init=50.0, tau_end=500.0, verbose=True)
        o_sf = compute_obj(a_sf)
        if o_sf < best_score:
            best_score = o_sf
            best_seq = a_sf.copy()
            print(f"  Final SmoothFISTA: {best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
