def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    for solver in [cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    # Verify feasibility
                    conv_check = A @ g_val
                    if np.max(conv_check) <= M_val * (1 + 1e-6) + 1e-9:
                        return g_val, M_val
        except Exception:
            continue

    # Fallback to linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                g_val = g_val / s
                return g_val, float(res.x[n])
    except Exception:
        pass

    return None, None


def smooth_gradient(a, beta):
    """
    Compute gradient of smooth surrogate f(a) = (1/beta)*log(sum_k exp(beta*c_k(a))).
    Uses FFT for efficiency.
    """
    n = len(a)
    conv = compute_conv(a)
    shifted = beta * (conv - conv.max())
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()

    # Gradient: grad_m = sum_k p_k * (a[k-m] + a[m-k]) (with boundary handling)
    # This is the cross-correlation of a with p (padded)
    # We can compute via FFT: grad = ifft(fft(a) * conj(fft(p))) but need to be careful
    # Actually, let's compute directly using the convolution structure
    # grad_m = sum_k p[k] * (d/d a_m) conv[k]
    # d/d a_m conv[k] = a[k-m] + a[m-k] (interpreting a[j]=0 for j outside [0,n-1])
    # So grad = p * a (cross-correlation in a specific sense)

    # More carefully: conv[a,a][k] = sum_i a[i]*a[k-i]
    # d/d a[m] conv[k] = a[k-m] + a[m-k] (if both indices valid)
    # grad[m] = sum_k p[k] * (a[k-m] + a[m-k])

    # This equals: sum_k p[k]*a[k-m] + sum_k p[k]*a[m-k]
    # First term: (p * a)[m] in some correlation sense
    # Let's just compute via FFT:
    # sum_k p[k] * a[k-m] = (p cross-correlate a)[m]
    # sum_k p[k] * a[m-k] = (p convolve a)[m] (flipped)

    # Actually the simplest: grad[m] = sum_k p[k] * (a[k-m] + a[m-k])
    # = sum_i a[i] * (p[m+i] + p[m-i])  (reindexing)
    # This is the convolution of a with (p + flip(p)) evaluated at m... approximately

    # Let's just do it directly with FFT:
    size = 1
    while size < 2 * n:
        size <<= 1
    A = rfft(a, size)
    P = rfft(p, size)
    # The cross-correlation approach
    # grad[m] = sum_k p[k] * a[k-m] + sum_k p[k] * a[m-k]
    # First sum: (p * a)[m] where * denotes cross-correlation = ifft(conj(P) * A) -- not quite
    # Let's just compute it element by element using the structure
    # Actually for efficiency with n up to ~1000, direct computation is O(n^2) which is fine

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        for m in range(n):
            idx1 = k - m
            idx2 = m - k
            if 0 <= idx1 < n:
                grad[m] += pk * a[idx1]
            if 0 <= idx2 < n and idx2 != idx1:
                grad[m] += pk * a[idx2]
            elif 0 <= idx2 < n and idx2 == idx1:
                grad[m] += pk * a[idx2]

    # Better: use FFT-based approach
    # conv[a,a][k] derivative wrt a[m] = a[k-m] (if valid) + a[m-k] (if valid, k!=m case)
    # Actually for the full convolution: d conv[k]/da[m] = a[k-m] + a[m-k] but
    # when k-m == m-k (i.e., k=2m... no that's not right)
    # Let me just use the direct O(n^2) which is fine for n < 2000

    return grad


def smooth_gradient_fft(a, beta):
    """Efficient gradient computation using FFT."""
    n = len(a)
    conv = compute_conv(a)
    shifted = beta * (conv - conv.max())
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()

    # grad[m] = sum_k p[k] * (a[k-m] + a[m-k])
    # Let's define b = [a, zeros(n-1)] and c = [p]
    # Then sum_k p[k] * a[k-m] is a cross-correlation
    # And sum_k p[k] * a[m-k] is a convolution of p (reversed) with a

    # Using FFT:
    size = 1
    while size < 2 * n + 1:
        size <<= 1

    A = fft(a, size)
    P = fft(p, size)

    # corr1[m] = sum_k p[k] * a[k-m] = ifft(conj(P) * A)[m] -- this gives correlation
    # Actually: corr[m] = sum_k p[k] * a[k+m] ... I need to be careful with conventions
    # Let me just use: ifft(conj(P)*A)[m] = sum_k conj(P)_k * A_k / size -- no
    # Cross-correlation: (p ⋆ a)[m] = sum_k p[k] * a[k+m] = ifft(conj(FFT(p)) * FFT(a))[m]
    # But I want sum_k p[k] * a[k-m] which is (p ⋆ a)[-m] = conjugate correlation

    # Let me just compute it directly - for n=800, O(n^2) = 640K operations, very fast
    grad = np.zeros(n)
    for m in range(n):
        s = 0.0
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-14:
                continue
            i1 = k - m
            i2 = m - k
            if 0 <= i1 < n:
                s += pk * a[i1]
            if 0 <= i2 < n:
                s += pk * a[i2]
        grad[m] = s

    return grad

def project_simplex(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def local_search(a, time_budget, verbose=False):

            a_try = a.copy()
            block = a_try[start_idx:start_idx + block_size]
            block = block + temp * block * rng.normal(0, 1, block_size)

        else:

                        a_try[j] *= (1 - temp * rng.uniform(0.5, 1.5))

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        a = a / a.sum()
    elif mode % 8 == 6:
        freqs = rng.integers(2, 8, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n) * 0.3)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining enhanced LP alternating, smooth optimization,
    local search, n-sweep, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"\nInitial: n={n_current}, obj={best_score:.8f}, target=1.5027")

    # === Phase 1: Quick LP polish ===
    print(f"\n=== Phase 1: Quick LP Polish ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.12, 120.0)
    if p1_budget > 20:
        a_p1 = lp_alternate_enhanced(best_seq, time_budget=p1_budget, verbose=True)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_p1:.8f}")

    # === Phase 2: Smooth optimization ===
    print(f"\n=== Phase 2: Smooth Optimization ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.10, 80.0)
    if p2_budget > 15:
        a_sm = smooth_optimize(best_seq, time_budget=p2_budget, verbose=False)
        obj_sm = compute_obj(a_sm)
        marker = " **" if obj_sm < best_score else ""
        print(f"  Smooth: obj={obj_sm:.8f}{marker}")
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()

    # Quick LP after smooth
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_p2 = lp_alternate_enhanced(best_seq, time_budget=min(remaining * 0.05, 40.0), verbose=False)
        obj_p2 = compute_obj(a_p2)
        if obj_p2 < best_score:
            best_score = obj_p2
            best_seq = a_p2.copy()
            print(f"  LP after smooth: obj={best_score:.8f} **")

    # === Phase 3: n-sweep ===
    print(f"\n=== Phase 3: n-Sweep ===")
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.18
    n_options = sorted(set([
        max(400, n_current - 100),
        max(400, n_current - 50),
        max(400, n_current - 20),
        max(400, n_current - 10),
        n_current,
        n_current + 10,
        n_current + 20,
        n_current + 50,
        n_current + 100,
        n_current + 150,
        700, 750, 768, 780, 790, 800, 810, 820, 830, 840, 850, 860, 880, 900, 960, 1000, 1024,
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1300]

    per_n_budget = p3_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_n_budget, remaining * 0.12)
        if alloc < 10:
            alloc = 10

        a_try = resample_sequence(best_seq, n_try)
        a_opt = lp_alternate_enhanced(a_try, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)
        marker = " **" if obj_opt < best_score else ""
        print(f"  n={n_try}: obj={obj_opt:.8f}{marker}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try

    # === Phase 4: Intensive LP alternating ===
    print(f"\n=== Phase 4: Intensive LP (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.30, 400.0)

    n_rounds = 5
    per_round = p4_budget / n_rounds
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        alloc = min(per_round, remaining * 0.2)
        if alloc < 20:
            break
        if rnd == 0:
            a_start = best_seq.copy()
        elif rnd == 1:
            # Small perturbation
            rng_r = np.random.default_rng(rnd * 7919 + n_current + seed)
            a_start = best_seq * np.exp(rng_r.normal(0, 0.02, n_current))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        elif rnd == 2:
            # Smooth then LP
            a_start = smooth_optimize(best_seq, time_budget=min(alloc * 0.3, 30.0), verbose=False)
        elif rnd == 3:
            # Larger perturbation
            rng_r = np.random.default_rng(rnd * 104729 + n_current + seed)
            a_start = best_seq * np.exp(rng_r.normal(0, 0.05, n_current))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        else:
            # Blend with diverse init
            a_div = diverse_init(n_current, rng, mode=rnd)
            a_start = 0.7 * best_seq + 0.3 * a_div
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()

        a_lp = lp_alternate_enhanced(a_start, time_budget=alloc, verbose=(rnd < 2))
        obj_lp = compute_obj(a_lp)
        marker = " **" if obj_lp < best_score else ""
        print(f"  LP round {rnd}: obj={obj_lp:.8f}{marker}")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()

    # === Phase 5: Local search ===
    print(f"\n=== Phase 5: Local Search ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.15, 100.0)
    if p5_budget > 15:
        a_ls = local_search(best_seq, time_budget=p5_budget, verbose=True)
        obj_ls = compute_obj(a_ls)
        marker = " **" if obj_ls < best_score else ""
        print(f"  Local search: obj={obj_ls:.8f}{marker}")
        if obj_ls < best_score:
            best_score = obj_ls
            best_seq = a_ls.copy()

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    alloc = min(remaining * 0.35, 60.0)
    if alloc > 15:
        a_final = lp_alternate_enhanced(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # One more local search
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.25, 30.0)
    if alloc > 10:
        a_ls2 = local_search(best_seq, time_budget=alloc, verbose=False)
        obj_ls2 = compute_obj(a_ls2)
        if obj_ls2 < best_score:
            best_score = obj_ls2
            best_seq = a_ls2.copy()
            print(f"  Final local: obj={best_score:.8f} **")

    # Last LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = lp_alternate_enhanced(best_seq, time_budget=min(remaining * 0.5, 25.0), verbose=False)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"=== Target: 1.5027, Achieved: {best_score:.8f}, Gap: {best_score - 1.5027:.8f} ===")

    return result
