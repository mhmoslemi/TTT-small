"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Block coordinate LP (much faster than full LP) + full LP polishing +
adaptive n exploration + spectral initialization.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def full_lp_solve(a):
    """Solve LP: min M s.t. conv(a,g)_k <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix A (2n-1) x n
    # A[k, i] = a_norm[i] if k-i is in [0, n-1], else 0
    # i.e., (A @ g)[k] = sum_i a_norm[i] * g[k-i] (valid when 0<=k-i<n)
    #        = sum_i a_norm[k-i] * g[i] (valid when 0<=k-i<n)
    
    A = np.zeros((2*n-1, n))
    for i in range(n):
        for k in range(max(0, i-(n-1)), min(2*n-2, i+n-1)+1):
            j = k - i  # j in [0, n-1]
            A[k, i] = a_norm[j]
    
    # linprog: min c^T x s.t. A_ineq @ x <= b_ineq, A_eq @ x = b_eq, bounds
    # Variables: [g_0, ..., g_{n-1}, M]
    # c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ineq = np.zeros((2*n-1, n+1))
    A_ineq[:, :n] = A
    A_ineq[:, n] = -1.0  # conv_k - M <= 0
    b_ineq = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s, float(res.x[n])
    return None, None

def block_lp_solve(a, start, end):
    """Solve LP optimizing only a[start:end] given rest fixed."""
    n = len(a)
    B = end - start
    rest_sum = float(np.sum(a) - np.sum(a[start:end]))
    
    # Variables: [g_0, ..., g_{B-1}, M] where g_i = a[start+i]
    # Objective: minimize M
    # Constraints:
    #   For each k in 0..2n-2: conv_k <= M
    #   sum(g) = 1 - rest_sum
    #   g >= 0
    
    # Compute which k values are affected by the block
    # a[start+i] contributes to conv[k] for k in [start+i, start+i+n-1]
    # So affected k range: [start, end-1+n-1] = [start, start+B+n-2]
    
    # But we also need to check all k for the M >= fixed_conv constraint
    
    # Build the LP
    # For efficiency, only include constraints that involve block variables
    # Plus a single constraint M >= max_fixed
    
    # First compute the "fixed" part of convolution (contributions from non-block variables)
    a_fixed = a.copy()
    a_fixed[start:end] = 0
    conv_fixed = compute_conv(a_fixed)
    
    # Now for each k, conv_k = conv_fixed_k + sum_{i in block} a[start+i] * a[k-(start+i)] (valid terms)
    # The block contribution to conv[k] is:

    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)

    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s

    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build (2n-1) x n matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    k = np.arange(2 * n - 1)
    j = np.arange(n)

    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[mask] = a[diff[mask]]
    return A

def full_lp_solve(a):
    """Solve LP: min M s.t. conv(a,g)_k <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.array([1.0])

                  bounds=bounds, method='highs',

        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:

    return None, None

def lp_iteration(a, rng, n_t_candidates=10):
    """One LP iteration: solve LP, try multiple t values, return best."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    g, M = full_lp_solve(a_norm)
    if g is None:
        return a, None, None
    
    best_obj = np.inf
    best_a = a.copy()
    
    t_candidates = np.linspace(0.05, 1.0, n_t_candidates)
    for t in t_candidates:
        a_try = (1 - t) * a_norm + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_a = a_try.copy()
    
    return best_a, best_obj, M

def coordinate_descent(a, time_budget, max_rounds=200):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for round_num in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for i in range(n):

                    break
            # Optimize a[i] by trying values
            a_save = a[i]
            # The contribution of a[i] to conv:
            # (a*a)[k] involves a[i] * a[k-i] for valid k-i
            # Try a few values around current
            candidates = [a_save * 0.5, a_save * 0.75, a_save, a_save * 1.25, a_save * 1.5]
            # Also try 0 and redistribute
            candidates = [max(0, c) for c in candidates]
            
            best_val = a_save
            best_local_obj = best_obj
            
            for val in candidates:
                a[i] = val
                s_new = float(np.sum(a))
                if s_new < 1e-10:
                    a[i] = a_save
                    continue
                a_norm = a / s_new
                obj = compute_obj(a_norm)
                if obj < best_local_obj:
                    best_local_obj = obj
                    best_val = val
            
            a[i] = best_val
            s_new = float(np.sum(a))
            a = a / s_new
            if best_local_obj < best_obj - 1e-12:
                best_obj = best_local_obj
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s, float(res.x[n])
    return None, None


def lp_optimize(a, time_budget, verbose=True, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    iter_count = 0
    stall = 0
    last_t = 1.0

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget:
            break
        g, M = full_lp_solve(a)
        if g is None:
            stall += 1
            if stall > 5:
                break
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            continue

        current_peak = float(np.max(compute_conv(a)))
        ratio = M / (current_peak + 1e-15)

        t_candidates = np.unique(np.concatenate([
            np.linspace(0.1, 1.0, 8),
            [max(0.05, last_t * 0.5), max(0.05, last_t * 0.8), last_t,
             min(1.0, last_t * 1.3), min(1.0, last_t * 1.5)]
        ]))
        t_candidates = t_candidates[t_candidates <= 1.0]

        iter_best_obj = best_obj
        iter_best_a = best_a.copy()
        iter_best_t = last_t

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < iter_best_obj:
                iter_best_obj = obj_try
                iter_best_a = a_try.copy()
                iter_best_t = t

        iter_count += 1
        if iter_best_obj < best_obj - 1e-13:
            best_obj = iter_best_obj
            best_a = iter_best_a.copy()
            a = best_a
            last_t = iter_best_t
            stall = 0
            if verbose and iter_count % 50 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={last_t:.3f}, M/peak={ratio:.4f}, {elapsed:.0f}s")
        else:
            stall += 1
            if stall <= 3:
                a = best_a * np.exp(rng.normal(0, 0.01 * stall, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
            elif stall <= 8:
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
            else:
                break

    return best_a


def coord_descent_fast(a, time_budget, verbose=True):
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    n_rounds = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget:
            break
        round_improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            old_val = a[i]
            # Try a few multiplicative adjustments
            multipliers = [0.3, 0.5, 0.7, 1.3, 1.5, 2.0]
            best_local = old_val
            best_local_obj = best_obj
            for mult in multipliers:
                val = old_val * mult
                a[i] = val
                s_new = float(np.sum(a))
                if s_new < 1e-10:
                    a[i] = old_val
                    continue
                a_n = a / s_new
                obj = compute_obj(a_n)
                if obj < best_local_obj:
                    best_local_obj = obj
                    best_local = val
            if best_local != old_val:
                a[i] = best_local
                a = a / a.sum()
                if best_local_obj < best_obj - 1e-13:
                    best_obj = best_local_obj
                    best_a = a.copy()
                    round_improved = True
            else:
                a[i] = old_val

        n_rounds += 1
        if not round_improved:
            no_improve += 1
            if no_improve >= 2:
                break
        else:
            no_improve = 0
        if verbose and n_rounds % 5 == 0:
            print(f"    CD round {n_rounds}: obj={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        a = np.linspace(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = (x - 0.5) ** 2 + 0.1
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = 1.0 / (0.1 + np.abs(x - 0.5))
        a = a / a.sum()
    elif mode == 5:
        freqs = rng.integers(1, 5, size=2)
        phases = rng.uniform(0, 2 * np.pi, size=2)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Two-hump
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.15) ** 2) + np.exp(-8 * (x - 0.85) ** 2) + 0.05
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj_parent = compute_obj(parent)
                best_score = obj_parent
                best_seq = parent.copy()
                n_parent = parent.size
                print(f"Parent (n={parent.size}): obj={obj_parent:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    # === Phase 1: LP optimization on parent (60% time) ===
    print(f"\n=== Phase 1: LP on n={n_parent} ===")
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.55
    if p1_budget > 60:
        a_opt = lp_optimize(best_seq, time_budget=p1_budget, verbose=True, rng=rng)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  LP done: obj={best_score:.8f} **")
        else:
            print(f"  LP done: obj={obj_opt:.8f} (no improvement)")

    # === Phase 2: Coordinate descent polish (10%) ===
    print(f"\n=== Phase 2: Coord Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.15
    if p2_budget > 20:
        a_cd = coord_descent_fast(best_seq, time_budget=p2_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD done: obj={best_score:.8f} **")
        else:
            print(f"  CD done: obj={obj_cd:.8f}")

    # === Phase 3: LP again on polished (10%) ===
    print(f"\n=== Phase 3: LP Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.2
    if p3_budget > 30:
        a_lp2 = lp_optimize(best_seq, time_budget=p3_budget, verbose=True, rng=rng)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  LP2 done: obj={best_score:.8f} **")
        else:
            print(f"  LP2 done: obj={obj_lp2:.8f}")

    # === Phase 4: Multi-n exploration (15%) ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(300, n_current - 200),
        max(300, n_current - 100),
        n_current + 100,
        n_current + 200,
        n_current + 300,
        500, 700, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000 and n != n_current]

    n_per_try = remaining / (2 * max(1, len(n_options[:5])))
    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(n_per_try, remaining * 0.3)
        if alloc < 20:
            alloc = 20

        a_try = resample_sequence(best_seq, n_try)
        a_try = lp_optimize(a_try, time_budget=alloc, verbose=False, rng=rng)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts (remaining time) ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        modes = [1, 2, 3, 4, 5, 6]
        n_start = len(best_seq)
        best_n_options = [n_start, max(400, n_start - 100), min(1500, n_start + 200)]

        per_restart = remaining / (len(modes) * len(best_n_options))
        tested = 0
        for mode in modes:
            for n_try in best_n_options:
                remaining = deadline - time.monotonic()
                if remaining < 30:
                    break
                alloc = min(per_restart, remaining * 0.25)
                if alloc < 15:
                    break
                a_try = diverse_init(n_try, rng, mode)
                a_try = lp_optimize(a_try, time_budget=alloc, verbose=False, rng=rng)
                obj_try = compute_obj(a_try)
                tested += 1
                if obj_try < best_score:
                    best_score = obj_try
                    best_seq = a_try.copy()
                    print(f"  ** mode={mode}, n={n_try}: obj={best_score:.8f} **")
                elif tested % 3 == 0:
                    print(f"  mode={mode}, n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = lp_optimize(best_seq, time_budget=min(remaining * 0.6, 120), verbose=True, rng=rng)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_final = coord_descent_fast(best_seq, time_budget=min(remaining * 0.8, 60), verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
