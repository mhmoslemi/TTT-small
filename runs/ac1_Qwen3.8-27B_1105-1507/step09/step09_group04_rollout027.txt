import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # (A @ g)[k] = sum_j A[k,j] * g[j] = sum_j a[k-j] * g[j] for valid k-j
        # A[k,j] = a[k-j] if 0 <= k-j < n, else 0
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, warm_start=None):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start
        except:
            pass

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def mirror_descent(a, time_budget, start_time, rng, verbose=False):
    """
    Exponentiated gradient / mirror descent for the simplex.
    a_new[i] = a[i] * exp(eta * g[i]) / Z
    where g is the negative subgradient (we want to minimize max conv).
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    eta = 1.0
    it = 0
    last_improve = 0
    
    while time.monotonic() - start_time < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient: grad f[k_star][m] = 2 * a[k_star - m] if 0 <= k_star - m < n
        grad = np.zeros(n)
        for m in range(n):
            idx = k_star - m
            if 0 <= idx < n:
                grad[m] = 2.0 * a[idx]
        
        # We want to DECREASE f, so move in direction -grad
        # Mirror descent: a_new[i] proportional to a[i] * exp(eta * (-grad[i] / ||grad||_inf))
        
        grad_max = float(np.max(np.abs(grad)))
        if grad_max < 1e-15:
            break
        grad_normed = -grad / grad_max
        
        # Exponentiated update
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a + eta * grad_normed
        # Subtract max for numerical stability
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new_s = float(np.sum(a_new))
        if a_new_s < 1e-10:
            break
        a_new /= a_new_s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            last_improve = it
            if verbose and it % 20 == 0:
                print(f"    MD it{it}: obj={best_obj:.8f}, eta={eta:.3f}")
        else:
            # Reduce learning rate
            eta *= 0.7
            if it - last_improve > 50:
                # Restart from best

                eta = 1.0
                last_improve = it
                if rng is not None:

                    a = np.maximum(a, 0)

        eta *= 0.995
        it += 1
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast coordinate descent, LP best-response iteration with Nesterov steps,
mirror descent (exponentiated gradient) polishing, cycling, and multi-length exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, warm_start=None):
    """Solve LP: min M s.t. (a_norm * g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def fast_coord_descent(a, time_budget, start_time, rng, verbose=False):
    """
    Fast coordinate descent with incremental convolution updates and
    adaptive golden section search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
    a = a / S
    conv = compute_conv(a)

    best_obj = float(np.max(conv))
    best_a = a.copy()
    gr = (np.sqrt(5) - 1) / 2.0
    stagnant = 0
    sweep = 0

    while time.monotonic() - start_time < time_budget:
        sweep += 1
        obj_before = float(np.max(conv))
        improved = False

        if rng is not None and sweep % 3 == 0:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for k in order:
            j_lo = k
            j_hi = k + n - 1
            d = a

            lo = -a[k] + 1e-15
            # Better upper bound: where the quadratic term makes things bad
            max_c = float(conv[j_lo:j_hi + 1].max())
            max_d = float(d.max())
            hi = max(2.0 / n, a[k] + max(0, (max_c - conv[2 * k]) / (2 * max_d + 1e-10)) if max_d > 1e-10 else 2.0 / n)
            hi = max(hi, lo + 1.0 / n)

            a1 = lo
            b1 = hi
            c1 = b1 - gr * (b1 - a1)
            d1 = a1 + gr * (b1 - a1)

            # Precompute base for speed
            base = conv[j_lo:j_hi + 1].copy()
            pos_2k = 2 * k - j_lo  # = k

            def eval_delta(delta):
                vals = base + 2.0 * delta * d
                if 0 <= pos_2k <= n - 1:
                    peak = float(base[pos_2k] + 2.0 * delta * d[pos_2k] + delta * delta)
                else:
                    peak = -1e30
                m = float(np.max(vals))
                if peak > m:
                    m = peak
                return m

            fc = eval_delta(c1)
            fd = eval_delta(d1)

            for _ in range(35):
                if fc < fd:
                    b1 = d1
                    d1 = c1
                    fd = fc
                    c1 = b1 - gr * (b1 - a1)
                    fc = eval_delta(c1)
                else:
                    a1 = c1
                    c1 = d1
                    fc = fd
                    d1 = a1 + gr * (b1 - a1)
                    fd = eval_delta(d1)

            delta_opt = (a1 + b1) / 2.0

            new_val = eval_delta(delta_opt)
            if new_val >= obj_before - 1e-14:
                continue

            a[k] += delta_opt
            a[k] = max(0.0, a[k])
            conv[j_lo:j_hi + 1] += 2.0 * delta_opt * d
            conv[2 * k] += delta_opt * delta_opt
            improved = True

            S_new = float(np.sum(a))
            if S_new > 1e-10:
                a /= S_new
                conv /= (S_new * S_new)
            else:
                a = best_a.copy()
                conv = compute_conv(a)
                break

        if improved:
            obj_after = float(np.max(conv))
            if obj_after < best_obj - 1e-12:
                best_obj = obj_after
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1

        if stagnant > 6:
            if rng is not None:
                scale = 0.008 * stagnant
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a /= s
                    conv = compute_conv(a)
                    stagnant = 0
                else:
                    break
            else:
                break

        if verbose and sweep % 30 == 0:
            print(f"    CD sweep {sweep}: obj={2*n*best_obj:.8f}, t={time.monotonic()-start_time:.0f}s")

    return best_a


def lp_best_response(a, time_budget, start_time, rng, verbose=False):
    """
    LP-based best-response iteration with Nesterov-like decreasing steps.
    At each step, solve LP to find best response g*, then blend.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    warm_g = None
    it = 0
    last_improve = 0
    momentum_a = a.copy()

    while time.monotonic() - start_time < time_budget:
        if time.monotonic() - start_time > time_budget:
            break

        g_lp, M_lp = solve_lp(a, warm_start=warm_g)
        if g_lp is None:
            break
        warm_g = g_lp.copy()
        it += 1

        # Nesterov-like momentum
        t_nest = 2.0 / (it + 1.5)
        momentum_a = a + t_nest * (a - best_a)  # extrapolation
        momentum_a = np.maximum(momentum_a, 0)
        sm = float(np.sum(momentum_a))
        if sm > 1e-10:
            momentum_a /= sm
        else:
            momentum_a = a.copy()

        # Try blending from both a and momentum_a
        bases = [a, momentum_a]
        t_candidates = [t_nest, min(1.0, t_nest * 2.5), 0.2, 0.4, 0.5, 0.5, 0.7, 0.9, 1.0]
        t_candidates = sorted(set([max(0.05, min(1.0, c)) for c in t_candidates]))

        best_step_obj = best_obj
        best_step_a = best_a.copy()

        for base in bases:
            for t in t_candidates:
                a_try = (1.0 - t) * base + t * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try /= s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_step_obj:
                    best_step_obj = obj_try
                    best_step_a = a_try.copy()

        if best_step_obj < best_obj - 1e-13:
            best_obj = best_step_obj
            best_a = best_step_a.copy()
            a = best_a
            last_improve = it
            if verbose and (it % 3 == 0 or best_obj < 1.503):
                print(f"    LP it{it}: obj={best_obj:.8f}, M_lp={M_lp:.6f}, gap={a@g_lp - best_obj/(2*n):.2e}")
        else:
            if rng is not None:
                scale = 0.015 + 0.005 * min(it // 10, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a /= a.sum()
            if it - last_improve > 25:
                break

    return best_a


def mirror_descent_polish(a, time_budget, start_time, rng, verbose=False):
    """
    Exponentiated gradient (mirror descent) on the simplex.
    Uses exact subgradient of the max convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    eta = 0.5
    it = 0
    last_improve = 0

    while time.monotonic() - start_time < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient: grad[m] = 2 * a[k_star - m] if valid
        grad = np.zeros(n)
        valid_m = np.arange(max(0, k_star - n + 1), min(n, k_star + 1))
        valid_idx = k_star - valid_m
        grad[valid_m] = 2.0 * a[valid_idx]

        gmax = float(np.max(np.abs(grad)))
        if gmax < 1e-15:
            break
        grad_normed = -grad / gmax

        # Exponentiated update
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a + eta * grad_normed
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new_s = float(np.sum(a_new))
        if a_new_s < 1e-10 or not np.isfinite(a_new_s):
            eta *= 0.5
            it += 1
            continue
        a_new /= a_new_s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            last_improve = it
            if verbose and it % 30 == 0:
                print(f"    MD it{it}: obj={best_obj:.8f}, eta={eta:.4f}")
        else:
            eta *= 0.8
            if it - last_improve > 40:
                a = best_a.copy()
                eta = 0.5
                if rng is not None:
                    a = a * np.exp(rng.normal(0, 0.008, n))
                    a = np.maximum(a, 0)
                    a /= a.sum()
                last_improve = it

        eta = max(0.01, eta * 0.998)
        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation (if available) or linear."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Arcsine distribution
        xx = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(np.maximum(1 - xx**2, 1e-8)))
        a = a / a.sum()
    elif mode == 3:
        # Raised cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 4:
        # Sine
        a = np.sin(np.pi * x)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 5:
        # Double Gaussian
        c1 = n // 4
        c2 = 3 * n // 4
        w = max(1, n // 10)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 6:
        # Power law
        a = x ** rng.uniform(-1, 3) + 0.001
        a = a / a.sum()
    elif mode == 7:
        # Sine mixture
        freqs = rng.choice([2, 3, 4, 5, 6], size=3, replace=False)
        phases = rng.uniform(0, 2 * np.pi, 3)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.25 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 8:
        # Triangular
        a = 1.0 - 2.0 * np.abs(x - 0.5)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def cycling_optimize(a, time_budget, start_time, rng, verbose=False):
    """
    Alternate between short CD bursts and LP steps.
    This helps escape local minima faster than running each method to convergence.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    warm_g = None
    cycle = 0

    while time.monotonic() - start_time < time_budget:
        remaining = time_budget - (time.monotonic() - start_time)
        if remaining < 10:
            break

        # CD burst (short)
        cd_budget = min(remaining * 0.4, 40.0)
        if cd_budget > 5:
            a_cd = fast_coord_descent(a, cd_budget, time.monotonic(), rng)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_obj:
                best_obj = obj_cd
                best_a = a_cd.copy()
                a = a_cd

        # LP step
        remaining = time_budget - (time.monotonic() - start_time)
        if remaining < 10:
            break
        g_lp, M_lp = solve_lp(a, warm_start=warm_g)
        if g_lp is None:
            break
        warm_g = g_lp

        # Try a few step sizes
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try /= s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                break

        cycle += 1
        if verbose and cycle % 5 == 0:
            print(f"    Cycle {cycle}: obj={best_obj:.8f}, t={time.monotonic()-start_time:.0f}s")

        # Perturb if stuck
        if cycle % 8 == 0 and rng is not None:
            a = best_a * np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 0)
            a /= a.sum()

    return best_a

            print(f"  P2 LP: obj={best_score:.8f} **")
        else:
            print(f"  P2 LP: obj={obj_r:.8f}")

    # === Phase 3: Fine CD ===
    print(f"\n=== Phase 3: Fine CD ===")
    remaining = deadline - time.monotonic()
    p3 = min(remaining * 0.18, 180.0)
    if p3 > 20:
        a_r = fast_coord_descent(best_seq, p3, time.monotonic(), rng, verbose=True)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  P3 CD: obj={best_score:.8f} **")
        else:
            print(f"  P3 CD: obj={obj_r:.8f}")

    # === Phase 4: Cycling (CD + LP alternating) ===
    print(f"\n=== Phase 4: Cycling ===")
    remaining = deadline - time.monotonic()
    p4 = min(remaining * 0.15, 150.0)
    if p4 > 20:
        a_r = cycling_optimize(best_seq, p4, time.monotonic(), rng, verbose=True)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  P4 Cycle: obj={best_score:.8f} **")
        else:
            print(f"  P4 Cycle: obj={obj_r:.8f}")

    # === Phase 5: Mirror Descent ===
    print(f"\n=== Phase 5: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p5 = min(remaining * 0.08, 80.0)
    if p5 > 15:
        a_r = mirror_descent_polish(best_seq, p5, time.monotonic(), rng, verbose=True)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  P5 MD: obj={best_score:.8f} **")
        else:
            print(f"  P5 MD: obj={obj_r:.8f}")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    n_options = sorted(set([
        n_cur + 200, n_cur + 400, n_cur + 600,
        max(400, n_cur - 200),
        1000, 1200, 1500, 2000
    ]))
    n_options = [n for n in n_options if 500 <= n <= 2500 and abs(n - n_cur) > 50]

    for n_try in n_options[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        # CD
        alloc1 = min(remaining * 0.35, 50.0)
        if alloc1 > 10:
            a_try = fast_coord_descent(a_try, alloc1, time.monotonic(), rng)

        # LP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.25, 35.0)
        if alloc2 > 10:
            a_try = lp_best_response(a_try, alloc2, time.monotonic(), rng)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_cur = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse Restarts ===
    print(f"\n=== Phase 7: Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 70:
            break

        mode = (r + 2) % 9
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.4 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r /= a_r.sum()

        alloc1 = min(remaining * 0.4, 40.0)
        if alloc1 > 10:
            a_r = fast_coord_descent(a_r, alloc1, time.monotonic(), rng)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.25, 25.0)
        if alloc2 > 10:
            a_r = lp_best_response(a_r, alloc2, time.monotonic(), rng)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Final Intensive Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # CD
    if remaining > 30:
        alloc = min(remaining * 0.35, 80.0)
        a_f = fast_coord_descent(best_seq, alloc, time.monotonic(), rng, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # LP
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.35, 60.0)
        a_f = lp_best_response(best_seq, alloc, time.monotonic(), rng, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # One more CD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.45, 45.0)
        a_f = fast_coord_descent(best_seq, alloc, time.monotonic(), rng)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
