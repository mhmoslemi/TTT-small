Multi-strategy optimizer for minimizing the step function objective
2*n*max(conv(a,a))/(sum(a))^2 using LP direction finding,

import numpy as np

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minstreak(a, mode='max'):
    """
    Solve LP to find a direction g on the simplex.
    mode='max': min M s.t. conv(a,g)[k] <= M for all k
    mode='top': min sum of conv(a,g)[k] for k in top peaks
    mode='l2': min sum of (conv(a,g)[k])^2 (approximated linearly)
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(g) == 1]
    
    if mode == 'max':
        M = cp.Variable()
        obj = M
        constraints.append(A @ g <= M)
    elif mode == 'top':
        # Focus on top peaks of conv(a,g)
        # Use current conv(a,a) peaks as proxy
        conv_aa = compute_conv(a_norm)
        top_k = np.argsort(conv_aa)[-max(10, n)//2:]
        obj = cp.sum(A[top_k] @ g)
    else:
        # Linear approximation of L2: use current peaks as weights
        conv_aa = compute_conv(a_norm)
        weights = conv_aa / np.sum(conv_aa)
        obj = weights @ (A @ g)
    
    prob = cp.Problem(cp.Minimize(obj), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.ECOS_BB]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(cp.Value(obj).value) if hasattr(cp, 'Value') else None
        except Exception:
            continue
    return None, None


def armijo_line_search(a, direction, f0, c1=1e-4, rho=0.5, max_iter=15):
    """
    Armijo backtracking line search.
    Find t such that f(a + t*(direction - a)) <= f0 + c1 * t * (direction-a)^T * grad_approx
    Simplified: just check if objective decreases.
    """
    n = len(a)
    t = 1.0
    for _ in range(max_iter):

        a_try = np.maximum(a_try, 0)

            t *= rho
            continue
        a_try = a_try / s
        f_try = compute_obj(a_try)
        if f_try < f0:
            return a_try, t
        t *= rho
    return a, 0.0


def power_sum_smoothed_gradient(a, p):
    """
    Compute gradient of (1/(2n-1)) * sum_k conv(a,a)[k]^p.
    Using FFT for efficiency.
    """
    n = len(a)
    conv = compute_conv(a)
    
    # weights_k = p * conv[k]^(p-1) / (2n-1)
    if p == 1:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = p * np.power(np.maximum(conv, 1e-30), p - 1) / (2 * n - 1)
    
    # gradient[i] = 2 * sum_k weights[k] * a[k-i] (with bounds)
    # This is 2 * conv(weights, a)[i] ... wait, let me think.
    # d/d a_i [sum_k weights[k] * (sum_j a[j]*a[k-j])]
    # = 2 * sum_k weights[k] * a[k-i] (where k-i is valid)
    # = 2 * (weights * a)[i] in some sense
    
    # Actually: d/da_i sum_k w_k * conv(a,a)[k]
    # conv(a,a)[k] = sum_j a_j * a_{k-j}
    # d/da_i: sum_k w_k * (a_{k-i} + a_{k-i}) [when i appears in both slots]
    # = 2 * sum_k w_k * a_{k-i} (constrained to valid indices)
    
    # This is 2 * conv(w, a)[i] where conv is the appropriate truncated version
    # Let me use FFT:
    N = 1
    while N < 3 * n:
        N <<= 1
    W = fft(weights, N)
    A = fft(a, N)
    grad_full = 2.0 * ifft(W * A).real
    grad = grad_full[:n]
    
    return grad, conv

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]

def solve_lp_direction(a, mode='max'):
    """
    Solve LP to find a descent direction g on the simplex.
    mode='max': min M s.t. conv(a,g)[k] <= M for all k
    mode='top': min sum of conv(a,g)[k] for top-peak positions
    mode='weighted': min weighted sum with weights proportional to current peaks
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(g) == 1]

    if mode == 'max':
        M = cp.Variable()
        objective = M
        constraints.append(A @ g <= M)

        n_top = max(5, n // 4)

    elif mode == 'weighted':
        conv_aa = compute_conv(a_norm)
        max_c = float(np.max(conv_aa))
        weights = np.maximum(conv_aa / max_c, 0.01)
        conv_ag = A @ g
        objective = cp.sum(cp.multiply(weights, conv_ag))
        constraints = []
    else:

def armijo_step(a, direction, f0, max_iter=12):
    """Backtracking line search from a toward direction."""
    t = 1.0
    rho = 0.5
    for _ in range(max_iter):
        a_try = (1.0 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            t *= rho
            continue
        a_try = a_try / s
        f_try = compute_obj(a_try)
        if f_try < f0:
            return a_try
        t *= rho
    return None


def power_sum_gradient(a, p):
    """
    Compute gradient of (1/(2n-1)) * sum_k conv(a,a)[k]^p using FFT.
    Returns (gradient, conv).
    """
    n = len(a)
    conv = compute_conv(a)

    if p <= 1.0 + 1e-9:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = p * np.power(np.maximum(conv, 1e-30), p - 1) / (2 * n - 1)

    N = 1
    while N < 3 * n:
        N <<= 1
    W = fft(weights, N)
    A = fft(a, N)
    grad_full = 2.0 * ifft(W * A).real
    grad = grad_full[:n]

    return grad, conv

def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse structured initializations."""
    x = np.arange(n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        # Two Gaussian peaks
        a = np.exp(-((x - n * 0.25) ** 2) / (2 * (n * 0.1) ** 2))
        a += np.exp(-((x - n * 0.75) ** 2) / (2 * (n * 0.1) ** 2))
    elif mode == 3:
        # Arcsine distribution
        t = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - t ** 2 + 1e-6))
    elif mode == 4:
        # Comb pattern
        period = max(2, n // 50)
        a = np.ones(n)
        a[::period] = 2.5
    elif mode == 5:
        # Exponential from center
        a = np.exp(-0.3 * np.abs(x - n / 2) / (n / 2))
    elif mode == 6:
        # Multi-frequency
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        t = x / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * t + p)
        a = np.maximum(a, 0.05)
    elif mode == 7:
        # Inverted parabola
        t = (x - n / 2) / (n / 2)
        a = 1.0 - 0.5 * t ** 2
    else:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def lp_optimize(a, time_budget=120.0, verbose=False):
    """
    LP-based optimization with cycling through multiple LP modes.
    Uses Armijo backtracking for step size.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(31415)

    it = 0
    stall = 0
    modes = ['max', 'top', 'weighted', 'max', 'top']  # cycle with more max

    while time.monotonic() - start < time_budget:
        mode = modes[it % len(modes)]
        g = solve_lp_direction(a, mode=mode)

        if g is None:
            stall += 1
            if stall > 15:
                break
            continue

        # Try full step first, then backtrack
        a_new = armijo_step(a, g, best_obj)

        if a_new is not None:
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
                it += 1
                continue

        # Try blending
        found = False
        for alpha in [0.6, 0.4, 0.25, 0.1]:
            a_try = (1 - alpha) * a + alpha * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                found = True
                break

        if found:
            stall = 0
        else:
            stall += 1
            if stall > 20:
                # Perturb to escape local minimum
                scale = 0.02 + 0.01 * min(stall - 20, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = float(np.sum(a))
                if s_a > 1e-10:
                    a = a / s_a
                else:
                    a = best_a.copy()
                stall = 0
                if it > 200:
                    break

        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it} ({mode}): obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def hybrid_optimize(a, time_budget=200.0, verbose=False):
    """
    Hybrid: alternate between LP steps and gradient steps.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    lr = 0.005
    v = np.zeros(n)

    while time.monotonic() - start < time_budget:
        # Every 3rd iteration: LP step
        if it % 3 == 0:
            g = solve_lp_direction(a, mode='max')
            if g is not None:
                a_new = armijo_step(a, g, best_obj)
                if a_new is not None:
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj - 1e-14:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new
                        it += 1
                        continue

        # Gradient step
        p = 8.0 + 16.0 * min(1.0, (time.monotonic() - start) / time_budget)
        grad, _ = power_sum_gradient(a, p)
        v = 0.8 * v - lr * grad
        a_new = project_simplex(a + v)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.7
            v *= 0.3

        it += 1
        lr = max(1e-7, lr * 0.9995)

        if verbose and it % 50 == 0:
            print(f"    HY it{it}: obj={best_obj:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization:
    Phase 1: LP-based optimization from parent
    Phase 2: Nesterov power-sum GD
    Phase 3: Peak surgery
    Phase 4: Hybrid LP+GD
    Phase 5: Multi-n exploration
    Phase 6: Diverse restarts
    Phase 7: Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent: {e}")

    if best_seq is None:
        best_seq = diverse_init(800, rng, mode=0)
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    # === Phase 1: LP-based optimization ===
    print(f"\n=== Phase 1: LP (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p1 = min(remaining * 0.25, 300.0)
    if p1 > 30:
        a1 = lp_optimize(best_seq, time_budget=p1, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  P1 done: obj={best_score:.8f} **")
        else:
            print(f"  P1 done: obj={obj1:.8f}")

    # === Phase 2: Nesterov power-sum GD ===
    print(f"\n=== Phase 2: NSGD ===")
    remaining = deadline - time.monotonic()
    p2 = min(remaining * 0.2, 200.0)
    if p2 > 20:
        a2 = nesterov_power_sum_gd(best_seq, time_budget=p2,
                                    p_init=4.0, p_end=256.0, lr_init=0.006,
                                    beta=0.85, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  P2 done: obj={best_score:.8f} **")
        else:
            print(f"  P2 done: obj={obj2:.8f}")

    # === Phase 3: Peak surgery ===
    print(f"\n=== Phase 3: Peak Surgery ===")
    remaining = deadline - time.monotonic()
    p3 = min(remaining * 0.1, 80.0)
    if p3 > 10:
        a3 = peak_surgery(best_seq, time_budget=p3, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  P3 done: obj={best_score:.8f} **")
        else:
            print(f"  P3 done: obj={obj3:.8f}")

    # === Phase 4: Hybrid LP+GD ===
    print(f"\n=== Phase 4: Hybrid ===")
    remaining = deadline - time.monotonic()
    p4 = min(remaining * 0.15, 150.0)
    if p4 > 20:
        a4 = hybrid_optimize(best_seq, time_budget=p4, verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  P4 done: obj={best_score:.8f} **")
        else:
            print(f"  P4 done: obj={obj4:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    n_candidates = [n for n in range(max(300, n_cur - 300), min(1500, n_cur + 400), 50)
                    if abs(n - n_cur) >= 100]
    # Add some specific values
    for extra in [400, 500, 600, 700, 900, 1000, 1100, 1200, 1300]:
        if extra not in n_candidates and 200 <= extra <= 1500 and abs(extra - n_cur) >= 100:
            n_candidates.append(extra)
    n_candidates = sorted(set(n_candidates))

    # Sort by proximity to current best n (try nearby first)
    n_candidates.sort(key=lambda x: abs(x - n_cur))

    for n_try in n_candidates[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 70:
            break

        a_try = resample_sequence(best_seq, n_try)

        alloc = min(remaining * 0.12, 40.0)
        if alloc > 15:
            a_try = lp_optimize(a_try, time_budget=alloc)

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_try = peak_surgery(a_try, time_budget=min(remaining * 0.04, 12.0))

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    used_modes = set()
    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        # Pick a mode not yet used
        mode = r
        while mode in used_modes and mode < 8:
            mode += 1
        used_modes.add(mode)

        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with current best (more weight on best)
        blend = 0.35
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        alloc = min(remaining * 0.12, 50.0)
        if alloc > 15:
            half = alloc * 0.6
            a_r = lp_optimize(a_r, time_budget=half)
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_r = peak_surgery(a_r, time_budget=min(remaining * 0.15, 15.0))

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    # LP
    if remaining > 40:
        alloc = min(remaining * 0.3, 100.0)
        a_f = lp_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # NSGD with high p
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.25, 60.0)
        a_f = nesterov_power_sum_gd(best_seq, time_budget=alloc,
                                     p_init=16.0, p_end=400.0, lr_init=0.003,
                                     beta=0.8)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final NSGD: obj={best_score:.8f} **")

    # Peak surgery
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.2, 30.0)
        a_f = peak_surgery(best_seq, time_budget=alloc)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final PS: obj={best_score:.8f} **")

    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 30.0)
        a_f = lp_optimize(best_seq, time_budget=alloc)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final micro-surgery
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.3, 15.0)
        a_f = peak_surgery(best_seq, time_budget=alloc)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Micro PS: obj={best_score:.8f} **")

    # Validation
    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
