import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    N = 1
    while N < size:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:size]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        length = min(n, 2 * n - 1 - j)
        A[start:start + length, j] = a[:length]
    return A


def solve_lp(a):

    Returns (g, M, dual_y) or (None, None, None)."""

    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)

def compute_conv_p(p, b, N):
    """Convolution of p and b using precomputed FFT size N."""
    P = rfft(p, N)
    B = rfft(b, N)
    size = len(p) + len(b) - 1
    return irfft(P * B, N)[:size]


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Cosine bump
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a + 1e-3
        a = a / a.sum()
    elif mode == 3:
        # Two bumps
        x = np.arange(n)
        a = np.exp(-((x - n//4)**2) / (n**2/32)) + np.exp(-((x - 3*n//4)**2) / (n**2/32))
        a = a / a.sum()
    elif mode == 4:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1, 1)
        a = a / a.sum()
    elif mode == 5:
        # Sinusoidal
        x = np.linspace(0, 1, n)
        f = rng.integers(2, 8)
        a = 0.5 + 0.3 * np.sin(2 * np.pi * f * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def perturb_sequence(a, rng, scale=0.03):
    """Apply a random perturbation."""
    n = len(a)
    method = rng.integers(0, 5)
    a = a.copy()
    if method == 0:
        a = a * np.exp(rng.normal(0, scale, n))
    elif method == 1:
        mask = rng.random(n) > 0.1
        a[mask] *= rng.uniform(1 - scale, 1 + scale, mask.sum())
    elif method == 2:
        # Shift
        shift = int(rng.integers(1, max(2, n // 20)))
        a = np.concatenate([a[-shift:], a[:-shift]])
    elif method == 3:
        # Fourier domain perturbation
        freqs = rfft(a, 2*n)
        freqs[1:-1] *= 1j * rng.normal(0, scale/10, len(freqs[1:-1]))
        a = irfft(freqs, 2*n)[:n]
        a = np.maximum(a, 0)
    else:
        # Add uniform noise
        a = a + rng.uniform(-scale, scale, n) * a.mean()
        a = np.maximum(a, 0)
    
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses intensive LP-based fixed-point iteration with adaptive blending,
periodic structural perturbations guided by LP dual variables, and multi-length exploration.
Key improvements: many more LP iterations, adaptive step schedules,
dual-guided perturbations, and careful time management.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    N = 1
    while N < size:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:size]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        length = min(n, 2 * n - 1 - j)
        A[start:start + length, j] = a[:length]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M, dual_y) or (None, None, None)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    cons_ineq = A @ g <= M
    cons_eq = cp.sum(g) == 1
    prob = cp.Problem(cp.Minimize(M), [cons_ineq, cons_eq])

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    try:
                        dy = np.asarray(cons_ineq.dual_value, dtype=np.float64)
                        dy = np.maximum(dy, 0)
                        dy_sum = float(np.sum(dy))
                        if dy_sum > 1e-15:
                            dual_y = dy / dy_sum
                        else:
                            dual_y = None
                    except Exception:
                        dual_y = None
                    return g_val, M_val, dual_y
        except Exception:
            continue
    return None, None, None


def lp_iterate(a, time_limit, max_iter=5000, t_schedule='adaptive', verbose=False):
    """

    Returns (best_a, best_obj, num_iters).
    """
    n = len(a)

    if s < 1e-10:
        return a, np.inf, 0
    a = a / s

    best_obj = compute_obj(a)

    it = 0
    consec_no_improve = 0

        if g is None:
            break

        frac = min(1.0, it / max_iter)

            candidates_t = [t, 0.5, 0.8, 1.0, 0.3, max(0.1, t * 0.5)]
        elif t_schedule == 'decreasing':

def dual_guided_perturb(a, dual_y, rng, scale=0.02):
    """Perturb sequence guided by LP dual variables (concentrated lags matter more)."""
    n = len(a)
    a = a.copy()

    if dual_y is not None:
        # Identify which lags are "hot" (high dual weight)
        top_lags = np.argsort(dual_y)[::-1][:10]
        # Compute contribution of each a[j] to hot lags
        pressure = np.zeros(n)
        for k in top_lags:
            k = int(k)
            w = dual_y[k]
            # a[j] contributes to conv[k] via a[j]*a[k-j]
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    pressure[j] += w * a[kj]
        max_p = np.max(pressure)
        if max_p > 1e-15:
            # Reduce high-pressure components, increase low-pressure
            a = a * np.exp(-scale * pressure / max_p)

    # Add small random perturbation
    a = a * np.exp(rng.normal(0, scale * 0.3, n))
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_perturb(a, rng, scale=0.03):
    """Apply a random perturbation."""
    n = len(a)
    method = rng.integers(0, 6)
    a = a.copy()

    elif method == 2:
        shift = int(rng.integers(1, max(2, n // 30)))
        a = np.concatenate([a[-shift:], a[:-shift]])
    elif method == 3:
        # Smooth perturbation in FFT domain
        N = 1
        while N < 2 * n:
            N <<= 1
        A = rfft(a, N)
        mask = np.ones(N // 2 + 1)
        mask[1:n // 3] = 1j * rng.normal(0, scale / 5, n // 3 - 1)
        mask[1:n // 3] = 1 + mask[1:n // 3]
        A = A * mask
        a = irfft(A, N)[:n]
        a = np.maximum(a, 0)
    elif method == 4:
        # Blend with uniform
        a = (1 - scale * 2) * a + scale * 2 / n
    else:
        # Gaussian noise proportional to a
        a = a + scale * a.mean() * rng.normal(0, 1, n)
        a = np.maximum(a, 0)

    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

            break

        a_new, obj_new, n_it = lp_iterate(a_current, time_limit=alloc,
                                          max_iter=3000, t_schedule='decreasing')
        total_lp_iters += n_it

        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_current = a_new.copy()
            print(f"  Final {fr}: obj={best_score:.10f} ({n_it} iters) **")
        else:
            a_current = a_new.copy()
            print(f"  Final {fr}: obj={obj_new:.10f} ({n_it} iters)")

            # Small perturbation for next round
            if remaining > 30:
                a_current = random_perturb(a_current, rng, scale=0.005)

    # === Validation ===
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, "
          f"LP_iters={total_lp_iters}, time={elapsed:.1f}s ===")

    return result
