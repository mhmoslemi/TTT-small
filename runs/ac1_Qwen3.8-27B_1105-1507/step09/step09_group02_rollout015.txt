def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        ks = range(j, min(j + n, 2 * n - 1))
        A[list(ks), j] = a[np.array(ks) - j]
    return A


def solve_lp(a, solver_list=None):
    """Solve LP: min M s.t. conv(a,g)<=M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def parabolic_line_search(a, g_lp, f0):

    Uses parabolic interpolation.
    Returns (best_t, best_a, best_f).

        if s < 1e-10:
            return np.inf
        a_try = a_try / s
        return compute_obj(a_try)

    best_f = min(f0, f1, f2, f3)
    best_t = 0.0 if best_f == f0 else t1 if best_f == f1 else t2 if best_f == f2 else t3
    
    # Parabolic interpolation for up to 5 iterations
    for _ in range(5):

        # t* = t2 - 0.5 * ((t2-t1)^2*(f2-f3) - (t2-t3)^2*(f2-f1)) / ((t2-t1)*(f2-f3) - (t2-t3)*(f2-f1))
        denom = (t2 - t1) * (f2 - f3) - (t2 - t3) * (f2 - f1)
        if abs(denom) < 1e-15:
            break
        t_star = t2 - 0.5 * ((t2 - t1)**2 * (f2 - f3) - (t2 - t3)**2 * (f2 - f1)) / denom
        t_star = max(0.01, min(0.99, t_star))
        
        f_star = eval_at(t_star)
        
        if f_star < best_f:

            best_t = t_star
        
        # Update the 3 points (keep the 3 best)
        points = [(t1, f1), (t2, f2), (t3, f3), (t_star, f_star)]
        points.sort(key=lambda x: x[1])
        (t1, f1), (t2, f2), (t3, f3) = points[0], points[1], points[2]

    a_best = np.maximum(a_best, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: LP polishing with parabolic line search for blend parameter,
FISTA smoothed optimization for fine-tuning, and multi-length exploration.
Key improvements over previous: proper parabolic line search for LP blending,
cleaner FISTA implementation, better time allocation, and adaptive restarts.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[k_start - j:k_end - j]
    return A


def solve_lp(a, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def parabolic_line_search(a, g_lp, f0, max_evals=12):
    """
    Find optimal blend parameter t for a_new = (1-t)*a + t*g_lp.
    Uses parabolic interpolation.
    Returns (best_t, best_a, best_f).
    """
    n = len(a)
    evals = 0

    def eval_at(t):
        nonlocal evals
        a_try = (1.0 - t) * a + t * g_lp
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            return np.inf
        a_try = a_try / s
        evals += 1
        return compute_obj(a_try)

    # Initial 3 points
    t1, t2, t3 = 0.2, 0.5, 0.8
    f1, f2, f3 = eval_at(t1), eval_at(t2), eval_at(t3)

    best_f = min(f0, f1, f2, f3)
    best_t = 0.0 if best_f == f0 else (t1 if best_f == f1 else (t2 if best_f == f2 else t3))

    # Parabolic interpolation
    while evals < max_evals:
        denom = (t2 - t1) * (f2 - f3) - (t2 - t3) * (f2 - f1)
        if abs(denom) < 1e-18:
            break
        t_star = t2 - 0.5 * ((t2 - t1)**2 * (f2 - f3) - (t2 - t3)**2 * (f2 - f1)) / denom
        t_star = max(0.02, min(0.98, t_star))

        if abs(t_star - t1) < 1e-4 or abs(t_star - t2) < 1e-4 or abs(t_star - t3) < 1e-4:
            break

        f_star = eval_at(t_star)

        if f_star < best_f:
            best_f = f_star
            best_t = t_star

        # Keep 3 best points
        points = sorted([(t1, f1), (t2, f2), (t3, f3), (t_star, f_star)], key=lambda x: x[1])
        (t1, f1), (t2, f2), (t3, f3) = points[0], points[1], points[2]

    a_best = (1.0 - best_t) * a + best_t * g_lp
    a_best = np.maximum(a_best, 0)
    a_best = a_best / np.sum(a_best)
    return best_t, a_best, best_f
