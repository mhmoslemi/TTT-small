def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_pair(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_mask = np.where(cond)[0]
    if len(rho_mask) == 0:
        return np.ones(n) / n
    rho = rho_mask[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    n = len(a)

    for k in range(2 * n - 1):

        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            B[k, j] = a[j]
    return B

def solve_lp_direction(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    B = build_conv_matrix_fast(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [B @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def lp_frank_wolfe(a, time_budget, rng, verbose=False):
    """
    LP-based Frank-Wolfe with convex line search.
    At each iteration:
    1. Solve LP to get g* (best direction)
    2. Precompute conv(a,a), conv(a,g*), conv(g*,g*)
    3. Golden section search on t to minimize max of the blended convolution
    4. Update a = (1-t)*a + t*g*
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    gr = (np.sqrt(5) - 1) / 2.0  # golden ratio conjugate
    stagnation = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        # Solve LP
        g_star, M_lp = solve_lp_direction(a)
        if g_star is None:
            break
        
        # Precompute convolutions
        conv_aa = compute_conv(a)
        conv_ag = compute_conv_pair(a, g_star)
        conv_gg = compute_conv(g_star)
        
        # Objective as function of t:
        # f(t) = max_k [(1-t)^2 * conv_aa[k] + 2t(1-t) * conv_ag[k] + t^2 * conv_gg[k]]
        # This is equivalent to max_k [conv_aa[k] + 2t(conv_ag[k] - conv_aa[k]) + t^2(conv_gg[k] - 2*conv_ag[k] + conv_aa[k])]
        # Let's just compute it directly for clarity
        
        lo_t = 0.0
        hi_t = 1.0
        
        def f_of_t(t):
            at = (1.0 - t) * a + t * g_star
            # We know sum(at) = 1 (both a and g_star sum to 1)
            # conv(at, at) = (1-t)^2 * conv_aa + 2t(1-t) * conv_ag + t^2 * conv_gg
            conv_at = (1-t)**2 * conv_aa + 2*t*(1-t) * conv_ag + t**2 * conv_gg
            return float(np.max(conv_at))
        
        # Golden section search
        c_t = hi_t - gr * (hi_t - lo_t)
        d_t = lo_t + gr * (hi_t - lo_t)
        fc = f_of_t(c_t)
        fd = f_of_t(d_t)
        
        for _ in range(60):
            if fc < fd:
                hi_t = d_t
                d_t = c_t
                fd = fc
                c_t = hi_t - gr * (hi_t - lo_t)
                fc = f_of_t(c_t)
            else:
                lo_t = c_t
                c_t = d_t
                fc = fd
                d_t = lo_t + gr * (hi_t - lo_t)
                fd = f_of_t(d_t)
        
        t_opt = (lo_t + hi_t) / 2.0
        
        # Apply update
        if t_opt < 1e-6:
            # No improvement
            stagnation += 1
            if stagnation > 5:
                # Perturb
                scale = 0.01 * (1 + 0.5 * stagnation)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_obj = compute_obj(a)
                stagnation = 0
            it += 1
            continue
        
        a_new = (1.0 - t_opt) * a + t_opt * g_star
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a = best_a.copy()
        else:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stagnation = 0
            else:
                stagnation += 1
                if stagnation > 5:
                    scale = 0.02 * (1 + 0.5 * stagnation)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    best_obj = compute_obj(a)
                    stagnation = 0
        
        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP-FW it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, time={elapsed:.0f}s")
    
    return best_a

            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv, float(M_var.value)
        except Exception:
            continue
    return None, None


def blended_max_conv(t, caa, cag, cgg):
    """Compute max of blended convolution at parameter t."""
    val = (1.0 - t) ** 2 * caa + 2.0 * t * (1.0 - t) * cag + t ** 2 * cgg
    return float(np.max(val))


def coord_descent(a, time_budget, rng, verbose=False):
    """
    Coordinate descent on normalized self-convolution peak.
    Uses golden section search per coordinate with incremental convolution updates.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    conv = compute_conv(a)

    best_obj = float(np.max(conv))
    best_a = a.copy()
    gr = (np.sqrt(5) - 1) / 2.0
    stagnant = 0
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        obj_before = float(np.max(conv))
        improved = False

        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for k in order:
            if time.monotonic() - start > time_budget:
                break

            # Direction: d[m] = a[m-k] for m in [k, k+n-1]
            d = np.zeros(2 * n - 1)
            d[k:k + n] = a

            lo_delta = -a[k]
            # Upper bound
            max_cur = float(np.max(conv))
            upper_est = np.sqrt(max(0.0, max_cur - conv[2 * k] + a[k] ** 2 + 0.1 * max_cur)) - a[k]
            hi_delta = max(upper_est, 3.0 / n)
            hi_delta = max(hi_delta, lo_delta + 2.0 / n)

            # Golden section search (40 iterations is plenty)
            a_gs = lo_delta
            b_gs = hi_delta
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)

            # Evaluate at c and d
            val_c = conv + 2.0 * c_gs * d
            val_c[2 * k] += c_gs * c_gs
            fc = float(np.max(val_c))

            val_d = conv + 2.0 * d_gs * d
            val_d[2 * k] += d_gs * d_gs
            fd = float(np.max(val_d))

            for _ in range(40):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    val_c = conv + 2.0 * c_gs * d
                    val_c[2 * k] += c_gs * c_gs
                    fc = float(np.max(val_c))
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    val_d = conv + 2.0 * d_gs * d
                    val_d[2 * k] += d_gs * d_gs
                    fd = float(np.max(val_d))

            delta_opt = (a_gs + b_gs) / 2.0

            # Check improvement
            val_new = conv + 2.0 * delta_opt * d
            val_new[2 * k] += delta_opt * delta_opt
            new_obj_est = float(np.max(val_new))

            if new_obj_est >= obj_before - 1e-15:
                continue

            # Apply update
            a[k] += delta_opt
            a[k] = max(0.0, a[k])
            conv += 2.0 * delta_opt * d
            conv[2 * k] += delta_opt * delta_opt

            # Re-normalize
            S_new = float(np.sum(a))
            if S_new < 1e-10:
                a[:] = 1.0 / n
                conv = compute_conv(a)
                improved = True
            else:
                a /= S_new
                conv /= (S_new * S_new)
                improved = True

        obj_after = float(np.max(conv))
        if obj_after < best_obj - 1e-13:
            best_obj = obj_after
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 3:
                if rng is not None:
                    scale = 0.01 * (1.0 + 0.5 * stagnant)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    S_p = float(np.sum(a))
                    if S_p > 1e-10:
                        a = a / S_p
                        conv = compute_conv(a)
                    best_obj = float(np.max(conv))
                    stagnant = 0
                else:
                    break

        if verbose:
            elapsed = time.monotonic() - start
            print(f"    CD: obj={2.0 * n * best_obj:.8f}, time={elapsed:.0f}s, stagnant={stagnant}")

    best_a = np.maximum(best_a, 0)
    S_b = float(np.sum(best_a))
    if S_b > 1e-10:
        best_a = best_a / S_b
    return best_a


def lp_frank_wolfe(a, time_budget, rng, verbose=False):
    """
    LP-based Frank-Wolfe with convex golden-section line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    gr = (np.sqrt(5) - 1) / 2.0
    stagnation = 0
    it = 0

    while time.monotonic() - start < time_budget:
        # Solve LP for best direction
        g_star, M_lp = solve_lp_direction(a)
        if g_star is None:
            break

        # Precompute cross-convolutions
        conv_aa = compute_conv(a)
        conv_ag = compute_conv_pair(a, g_star)
        conv_gg = compute_conv(g_star)

        # Golden section search on t in [0, 1]
        lo_t = 0.0
        hi_t = 1.0
        c_t = hi_t - gr * (hi_t - lo_t)
        d_t = lo_t + gr * (hi_t - lo_t)
        fc = blended_max_conv(c_t, conv_aa, conv_ag, conv_gg)
        fd = blended_max_conv(d_t, conv_aa, conv_ag, conv_gg)

        for _ in range(55):
            if fc < fd:
                hi_t = d_t
                d_t = c_t
                fd = fc
                c_t = hi_t - gr * (hi_t - lo_t)
                fc = blended_max_conv(c_t, conv_aa, conv_ag, conv_gg)
            else:
                lo_t = c_t
                c_t = d_t
                fc = fd
                d_t = lo_t + gr * (hi_t - lo_t)
                fd = blended_max_conv(d_t, conv_aa, conv_ag, conv_gg)

        t_opt = (lo_t + hi_t) / 2.0

        # Evaluate actual objective at t_opt
        a_new = (1.0 - t_opt) * a + t_opt * g_star
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a = best_a.copy()
            stagnation += 1
        else:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stagnation = 0
            else:
                # Try smaller step
                t_small = t_opt * 0.5
                a_try = (1.0 - t_small) * a + t_small * g_star
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        stagnation = 0
                    else:
                        stagnation += 1
                else:
                    stagnation += 1

        it += 1
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP-FW it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, time={elapsed:.0f}s")

        if stagnation > 8:
            # Perturb to escape
            if rng is not None:
                scale = 0.015 * (1.0 + 0.3 * stagnant)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                S_p = float(np.sum(a))
                if S_p > 1e-10:
                    a = a / S_p
                best_obj = compute_obj(a)
                stagnation = 0
            elif it < 3:
                continue
            else:
                break

        # Early stop if LP gives no improvement potential
        if M_lp is not None:
            cur_max = float(np.max(compute_conv(a)))
            if M_lp > cur_max * 0.995 and it > 20:
                break

    return best_a


def subgrad_descent(a, time_budget, rng, verbose=False):
    """
    Subgradient descent with Chebyshev step sizes and simplex projection.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    C = np.sqrt(2.0)  # diameter of simplex

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient: s[i] = a[k_star - i] + a[i - k_star] (with bounds)
        sg = np.zeros(n)
        for i in range(n):
            idx1 = k_star - i
            idx2 = i - k_star
            if 0 <= idx1 < n:
                sg[i] += a[idx1]
            if 0 <= idx2 < n:
                sg[i] += a[idx2]

        # Chebyshev step size
        t = it + 1
        eta = C / (t * (np.linalg.norm(sg) + 1e-15))

        # Update
        a_new = a - eta * sg
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        it += 1

        # Periodic check
        if it % 500 == 0 and verbose:
            elapsed = time.monotonic() - start
            print(f"    SubGrad it{it}: obj={best_obj:.8f}, time={elapsed:.0f}s")

        # Stagnation detection
        if it > 100 and it % 500 == 0:
            # Check if we've been stuck
            pass

        # Reseed if stuck
        if it % 2000 == 0 and it > 1000:
            obj_check = compute_obj(a)
            if abs(obj_check - best_obj) < 1e-10:
                if rng is not None:
                    a = best_a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 0)
                    S_p = float(np.sum(a))
                    if S_p > 1e-10:
                        a = a / S_p

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [max(1, n // 8), max(1, n // 8)]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2.0 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        c1 = rng.integers(0, max(1, n // 3))
        c2 = rng.integers(max(1, 2 * n // 3), n)
        w = max(1, n // 12)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2.0 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2.0 * w ** 2))
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
    elif mode == 8:
        # Arc-sine distribution
        x = np.linspace(1e-4, 1 - 1e-4, n)
        a = 1.0 / (np.pi * np.sqrt(x * (1 - x)))
        a = a / a.sum()
    elif mode == 9:
        # Beta(0.5, 0.5) = arc-sine, slightly different parameterization
        x = np.linspace(0.001, 0.999, n)
        a = 1.0 / np.sqrt(x * (1 - x))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)


def quick_cd(a, num_sweeps, rng):
    """Quick coordinate descent for a fixed number of sweeps."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    conv = compute_conv(a)
    gr = (np.sqrt(5) - 1) / 2.0

    for sweep in range(num_sweeps):
        order = rng.permutation(n) if rng is not None else np.arange(n)
        for k in order:
            d = np.zeros(2 * n - 1)
            d[k:k + n] = a

            lo_delta = -a[k]
            max_cur = float(np.max(conv))
            upper_est = np.sqrt(max(0.0, max_cur - conv[2 * k] + a[k] ** 2 + 0.1 * max_cur)) - a[k]
            hi_delta = max(upper_est, 3.0 / n)
            hi_delta = max(hi_delta, lo_delta + 2.0 / n)

            a_gs = lo_delta
            b_gs = hi_delta
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)

            val_c = conv + 2.0 * c_gs * d
            val_c[2 * k] += c_gs * c_gs
            fc = float(np.max(val_c))

            val_d = conv + 2.0 * d_gs * d
            val_d[2 * k] += d_gs * d_gs
            fd = float(np.max(val_d))

            for _ in range(30):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    val_c = conv + 2.0 * c_gs * d
                    val_c[2 * k] += c_gs * c_gs
                    fc = float(np.max(val_c))
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    val_d = conv + 2.0 * d_gs * d
                    val_d[2 * k] += d_gs * d_gs
                    fd = float(np.max(val_d))

            delta_opt = (a_gs + b_gs) / 2.0

            val_new = conv + 2.0 * delta_opt * d
            val_new[2 * k] += delta_opt * delta_opt
            new_obj_est = float(np.max(val_new))
            obj_before = float(np.max(conv))

            if new_obj_est >= obj_before - 1e-15:
                continue

            a[k] += delta_opt
            a[k] = max(0.0, a[k])
            conv += 2.0 * delta_opt * d
            conv[2 * k] += delta_opt * delta_opt

            S_new = float(np.sum(a))
            if S_new < 1e-10:
                a[:] = 1.0 / n
                conv = compute_conv(a)
            else:
                a /= S_new
                conv /= (S_new * S_new)

    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Quick CD on parent
    Phase 2: LP-FW with convex line search (main workhorse)
    Phase 3: Multi-n exploration with CD + LP-FW
    Phase 4: Subgradient descent polish
    Phase 5: Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # === Initialization from parent ===
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = diverse_init(n_target, rng, mode=8)  # arc-sine
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Quick Coordinate Descent ===
    print(f"\n=== Phase 1: Quick CD (n={n_target}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.12, 120.0)
    if p1_budget > 10:
        a_cd = coord_descent(best_seq, p1_budget, rng, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase1 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 CD: obj={obj_cd:.8f}")

    # === Phase 2: LP-Frank-Wolfe (main workhorse on current n) ===
    print(f"\n=== Phase 2: LP-FW (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 400.0)
    if p2_budget > 30:
        a_lp = lp_frank_wolfe(best_seq, p2_budget, rng, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP-FW: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP-FW: obj={obj_lp:.8f}")

    # Quick CD after LP
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_cd2 = coord_descent(best_seq, min(remaining * 0.08, 60.0), rng)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  Post-LP CD: obj={best_score:.8f} **")

    # === Phase 3: Multi-n Exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    n_current = len(best_seq)

    # Candidate n values: larger values give better asymptotic behavior
    n_candidates = []
    for delta_n in [200, 400, 600, 1000, 1500, 2000, 3000]:
        n_try = n_current + delta_n
        if n_current <= n_try <= 5000:
            n_candidates.append(n_try)
    # Also try some smaller
    for delta_n in [200, 400]:
        n_try = n_current - delta_n
        if 400 <= n_try < n_current:
            n_candidates.append(n_try)

    n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - n_current))
    # Prioritize larger n (better asymptotics)
    n_candidates = sorted(n_candidates, reverse=True)

    explored = 0
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        if explored >= 4:
            break

        # Time allocation for this n
        alloc_total = min(remaining * 0.25, 250.0)
        if alloc_total < 40:
            break

        # Resample
        a_try = resample_seq(best_seq, n_try)
        a_try = a_try / a_try.sum()

        # Quick CD
        alloc_cd = min(alloc_total * 0.3, 60.0)
        if alloc_cd > 10:
            a_try = coord_descent(a_try, alloc_cd, rng)

        # LP-FW
        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.15, alloc_total * 0.5, 120.0)
        if alloc_lp > 20:
            a_try = lp_frank_wolfe(a_try, alloc_lp, rng)

        # Final CD
        remaining = deadline - time.monotonic()
        alloc_cd2 = min(remaining * 0.05, 30.0)
        if alloc_cd2 > 5:
            a_try = coord_descent(a_try, alloc_cd2, rng)

        obj_try = compute_obj(a_try)
        explored += 1
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Subgradient Descent Polish ===
    print(f"\n=== Phase 4: Subgradient Descent ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 120.0)
    if p4_budget > 20:
        a_sg = subgrad_descent(best_seq, p4_budget, rng, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Phase4 SubGrad: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 SubGrad: obj={obj_sg:.8f}")

    # === Phase 5: Final Intensive Polish ===
    print(f"\n=== Phase 5: Final Polish (n={len(best_seq)}) ===")

    # LP-FW again
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.3, 150.0)
        a_lp_f = lp_frank_wolfe(best_seq, alloc, rng, verbose=True)
        obj_lp_f = compute_obj(a_lp_f)
        if obj_lp_f < best_score:
            best_score = obj_lp_f
            best_seq = a_lp_f.copy()
            print(f"  Final LP-FW: obj={best_score:.8f} **")

    # CD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 80.0)
        a_cd_f = coord_descent(best_seq, alloc, rng, verbose=True)
        obj_cd_f = compute_obj(a_cd_f)
        if obj_cd_f < best_score:
            best_score = obj_cd_f
            best_seq = a_cd_f.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # One more short LP
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 60.0)
        a_lp_f2 = lp_frank_wolfe(best_seq, alloc, rng)
        obj_lp_f2 = compute_obj(a_lp_f2)
        if obj_lp_f2 < best_score:
            best_score = obj_lp_f2
            best_seq = a_lp_f2.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final micro-CD
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.5, 30.0)
        a_cd_f2 = coord_descent(best_seq, alloc, rng)
        obj_cd_f2 = compute_obj(a_cd_f2)
        if obj_cd_f2 < best_score:
            best_score = obj_cd_f2
            best_seq = a_cd_f2.copy()
            print(f"  Final micro-CD: obj={best_score:.8f} **")

    # === Final Validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0)
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Verify with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
