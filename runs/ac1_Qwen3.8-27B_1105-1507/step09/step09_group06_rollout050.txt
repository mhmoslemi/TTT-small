```python
def diverse_init(n, rng, mode=0):
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x)
        return a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.cos(2 * np.pi * x)
        return a / a.sum()
    elif mode == 4:
        # Block structure
        block = n // 4
        a = np.concatenate([np.ones(block), np.zeros(n - 2*block), np.ones(block)])
        a = a + 0.1 * rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 5:
        # Gaussian mixture
        a = np.zeros(n)
        for _ in range(rng.integers(2, 6)):
            c = rng.integers(0, n)
            w = rng.integers(n//10, n//3)
            a += np.exp(-((np.arange(n) - c)**2) / (2*w**2))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        a = rng.uniform(0.1, 1, n)
        return a / a.sum()

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
```

```python
def golden_section_line_search_fast(a, g):
    """Fast line search using precomputed convolutions."""
    n = len(a)
    a_n = a / a.sum()
    g_n = g / g.sum()
    
    conv_a = compute_conv(a_n)
    conv_ag = compute_conv(a_n)  # Wait, this is conv(a,a), not conv(a,g)
    
    # Actually, we need conv(a, g) which is NOT the same as conv(a, a)
    # conv(a, g) = ∑_i a_i * g_{k-i}
    # This is a cross-correlation, not an autocorrelation
    # We can compute it as:
    # conv_ag[k] = ∑_i a[i] * g[k-i]
    # which is the same as np.convolve(a, g)
    
    conv_ag = np.convolve(a_n, g_n, mode='full')  # length 2n-1
    
    # Now f(t) = 2n * max((1-t)*conv_a + t*conv_ag) / ((1-t)*sum_a + t*sum_g)^2
    # Since we normalize, sum_a = sum_g = 1, so the denominator is 1
    # f(t) = 2n * max((1-t)*conv_a + t*conv_ag)
    
    # This is 2n * max_k [(1-t)*conv_a[k] + t*conv_ag[k]]
    # = 2n * max_k [conv_a[k] + t*(conv_ag[k] - conv_a[k])]
    
    diff = conv_ag - conv_a
    
    # The max of a set of linear functions of t is the upper envelope
    # We want to find t in [0,1] that minimizes this upper envelope
    
    # Coarse grid
    best_t = 0
    best_val = np.inf
    for t in np.linspace(0, 1, 51):
        val = 2 * n * np.max(conv_a + t * diff)
        if val < best_val:
            best_val = val
            best_t = t
    
    # Fine golden section
    if 0 < best_t < 1:
        gr = (np.sqrt(5) - 1) / 2
        lo = max(0, best_t - 0.02)
        hi = min(1, best_t + 0.02)
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)
        fc = 2 * n * np.max(conv_a + c * diff)
        fd = 2 * n * np.max(conv_a + d * diff)
        for _ in range(50):
            if fc < fd:
                hi, d, fd = d, c, fc
                c = hi - gr * (hi - lo)
                fc = 2 * n * np.max(conv_a + c * diff)
            else:
                lo, c, fc = c, d, fc
                d = lo + gr * (hi - lo)
                fd = 2 * n * np.max(conv_a + d * diff)
            if hi - lo < 1e-10:
                break
        best_t = (lo + hi) / 2
        best_val = 2 * n * np.max(conv_a + best_t * diff)
    
    a_new = (1 - best_t) * a_n + best_t * g_n
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a_n, best_val, best_t
    return a_new / s, compute_obj(a_new / s), best_t
```

conv(x,x) = (1-t)^2 conv(a,a) + 2t(1-t) conv(a,g) + t^2 conv(g,g)

def compute_conv(a):
    """Full linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]
