"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_of_max_conv(a, k):
    """
    Compute subgradient of max_k (a*a)[k] at position k.
    Returns vector g[i] = 2*a[k-i] if max(0,k-n+1) <= i <= min(k, n-1), else 0.
    """
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if lo <= hi:
        indices = np.arange(lo, hi + 1)
        g[indices] = 2.0 * a[k - indices]
    return g


def nesterov_exact_subgrad(a, time_budget=200.0, verbose=False):
    """
    Nesterov accelerated exact subgradient method on the simplex.
    Minimizes max_k (a*a)[k] subject to a>=0, sum(a)=1.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = float(np.max(compute_conv(a)))
    start = time.monotonic()

    # Nesterov parameters
    t = 1.0
    a_prev = a.copy()
    lr_base = 3.0 / np.sqrt(n)
    lr = lr_base
    it = 0
    best_it = 0
    stagnant = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start

        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = a + beta * (a - a_prev)
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute subgradient at y
        conv_y = compute_conv(y)
        k_star = int(np.argmax(conv_y))
        g = subgradient_of_max_conv(y, k_star)

        # Adaptive learning rate (backtracking)
        # Try to reduce the objective
        a_try = y - lr * g
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            a_try = best_a.copy()
        else:
            a_try = a_try / s_try

        obj_try = float(np.max(compute_conv(a_try)))

        if obj_try < float(np.max(conv_y)):
            # Accept step, maybe increase lr slightly
            lr = min(lr * 1.05, 10.0 / np.sqrt(n))
        else:
            # Reject, decrease lr
            lr = max(lr * 0.8, 0.01 / np.sqrt(n))
            a_try = y.copy()
            obj_try = float(np.max(conv_y))

        a_prev = a.copy()
        a = a_try
        t = t_new
        it += 1

        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a.copy()
            best_it = it
            stagnant = 0
        else:
            stagnant += 1

        # Reset if too stagnant
        if stagnant > 5000:
            a = best_a.copy()
            a_prev = a.copy()
            t = 1.0
            lr = lr_base
            stagnant = 0

        if verbose and it % 5000 == 0:
            print(f"    NES it{it}: obj={2*n*best_obj:.8f}, lr={lr:.4f}, t={elapsed:.0f}s")

    return best_a


def paired_coord_descent(a, time_budget=150.0, verbose=False, rng=None):
    """
    Paired coordinate descent: update pairs (a[i], a[j]) where i+j = k* (argmax conv position).
    This directly targets the dominant convolution term.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = float(np.max(conv))
    start = time.monotonic()
    iter_count = 0

    while time.monotonic() - start < time_budget:
        k_star = int(np.argmax(conv))
        
        # Find pairs (i, k_star - i) that contribute to conv[k_star]
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        
        # For each valid pair (i, k_star-i) with i < k_star-i, do a 2D update
        pairs_updated = 0
        for i in range(lo, hi + 1):
            j = k_star - i
            if j >= n or j < i:
                continue
            if j >= n:
                continue
                
            # conv[k_star] contains a[i]*a[j] (and a[j]*a[i] if i != j)
            # We want to reduce a[i]*a[j] while maintaining sum(a) = 1
            # Strategy: transfer mass from the larger to the smaller
            # For i != j: the pair contributes 2*a[i]*a[j] to conv[k_star]
            # For i == j: contributes a[i]^2
            
            # Simple: reduce the product by making them more equal
            # a[i]*a[j] is minimized (given sum a[i]+a[j]=const) when they're equal
            # But we also need to consider other convolution terms...
            
            # Use a small perturbation toward equality
            diff = a[i] - a[j]
            if abs(diff) < 1e-10:
                continue
            
            # Move a small amount from larger to smaller
            step = 0.1 * abs(diff)
            if a[i] > a[j]:
                a[i] -= step
                a[j] += step
            else:
                a[i] += step
                a[j] -= step
            
            a[i] = max(0, a[i])
            a[j] = max(0, a[j])
            pairs_updated += 1
        
        # Also do a global CD step on the most problematic coordinate
        k_star2 = int(np.argmax(compute_conv(a)))
        
        # Re-normalize
        s_new = float(np.sum(a))
        if s_new < 1e-10:
            a = best_a.copy()
            conv = compute_conv(a)
            break
        a = a / s_new
        conv = compute_conv(a)
        obj = float(np.max(conv))
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
        
        iter_count += 1
        if verbose and iter_count % 100 == 0:
            print(f"    PCD it{iter_count}: obj={2*n*best_obj:.8f}, t={time.monotonic()-start:.0f}s")

        # Add random perturbation periodically
        if iter_count % 500 == 0 and rng is not None:
            a = best_a * np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            conv = compute_conv(a)
            best_obj = float(np.max(conv))

    return best_a

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)

        return np.array([])

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
