from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu * rho > (cssv - 1.0)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_val = idx[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient_fft(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT.
    Returns (gradient, convolution values).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def build_conv_matrix(a):
    """Build matrix A_mat (2n-1 x n) such that A_mat @ g = convolve(a, g)."""
    n = len(a)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A_mat[j:j + r, j] = a[:r]
    return A_mat


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A_mat = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def nesterov_smoothed_descent(a, time_budget=300.0, tau_init=10.0, tau_end=500.0,

    Nesterov accelerated gradient descent on smoothed objective over simplex.

    lr = 4.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))

        # Cosine annealing for tau: start smooth, end sharp
        tau = tau_init + (tau_end - tau_init) * (1.0 - np.cos(np.pi * progress)) / 2.0

        # Periodic tau reset to escape local minima
        if it > 0 and it % 5000 == 0:
            tau = tau_init * 2.0

        if it % 3000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)

        # Barzilai-Borwein step size
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr_new = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 15.0 / np.sqrt(n))
                lr = lr_new

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1

        # Decay learning rate slowly
        lr = max(1e-8, lr * 0.9999)

        # Reset if stuck
        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 4.0 / np.sqrt(n)
            best_it = it

    return best_a


def coordinate_descent(a, time_budget=100.0, verbose=False, seed=0):
    """
    Cyclic coordinate descent with 1D line search on the smooth objective.
    For each coordinate, perform a golden section search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Precompute helper: for coordinate i, the "effective convolution" change
    # When we change a[i] from old to x (keeping sum=1 by adjusting another coord),
    # it's complex. Instead, we use a simpler approach:
    # Change a[i] by delta, then project back to simplex.

    it = 0
    while time.monotonic() - start < time_budget:
        # Cyclic order with random perturbation
        order = np.arange(n)
        rng.shuffle(order)

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            # Try to improve by adjusting a[i]
            # Use a small step in both directions
            old_val = a[i]
            step = 0.01 / np.sqrt(n)

            best_local_obj = best_obj
            best_local_val = old_val

            for delta in [step, -step, 2*step, -2*step, step*0.5, -step*0.5]:
                a_try = a.copy()
                a_try[i] = max(0, a_try[i] + delta)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_local_obj:
                    best_local_obj = obj_try
                    best_local_val = max(0, old_val + delta)

            if best_local_val != old_val:
                a[i] = best_local_val
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                if best_local_obj < best_obj:
                    best_obj = best_local_obj
                    best_a = a.copy()

        it += 1

    return best_a


def peak_aware_perturbation(a, conv, scale=0.05, seed=0):
    """
    Perturb the sequence with noise weighted by the autocorrelation peaks.
    Positions that contribute to peak values get more perturbation.
    """
    n = len(a)
    rng = np.random.default_rng(seed)

    # Identify which positions contribute to the top peaks
    top_k = np.argsort(conv)[::-1][:10]
    contrib = np.zeros(n)
    for k in top_k:
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if 0 <= j < n:
                contrib[i] += a[i] * a[j]

    # Normalize contribution
    max_c = contrib.max() if contrib.max() > 0 else 1.0
    contrib = contrib / max_c

    # Perturbation: larger where contribution is high
    noise = rng.normal(0, 1, n) * scale * (0.5 + contrib)
    a_new = a * np.exp(noise)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 6, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        c1 = rng.integers(0, n // 3)
        c2 = rng.integers(2 * n // 3, n)
        w = max(1, n // 12)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 8:
        # Smooth random with low-pass
        a = rng.uniform(0, 1, n)
        # Simple moving average
        k = max(2, n // 50)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 9:
        # Tapered (Hann window)
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def gd_lp_alternating(a, time_budget=300.0, verbose=False, seed=0):
    """
    Alternate between smoothed GD and LP polishing.
    This is the main optimization loop.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    gd_time_per_phase = 30.0
    lp_time_per_phase = 30.0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 20:
            break

        # GD phase
        gd_alloc = min(gd_time_per_phase, remaining * 0.4)
        if gd_alloc > 10:
            a_sgd = nesterov_smoothed_descent(a, time_budget=gd_alloc,
                                              tau_init=10.0, tau_end=400.0,
                                              verbose=False, seed=seed)
            obj_sgd = compute_obj(a_sgd)
            if obj_sgd < best_obj:
                best_obj = obj_sgd
                best_a = a_sgd.copy()
                a = a_sgd.copy()

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 20:
            break

        # LP phase
        lp_alloc = min(lp_time_per_phase, remaining * 0.4)
        if lp_alloc > 10:
            a_lp = lp_polish_aggressive(a, time_budget=lp_alloc,
                                        verbose=verbose, seed=seed)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()
                a = a_lp.copy()
            else:
                # Perturb
                conv = compute_conv(a)
                a = peak_aware_perturbation(a, conv, scale=0.03, seed=seed + int(time.monotonic()*1000))
                a = 0.8 * a + 0.2 * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()

        if verbose:
            elapsed = time.monotonic() - start
            print(f"    GD-LP alt: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a
