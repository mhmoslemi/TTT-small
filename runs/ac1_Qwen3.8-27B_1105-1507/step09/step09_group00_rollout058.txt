"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based steepest descent with golden-section line search, 
diverse asymmetric initialization, and adaptive perturbation for escape.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_normalized, g) where sum(a_normalized)=1."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = max(0, 0)  # k - (n-1) >= 0 => k >= n-1, but we want k-j in [0, n-1]
        # (a*g)[k] = sum_j a[j] * g[k-j], valid when 0<=j<n and 0<=k-j<n
        # So for fixed j, k ranges from j to j+n-1
        k_start = j
        k_end = min(2 * n - 1, j + n)  # exclusive
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, solver_list=None):
    """
    Solve LP: min M s.t. conv(a, g)[k] <= M for all k, sum(g)=1, g>=0.
    a should be normalized (sum=1).
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a = a / s
    
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.MOSEK, cp.SCS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    g_val = g_val / s_val
                    return g_val, float(M.value)
        except Exception:
            continue
    
    return None, None

def asymmetric_init(n, rng, mode=0):
    """Generate asymmetric non-negative initializations."""
    if mode == 0:
        # Ramp
        a = np.linspace(0.1, 2.0, n)
    elif mode == 1:
        # One-sided exponential
        a = np.exp(-np.arange(n) / (n * 0.3))
    elif mode == 2:
        # Asymmetric Gaussian
        center = n * rng.uniform(0.2, 0.4)
        width = n * rng.uniform(0.2, 0.5)
        a = np.exp(-((np.arange(n) - center) ** 2) / (2 * width ** 2))
    elif mode == 3:
        # Sawtooth
        a = np.concatenate([np.linspace(1, 0.1, n//2), np.linspace(0.1, 2, n - n//2)])
    elif mode == 4:
        # Random with asymmetry
        a = rng.uniform(0.5, 1.0, n) * np.linspace(1, 0.3, n)
    elif mode == 5:
        # Two asymmetric peaks
        c1 = n * rng.uniform(0.1, 0.3)
        c2 = n * rng.uniform(0.6, 0.9)
        w1 = n * rng.uniform(0.05, 0.15)
        w2 = n * rng.uniform(0.1, 0.3)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w1 ** 2)) + \
            0.5 * np.exp(-((np.arange(n) - c2) ** 2) / (2 * w2 ** 2))
    elif mode == 6:
        # Polynomial asymmetric
        x = np.linspace(0, 1, n)
        a = 1 + x + 0.5 * x**2
    elif mode == 7:
        # Sinusoidal asymmetric
        x = np.linspace(0, 1, n)
        a = 1 + 0.3 * np.sin(2 * np.pi * x + np.pi/4) + 0.2 * x
    elif mode == 8:
        # Piecewise: high at start, low in middle, medium at end
        a = np.ones(n)
        a[:n//4] *= 2.0
        a[n//4:3*n//4] *= 0.5
        a[3*n//4:] *= 1.0
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-6)
    a = a / a.sum()
    return a


def spectral_asymmetric_init(n, rng):
    """Initialize using asymmetric spectral components."""
    size = 1
    while size < 2 * n:
        size <<= 1
    freqs = np.fft.rfftfreq(size)
    # Asymmetric amplitude: boost positive frequencies
    mag = np.exp(-freqs * n * 0.3)
    # Add phase to create asymmetry
    phases = rng.uniform(-np.pi, np.pi, size=size // 2 + 1)
    # Force real-valued by conjugate symmetry
    F = mag * np.exp(1j * phases)
    # Ensure a[i] >= 0 by using appropriate phases
    a = np.fft.irfft(F, size)[:n]
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def smoothed_subgradient(a, tau=50.0):
    """Compute smoothed subgradient of max(conv(a,a)) using softmax weighting."""
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    weights = np.exp(tau * (conv - max_c))
    weights = weights / weights.sum()
    
    # Gradient of conv(a,a)[k] w.r.t. a[j] is 2*a[k-j] (for valid k-j)
    # So gradient of sum_k weights[k] * conv(a,a)[k] = 2 * conv(weights, a)
    # (with appropriate indexing)
    
    size = 1
    while size < 3 * n:
        size <<= 1
    W = rfft(weights, size)
    A = rfft(a, size)
    grad = irfft(W * A, size)[:n]
    return 2.0 * grad

import numpy as np
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g) where sum(a)=1."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a):

def smoothed_subgradient(a, tau=50.0):
    """Compute smoothed subgradient of max(conv(a,a)) using softmax weighting."""
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    weights = np.exp(tau * (conv - max_c))
    sum_w = weights.sum()
    if sum_w < 1e-300:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
    else:
        weights = weights / sum_w

    size = 1
    while size < 3 * n:
        size <<= 1
    W = rfft(weights, size)
    A = rfft(a, size)
    grad = irfft(W * A, size)[:n]
    return 2.0 * grad

        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

    return best_a


def asymmetric_init(n, rng, mode=0):
    """Generate diverse asymmetric non-negative initializations."""
    if mode == 0:
        a = np.linspace(0.1, 2.0, n)
    elif mode == 1:
        a = np.exp(-np.arange(n) / (n * rng.uniform(0.2, 0.5)))
    elif mode == 2:
        center = n * rng.uniform(0.15, 0.35)
        width = n * rng.uniform(0.15, 0.4)
        a = np.exp(-((np.arange(n) - center) ** 2) / (2 * width ** 2))
    elif mode == 3:
        a = np.concatenate([np.linspace(1.5, 0.2, n // 2), np.linspace(0.2, 1.0, n - n // 2)])
    elif mode == 4:
        a = rng.uniform(0.5, 1.0, n) * np.linspace(1.5, 0.3, n)
    elif mode == 5:
        c1 = n * rng.uniform(0.1, 0.3)
        c2 = n * rng.uniform(0.6, 0.85)
        w1 = n * rng.uniform(0.05, 0.12)
        w2 = n * rng.uniform(0.1, 0.25)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w1 ** 2)) + \
            0.4 * np.exp(-((np.arange(n) - c2) ** 2) / (2 * w2 ** 2))
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = 1.0 + 1.5 * x + 0.5 * x ** 2
    elif mode == 7:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.sin(2 * np.pi * 1.5 * x + 0.8) + 0.5 * x
    elif mode == 8:
        a = np.ones(n)
        a[:n // 3] *= 1.8
        a[n // 3:2 * n // 3] *= 0.4
        a[2 * n // 3:] *= 1.0
    elif mode == 9:
        # Geometric
        r = rng.uniform(0.95, 0.999)
        a = r ** np.arange(n)
    elif mode == 10:
        # Two-stage exponential
        a = np.exp(-np.arange(n) * rng.uniform(0.005, 0.02)) + \
            0.3 * np.exp(-(np.arange(n) - n * 0.7) ** 2 / (n * 0.1))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-6)
    a = a / a.sum()
    return a


def parametric_scan(n, time_budget=40.0):
    """Scan over parametric families to find good initializations."""
    best_a = None
    best_obj = np.inf
    start = time.monotonic()

    # Geometric sequences
    for r in np.linspace(0.90, 0.999, 30):
        if time.monotonic() - start > time_budget:
            break
        a = r ** np.arange(n)
        a = a / a.sum()
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    # Asymmetric Gaussians
    for center_frac in np.linspace(0.1, 0.5, 15):
        for width_frac in np.linspace(0.1, 0.5, 10):
            if time.monotonic() - start > time_budget:
                break
            c = center_frac * n
            w = width_frac * n
            a = np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
            a = a / a.sum()
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    # Power law asymmetric
    for alpha in np.linspace(0.5, 5.0, 20):
        if time.monotonic() - start > time_budget:
            break
        x = np.linspace(1e-6, 1, n)
        a = x ** alpha
        a = a / a.sum()
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def mirror_break_symmetry(a, rng):
    """Break mirror symmetry of a sequence by adding asymmetric perturbation."""
    n = len(a)
    # Create a strictly increasing perturbation
    ramp = np.linspace(0, 1, n)
    noise = rng.normal(0, 0.01, n)
    perturb = 0.02 * ramp + noise
    a_new = a * np.exp(perturb)
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    return a_new


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining LP steepest descent with golden-section line search,
    parametric scanning, diverse asymmetric initialization, and multi-phase strategy.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_default = 800
    parent_loaded = False
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_default = parent.size
                parent_loaded = True
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if not parent_loaded:
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    n = len(best_seq)

    # === Phase 1: Parametric scan for good asymmetric starts ===
    print(f"\n=== Phase 1: Parametric Scan ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.05, 45.0)
    if p1_budget > 10:
        a_param, obj_param = parametric_scan(n, time_budget=p1_budget)
        if obj_param < best_score:
            best_score = obj_param
            best_seq = a_param.copy()
            print(f"  Parametric: obj={best_score:.8f} **")
        else:
            print(f"  Parametric: obj={obj_param:.8f}")

    # === Phase 2: LP Steepest Descent on parent ===
    print(f"\n=== Phase 2: LP Steepest Descent (parent start) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 280.0)
    if p2_budget > 30:
        # Restore parent for this phase
        if parent_loaded:
            try:
                parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
                parent_arr = np.clip(parent_arr, 0, None)
                a_start = parent_arr / parent_arr.sum()
            except Exception:
                a_start = best_seq.copy()
        else:
            a_start = best_seq.copy()

        a2 = lp_steepest_descent(a_start, time_budget=p2_budget, verbose=True, seed=42)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj2:.8f}")

    # === Phase 3: Gradient polish ===
    print(f"\n=== Phase 3: Gradient Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.08, 70.0)
    if p3_budget > 15:
        a3 = gradient_polish(best_seq, time_budget=p3_budget, tau_init=30.0, tau_end=600.0)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj3:.8f}")

    # === Phase 4: LP Steepest Descent (second round, from improved point) ===
    print(f"\n=== Phase 4: LP Steepest Descent (round 2) ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.2, 250.0)
    if p4_budget > 30:
        a4 = lp_steepest_descent(best_seq, time_budget=p4_budget, verbose=True, seed=777)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase 4: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4: obj={obj4:.8f}")

    # === Phase 5: Diverse asymmetric restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()

    restart_modes = [0, 1, 3, 5, 6, 9, 10]
    n_restarts = min(len(restart_modes), 7)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = restart_modes[r]
        a_r = asymmetric_init(n, rng, mode=mode)

        # Blend with current best
        blend = 0.25 + 0.05 * r
        a_r = (1.0 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.12, 100.0)
        if alloc > 25:
            a_r = lp_steepest_descent(a_r, time_budget=alloc * 0.75, seed=2000 + r * 100)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_r = gradient_polish(a_r, time_budget=min(remaining * 0.3, 25.0))

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()

    n_current = len(best_seq)
    n_options_list = [
        max(600, n_current - 150), max(600, n_current - 50),
        min(1400, n_current + 50), min(1400, n_current + 100),
        min(1400, n_current + 200), n_current + 300
    ]
    # Add some fixed options
    n_options_list.extend([700, 850, 950, 1000, 1100, 1200])

    # Deduplicate and filter
    seen = set()
    n_options = []
    for x in sorted(n_options_list):
        if 500 <= x <= 1500 and x not in seen and x != n_current:
            seen.add(x)
            n_options.append(x)
    n_options = n_options[:5]

    print(f"  Current n={n_current}, trying: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 70:
            break

        # Resample best_seq to new length
        x_old = np.linspace(0, 1, len(best_seq))
        x_new = np.linspace(0, 1, n_try)
        a_try = np.interp(x_new, x_old, best_seq)
        a_try = np.maximum(a_try, 0)
        a_try = a_try / a_try.sum()

        alloc = min(remaining * 0.18, 120.0)
        if alloc > 30:
            a_try = lp_steepest_descent(a_try, time_budget=alloc * 0.7, seed=n_try)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_try = gradient_polish(a_try, time_budget=min(remaining * 0.3, 25.0))

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final intensive LP polishing ===
    print(f"\n=== Phase 7: Final LP Polish ===")
    remaining = deadline - time.monotonic()

    for rnd in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        alloc = min(remaining * 0.2, 120.0)
        a_final = lp_steepest_descent(best_seq, time_budget=alloc, verbose=False,
                                      seed=9000 + rnd * 100)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  Final LP {rnd}: obj={obj_final:.8f}")
            # Small asymmetric perturbation
            if rnd < 4:
                best_seq = mirror_break_symmetry(best_seq, rng)

    # === Phase 8: Ultra-fine gradient finish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 50.0)
        a_gp = gradient_polish(best_seq, time_budget=alloc, tau_init=100.0, tau_end=1000.0)
        obj_gp = compute_obj(a_gp)
        if obj_gp < best_score:
            best_score = obj_gp
            best_seq = a_gp.copy()
            print(f"  Final gradient: obj={best_score:.8f} **")

    # One more LP if time permits
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.4, 80.0)
        a_last = lp_steepest_descent(best_seq, time_budget=alloc, verbose=True, seed=99999)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Ultra-final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
