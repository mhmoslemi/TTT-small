import numpy as np

from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def enforce_symmetry(a):
    return (a + a[::-1]) / 2.0


def normalize(a):
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def solve_lp_response(a, max_constraints=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for k in active set, sum(g)=1, g>=0.
    Returns (g_opt, M_opt).
    """
    n = len(a)
    a = normalize(a)
    
    # Determine which constraints to include
    conv_aa = compute_conv(a)
    if max_constraints is not None and max_constraints < 2 * n - 1:
        # Only include top constraints
        top_idx = np.argsort(conv_aa)[::-1][:max_constraints]
        active_k = sorted(set([int(k) for k in top_idx]))
    else:
        active_k = list(range(2 * n - 1))
    
    nc = len(active_k)
    
    # Build constraint matrix: A[g] = (a*g)[k] for k in active_k
    # A is nc x n
    A = np.zeros((nc, n))
    for ci, k in enumerate(active_k):
        k = int(k)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            A[ci, idx] = a[k - idx]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g, symmetric=True, n_iter=30):
    """Find optimal t in [0,1] to minimize obj((1-t)*a + t*g)."""
    best_t = 0.0
    best_obj = compute_obj(a)
    
    # Ternary search
    lo, hi = 0.0, 1.0
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        
        a1 = (1 - m1) * a + m1 * g
        if symmetric:
            a1 = enforce_symmetry(a1)
        a1 = np.maximum(a1, 0)
        a1 = normalize(a1)
        obj1 = compute_obj(a1)
        
        a2 = (1 - m2) * a + m2 * g
        if symmetric:
            a2 = enforce_symmetry(a2)
        a2 = np.maximum(a2, 0)
        a2 = normalize(a2)
        obj2 = compute_obj(a2)
        
        if obj1 < obj2:
            hi = m2
        else:
            lo = m1
    
    best_t = (lo + hi) / 2
    a_best = (1 - best_t) * a + best_t * g
    if symmetric:
        a_best = enforce_symmetry(a_best)
    a_best = np.maximum(a_best, 0)
    a_best = normalize(a_best)
    best_obj = compute_obj(a_best)
    
    return a_best, best_obj, best_t

def subgradient_accelerated(a, time_budget, verbose=False, symmetric=True):
    """Accelerated subgradient method with top-K peaks."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = normalize(a)
    if symmetric:
        a = enforce_symmetry(a)
        a = normalize(a)

    start = time.monotonic()
    
    K = min(50, 2 * n - 1)  # number of peaks to use
    lr = 1.0
    momentum = 0.5
    v = np.zeros(n)
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        
        # Weighted subgradient
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        # Softmax weights
        weights = np.exp(5.0 * (peak_vals - max_pv) / max(max_pv, 1e-15))
        weights /= weights.sum()
        
        h = np.zeros(n)
        for w, k in zip(weights, top_idx):

                h[idx] += w * 2.0 * a[k - idx]
        
        # Update with momentum
        v = momentum * v + lr * h
        a_new = a - v
        
        if symmetric:
            a_new = enforce_symmetry(a_new)
        a_new = np.maximum(a_new, 0)
        a_new = normalize(a_new)

def resample_sequence(a, new_n, symmetric=True):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    if symmetric:
        a_new = enforce_symmetry(a_new)
    return normalize(a_new)
