def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def make_symmetric(a):
    """Make a sequence symmetric."""
    n = len(a)
    return (a + a[::-1]) / 2.0

    g = cp.Variable(n, nonneg=True)

    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    prob = cp.Problem(cp.Minimize(M), constraints)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses symmetry exploitation, LP-based polishing with fast blend search,
parametric initialization sweep, multi-n exploration, and peak-targeted
subgradient refinement. Key improvements over previous: symmetry-reduced
evaluation, faster blend search (no golden section), parametric init sweep,
and more aggressive multi-n exploration with better time allocation.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def make_symmetric(a):
    """Make a sequence symmetric: a[i] = a[n-1-i]."""
    n = len(a)
    return (a + a[::-1]) / 2.0


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_value) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue
    return None, None


def parametric_seq(n, mode=0, param=1.0, rng=None):
    """Generate parametric initial sequences."""
    x = np.arange(n) / max(n - 1, 1)
    xc = x - 0.5  # centered in [-0.5, 0.5]

    if mode == 0:
        # Uniform
        a = np.ones(n)
    elif mode == 1:
        # Raised cosine power: (cos(pi*xc))^p clipped
        a = np.cos(np.pi * xc) ** param
        a = np.maximum(a, 0)
    elif mode == 2:
        # Gaussian
        sigma = param
        a = np.exp(-xc ** 2 / (2 * sigma ** 2))
    elif mode == 3:
        # Parabolic
        a = np.maximum(1 - 4 * xc ** 2 * param, 0)
    elif mode == 4:
        # Power law
        a = np.maximum(np.abs(1 - 2 * xc), 0) ** param
    elif mode == 5:
        # B-power
        a = (1 - np.abs(2 * xc)) ** param
        a = np.maximum(a, 0)
    elif mode == 6:
        # Double bump
        c1, c2 = 0.25, 0.75
        w = param
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
    else:
        a = np.ones(n)

    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    n = len(a)

        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.0 / np.sqrt(n)
    momentum = 0.8
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        if max_val < 1e-15:
            break

        # Soft top-K gradient
        k = min(10, len(conv))
        top_idx = np.argsort(conv)[::-1][:k]
        peak_vals = conv[top_idx]
        weights = np.exp(5.0 * (peak_vals - peak_vals[0]) / max(peak_vals[0], 1e-15))
        weights /= weights.sum()

        g = np.zeros(n)
        for w, ki in zip(weights, top_idx):
            ki = int(ki)

            i_end = min(n - 1, ki)
            j_start = ki - i_end
            j_end = ki - i_start
            g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        a_new = make_symmetric(a_new)

            best_seq = a_lp.copy()
            print(f"  Main LP: obj={best_score:.8f} **")
        else:
            print(f"  Main LP: obj={obj_lp:.8f}")

    # === Phase 4: Multi-n exploration ===
    print("\n=== Phase 4: Multi-n Exploration ===")
    n_current = len(best_seq)
    n_options = sorted(set([
        max(400, n_current - 200),
        max(500, n_current - 100),
        n_current + 100,
        n_current + 200,
        n_current + 400,
        600, 800, 1000, 1200, 1400, 1600
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1800 and abs(n - n_current) > 50]

    for n_try in n_options[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)

        # Quick warmup
        alloc1 = min(remaining * 0.1, 25.0)
        if alloc1 > 8:
            a_try = quick_gd(a_try, time_budget=alloc1)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.12, 100.0)
        if alloc2 > 15:
            a_try = lp_polish(a_try, time_budget=alloc2, seed=n_try % 1000)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Perturbation + Re-LP ===
    print("\n=== Phase 5: Perturbation + Re-LP ===")
    for r in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        scale = 0.003 + 0.004 * r
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = make_symmetric(np.maximum(a_pert, 0))
        a_pert = a_pert / a_pert.sum()

        alloc = min(remaining * 0.15, 80.0)
        if alloc > 15:
            a_pert = lp_polish(a_pert, time_budget=alloc, seed=200 + r)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Perturb {r} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Perturb {r} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 6: Peak-targeted subgradient ===
    print("\n=== Phase 6: Peak Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 60.0)
        a_sg = peak_subgradient(best_seq, time_budget=alloc, K=8)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Peak SG: obj={best_score:.8f} **")
        else:
            print(f"  Peak SG: obj={obj_sg:.8f}")

    # === Phase 7: Final intensive LP ===
    print("\n=== Phase 7: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.7, 200.0)
        a_final = lp_polish(best_seq, time_budget=alloc, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

        # One more subgradient pass
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc2 = min(remaining * 0.3, 30.0)
            a_sg2 = peak_subgradient(best_seq, time_budget=alloc2, K=12)
            obj_sg2 = compute_obj(a_sg2)
            if obj_sg2 < best_score:
                best_score = obj_sg2
                best_seq = a_sg2.copy()
                print(f"  Final SG: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
