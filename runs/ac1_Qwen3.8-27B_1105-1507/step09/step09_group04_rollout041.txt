def project_simplex(v):

def coord_descent(a, time_budget, rng=None, verbose=False):

    n = len(a)

    gr = (np.sqrt(5) - 1) / 2.0
    start = time.monotonic()
    stagnant = 0
    
    while time.monotonic() - start < time_budget:
        sweep_start = time.monotonic()

        if rng is not None and np.random.random() < 0.3:

            # Direction vector for changing a[k]

            lo = -a[k]
            hi = 3.0 / n
            
            # Golden section

            b1 = hi
            c1 = b1 - gr * (b1 - a1)
            d1 = a1 + gr * (b1 - a1)

                val = conv + 2.0 * delta * d

                return float(np.max(val))
            
            fc = obj_delta(c1)

            for _ in range(30):
                if fc < fd:
                    b1 = d1
                    d1 = c1
                    fd = fc
                    c1 = b1 - gr * (b1 - a1)
                    fc = obj_delta(c1)
                else:
                    a1 = c1
                    c1 = d1
                    fc = fd
                    d1 = a1 + gr * (b1 - a1)

            new_obj = obj_delta(delta)

            conv += 2.0 * delta * d
            conv[2 * k] += delta * delta

            if S_new < 1e-10:

                conv = compute_conv(a)

def peak_targeting(a, rng, n_targets=3, n_iters=1000, time_budget=30):
    """
    Targeted peak reduction: identify top peaks and reduce contributions.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    a = a / max(s, 1e-10)

    start = time.monotonic()
    
    lr = 1.0 / (2.0 * n * 0.05)  # small step
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        
        # Find top peaks
        top_k = np.argsort(conv)[::-1][:n_targets]
        
        # Randomly pick one
        k_star = top_k[rng.integers(0, n_targets)]
        
        # Subgradient direction (template that increases peak at k_star)
        # g_j = 2 * a_{k_star - j} for valid j
        g = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < 2 * n - 1:
                # The contribution to (a*a)_{k_star} from a_j is a_{k_star-j}
                # if k_star - j is in [0, n-1]
                if 0 <= k_star - j <= n - 1:
                    g[j] = 2.0 * a[k_star - j]
        
        # Step: reduce in the direction of the subgradient
        a_new = a - lr * g
        a_new = project_simplex(a_new)
        
        # Check
        conv_new = compute_conv(a_new)
        peak_new = float(np.max(conv_new))
        
        if peak_new < best_peak - 1e-13:
            best_peak = peak_new
            best_a = a_new.copy()
            a = a_new

            lr *= 0.95
        
        # Periodic restart

                a = best_a.copy()
    
    return best_a

def poly_init(n, degree=4, rng=None):
    """Initialize with a polynomial of given degree."""
    if rng is None:
        rng = np.random.default_rng(42)
    x = np.linspace(0, 1, n)
    # Random polynomial with positive values
    coeffs = rng.uniform(0.1, 1.0, degree + 1)
    coeffs[0] = 1.0  # Ensure positive constant term
    a = np.zeros(n)
    for d in range(degree + 1):
        a += coeffs[d] * x**d
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:

        a = a / a.sum()
    elif mode == 2:
        # Bimodal

            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))

    elif mode == 3:
        # Half-sine
        x = np.linspace(0, np.pi, n)
        a = np.sin(x)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 4:
        # Cosine window
        x = np.linspace(-1, 1, n)
        a = 1.0 - 0.5 * x**2
        a = a / a.sum()
    elif mode == 5:
        # Polynomial
        return poly_init(n, degree=4, rng=rng)
    elif mode == 6:
        # Geometric
        r = rng.uniform(0.8, 0.99)
        a = r ** np.arange(n)
        a = a / a.sum()

"""
Optimizes non-negative step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses multi-scale smoothed gradient descent (FISTA continuation), improved coordinate descent,
peak-targeted subgradient method, polynomial subspace initialization, and multi-length exploration.
Key innovations: continuation method for tau in FISTA, peak-targeted corrections,
adaptive step sizes, and diverse restart strategies.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_grad_and_value(a, tau):
    """Compute smoothed max value and its gradient. Returns (smoothed_val, gradient, conv)."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    smoothed_val = max_c + np.log(sum_exp + 1e-300) / tau

    N = 1
    while N < 3 * n - 1:
        N <<= 1
    A = fft(a, N)
    P = fft(p, N)
    g = ifft(A * np.conj(P)).real[:n]
    grad = 2.0 * g

    return smoothed_val, grad, conv


def gs_evaluate(conv, d, k, delta):
    """Evaluate max of (conv + 2*delta*d) with delta^2 added at position 2k."""
    val = conv + 2.0 * delta * d
    val[2 * k] += delta * delta
    return float(np.max(val))


def fista_continuation(a, time_budget, tau_schedule, verbose=False):
    """FISTA with continuation on tau. a is on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_peak = float(np.max(compute_conv(a)))
    start = time.monotonic()

    for tau_level, tau in enumerate(tau_schedule):
        remaining = time_budget - (time.monotonic() - start)
        levels_left = len(tau_schedule) - tau_level
        level_budget = max(5.0, remaining * 0.7 / levels_left)

        if time.monotonic() - start >= time_budget - 5:
            break

        level_start = time.monotonic()
        x = a.copy()
        y = a.copy()
        t_k = 1.0
        conv_init = compute_conv(x)
        peak_init = float(np.max(conv_init))
        lr = 1.0 / (4.0 * tau * max(peak_init, 1e-6))
        lr = min(lr, 1.0 / (tau * 1.1))
        lr = max(lr, 1e-6)

        it = 0
        best_x = x.copy()
        best_f = peak_init

        while time.monotonic() - level_start < level_budget:
            if time.monotonic() - start >= time_budget - 2:
                break

            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
            y = x + ((t_k - 1.0) / t_new) * (x - y)
            y = np.maximum(y, 0)
            s_y = float(np.sum(y))
            if s_y < 1e-10:
                y = x.copy()
            else:
                y = y / s_y

            smoothed_val, grad, conv_y = smoothed_grad_and_value(y, tau)

            x_new = y - lr * grad
            x_new = project_simplex(x_new)

            f_new = float(np.max(compute_conv(x_new)))

            if f_new > best_f * 1.05 + 1e-10:
                lr = max(lr * 0.8, 1e-8)
            elif f_new < best_f - 1e-12:
                lr = min(lr * 1.005, 1.0 / (tau * 0.5))

            x_prev = x

            t_k = t_new

def coord_descent(a, time_budget, rng, verbose=False):
    """Coordinate descent with golden section search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
    a = a / S
    conv = compute_conv(a)

    best_obj = float(np.max(conv))
    best_a = a.copy()
    gr = (np.sqrt(5) - 1) / 2.0
    start = time.monotonic()
    stagnant = 0
    sweep_count = 0

    while time.monotonic() - start < time_budget:
        obj_before = float(np.max(conv))

        if rng is not None and rng.random() < 0.3:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for k in order:
            if time.monotonic() - start > time_budget:
                break

            d = np.zeros(2 * n - 1)
            d[k:k + n] = a

            lo = -a[k]
            hi = max(3.0 / n, lo + 1.0 / n)

            a1 = lo
            b1 = hi
            c1 = b1 - gr * (b1 - a1)
            d1 = a1 + gr * (b1 - a1)

            fc = gs_evaluate(conv, d, k, c1)
            fd = gs_evaluate(conv, d, k, d1)

            for _ in range(35):
                if fc < fd:
                    b1 = d1
                    d1 = c1
                    fd = fc
                    c1 = b1 - gr * (b1 - a1)
                    fc = gs_evaluate(conv, d, k, c1)
                else:
                    a1 = c1
                    c1 = d1
                    fc = fd
                    d1 = a1 + gr * (b1 - a1)
                    fd = gs_evaluate(conv, d, k, d1)

            delta = (a1 + b1) / 2.0
            new_obj = gs_evaluate(conv, d, k, delta)

            if new_obj >= obj_before - 1e-14:
                continue

            a[k] = max(0.0, a[k] + delta)
            conv += 2.0 * delta * d
            conv[2 * k] += delta * delta

            S_new = float(np.sum(a))
            if S_new < 1e-10:
                a[:] = 1.0 / n
                conv = compute_conv(a)
                break
            a /= S_new
            conv /= (S_new * S_new)

        sweep_count += 1
        obj_after = float(np.max(conv))
        if obj_after < best_obj - 1e-14:
            best_obj = obj_after
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 4:
                if rng is not None:
                    scale = 0.01 * min(stagnant, 15)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    conv = compute_conv(a)
                    best_obj = float(np.max(conv))
                    stagnant = 0

        if verbose and sweep_count % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep_count}: obj={2*n*best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def peak_targeting(a, rng, time_budget=30, n_targets=3):
    """Targeted peak reduction using subgradients at peak positions."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    a = a / max(s, 1e-10)

    best_a = a.copy()
    best_peak = float(np.max(compute_conv(a)))
    start = time.monotonic()
    lr = 0.01 / n
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_k = np.argsort(conv)[::-1][:n_targets]
        k_star = top_k[rng.integers(0, n_targets)]

        g = np.zeros(n)
        lo_j = max(0, k_star - (n - 1))
        hi_j = min(k_star, n - 1)
        for j in range(lo_j, hi_j + 1):
            g[j] = 2.0 * a[k_star - j]

        a_new = a - lr * g
        a_new = project_simplex(a_new)

        conv_new = compute_conv(a_new)
        peak_new = float(np.max(conv_new))

        if peak_new < best_peak - 1e-13:
            best_peak = peak_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.005, 0.1 / n)
        else:
            lr = max(lr * 0.95, 1e-6 / n)

        if it % 500 == 0:
            a = 0.5 * a + 0.5 * best_a
            a = project_simplex(a)

        it += 1

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def diverse_init(n, rng, mode=0):

    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:

    elif mode == 2:
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 8)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0, np.pi, n)
        a = np.sin(x)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 4:

        a = 1.0 - 0.5 * x ** 2
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        r = rng.uniform(0, 6)
        s = rng.uniform(0.1, 1.0)
        a = 1.0 + s * np.cos(r * x * np.pi)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        r = rng.uniform(0.85, 0.99)
        a = r ** np.arange(n)
        a = a / a.sum()
    elif mode == 7:
        x = np.arange(n) / max(n - 1, 1)
        a = 1.0 - np.abs(2.0 * x - 1.0)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n

    # === Phase 2: Coordinate Descent (main workhorse) ===
    print(f"\n=== Phase 2: Coordinate Descent ===")
    rem = deadline - time.monotonic()
    p2_budget = min(rem * 0.30, 400.0)
    if p2_budget > 30:
        a_out = coord_descent(best_seq, time_budget=p2_budget, rng=rng, verbose=True)
        obj_out = compute_obj(a_out)
        if obj_out < best_score:
            best_score = obj_out
            best_seq = a_out.copy()
            print(f"  Phase2 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 CD: obj={obj_out:.8f}")

    # === Phase 3: Peak Targeting ===
    print(f"\n=== Phase 3: Peak Targeting ===")
    rem = deadline - time.monotonic()
    p3_budget = min(rem * 0.08, 100.0)
    if p3_budget > 15:
        a_out = peak_targeting(best_seq, rng, time_budget=p3_budget, n_targets=3)
        obj_out = compute_obj(a_out)
        if obj_out < best_score:
            best_score = obj_out
            best_seq = a_out.copy()
            print(f"  Phase3 PeakTarget: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 PeakTarget: obj={obj_out:.8f}")

    # === Phase 4: FISTA (high tau) ===
    print(f"\n=== Phase 4: FISTA High-Tau ===")
    rem = deadline - time.monotonic()
    p4_budget = min(rem * 0.15, 200.0)
    if p4_budget > 30:
        a_out = fista_continuation(best_seq, time_budget=p4_budget,
                                   tau_schedule=(500, 1000, 2000, 5000, 10000),
                                   verbose=True)
        obj_out = compute_obj(a_out)
        if obj_out < best_score:
            best_score = obj_out
            best_seq = a_out.copy()
            print(f"  Phase4 FISTA: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 FISTA: obj={obj_out:.8f}")

    # === Phase 5: Coordinate Descent (fine) ===
    print(f"\n=== Phase 5: Fine CD ===")
    rem = deadline - time.monotonic()
    p5_budget = min(rem * 0.20, 250.0)
    if p5_budget > 30:
        a_out = coord_descent(best_seq, time_budget=p5_budget, rng=rng, verbose=True)
        obj_out = compute_obj(a_out)
        if obj_out < best_score:
            best_score = obj_out
            best_seq = a_out.copy()
            print(f"  Phase5 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase5 CD: obj={obj_out:.8f}")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_current = len(best_seq)

    n_candidates = sorted(set([
        max(300, n_current - 200), max(300, n_current - 100),
        n_current + 100, n_current + 200, n_current + 400,
        600, 800, 1000, 1200, 1500, 1800
    ]))
    n_candidates = [nn for nn in n_candidates if 300 <= nn <= 2000 and abs(nn - n_current) > 40]

    for n_try in n_candidates[:5]:
        rem = deadline - time.monotonic()
        if rem < 100:
            break

        a_try = resample_seq(best_seq, n_try)

        alloc1 = min(rem * 0.35, 100.0)
        if alloc1 > 20:
            a_try = coord_descent(a_try, time_budget=alloc1, rng=rng)

        rem = deadline - time.monotonic()
        alloc2 = min(rem * 0.15, 50.0)
        if alloc2 > 15:
            a_try = fista_continuation(a_try, time_budget=alloc2,
                                       tau_schedule=(200, 1000, 5000))

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse Restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    n_best = len(best_seq)

    for r in range(4):
        rem = deadline - time.monotonic()
        if rem < 120:
            break

        mode = (r + 1) % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.2 + 0.1 * r
        a_r = (1.0 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc1 = min(rem * 0.35, 100.0)
        if alloc1 > 20:
            a_r = coord_descent(a_r, time_budget=alloc1 * 0.6, rng=rng)

        rem = deadline - time.monotonic()
        alloc2 = min(rem * 0.2, 50.0)
        if alloc2 > 15:
            a_r = fista_continuation(a_r, time_budget=alloc2,
                                     tau_schedule=(100, 500, 2000))

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Final Intensive Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    rem = deadline - time.monotonic()

    if rem > 40:
        alloc = min(rem * 0.35, 120.0)
        a_out = coord_descent(best_seq, time_budget=alloc, rng=rng, verbose=True)
        obj_out = compute_obj(a_out)
        if obj_out < best_score:
            best_score = obj_out
            best_seq = a_out.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    rem = deadline - time.monotonic()
    if rem > 25:
        alloc = min(rem * 0.3, 60.0)
        a_out = fista_continuation(best_seq, time_budget=alloc,
                                   tau_schedule=(2000, 5000, 10000, 20000))
        obj_out = compute_obj(a_out)
        if obj_out < best_score:
            best_score = obj_out
            best_seq = a_out.copy()
            print(f"  Final FISTA: obj={best_score:.8f} **")

    rem = deadline - time.monotonic()
    if rem > 20:
        alloc = min(rem * 0.4, 50.0)
        a_out = coord_descent(best_seq, time_budget=alloc, rng=rng)
        obj_out = compute_obj(a_out)
        if obj_out < best_score:
            best_score = obj_out
            best_seq = a_out.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        final_score = val
    else:
        result = [1.0] * 100
        final_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={final_score:.8f}, time={elapsed:.1f}s ===")

    return result
