def solve_lp(a, perturb=None, solver_pref=None):
    """
    Solve LP: min M (+ eps*perturb^T x) s.t. conv(a,x)[k] <= M, sum(x)=1, x>=0.

    n = len(a)

    M = cp.Variable()

def active_constraint_analysis(a, g, M_lp, tolerance=1e-6):
    """
    Analyze which convolution constraints are active at the LP solution.
    Returns array of active multipliers (proxy for dual variables).
    """
    n = len(a)
    conv_ag = np.zeros(2 * n - 1)
    # conv(a, g)[k] = sum_j a[k-j] * g[j]
    for j in range(n):
        if g[j] > 1e-12:
            k_start = j
            k_end = min(j + n, 2 * n - 1)
            conv_ag[k_start:k_end] += a[:k_end - k_start] * g[j]

    # Active constraints: those where conv(a,g)[k] is close to M_lp
    slack = M_lp - conv_ag
    active = np.maximum(0, 1.0 - slack / max(M_lp, 1e-12))
    return active, conv_ag

def smoothed_subgradient_step(a, step_size, num_points=1):
    """
    Take a subgradient step on the exact max objective.
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = conv.max()
    # Find all indices achieving (near) the max
    threshold = max_val - 1e-10 * max_val
    active_indices = np.where(conv >= threshold)[0]

    if len(active_indices) == 0:
        return a

    # Subgradient: sum of cross-correlations at active indices
    grad = np.zeros(n)
    for k in active_indices:
        k = int(k)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += a[idx]
    grad *= 2.0 / len(active_indices)

    # Projected gradient step onto simplex (approximate via negative part + renormalize)
    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""

    elif mode == 3:

        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:

        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:

    elif mode == 7:
