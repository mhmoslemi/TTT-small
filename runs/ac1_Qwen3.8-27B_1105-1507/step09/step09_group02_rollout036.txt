def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

def compute_obj(a):

    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * np.max(conv)

def lp_solve(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build convolution matrix
    # conv(a, g)[k] = sum_i a_norm[i] * g[k-i]
    # We want: for all k in 0..2n-2, conv(a,g)[k] <= M
    
    # Use scipy sparse for large n
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # g[j] appears in conv[k] for k = j..j+n-1 with coefficient a_norm[min(n-1, k-j)]
        # Actually: conv(a,g)[k] = sum_i a[i]*g[k-i], so g[j] appears when k-i=j, i.e., i=k-j
        # conv[k] gets contribution a[k-j]*g[j] for valid i=k-j in [0,n-1]
        # So for each j, and for each k from j to j+n-1:
        #   conv[k] += a[k-j] * g[j]
        #   i.e., a[(k-j)] * g[j] where i = k-j must be in [0, n-1]
        #   So k-j in [0, n-1], meaning k in [j, j+n-1]
        for i in range(n):
            k = i + j
            if k < 2*n - 1:
                rows.append(k)
                cols.append(j)
                vals.append(a_norm[i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except:
            continue
    return None, None

def lp_polish(a, time_budget, verbose=False):
    """LP-based polishing with adaptive blending."""
    n = len(a)

        return a

        if g_lp is None:
            break

        best_t_obj = best_obj

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)
