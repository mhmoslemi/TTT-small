"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated evaluation, Nesterov smoothed gradient descent,
top-k LP polishing with golden section line search, spectral shaping,
and multi-length exploration.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_gradient(a, tau):
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def nesterov_smoothed_descent(a, time_budget=100.0, tau_init=10.0, tau_end=500.0, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient(y, tau)

        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.3 * lr + 0.7 * min(max(bb_lr, 1e-8), 15.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9997)

        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def build_conv_matrix_rows(a, rows):
    """Build rows of the convolution matrix A for specific lag indices."""
    n = len(a)
    A = np.zeros((len(rows), n))
    for idx, k in enumerate(rows):
        k = int(k)
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[idx, j] = a[k - j]
    return A


def solve_lp_topk(a, top_k=30):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for top-k entries, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv = compute_conv(a_norm)
    
    # Get top-k indices
    k = min(top_k, len(conv))
    top_indices = np.argsort(conv)[::-1][:k]
    
    A = build_conv_matrix_rows(a_norm, top_indices)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_full(a):
    """Solve full LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix_rows(a_norm, list(range(2 * n - 1)))
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def golden_search(f, a, b, tol=1e-6, max_iter=50):
    """Golden section search for minimum of f on [a, b]."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2


def lp_polish_with_golden(a, max_iter=50, time_limit=60.0, use_full_lp_every=5, top_k=30, verbose=False):
    """
    LP polishing with golden section search for blend parameter.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf, 0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    no_improve = 0
    start = time.monotonic()
    it = 0

    while it < max_iter and (time.monotonic() - start) < time_limit:
        # Solve LP
        if it % use_full_lp_every == 0:
            g_lp, M_lp = solve_lp_full(a)
        else:
            g_lp, M_lp = solve_lp_topk(a, top_k=top_k)
        
        if g_lp is None:
            # Fallback to top-k
            g_lp, M_lp = solve_lp_topk(a, top_k=top_k)
            if g_lp is None:
                break

        # Golden section search for optimal t
        def obj_at_t(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return np.inf
            return compute_obj(a_try / s_try)

        t_opt = golden_search(obj_at_t, 0.01, 1.0, tol=0.01, max_iter=20)
        
        a_new = (1.0 - t_opt) * a + t_opt * g_lp
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        it += 1
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            if verbose and it % 5 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={t_opt:.3f}, M={M_lp:.6f}")
        else:
            no_improve += 1
            if no_improve >= 3:
                break

    return best_a, best_obj, it


def spectral_shaping(a, n_iter=10, verbose=False):
    """
    Spectral shaping: try to flatten the power spectrum |FFT(a)|^2.
    This reduces the peak of the inverse FFT (i.e., the convolution).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    for i in range(n_iter):
        A = fft(a, 2 * n)
        mag = np.abs(A)
        # Try to make mag more uniform
        target_mag = np.ones(len(mag))
        # Blend current mag with target
        blend = 0.3
        new_mag = (1 - blend) * mag + blend * target_mag
        # Preserve the zero frequency (DC) component
        new_mag[0] = mag[0]
        # Preserve conjugate symmetry (for real signal)
        new_mag[n:] = new_mag[n-1:0:-1] if n <= len(mag) else new_mag
        
        # Reconstruct
        A_new = new_mag * np.exp(1j * np.angle(A))
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller blend
            blend = 0.1
            new_mag = (1 - blend) * mag + blend * target_mag
            new_mag[0] = mag[0]
            new_mag[n:] = new_mag[n-1:0:-1] if n <= len(mag) else new_mag
            A_new = new_mag * np.exp(1j * np.angle(A))
            a_new = ifft(A_new).real[:n]
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                break

    return best_a


def perturb_peak_targeted(a, rng, scale=0.05):
    """
    Perturbation that specifically targets the peak convolution entries.
    Reduces contributions to the peak and increases non-peak contributions.
    """
    n = len(a)
    a = a.copy()
    a_norm = a / float(np.sum(a))
    conv = compute_conv(a_norm)
    max_conv = float(np.max(conv))
    
    # Find top peaks
    top_k = np.argsort(conv)[::-1][:10]
    
    # Compute gradient of each peak w.r.t. a
    pressure = np.zeros(n)
    for k in top_k:
        k = int(k)
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            pressure[j] += conv[k] * a_norm[k - j]
    
    # Normalize pressure
    max_p = np.max(np.abs(pressure))
    if max_p > 1e-15:
        pressure = pressure / max_p
    
    # Reduce where pressure is high, increase where it's low
    a = a * np.exp(-scale * pressure + scale * rng.normal(0, 0.1, n))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_norm
    return a / s


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated evaluation, Nesterov smoothed gradient descent,
top-k LP polishing with golden section line search, spectral shaping,
peak-targeted perturbation, and multi-length exploration.
Key improvements over previous: golden section blend search, top-k LP for speed,
spectral shaping moves, and targeted anti-peak perturbation.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of (a*a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def nesterov_smoothed_descent(a, time_budget=100.0, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))

    prev_g = None

    best_it = 0

        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:

def build_conv_matrix_rows(a, row_indices):
    """Build rows of convolution matrix A where A[k,j] = a[k-j] for valid j."""
    n = len(a)
    A = np.zeros((len(row_indices), n))
    for idx, k in enumerate(row_indices):
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n, k + 1)
        for j in range(j_lo, j_hi):
            A[idx, j] = a[k - j]
    return A


def solve_lp_topk(a, top_k=30):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for top-k lag indices, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv = compute_conv(a_norm)

    k = min(top_k, len(conv))
    top_indices = np.argsort(conv)[::-1][:k]

    A = build_conv_matrix_rows(a_norm, top_indices)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_full(a):
    """Solve full LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix_rows(a_norm, list(range(2 * n - 1)))

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def golden_search(f, a_lo, a_hi, tol=0.01, max_iter=25):
    """Golden section search for minimum of f on [a_lo, a_hi]. Returns optimal point."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if a_hi - a_lo < tol:
            break
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = f(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = f(d)
    return (a_lo + a_hi) / 2.0


def lp_polish_with_golden(a, max_iter=50, time_limit=60.0,

def spectral_shaping_step(a, blend=0.2):
    """
    One step of spectral shaping: flatten the power spectrum to reduce convolution peak.
    Returns modified sequence.
    """
    n = len(a)
    N = 2 * n
    A = fft(a, N)
    mag = np.abs(A)

    # Target: uniform magnitude (flat spectrum)
    target_mag = np.full(N, np.sqrt(np.mean(mag ** 2)))

    # Blend current magnitude with target
    new_mag = (1.0 - blend) * mag + blend * target_mag

    # Preserve DC component
    new_mag[0] = mag[0]
    # Enforce conjugate symmetry for real signal
    for i in range(1, N // 2 + 1):
        if i < N:
            new_mag[N - i] = new_mag[i]

    # Reconstruct with original phases
    A_new = new_mag * np.exp(1j * np.angle(A))
    a_new = ifft(A_new).real[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def perturb_peak_targeted(a, rng, scale=0.04):
    """
    Perturbation that specifically targets peak convolution entries.
    Reduces weight where it contributes to peaks, adds random noise elsewhere.
    """
    n = len(a)
    a = a.copy()
    a_norm = a / float(np.sum(a))
    conv = compute_conv(a_norm)

    # Find top peaks
    n_peaks = min(15, len(conv))
    top_idx = np.argsort(conv)[::-1][:n_peaks]

    # Compute "pressure" from peak entries
    pressure = np.zeros(n)
    for k in top_idx:
        k = int(k)
        weight = conv[k]
        j_lo = max(0, k - n + 1)
        j_hi = min(n, k + 1)
        for j in range(j_lo, j_hi):
            pressure[j] += weight * a_norm[k - j]

    # Normalize
    max_p = np.max(np.abs(pressure))
    if max_p > 1e-15:
        pressure = pressure / max_p

    # Apply exponential tilt: reduce where pressure is high
    a = a * np.exp(-scale * pressure + scale * 0.3 * rng.standard_normal(n))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_norm
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    p1_budget = min(remaining * 0.08, 80.0)
    if p1_budget > 15:
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=p1_budget,
                                          tau_init=5.0, tau_end=500.0, verbose=False)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase1 SGD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 SGD: obj={obj_sgd:.8f}")

    # === Phase 2: Intensive LP polishing with golden section ===
    print("\n=== Phase 2: LP Polish (golden section) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.40, 480.0)
    if p2_budget > 30:
        a_current = best_seq.copy()
        n_rounds = 10
        round_budget = p2_budget / n_rounds

        for rnd in range(n_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break
            alloc = min(round_budget, remaining * 0.35)
            if alloc < 12:
                break

            # Use increasing top_k and more frequent full LP in later rounds
            top_k_val = 30 + 10 * min(rnd, 3)
            full_every = max(2, 5 - rnd // 3)

            a_new, obj_new, iters = lp_polish_with_golden(
                a_current, max_iter=60, time_limit=alloc,
                use_full_lp_every=full_every, top_k=top_k_val,
                verbose=(rnd == 0))

            if obj_new < best_score - 1e-12:
                best_score = obj_new
                best_seq = a_new.copy()
                a_current = a_new.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} ({iters} iters) **")
            else:
                print(f"  LP round {rnd}: obj={obj_new:.8f} ({iters} iters)")

                # Perturb to escape local minimum
                method = rng.random()
                if method < 0.35:
                    a_current = perturb_peak_targeted(a_new, rng,
                                                      scale=0.02 + 0.01 * rnd)
                elif method < 0.6:
                    a_current = a_new * np.exp(rng.normal(0, 0.025, len(a_new)))
                    a_current = np.maximum(a_current, 0)
                    a_current = a_current / a_current.sum()
                elif method < 0.8:
                    # Spectral perturbation
                    A = fft(a_new, 2 * len(a_new))
                    A[1:-1] *= np.exp(1j * rng.normal(0, 0.04, len(A[1:-1])))
                    a_current = ifft(A).real[:len(a_new)]
                    a_current = np.maximum(a_current, 0)
                    sc = a_current.sum()
                    if sc > 1e-10:
                        a_current = a_current / sc
                    else:
                        a_current = a_new.copy()
                else:
                    # Shift perturbation
                    shift = int(rng.integers(1, max(2, len(a_new) // 12)))
                    a_current = np.roll(a_new, shift)

                # Quick LP after perturbation
                remaining = deadline - time.monotonic()
                alloc2 = min(alloc * 0.45, remaining * 0.2)
                if alloc2 > 8:
                    a_pert, obj_pert, _ = lp_polish_with_golden(
                        a_current, max_iter=25, time_limit=alloc2,
                        use_full_lp_every=4, top_k=30)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        a_current = a_pert.copy()
                        print(f"  LP round {rnd} (perturbed): obj={best_score:.8f} **")

    # === Phase 3: Spectral shaping ===
    print("\n=== Phase 3: Spectral Shaping ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        spec_start = time.monotonic()
        spec_alloc = min(remaining * 0.06, 50.0)
        a_spec = best_seq.copy()

        for i in range(30):
            if time.monotonic() - spec_start > spec_alloc:
                break
            blend = 0.15 * (1.0 - i / 30.0) + 0.05
            a_spec = spectral_shaping_step(a_spec, blend=blend)
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_score:
                best_score = obj_spec
                best_seq = a_spec.copy()
                if i % 5 == 0:
                    print(f"  Spec {i}: obj={best_score:.8f} **")

        # LP polish after spectral shaping
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_lp, obj_lp, _ = lp_polish_with_golden(
                best_seq, max_iter=25, time_limit=min(remaining * 0.08, 35.0),
                use_full_lp_every=3, top_k=40)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Post-spec LP: obj={best_score:.8f} **")

    # === Phase 4: Multi-n exploration ===
    print("\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_candidates = []
    for delta in [-100, -50, 50, 100, 200, 300, 400]:
        n_try = n_current + delta
        if 500 <= n_try <= 1500:
            n_candidates.append(n_try)
    # Add some absolute values
    for n_abs in [600, 700, 800, 900, 1000, 1100, 1200]:
        if 500 <= n_abs <= 1500:
            n_candidates.append(n_abs)

    n_options = sorted(set(n_candidates) - {n_current})
    n_options.sort(key=lambda x: abs(x - n_current))

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        # Short SMOD
        remaining = deadline - time.monotonic()
        alloc1 = min(remaining * 0.04, 25.0)
        if alloc1 > 8:
            a_sgd = nesterov_smoothed_descent(a_try, time_budget=alloc1,
                                              tau_init=10.0, tau_end=300.0)
            if compute_obj(a_sgd) < compute_obj(a_try):
                a_try = a_sgd

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.10, 50.0)
        if alloc2 > 8:
            a_try, obj_try, _ = lp_polish_with_golden(
                a_try, max_iter=30, time_limit=alloc2,
                use_full_lp_every=4, top_k=35)
        else:
            obj_try = compute_obj(a_try)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts ===
    print("\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        a_r = best_seq.copy()
        if r == 0:
            a_r = a_r * np.exp(rng.normal(0, 0.12, n_best))
        elif r == 1:
            shift = int(rng.integers(1, max(2, n_best // 10)))
            a_r = np.roll(a_r, shift)
        elif r == 2:
            A = fft(a_r, 2 * n_best)
            A[1:-1] *= np.exp(1j * rng.normal(0, 0.15, len(A[1:-1])))
            a_r = ifft(A).real[:n_best]
        elif r == 3:
            a_r = perturb_peak_targeted(a_r, rng, scale=0.08)
        elif r == 4:
            # Blend with uniform
            a_r = 0.7 * a_r + 0.3 / n_best

        a_r = np.maximum(a_r, 0)
        s = a_r.sum()
        if s > 1e-10:
            a_r = a_r / s
        else:
            a_r = best_seq.copy()

        alloc = min(remaining * 0.12, 45.0)
        if alloc > 12:
            a_r, obj_r, _ = lp_polish_with_golden(
                a_r, max_iter=30, time_limit=alloc,
                use_full_lp_every=4, top_k=35)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    print("\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    for lp_rnd in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 12:
            break
        alloc = min(remaining * 0.18, 50.0)
        if alloc < 8:
            break

        a_current = best_seq.copy()
        if lp_rnd > 0 and lp_rnd % 2 == 0:
            a_current = perturb_peak_targeted(a_current, rng, scale=0.012)

        full_every = max(2, 4 - lp_rnd // 2)
        top_k_val = min(50, 35 + 5 * lp_rnd)

        a_lp, obj_lp, iters = lp_polish_with_golden(
            a_current, max_iter=40, time_limit=alloc,
            use_full_lp_every=full_every, top_k=top_k_val)

        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP {lp_rnd}: obj={best_score:.8f} ({iters} iters) **")
        else:
            print(f"  Final LP {lp_rnd}: obj={obj_lp:.8f} ({iters} iters)")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
