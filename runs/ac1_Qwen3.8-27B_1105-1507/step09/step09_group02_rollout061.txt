"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Structured monotone initializations (power-law, exponential) + heavy LP polishing
with adaptive blending + spectral shaping + Nesterov smoothed GD + multi-n exploration.
Key insight: monotone sequences (heavily weighted to one side) reduce the peak autocorrelation
below the uniform baseline of 2.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_grad_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n], conv


def nesterov_smoothed_descent(a, time_budget, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)

    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        if it % 5000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_grad_fft(y, tau)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            consecutive_fails = 0

                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t_val:.3f}, M={M_lp:.6f}")
        else:

            if consecutive_fails > 20:

                a = np.maximum(a, 0)
                a = a / a.sum()
                consecutive_fails = 0
            if consecutive_fails > 50:

def spectral_shaping(a, time_budget, verbose=False):

    """

    s = float(np.sum(a))

        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(77)

    N = 1
    while N < 2 * n:
        N <<= 1

    it = 0
    alpha = 0.5  # spectral flattening parameter

    while time.monotonic() - start < time_budget:
        F = fft(a, N)
        mag = np.abs(F)
        phase = np.angle(F)

        # Flattening: reduce dynamic range of magnitude spectrum
        mag_normalized = mag / (np.max(mag) + 1e-15)
        # Apply power compression (alpha < 1 compresses, alpha > 1 expands)
        mag_new = mag_normalized ** alpha
        mag_new = mag_new / np.max(mag_new) * np.max(mag)

        F_new = mag_new * np.exp(1j * phase)
        a_candidate = np.real(ifft(F_new, N))[:n]
        a_candidate = np.maximum(a_candidate, 0)
        s_cand = float(np.sum(a_candidate))
        if s_cand < 1e-10:
            it += 1
            continue
        a_candidate = a_candidate / s_cand

        obj_cand = compute_obj(a_candidate)
        if obj_cand < best_obj - 1e-12:
            best_obj = obj_cand
            best_a = a_candidate.copy()
            a = a_candidate
            it += 1
            # Slowly increase alpha (less aggressive flattening over time)
            alpha = min(0.9, alpha * 1.01)
        else:
            # Try with random phase perturbation
            if it % 10 == 0:
                phase_pert = phase + rng.normal(0, 0.01, N)
                F_pert = mag_new * np.exp(1j * phase_pert)
                a_pert = np.real(ifft(F_pert, N))[:n]
                a_pert = np.maximum(a_pert, 0)
                s_pert = float(np.sum(a_pert))
                if s_pert > 1e-10:
                    a_pert = a_pert / s_pert
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_obj - 1e-12:
                        best_obj = obj_pert
                        best_a = a_pert.copy()
                        a = a_pert

            it += 1
            if it > 500:
                break

        if verbose and it % 50 == 0:
            print(f"    Spectral it{it}: obj={best_obj:.8f}, alpha={alpha:.3f}")

    return best_a

    s = float(np.sum(a))
    if s < 1e-10:
        return a
