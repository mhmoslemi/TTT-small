def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    # For zero-padded a (length N), this is sum_k p[k]*a[k-i] for valid indices
    # We want grad[i] = 2 * sum_k p[k] * a[k-i] for k-i in [0, n-1]
    # The FFT result gives us this (with wrap-around, but since a is zero-padded
    # to length N >= 2n-1, and p has support in [0, 2n-2], the wrap-around
    # doesn't affect the first n values)
    
    grad = 2.0 * xc[:n]
    smoothed_obj = 2.0 * n * (max_conv + np.log(total) / tau)
    return smoothed_obj, grad, conv


def mirror_descent(a, time_budget=100.0, tau_init=5.0, tau_end=200.0, 
                   verbose=False, seed=42):
    """
    Exponentiated gradient / mirror descent for minimizing the smoothed objective.
    Uses the entropy mirror map, so the update is multiplicative:
    a_new[i] ∝ a[i] * exp(-eta * grad[i])
    This naturally respects non-negativity.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)  # ensure strictly positive
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    eta = 1.0 / np.sqrt(n)
    y = a.copy()  # "hidden" variable in mirror descent
    no_improve = 0
    
    it = 0
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)
        
        if it % 2000 == 0 and verbose:
            print(f"    MD it{it}: best={best_obj:.8f}, cur={compute_obj(y):.8f}, tau={tau:.1f}, eta={eta:.4f}, t={elapsed:.0f}s")
        
        # Compute gradient at y
        _, grad, conv = smoothed_obj_and_grad(y, tau)
        
        # Mirror descent update: y_new = exp(-eta * grad) / sum(exp(-eta * grad))
        # But we want to stay close to previous iterate, so:
        # y_new[i] ∝ y[i] * exp(-eta * grad[i])
        log_y = np.log(y + 1e-30)
        log_y_new = log_y - eta * grad
        # Numerical stability
        log_y_new -= np.max(log_y_new)
        y_new = np.exp(log_y_new)
        y_new = y_new / y_new.sum()
        
        # Evaluate
        obj_new = compute_obj(y_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = y_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        y = y_new
        it += 1
        
        # Adaptive step size
        if no_improve > 50:
            eta *= 0.8
            no_improve = 0
            if eta < 1e-6:
                eta = 0.5 / np.sqrt(n)
                # Perturb to escape
                y = best_a * np.exp(rng.normal(0, 0.01, n))
                y = np.maximum(y, 1e-15)
                y = y / y.sum()
        
        # Periodic re-initialization from best
        if it % 5000 == 0 and compute_obj(y) > best_obj * 1.05:
            y = best_a.copy()
            eta = 0.5 / np.sqrt(n)
    
    return best_a


def peak_directed_update(a, time_budget=80.0, K=10, verbose=False, seed=42):
    """
    Directly target the top K peaks of the autocorrelation.
    For each peak, reduce the contributing pairs (i, k-i) proportionally.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    lr = 0.1
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Find top K peaks
        top_idx = np.argsort(conv)[::-1][:K]
        max_val = conv[0]
        
        if max_val < 1e-10:
            break
        
        # Compute gradient as weighted sum over top K peaks
        g = np.zeros(n)
        weights = np.exp((conv[top_idx] - max_val) * 10.0 / max_val)
        weights /= weights.sum()
        
        for w, k in zip(weights, top_idx):
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    g[i] += w * a[j]
        
        # Project gradient to simplex tangent
        g = g - np.mean(g)
        
        # Multiplicative update (slightly different from mirror descent)
        a_try = a * np.exp(-lr * g)
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-10:
            a_try = best_a
            lr *= 0.5
        else:
            a_try = a_try / s
        
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            no_improve = 0
            lr = min(lr * 1.02, 1.0)
        else:
            no_improve += 1
            lr *= 0.7
            if no_improve > 100:
                # Larger perturbation
                a = best_a * np.exp(rng.normal(0, 0.02 + 0.01 * no_improve / 50, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                no_improve = 0
                lr = 0.1
        
        it += 1
    
    return best_a


def spectral_shape(a, time_budget=60.0, verbose=False, seed=42):
    """
    Shape the Fourier spectrum to flatten the autocorrelation.
    The autocorrelation is IFFT(|FFT(a)|^2) (circular, with zero-padding).
    We directly adjust the power spectrum to reduce peaks in the time domain.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    lr = 0.3
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        # Compute power spectrum
        A = fft(a, N)
        P = np.abs(A) ** 2  # power spectrum
        
        # Autocorrelation from spectrum (circular, zero-padded)
        ac = ifft(P).real
        
        # Find peak in autocorrelation
        max_idx = np.argmax(ac)
        max_val = ac[max_idx]
        
        # We want to reduce the contribution to this peak.
        # The peak at lag k is: ac[k] = (1/N) * sum_m P[m] * exp(2πi * m * k / N)
        # To reduce ac[k], we can reduce P[m] where the phase aligns.
        # Simple approach: reduce P uniformly in proportion to its contribution to the peak.
        
        # Contribution of each frequency to the peak:
        phases = np.exp(2j * np.pi * np.arange(N) * max_idx / N)
        contributions = np.abs(P * phases).real  # real part of P * exp(i*phase)
        
        # Normalize contributions
        total_contrib = contributions.sum()
        if total_contrib < 1e-15:
            break
        
        # Reduce high contributors
        P_new = P * (1.0 - lr * contributions / total_contrib)
        P_new = np.maximum(P_new, 0)
        
        # Also add a small amount to low contributors (to maintain sum constraint)
        P_new = P_new + lr * P.min() * 0.1
        
        # Reconstruct a from modified spectrum
        # We need |A_new|^2 = P_new, so A_new = sqrt(P_new) * exp(i*angle(A))
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
        a_try = ifft(A_new).real[:n]
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s > 1e-10:
            a_try = a_try / s
        else:
            a_try = a
        
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.8
            if no_improve > 50:
                break
        
        it += 1
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, M_target=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue
    
    return None, None


def geometric_blend(a, g, t):
    """Geometric (multiplicative) blend: a_new[i] = a[i]^(1-t) * g[i]^t, normalized."""
    a = np.maximum(a, 1e-15)
    g = np.maximum(g, 1e-15)
    log_a = (1 - t) * np.log(a) + t * np.log(g)
    log_a -= np.max(log_a)
    result = np.exp(log_a)
    result = result / result.sum()
    return result


def lp_geometric_polish(a, time_budget=200.0, verbose=False, seed=42):
    """
    LP-based polishing with geometric blending and adaptive alpha.
    Different from previous: uses geometric (multiplicative) blend instead of linear,
    and sweeps over multiple alpha values per LP solve.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    iter_count = 0
    no_improve = 0
    
    # Alpha schedule: start with small, increase
    alpha_schedule = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 5:
            break
        
        # Solve LP
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Try multiple geometric blend factors
        best_t = 0.0
        best_obj_try = best_obj
        best_a_try = best_a
        
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = geometric_blend(best_a, g_lp, t)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_try:
                best_obj_try = obj_try
                best_a_try = a_try
                best_t = t
        
        iter_count += 1
        
        if best_obj_try < best_obj - 1e-14:
            best_obj = best_obj_try
            best_a = best_a_try.copy()
            a = best_a.copy()
            no_improve = 0
            if verbose and iter_count % 3 == 0:
                print(f"    LP-geo it{iter_count}: obj={best_obj:.8f}, t={best_t:.2f}, M={M_lp:.6f}, el={elapsed:.0f}s")
        else:
            no_improve += 1
            if no_improve >= 3:
                # Perturb
                scale = 0.005 + 0.005 * no_improve
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                no_improve = 0
                if iter_count > 40:
                    break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    from scipy.interpolate import CubicSpline
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    if n >= 4:
        cs = CubicSpline(x_old, a, bc_type='natural')
        a_new = cs(x_new)
    else:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-15)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-0.999, 0.999, n)
        a = 1.0 / np.sqrt(1 - x**2)
        a = a / a.sum()
    elif mode == 4:
        # Multi-frequency
        freqs = rng.integers(1, 8, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        amps = rng.uniform(0.1, 0.5, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        # Periodic with random period
        period = rng.integers(max(3, n // 20), max(4, n // 5))
        a = np.zeros(n)
        for i in range(n):
            a[i] = 1.0 + 0.5 * np.cos(2 * np.pi * i / period)
        a = a / a.sum()
    elif mode == 8:
        # zietz pattern (alternating high/low)
        a = np.ones(n)
        a[n // 3: 2 * n // 3] = 0.3
        a = a / a.sum()
    elif mode == 9:
        # Random with structure
        a = rng.uniform(0, 1, n)
        # Smooth it
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def nesterov_smoothed_descent(a, time_budget=200.0, tau_init=5.0, tau_end=300.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.85
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    y = a.copy()
    
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)
        
        if it % 5000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")
        
        # Nesterov extrapolation
        y = a + beta * v
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Gradient
        _, grad, _ = smoothed_obj_and_grad(y, tau)
        
        # Update
        v_new = y - a
        a_new = y - lr * grad
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        v = a_new - y + beta * v
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9998)
        
        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.3
            best_it = it
    
    return best_a
