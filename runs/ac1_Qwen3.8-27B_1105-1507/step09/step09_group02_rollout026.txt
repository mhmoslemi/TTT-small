def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coord_newton_step(a, i, s=1.0):
    """
    Find the optimal value of a[i] given all other coordinates fixed.
    Uses the piecewise-quadratic structure for exact minimization.
    Returns the optimal value x*.
    """
    n = len(a)
    x_old = a[i]
    
    # Compute base_k, c_k, d_k
    # conv(a',a')[k] = base_k + 2*x*c_k + x^2*d_k
    # where a' is a with a[i] = x
    
    # base_k = conv without the a[i] contribution
    # c_k = a[k-i] if valid and k != 2i
    # d_k = 1 if k == 2i, else 0
    
    # First, compute the full convolution
    conv_full = compute_conv(a)
    
    # Remove the contribution of a[i]
    # a[i] contributes to conv at lags k where:
    # - j = i: a[i]*a[k-i] valid if max(0,k-n+1) <= i <= min(k,n-1) and 0 <= k-i < n
    # - j = k-i: a[k-i]*a[i] valid if max(0,k-n+1) <= k-i <= min(k,n-1) and 0 <= i < n (always)
    #   which means 0 <= k-i < n, i.e., i <= k <= i+n-1
    # - double counted when j = i and j = k-i, i.e., k = 2i
    
    base = conv_full.copy()
    c = np.zeros(2 * n - 1)
    d = np.zeros(2 * n - 1)
    
    # Contribution of a[i] to conv:
    # For lag k, if 0 <= k-i < n: add x*a[k-i] (from j=i)
    # For lag k, if 0 <= i < n and k-i >= 0 and k-i < n: add a[k-i]*x (from j=k-i)
    # These are the same when i = k-i, i.e., k = 2i, giving x^2
    
    # So: if i <= k <= i+n-1 (i.e., 0 <= k-i < n):
    #   if k == 2i: contribution is x^2
    #   else: contribution is 2*x*a[k-i]
    
    if i + n - 1 >= 0:  # always true
        k_start = i  # k-i >= 0 => k >= i
        k_end = min(i + n - 1, 2 * n - 2)  # k-i < n => k < i+n, and k <= 2n-2
        
        if k_start <= k_end:
            # base: remove the current a[i] contribution
            for k in range(k_start, k_end + 1):
                if k == 2 * i:
                    base[k] -= x_old * x_old  # remove x^2
                    d[k] = 1.0
                else:
                    base[k] -= 2.0 * x_old * a[k - i]  # remove 2*x*a[k-i]
                    c[k] = a[k - i]
    
    # Now minimize max_k(base_k + 2*x*c_k + x^2*d_k) / (s - x_old + x)^2
    # where s = sum(a) = 1 (normalized)
    s_new_base = s - x_old  # sum of all other elements
    
    # Find optimal x
    # Use iterative approach: start with current x, then do "best response" iterations
    
    x = x_old
    for _ in range(10):
        # Find the argmax
        vals = base + 2.0 * x * c + x * x * d
        k_max = int(np.argmax(vals))
        
        # For the active constraint at k_max, find optimal x
        b_k = base[k_max]
        c_k = c[k_max]
        d_k = d[k_max]
        
        # Minimize (b_k + 2*x*c_k + x^2*d_k) / (s_new_base + x)^2
        # Derivative = 0:
        # (2*c_k + 2*x*d_k)*(s_new_base + x) - (b_k + 2*x*c_k + x^2*d_k)*2 = 0
        # 2*(c_k + x*d_k)*(s_new_base + x) - 2*(b_k + 2*x*c_k + x^2*d_k) = 0
        # (c_k + x*d_k)*(s_new_base + x) = b_k + 2*x*c_k + x^2*d_k
        # c_k*s_new_base + c_k*x + x*d_k*s_new_base + x^2*d_k = b_k + 2*x*c_k + x^2*d_k
        # c_k*s_new_base - b_k + x*(c_k + d_k*s_new_base - 2*c_k) = 0
        # c_k*s_new_base - b_k + x*(d_k*s_new_base - c_k) = 0
        
        denom = d_k * s_new_base - c_k
        if abs(denom) > 1e-15:
            x_new = (b_k - c_k * s_new_base) / denom
            x_new = max(0.0, min(x_new, 2.0 * s_new_base / (n * 0.1)))  # reasonable bound
        else:
            # Linear case: the ratio is (b_k + 2*x*c_k) / (s_new_base + x)^2
            # This is minimized at x -> infinity if c_k > 0, or x = 0 if c_k <= 0
            # Actually for the linear case (d_k=0):
            # (b_k + 2*x*c_k) / (s_new_base + x)^2
            # If c_k > 0: minimize by making numerator small, so x = 0 if b_k > 0
            # If c_k < 0: the numerator decreases with x, but so does the denominator effect
            x_new = x  # keep current
            
        # Dampen the update
        x = 0.5 * x + 0.5 * x_new
    
    return max(0.0, min(x, s / 0.01))  # clip to reasonable range


def coordinate_descent(a, time_budget, verbose=False):
    """
    Coordinate descent with exact line search on the normalized sequence.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s  # normalize to sum 1
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    
    # Use random + greedy ordering
    order = np.arange(n)
    rng = np.random.default_rng(123)
    
    while time.monotonic() - start < time_budget:
        # Shuffle order periodically
        if it % n == 0:
            rng.shuffle(order)
        
        improved = False
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            x_new = coord_newton_step(a, i, s=1.0)
            a[i] = x_new
            obj_new = compute_obj(a)
            
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a.copy()
                improved = True
            else:
                a[i] = best_a[i]  # revert
                if obj_new < best_obj:
                    pass  # still slightly better but within tolerance
        
        it += 1
        if not improved:
            stall += 1
            if stall > 3:
                # Perturb to escape
                a = best_a * np.exp(rng.normal(0, 0.005 * stall, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
        
        if verbose and it % 5 == 0:
            print(f"    CD sweep {it}: obj={best_obj:.10f}")
    
    return best_a


def build_conv_matrix(a):
    """Build matrix (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, solver_list=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    if not HAS_CVXPY:
        return None, None
    
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix efficiently
    # conv(a, g)[k] = sum_j a[j] * g[k-j] (with valid indices)
    # We need: for each k, sum_j a[j]*g[k-j] <= M
    # This is: sum_j (a flipped and shifted) * g <= M
    
    # For efficiency, build the matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # a[j] * g[k-j] contributes to position k
        # k-j must be in [0, n-1], so k in [j, j+n-1]
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def spectral_init(n, rng):
    """Generate a sequence with approximately flat power spectrum."""
    # Random phases
    n_fft = 1
    while n_fft < 2 * n:
        n_fft <<= 1
    phases = rng.uniform(0, 2 * np.pi, n_fft // 2 + 1)
    # Create a "flat" spectrum with slight random variation
    mags = np.ones(n_fft // 2 + 1) * (1 + 0.1 * rng.normal(0, 1, n_fft // 2 + 1))
    mags = np.maximum(mags, 0.1)
    # Build complex spectrum
    spec = mags * np.exp(1j * phases)
    spec[0] = abs(spec[0])  # DC term real
    if n_fft % 2 == 0:
        spec[n_fft // 2] = abs(spec[n_fft // 2])  # Nyquist term real
    # Inverse FFT
    a = irfft(spec, n_fft)[:n]
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        a = spectral_init(n, rng)
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(-0.999, 0.999, n)
        a = 1.0 / (np.sqrt(np.maximum(1 - x**2, 1e-8)))
        a = a / a.sum()
    elif mode == 4:
        # Multi-lobe
        k = rng.integers(2, 8)
        x = np.arange(n) / n
        a = np.zeros(n)
        for j in range(k):
            center = (j + 0.5) / k
            width = 1.0 / (k * 3)
            a += np.exp(-((x - center) ** 2) / (2 * width ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        alpha = rng.uniform(-3, 3)
        a = x ** alpha
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 6:
        # Product of sinusoids (Barker-like)
        x = np.arange(n)
        freqs = rng.uniform(0.5, 5.0, 3)
        a = np.ones(n)
        for f in freqs:
            a *= (1 + 0.5 * np.sin(2 * np.pi * f * x / n))
        a = np.maximum(a, 0.01)
        a = a / a.sum()

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
