"""
Optimizes step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-p norm smoothed optimization, fixed LP direction finding,
peak shaving, and multi-scale exploration with Nesterov acceleration.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.signal import correlate
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a, top_k=None):
    """
    Solve LP: minimize (sum of top_k of conv(a,g)) s.t. sum(g)=1, g>=0.
    If top_k is None, minimize max(conv(a,g)).
    Returns the optimal g (normalized to sum=1).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    
    if top_k is not None:
        # Minimize sum of top_k: use auxiliary variables
        # t_k >= conv(a,g)[k] for all k, minimize sum(t_k) over top k
        # This is equivalent to: min sum_k t_k s.t. t_k >= A[k,:]@g, t_k >= 0
        # But that minimizes ALL t_k, not just top_k.
        # For top_k, we can use: min sum_k max(0, A[k,:]@g - threshold)
        # Simpler: just minimize max (which is top-1)
        M = cp.Variable()
        obj = M
        constraints = [A @ g <= M, cp.sum(g) == 1]
    else:
        M = cp.Variable()
        obj = M
        constraints = [A @ g <= M, cp.sum(g) == 1]
    
    prob = cp.Problem(cp.Minimize(obj), constraints)
    
    solvers = [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def lp_norm_gradient(a, p):
    """
    Compute gradient of ||conv(a,a)||_p where a is normalized (sum=1).
    Uses FFT for efficiency.
    """
    n = len(a)
    c = compute_conv(a)
    c = np.maximum(c, 0)  # numerical safety
    
    c_max = float(np.max(c))
    if c_max < 1e-15:
        return np.zeros(n)
    
    # w[k] = c[k]^(p-1) / sum(c^p)^(1/p) * (1/p) ... 
    # Actually, let's compute the gradient of f(a) = (sum_k c[k]^p)^(1/p)
    # df/da[i] = (1/p) * (sum c[k]^p)^(1/p - 1) * sum_k p * c[k]^(p-1) * dc[k]/da[i]
    # = (sum c[k]^p)^(1/p - 1) * sum_k c[k]^(p-1) * dc[k]/da[i]
    # = ||c||_p^(1-p) * sum_k c[k]^(p-1) * 2*a[k-i] (for valid k-i)
    
    w = np.power(c, p - 1)
    
    # g[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
    # This is the cross-correlation of a and w
    N = 1
    while N < 3 * n:
        N <<= 1
    A_fft = fft(a, N)
    W_fft = fft(w, N)
    g_full = ifft(np.conj(A_fft) * W_fft).real
    g = g_full[:n]
    
    norm_p = float(np.sum(c ** p)) ** (1.0 / p)
    scale = 2.0 * (norm_p ** (1.0 - p))
    
    return scale * g


def lp_norm_smoothed_descent(a, time_budget, p_init=20, p_end=200, lr_init=None, verbose=False):
    """
    Nesterov accelerated projected gradient descent on ||conv(a,a)||_p.
    a is normalized to sum=1.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if lr_init is None:
        lr_init = 5.0 / np.sqrt(n)
    
    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        p = p_init * (p_end / p_init) ** (progress ** 0.5)
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:

        else:
            y = y / s_y
        
        # Compute gradient
        g = lp_norm_gradient(y, p)

        v = beta * v - lr * g
        a_new = y + v
        a_new = project_simplex(a_new)

            best_a = a_new.copy()

            v = np.zeros(n)

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
