def compute_conv(a):

    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):

    n = len(a)

    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix such that A_mat @ g = convolve(a, g)."""
    n = len(a)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A_mat[j:j + r, j] = a[:r]
    return A_mat

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Multi-length LP-driven optimization with pure-LP fixed-point iteration,
adaptive blending, top-k peak gradient descent, diversity injection, and
coordinate descent polishing. Key differences from prior art: (1) pure LP
iteration (t=1) before blending, (2) dual-guided gradient focusing, (3)
wider n exploration, (4) more aggressive diversity injection, (5) better
time allocation focused on most promising lengths.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build sparse-friendly matrix A_mat (2n-1 x n) such that A_mat @ g = convolve(a, g)."""
    n = len(a)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A_mat[j:j + r, j] = a[:r]
    return A_mat


def solve_lp(a, timeout=10.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_opt, M_opt, dual_vars) or (None, None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A_mat = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    dual_vars = None
    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.MOSEK]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    # Try to get dual variables for the inequality constraints
                    try:
                        dual_cons = constraints[0].dual_value
                        if dual_cons is not None:
                            dual_vars = np.maximum(dual_cons, 0)
                    except Exception:
                        dual_vars = None
                    return g_val, M_val, dual_vars
        except Exception:
            continue
    return None, None, None


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def topk_gradient_fft(a, k=5):
    """
    Compute gradient using hard top-k selection for sharper focus on critical peaks.
    Returns (gradient, convolution).
    """
    n = len(a)
    conv = compute_conv(a)
    top_k_idx = np.argsort(conv)[::-1][:k]
    weights = np.zeros(len(conv))
    for idx in top_k_idx:
        weights[idx] = 1.0 / k

    N = 1
    while N < 3 * n:
        N <<= 1
    W = fft(weights, N)
    A = fft(a, N)
    g_full = ifft(W * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def gradient_descent_phase(a, time_budget=60.0, tau_init=20.0, tau_end=800.0,

    best_obj = compute_obj(a)

        # Nesterov lookahead

        s_y = float(np.sum(y))

            best_a = a_new.copy()
            stagnant = 0
        else:
            stagnant += 1

        a = a_new
        it += 1

        # Decay and reset

        if stagnant > 20000:
            a = best_a.copy()
            v = np.zeros(n)

def coordinate_descent(a, time_budget=60.0, seed=0):
    """
    Coordinate descent with adaptive step sizes.
    For each coordinate, try multiple step sizes and keep the best.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Precompute: for each coordinate p, which lags are affected
    # conv_k changes when a_p changes: lags k where a_{k-p} exists (0 <= k-p < n)
    # i.e., k in [p, p+n-1]

    base_step = 0.005 / np.sqrt(n)

    while time.monotonic() - start < time_budget:
        order = np.arange(n)
        rng.shuffle(order)

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            old_val = a[i]
            if old_val < 1e-15:
                # Try turning it on
                for step in [base_step, 2 * base_step]:
                    a_try = a.copy()
                    a_try[i] = step
                    s_try = float(np.sum(a_try))
                    if s_try > 1e-10:
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try.copy()
            else:
                # Try various perturbations
                for step in [base_step, -base_step, 2 * base_step, -2 * base_step,
                             base_step * 0.3, -base_step * 0.3]:
                    a_try = a.copy()
                    a_try[i] = max(0.0, a_try[i] + step)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-15:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try.copy()

def lp_fixed_point_iteration(a, time_budget=120.0, seed=0, verbose=False):

        return a
    a = a / s

    start = time.monotonic()
    rng = np.random.default_rng(seed)

    recent_objs = []
    t_blend = 1.0  # Start with pure LP iteration
    it = 0
    consecutive_no_improve = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp, dual_vars = solve_lp(a, timeout=10.0)
        if g_lp is None:
            break

        it += 1

        # Try pure LP solution first, then blends
        candidates = [(1.0, g_lp)]
        # Add blend candidates
        for t in [0.5, 0.3, 0.2, 0.1, 0.8]:
            a_cand = (1.0 - t) * a + t * g_lp
            a_cand = np.maximum(a_cand, 0)
            a_cand = a_cand / a_cand.sum()
            candidates.append((t, a_cand))

        best_cand_obj = best_obj
        best_cand_a = best_a.copy()
        best_t = t_blend

        for t, a_cand in candidates:
            obj_cand = compute_obj(a_cand)
            if obj_cand < best_cand_obj:
                best_cand_obj = obj_cand
                best_cand_a = a_cand.copy()
                best_t = t

        if best_cand_obj < best_obj - 1e-14:
            best_obj = best_cand_obj
            best_a = best_cand_a.copy()
            a = best_cand_a.copy()
            consecutive_no_improve = 0
            # If pure LP worked, keep t=1
            if best_t == 1.0:
                t_blend = 1.0
            else:
                t_blend = max(0.1, t_blend * 0.8)
        else:
            consecutive_no_improve += 1
            # Deteriorating: reduce blend factor and inject diversity

            a = (1.0 - 0.05) * a + 0.05 * rng.dirichlet(np.ones(n))
            a = a / a.sum()

                a = a / a.sum()
                consecutive_no_improve = 0
                if time.monotonic() - start > time_budget * 0.5:

            print(f"    LP-FP it{it}: obj={best_obj:.8f}, t={t_blend:.3f}, M={M_lp:.6f}, {elapsed:.0f}s")

def gp_random_search(a, time_budget=60.0, seed=0):

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    elif mode == 4:
        freqs = rng.integers(1, 8, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Smooth random
        a = rng.uniform(0, 1, n)
        k = max(2, n // 40)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 6:
        # Bimodal
        c1, c2 = n // 3, 2 * n // 3
        w = max(1, n // 10)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 7:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 8:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1, 1)
        a = a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n))
    return a


def optimize_for_length(a_init, n, time_budget=200.0, seed=0, verbose=False):
    """
    Full optimization pipeline for a given length:
    1. LP fixed-point iteration
    2. Gradient descent (smoothed)
    3. LP fixed-point iteration again
    4. Top-k gradient descent
    5. Random search
    6. Coordinate descent
    """
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def update_best(a_try):
        nonlocal best_a, best_obj
        if a_try is not None:
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try / a_try.sum()
                if verbose:
                    print(f"      n={n}: obj={best_obj:.8f} **")
        return best_a

    # Phase 1: LP fixed-point
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 30:
        if verbose:
            print(f"    [n={n}] LP-FP phase ({min(remaining*0.35, 80):.0f}s)")
        a_lp = lp_fixed_point_iteration(best_a, time_budget=min(remaining * 0.35, 80),
                                         seed=seed, verbose=verbose)
        best_a = update_best(a_lp)

    # Phase 2: Smoothed gradient descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        if verbose:
            print(f"    [n={n}] GD phase ({min(remaining*0.25, 60):.0f}s)")
        a_gd = gradient_descent_phase(best_a, time_budget=min(remaining * 0.25, 60),
                                       tau_init=20.0, tau_end=600.0, seed=seed+1)
        best_a = update_best(a_gd)

    # Phase 3: LP again from new point
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 25:
        if verbose:
            print(f"    [n={n}] LP-FP phase 2 ({min(remaining*0.25, 60):.0f}s)")
        a_lp2 = lp_fixed_point_iteration(best_a, time_budget=min(remaining * 0.25, 60),
                                          seed=seed+2, verbose=False)
        best_a = update_best(a_lp2)

    # Phase 4: Top-k gradient (sharper)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        if verbose:
            print(f"    [n={n}] Top-k GD phase ({min(remaining*0.15, 40):.0f}s)")
        a_tk = gradient_descent_phase(best_a, time_budget=min(remaining * 0.15, 40),
                                       tau_init=50.0, tau_end=1000.0,
                                       seed=seed+3, topk_mode=True, topk_count=3)
        best_a = update_best(a_tk)

    # Phase 5: Random search for diversity
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        if verbose:
            print(f"    [n={n}] Random search phase ({min(remaining*0.1, 30):.0f}s)")
        a_rs = gp_random_search(best_a, time_budget=min(remaining * 0.1, 30), seed=seed+4)
        best_a = update_best(a_rs)

    # Phase 6: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        if verbose:
            print(f"    [n={n}] CD polish ({min(remaining*0.2, 40):.0f}s)")
        a_cd = coordinate_descent(best_a, time_budget=min(remaining * 0.2, 40), seed=seed+5)
        best_a = update_best(a_cd)

    # Final LP if time
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        a_final_lp = lp_fixed_point_iteration(best_a, time_budget=min(remaining * 0.5, 40),
                                               seed=seed+6, verbose=False)
        best_a = update_best(a_final_lp)

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Explores multiple sequence lengths with
    LP-driven optimization, gradient descent, and coordinate descent.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0  # 30s safety margin

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_parent = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    # Determine which lengths to try
    # Focus on lengths around the parent and some larger ones
    n_candidates = sorted(set([
        n_parent,
        n_parent - 100 if n_parent > 500 else 500,
        n_parent + 100,
        n_parent + 200,
        n_parent + 400,
        600, 800, 1000, 1200, 1500, 2000
    ]))
    # Filter to reasonable range
    n_candidates = [n for n in n_candidates if 400 <= n <= 2500]
    # Sort by proximity to parent first, then novel
    n_candidates = sorted(n_candidates, key=lambda x: abs(x - n_parent))
    # Limit to avoid too many
    n_candidates = n_candidates[:8]

    print(f"\nTarget lengths to explore: {n_candidates}")
    print(f"Initial best: {best_score:.8f} (n={len(best_seq)})")

    # Allocate time: give more time to lengths near the current best
    # Use a "tournament" approach: quick trial on each, then focus on best 2-3

    # Phase A: Quick trial on each length (30s each)
    trial_results = {}
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        if n_try == len(best_seq):
            trial_results[n_try] = (best_seq.copy(), best_score)
            continue

        a_init = resample_sequence(best_seq, n_try)
        a_init = a_init / a_init.sum()
        # Quick optimization (just LP for 30s)
        a_quick = lp_fixed_point_iteration(a_init, time_budget=30.0, seed=seed + n_try)
        obj_quick = compute_obj(a_quick)
        trial_results[n_try] = (a_quick, obj_quick)
        print(f"  Trial n={n_try}: obj={obj_quick:.8f}")

    # Phase B: Focus on the top 2-3 lengths
    sorted_trials = sorted(trial_results.items(), key=lambda x: x[1][1])
    focus_lengths = [n for n, _ in sorted_trials[:3]]
    print(f"\nFocus lengths: {focus_lengths}")

    for n_focus in focus_lengths:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_start = trial_results[n_focus][0]
        # Make sure it's the right length
        if len(a_start) != n_focus:
            a_start = resample_sequence(a_start, n_focus)
            a_start = a_start / a_start.sum()

        # Full optimization pipeline
        alloc = min(remaining * 0.5, 300.0)
        if verbose_alloc := alloc > 50:
            a_opt = optimize_for_length(a_start, n_focus, time_budget=alloc,
                                         seed=seed + n_focus * 10, verbose=True)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  ** n={n_focus}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_focus}: obj={obj_opt:.8f}")

    # Phase C: Diverse restarts on the best length
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    print(f"\nPhase C: Diverse restarts (n={n_best}), remaining={remaining:.0f}s")

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        mode = (r * 3) % 9
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with current best to keep it in the right basin
        blend = 0.3
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.25, 100.0)
        if alloc > 30:
            a_opt = optimize_for_length(a_r, n_best, time_budget=alloc,
                                         seed=seed + 1000 + r * 100, verbose=False)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_opt:.8f}")

    # Phase D: Final intensive polish on the best sequence
    remaining = deadline - time.monotonic()
    print(f"\nPhase D: Final polish, remaining={remaining:.0f}s")

    if remaining > 50:
        # Fast LP
        alloc = min(remaining * 0.3, 80.0)
        a_lp = lp_fixed_point_iteration(best_seq, time_budget=alloc,
                                         seed=seed + 5000, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        # High-tau GD
        alloc = min(remaining * 0.3, 60.0)
        a_gd = gradient_descent_phase(best_seq, time_budget=alloc,
                                       tau_init=100.0, tau_end=2000.0,
                                       seed=seed + 6000)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Coordinate descent
        alloc = min(remaining * 0.4, 50.0)
        a_cd = coordinate_descent(best_seq, time_budget=alloc, seed=seed + 7000)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
