def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def golden_section_line_search(a, g, n_evals=8):

def spectral_shape_init(n, mode=0, rng=None):
    """Generate initialization with spectral shaping."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    if mode == 0:
        # Uniform
        return np.ones(n) / n
    elif mode == 1:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
        return a / a.sum()
    elif mode == 2:
        # Two-peak comb
        a = np.zeros(n)
        a[::2] = 1.0
        return a / a.sum()
    elif mode == 3:
        # Modulated uniform
        x = np.arange(n)
        c = rng.uniform(0.3, 0.7)
        freq = rng.uniform(0.3, 2.0)
        a = 1.0 + c * np.cos(2 * np.pi * freq * x / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 4:
        # Gaussian + comb
        x = np.arange(n)
        center = n // 2
        a = np.exp(-0.5 * ((x - center) / (n * 0.25)) ** 2)
        a += 0.3 * (1 + np.cos(np.pi * x / n * 3))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 5:
        # Benford-like
        x = np.arange(1, n + 1)
        a = 1.0 / np.log10(x + 1)
        return a / a.sum()
    elif mode == 6:
        # Random smooth
        raw = np.zeros(n)
        for _ in range(5):
            center = rng.integers(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            raw += np.exp(-((np.arange(n) - center) ** 2) / (2 * width ** 2))
        return raw / raw.sum()
    else:
        # Arithmetic progression based
        r = rng.uniform(0.9, 1.1)
        a = r ** np.arange(n)
        return a / a.sum()

    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) spectral-shape diverse initializations, (2) fast projected gradient
peak-shaving with adaptive tau, (3) dual-guided LP with active-set refinement,
(4) pair-based coordinate descent for fine-tuning, (5) systematic multi-n
exploration, and (6) simulated-annealing restarts. Target: beat 1.5027.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak_full(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_active(a, active_indices):
    """Solve LP: min sum(conv(a,g)[k] for k in active), sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    active_A = A[active_indices, :]
    g = cp.Variable(n, nonneg=True)
    obj = cp.sum(active_A @ g)
    constraints = [cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(obj), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)

                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))

def golden_section_line_search(a, g, n_evals=6):
    """Find optimal t in [0,1] minimizing compute_obj((1-t)*a + t*g)."""
    phi = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)
        a1 = np.maximum((1 - t1) * a + t1 * g, 0)
        s1 = float(np.sum(a1))
        f1 = compute_obj(a1 / s1) if s1 > 1e-10 else np.inf
        a2 = np.maximum((1 - t2) * a + t2 * g, 0)
        s2 = float(np.sum(a2))
        f2 = compute_obj(a2 / s2) if s2 > 1e-10 else np.inf
        if f1 < f2:
            hi = t2
        else:
            lo = t1
    t_best = (lo + hi) / 2
    a_best = np.maximum((1 - t_best) * a + t_best * g, 0)
    s_best = float(np.sum(a_best))
    if s_best > 1e-10:
        a_best = a_best / s_best
    return a_best


def smoothed_gradient_fft(a, tau):
    """Gradient of log-sum-exp smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv

        it += 1
        if verbose and it % 200 == 0:
            print(f"      PS it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.4f}")

    return best_a


def pair_cd(a, time_budget=60.0, verbose=False):
    """
    Pair-based coordinate descent: for the dominant peak, find (src, tgt) pairs
    that reduce it when moving mass from src to tgt.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(9999)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_k = int(np.argmax(conv))

        # Find top contributing pairs to the peak
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)

        if i_start > i_end:
            break

        # Compute contributions: a[i] * a[max_k - i]
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = max_k - i_vals
        contribs = a[i_vals] * a[j_vals]
        top_pairs = np.argsort(contribs)[::-1][:5]

        improved = False
        for tp in top_pairs:
            src = int(i_vals[tp])
            # Target: a position that minimizes new peak contributions
            # Heuristic: positions far from current peak contributors
            targets = [
                (src + n // 3) % n,
                (src + 2 * n // 3) % n,
                rng.integers(0, n),
                n - 1 - src,
                (src + n // 2) % n,
            ]

            for tgt in targets:
                tgt = int(tgt) % n
                if tgt == src:
                    continue
                for delta_frac in [0.02, 0.01, 0.005, 0.002]:
                    delta = delta_frac * a[src]
                    if delta > a[src] * 0.9:
                        continue
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[tgt] += delta
                    a_try = a_try / a_try.sum()
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-15:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break

        if verbose and it % 50 == 0:
            print(f"      PCD it{it}: obj={best_obj:.8f}")

    return best_a


def dual_guided_lp(a, time_budget=100.0, verbose=False):
    """
    LP-based optimization with active-set refinement.
    Stage 1: min max LP -> find active constraints
    Stage 2: min sum on active set -> refined direction
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(7777)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        g1, M1 = solve_lp_minpeak_full(a)
        if g1 is None:
            break

        # Find active set: where conv(a, g1) is near max
        n_a = len(a)
        size = 1
        while size < 2 * n_a:
            size <<= 1
        fa = rfft(a, size)
        fg = rfft(g1, size)
        conv_ag1 = irfft(fa * fg, size)[:2 * n_a - 1]

        max_val = float(np.max(conv_ag1))
        tol = 0.003 * max(max_val, 1e-10)
        active = np.where(conv_ag1 >= max_val - tol)[0]

        if len(active) == 0:
            active = np.array([np.argmax(conv_ag1)])
        # Limit to top 40 for efficiency
        if len(active) > 40:
            active = active[np.argsort(conv_ag1[active])[::-1][:40]]

        # Solve reduced LP
        g2 = solve_lp_active(a, active)

        # Choose best direction via line search
        if g2 is not None:
            test1 = golden_section_line_search(a, g1, n_evals=4)
            obj1 = compute_obj(test1)
            test2 = golden_section_line_search(a, g2, n_evals=4)
            obj2 = compute_obj(test2)
            if obj2 < obj1:
                a_new, obj_new = test2, obj2
            else:
                a_new, obj_new = test1, obj1
        else:
            a_new = golden_section_line_search(a, g1, n_evals=5)
            obj_new = compute_obj(a_new)

        it += 1
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 3 == 0:
                print(f"      DGLP it{it}: obj={best_obj:.8f}, M={M1:.6f}, |S|={len(active)}")
        else:
            stall += 1
            if stall > 15:
                scale = min(0.01 + 0.005 * stall, 0.08)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if it > 150:
                    break

    return best_a


def spectral_shape_init(n, mode=0, rng=None):
    """Generate diverse initializations with spectral structure."""
    if rng is None:
        rng = np.random.default_rng(0)

    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
        return a / a.sum()
    elif mode == 2:
        a = np.zeros(n)
        a[::2] = 1.0
        return a / a.sum()
    elif mode == 3:
        x = np.arange(n)
        c = rng.uniform(0.3, 0.7)
        freq = rng.uniform(0.3, 2.0)
        a = 1.0 + c * np.cos(2 * np.pi * freq * x / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 4:
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - n // 2) / (n * 0.25)) ** 2)
        a += 0.3 * (1 + np.cos(3 * np.pi * x / n))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 5:
        x = np.arange(1, n + 1)
        a = 1.0 / np.log10(x + 1)
        return a / a.sum()
    elif mode == 6:
        raw = np.zeros(n)
        for _ in range(5):
            center = rng.integers(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            raw += np.exp(-((np.arange(n) - center) ** 2) / (2 * width ** 2))
        return raw / raw.sum()
    else:
        r = rng.uniform(0.92, 1.08)
        a = r ** np.arange(n)
        return a / a.sum()


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def optimize_at_n(a_init, time_budget=100.0, verbose=False):
    """
    Optimize a fixed-length sequence using a pipeline:
    1. Peak shaving (fast, adaptive tau)
    2. Dual-guided LP (principled direction)
    3. Pair CD (fine-tuning)
    4. Final peak shaving
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    # Step 1: Peak shaving
    t1 = min(time_budget * 0.35, 50.0)
    if t1 > 10:
        a_ps = peak_shaving(a, time_budget=t1, top_k=5, verbose=verbose)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_obj:
            best_obj = obj_ps
            best_a = a_ps.copy()
            if verbose:
                print(f"        PS: {best_obj:.8f} **")

    # Step 2: Dual-guided LP
    t2 = min(time_budget * 0.35, 70.0)
    if t2 > 15:
        a_lp = dual_guided_lp(best_a, time_budget=t2, verbose=verbose)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            if verbose:
                print(f"        LP: {best_obj:.8f} **")

    # Step 3: Pair CD
    t3 = min(time_budget * 0.15, 30.0)
    if t3 > 8:
        a_cd = pair_cd(best_a, time_budget=t3)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()
            if verbose:
                print(f"        PCD: {best_obj:.8f} **")

    # Step 4: Final peak shaving
    t4 = min(time_budget * 0.15, 30.0)
    if t4 > 8:
        a_final = peak_shaving(best_a, time_budget=t4, top_k=3)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
            if verbose:
                print(f"        PS2: {best_obj:.8f} **")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization for step function coefficients.
    Strategy: start from parent, optimize with peak shaving + LP,
    explore multiple n values, diverse restarts, and intensive final polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent: {e}")

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Optimize at current n ===
    print(f"\n=== Phase 1: Optimize n={n_current} (budget ~35%) ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.35, 380.0)
    a_opt = optimize_at_n(best_seq, time_budget=alloc, verbose=True)
    obj_opt = compute_obj(a_opt)
    if obj_opt < best_score:
        best_score = obj_opt
        best_seq = a_opt.copy()
        print(f"  Phase 1 result: obj={best_score:.8f} **")
    else:
        print(f"  Phase 1 result: obj={obj_opt:.8f}")

    # === Phase 2: Systematic n exploration ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()

    # Try nearby n values and some outliers
    n_candidates = sorted(set([
        n_current - 50, n_current - 100, n_current + 50, n_current + 100,
        n_current + 200, n_current + 300,
        max(300, n_current - 200), min(2500, n_current + 500),
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 2500 and abs(n - n_current) > 30]
    n_candidates.sort(key=lambda x: abs(x - n_current))

    n_tried = 0
    for n_try in n_candidates[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 70:
            break
        if n_tried >= 5:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc_n = min(remaining * 0.12, 70.0)
        a_try_opt = optimize_at_n(a_try, time_budget=alloc_n)
        obj_try = compute_obj(a_try_opt)

        n_tried += 1
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try_opt.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 3: Diverse restarts ===
    print(f"\n=== Phase 3: Diverse Restarts ===")
    remaining = deadline - time.monotonic()

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = (r * 2 + 1) % 7
        a_r = spectral_shape_init(n_current, mode=mode, rng=rng)
        # Blend with current best (more weight on best as we go)
        blend = 0.5 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc_r = min(remaining * 0.12, 60.0)
        a_r_opt = optimize_at_n(a_r, time_budget=alloc_r)
        obj_r = compute_obj(a_r_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 4: Intensive final polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Long peak shaving with high tau
    if remaining > 30:
        alloc = min(remaining * 0.3, 120.0)
        a_ps = peak_shaving(best_seq, time_budget=alloc, top_k=3)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  Final PS: obj={best_score:.8f} **")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.3, 60.0)
        a_lp = dual_guided_lp(best_seq, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Pair CD polish
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.2, 30.0)
        a_cd = pair_cd(best_seq, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final PCD: obj={best_score:.8f} **")

    # Last peak shave
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.3, 30.0)
        a_ps2 = peak_shaving(best_seq, time_budget=alloc, top_k=2)
        obj_ps2 = compute_obj(a_ps2)
        if obj_ps2 < best_score:
            best_score = obj_ps2
            best_seq = a_ps2.copy()
            print(f"  Final PS2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
