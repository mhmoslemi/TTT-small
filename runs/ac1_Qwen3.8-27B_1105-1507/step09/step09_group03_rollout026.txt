import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
        # So for column j, A[k, j] = a[k-j] for k = j, j+1, ..., j+n-1
        # But k goes from 0 to 2n-2
        start = max(0, j - (n - 1))  # Actually let me think again
        # conv(a,g)[k] = sum_j a[k-j] * g[j] for valid indices
        # So A[k,j] = a[k-j] if 0 <= k-j < n

        # Valid: 0 <= k-j < n => j <= k <= j+n-1

        k_start = j

        A[k_start:k_end+1, j] = a[0:k_end - k_start + 1]
    return A


def solve_lp_direction(a, warm_start=None):

def line_search(a, g, n_evals=10):

    phi = (np.sqrt(5) - 1) / 2

    a_center = 0.5 * a + 0.5 * g

    for _ in range(n_evals):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)
        
        a1 = (1 - t1) * a + t1 * g
        a1 = np.maximum(a1, 0)
        s1 = float(np.sum(a1))
        f1 = compute_obj(a1) if s1 > 1e-10 else np.inf

        f2 = compute_obj(a2) if s2 > 1e-10 else np.inf
        
        if f1 < f2:
            hi = t2
        else:
            lo = t1
    
    t_best = (lo + hi) / 2
    a_best = (1 - t_best) * a + t_best * g
    a_best = np.maximum(a_best, 0)
    s_best = float(np.sum(a_best))
    if s_best > 1e-10:
        a_best = a_best / s_best

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):

            a += 0.2 * np.sin(2 * np.pi * f * x + p)

        a = 0.5 + 0.5 * np.cos(x)

def smooth_descent(a, time_budget=60.0, tau=100.0, verbose=False):
    """
    Gradient descent on the log-sum-exp smoothed objective.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 2.0 / np.sqrt(n)
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Softmax weights for smoothing
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient: d/da sum_k p[k] * conv(a,a)[k]
        # = sum_k p[k] * d(conv(a,a)[k])/da
        # d(conv(a,a)[k])/da[i] = a[k-i] + a[i-k] (with boundary handling)
        # Using FFT: gradient = 2 * ifft(fft(p) * conj(fft(a)))[:n]
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        grad_full = ifft(P * np.conj(A)).real
        grad = 2.0 * grad_full[:n]
        
        # Project gradient onto tangent space of simplex
        grad_tangent = grad - np.mean(grad)
        
        a_new = a - lr * grad_tangent
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            # Reduce learning rate
            lr *= 0.9

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: LP-based Frank-Wolfe with dual-guided perturbation, spectral initialization,
multi-n exploration, convex combination pool, and peak-specific coordinate descent.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        k_start = j
        k_end = min(m, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    m_val = float(M.value)
                    return g_val, m_val
        except Exception:
            continue
    return None, None


def line_search(a, direction, n_evals=8):
    """Golden section search for best t in [0,1]."""
    phi = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    
    for _ in range(n_evals):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)
        
        a1 = (1 - t1) * a + t1 * direction
        a1 = np.maximum(a1, 0)
        s1 = float(np.sum(a1))
        f1 = compute_obj(a1) if s1 > 1e-10 else np.inf
        
        a2 = (1 - t2) * a + t2 * direction
        a2 = np.maximum(a2, 0)
        s2 = float(np.sum(a2))
        f2 = compute_obj(a2) if s2 > 1e-10 else np.inf
        
        if f1 < f2:
            hi = t2
        else:
            lo = t1
    
    t_best = (lo + hi) / 2
    a_best = (1 - t_best) * a + t_best * direction
    a_best = np.maximum(a_best, 0)
    s_best = float(np.sum(a_best))
    if s_best > 1e-10:
        a_best = a_best / s_best
    return a_best


def lp_frank_wolfe(a, time_budget=100.0, verbose=False):
    """LP-based Frank-Wolfe with momentum and adaptive perturbation."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(54321)
    
    it = 0
    stall = 0
    momentum = 0.4
    v = np.zeros(n)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break
        
        g, M = solve_lp_direction(a)
        if g is None:
            # Fallback: simple gradient step
            conv = compute_conv(a)
            k_max = int(np.argmax(conv))
            # Gradient of f_{k_max} at a
            grad = np.zeros(n)
            for j in range(n):
                idx = k_max - j
                if 0 <= idx < n:
                    grad[j] = 2.0 * a[idx]
            # Project onto tangent space
            grad_tan = grad - np.mean(grad)
            g = np.maximum(a - 0.1 * grad_tan, 0)
            s_g = float(np.sum(g))
            if s_g > 1e-10:
                g = g / s_g
            else:
                g = a.copy()
            M = float(np.max(conv))
        
        # Momentum
        v = momentum * v + (g - a)
        a_mom = np.maximum(a + v, 0)
        s_mom = float(np.sum(a_mom))
        if s_mom < 1e-10:
            a_mom = a.copy()
            v = np.zeros(n)
        else:
            a_mom = a_mom / s_mom
        
        # Line search
        a_new = line_search(a, a_mom, n_evals=6)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Try reduced steps
            found = False
            for scale in [0.5, 0.25, 0.1]:
                a_try = (1 - scale) * a + scale * g
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    stall = 0
                    found = True
                    break
            
            if not found:
                stall += 1
                momentum = max(0.1, momentum * 0.9)
                v *= 0.2
                a = best_a.copy()
                
                if stall > 3:
                    # Adaptive perturbation
                    pert_scale = 0.005 + 0.003 * min(stall, 30)
                    a = best_a * np.exp(rng.normal(0, pert_scale, n))
                    a = np.maximum(a, 0)
                    s_a = float(np.sum(a))
                    if s_a > 1e-10:
                        a = a / s_a
                    else:
                        a = best_a.copy()
                
                if stall > 60:
                    break
        
        it += 1
        momentum = min(0.75, 0.4 + 0.003 * it)
        
        if verbose and it % 5 == 0:
            print(f"    FW it{it}: obj={best_obj:.8f}, M={M:.6f}, stall={stall}")
    
    return best_a


def peak_coordinate_descent(a, time_budget=60.0, verbose=False):
    """Coordinate descent targeting the dominant peak."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))
        
        # Find dominant contributors to the peak
        i_lo = max(0, k_max - (n - 1))
        i_hi = min(n - 1, k_max)
        
        if i_lo > i_hi:
            break
        
        # Get contributions
        contribs = []
        for i in range(i_lo, i_hi + 1):
            j = k_max - i
            if 0 <= j < n:
                contribs.append((a[i] * a[j], i, j))
        
        contribs.sort(reverse=True)
        
        improved = False
        for cv, src, dst in contribs[:5]:
            for delta_frac in [0.02, 0.01, 0.005]:
                delta = delta_frac * a[src]
                if delta < 1e-15:
                    continue
                
                # Try several targets
                targets = set()
                targets.add(src - max(1, n // 10) if src - max(1, n // 10) >= 0 else 0)
                targets.add(src + max(1, n // 10) if src + max(1, n // 10) < n else n - 1)
                targets.add(int(rng.integers(0, n)))
                targets.add(int(rng.integers(0, n)))
                targets.add(n - 1 - src)
                
                for tgt in targets:
                    tgt = int(tgt) % n
                    if tgt == src:
                        continue
                    
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[tgt] += delta
                    if a_try[src] < 0:
                        continue
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 40:
                break
        
        if verbose and it % 20 == 0:
            print(f"    PCD it{it}: obj={best_obj:.8f}")
    
    return best_a


def resample_sequence(a, new_n):
    """Resample via nearest neighbor (preserves structure better)."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    # Nearest neighbor resampling
    idx = np.round(np.linspace(0, n - 1, new_n)).astype(int)
    return a[idx].copy()


def resample_linear(a, new_n):
    """Resample via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def spectral_init(n, rng):
    """
    Initialize with a sequence whose spectrum has a desired shape.
    Try top-hat, Gaussian, and rectangular spectral shapes.
    """
    mode = rng.integers(0, 4)
    x = np.linspace(0, 1, n)
    
    if mode == 0:
        # Uniform (white noise in spectrum)
        a = np.ones(n)
    elif mode == 1:
        # Cosine taper
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif mode == 2:
        # Softplus of random Fourier features
        freqs = rng.integers(1, 3, size=2)
        phases = rng.uniform(0, 2 * np.pi, size=2)
        a = np.ones(n) * 0.8
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
    else:
        # Triangular
        a = 1.0 - np.abs(2 * x - 1)
    
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def smooth_descent(a, time_budget=60.0, tau_start=30.0, tau_end=500.0, verbose=False):
    """Gradient descent on log-sum-exp smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 3.0 / np.sqrt(n)
    beta = 0.85  # Nesterov momentum
    v = np.zeros(n)
    y = a.copy()
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget * 0.9, 1e-6))
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.5)
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = a.copy()
        else:
            y = y / s_y
        
        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient via FFT
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_ft = fft(y, N)
        grad_full = ifft(P * np.conj(A_ft)).real
        grad = 2.0 * grad_full[:n]
        
        # Project to tangent space
        grad_tan = grad - np.mean(grad)
        
        # Nesterov update
        v = beta * v - lr * grad_tan
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)
        
        if it - best_it > 10000:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it
            lr = 3.0 / np.sqrt(n) * 0.5
        
        if verbose and it % 100 == 0:
            print(f"    SD it{it}: obj={best_obj:.8f}, tau={tau:.0f}")
    
    return best_a
