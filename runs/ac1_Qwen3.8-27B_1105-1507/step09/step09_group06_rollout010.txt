"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Primary method: LP-based alternating optimization (block coordinate descent) where
each iteration solves an LP to find the best direction, then blends.
Secondary: Smoothed gradient descent for exploration and escape from local minima.
Explores multiple sequence lengths and diverse initializations.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix (2n-1 x n) such that mat @ g = convolve(a, g)."""
    n = len(a)
    mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        mat[j:j + r, j] = a[:r]
    return mat


def solve_lp_minpeak(a, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A_mat = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    while N < 3 * n:

    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv

            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 15.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 4.0 / np.sqrt(n)
            best_it = it

    return best_a


def find_optimal_t(a, g, n_candidates=15):
    """Find optimal blending parameter t in [(1-t)*a + t*g] that minimizes objective."""
    best_t = 0.0
    best_obj = compute_obj(a)

    # Logarithmically spaced candidates
    ts = np.concatenate([
        np.linspace(0.01, 0.3, 5),
        np.linspace(0.3, 0.7, 5),
        np.linspace(0.7, 1.0, 5)
    ])

    # Quick pass
    for t in ts:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t

    # Refine around best_t
    if best_t > 0:
        step = 0.02
        for _ in range(3):
            for t_try in [best_t - step, best_t + step]:
                if 0 < t_try < 1:
                    a_try = (1.0 - t_try) * a + t_try * g
                    a_try = np.maximum(a_try, 0)
                    s = float(np.sum(a_try))
                    if s < 1e-10:
                        continue
                    a_try = a_try / s
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_t = t_try
            step *= 0.5

    return best_t, best_obj


def lp_alternating(a, time_budget=100.0, verbose=False, seed=0):
    """
    LP-based alternating optimization.
    Each iteration: solve LP for optimal g, find best t, update a.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    consecutive_no_improve = 0
    last_m = None

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, M_lp = solve_lp_minpeak(a, verbose=verbose)
        if g_lp is None:
            break

        # If M_lp is not improving, we might be at a fixed point
        if last_m is not None and abs(M_lp - last_m) < 1e-10:
            consecutive_no_improve += 1
        else:
            consecutive_no_improve = 0
        last_m = M_lp

        # Find optimal blend
        t_opt, obj_at_t = find_optimal_t(a, g_lp)

        if obj_at_t < best_obj - 1e-12:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            best_obj = obj_at_t
            best_a = a_new.copy()
            a = a_new.copy()
        elif t_opt < 0.02 or (time.monotonic() - start) > time_budget * 0.5:
            # If we can't improve much, try perturbing
            a = a * (1.0 + 0.02 * rng.normal(0, 1, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
        else:
            # Try a small step anyway to make progress
            t_small = 0.05
            a_new = (1.0 - t_small) * a + t_small * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            obj_new = compute_obj(a_new)
            if obj_new <= best_obj + 1e-10:
                a = a_new
            # Periodic perturbation
            if it % 20 == 0:
                a = a * (1.0 + 0.03 * rng.normal(0, 1, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                a = 0.8 * a + 0.2 * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()

        it += 1
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"      LP it{it}: obj={best_obj:.8f}, M={M_lp:.6f}, t={t_opt:.3f}, time={elapsed:.0f}s")

    return best_a


def peak_aware_perturbation(a, conv, scale=0.05, seed=0):
    """Perturb sequence with noise weighted by autocorrelation peak contributions."""
    n = len(a)
    rng = np.random.default_rng(seed)
    top_k = np.argsort(conv)[::-1][:10]
    contrib = np.zeros(n)
    for k in top_k:
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if 0 <= j < n:
                contrib[i] += a[i] * a[j]
    max_c = contrib.max() if contrib.max() > 0 else 1.0
    contrib = contrib / max_c
    noise = rng.normal(0, 1, n) * scale * (0.5 + contrib)
    a_new = a * np.exp(noise)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # U-shaped (high at edges)
        x = np.linspace(0, 1, n)
        a = x**2 + (1-x)**2
        a = a / a.sum()
    elif mode == 3:
        # W-shaped
        x = np.linspace(0, 1, n)
        a = (np.sin(2*np.pi*x)**4 + 0.1)
        a = a / a.sum()
    elif mode == 4:
        # Two bumps
        x = np.linspace(0, 1, n)
        a = np.exp(-20*(x-0.25)**2) + np.exp(-20*(x-0.75)**2) + 0.01
        a = a / a.sum()
    elif mode == 5:
        # Cosine taper
        x = np.linspace(0, np.pi, n)
        a = np.cos(x/2)**2
        a = a / a.sum()
    elif mode == 6:
        # Inverse cosine (U-shaped smooth)
        x = np.linspace(0.01, 0.99, n)
        a = 1.0 / np.cos(np.pi * (x - 0.5))**2
        a = a / a.sum()
    elif mode == 7:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x**(-0.3) + (1-x)**(-0.3)
        a = a / a.sum()
    elif mode == 8:
        # Smooth random
        a = rng.uniform(0, 1, n)
        k = max(2, n // 40)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 9:
        # Raised cosine with notch
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x)**2 - 0.3 * np.exp(-50*(x-0.5)**2)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def optimize_single_length(a_init, time_budget, seed=0, verbose=False):
    """Full optimization pipeline for a single sequence length."""
    n = len(a_init)
    a_init = np.maximum(a_init, 0).astype(np.float64)
    s = float(np.sum(a_init))
    if s < 1e-10:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()

    # Phase 1: Quick GD to get into the basin (short)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        gd_budget = min(remaining * 0.15, 30.0)
        a_gd = nesterov_smoothed_descent(best_a, time_budget=gd_budget,
                                          tau_init=10.0, tau_end=300.0, seed=seed)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_obj:
            best_obj = obj_gd
            best_a = a_gd.copy()
            if verbose:
                print(f"    GD: obj={best_obj:.8f}")

    # Phase 2: Main LP alternating (bulk of time)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        lp_budget = remaining * 0.6
        a_lp = lp_alternating(best_a, time_budget=lp_budget,
                              verbose=verbose, seed=seed)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            if verbose:
                print(f"    LP alt 1: obj={best_obj:.8f}")

    # Phase 3: Another LP round with perturbation
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        a_pert = peak_aware_perturbation(best_a, compute_conv(best_a),
                                          scale=0.03, seed=seed+100)
        a_pert = 0.7 * a_pert + 0.3 * best_a
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        lp_budget2 = min(remaining * 0.3, 60.0)
        a_lp2 = lp_alternating(a_pert, time_budget=lp_budget2,
                               verbose=verbose, seed=seed+200)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2.copy()
            if verbose:
                print(f"    LP alt 2: obj={best_obj:.8f}")

    # Phase 4: Final GD polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        gd_budget2 = min(remaining * 0.4, 30.0)
        a_gd2 = nesterov_smoothed_descent(best_a, time_budget=gd_budget2,
                                           tau_init=50.0, tau_end=800.0, seed=seed+300)
        obj_gd2 = compute_obj(a_gd2)
        if obj_gd2 < best_obj:
            best_obj = obj_gd2
            best_a = a_gd2.copy()

    # Phase 5: Final LP polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        a_final = lp_alternating(best_a, time_budget=min(remaining * 0.5, 40.0),
                                  verbose=verbose, seed=seed+400)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: explores multiple sequence lengths and initialization strategies.
    Uses LP-based alternating optimization as the primary method.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Phase 0: Load parent
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_parent = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_parent) / n_parent
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_parent}): obj={best_score:.8f}")

    # === Phase 1: Optimize parent length with LP ===
    print(f"\n=== Phase 1: LP optimization at n={n_parent} ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 30:
        a_opt, obj_opt = optimize_single_length(best_seq, p1_budget,
                                                 seed=seed, verbose=True)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")

    # === Phase 2: Explore different n values ===
    print(f"\n=== Phase 2: Multi-n exploration ===")
    n_current = len(best_seq)
    n_options = sorted(set([
        max(400, n_current - 200),
        max(300, n_current - 100),
        n_current + 100,
        n_current + 200,
        n_current + 400,
        n_current + 600,
        600, 800, 1000, 1200, 1500, 2000
    ]))
    # Filter to reasonable range and distinct from current
    n_options = [n for n in n_options if 300 <= n <= 2500 and abs(n - n_current) >= 80]
    n_options = n_options[:5]  # Try at most 5 different lengths

    for idx, n_try in enumerate(n_options):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        # Resample best sequence to new length
        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        # Optimize at this length
        alloc = min(remaining * 0.3, 180.0)
        if alloc > 30:
            a_opt, obj_opt = optimize_single_length(a_try, alloc,
                                                     seed=seed + 1000 + idx,
                                                     verbose=True)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                n_current = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 3: Diverse restarts at best n ===
    print(f"\n=== Phase 3: Diverse restarts at n={len(best_seq)} ===")
    n_best = len(best_seq)

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = (r * 3 + 2) % 10
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with best to stay in good basin
        blend = 0.3 + 0.05 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.25, 120.0)
        if alloc > 25:
            a_opt, obj_opt = optimize_single_length(a_r, alloc,
                                                     seed=seed + 2000 + r,
                                                     verbose=False)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_opt:.8f}")

    # === Phase 4: Intensive final polish ===
    print(f"\n=== Phase 4: Final intensive polish ===")
    remaining = deadline - time.monotonic()

    # Long LP optimization on best
    if remaining > 30:
        lp_alloc = min(remaining * 0.5, 200.0)
        a_final = lp_alternating(best_seq, time_budget=lp_alloc,
                                  verbose=True, seed=seed + 5000)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Quick GD polish
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_gd = nesterov_smoothed_descent(best_seq, time_budget=min(remaining * 0.4, 40.0),
                                          tau_init=100.0, tau_end=2000.0, seed=seed+6000)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_lp = lp_alternating(best_seq, time_budget=min(remaining * 0.5, 50.0),
                               verbose=True, seed=seed+7000)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP 2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
