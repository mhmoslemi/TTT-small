import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_minmax(a, active_indices=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    
    if active_indices is not None and len(active_indices) < 2 * n - 1:
        A_use = A[active_indices, :]
        M = cp.Variable()
        constraints = [A_use @ g <= M, cp.sum(g) == 1]
    else:
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    try:
        prob.solve(solver=cp.CLARABEL)
        if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
            g_val = np.maximum(g.value, 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(M.value)
    except Exception:
        pass
    
    # Fallback to other solvers
    for solver in [cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None

def smoothed_gradient(a, tau):
    """Compute smoothed gradient of max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    # Gradient of sum_k p[k] * conv[k] w.r.t. a[i]
    # conv[k] = sum_j a[j] * a[k-j]
    # d(conv[k])/da[i] = a[k-i] + a[i-k] (for valid indices)
    # So gradient[i] = sum_k p[k] * (a[k-i] + a[i-k])
    
    # Use FFT for efficiency
    N = 1
    while N < 4 * n:
        N <<= 1
    P = rfft(p[:N] if N <= len(p) else np.pad(p, (0, N - len(p))), N)
    A = rfft(a, N)
    
    # cross-correlation
    g = irfft(P * A, N)[:n] * 2  # factor of 2 for the symmetry
    
    return g, conv

def resample_sequence(a, new_n):

Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based polishing with adaptive active sets, L2-norm gradient descent (flat spectrum design),
Nesterov accelerated smoothed-max gradient, multi-length exploration, and diverse restarts.
Key innovations: L2-autocorrelation minimization as a warm-start for peak reduction,
improved momentum scheduling, and efficient CLARABEL-based LP solves.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minmax(a, active_indices=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()

    if active_indices is not None and 0 < len(active_indices) < 2 * n - 1:
        A_use = A[active_indices, :]
        constraints = [A_use @ g <= M, cp.sum(g) == 1]
    else:
        constraints = [A @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

            g_lp2, M_lp2 = solve_lp_minmax(a, active_indices=active)
            if g_lp2 is not None:
                g_dir = 0.5 * g_lp + 0.5 * g_lp2
            else:
                g_dir = g_lp
        else:
            g_dir = g_lp

        # Momentum update
        v = momentum * v + (g_dir - a)
        a_new = project_simplex(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            found = False
            for scale in [0.5, 0.25, 0.1, 0.05]:
                a_try = project_simplex(a + scale * v)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-15:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    v = scale * v
                    stall = 0
                    found = True
                    break
            if not found:
                stall += 1
                v *= 0.3
                if stall > 25:
                    a = best_a * np.exp(rng.normal(0, 0.02, n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    v = np.zeros(n)
                    momentum = max(0.3, momentum - 0.05)
                    stall = 0
                    if it > 150:
                        break

        it += 1
        momentum = min(0.85, 0.5 + 0.003 * it)

        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.10f}, M={M_lp:.8f}, |act|={len(active)}, t={elapsed:.0f}s")

    return best_a


def l2_gradient_descent(a, time_budget=150.0, verbose=False):
    """
    Minimize the L2 norm of the autocorrelation (sum_k conv[k]^2).
    This promotes a flat spectrum, which is a good starting point for peak minimization.
    Uses Nesterov acceleration with adaptive learning rate.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 1.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute L2 gradient: d(sum conv[k]^2)/da[m] = 2 * sum_k conv[k] * (a[k-m] + a[m-k])
        conv = compute_conv(y)
        # gradient[m] = 2 * (conv * flip(a))_cross[m]
        # Use FFT: gradient = 2 * irfft(FFT(conv) * conj(FFT(a)))
        N = 1
        while N < 3 * n:
            N <<= 1
        C = fft(conv, N)
        A_f = fft(y, N)
        grad = 2.0 * ifft(C * np.conj(A_f)).real[:n]

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9998)

        if it - best_it > 3000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 1.0 / np.sqrt(n) * 0.5
            best_it = it

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    L2 it{it}: obj={best_obj:.10f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a


def smoothed_max_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g = 2.0 * ifft(P * np.conj(A)).real[:n]
    return g, conv


def nesterov_smoothed_max(a, time_budget=150.0, tau_start=30.0, tau_end=1000.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed max of autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1))
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.6)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_max_gradient(y, tau)

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)

        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    NSM it{it}: obj={best_obj:.10f}, tau={tau:.1f}, t={elapsed:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = 1.0 / (1.0 + np.exp(-6.0 * (x - 0.5)))
    elif mode == 2:
        a = np.sin(np.pi * x) ** 2
    elif mode == 3:
        a = np.exp(-3.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-3.0 * (x - 0.7) ** 2)
    elif mode == 4:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
    elif mode == 5:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / (0.15 ** 2))
    elif mode == 6:
        a = 1.0 / (1.0 + 4.0 * (x - 0.5) ** 2)
    else:
        a = rng.uniform(0.1, 1.0, n)
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phases: LP polish -> L2 flat-spectrum GD -> LP polish -> Smoothed-max GD ->
    multi-length exploration -> diverse restarts -> final polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception as e:
        print(f"No parent available: {e}")

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Uniform init (n={n0}): obj={best_score:.10f}")

    # === Phase 1: LP Polish ===
    print(f"\n=== Phase 1: LP Polish ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 250.0)
    if p1_budget > 30:
        a1 = lp_polish(best_seq, time_budget=p1_budget, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase 1: obj={best_score:.10f} **")
        else:
            print(f"  Phase 1: obj={obj1:.10f}")

    # === Phase 2: L2 Flat-Spectrum Gradient Descent ===
    print(f"\n=== Phase 2: L2 Flat-Spectrum GD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 120.0)
    if p2_budget > 20:
        a2 = l2_gradient_descent(best_seq, time_budget=p2_budget, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase 2: obj={best_score:.10f} **")
        else:
            print(f"  Phase 2: obj={obj2:.10f}")

    # === Phase 3: LP Polish Round 2 ===
    print(f"\n=== Phase 3: LP Polish Round 2 ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.20, 180.0)
    if p3_budget > 20:
        a3 = lp_polish(best_seq, time_budget=p3_budget, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase 3: obj={best_score:.10f} **")
        else:
            print(f"  Phase 3: obj={obj3:.10f}")

    # === Phase 4: Smoothed Max Gradient Descent ===
    print(f"\n=== Phase 4: Smoothed Max GD ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 120.0)
    if p4_budget > 20:
        a4 = nesterov_smoothed_max(best_seq, time_budget=p4_budget,
                                   tau_start=50.0, tau_end=800.0, verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase 4: obj={best_score:.10f} **")
        else:
            print(f"  Phase 4: obj={obj4:.10f}")

    # === Phase 5: Multi-length Exploration ===
    print(f"\n=== Phase 5: Multi-length ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(300, n_best - 100),
        max(300, n_best - 50),
        n_best + 50,
        n_best + 100,
        n_best + 200,
        600, 700, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500 and abs(n - n_best) > 30]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.12, 40.0)
        if alloc > 15:
            a_try = lp_polish(a_try, time_budget=alloc * 0.7)
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_try = nesterov_smoothed_max(a_try, time_budget=min(alloc * 0.3, remaining * 0.5),
                                              tau_start=80, tau_end=500)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_r = diverse_init(n_best, rng, mode=r)

        # Blend with current best
        blend = 0.2 + 0.05 * r
        a_r = (1.0 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.12, 50.0)
        if alloc > 20:
            a_r = lp_polish(a_r, time_budget=alloc * 0.5)
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_r = nesterov_smoothed_max(a_r, time_budget=min(alloc * 0.3, remaining * 0.5),
                                            tau_start=60, tau_end=600)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={r}): obj={best_score:.10f} **")
        else:
            print(f"  Restart {r} (mode={r}): obj={obj_r:.10f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 40:
        alloc = min(remaining * 0.35, 100.0)
        a_f = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.10f} **")
        else:
            print(f"  Final LP: obj={obj_f:.10f}")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.35, 60.0)
        a_f = nesterov_smoothed_max(best_seq, time_budget=alloc,
                                    tau_start=100, tau_end=2000, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final SMGD: obj={best_score:.10f} **")
        else:
            print(f"  Final SMGD: obj={obj_f:.10f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 40.0)
        a_f = lp_polish(best_seq, time_budget=alloc)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP2: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
