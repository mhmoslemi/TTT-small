"""
Optimizes step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast scipy HiGHS LP solver for two-stage LP polishing,
dual-based cross-correlation flattening, and multi-length exploration.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    n = len(a)
    if n == 1:
        return np.array([a[0]**2])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(2 * n - 1, j + n)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_lp_minpeak(a):
    """
    Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    
    if res.success:
        g = np.maximum(res.x[:n], 0)
        M = res.x[-1]
        s = g.sum()
        if s > 1e-10:
            g = g / s
        return g, M
    return None, None


def solve_lp_min_sum_active(a, active_indices):
    """
    Solve: min sum_{k in active} conv(a,g)[k] s.t. sum(g)=1, g>=0.
    Returns g or None.
    """
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # Objective: minimize sum_{k in active} (A@g)[k] = (sum of active rows of A) @ g
    c = np.zeros(n)
    for k in active_indices:
        c += A[k]
    
    A_eq = np.ones((1, n))
    b_eq = np.ones(1)
    
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=None, b_ub=None, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    
    if res.success:
        g = np.maximum(res.x, 0)
        s = g.sum()
        if s > 1e-10:
            g = g / s
        return g
    return None


def line_search(a, g, n_evals=10):
    """Golden section line search for best t in [0,1]."""
    phi = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    
    for _ in range(n_evals):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)
        
        a1 = np.maximum((1 - t1) * a + t1 * g, 0)
        s1 = a1.sum()
        f1 = compute_obj(a1) if s1 > 1e-10 else np.inf
        
        a2 = np.maximum((1 - t2) * a + t2 * g, 0)
        s2 = a2.sum()
        f2 = compute_obj(a2) if s2 > 1e-10 else np.inf
        
        if f1 < f2:
            hi = t2
        else:
            lo = t1
    
    t = (lo + hi) / 2
    a_best = np.maximum((1 - t) * a + t * g, 0)
    s = a_best.sum()
    if s > 1e-10:
        a_best = a_best / s
    return a_best

def subgradient_descent(a, time_budget=100.0, verbose=False):
    """
    Subgradient descent on the smoothed max-conv objective.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    
    lr = 2.0 / np.sqrt(n)
    beta = 0.8  # momentum
    v = np.zeros(n)
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        
        # Find active set (top peaks)
        threshold = max_val - 0.01 * max_val
        active = np.where(conv >= threshold)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            for j in range(n):
                if 0 <= k - j < n:
                    if j == k - j:  # i.e., 2j = k
                        grad[j] += a[j] / len(active)
                    else:
                        grad[j] += 2 * a[k - j] / len(active)
        
        # Projected gradient step on simplex
        v = beta * v + grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()
        
        obj_new = compute_obj(a_new)
        
        it += 1
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 20 == 0:
                print(f"    SGD it{it}: obj={best_obj:.8f}")
        else:
            stall += 1
            if stall % 10 == 0:
                lr *= 0.9
                a = best_a.copy()
                v = np.zeros(n)
            if stall > 100:
                break
        
        lr = max(1e-6, lr * 0.999)
    
    return best_a


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if new_n == 1:
        return np.array([float(a.sum())])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

The Fuchs method works as follows:
1. Start with initial a
2. Compute autocorrelation r = a*a
3. Find the peak (excluding the main lobe at center)
4. Adjust a to reduce that peak

5. Repeat

1. Compute FFT of a
2. Modify the phase to reduce sidelobes
3. Invert FFT
4. Threshold (set negative values to 0)
5. Repeat

        # The "center" of the full autocorrelation is at index n-1
        # We want to minimize the max excluding the center
        peak_idx = int(np.argmax(np.concatenate([R[:n-1], R[n:]])))
        if peak_idx >= n - 1:
            peak_idx += 1
        
        peak_val = R[peak_idx]

        # Scale the spectrum to reduce sidelobes

        # Reduce a at positions that contribute most to this peak

```python
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    n = len(a)
    if n == 1:
        return np.array([a[0]**2])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a / s)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_fast(a):
    """Build A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(2 * n - 1, j + n)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_partial_lp(a, top_k=50):
    """
    Solve partial LP: min M s.t. conv(a,g)[k] <= M for top K peaks, sum(g)=1, g>=0.
    Much faster than full LP.
    """
    n = len(a)
    a = a / np.sum(a)
    
    # Compute current conv to find top peaks
    conv = compute_conv(a)
    top_indices = np.argsort(conv)[::-1][:top_k]
    
    # Build partial A
    A = build_conv_matrix_fast(a)
    A_partial = A[top_indices, :]
    
    # LP: min M
    # s.t. A_partial @ g <= M (top_k constraints)
    #      sum(g) = 1
    #      g >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A_partial, -np.ones((top_k, 1))])
    b_ub = np.zeros(top_k)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
    
    if res.success:
        g = np.maximum(res.x[:n], 0)
        s = g.sum()
        if s > 1e-10:
            g = g / s
        return g, float(res.x[-1])
    return None, None


def solve_full_lp(a):
    """Solve full LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix_fast(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    
    if res.success:
        g = np.maximum(res.x[:n], 0)
        M = float(res.x[-1])
        s = g.sum()
        if s > 1e-10:
            g = g / s
        return g, M
    return None, None


def golden_line_search(a, g, a_best, obj_best, n_evals=8):
    """Golden section search, keeps track of best."""
    phi = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    
    for _ in range(n_evals):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)
        
        a1 = np.maximum((1 - t1) * a + t1 * g, 0)
        s1 = a1.sum()
        f1 = compute_obj(a1) if s1 > 1e-10 else np.inf
        if f1 < obj_best:
            obj_best = f1
            a_best = a1 / s1
        
        a2 = np.maximum((1 - t2) * a + t2 * g, 0)
        s2 = a2.sum()
        f2 = compute_obj(a2) if s2 > 1e-10 else np.inf
        if f2 < obj_best:
            obj_best = f2
            a_best = a2 / s2
        
        if f1 < f2:
            hi = t2
        else:
            lo = t1
    
    return a_best, obj_best


def fast_two_stage_lp(a, time_budget=300.0, verbose=False, top_k=50):
    """
    Fast two-stage LP using scipy HiGHS with partial constraints.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        # Stage 1: partial LP
        g1, M1 = solve_partial_lp(a, top_k=top_k)
        if g1 is None:
            # Try full LP
            g1, M1 = solve_full_lp(a)
            if g1 is None:
                break
        
        # Find active set from stage 1 solution
        A = build_conv_matrix_fast(a)
        conv_ag = A @ g1
        max_val = float(np.max(conv_ag))
        
        # Active set: where conv_ag is close to max
        tol = 0.01 * max_val
        active = np.where(conv_ag >= max_val - tol)[0]
        
        # Stage 2: minimize sum at active
        if len(active) > 0 and len(active) < 2 * n:
            c2 = np.sum(A[active, :], axis=0)
            res2 = linprog(c2, A_ub=None, b_ub=None, 
                          A_eq=np.ones((1, n)), b_eq=np.ones(1),
                          bounds=[(0, None)] * n, method='highs')
            if res2.success:
                g2 = np.maximum(res2.x, 0)
                s2 = g2.sum()
                if s2 > 1e-10:
                    g2 = g2 / s2
                g_blend = 0.5 * g1 + 0.5 * g2
            else:
                g_blend = g1
        else:
            g_blend = g1
        
        # Line search
        a, best_obj = golden_line_search(a, g_blend, best_a.copy(), best_obj, n_evals=6)
        
        it += 1
        if best_obj < compute_obj(best_a) - 1e-13:
            best_a = a.copy()
            stall = 0
            if verbose and it % 10 == 0:
                elapsed = time.monotonic() - start
                print(f"    it{it}: obj={best_obj:.8f}, M={M1:.6f}, |S|={len(active)}, t={elapsed:.0f}s")
        else:
            stall += 1
            # Perturb to escape
            noise_scale = 0.02 + 0.01 * min(stall, 10)
            a = np.maximum(best_a * np.exp(rng.normal(0, noise_scale, n)), 0)
            s_a = a.sum()
            if s_a > 1e-10:
                a = a / s_a
            else:
                a = best_a.copy()
            if stall > 50:
                break
    
    return best_a
