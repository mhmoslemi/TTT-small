```python
import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses: fast LP with top-k active set, FFT-based smooth gradient with Nesterov acceleration,
simulated annealing polish, multi-n exploration, and spectral initialization.
Key improvements: PDLp solver for speed, top-k active constraints, adaptive step sizes,
and a novel simulated annealing phase to escape local minima.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp_fast(a_norm, active_indices=None, solver=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M (for all k or active_indices), sum(g)=1, g>=0.
    Returns g (normalized to sum=1) or None.
    """
    n = len(a_norm)
    A = build_conv_matrix(a_norm)

    if active_indices is not None:
        A_use = A[active_indices, :]
    else:
        A_use = A

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_use @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if solver is None:
        solvers = [cp.CLARABEL, cp.PDLP, cp.MOSEK, cp.SCS, cp.ECOS]
    else:
        solvers = [solver]

    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    return g_val / s_val
        except Exception:
            continue
    return None


def backtracking_line_search(a, g, best_obj, max_iter=15):
    """Backtracking line search to find best t in [0,1] for a + t*(g-a)."""
    current = compute_obj(a)
    best_a = a.copy()
    best_f = min(current, best_obj)
    t = 1.0
    for _ in range(max_iter):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            t *= 0.5
            continue
        a_try = a_try / s
        f_try = compute_obj(a_try)
        if f_try < best_f:
            best_f = f_try
            best_a = a_try.copy()
            if t < 0.5:
                break
            t = min(1.0, t * 1.2)
        else:
            t *= 0.5
            if t < 0.001:
                break
    return best_a


def smooth_gradient(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of (a*a).
    Returns (value, gradient_array_of_length_n).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    Z = float(np.sum(exp_vals))
    p = exp_vals / Z
    f_val = max_conv + np.log(Z) / tau

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    grad = 2.0 * g_full[:n]
    return f_val, grad


def nesterov_smooth(a, time_budget, tau_init=20.0, tau_end=500.0, verbose=False):
    """
    Nesterov accelerated gradient descent on smoothed objective over simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    momentum = 0.85

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)

        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Gradient
        f_val, grad = smooth_gradient(y, tau)

        # Gradient step
        y_grad = y - lr * grad
        a_new = project_simplex(y_grad)

        # Update momentum
        v = a_new - y
        a = a_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        lr *= 0.9999
        it += 1

        if it % 5000 == 0 and verbose:
            print(f"      NS it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")

        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)
            best_it = it

    return best_a


def lp_polish_topk(a, time_budget, k_active=10, verbose=False):
    """
    LP-based polishing with top-k active constraints.
    Iteratively: solve LP with top-k peaks as active, blend, repeat.
    """
    n = len(a)

        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    stall = 0
    it = 0
    step_scale = 1.0

    while time.monotonic() - start < time_budget:
        # Find top-k active constraints
        conv = compute_conv(a)
        top_k = np.argsort(conv)[::-1][:k_active]

        # Solve LP with these active constraints
        g = solve_lp_fast(a, active_indices=top_k)
        if g is None:
            # Try full LP
            g = solve_lp_fast(a, active_indices=None)
            if g is None:
                break

        # Backtracking line search
        a_new = backtracking_line_search(a, g, best_obj)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            step_scale = min(1.0, step_scale * 1.05)
        else:
            stall += 1
            step_scale *= 0.7
            if stall > 20:
                # Perturb to escape
                noise_scale = 0.01 + 0.005 * min(stall // 10, 5)
                a = best_a * np.exp(rng.normal(0, noise_scale, n))
                a = np.maximum(a, 0)
                s_a = float(np.sum(a))
                if s_a > 1e-10:
                    a = a / s_a
                else:
                    a = best_a.copy()
                stall = 0
                step_scale = 1.0
                if stall > 100:
                    break

        it += 1
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"      LP it{it}: obj={best_obj:.8f}, stall={stall}, time={elapsed:.0f}s")

        if it > 500:
            break

    return best_a


def simulated_annealing(a, time_budget, verbose=False):
    """
    Simulated annealing polish: random coordinate perturbations with SA acceptance.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99999)

    T = 0.01  # initial temperature
    T_min = 1e-6
    cooling = 0.9995

    it = 0
    while time.monotonic() - start < time_budget and T > T_min:
        # Random perturbation: move mass from one coordinate to another
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Amount to move
        delta = rng.uniform(0.001, 0.02) * max(a[i], 1.0 / n)
        delta = min(delta, a[i] * 0.9)

        a_try = a.copy()
        a_try[i] -= delta
        a_try[j] += delta
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            T *= cooling
            continue
        a_try = a_try / s_try

        # Also try scale change (boost one, reduce another proportionally)
        if rng.random() < 0.3:
            ratio = np.exp(rng.normal(0, 0.1))
            a_try2 = a.copy()
            a_try2[i] *= ratio
            a_try2[j] /= ratio
            a_try2 = np.maximum(a_try2, 0)
            s_try2 = float(np.sum(a_try2))
            if s_try2 > 1e-10:
                a_try2 = a_try2 / s_try2
                obj_try2 = compute_obj(a_try2)
                if obj_try2 < best_obj:
                    best_obj = obj_try2
                    best_a = a_try2.copy()
                    a = a_try2
                elif rng.random() < np.exp(-(obj_try2 - best_obj) / T):
                    a = a_try2

        obj_try = compute_obj(a_try)
        delta_obj = obj_try - best_obj

        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / T):
            a = a_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

        T *= cooling
        it += 1

        if verbose and it % 500 == 0:
            print(f"      SA it{it}: obj={best_obj:.8f}, T={T:.6f}")

    return best_a


def spectral_init(n, rng, mode=0):
    """Generate initialization using spectral methods."""
    if mode == 0:
        # Uniform
        return np.ones(n) / n
    elif mode == 1:
        # Cosine taper (reduces edge effects)
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 - np.cos(x))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif mode == 2:
        # Anti-palindromic: high at start, low at end
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * x
        return a / a.sum()
    elif mode == 3:
        # Two-hump anti-symmetric
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.25) ** 2) / 0.05) + 0.5 * np.exp(-((x - 0.75) ** 2) / 0.05)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif mode == 4:
        # Random (smoothed)
        a = rng.uniform(0.5, 1.5, n)
        # Smooth
        k = max(1, n // 50)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif mode == 5:
        # Linear ramp
        x = np.linspace(0.1, 1.0, n)
        a = x ** rng.uniform(0.5, 3.0)
        return a / a.sum()
    elif mode == 6:
        # Gaussian centered
        x = np.linspace(0, 1, n)
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.1, 0.4)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        # Random Fourier
        freqs = rng.uniform(0, 5, 3)
        phases = rng.uniform(0, 2 * np.pi, 3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 1e-10)
        return a / a.sum()


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization:
    1. LP polishing with top-k active set on parent
    2. Nesterov smooth gradient
    3. Multi-n exploration
    4. Simulated annealing
    5. Final combined polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent: {e}")

    if best_seq is None:
        n0 = 800
        best_seq = spectral_init(n0, rng, mode=1)
        best_score = compute_obj(best_seq)
        print(f"Init (n={n0}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: LP polishing with top-k ===
    print(f"\n=== Phase 1: LP Polish (top-k) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 30:
        a1 = lp_polish_topk(best_seq, time_budget=p1_budget, k_active=10, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj1:.8f}")

    # === Phase 2: Nesterov smooth gradient ===
    print(f"\n=== Phase 2: Nesterov Smooth GD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 250.0)
    if p2_budget > 30:
        a2 = nesterov_smooth(best_seq, time_budget=p2_budget, tau_init=20.0, tau_end=400.0, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj2:.8f}")

    # === Phase 3: Second LP pass ===
    print(f"\n=== Phase 3: LP Polish (top-k=20) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 200.0)
    if p3_budget > 20:
        a3 = lp_polish_topk(best_seq, time_budget=p3_budget, k_active=20, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj3:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(200, n_best - 200),
        max(200, n_best - 100),
        n_best + 100,
        n_best + 200,
        min(2000, n_best + 400),
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000 and abs(n - n_best) > 50]

    for n_try in n_options[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.25, 80.0)
        if alloc > 15:
            a_try = lp_polish_topk(a_try, time_budget=alloc, k_active=10)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_try = nesterov_smooth(a_try, time_budget=min(remaining * 0.3, 40.0),
                                         tau_init=30.0, tau_end=300.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = r % 7
        a_r = spectral_init(n_best, rng, mode=mode)
        blend = 0.4
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.25, 60.0)
        if alloc > 15:
            half = alloc / 2
            a_r = lp_polish_topk(a_r, time_budget=half, k_active=10)
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_r = nesterov_smooth(a_r, time_budget=min(remaining * 0.4, 30.0),
                                       tau_init=50.0, tau_end=400.0)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Simulated annealing ===
    print(f"\n=== Phase 6: Simulated Annealing ===")
    remaining = deadline - time.monotonic()
    sa_budget = min(remaining * 0.15, 100.0)
    if sa_budget > 10:
        a_sa = simulated_annealing(best_seq, time_budget=sa_budget, verbose=True)
        obj_sa = compute_obj(a_sa)
        if obj_sa < best_score:
            best_score = obj_sa
            best_seq = a_sa.copy()
            print(f"  SA: obj={best_score:.8f} **")
        else:
            print(f"  SA: obj={obj_sa:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 20:
        alloc = min(remaining * 0.4, 80.0)
        a_f1 = lp_polish_topk(best_seq, time_budget=alloc, k_active=15, verbose=True)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 50.0)
        a_f2 = nesterov_smooth(best_seq, time_budget=alloc, tau_init=100.0, tau_end=800.0)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 40.0)
        a_f3 = lp_polish_topk(best_seq, time_budget=alloc, k_active=30, verbose=True)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Final LP30: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
