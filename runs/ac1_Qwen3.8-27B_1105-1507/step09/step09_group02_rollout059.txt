```python
    M = cp.Variable()
    constraints = [2 * (A @ g) - b <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def solve_lp_reg(a, reg):
    """LP with regularization: min M + reg*sum(g^2) s.t. 2*(a*g)-(a*a) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    b = compute_conv(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [2 * (A @ g) - b <= M, cp.sum(g) == 1]

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:

                g_val = np.maximum(g.value, 0)

        except Exception:

    return None


def line_search(a, direction, best_obj, n_evals=15):
    """
    Line search along direction: find best t in [0,1] for (1-t)*a + t*direction.
    Uses adaptive evaluation.
    """
    a = np.maximum(a, 0)
    direction = np.maximum(direction, 0)
    
    # Sort t candidates by expected usefulness
    t_candidates = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]
    
    best_t = 0.0
    best_t_obj = best_obj
    best_a = a.copy()
    
    for t in t_candidates:
        a_try = (1 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_t_obj:
            best_t_obj = obj_try
            best_t = t
            best_a = a_try.copy()
    
    # Refine around best_t with finer steps
    if best_t > 0:
        t_low = max(0, best_t - 0.1)
        t_high = min(1.0, best_t + 0.1)
        for t in np.linspace(t_low, t_high, 10):
            a_try = (1 - t) * a + t * direction
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a = a_try.copy()
    
    return best_a, best_t_obj, best_t

def smoothed_gradient_descent(a, time_budget, tau_init=10.0, tau_end=500.0):

    n = len(a)

    if s < 1e-10:
        return a

    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)
        
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))

        else:
            y = y / s_y

        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient: g_i = 2 * sum_k p[k] * y[k-i] (from the convolution structure)
        # This is 2 * (p reversed) * y, specifically:
        # g_i = 2 * sum_{k} p[k] * y[k-i] where 0 <= k-i < n
        # = 2 * (p * rev_y)[n-1-i] ... let me think more carefully
        # Q_k = sum_{i+j=k} y[i]*y[j], dQ_k/dy_i = sum_j y[j] * [i+j=k] = sum_j y[k-i] * [j=k-i]
        # = y[k-i] if 0 <= k-i < n

        # This is 2 * (p convolved with y, but only the "valid" part)
        # Actually it's: grad = 2 * ifft(fft(p, 2n) * conj(fft(y, 2n)))[:n]

        A = fft(y, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:

        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)
        
        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it
    
    return best_a


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def peak_shave(a, time_budget):
    """Peak shaving with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.05
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Subgradient from top peaks
        K = min(10, 2 * n - 1)
        top_idx = np.argsort(conv)[::-1][:K]
        
        g = np.zeros(n)
        for k in top_idx:
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += a[j_start:j_end + 1][::-1]
        g /= K
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        a_new = a - lr / g_norm * g
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall % 10 == 0:
                lr *= 0.7
            if lr < 1e-10:
                break
    
    return best_a


def resample(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
```

Wait, I need to double-check the LP v2 formulation. 

The LP v2 is: min M s.t. 2*(A@g)[k] - b[k] <= M for all k, sum(g) = 1, g >= 0.

Where A is the conv matrix of a (so A@g = conv(a,g)) and b = conv(a,a).

This comes from the linearization:
conv(g,g)_k ≈ 2*(a*g)_k - (a*a)_k

Let me verify: if g = a + eps*d, then
conv(g,g)_k = conv(a+d*eps, a+d*eps)_k = (a*a)_k + 2*eps*(a*d)_k + eps^2*(d*d)_k
≈ (a*a)_k + 2*eps*(a*d)_k

And 2*(a*g)_k - (a*a)_k = 2*(a*a)_k + 2*eps*(a*d)_k - (a*a)_k = (a*a)_k + 2*eps*(a*d)_k ✓

Good, the linearization is correct.

Now, one concern: the LP v2 might give a direction that's not very different from a (since we're linearizing around a). If a is already good, the LP solution g* might be very close to a, and the line search won't find much improvement. This is fine - it means we're near a local minimum.

Let me also think about whether there's a better approach. One idea: **alternating LP**. Instead of fixing a and solving for g, we could:
1. Solve LP to get g from a
2. Then solve LP to get a' from g (i.e., solve min M s.t. 2*(g*a')[k] - (g*g)[k] <= M)
3. Alternate

This is like a "ping-pong" or "alternating minimization" approach. Each step is an LP, and we're alternating between the two variables in the bilinear form conv(., .).

Actually, this is exactly what the previous code does (sort of): it sets a_new = (1-t)*a + t*g, which moves a towards g. Then in the next iteration, it computes a new LP direction from the updated a. So it's already doing something like this.

Let me think about whether there's a way to speed up the LP solving. For n=800, the LP has 800 variables and 2*800-1 = 1599 constraints. This is a moderate-sized LP. MOSEK should handle it in milliseconds to a few seconds.

Actually, let me reconsider the time budget. With 1500 seconds and LPs that take maybe 1-5 seconds each, we can do 300-1500 LP iterations. That's quite a lot. The bottleneck might be the line search (each evaluation requires an FFT convolution).

Let me optimize the line search: instead of evaluating the objective at many t values, we can compute conv((1-t)*a + t*g, (1-t)*a + t*g) analytically:
= (1-t)^2 * conv(a,a) + 2*t*(1-t) * conv(a,g) + t^2 * conv(g,g)

We can precompute conv(a,a), conv(a,g), and conv(g,g), then evaluate the max at various t values very cheaply (just max of a vector that's a quadratic in t).

This is a great optimization! Let me implement it.

```python
def line_search_fast(a, g, best_obj):
    """
    Fast line search using precomputed convolutions.
    conv((1-t)a + tg, (1-t)a + tg) = (1-t)^2*conv_aa + 2t(1-t)*conv_ag + t^2*conv_gg
    """
    conv_aa = compute_conv(a)
    conv_ag = compute_conv(a)  # Wait, this is conv(a,a) again, I need conv(a,g)
    
    # Actually I need:
    # conv((1-t)a + tg, (1-t)a + tg) 
    # = (1-t)^2 conv(a,a) + 2t(1-t) conv(a,g) + t^2 conv(g,g)
    
    conv_aa = compute_conv(a)
    conv_gg = compute_conv(g)
    # conv(a,g) = ifft(fft(a) * fft(g))
    N = 1
    while N < 2 * len(a) - 1:
        N <<= 1
    from scipy.fft import rfft, irfft
    fa = rfft(a, N)
    fg = rfft(g, N)
    conv_ag = irfft(fa * fg, N)[:2*len(a)-1]
    
    # Now max over k of [(1-t)^2*conv_aa[k] + 2t(1-t)*conv_ag[k] + t^2*conv_gg[k]]
    # This is max over k of [conv_aa[k] - 2conv_aa[k]*t + conv_aa[k]*t^2 + 2conv_ag[k]*t - 2conv_ag[k]*t^2 + conv_gg[k]*t^2]
    # = max over k of [conv_aa[k] + 2(conv_ag[k]-conv_aa[k])*t + (conv_aa[k]-2conv_ag[k]+conv_gg[k])*t^2]
    # = max over k of [c0[k] + c1[k]*t + c2[k]*t^2]
    
    c0 = conv_aa
    c1 = 2 * (conv_ag - conv_aa)
    c2 = conv_aa - 2 * conv_ag + conv_gg
    
    # For each t, the objective is 2*n*max_k(c0[k] + c1[k]*t + c2[k]*t^2)
    # (since sum((1-t)a+tg) = (1-t)*1 + t*1 = 1 when a and g are both normalized)
    
    n = len(a)
    # Evaluate at many t values
    t_vals = np.linspace(0, 1, 200)
    # For each k, the value at t is c0[k] + c1[k]*t + c2[k]*t^2
    # We want max over k for each t
    # This is: np.max(c0[None,:] + c1[None,:]*t_vals[:,None] + c2[None,:]*t_vals[:,None], axis=1)
    
    # This creates a (200, 2n-1) array which for n=800 is 200*1599 ~ 320K, fine
    vals = c0[np.newaxis, :] + c1[np.newaxis, :] * t_vals[:, np.newaxis] + c2[np.newaxis, :] * t_vals[np.newaxis, :]
    max_vals = np.max(vals, axis=1)
    obj_vals = 2.0 * n * max_vals  # since sum = 1
    
    best_idx = np.argmin(obj_vals)
    best_t = t_vals[best_idx]
    best_obj_t = obj_vals[best_idx]
    
    # Refine around best_t
    t_low = max(0, best_t - 0.05)
    t_high = min(1.0, best_t + 0.05)
    t_fine = np.linspace(t_low, t_high, 100)
    vals_fine = c0[np.newaxis, :] + c1[np.newaxis, :] * t_fine[:, np.newaxis] + c2[np.newaxis, :] * t_fine[np.newaxis, :]
    max_vals_fine = np.max(vals_fine, axis=1)
    obj_fine = 2.0 * n * max_vals_fine
    
    best_fine_idx = np.argmin(obj_fine)
    best_t = t_fine[best_fine_idx]
    best_obj_t = obj_fine[best_fine_idx]
    
    a_new = (1 - best_t) * a + best_t * g
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new > 1e-10:
        a_new = a_new / s_new
    
    return a_new, best_obj_t, best_t
```

```python
"""
Optimized step function search using improved LP linearization (2*(a*g)-(a*a) <= M),
exact fast line search via precomputed convolutions, smoothed gradient descent,
peak shaving, and multi-length exploration.
Key innovation: fast exact line search decomposes conv((1-t)a+tg,(1-t)a+tg) into
quadratic-in-t vector, enabling 300+ line search evaluations per LP solve.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_cross_conv(a, b):
    """Cross convolution a*b via FFT. Returns array of length len(a)+len(b)-1."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_v2(a):
    """LP v2: min M s.t. 2*(a*g)[k] - (a*a)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    b = compute_conv(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [2 * (A @ g) - b <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def solve_lp_v1(a):
    """LP v1: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def solve_lp_reg(a, reg):
    """LP with quadratic regularization for smoothness."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    b = compute_conv(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [2 * (A @ g) - b <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M + reg * cp.sum_squares(g)), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def fast_line_search(a, g, conv_aa):
    """
    Exact fast line search: minimize max_k [(1-t)^2*conv_aa[k] + 2t(1-t)*conv_ag[k] + t^2*conv_gg[k]]
    over t in [0,1]. Returns (a_new, obj_at_best_t, best_t).
    """
    n = len(a)
    conv_gg = compute_conv(g)
    conv_ag = compute_cross_conv(a, g)
    # Ensure same length
    L = 2 * n - 1
    conv_aa = conv_aa[:L]
    conv_gg = conv_gg[:L]
    conv_ag = conv_ag[:L]

    # Quadratic coefficients: val(t) = c0 + c1*t + c2*t^2
    c0 = conv_aa
    c1 = 2.0 * (conv_ag - conv_aa)
    c2 = conv_aa - 2.0 * conv_ag + conv_gg

    # Coarse search
    t_vals = np.linspace(0.0, 1.0, 300)
    vals = c0[np.newaxis, :] + c1[np.newaxis, :] * t_vals[:, np.newaxis] + c2[np.newaxis, :] * t_vals[np.newaxis, :]
    max_vals = np.max(vals, axis=1)
    obj_coarse = 2.0 * n * max_vals

    best_idx = int(np.argmin(obj_coarse))
    best_t = t_vals[best_idx]

    # Fine search around best
    t_low = max(0.0, best_t - 0.03)
    t_high = min(1.0, best_t + 0.03)
    t_fine = np.linspace(t_low, t_high, 200)
    vals_fine = c0[np.newaxis, :] + c1[np.newaxis, :] * t_fine[:, np.newaxis] + c2[np.newaxis, :] * t_fine[np.newaxis, :]
    max_fine = np.max(vals_fine, axis=1)
    obj_fine = 2.0 * n * max_fine

    best_fine_idx = int(np.argmin(obj_fine))
    best_t = t_fine[best_fine_idx]
    best_obj = obj_fine[best_fine_idx]

    a_new = (1.0 - best_t) * a + best_t * g
    a_new = np.maximum(a_new, 0.0)
    s_new = float(np.sum(a_new))
    if s_new > 1e-10:
        a_new = a_new / s_new
    else:
        a_new = a.copy()

    return a_new, float(best_obj), float(best_t)


def lp_polish_fast(a, time_budget, verbose=False, use_v2=True, reg=0.0):
    """LP polish with fast line search. Main optimization loop."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    conv_aa = compute_conv(a)
    start = time.monotonic()
    it = 0
    rng = np.random.default_rng(42)
    stall = 0
    last_t = 0.3

    while time.monotonic() - start < time_budget:
        # Solve LP
        if reg > 0:
            g = solve_lp_reg(a, reg)
        elif use_v2:
            g = solve_lp_v2(a)
        else:
            g = solve_lp_v1(a)

        if g is None:
            break

        # Fast line search
        a_new, obj_new, t_best = fast_line_search(a, g, conv_aa)

        it += 1
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = best_a
            conv_aa = compute_conv(a)
            last_t = t_best
            stall = 0
            if verbose and it % 10 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it{it}: obj={best_obj:.10f}, t={t_best:.4f}, time={elapsed:.0f}s")
        else:
            stall += 1
            if stall == 5:
                a = best_a * np.exp(rng.normal(0, 0.008, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv_aa = compute_conv(a)
            elif stall == 10:
                freq = rng.integers(2, 15)
                phase = rng.uniform(0, 2 * np.pi)
                pert = 1.0 + 0.03 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
                a = best_a * pert
                a = np.maximum(a, 0.0001)
                a = a / a.sum()
                conv_aa = compute_conv(a)
            elif stall == 20:
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv_aa = compute_conv(a)
                use_v2 = not use_v2
            elif stall == 35:
                a = best_a * np.exp(rng.normal(0, 0.08, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv_aa = compute_conv(a)
            elif stall > 50:
                if verbose:
                    print(f"    LP: giving up after {stall} stalls")
                break

    return best_a


def smoothed_grad_fft(a, tau):
    """Compute log-sum-exp smoothed gradient of max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def nesterov_smoothed_descent(a, time_budget, tau_init=10.0, tau_end=500.0):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1.0))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_grad_fft(y, tau)

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)

        if it - best_it > 10000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def peak_shave(a, time_budget):
    """Peak shaving via subgradient on top peaks."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.03
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:10]

        g = np.zeros(n)
        for k in top_idx:
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += a[j_start:j_end + 1][::-1]
        g /= len(top_idx)

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        a_new = a - lr / g_norm * g
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall % 10 == 0:
                lr *= 0.7
            if lr < 1e-10:
                break

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Final GD: obj={best_score:.10f} **")
        else:
            print(f"  Final GD: obj={obj_sgd:.10f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.25, 50.0)
        a_ps = peak_shave(best_seq, time_budget=alloc)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  Final PS: obj={best_score:.10f} **")
        else:
            print(f"  Final PS: obj={obj_ps:.10f}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 40.0)
        a_lp = lp_polish_fast(best_seq, time_budget=alloc, use_v2=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP2: obj={best_score:.10f} **")
        else:
            print(f"  Final LP2: obj={obj_lp:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
```
