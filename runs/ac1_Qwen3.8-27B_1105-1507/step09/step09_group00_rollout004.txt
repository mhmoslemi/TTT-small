def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix_row(a, k):
    """Build the k-th row of the convolution matrix.
    Returns a length-n vector where (M[k,:] @ g) = conv(a,g)[k]."""
    n = len(a)
    row = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n, k + 1)
    for j in range(j_start, j_end):
        row[j] = a[k - j]
    return row

def solve_active_lp(a, active_set, verbose=False):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for k in active_set, sum(g)=1, g>=0."""
    n = len(a)
    K = len(active_set)
    
    # Build constraint matrix
    A = np.zeros((K, n))
    for i, k in enumerate(active_set):
        A[i] = build_conv_matrix_row(a, k)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except:
            continue
    return None, None

def smoothed_gd(a, time_budget=60.0, tau_start=10, tau_end=500, verbose=False):
    """Smoothed gradient descent on the probability simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    beta = 0.9
    it = 0
    
    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress
        lr = 4.0 / np.sqrt(n) * (1.0 - 0.5 * progress)
        
        # Nesterov
        y = a + beta * v
        y = np.maximum(y, 0)
        y = y / y.sum()
        
        # Smoothed gradient
        conv = compute_conv(y)
        max_c = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_c))
        p = exp_vals / exp_vals.sum()
        
        # Gradient: d/da [sum_k p_k * conv(a,a)[k]] = 2 * (p * a) (circular-ish)
        # Actually conv(a,a)[k] = sum_j a[j]*a[k-j]
        # d/da_i conv(a,a)[k] = a[k-i] + a[i-k] (for valid indices)
        # So gradient = 2 * sum_k p_k * (a[k-i] + a[i-k]) 
        #              = 2 * conv(p, a)[i] (with appropriate indexing)
        
        # Use FFT: gradient = 2 * ifft(FFT(p) * conj(FFT(a)))[:n]
        size = 1
        while size < 3*n:
            size <<= 1
        from scipy.fft import fft, ifft
        P = fft(p, size)
        A = fft(a, size)
        grad_full = ifft(P * np.conj(A)).real
        grad = 2.0 * grad_full[:n]
        
        # Update
        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        it += 1
    
    return best_a
