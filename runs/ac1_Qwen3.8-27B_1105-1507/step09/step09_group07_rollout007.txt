import numpy as np

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_gradient_fft(a, beta):
    """
    Compute gradient of log-sum-exp approximation of max(conv(a,a)).
    Uses FFT for efficiency.
    f_beta(a) = (1/beta) * log(sum_k exp(beta * conv[k]))
    grad[i] = sum_k w[k] * d(conv[k])/d(a[i]) = 2 * sum_k w[k] * a[k-i] (with bounds)
    This is 2 * cross_correlation(w, a)[i]
    """
    n = len(a)
    conv = compute_conv(a)
    shifted = beta * (conv - conv.max())
    weights = np.exp(shifted)
    weights /= weights.sum()

    # Compute cross-correlation: grad[i] = 2 * sum_k weights[k] * a[k-i]
    # where a[j] = 0 for j < 0 or j >= n
    size = 1
    while size < 2 * n - 1 + n - 1:
        size <<= 1

    W = fft(weights, size)
    A = fft(a, size)
    # cross-correlation w ⋆ a at lag i = sum_k w[k] * a[k-i]
    # In frequency domain: IFFT(conj(FFT(w)) * FFT(a)) gives cross-correlation
    # Actually: IFFT(FFT(w) * conj(FFT(a))) gives sum_k w[k+i] * conj(a[k]) ... 
    # Let me be more careful.
    # We want c[i] = sum_k w[k] * a[k-i] for i = 0,...,n-1
    # This is the cross-correlation of w and a.
    # corr[i] = sum_k w[k] * a[k-i]
    # In frequency domain: FFT(corr) = conj(FFT(w)) * FFT(a) (with zero padding)
    # Wait, let me verify: if c[m] = sum_n x[n] * conj(y[n-m])
    # then FFT(c) = conj(FFT(x)) * FFT(y)
    # Our case: c[i] = sum_k w[k] * a[k-i]
    # Let m = i, so c[m] = sum_k w[k] * a[k-m]
    # This is like cross-correlation of a and w: (a ⋆ w)[m] = sum_k a[k] * conj(w[k-m])
    # Hmm, let me just use the direct relation.
    # c[i] = sum_k w[k] * a[k-i]
    # = sum_j w[j+i] * a[j] (substituting j = k-i)
    # This is the cross-correlation of a and w at lag i: (a ⋆ w)[i] = sum_j a[j] * conj(w[j+i])
    # For real signals: (a ⋆ w)[i] = sum_j a[j] * w[j+i]
    # FFT relation: FFT(a ⋆ w) = conj(FFT(a)) * FFT(w)
    # So: c = IFFT(conj(FFT(a)) * FFT(w)), then take first n elements.
    
    corr = ifft(conj(A) * W).real[:n]
    grad = 2.0 * corr
    return grad


def smooth_optimize(a, time_budget, beta_start=20.0, beta_end=500.0, verbose=False):
    """
    Smooth optimization using log-sum-exp surrogate with projected gradient.
    Uses FFT for fast gradient computation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.1
    it = 0
    total_iters = max(1, int(time_budget * 2))
    stall = 0

    while time.monotonic() - start < time_budget:
        frac = min(1.0, it / total_iters)
        beta = beta_start * (beta_end / beta_start) ** (frac ** 0.5)

        grad = smooth_gradient_fft(a, beta)
        
        # Projected gradient step with backtracking
        step = lr
        for _ in range(10):
            a_try = a - step * grad
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj + 1e-10:
                break
            step *= 0.5

        obj_new = compute_obj(a_try)
        if obj_new < best_obj - 1e-9:
            best_obj = obj_new
            best_a = a_try.copy()
            a = a_try.copy()
            stall = 0
            lr = min(0.5, lr * 1.01)
        else:
            a = a_try.copy()
            stall += 1
            lr = max(1e-6, lr * 0.8)
            if stall > 50:
                # Add noise to escape
                rng = np.random.default_rng(it * 12345 + n)
                a = a * np.exp(rng.normal(0, 0.02, n))
                a = project_simplex(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                lr = 0.1

        it += 1
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"          Smooth it{it}: obj={best_obj:.8f}, beta={beta:.1f}, "
                  f"lr={lr:.5f}, stall={stall}, elapsed={elapsed:.1f}s")

    return best_a


def build_conv_matrix_sparse(a):
    """Build the convolution constraint matrix A where (A@g)[k] = (a*g)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                rows.append(k)
                cols.append(i)
                vals.append(a[j])
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp(a, objective='max'):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    objective: 'max' (minimize max conv), 'topk' (minimize sum of top-k), 'second' (minimize second max)
    Returns (g_normalized, M_opt).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    A_sparse = build_conv_matrix_sparse(a_norm)
    A = A_sparse.toarray()

    # Try cvxpy first
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver, **{'max_time': 30})
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    # Fallback to linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def solve_lp_topk(a, k=5):
    """
    Solve LP: min t s.t. (a*g)[k] <= M for all k (M fixed at current max),
    sum of top-k conv values <= t, sum(g)=1, g>=0.
    Simplified: just minimize sum of all conv values above a threshold.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_sparse(a_norm).toarray()
    current_conv = compute_conv(a_norm)
    M = float(np.max(current_conv))
    
    # Add penalty for values above 0.8*M
    threshold = 0.7 * M
    penalty = np.zeros(2 * n - 1)
    penalty[current_conv > threshold] = 1.0
    
    try:
        g = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        obj = cp.Minimize(cp.sum(penalty * (A @ g)))
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.CLARABEL, **{'max_time': 20})
        if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
            g_val = np.maximum(g.value, 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def lp_alternate(a, time_budget, verbose=False):
    """
    Improved LP alternating with adaptive step sizes, momentum, and diversity.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = np.zeros(n)
    mu = 0.3
    stall = 0
    patience = 40
    iteration = 0
    last_good_t = 0.3
    use_topk = False

    while time.monotonic() - start < time_budget:
        iteration += 1
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            # Try topk LP every few iterations
            if iteration % 5 == 0:
                g_topk = solve_lp_topk(a, k=5)
                if g_topk is not None:
                    g_lp = g_topk
                    M_lp = M_lp or float(np.max(compute_conv(a)))
                else:
                    break
            else:
                break

        current_max_c = float(np.max(compute_conv(a)))
        similarity = 1.0 - np.linalg.norm(a - g_lp) / (np.linalg.norm(a) + np.linalg.norm(g_lp) + 1e-15)
        
        if similarity > 0.999:
            stall += 3
        elif M_lp < current_max_c * 0.998:
            stall = max(0, stall - 2)
        else:
            stall += 1

        # Generate candidate step sizes
        candidate_ts = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 1.0]
        if 0 < last_good_t < 1:
            candidate_ts.extend([
                max(0.02, last_good_t * 0.5),
                max(0.02, last_good_t * 0.75),
                min(1.0, last_good_t * 1.25),
                min(1.0, last_good_t * 1.5),
            ])
        candidate_ts = sorted(set(t for t in candidate_ts if 0 < t <= 1.0))

        # Use momentum to bias the LP solution
        g_momentum = (1 - mu) * g_lp + mu * momentum
        g_momentum = project_simplex(g_momentum)

        best_t_obj = np.inf
        best_t_val = last_good_t
        best_a_local = best_a.copy()

        for t in candidate_ts:
            for g_use in [g_lp, g_momentum]:
                a_try = (1.0 - t) * a + t * g_use
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_t_obj:
                    best_t_obj = obj_try
                    best_t_val = t
                    best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-10:
            # Update momentum
            momentum = mu * momentum + (1 - mu) * (best_a_local - a)
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local.copy()
            last_good_t = best_t_val
            stall = 0
        else:
            stall += 1
            if stall > patience:
                # Reset with perturbation
                rng = np.random.default_rng(iteration * 7919 + n)
                scale = min(0.5, 0.03 + 0.01 * (stall // patience))
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = project_simplex(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                last_good_t = 0.4
                momentum = np.zeros(n)
                use_topk = not use_topk

        if verbose and iteration % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"        LP-alt it{iteration}: obj={best_obj:.8f}, t={last_good_t:.3f}, "
                  f"stall={stall}, elapsed={elapsed:.1f}s")

    return best_a


def peak_shave(a, time_budget, verbose=False):
    """
    Targeted peak shaving: identify the lag that achieves max convolution,
    find the main contributors, and move mass to reduce that specific peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0

    while time.monotonic() - start < time_budget:
        round_num += 1
        conv = compute_conv(a)
        peak_lag = int(np.argmax(conv))
        peak_val = float(conv[peak_lag])

        # Find contributors to this peak
        # conv[peak_lag] = sum_i a[i]*a[peak_lag - i] for valid i
        contributors = []
        for i in range(n):
            j = peak_lag - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)

        # Find "quiet" positions: positions that contribute least to the peak
        # and are far from the peak's contributor positions
        contrib_to_peak = np.zeros(n)
        for contrib, i, j in contributors:
            contrib_to_peak[i] += contrib
            contrib_to_peak[j] += contrib

        # Quiet positions are those with low contribution and low a value
        quiet_score = -contrib_to_peak - 0.1 * a
        quiet_positions = np.argsort(quiet_score)[-8:]

        improved = False
        for contrib, i_src, j_src in contributors[:8]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-12:
                continue
            src = i_src if a[i_src] >= a[j_src] else j_src
            
            for target in quiet_positions:
                if target == src:
                    continue
                for frac in [0.003, 0.005, 0.01, 0.015, 0.02, 0.03]:
                    delta = frac * a[src]
                    if delta <= 0 or delta >= 0.9 * a[src]:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break
            if improved:
                break

        if not improved:
            # Try swapping: move mass from a high-contributor to a low-contributor
            # at a different peak lag
            conv = compute_conv(a)
            top_lags = np.argsort(conv)[::-1][:3]
            
            for lag in top_lags:
                lag = int(lag)
                for contrib, i_src, j_src in contributors[:5]:
                    if contrib < 1e-12:
                        continue
                    src = i_src if a[i_src] >= a[j_src] else j_src
                    for target in range(n):
                        if target == src:
                            continue
                        # Check if this target would help
                        for frac in [0.005, 0.01, 0.02]:
                            delta = frac * a[src]
                            if delta <= 0 or delta >= 0.9 * a[src]:
                                continue
                            a[src] -= delta
                            a[target] += delta
                            new_conv = compute_conv(a)
                            new_obj = 2.0 * n * float(np.max(new_conv))
                            if new_obj < best_obj - 1e-12:
                                best_obj = new_obj
                                best_a = a.copy()
                                improved = True
                                break
                            else:
                                a[src] += delta
                                a[target] -= delta
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break

            if not improved and round_num > 20:
                break
            elif not improved:
                break

        if verbose and round_num % 10 == 0:
            print(f"          PeakShave r{round_num}: obj={best_obj:.8f}, peak_lag={peak_lag}")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode % 10 == 0:
        a = np.ones(n) / n
    elif mode % 10 == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode % 10 == 2:
        # U-shape (low in middle)
        x = np.linspace(0, 1, n)
        a = (x - 0.5) ** 2 + 0.1
        a = a / a.sum()
    elif mode % 10 == 3:
        # Linear ramp
        x = np.linspace(0, 1, n)
        a = x + 0.1
        a = a / a.sum()
    elif mode % 10 == 4:
        # Two peaks
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.2) ** 2) + np.exp(-20 * (x - 0.8) ** 2)
        a = a / a.sum()
    elif mode % 10 == 5:
        # Cosine modulation
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 6:
        # Random Fourier series
        freqs = rng.integers(2, 10, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 7:
        # Exponential decay from both ends
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * x) + np.exp(-3 * (1 - x))
        a = a / a.sum()
    elif mode % 10 == 8:
        # W-shaped
        x = np.linspace(0, 1, n)
        a = np.exp(-30 * (x - 0) ** 2) + np.exp(-30 * (x - 0.5) ** 2) + np.exp(-30 * (x - 1.0) ** 2)
        a = a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n) * 0.3)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            rng_r = np.random.default_rng(rnd * 104729 + n_current)
            a_start = best_seq * np.exp(rng_r.normal(0, 0.1, n_current))
            a_start = project_simplex(a_start)

        a_lp = lp_alternate(a_start, time_budget=alloc, verbose=(rnd == 0))
        obj_lp = compute_obj(a_lp)
        marker = " **" if obj_lp < best_score else ""
        print(f"  LP round {rnd}: obj={obj_lp:.8f}{marker}")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()

    # === Phase 4: n-sweep (20%) ===
    print(f"\n=== Phase 4: n-Sweep ===")
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.20
    n_options = sorted(set([
        n_current - 20, n_current - 10, n_current,
        n_current + 10, n_current + 20, n_current + 50,
        600, 640, 680, 700, 720, 740, 760, 780, 800,
        820, 840, 860, 880, 900, 920, 960, 1000, 1024, 1080, 1200,
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1300]

    per_n = p4_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_n * 1.5, remaining * 0.1, 40.0)
        if alloc < 10:
            alloc = 10

        a_try = resample_sequence(best_seq, n_try)
        a_opt = lp_alternate(a_try, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)
        marker = " **" if obj_opt < best_score else ""
        print(f"  n={n_try}: obj={obj_opt:.8f}{marker}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try

    # === Phase 5: Peak shaving + more LP (15%) ===
    print(f"\n=== Phase 5: Peak Shaving + LP ===")
    remaining = deadline - time.monotonic()
    p5_budget = remaining * 0.15

    alloc_ps = min(p5_budget * 0.4, 60.0)
    if alloc_ps > 10:
        a_ps = peak_shave(best_seq, time_budget=alloc_ps, verbose=True)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  PeakShave: obj={best_score:.8f} **")
        else:
            print(f"  PeakShave: obj={obj_ps:.8f}")

    remaining = deadline - time.monotonic()
    alloc_lp = min(remaining * 0.3, 80.0)
    if alloc_lp > 20:
        a_lp2 = lp_alternate(best_seq, time_budget=alloc_lp, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp2:.8f}")

    # === Phase 6: Diverse restarts (7%) ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p6_budget = remaining * 0.07
    n_use = len(best_seq)
    n_restarts = 4
    per_restart = p6_budget / max(1, n_restarts)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_restart * 1.5, remaining * 0.15)
        if alloc < 15:
            break

        mode = (r * 3 + 2) % 10
        a_r = diverse_init(n_use, rng, mode=mode)
        # Blend with best
        blend = 0.4 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = project_simplex(a_r)

        # Quick smooth + LP
        half = alloc / 2
        if half > 10:
            a_r = smooth_optimize(a_r, time_budget=half,
                                  beta_start=20, beta_end=200, verbose=False)
        remaining = deadline - time.monotonic()
        lp_alloc = min(remaining * 0.1, 30.0)
        if lp_alloc > 10:
            a_r = lp_alternate(a_r, time_budget=lp_alloc, verbose=False)

        obj_r = compute_obj(a_r)
        marker = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}{marker}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()

    # === Phase 7: Final polish (5%) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Smooth optimization with high beta
    alloc = min(remaining * 0.3, 40.0)
    if alloc > 10:
        a_fin = smooth_optimize(best_seq, time_budget=alloc,
                                beta_start=50.0, beta_end=800.0, verbose=False)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Final Smooth: obj={best_score:.8f} **")

    # LP polish
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.4, 40.0)
    if alloc > 15:
        a_fin = lp_alternate(best_seq, time_budget=alloc, verbose=False)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # One more peak shave
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_fin = peak_shave(best_seq, time_budget=min(remaining * 0.4, 20.0), verbose=False)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Final PeakShave: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"=== Target: 1.5027, Achieved: {best_score:.8f}, Gap: {best_score - 1.5027:.8f} ===")

    return result
