"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent (multiplicative weights), LP polishing with adaptive targets,
frequency-domain structured perturbation, and multi-length exploration.
Key innovations: entropic mirror descent for better simplex navigation,
peak-target LP for faster convergence, and spectral diversity for restarts.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad_mirror(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT.
    Returns (gradient, max_conv, conv).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, max_conv, conv


def mirror_descent(a, time_budget=300.0, tau_init=20.0, tau_end=800.0,
                   verbose=False, seed=0):
    """
    Mirror descent (multiplicative weights) with Nesterov acceleration.
    Uses entropic mirror for simplex optimization.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    s = float(np.sum(a))
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))

        # Exponential tau annealing
        tau = tau_init * (tau_end / tau_init) ** progress

        # Periodic reset
        if it > 0 and it % 4000 == 0:
            tau = tau_init * 3.0
            lr = 2.0 / np.sqrt(n)

        if it % 5000 == 0 and verbose:
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        g, max_conv, conv = smoothed_grad_mirror(y, tau)

        # Barzilai-Borwein style adaptive step
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr_new = 0.3 * lr + 0.7 * min(max(bb_lr, 1e-10), 10.0 / np.sqrt(n))
                lr = lr_new

        # Mirror descent update: multiplicative
        a_new = y * np.exp(-lr * g)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        v = beta * v - lr * g
        # Blend with Nesterov
        a_new = 0.7 * a_new + 0.3 * np.maximum(y + v, 1e-15)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.99995)

        # Reset if stuck
        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)
            best_it = it

    return best_a


def build_conv_matrix(a):
    """Build matrix A_mat (2n-1 x n) such that A_mat @ g = convolve(a, g)."""
    n = len(a)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A_mat[j:j + r, j] = a[:r]
    return A_mat


def solve_lp_minpeak(a, M_target=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    If M_target is given, solve: max sum(g_extra) s.t. conv(a,g)[k] <= M_target.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A_mat = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [cp.sum(g) == 1]

    if M_target is not None:
        constraints.append(A_mat @ g <= M_target)
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)  # This is always 1, not useful
        # Instead: minimize distance to g from a
        # Or: just minimize M with a tighter initial guess
        pass

    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

    a_norm = a / np.sum(a)
    A_mat = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    constraints = [A_mat @ g <= M_target, cp.sum(g) == 1]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)  # trivial, always 1

    # Better: minimize ||g - a||_1 subject to constraints
    prob = cp.Problem(cp.Minimize(cp.norm1(g - a_norm)), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def lp_polish(a, time_budget=200.0, verbose=False, seed=0):
    """
    LP polishing with adaptive M targets and peak-aware perturbation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    last_t = 0.3
    consecutive_fails = 0
    current_max = float(np.max(compute_conv(a)))

    while time.monotonic() - start < time_budget:
        # Solve LP to find the best direction
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Try multiple blending factors
        candidates = [0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        candidates.append(max(0.02, last_t * 0.5))
        candidates.append(min(1.0, last_t * 1.5))
        candidates = sorted(set([max(0.01, min(1.0, c)) for c in candidates]))

        best_t_obj = best_obj
        best_t_val = last_t
        best_a_local = best_a.copy()

        for t in candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        iter_count += 1

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a.copy()
            last_t = best_t_val
            consecutive_fails = 0
            current_max = float(np.max(compute_conv(a)))

            if verbose and iter_count % 3 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t_val:.2f}, M_lp={M_lp:.6f}")
        else:
            consecutive_fails += 1
            # Perturb and try again
            conv = compute_conv(a)
            # Peak-aware perturbation
            peak_idx = int(np.argmax(conv))
            contrib = np.zeros(n)
            i_start = max(0, peak_idx - (n - 1))
            i_end = min(n - 1, peak_idx)
            for i in range(i_start, i_end + 1):
                j = peak_idx - i
                if 0 <= j < n:
                    contrib[i] += a[i] * a[j]

            max_c = contrib.max() if contrib.max() > 0 else 1.0
            scale = 0.02 + 0.003 * min(consecutive_fails, 15)
            noise = rng.normal(0, 1, n) * scale * (0.5 + contrib / max_c)
            a = a * np.exp(noise)
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            a = 0.8 * a + 0.2 * best_a
            a = a / a.sum()

            if consecutive_fails > 30:
                # Try target LP
                target_M = current_max * 0.95
                g_target = solve_lp_target(a, target_M)
                if g_target is not None:
                    a_try = 0.5 * a + 0.5 * g_target
                    a_try = np.maximum(a_try, 0)
                    a_try = a_try / a_try.sum()
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = best_a.copy()
                        consecutive_fails = 0

            if consecutive_fails > 50:
                break

    return best_a


def spectral_perturbation(a, scale=0.1, seed=0):
    """
    Perturb in frequency domain to create diverse candidates.
    """
    n = len(a)
    rng = np.random.default_rng(seed)
    N = 1
    while N < 2 * n:
        N <<= 1

    A = fft(a, N)
    # Perturb the magnitude
    mag = np.abs(A)
    phase = np.angle(A)
    noise = rng.normal(0, scale, N)
    mag_new = mag * np.exp(noise)
    A_new = mag_new * np.exp(1j * phase)
    a_new = ifft(A_new).real[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def freq_flatness_perturbation(a, seed=0):
    """
    Perturb to flatten the frequency response (reduce peak in autocorrelation).
    """
    n = len(a)
    rng = np.random.default_rng(seed)
    N = 1
    while N < 2 * n:
        N <<= 1

    A = fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)

    # Compute target: geometric mean of magnitudes
    log_mag = np.log(mag + 1e-15)
    target_log = np.mean(log_mag)

    # Blend towards flat spectrum
    blend = 0.1 + 0.05 * rng.random()
    log_mag_new = (1 - blend) * log_mag + blend * target_log
    mag_new = np.exp(log_mag_new)

    A_new = mag_new * np.exp(1j * phase)
    a_new = ifft(A_new).real[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def gd_lp_alternating(a, time_budget=300.0, verbose=False, seed=0):
    """
    Alternate between mirror descent and LP polishing.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    a = a / s

    best_a = a.copy()

    start = time.monotonic()

    phase_idx = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break

        if phase_idx % 3 == 0:
            # Mirror descent phase
            alloc = min(remaining * 0.3, 60.0)

                a_md = mirror_descent(a, time_budget=alloc,
                                      tau_init=15.0, tau_end=600.0,

                obj_md = compute_obj(a_md)
                if obj_md < best_obj:
                    best_obj = obj_md
                    best_a = a_md.copy()

        elif phase_idx % 3 == 1:
            # LP phase
            alloc = min(remaining * 0.3, 60.0)

                obj_lp = compute_obj(a_lp)
                if obj_lp < best_obj:
                    best_obj = obj_lp
                    best_a = a_lp.copy()
                    a = a_lp.copy()
                else:
                    # Spectral perturbation
                    a = spectral_perturbation(a, scale=0.05, seed=seed + phase_idx)
                    a = 0.7 * a + 0.3 * best_a
                    a = a / a.sum()

        else:
            # Frequency flattening phase
            a = freq_flatness_perturbation(a, seed=seed + phase_idx)
            alloc = min(remaining * 0.2, 30.0)

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / (np.sqrt(1 - x**2) + 0.1)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** rng.uniform(0.5, 2.0)
        a = a / a.sum()
    elif mode == 5:
        freqs = rng.integers(1, 8, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Lacunary series (gives interesting autocorrelation structure)
        a = np.zeros(n)
        for k in range(5):
            freq = 2 ** k
            a += np.abs(np.sin(2 * np.pi * freq * np.arange(n) / n))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 7:
        # Gaussian mixture
        ncomp = rng.integers(2, 5)
        a = np.zeros(n)
        for _ in range(ncomp):
            c = rng.integers(0, n)
            w = max(1, int(n * rng.uniform(0.02, 0.1)))
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 8:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1, 1)
        a = a / a.sum()
    elif mode == 9:
        # Chebyshev-like
        x = np.linspace(-1, 1, n)
        a = 1 / (np.sqrt(np.maximum(1 - x**2, 1e-6)) * 0.5 + 0.5)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

Uses entropic mirror descent (multiplicative weights), LP polishing with adaptive
targets, frequency-domain structured perturbation, pairwise coordinate refinement,

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad_fft(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT.
    Returns (gradient, max_conv, conv).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, max_conv, conv


def exact_subgrad(a):
    """
    Compute exact subgradient of max_k conv(a,a)[k] at the active lag.
    Returns (gradient, max_conv, active_lag).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    # Find all active lags (within tolerance)
    tol = 1e-6 * max_conv
    active = np.where(conv >= max_conv - tol)[0]
    g = np.zeros(n)
    for k in active:
        k = int(k)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                g[i] += a[j] / len(active)
            if 2 * i == k:
                g[i] += a[i] / len(active)
    return g, max_conv, int(active[0])

def build_conv_matrix(a):
    """Build matrix A_mat (2n-1 x n) such that A_mat @ g = linear_convolve(a, g)."""
    n = len(a)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A_mat[j:j + r, j] = a[:r]
    return A_mat


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A_mat = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_target(a, M_target):
    """
    Solve LP: minimize ||g - a||_1 s.t. conv(a_norm, g)[k] <= M_target, sum(g)=1, g>=0.
    Returns g or None if infeasible.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A_mat = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    constraints = [A_mat @ g <= M_target, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(cp.norm1(g - a_norm)), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None

    A_new = mag_new * np.exp(1j * phase)

    s = float(np.sum(a_new))
    if s > 1e-10:

    return a_new

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)

            print(f"  Phase1 MD: obj={obj_md:.8f}")

    # === Phase 2: Main GD-LP Alternating ===
    print(f"\n=== Phase 2: GD-LP Alternating (main workhorse) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 480.0)
    if p2_budget > 30:
        a_alt = gd_lp_alternating(best_seq, time_budget=p2_budget,
                                   verbose=True, seed=seed)
        obj_alt = compute_obj(a_alt)
        if obj_alt < best_score:
            best_score = obj_alt
            best_seq = a_alt.copy()
            print(f"  Phase2 GD-LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 GD-LP: obj={obj_alt:.8f}")

    # === Phase 3: Aggressive LP Polish ===
    print(f"\n=== Phase 3: Aggressive LP Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.18, 220.0)
    if p3_budget > 20:
        a_lp = lp_polish(best_seq, time_budget=p3_budget,
                          verbose=True, seed=seed + 100)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase3 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 LP: obj={obj_lp:.8f}")
            # Try from perturbed start
            a_pert = spectral_perturbation(best_seq, scale=0.1, seed=seed + 200)
            a_pert = 0.5 * a_pert + 0.5 * best_seq
            a_pert = a_pert / a_pert.sum()
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_lp2 = lp_polish(a_pert, time_budget=min(remaining * 0.15, 60),
                                   verbose=False, seed=seed + 300)
                obj_lp2 = compute_obj(a_lp2)
                if obj_lp2 < best_score:
                    best_score = obj_lp2
                    best_seq = a_lp2.copy()
                    print(f"  Phase3 LP (pert): obj={best_score:.8f} **")

    # === Phase 4: Pairwise Refinement ===
    print(f"\n=== Phase 4: Pairwise Refinement ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.08, 80.0)
    if p4_budget > 10:
        a_pw = pairwise_refine(best_seq, time_budget=p4_budget, seed=seed + 400)
        obj_pw = compute_obj(a_pw)
        if obj_pw < best_score:
            best_score = obj_pw
            best_seq = a_pw.copy()
            print(f"  Phase4 Pairwise: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 Pairwise: obj={obj_pw:.8f}")

    # === Phase 5: Multi-length Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = [n_current - 100, n_current + 100, n_current + 200,
                 n_current + 300, 600, 1000, 1200, 1500]
    n_options = sorted(set([n for n in n_options if 400 <= n <= 2500 and abs(n - n_current) >= 80]))
    n_options = n_options[:3]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        alloc = min(remaining * 0.25, 100.0)
        if alloc > 20:
            a_try = gd_lp_alternating(a_try, time_budget=alloc,
                                       verbose=False, seed=seed + n_try)
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_try = lp_polish(a_try, time_budget=min(remaining * 0.2, 40),
                               verbose=False, seed=seed + n_try + 1)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = (r * 3 + 2) % 10
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.1 + 0.06 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 1e-10)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.18, 80.0)
        if alloc > 15:
            a_r = gd_lp_alternating(a_r, time_budget=alloc,
                                     verbose=False, seed=seed + 1000 + r)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau mirror descent
    if remaining > 20:
        alloc = min(remaining * 0.25, 50.0)
        a_md = mirror_descent(best_seq, time_budget=alloc,
                              tau_init=50.0, tau_end=2000.0,
                              verbose=True, seed=seed + 2000, use_exact_final=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Final MD (high tau): obj={best_score:.8f} **")
        else:
            print(f"  Final MD (high tau): obj={obj_md:.8f}")

    # LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.25, 50.0)
        a_lp = lp_polish(best_seq, time_budget=alloc,
                          verbose=True, seed=seed + 3000)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_lp:.8f}")

    # One more GD-LP cycle
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 40.0)
        a_final = gd_lp_alternating(best_seq, time_budget=alloc,
                                     verbose=False, seed=seed + 4000)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final GD-LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
