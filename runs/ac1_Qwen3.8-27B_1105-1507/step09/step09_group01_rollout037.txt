def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_peak(a, M_target=None, L_target=None):
    """
    Solve LP with optional peak and floor constraints.
    If L_target is given: min (M - L) s.t. L <= conv(a,g)[k] <= M, sum(g)=1, g>=0
    Otherwise: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    num_rows = 2 * n - 1

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [cp.sum(g) == 1]

    if L_target is not None:
        L = cp.Variable()
        constraints.append(A @ g <= M)
        constraints.append(A @ g >= L)
        if M_target is not None:
            constraints.append(M <= M_target)
        prob = cp.Problem(cp.Minimize(M - L), constraints)
    else:
        constraints.append(A @ g <= M)
        if M_target is not None:
            constraints.append(M <= M_target)
        prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
            elif prob.status == 'infeasible':
                return None
        except Exception:
            continue
    return None


def spectral_shaping(a, time_budget=100.0, alpha=0.1, verbose=False):
    """
    Novel spectral shaping method: work in frequency domain to flatten
    the convolution. Iteratively reduce peak lags by modifying the
    magnitude spectrum.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 3 * n:
        N <<= 1

    it = 0
    stall = 0
    current_alpha = alpha

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        mean_conv = float(np.mean(conv))
        peak_idx = int(np.argmax(conv))
        peak_val = float(conv[peak_idx])

        if peak_val <= mean_conv * (1 + 1e-10):
            break

        F = fft(a, N)
        magnitude = np.abs(F)
        phase = np.angle(F)

        # Compute the contribution of each frequency to the peak lag
        # b[k] = sum_f |F(f)|^2 * exp(2*pi*i*f*k/N) / N
        # To reduce b[peak_idx], reduce |F(f)| where the phase aligns
        freqs = np.arange(N)
        alignment = np.cos(2.0 * np.pi * freqs * peak_idx / N)

        # Only reduce where alignment is positive (constructive interference at peak)
        reduction = np.where(alignment > 0,
                            current_alpha * alignment,
                            0.0)

        new_magnitude = magnitude * (1.0 - reduction)
        new_magnitude = np.maximum(new_magnitude, 1e-10)

        F_new = new_magnitude * np.exp(1j * phase)
        a_candidate = np.real(ifft(F_new, N))[:n]

        # Project to non-negative, normalize
        a_candidate = np.maximum(a_candidate, 0.0)
        s_cand = float(np.sum(a_candidate))
        if s_cand < 1e-10:
            a_candidate = best_a.copy()
            current_alpha *= 0.5
            stall += 1
        else:
            a_candidate = a_candidate / s_cand
            obj_cand = compute_obj(a_candidate)

            if obj_cand < best_obj - 1e-14:
                best_obj = obj_cand
                best_a = a_candidate.copy()
                a = a_candidate
                stall = 0
                current_alpha = min(current_alpha * 1.05, 0.5)
            else:
                stall += 1
                current_alpha *= 0.7

        if stall > 50:
            break

        it += 1
        if verbose and it % 50 == 0:
            print(f"    Spectral it{it}: obj={best_obj:.8f}, alpha={current_alpha:.4f}")

    return best_a


def spectral_shaping_v2(a, time_budget=100.0, verbose=False):
    """
    Improved spectral shaping: use log-magnitude modification and
    multiple peak lags simultaneously.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 3 * n:
        N <<= 1
    freqs = np.arange(N)

    it = 0
    stall = 0
    tau = 5.0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Weighted peaks using softmax
        weights = np.exp(tau * (conv - max_conv) / max_conv)
        total_w = float(np.sum(weights))
        if total_w < 1e-10:
            break
        weights = weights / total_w

        # Top K peaks
        K = min(5, 2 * n - 1)
        top_idx = np.argsort(conv)[::-1][:K]
        top_w = weights[top_idx]
        top_w = top_w / float(np.sum(top_w))

        F = fft(a, N)
        log_mag = np.log(np.maximum(np.abs(F), 1e-15))
        phase = np.angle(F)

        # For each peak lag, reduce log-magnitude where constructive
        delta = np.zeros(N)
        for w, k in zip(top_w, top_idx):
            alignment = np.cos(2.0 * np.pi * freqs * k / N)
            delta += w * alignment
        delta = np.maximum(delta, 0)

        step = 0.05
        new_log_mag = log_mag - step * delta
        F_new = np.exp(new_log_mag) * np.exp(1j * phase)
        a_candidate = np.real(ifft(F_new, N))[:n]

        a_candidate = np.maximum(a_candidate, 0.0)
        s_cand = float(np.sum(a_candidate))
        if s_cand < 1e-10:
            a_candidate = best_a.copy()
            step *= 0.5
            stall += 1
        else:
            a_candidate = a_candidate / s_cand
            obj_cand = compute_obj(a_candidate)

            if obj_cand < best_obj - 1e-14:
                best_obj = obj_cand
                best_a = a_candidate.copy()
                a = a_candidate
                stall = 0
                step = min(step * 1.1, 0.2)
                tau = min(tau * 1.05, 50.0)
            else:
                stall += 1
                step *= 0.6

        if stall > 80:
            break

        it += 1
        if verbose and it % 100 == 0:
            print(f"    SpecV2 it{it}: obj={best_obj:.8f}, step={step:.4f}, tau={tau:.1f}")

    return best_a


def weighted_subgradient(a, time_budget=100.0, K=3, verbose=False):
    """
    Subgradient method with adaptive K and Barzilai-Borwein step size.
    Uses weighted combination of top-K peak subgradients.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 3.0 / np.sqrt(n)
    momentum = 0.6
    v = np.zeros(n)
    stall = 0
    it = 0
    prev_a = a.copy()
    prev_obj = best_obj
    prev_lr = lr

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        if max_conv < 1e-15:
            break

        # Adaptive K: start small, grow as we improve
        effective_K = min(K + it // 200, 10)
        top_idx = np.argsort(conv)[::-1][:effective_K]
        peak_vals = conv[top_idx]

        # Softmax weights on top peaks
        scale = 5.0
        w = np.exp(scale * (peak_vals - peak_vals[0]) / max(peak_vals[0], 1e-15))
        w = w / float(np.sum(w))

        # Compute subgradient: gradient of b[k] w.r.t. a is 2*(a reversed and shifted)
        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k = int(k)
            # b[k] = sum_i a[i]*a[k-i], grad_a = 2 * a[k-i] (reversed)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += wi * 2.0 * a[j_start:j_end + 1][::-1]

        # Barzilai-Borwein step size
        if it > 0 and prev_lr > 1e-12:
            s_vec = a - prev_a
            y_vec = g - 0  # We don't have prev_g easily, use simple BB
            ss = float(np.dot(s_vec, s_vec))
            if ss > 1e-15:
                # Use objective-based BB
                obj_diff = compute_obj(a) - prev_obj
                if abs(obj_diff) > 1e-15:
                    bb_lr = ss / (2.0 * abs(obj_diff) * max_conv + 1e-10)
                    lr = 0.5 * lr + 0.5 * np.clip(bb_lr, 1e-8, 10.0 / np.sqrt(n))

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 100:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break
            else:
                lr *= 0.99

        prev_a = a.copy()
        prev_obj = obj_new
        prev_lr = lr
        it += 1

        if verbose and it % 200 == 0:
            print(f"    Subgrad it{it}: obj={best_obj:.8f}, lr={lr:.6f}")

    return best_a


def lp_polish_improved(a, time_budget=200.0, verbose=False, seed=42):
    """
    Improved LP polishing with:
    1. Standard peak-minimizing LP
    2. Range-minimizing LP (minimize M - L)
    3. Quadratic step size extrapolation
    4. Periodic constraint tightening
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    no_improve = 0
    last_t = 0.5
    t_history = []

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start

        # Choose LP type based on iteration
        use_flattening = (iter_count % 3 == 1)

        if use_flattening:
            # Range-minimizing LP
            conv_current = compute_conv(a)
            L_target = float(np.max(conv_current)) * 0.7
            g_lp = solve_lp_peak(a, L_target=L_target)
        else:
            # Peak-minimizing LP
            M_current = float(np.max(compute_conv(a)))
            M_target = M_current * 0.95
            g_lp = solve_lp_peak(a, M_target=M_target)

        if g_lp is None:
            if use_flattening:
                g_lp = solve_lp_peak(a)
            if g_lp is None:
                break

        # Step size: use history for quadratic extrapolation
        if len(t_history) >= 2:
            # Try to predict good t from recent history
            t_candidates = [0.3, 0.5, 0.7, last_t, last_t * 0.5, last_t * 1.5]
            t_candidates = [max(0.02, min(1.0, t)) for t in t_candidates]
            t_candidates = list(set([round(t, 3) for t in t_candidates]))
        else:
            t_candidates = [0.3, 0.5, 0.7]

        # Evaluate several step sizes (cheaper than golden section for small number)
        best_t = 0.5
        best_t_obj = compute_obj(a)

        for t_try in t_candidates:
            a_try = (1.0 - t_try) * best_a + t_try * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t_try

        a_try = (1.0 - best_t) * best_a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        a_try = a_try / float(np.sum(a_try))
        obj_try = compute_obj(a_try)

        iter_count += 1

        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            last_t = best_t
            t_history.append(best_t)
            if len(t_history) > 5:
                t_history.pop(0)
            no_improve = 0
            if verbose and iter_count % 5 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t:.3f}, flat={use_flattening}, el={elapsed:.0f}s")
        else:
            no_improve += 1
            if no_improve == 2:
                # Try smaller step
                t_small = max(0.02, last_t * 0.3)
                a_try2 = (1.0 - t_small) * best_a + t_small * g_lp
                a_try2 = np.maximum(a_try2, 0)
                a_try2 = a_try2 / float(np.sum(a_try2))
                obj_try2 = compute_obj(a_try2)
                if obj_try2 < best_obj - 1e-14:
                    best_obj = obj_try2
                    best_a = a_try2.copy()
                    no_improve = 0
                    if verbose:
                        print(f"    LP small it{iter_count}: obj={best_obj:.8f}")
            elif no_improve >= 4:
                # Perturb
                scale = 0.005 * min(no_improve, 10)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / float(np.sum(a_pert))
                no_improve = 0
                if iter_count > 40:
                    break

        # Constraint tightening every 6 iterations
        if iter_count % 6 == 3:
            conv_cur = compute_conv(best_a)
            M_cur = float(np.max(conv_cur))
            for frac in [0.90, 0.93, 0.96]:
                g_tight = solve_lp_peak(best_a, M_target=M_cur * frac)
                if g_tight is not None:
                    for t_frac in [0.3, 0.5, 0.7]:
                        a_try3 = (1.0 - t_frac) * best_a + t_frac * g_tight
                        a_try3 = np.maximum(a_try3, 0)
                        s3 = float(np.sum(a_try3))
                        if s3 < 1e-10:
                            continue
                        a_try3 = a_try3 / s3
                        obj_try3 = compute_obj(a_try3)
                        if obj_try3 < best_obj - 1e-14:
                            best_obj = obj_try3
                            best_a = a_try3.copy()
                            no_improve = 0
                            if verbose:
                                print(f"    LP-tight it{iter_count}: obj={best_obj:.8f}, frac={frac}, t={t_frac}")
                            break

    return best_a


def nesterov_smoothed_descent(a, time_budget=200.0, tau_init=10.0, tau_end=400.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        if it % 5000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Smoothed gradient via FFT
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(y, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)

        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Bell-shaped
        x = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (x - 0.5) ** 2)
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-8))
        a = a / a.sum()
    elif mode == 4:
        # Multi-frequency
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1, 1)
        a = a / a.sum()
    elif mode == 6:
        # Two bumps
        c1, c2 = 0.3, 0.7
        w = 0.1
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - c1) / w) ** 2) + np.exp(-((x - c2) / w) ** 2)
        a = a / a.sum()
    elif mode == 7:
        # Cosine taper
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 8:
        # Sine bump
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** rng.uniform(0.5, 3.0)
        a = a / a.sum()
    elif mode == 9:
        # Random smooth
        n_ctrl = 5
        pts = sorted(rng.uniform(0, 1, n_ctrl).tolist())
        vals = rng.uniform(0.1, 1.0, n_ctrl)
        x = np.linspace(0, 1, n)
        a = np.interp(x, pts, vals)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):

    Main search: combines spectral shaping (novel), improved LP with
    flattening, adaptive subgradient, multi-n exploration, and
    perturbative restarts to minimize the step function objective.
    """

    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 0: Try diverse initializations at current n ===
    print(f"\n=== Phase 0: Diverse Init Search ===")
    remaining = deadline - time.monotonic()
    p0_budget = min(remaining * 0.05, 60.0)
    if p0_budget > 10:
        for mode in range(10):
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            a_try = diverse_init(n_target, rng, mode)
            # Quick polish
            quick_budget = min(10.0, remaining * 0.3)
            if quick_budget > 5:
                a_try = lp_polish_improved(a_try, time_budget=quick_budget, seed=mode)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Init mode {mode}: obj={best_score:.8f} **")
            else:
                print(f"  Init mode {mode}: obj={obj_try:.8f}")

    # === Phase 1: Spectral Shaping (NEW) ===
    print(f"\n=== Phase 1: Spectral Shaping ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.1, 100.0)
    if p1_budget > 20:
        a_spec = spectral_shaping_v2(best_seq, time_budget=p1_budget, verbose=True)
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Spectral V2: obj={best_score:.8f} **")
        else:
            print(f"  Spectral V2: obj={obj_spec:.8f}")

        # Also try v1
        remaining = deadline - time.monotonic()
        p1b_budget = min(remaining * 0.05, 40.0)
        if p1b_budget > 15:
            a_spec1 = spectral_shaping(best_seq, time_budget=p1b_budget)
            obj_spec1 = compute_obj(a_spec1)
            if obj_spec1 < best_score:
                best_score = obj_spec1
                best_seq = a_spec1.copy()
                print(f"  Spectral V1: obj={best_score:.8f} **")

    # === Phase 2: Intensive LP Polishing ===
    print(f"\n=== Phase 2: Intensive LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.30, 350.0)
    if p2_budget > 30:
        a_lp = lp_polish_improved(best_seq, time_budget=p2_budget, verbose=True, seed=42)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f}")

    # === Phase 3: Subgradient refinement ===
    print(f"\n=== Phase 3: Subgradient Refine ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.08, 80.0)
    if p3_budget > 15:
        a_sg = weighted_subgradient(best_seq, time_budget=p3_budget, K=5, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj_sg:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 300), max(400, n_current - 200),
        max(400, n_current - 100),
        n_current + 100, n_current + 200, n_current + 300, n_current + 500,
        600, 800, 1000, 1200, 1500, 1800, 2000
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2500 and abs(n - n_current) > 50]

    for n_try in n_options[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)

        # Quick spectral + LP
        alloc1 = min(remaining * 0.1, 30.0)
        if alloc1 > 10:
            a_try = spectral_shaping_v2(a_try, time_budget=alloc1)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 80.0)
        if alloc2 > 20:
            a_try = lp_polish_improved(a_try, time_budget=alloc2, seed=n_try % 1000)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Perturbation + Re-optimization ===
    print(f"\n=== Phase 5: Perturbation + Re-optimization ===")
    remaining = deadline - time.monotonic()

    for r in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        scales = [0.003, 0.008, 0.015, 0.025, 0.04, 0.06, 0.08, 0.12, 0.15, 0.2]
        scale = scales[r % len(scales)]
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / float(np.sum(a_pert))

        # Choose method based on perturbation size
        alloc = min(remaining * 0.2, 80.0)
        if alloc > 15:
            if scale < 0.05:
                a_pert = lp_polish_improved(a_pert, time_budget=alloc, seed=200 + r)
            else:
                a_pert = spectral_shaping_v2(a_pert, time_budget=alloc * 0.5)
                remaining = deadline - time.monotonic()
                alloc2 = min(remaining * 0.1, 40.0)
                if alloc2 > 15:
                    a_pert = lp_polish_improved(a_pert, time_budget=alloc2, seed=300 + r)

            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Perturb {r} (scale={scale:.3f}): obj={best_score:.8f} **")
            else:
                print(f"  Perturb {r} (scale={scale:.3f}): obj={obj_pert:.8f}")

    # === Phase 6: Final intensive LP ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 150.0)
        a_final = lp_polish_improved(best_seq, time_budget=alloc, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # Quick final subgradient
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.25, 40.0)
        a_sg = weighted_subgradient(best_seq, time_budget=alloc, K=8)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # Quick final spectral
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.3, 30.0)
        a_sp = spectral_shaping_v2(best_seq, time_budget=alloc)
        obj_sp = compute_obj(a_sp)
        if obj_sp < best_score:
            best_score = obj_sp
            best_seq = a_sp.copy()
            print(f"  Final Spec: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
