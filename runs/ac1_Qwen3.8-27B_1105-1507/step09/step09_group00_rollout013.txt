"""

Uses top-k focused log-sum-exp gradient descent with FFT acceleration,

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def topk_smoothed_gradient(a, tau, k=50):
    """
    Compute gradient of log-sum-exp of top-k values of conv(a,a).
    Returns (gradient, conv).
    """
    n = len(a)
    conv = compute_conv(a)
    
    if k >= len(conv):
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp
    else:
        # Focus on top-k lags
        topk_idx = np.argpartition(conv, -k)[-k:]
        conv_topk = conv[topk_idx]
        max_topk = float(np.max(conv_topk))
        exp_topk = np.exp(tau * (conv_topk - max_topk))
        sum_exp = float(np.sum(exp_topk))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[topk_idx[np.argmax(conv_topk)]] = 1.0
        else:
            p_topk = exp_topk / sum_exp
            p = np.zeros(len(conv))
            p[topk_idx] = p_topk
    
    # Gradient via FFT: g[i] = 2 * sum_k p[k] * conv(a, e_i)[k]
    # conv(a, e_i)[k] = a[k-i] (for valid k-i)
    # So g[i] = 2 * sum_k p[k] * a[k-i]
    # This is a cross-correlation of p and a
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    
    return g, conv


def anneal_tau(elapsed, total, tau_init, tau_end, warmup=0.05):
    """Cosine annealing for tau."""
    if total <= 0:
        return tau_end
    frac = min(1.0, elapsed / total)
    if frac < warmup:
        return tau_init
    cos_val = 0.5 * (1 + np.cos(np.pi * (frac - warmup) / (1 - warmup)))
    return tau_init + (tau_end - tau_init) * (1 - cos_val)

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        # Build constraint matrix A such that A @ g = conv(a, g)
        # conv(a,g)[k] = sum_j a[k-j] * g[j] for valid (k-j, j)
        # A is (2n-1) x n

            # k-j ranges from 0 to n-1, so k ranges from j to j+n-1

        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]

                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        g_val = g_val / s_g
                        solved = True
                        break
            except Exception:
                continue
        
        if not solved:
            break

        best_blend_obj = best_obj
        best_blend = best_a.copy()

            a_try = np.maximum(a_try, 0)

            a = np.maximum(a, 0)

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, np.maximum(a, 0))
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def multi_peak_init(n, n_peaks, rng):
    """Initialize with n_peaks Gaussian bumps."""
    a = np.zeros(n)
    centers = rng.integers(0, n, size=n_peaks)
    widths = rng.uniform(n * 0.01, n * 0.05, size=n_peaks)
    heights = rng.uniform(0.5, 1.5, size=n_peaks)
    x = np.arange(n)
    for c, w, h in zip(centers, widths, heights):
        a += h * np.exp(-((x - c) ** 2) / (2 * w ** 2))
    a = np.maximum(a, 0.001)
    return a / a.sum()


def frequency_init(n, rng, n_modes=10):
    """Initialize using random Fourier modes (guaranteed non-negative via exp)."""
    x = np.arange(n)
    log_a = np.zeros(n)
    for k in range(1, n_modes + 1):
        amp = rng.uniform(0.1, 0.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        log_a += amp * np.cos(2 * np.pi * k * x / n + phase)
    a = np.exp(log_a - log_a.max())
    return a / a.sum()


def support_prune(a, keep_frac=0.8):
    """Zero out the smallest elements (prune support)."""
    n = len(a)
    thresh_idx = int(n * (1 - keep_frac))
    if thresh_idx <= 0:
        return a
    thresh = np.partition(a, thresh_idx)[thresh_idx]
    a_new = np.where(a >= thresh, a, 0).astype(np.float64)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a
    return a_new / s

The objective is `2n * max(conv) / (sum a)^2`. If we normalize `sum a = 1`, it's `2n * max(conv)`.

The average value of conv is `1/(2n-1)`. So we want `max(conv)` to be as close to `1/(2n-1)` as possible.

The ratio `max(conv) / avg(conv) = max(conv) * (2n-1)` is the "peak factor" or "crest factor". We want to minimize this.

For a uniform sequence, `max(conv) = 1/n` and `avg = 1/(2n-1)`, so peak factor = `(2n-1)/n ≈ 2`.
