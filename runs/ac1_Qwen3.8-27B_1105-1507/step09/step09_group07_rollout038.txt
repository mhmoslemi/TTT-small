import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a):
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None

The Repeated Median Heuristic (RMH) works as follows:
1. Compute the autocorrelation `c = a*a`
2. Find the peak lag `k*`
3. Create a new sequence by taking the median of `a` and its permutations that would reduce the peak

For our non-negative case, a simplified version would be:
1. Compute `c = a*a`, find peak lag `k*`
2. For each position `j`, compute `a[k*-j]` (the "conflicting" value)
3. Sort positions by `a[k*-j]` (descending)
4. Move mass from high-conflict positions to low-conflict positions

This is essentially what the gradient method does, but the median heuristic is more aggressive (it reorders rather than just perturbs).

A more practical version:
1. Compute `c = a*a`, find peak lag `k*`
2. Sort indices by `a[k*-j]` descending
3. Take the top `p%` and bottom `p%` indices
4. Swap their values (or average them)
5. Repeat

This is a "permutation-based" heuristic that directly reduces the peak by swapping the most conflicting values.

"""
Optimized search for minimum peak autocorrelation of non-negative step functions.
Exploits the special gradient structure (grad c[k]/da[j] = 2*a[k-j]) for fast
soft-max gradient descent, combined with LP alternating for polishing,
adaptive n-sweep, and diverse restarts.
Key improvements: FFT-based gradient computation, symmetric LP, aggressive
exploration with multi-scale restarts.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def softmax_gradient_step(a, beta, lr, momentum_v=None, momentum=0.85):
    """
    One soft-max gradient step exploiting grad c[k]/da[j] = 2*a[k-j].
    Returns (a_new, momentum_v_new, obj_improved).
    """
    n = len(a)
    conv = compute_conv(a)

    # Soft-max weights
    shifted = beta * (conv - conv.max())
    shifted = np.clip(shifted, -50, 0)
    weights = np.exp(shifted)
    weights /= weights.sum()

    # Gradient: grad[j] = 2 * sum_k weights[k] * a[k-j]
    # Compute via FFT: cross[j+n-1] where cross = conv(weights, a_flip)
    a_flip = a[::-1]
    size = 1
    while size < 3 * n:
        size <<= 1
    W = rfft(weights, size)
    C = rfft(a_flip, size)
    cross = irfft(W * C, size)
    grad = 2.0 * cross[n - 1:n - 1 + n]

    # Project to sum=0
    grad -= np.mean(grad)

    # Momentum
    if momentum_v is None:
        momentum_v = np.zeros(n)
    momentum_v = momentum * momentum_v - lr * grad

    a_new = a + momentum_v
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        return a, momentum_v, False
    a_new = a_new / s_new
    return a_new, momentum_v, True


def gradient_descent_soft(a, time_budget, beta_start=50.0, beta_end=10000.0,
                          lr_init=None, verbose=False):
    """
    Gradient descent using soft-max surrogate with momentum.
    Exploits the special gradient structure for efficiency.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if lr_init is None:
        lr = 2.0 / n
    else:
        lr = lr_init
    momentum = 0.85
    momentum_v = None
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        beta = beta_start * (beta_end / beta_start) ** frac

        a_new, momentum_v, ok = softmax_gradient_step(a, beta, lr, momentum_v, momentum)
        if not ok:
            lr *= 0.9
            continue

        obj_new = compute_obj(a_new)
        it += 1

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
            lr = min(5.0 / n, lr * 1.002)
            # Decay momentum slightly as we converge
            momentum = max(0.7, 0.85 - 0.1 * frac)
        else:
            no_improve += 1
            lr = max(1e-6, lr * 0.97)
            if no_improve > 500:
                # Restart with perturbation
                rng = np.random.default_rng(it + n)
                scale = 0.05 + 0.05 * (no_improve // 1000)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                momentum_v = None
                no_improve = 0
                lr = 2.0 / n
                momentum = 0.85

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"          GradSoft it{it}: obj={best_obj:.8f}, beta={beta:.0f}, "
                  f"lr={lr:.6f}, mom={momentum:.3f}, elapsed={elapsed:.1f}s")

    return best_a


def subgradient_top_lags(a, time_budget, num_active=5, verbose=False):
    """
    Subgradient method using top active lags.
    Directly computes gradient from the most active lags.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.0 / n
    momentum_v = np.zeros(n)
    momentum = 0.8
    no_improve = 0
    step = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = conv.max()

        # Find active lags
        sorted_idx = np.argsort(conv)[::-1][:num_active]
        active_lags = sorted_idx

        # Compute gradient: 2 * sum_k a[k-j] for active k
        a_flip = a[::-1]
        size = 1
        while size < 3 * n:
            size <<= 1

        grad = np.zeros(n)
        for k in active_lags:
            k = int(k)
            # a[k-j] for j=0..n-1: valid when 0 <= k-j < n
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                m_start = k - j_end
                m_end = k - j_start
                grad[j_start:j_end + 1] += a[m_start:m_end + 1]

        grad *= 2.0
        grad -= np.mean(grad)

        momentum_v = momentum * momentum_v - lr * grad
        a_new = a + momentum_v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        step += 1

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
            lr = min(3.0 / n, lr * 1.005)
        else:
            no_improve += 1
            lr = max(1e-7, lr * 0.98)
            if no_improve > 300:
                rng = np.random.default_rng(step + n * 7)
                a = best_a * np.exp(rng.normal(0, 0.08, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                momentum_v = np.zeros(n)
                no_improve = 0
                lr = 1.5 / n

        if verbose and step % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"          SubGrad it{step}: obj={best_obj:.8f}, lr={lr:.6f}, "
                  f"active={len(active_lags)}, elapsed={elapsed:.1f}s")

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    try:
        from scipy.interpolate import PchipInterpolator
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, new_n)
        interp = PchipInterpolator(x_old, a, extrapolate=False)
        a_new = interp(x_new)
    except Exception:
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, new_n)
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode % 10 == 0:
        a = np.ones(n) / n
    elif mode % 10 == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode % 10 == 2:
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 3:
        a = np.exp(-3 * (x - 0.25) ** 2) + np.exp(-3 * (x - 0.75) ** 2) + 0.1
        a = a / a.sum()
    elif mode % 10 == 4:
        a = x * (1 - x) * 4 + 0.1
        a = a / a.sum()
    elif mode % 10 == 5:
        freqs = rng.integers(2, 10, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 6:
        a = 1.0 + 0.4 * np.sin(2 * np.pi * 3 * x + 0.5) + 0.2 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 7:
        a = rng.dirichlet(np.ones(n) * 0.3)
    elif mode % 10 == 8:
        # Anti-palindromic bias
        a = 1.0 + 0.3 * (2 * x - 1)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    else:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 5 * x) * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    return a


def swap_heuristic(a, time_budget, verbose=False):
    """
    Swap-based heuristic: identify the peak lag, then swap values at
    positions that contribute most to the peak with positions that
    contribute least.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0

    while time.monotonic() - start < time_budget:
        round_num += 1
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Compute contribution of each position to the peak
        # a[j] * a[k_star - j]
        contrib = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        if j_start <= j_end:
            m_start = k_star - j_end
            m_end = k_star - j_start
            contrib[j_start:j_end + 1] = a[j_start:j_end + 1] * a[m_start:m_end + 1]

        # Sort by contribution
        sorted_idx = np.argsort(contrib)[::-1]
        # Top contributors (to reduce) and bottom (to increase)
        n_swap = max(3, n // 20)
        reduce_positions = sorted_idx[:n_swap]
        increase_positions = sorted_idx[-n_swap:]

        improved = False
        for i in reduce_positions:
            for j in increase_positions:
                if i == j:
                    continue
                # Try swap
                old_i, old_j = a[i], a[j]
                a[i], a[j] = a[j], a[i]
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                else:
                    a[i], a[j] = old_i, old_j

        if not improved and round_num > 5:
            break

        if verbose and round_num % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"          Swap it{round_num}: obj={best_obj:.8f}, k*={k_star}, elapsed={elapsed:.1f}s")

    return best_a

        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"\nInitial: n={n_current}, obj={best_score:.8f}, target=1.5027")

    # === Phase 1: Fast gradient descent (25%) ===
    print(f"\n=== Phase 1: Fast Gradient Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 350.0)
    if p1_budget > 30:
        a_p1 = gradient_descent_soft(best_seq, time_budget=p1_budget,
                                     beta_start=20, beta_end=5000, verbose=True)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Phase 1 (grad): obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 (grad): obj={obj_p1:.8f}")

    # Also try subgradient
    remaining = deadline - time.monotonic()
    p1b_budget = min(remaining * 0.05, 60.0)
    if p1b_budget > 15:
        a_p1b = subgradient_top_lags(best_seq, time_budget=p1b_budget,
                                     num_active=5, verbose=False)
        obj_p1b = compute_obj(a_p1b)
        if obj_p1b < best_score:
            best_score = obj_p1b
            best_seq = a_p1b.copy()
            print(f"  Phase 1b (subgrad): obj={best_score:.8f} **")
        else:
            print(f"  Phase 1b (subgrad): obj={obj_p1b:.8f}")

    # === Phase 2: LP Alternating (25%) ===
    print(f"\n=== Phase 2: LP Alternating ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 350.0)
    if p2_budget > 30:
        a_p2 = lp_alternate_improved(best_seq, time_budget=p2_budget, verbose=True)
        obj_p2 = compute_obj(a_p2)
        if obj_p2 < best_score:
            best_score = obj_p2
            best_seq = a_p2.copy()
            print(f"  Phase 2 (LP): obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 (LP): obj={obj_p2:.8f}")

    # === Phase 3: n-sweep (15%) ===
    print(f"\n=== Phase 3: n-Sweep ===")
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.15

    n_options = sorted(set([
        n_current - 50, n_current - 30, n_current - 20, n_current - 10,
        n_current,
        n_current + 10, n_current + 20, n_current + 30, n_current + 50,
        700, 720, 750, 768, 780, 800, 820, 840, 860, 880, 900, 920, 960, 1000,
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1200]

    per_n = p3_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_n, remaining * 0.12)
        if alloc < 10:
            alloc = 10

        a_try = resample_sequence(best_seq, n_try)

        # Quick gradient + LP
        a_opt = gradient_descent_soft(a_try, time_budget=alloc * 0.6,
                                      beta_start=50, beta_end=3000, verbose=False)
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_opt = lp_alternate_improved(a_opt, time_budget=min(alloc * 0.4, remaining * 0.3),
                                          verbose=False)

        obj_opt = compute_obj(a_opt)
        marker = " **" if obj_opt < best_score else ""
        print(f"  n={n_try}: obj={obj_opt:.8f}{marker}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try

    # === Phase 4: Intensive gradient + LP cycles (20%) ===
    print(f"\n=== Phase 4: Intensive Cycles (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.20
    n_cycles = 4
    per_cycle = p4_budget / n_cycles

    for cyc in range(n_cycles):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(per_cycle, remaining * 0.25)
        if alloc < 20:
            break

        # Start with perturbation for later cycles
        if cyc == 0:
            a_start = best_seq.copy()
        else:
            rng_c = np.random.default_rng(cyc * 49297 + n_current)
            scale = 0.04 + 0.03 * cyc
            a_start = best_seq * np.exp(rng_c.normal(0, scale, n_current))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()

        # Gradient phase (60% of cycle)
        a_grad = gradient_descent_soft(a_start, time_budget=alloc * 0.5,
                                       beta_start=30, beta_end=8000, verbose=False)
        # LP phase (40% of cycle)
        remaining = deadline - time.monotonic()
        lp_alloc = min(alloc * 0.5, remaining * 0.3)
        if lp_alloc > 10:
            a_lp = lp_alternate_improved(a_grad, time_budget=lp_alloc, verbose=False)
        else:
            a_lp = a_grad

        obj_cyc = compute_obj(a_lp)
        marker = " **" if obj_cyc < best_score else ""
        print(f"  Cycle {cyc}: obj={obj_cyc:.8f}{marker}")
        if obj_cyc < best_score:
            best_score = obj_cyc
            best_seq = a_lp.copy()

    # === Phase 5: Swap heuristic + diverse restarts (10%) ===
    print(f"\n=== Phase 5: Swap + Restarts ===")
    remaining = deadline - time.monotonic()
    p5_budget = remaining * 0.10

    # Swap heuristic
    alloc_s = min(p5_budget * 0.4, 40.0)
    if alloc_s > 10:
        a_swap = swap_heuristic(best_seq, time_budget=alloc_s, verbose=False)
        obj_swap = compute_obj(a_swap)
        marker = " **" if obj_swap < best_score else ""
        print(f"  Swap: obj={obj_swap:.8f}{marker}")
        if obj_swap < best_score:
            best_score = obj_swap
            best_seq = a_swap.copy()

    # Diverse restarts
    remaining = deadline - time.monotonic()
    n_restarts = 4
    per_restart = (remaining * 0.08) / max(1, n_restarts)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_restart, remaining * 0.2)
        if alloc < 15:
            break

        mode = (r * 7 + 2) % 10
        a_r = diverse_init(n_current, rng, mode=mode)
        # Blend with best
        blend = 0.3 + 0.15 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        # Quick gradient
        a_r = gradient_descent_soft(a_r, time_budget=alloc * 0.6,
                                    beta_start=30, beta_end=5000, verbose=False)
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_r = lp_alternate_improved(a_r, time_budget=min(alloc * 0.4, remaining * 0.2),
                                        verbose=False)

        obj_r = compute_obj(a_r)
        marker = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}{marker}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()

    # === Phase 6: Final polish (10%) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Long gradient run
    alloc_g = min(remaining * 0.4, 100.0)
    if alloc_g > 20:
        a_final = gradient_descent_soft(best_seq, time_budget=alloc_g,
                                        beta_start=100, beta_end=20000, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final grad: obj={best_score:.8f} **")

    # LP polish
    remaining = deadline - time.monotonic()
    alloc_lp = min(remaining * 0.4, 100.0)
    if alloc_lp > 15:
        a_final = lp_alternate_improved(best_seq, time_budget=alloc_lp, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # One more gradient
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = gradient_descent_soft(best_seq, time_budget=min(remaining * 0.6, 40.0),
                                        beta_start=200, beta_end=50000, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Last grad: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"=== Target: 1.5027, Achieved: {best_score:.8f}, Gap: {best_score - 1.5027:.8f} ===")

    return result
