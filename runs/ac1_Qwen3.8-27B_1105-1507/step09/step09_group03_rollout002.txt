import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = max(0, j - (n - 1))
        hi = min(2 * n - 1, j + n)
        A[lo:hi, j] = a[j - lo:j - lo + (hi - lo)]
    return A


def solve_lp_minpeak(a, active_indices=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Or minimize sum of active conv terms if active_indices given.
    Returns (g_normalized, objective_value).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)

    if active_indices is not None:
        active_A = A[active_indices, :]
        obj = cp.sum(active_A @ g)
        constraints = [cp.sum(g) == 1]
    else:
        M = cp.Variable()
        obj = M
        constraints = [A @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(obj), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    if active_indices is None:
                        return g_val, float(M.value)
                    else:
                        return g_val, float(obj.value)
        except Exception:
            continue
    return None, None


def golden_section_line_search(a, g, n_evals=8):
    """Find optimal t in [0,1] minimizing compute_obj((1-t)*a + t*g)."""
    phi = (np.sqrt(5) - 1) / 2
    lo = 0.0
    hi = 1.0

    for _ in range(n_evals):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)
        a1 = np.maximum((1 - t1) * a + t1 * g, 0)
        s1 = float(np.sum(a1))
        f1 = compute_obj(a1) if s1 > 1e-10 else np.inf
        a2 = np.maximum((1 - t2) * a + t2 * g, 0)
        s2 = float(np.sum(a2))
        f2 = compute_obj(a2) if s2 > 1e-10 else np.inf
        if f1 < f2:
            hi = t2
        else:
            lo = t1

    t_best = (lo + hi) / 2
    a_best = np.maximum((1 - t_best) * a + t_best * g, 0)
    s_best = float(np.sum(a_best))
    if s_best > 1e-10:
        a_best = a_best / s_best
    return a_best

    Solve min M s.t. conv(a,g) <= M, get g, blend with golden section.
    Use active set refinement for later iterations.
    """

    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    it = 0
    stall = 0
    use_active = False
    active_indices = None

    while time.monotonic() - start < time_budget:
        if use_active and active_indices is not None:
            g, obj_val = solve_lp_minpeak(a, active_indices=active_indices)

        if g is None:
            break

        # Find active constraints
        conv_ag = build_conv_matrix(a) @ g
        max_conv_ag = float(np.max(conv_ag))
        tol = 0.02 * max_conv_ag
        active = np.where(conv_ag >= max_conv_ag - tol)[0]
        active_indices = active

        # Blend with golden section
        a_new = golden_section_line_search(a, g, n_evals=6)
        obj_new = compute_obj(a_new)
        it += 1

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if it > 5:
                use_active = True
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, M={obj_val:.6f}, |S|={len(active)}")
        else:
            stall += 1
            use_active = False
            # Perturb
            scale = 0.01 + 0.005 * min(stall, 10)
            a = best_a * np.exp(rng.normal(0, min(scale, 0.08), n))
            a = np.maximum(a, 0)
            s_a = float(np.sum(a))
            if s_a > 1e-10:
                a = a / s_a
            else:
                a = best_a.copy()
            if stall > 60:
                break

    return best_a


def peak_averaged_subgradient(a, K=5):

    Compute averaged subgradient of max_k Q_k(a) over top-K peaks.

    conv = compute_conv(a)

        weights[k] = 1.0 / K

    grad = np.zeros(n)
    for k in top_k:
        w = 1.0 / K
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            j = k - i
            if 0 <= j < n:
                # Q_k = sum_i a_i * a_{k-i}, dQ_k/da_i = a_{k-i} + a_i * (1 if 2i==k else 0)
                # Actually dQ_k/da_i = a_{k-i} (from the term a_i*a_{k-i})
                # Plus from the term a_{k-i}*a_i (same when i and k-i swapped)

                # dQ_k/da_i = a_{k-i} (for the pair (i, k-i)) 
                # But also if k-i != i, the pair (k-i, i) contributes a_i as well
                # So dQ_k/da_i = a_{k-i} + a_{k-i} = 2*a_{k-i} if i != k-i
                # And dQ_k/da_i = a_i if i == k-i (i.e., 2i = k)
                if 2 * i == k:
                    grad[i] += w * a[i]
                else:
                    grad[i] += w * a[k - i]

    return 2.0 * grad, conv, top_k


def nesterov_peak_subgrad(a, time_budget=200.0, K=5, verbose=False):
    """
    Nesterov-accelerated subgradient descent using top-K peak averaging.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        if it % 500 == 0 and verbose:
            elapsed = time.monotonic() - start
            print(f"    PSG it{it}: obj={best_obj:.8f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        grad, conv, _ = peak_averaged_subgradient(y, K=K)

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-6, lr * 0.9997)

        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)
            best_it = it

    return best_a


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def nesterov_smoothed_descent(a, time_budget=200.0, tau_init=20.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 4.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        if it % 2000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)

        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.3 * lr + 0.7 * min(max(bb_lr, 1e-8), 20.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-7, lr * 0.9999)

        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)
            best_it = it

    return best_a


def aggressive_cd(a, time_budget=150.0, verbose=False):
    """
    Aggressive coordinate descent: for each top peak, try all source-target pairs
    within the contributing range, with optimal line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(4242)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Get top 3 peaks
        top_peaks = np.argsort(conv)[::-1][:3]

        improved = False
        for k_peak in top_peaks:
            i_start = max(0, k_peak - (n - 1))
            i_end = min(n - 1, k_peak)

            if i_start > i_end:
                continue

            # Get contributions
            contribs = []
            for i in range(i_start, i_end + 1):
                j = k_peak - i
                if 0 <= j < n:
                    contribs.append((a[i] * a[j], i, j))

            contribs.sort(reverse=True)

            # Try top contributors
            for contrib_val, src, dst in contribs[:5]:
                if contrib_val < 1e-12:
                    continue

                # Try moving some mass from src (or dst) to random targets
                for source in [src, dst]:
                    if source == dst and source == src:
                        continue
                    if a[source] < 1e-10:
                        continue

                    # Try several targets: far from peak, random, reflected
                    targets = set()
                    targets.add(int(rng.integers(0, n)))
                    targets.add(int(rng.integers(0, n)))
                    targets.add(n - 1 - source)
                    targets.add(max(0, source - n // 3))
                    targets.add(min(n - 1, source + n // 3))
                    targets.add(int(k_peak - source + n - 1) % n)  # reflected

                    for tgt in targets:
                        tgt = int(tgt) % n
                        if tgt == source:
                            continue

                        # Try several delta fractions
                        for delta_frac in [0.1, 0.05, 0.02, 0.01]:
                            delta = delta_frac * a[source]
                            if delta <= 0 or delta > a[source]:
                                continue

                            a_try = a.copy()
                            a_try[source] -= delta
                            a_try[tgt] += delta
                            s_try = float(np.sum(a_try))
                            if s_try < 1e-10:
                                continue
                            a_try = a_try / s_try
                            obj_try = compute_obj(a_try)

                            if obj_try < best_obj - 1e-15:
                                best_obj = obj_try
                                best_a = a_try.copy()
                                a = a_try
                                improved = True
                                break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break

        if verbose and it % 50 == 0:
            print(f"    CD it{it}: obj={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if new_n < 2:
        return a[:max(1, new_n)].copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a, kind='cubic')
    return np.maximum(a_new, 0).astype(np.float64)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Comb pattern
        p = max(2, n // 40)
        a = np.zeros(n)
        for i in range(0, n, p):
            a[i] = 1.0
        a = a / a.sum()
    elif mode == 3:
        # Triangular
        x = np.linspace(0, 1, n)
        a = np.maximum(0, 1 - np.abs(2 * x - 1))
        a = a / a.sum()
    elif mode == 4:
        # Two Gaussian bumps
        x = np.arange(n)
        a = np.exp(-((x - n // 4) ** 2) / (2 * (n // 10) ** 2)) + \
            np.exp(-((x - 3 * n // 4) ** 2) / (2 * (n // 10) ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 6:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    else:
        # Sine-modulated
        x = np.linspace(0, 1, n)
        f = rng.integers(2, 8)
        a = np.maximum(0.05, 1.0 + 0.5 * np.sin(2 * np.pi * f * x))
        a = a / a.sum()
    return a


def accelerated_lp_momentum(a, time_budget=150.0, momentum=0.4, verbose=False):
    """Accelerated LP with heavy-ball momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        v = momentum * v + (g_lp - a)
        a_new = project_simplex(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            found = False
            for scale in [0.5, 0.25, 0.1]:
                a_try = project_simplex(a + scale * v)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    v = scale * v
                    stall = 0
                    found = True
                    break
            if not found:
                stall += 1
                v *= 0.5
                a = best_a.copy()
                if stall > 30:
                    break

        it += 1
        if verbose and it % 10 == 0:
            print(f"    ALP it{it}: obj={best_obj:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization combining LP, subgradient, smoothed GD, and coordinate descent.
    Explores multiple sequence lengths and uses diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent available: {e}")

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: LP-based optimization ===
    print(f"\n=== Phase 1: LP Polish ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 30:
        a_lp = lp_polish(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 2: Peak-averaged subgradient ===
    print(f"\n=== Phase 2: Peak-Averaged Subgradient ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 250.0)
    if p2_budget > 30:
        a_psg = nesterov_peak_subgrad(best_seq, time_budget=p2_budget, K=3, verbose=True)
        obj_psg = compute_obj(a_psg)
        if obj_psg < best_score:
            best_score = obj_psg
            best_seq = a_psg.copy()
            print(f"  PSG(K=3): obj={best_score:.8f} **")
        else:
            print(f"  PSG(K=3): obj={obj_psg:.8f}")

        # Also try K=5
        remaining = deadline - time.monotonic()
        if remaining > 60:
            a_psg2 = nesterov_peak_subgrad(best_seq, time_budget=min(remaining * 0.3, 120.0), K=5, verbose=True)
            obj_psg2 = compute_obj(a_psg2)
            if obj_psg2 < best_score:
                best_score = obj_psg2
                best_seq = a_psg2.copy()
                print(f"  PSG(K=5): obj={best_score:.8f} **")
            else:
                print(f"  PSG(K=5): obj={obj_psg2:.8f}")

    # === Phase 3: Smoothed GD ===
    print(f"\n=== Phase 3: Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 200.0)
    if p3_budget > 30:
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=p3_budget,
                                          tau_init=30.0, tau_end=600.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  GD: obj={best_score:.8f} **")
        else:
            print(f"  GD: obj={obj_sgd:.8f}")

    # === Phase 4: Aggressive Coordinate Descent ===
    print(f"\n=== Phase 4: Aggressive CD ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 150.0)
    if p4_budget > 20:
        a_cd = aggressive_cd(best_seq, time_budget=p4_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        n_current - 200, n_current - 100, n_current + 100, n_current + 200, n_current + 400,
        400, 600, 1000, 1200, 1500, 2000
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2500 and abs(n - n_current) > 80]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()

        a_try = resample_sequence(best_seq, n_try)

        # Quick LP polish
        alloc = min(remaining * 0.15, 50.0)
        if alloc > 15:
            a_try = lp_polish(a_try, time_budget=alloc)

        # Quick CD
        remaining = deadline - time.monotonic()
        if remaining > 30:
            cd_alloc = min(remaining * 0.1, 30.0)
            if cd_alloc > 10:
                a_try = aggressive_cd(a_try, time_budget=cd_alloc)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(4):

        if remaining < 80:
            break

        mode = (r + 1) % 7
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.4
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        # LP + CD
        alloc = min(remaining * 0.2, 70.0)
        if alloc > 20:
            half = alloc / 2
            a_r = lp_polish(a_r, time_budget=half)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_r = aggressive_cd(a_r, time_budget=min(remaining * 0.2, 30.0))

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP
    if remaining > 40:
        alloc = min(remaining * 0.3, 80.0)
        a_final = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # PSG with K=1 (single peak focus)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 60.0)
        a_psg = nesterov_peak_subgrad(best_seq, time_budget=alloc, K=1, verbose=True)
        obj_psg = compute_obj(a_psg)
        if obj_psg < best_score:
            best_score = obj_psg
            best_seq = a_psg.copy()
            print(f"  Final PSG(K=1): obj={best_score:.8f} **")

    # CD
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.25, 50.0)
        a_cd = aggressive_cd(best_seq, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 40.0)
        a_lp = lp_polish(best_seq, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
