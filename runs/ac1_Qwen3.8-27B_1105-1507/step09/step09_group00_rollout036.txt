import numpy as np

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n - j, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def smoothed_gradient_fft(a, tau):

    A = fft(a, N)

        if s2 > 1e-10:
            a2 = a2 / s2
            obj2 = compute_obj(a2)
        else:
            obj2 = np.inf

        # Also try t = last_step as a candidate
        t_trial = min(max(last_step, t_lo), t_hi)
        if abs(t_trial - t1) > 0.01 and abs(t_trial - t2) > 0.01:
            a_trial = (1 - t_trial) * a + t_trial * g_lp
            a_trial = np.maximum(a_trial, 0)
            s_trial = float(np.sum(a_trial))
            if s_trial > 1e-10:
                a_trial = a_trial / s_trial
                obj_trial = compute_obj(a_trial)
            else:
                obj_trial = np.inf
            # Include in comparison
            if obj_trial < min(obj1, obj2):
                obj1, obj2, t1, t2 = obj_trial, min(obj1, obj2), t_trial, (t1 if obj1 < obj2 else t2)

        # Narrow interval (up to 3 iterations of golden section)
        for _ in range(3):
            if obj1 < obj2:
                t_hi = t2
                t2 = t1
                obj2 = obj1
                t1 = t_hi - gr * (t_hi - t_lo)
                a1 = (1 - t1) * a + t1 * g_lp
                a1 = np.maximum(a1, 0)
                s1 = float(np.sum(a1))
                obj1 = compute_obj(a1 / s1) if s1 > 1e-10 else np.inf
            else:
                t_lo = t1
                t1 = t2
                obj1 = obj2
                t2 = t_lo + gr * (t_hi - t_lo)
                a2 = (1 - t2) * a + t2 * g_lp
                a2 = np.maximum(a2, 0)
                s2 = float(np.sum(a2))
                obj2 = compute_obj(a2 / s2) if s2 > 1e-10 else np.inf

        # Best t from the search
        if obj1 < obj2:
            best_t = t1
            best_obj_t = obj1
        else:
            best_t = t2
            best_obj_t = obj2

        it += 1
        if best_obj_t < best_obj - 1e-12:
            best_obj = best_obj_t
            best_a = ((1 - best_t) * a + best_t * g_lp).copy()
            best_a = np.maximum(best_a, 0)
            best_a = best_a / np.sum(best_a)
            a = best_a.copy()
            last_step = best_t
            stall_count = 0
            best_it = it
            if verbose and it % 10 == 0:
                print(f"    LP-LS it{it}: obj={best_obj:.8f}, t={best_t:.4f}, M_lp={M_lp:.6f}")
        else:
            stall_count += 1
            # Perturb to escape
            if stall_count > 3:
                # Try a random perturbation in frequency domain
                n_pert = min(5, n // 4)
                pert = np.zeros(n)
                for k in range(1, n_pert + 1):
                    freq = k * 2 * np.pi / n
                    amp = 0.01 * rng.uniform(0.5, 2.0) / k
                    phase = rng.uniform(0, 2 * np.pi)
                    pert += amp * np.cos(freq * np.arange(n) + phase)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                s_p = float(np.sum(a))
                if s_p < 1e-10:
                    a = best_a.copy()
                else:
                    a = a / s_p
                stall_count = 0
                last_step = 0.1  # Reset step size

            if stall_count > 8:
                break

        # Reset if stuck for too long
        if it - best_it > 50:
            a = best_a.copy()
            stall_count = 0
            last_step = 0.05

    return best_a


def high_tau_gradient_descent(a, time_budget=120.0, tau_init=200.0, tau_end=5000.0,
                               verbose=True, seed=0):
    """
    High-tau smoothed gradient descent with Nesterov acceleration.
    Uses very high tau to approximate the true max operator.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    grad_norms = []

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Linear tau schedule (slower increase for more precision)
        tau = tau_init + (tau_end - tau_init) * progress ** 0.7

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute gradient at y
        g, conv = smoothed_gradient_fft(y, tau)
        g_norm = float(np.linalg.norm(g))
        grad_norms.append(g_norm)
        if len(grad_norms) > 200:
            grad_norms = grad_norms[-200:]

        # Adaptive learning rate
        avg_gnorm = float(np.mean(grad_norms))
        if avg_gnorm > 1e-15:
            adaptive_lr = 0.5 / (avg_gnorm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.05 * base_lr), 2.0 * base_lr)
            lr = 0.7 * lr + 0.3 * adaptive_lr

        # Barzilai-Borwein
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                bb_lr = min(max(bb_lr, 1e-8), 3.0 / np.sqrt(n))
                lr = 0.6 * lr + 0.4 * bb_lr

        # Update
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1

        # Slow decay
        lr = max(1e-9, lr * 0.99995)

        # Reset if stuck
        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.3
            best_it = it

        if verbose and it % 2000 == 0:
            print(f"    HT-GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.5f}")

    return best_a


def kkt_verify_and_refine(a, time_budget=60.0, verbose=True):
    """
    Verify KKT conditions and refine using active set information.
    Identifies active lags and does targeted optimization.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(50):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Identify active set (within 1% of max)
        threshold = max_conv * 0.995
        active_mask = conv >= threshold
        K_active = np.where(active_mask)[0]

        if len(K_active) == 0:
            break

        # Compute "hard" gradient: average gradient over active lags
        # grad_k[a_i] = a[k-i] + a[i-k] (with zero extension)
        hard_grad = np.zeros(n)
        for k in K_active:
            # Contribution from lag k
            for i in range(n):
                val = 0.0
                j1 = k - i
                j2 = i - k
                if 0 <= j1 < n:
                    val += a[j1]
                if 0 <= j2 < n:
                    val += a[j2]
                hard_grad[i] += val
        hard_grad /= len(K_active)

        # Project gradient onto tangent space of simplex
        # (subtract mean to maintain sum constraint)
        hard_grad_centered = hard_grad - np.mean(hard_grad)

        # Line search: a - alpha * grad_centered, project to simplex
        best_alpha_obj = best_obj
        best_alpha_a = best_a.copy()

        alphas = [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]
        for alpha in alphas:
            trial = a - alpha * hard_grad_centered
            trial = np.maximum(trial, 0)
            s_trial = float(np.sum(trial))
            if s_trial < 1e-10:
                continue
            trial = trial / s_trial
            obj_trial = compute_obj(trial)
            if obj_trial < best_alpha_obj:
                best_alpha_obj = obj_trial
                best_alpha_a = trial.copy()

        if best_alpha_obj < best_obj - 1e-12:
            best_obj = best_alpha_obj
            best_a = best_alpha_a.copy()
            a = best_a
            if verbose:
                print(f"    KKT it{it}: obj={best_obj:.8f}, |K|={len(K_active)}")
        else:
            break

    return best_a


def multi_n_search(best_seq, best_score, time_budget=200.0, verbose=True):
    """
    Try different sequence lengths to find a better n.
    """
    start = time.monotonic()
    n_best = len(best_seq)

    # Generate candidate n values
    n_candidates = sorted(set([
        n_best - 100, n_best - 50, n_best + 50, n_best + 100,
        n_best + 200, n_best + 400, n_best - 200,
        600, 700, 900, 1000, 1200, 1500
    ]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 2000]
    n_candidates = [n for n in n_candidates if abs(n - n_best) > 20]
    n_candidates = n_candidates[:8]

    if verbose:
        print(f"  Multi-n candidates: {n_candidates}")

    for n_try in n_candidates:
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 30:
            break

        # Resample
        n_old = len(best_seq)
        if n_old == n_try:
            continue
        x_old = np.linspace(0, 1, n_old)
        x_new = np.linspace(0, 1, n_try)
        a_try = np.interp(x_new, x_old, best_seq)
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # Quick LP polish
        alloc_lp = min(remaining * 0.5, 40.0)
        if alloc_lp > 15:
            a_try = lp_alternate_with_linesearch(a_try, time_budget=alloc_lp, verbose=False)

        # Quick GD
        remaining = time_budget - (time.monotonic() - start)
        alloc_gd = min(remaining * 0.3, 30.0)
        if alloc_gd > 10:
            a_try = high_tau_gradient_descent(a_try, time_budget=alloc_gd,
                                              tau_init=100.0, tau_end=2000.0, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            if verbose:
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
        elif verbose:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    return best_seq, best_score


def diverse_perturbation(a, rng, strength=0.03):
    """Apply a structured perturbation to escape local optima."""
    n = len(a)
    # Random smooth perturbation
    n_modes = max(2, n // 100)
    pert = np.zeros(n)
    x = np.arange(n)
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength / k * rng.uniform(0.5, 1.5)
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * x + phase)

    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: aggressive LP alternating with line search, high-tau GD polish,
    KKT refinement, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        n_current = 800
        best_seq = np.ones(n_current) / n_current
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")

    # === Phase 1: Aggressive LP Alternating with Line Search (main workhorse) ===
    print(f"\n=== Phase 1: LP Alternating with Line Search ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.45, 600.0)
    if p1_budget > 60:
        # Multiple rounds with different seeds for robustness
        for rnd in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = p1_budget / (4 - rnd)
            alloc = min(alloc, remaining * 0.6)
            if alloc < 20:
                break

            a_lp = lp_alternate_with_linesearch(best_seq, time_budget=alloc,
                                                verbose=True, seed=1000 + rnd * 7)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement)")
                # Perturb before next round
                if rnd < 3:
                    strength = 0.01 * (rnd + 1)
                    best_seq = diverse_perturbation(best_seq, rng, strength=strength)

    # === Phase 2: High-tau Gradient Descent Polish ===
    print(f"\n=== Phase 2: High-Tau GD Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 250.0)
    if p2_budget > 30:
        a_gd = high_tau_gradient_descent(best_seq, time_budget=p2_budget,
                                         tau_init=100.0, tau_end=5000.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase2 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 GD: obj={obj_gd:.8f} (no improvement)")

    # === Phase 3: KKT Verification and Refinement ===
    print(f"\n=== Phase 3: KKT Refinement ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 80.0)
    if p3_budget > 20:
        a_kkt = kkt_verify_and_refine(best_seq, time_budget=p3_budget, verbose=True)
        obj_kkt = compute_obj(a_kkt)
        if obj_kkt < best_score:
            best_score = obj_kkt
            best_seq = a_kkt.copy()
            print(f"  KKT: obj={best_score:.8f} **")
        else:
            print(f"  KKT: obj={obj_kkt:.8f} (no improvement)")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 200.0)
    if p4_budget > 40:
        best_seq, best_score = multi_n_search(best_seq, best_score,
                                              time_budget=p4_budget, verbose=True)

    # === Phase 5: Final Intensive Polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    if remaining > 40:
        alloc = min(remaining * 0.3, 120.0)
        a_lp = lp_alternate_with_linesearch(best_seq, time_budget=alloc,
                                            verbose=True, seed=99999)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_lp:.8f}")

    # Ultra-high tau GD
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 100.0)
        a_gd = high_tau_gradient_descent(best_seq, time_budget=alloc,
                                         tau_init=500.0, tau_end=10000.0, verbose=False)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Ultra GD: obj={best_score:.8f} **")
        else:
            print(f"  Ultra GD: obj={obj_gd:.8f}")

    # One more KKT pass
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 50.0)
        a_kkt = kkt_verify_and_refine(best_seq, time_budget=alloc, verbose=False)
        obj_kkt = compute_obj(a_kkt)
        if obj_kkt < best_score:
            best_score = obj_kkt
            best_seq = a_kkt.copy()
            print(f"  Final KKT: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
