"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of:
- High-accuracy FFT-smoothed gradient descent with Nesterov momentum
- Peak-flooding coordinate descent (new: targets top peaks directly)
- Aggressive LP-based direction finding with adaptive blending
- Parametric spectral optimization (new: low-dimensional sine-series fitting)
- Multi-length exploration with resampling
Key new ideas: peak-flooding reduces active peaks via projected gradient,

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of (a*a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def nesterov_smoothed_descent(a, time_budget=200.0, tau_init=5.0, tau_end=800.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 4.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        if it % 10000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient(y, tau)

        if prev_g is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.3 * lr + 0.7 * min(max(bb_lr, 1e-9), 20.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1

        lr = max(1e-9, lr * 0.9998)

        if it - best_it > 40000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 4.0 / np.sqrt(n) * 0.4
            best_it = it

    return best_a


def peak_flooding(a, time_budget=100.0, num_peaks=3, verbose=False):
    """
    Peak-flooding: iteratively target the top peaks of the convolution
    and reduce them via projected gradient steps on the simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    no_improve = 0
    step_scale = 0.5

    while time.monotonic() - start < time_budget and no_improve < 200:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        obj = 2.0 * n * max_conv

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Find top peaks
        top_indices = np.argsort(conv)[::-1][:num_peaks]

        # Build combined gradient targeting top peaks
        combined_grad = np.zeros(n)
        for k in top_indices:
            k = int(k)
            # d(conv[k])/da[i] = 2*a[k-i] if 0 <= k-i < n
            for i in range(min(k, n)):
                combined_grad[i] += 2.0 * a[k - i]
            # Also from the "other side": d(conv[k])/da[i] includes a[i-k]
            for i in range(k + 1, k + n):
                if i < n:
                    combined_grad[i] += 2.0 * a[i - k]

        # Project gradient onto simplex tangent (zero sum)
        combined_grad -= np.mean(combined_grad)

        # Adaptive step size via line search
        objs = []
        for t in [step_scale, step_scale * 0.5, step_scale * 0.25, step_scale * 0.1,
                  step_scale * 2.0, step_scale * 0.75]:
            a_try = a - t * combined_grad
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            o = compute_obj(a_try)
            objs.append((o, a_try, t))

        if not objs:
            break

        objs.sort(key=lambda x: x[0])
        best_o, best_try, best_t = objs[0]

        if best_o < best_obj:
            a = best_try
            step_scale = best_t * 1.1
            best_obj = best_o
            best_a = a.copy()
        else:
            step_scale *= 0.7
            if step_scale < 1e-6:
                break

        it += 1
        if verbose and it % 50 == 0:
            print(f"    PF it{it}: obj={best_obj:.8f}, step={step_scale:.4f}, t={time.monotonic()-start:.0f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish_round(a, max_iter=100, time_limit=100.0, verbose=False):
    """LP polishing: repeatedly solve LP for optimal g given a, then blend."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf, 0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    no_improve = 0
    last_t = 0.5
    start = time.monotonic()
    it = 0

    while it < max_iter and (time.monotonic() - start) < time_limit:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        candidates = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8,
                      0.9, 1.0,
                      max(0.03, last_t * 0.5), max(0.03, last_t * 0.7),
                      max(0.03, last_t * 1.1), min(1.0, last_t * 1.3)]

        best_t_obj = np.inf
        best_t_val = last_t
        best_a_local = best_a.copy()

        for t in set(candidates):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            no_improve = 0
            if verbose and it % 5 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}, M={M_lp:.8f}")
        else:
            no_improve += 1
            if no_improve >= 4:
                break

    return best_a, best_obj, it


def parametric_optimize(a_init, time_budget=100.0, max_freqs=8, verbose=False):

                obj = compute_obj(a_opt)
                if verbose:
                    print(f"    Parametric: obj={obj:.8f}")
                return a_opt
    except Exception as e:
        if verbose:
            print(f"    Parametric failed: {e}")

    return a_init


def perturb_sequence(a, rng, method=0, scale=0.03):
    """Apply a perturbation to escape local minima."""
    n = len(a)
    a = a.copy()
    if method == 0:
        a = a * np.exp(rng.normal(0, scale, n))
    elif method == 1:
        noise = rng.uniform(1 - scale, 1 + scale, n)
        a = a * noise
    elif method == 2:
        mask = rng.random(n) > scale * 2
        a = a * mask
    elif method == 3:
        shift = int(rng.integers(1, max(2, n // 10)))
        a = np.roll(a, shift)
    elif method == 4:
        a = (1 - scale) * a + scale / n
    elif method == 5:
        N = 1
        while N < 2 * n:
            N <<= 1
        A = fft(a, N)
        A[1:N // 2] *= np.exp(1j * rng.normal(0, scale, len(A[1:N // 2])))
        A[N // 2 + 1:] = np.conj(np.flip(A[1:N // 2]))
        a = ifft(A).real[:n]
        a = np.maximum(a, 0)
    elif method == 6:
        # Anti-peak perturbation: reduce mass at locations contributing to peaks
        conv = compute_conv(a / a.sum())
        top_k = np.argsort(conv)[::-1][:min(5, len(conv))]
        deltas = np.zeros(n)
        for k in top_k:
            k = int(k)
            for j in range(min(k, n)):
                deltas[j] += a[k - j]
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                deltas[j] += a[k - j]
        deltas /= (float(np.sum(deltas)) + 1e-10)
        a = a - scale * 0.3 * (n * deltas - 1.0)
        a = np.maximum(a, 0)
    else:
        a = a * np.exp(rng.normal(0, scale, n))

    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = list(rng.integers(1, 6, size=3))
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1.5, 1.5)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        # Two-peak with gap
        a = np.zeros(n)
        a[:n // 3] = 1.0
        a[2 * n // 3:] = 1.0
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining smoothed GD, peak-flooding, LP polishing,
    parametric optimization, and multi-length exploration.
    Returns list of non-negative floats minimizing the evaluation objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: High-accuracy smoothed GD (warm start) ===
    print(f"\n=== Phase 1: Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.12, 120.0)
    if p1_budget > 20:
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=p1_budget,
                                          tau_init=20.0, tau_end=1000.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  ** Phase1 SGD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 SGD: obj={obj_sgd:.8f}")

    # === Phase 2: Peak-flooding (new method) ===
    print(f"\n=== Phase 2: Peak Flooding ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.1, 100.0)
    if p2_budget > 20:
        a_pf = peak_flooding(best_seq, time_budget=p2_budget, num_peaks=3, verbose=False)
        obj_pf = compute_obj(a_pf)
        if obj_pf < best_score:
            best_score = obj_pf
            best_seq = a_pf.copy()
            print(f"  ** Peak Flooding: obj={best_score:.8f} **")
        else:
            print(f"  Peak Flooding: obj={obj_pf:.8f}")

    # === Phase 3: Intensive LP polishing ===
    print(f"\n=== Phase 3: LP Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.25, 300.0)
    if p3_budget > 30:
        a_current = best_seq.copy()
        num_rounds = 6
        round_budget = p3_budget / num_rounds

        for rnd in range(num_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(round_budget, remaining * 0.4)
            if alloc < 15:
                break

            a_new, obj_new, iters = lp_polish_round(a_current, max_iter=80,
                                                     time_limit=alloc, verbose=(rnd == 0))
            if obj_new < best_score - 1e-12:
                best_score = obj_new
                best_seq = a_new.copy()
                a_current = a_new.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} ({iters} iters) **")
            else:
                print(f"  LP round {rnd}: obj={obj_new:.8f} ({iters} iters)")
                # Perturb and retry
                method = int(rng.integers(0, 7))
                scale = 0.015 + 0.01 * rnd
                a_current = perturb_sequence(a_new, rng, method=method, scale=scale)
                remaining = deadline - time.monotonic()
                alloc2 = min(alloc * 0.4, remaining * 0.2)
                if alloc2 > 10:
                    a_pert, obj_pert, _ = lp_polish_round(a_current, max_iter=30, time_limit=alloc2)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        a_current = a_pert.copy()
                        print(f"  LP round {rnd} (perturbed): obj={best_score:.8f} **")

    # === Phase 4: Parametric optimization (new) ===
    print(f"\n=== Phase 4: Parametric Opt ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.08, 80.0)
    if p4_budget > 20:
        a_param = parametric_optimize(best_seq, time_budget=p4_budget,
                                      max_freqs=8, verbose=True)
        obj_param = compute_obj(a_param)
        if obj_param < best_score:
            best_score = obj_param
            best_seq = a_param.copy()
            print(f"  ** Parametric: obj={best_score:.8f} **")
        else:
            print(f"  Parametric: obj={obj_param:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 100), max(400, n_current - 50),
        n_current + 50, n_current + 100, n_current + 150, n_current + 200,
        700, 750, 800, 850, 900, 950, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1400 and abs(n - n_current) > 20]
    n_options.sort(key=lambda x: abs(x - n_current))

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick smoothed GD
        alloc1 = min(remaining * 0.08, 40.0)
        if alloc1 > 10:
            a_sgd = nesterov_smoothed_descent(a_try, time_budget=alloc1,
                                              tau_init=20.0, tau_end=400.0)
            if compute_obj(a_sgd) < compute_obj(a_try):
                a_try = a_sgd

        # Peak flooding
        remaining = deadline - time.monotonic()
        alloc_pf = min(remaining * 0.05, 25.0)
        if alloc_pf > 5:
            a_try = peak_flooding(a_try, time_budget=alloc_pf, num_peaks=3)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.12, 60.0)
        if alloc2 > 10:
            a_try, obj_try, _ = lp_polish_round(a_try, max_iter=40, time_limit=alloc2)
        else:
            obj_try = compute_obj(a_try)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = (r * 3 + 1) % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.1 + 0.05 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.15, 60.0)
        if alloc > 25:
            third = alloc / 3.0
            a_r = nesterov_smoothed_descent(a_r, time_budget=third,
                                            tau_init=20.0, tau_end=300.0)
            remaining = deadline - time.monotonic()
            a_r = peak_flooding(a_r, time_budget=min(third, remaining * 0.1), num_peaks=2)
            remaining = deadline - time.monotonic()
            lp_alloc = min(third, remaining * 0.1)
            if lp_alloc > 10:
                a_r, obj_r, _ = lp_polish_round(a_r, max_iter=30, time_limit=lp_alloc)
            else:
                obj_r = compute_obj(a_r)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau smoothed GD
    if remaining > 40:
        alloc = min(remaining * 0.2, 60.0)
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                          tau_init=100.0, tau_end=2000.0)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Final SGD: obj={best_score:.8f} **")
        else:
            print(f"  Final SGD: obj={obj_sgd:.8f}")

    # Peak flooding
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 50.0)
        a_pf = peak_flooding(best_seq, time_budget=alloc, num_peaks=5, verbose=False)
        obj_pf = compute_obj(a_pf)
        if obj_pf < best_score:
            best_score = obj_pf
            best_seq = a_pf.copy()
            print(f"  Final PF: obj={best_score:.8f} **")
        else:
            print(f"  Final PF: obj={obj_pf:.8f}")

    # Multiple LP polish rounds
    for lp_rnd in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * 0.2, 60.0)
        if alloc < 10:
            break
        a_current = best_seq.copy()

        if lp_rnd > 0:
            a_current = perturb_sequence(a_current, rng, method=lp_rnd % 6, scale=0.015)

        a_lp, obj_lp, iters = lp_polish_round(a_current, max_iter=80, time_limit=alloc)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP {lp_rnd}: obj={best_score:.8f} ({iters} iters) **")
        else:
            print(f"  Final LP {lp_rnd}: obj={obj_lp:.8f} ({iters} iters)")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
