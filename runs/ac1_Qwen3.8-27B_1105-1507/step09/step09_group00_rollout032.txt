Uses LP-based fixed-point iteration with both asymmetric and symmetric linearizations,

import numpy as np

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def compute_obj(a):

    if n == 0:
        return np.inf
    s = float(np.sum(a))

        return np.inf

    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def solve_lp_asymmetric(a, solver=None):
    """Solve: min M s.t. conv(a, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build A matrix: A[k,j] = a_norm[k-j] if valid
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a_norm
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver is None:
        solvers = [cp.MOSEK, cp.GLOP, cp.CLARABEL, cp.ECOS, cp.SCS, cp.CBC]

def solve_lp_symmetric(a, solver=None):

def smoothed_gradient(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    # Gradient: d/d a[j] of sum_k p[k] * conv(a,a)[k]
    # = sum_k p[k] * 2 * a[k-j] (for valid k-j)
    # This is 2 * cross_correlation(p, a)
    size = 1
    while size < 3 * n:
        size <<= 1
    P = np.fft.fft(p, size)
    A = np.fft.fft(a, size)
    g_full = np.fft.ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv

        frac = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * frac ** 2
        
        g, conv = smoothed_gradient(a, tau)

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2 (scale-invariant)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def solve_lp_asymmetric(a):
    """Solve: min M s.t. conv(a, g)[k] <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.GLOP, cp.CLARABEL, cp.ECOS, cp.SCS, cp.CBC]
    for solver in solvers:

                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))

def solve_lp_symmetric(a):
    """Solve SCA LP: min M s.t. 2*conv(a*,a)[k] <= M + c[k], sum(a)=1, a>=0."""

                sv = float(np.sum(a_val))

def smoothed_gradient(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) via log-sum-exp + FFT cross-correlation."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    N = 1
    while N < 3 * n:
        N <<= 1
    P = np.fft.fft(p, N)
    A = np.fft.fft(a, N)
    g_full = np.fft.ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s
