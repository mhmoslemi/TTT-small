    """
    Solve: min M s.t. (g * a)_k <= M for all k,
           sum(a) = 1, a >= 0
    Returns a_optimal (normalized to sum=1).
    """
    # (g * a)_k = sum_{i=0}^{min(n-1, k)} g_i * a_{k-i} for k = 0, ..., 2n-2

        a_val = a_val / a_val.sum()

  - Given a, find g that minimizes max_k (a*g)_k subject to sum(g)=1, g>=0

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_convolve_matrix(seq):
    """
    Build matrix A (2n-1 x n) such that A @ x = convolve(seq, x).
    A[k, j] = seq[k-j] if 0 <= k-j <= n-1, else 0
    """
    n = len(seq)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # j ranges such that 0 <= k-j <= n-1
        # => k-(n-1) <= j <= k
        j_start = max(0, k - (n-1))
        j_end = min(n-1, k)
        for j in range(j_start, j_end+1):
            A[k, j] = seq[k-j]
    return A

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: Alternating LP method that alternates between two convex LPs:
  - Given g, find a minimizing max_k (g*a)_k on the simplex
  - Given a, find g minimizing max_k (a*g)_k on the simplex
Combined with QP subproblems, Nesterov smoothed gradient descent,
adaptive blending, multi-length exploration, and perturbation-based restarts.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(seq):
    """Build matrix A (2n-1 x n) such that A @ x = convolve(seq, x)."""
    n = len(seq)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = seq
    return A


def solve_lp_min_peak(fixed_seq, n, time_limit=30.0):
    """
    Solve: min M s.t. (fixed_seq * x)_k <= M for all k, sum(x)=1, x>=0.
    Returns (x_opt, M_opt) or (None, None).
    """
    A = build_conv_matrix(fixed_seq)
    x = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ x <= M, cp.sum(x) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            deadline = time.monotonic() + time_limit
            prob.solve(solver=solver)
            if x.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                x_val = np.maximum(x.value, 0)
                s = float(np.sum(x_val))
                if s > 1e-10:
                    return x_val / s, float(M.value)
            if time.monotonic() > deadline:
                break
        except Exception:
            continue
    return None, None


def solve_qp_min_energy(fixed_seq, n, time_limit=30.0):
    """
    Solve: min sum_k ((fixed_seq * x)_k)^2 s.t. sum(x)=1, x>=0.
    This is a convex QP since it minimizes ||A @ x||^2.
    Returns x_opt or None.
    """
    A = build_conv_matrix(fixed_seq)
    P = A.T @ A  # n x n, PSD
    x = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(x) == 1]
    prob = cp.Problem(cp.Minimize(cp.quad_form(x, cp.psd_wrap(P))), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if x.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                x_val = np.maximum(x.value, 0)
                s = float(np.sum(x_val))
                if s > 1e-10:
                    return x_val / s
        except Exception:
            continue
    return None


def solve_combined_lp_qp(fixed_seq, n, alpha=0.5, time_limit=30.0):
    """
    Solve: min alpha*M + (1-alpha)*||A@x||^2 s.t. A@x <= M, sum(x)=1, x>=0.
    Convex QP with linear constraints.
    """
    A = build_conv_matrix(fixed_seq)
    P = A.T @ A
    x = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ x <= M, cp.sum(x) == 1]
    obj = alpha * M + (1.0 - alpha) * cp.quad_form(x, cp.psd_wrap(P))
    prob = cp.Problem(cp.Minimize(obj), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if x.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                x_val = np.maximum(x.value, 0)
                s = float(np.sum(x_val))
                if s > 1e-10:
                    return x_val / s
        except Exception:
            continue
    return None


def alternating_lp(a, time_budget, rng, verbose=True):
    """
    Alternating LP method:
    - Given g=a, solve LP for new a that minimizes max_k (g*a_new)_k
    - Given a_new, solve LP for new g that minimizes max_k (a_new*g)_k
    - Blend and update
    With QP variants and adaptive step sizes.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    g = a.copy()
    start = time.monotonic()
    it = 0
    blend_t = 0.8
    stall = 0
    method_cycle = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 5:
            break

        # Cycle through different LP formulations
        method = method_cycle % 3
        method_cycle += 1

        candidate = None
        if method == 0:
            # Standard LP: min max_k (g * x)_k
            candidate, M_val = solve_lp_min_peak(g, n, time_limit=min(60, remaining * 0.3))
        elif method == 1:
            # QP: min ||g * x||^2
            candidate = solve_qp_min_energy(g, n, time_limit=min(60, remaining * 0.3))
            if candidate is None:
                candidate, M_val = solve_lp_min_peak(g, n, time_limit=min(60, remaining * 0.3))
        else:
            # Combined LP+QP
            candidate = solve_combined_lp_qp(g, n, alpha=0.3, time_limit=min(60, remaining * 0.3))
            if candidate is None:
                candidate, M_val = solve_lp_min_peak(g, n, time_limit=min(60, remaining * 0.3))

        if candidate is None:
            stall += 1
            if stall > 10:
                break
            continue

        # Try different blending factors
        best_candidate = None
        best_cand_obj = best_obj

        # Adaptive blend range
        if stall == 0:
            t_range = [blend_t, 0.5, 1.0, 0.7, 0.3]
        else:
            t_range = [0.3, 0.5, 0.7, 1.0, 0.1]

        for t in t_range:
            a_try = (1.0 - t) * a + t * candidate
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_cand_obj:
                best_cand_obj = obj_try
                best_candidate = a_try.copy()

        it += 1

        if best_candidate is not None:
            best_obj = best_cand_obj
            best_a = best_candidate.copy()
            a = best_candidate.copy()
            g = best_candidate.copy()
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    AltLP it{it}: obj={best_obj:.8f}, method={method}, t={t_range[0]:.2f}")
        else:
            stall += 1
            if verbose and stall % 5 == 0:
                print(f"    AltLP it{it}: no improvement, stall={stall}")
            if stall > 20:
                # Perturb to escape
                scale = 0.01 * min(stall, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                g = a.copy()
                stall = 0
            if stall > 50:
                break

    return best_a


def smoothed_grad_fft(a, tau):
    """Compute gradient of smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n], conv


def nesterov_smoothed_descent(a, time_budget, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_grad_fft(y, tau)

        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.7 * lr + 0.3 * min(max(bb_lr, 1e-8), 5.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)

        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it

        if verbose and it % 5000 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}")

    return best_a


def dual_guided_subgradient(a, time_budget, K=15, verbose=False):
    """
    Subgradient method using active constraint weights from the LP dual.
    At each step, identify top peaks and use weighted subgradient with
    adaptive step size (Polyak-type).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Estimate lower bound (use 1/n as crude bound for max(a*a) when sum=1)
    lb = 1.0 / n * 0.5
    gap = best_obj / (2.0 * n) - lb
    lr = 2.0 * gap / n
    momentum = 0.5
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        current_val = max_conv / (1.0 / n)  # normalized peak

        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break

        # Soft-max weights
        w = np.exp((peak_vals - max_pv) * 5.0 / (max_pv + 1e-15))
        w = w / w.sum()

        # Compute subgradient
        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += wi * a[j_start:j_end + 1][::-1]

        # Polyak step size
        gap = max_conv - lb
        if gap > 1e-12:
            g_norm_sq = float(np.dot(g, g))
            if g_norm_sq > 1e-15:
                lr_p = 2.0 * gap / (g_norm_sq * n)
                lr = 0.8 * lr + 0.2 * min(lr_p, 1.0 / np.sqrt(n))

        v = momentum * v + g
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            momentum = min(0.9, momentum + 0.01)
        else:
            stall += 1
            if stall > 50:
                lr *= 0.6
                v *= 0.5
                momentum = 0.3
                stall = 0
            if stall > 200:
                # Perturb
                scale = 0.005 * min(stall, 10)
                a = best_a * np.exp(np.random.default_rng(it).normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                v = np.zeros(n)
                stall = 0

        it += 1
        if verbose and it % 500 == 0:
            print(f"    DSG it{it}: obj={best_obj:.8f}, lr={lr:.6f}")

    return best_a


def peak_shave_direct(a, time_budget, verbose=False):
    """Direct peak shaving with backtracking line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.5 / np.sqrt(n)
    K = 10
    it = 0
    fail_count = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]

        # Compute descent direction (average gradient of top peaks)
        g = np.zeros(n)
        for k in top_idx:
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += a[j_start:j_end + 1][::-1]
        g /= K

        # Backtracking line search
        alpha = lr
        improved = False
        for _ in range(10):
            a_new = a - alpha * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                alpha *= 0.5
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj * (1 - 1e-10):
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            alpha *= 0.5

        if improved:
            lr = min(lr * 1.1, 2.0 / np.sqrt(n))
            fail_count = 0
        else:
            lr *= 0.7
            fail_count += 1
            if fail_count > 30:
                break

        it += 1
        if verbose and it % 200 == 0:
            print(f"    PS it{it}: obj={best_obj:.8f}, lr={lr:.6f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Symmetric initialization
        half = n // 2 + 1
        h = rng.uniform(0.1, 1.0, half)
        a = np.zeros(n)
        a[:half] = h
        a[half:] = h[:n - half][::-1]
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a
