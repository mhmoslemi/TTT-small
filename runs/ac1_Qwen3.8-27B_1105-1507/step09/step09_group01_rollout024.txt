def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp_fast(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Uses HiGHS."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    # x = [g (n), M (1)]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        g = np.maximum(res.x[:n], 0)
        s = g.sum()
        if s > 1e-10:
            g = g / s
        return g, float(res.x[n])
    return None, None


def solve_lp_cvxpy(a):
    """Fallback LP solver using cvxpy."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp(a):
    """Try fast solver first, fall back to cvxpy."""
    try:
        result = solve_lp_fast(a)
        if result[0] is not None:
            return result
    except Exception:
        pass
    return solve_lp_cvxpy(a)

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.arange(n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1 + 1e-10)))
        a = a / a.sum()
    elif mode == 3:
        # Triangle (peak in middle)
        a = 1.0 - 2.0 * np.abs(x - (n-1)/2.0) / (n - 1 + 1e-10)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Two bumps
        c1, c2 = n // 4, 3 * n // 4
        w = n // 8
        a = np.exp(-((x - c1)**2) / (2*w*w)) + np.exp(-((x - c2)**2) / (2*w*w))
        a = a / a.sum()
    elif mode == 5:
        # Power law
        a = (x + 1) ** rng.uniform(-1, 1)
        a = a / a.sum()
    elif mode == 6:
        # Sine modulation
        f = rng.uniform(1, 4)
        a = 1 + 0.5 * np.sin(2 * np.pi * f * x / n)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

def perturbation_search(a, time_budget=100.0, rng=None, verbose=False):
    """Try diverse perturbations and keep the best."""
    n = len(a)

    ]
    
    for scale, ptype in strategies:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 20:

            best_a = a_pert.copy()
