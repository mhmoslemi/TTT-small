```python
def build_conv_matrix_fixed_fixed(fixed_seq, n_var):
    """Build matrix A such that A @ x = conv(fixed_seq, x)[:n_var + len(fixed_seq) - 1]"""
    n_f = len(fixed_seq)
    A = np.zeros((n_f + n_var - 1, n_var))
    for i in range(n_var):
        # conv(fixed_seq, x)[k] = sum_j fixed_seq[j] * x[k-j]
        # For fixed i (index of x), contribution to A[k, i] = fixed_seq[k-i]
        for k in range(n_f + n_var - 1):
            j = k - i
            if 0 <= j < n_f:
                A[k, i] = fixed_seq[j]
    return A
```

```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses alternating LP (fix one variable, solve for other) with adaptive blending,
FFT-accelerated gradient descent for polishing, and multi-scale exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_ab(f, g):
    """Full linear convolution of f and g via FFT."""
    n_f = len(f)
    n_g = len(g)
    size = 1
    while size < n_f + n_g - 1:
        size <<= 1
    ff = rfft(f, size)
    fg = rfft(g, size)
    return irfft(ff * fg, size)[:n_f + n_g - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(fixed, n_var):
    """Build matrix A (len(n_var)+len(fixed)-1 x n_var) such that A @ x = conv(fixed, x)."""
    n_f = len(fixed)
    n_rows = n_f + n_var - 1
    A = np.zeros((n_rows, n_var))
    # A[k, i] = fixed[k-i] if 0 <= k-i < n_f
    # For each column i, A[:, i] has fixed shifted by i
    for i in range(n_var):
        start = i
        end = min(n_rows, i + n_f)
        length = end - start
        if length > 0:
            A[start:end, i] = fixed[:length]
    return A


def solve_lp_for_var(fixed, n_var, solver_list=None):
    """
    Solve LP: min M s.t. conv(fixed, x)[k] <= M for all k, sum(x)=1, x>=0.
    Returns (optimal_x, optimal_M) or (None, None).
    """
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    
    A = build_conv_matrix(fixed, n_var)
    x = cp.Variable(n_var, nonneg=True)
    M = cp.Variable()
    constraints = [A @ x <= M, cp.sum(x) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if x.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                x_val = np.maximum(x.value, 0)
                s = float(np.sum(x_val))
                if s > 1e-10:
                    return x_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    N = 1
    while N < 3 * n:
        N <<= 1
    
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    
    return g, conv

def alternating_lp(a, time_budget=600.0, verbose=True, max_iter=500):
    """
    Alternating LP: fix one side, solve LP for the other, iterate.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    g = a.copy()  # Initialize g as copy of a
    t_blend = 1.0  # Start with full LP update
    stall_count = 0
    oscillation_count = 0
    prev_obj = best_obj
    it = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # Step 1: Solve for g given a (fixed)
        g_sol, M_g = solve_lp_for_var(a, n)
        if g_sol is None:
            break
        
        # Step 2: Solve for a_new given g_sol (fixed)
        a_sol, M_a = solve_lp_for_var(g_sol, n)
        if a_sol is None:
            break
        
        # Check for oscillation: if a_sol is far from a, reduce blend
        dist = np.linalg.norm(a_sol - a)
        if dist > 0.5:
            t_blend = max(0.1, t_blend * 0.5)
            oscillation_count += 1
        else:
            oscillation_count = max(0, oscillation_count - 1)
            t_blend = min(1.0, t_blend * 1.1)
        
        # Blend
        a_new = (1.0 - t_blend) * a + t_blend * a_sol
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        it += 1
        
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    ALP it{it}: obj={best_obj:.8f}, M_a={M_a:.6f}, t={t_blend:.3f}, dist={dist:.4f}, elapsed={elapsed:.0f}s")
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 20:
                # Perturb to escape
                rng = np.random.default_rng(it * 7919)
                pert = rng.normal(0, 0.01, n)
                a_new = a_new * np.exp(pert)
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                stall_count = 0
                if it > 200:
                    break
        
        a = a_new
        g = g_sol
    
    return best_a


def fourier_perturb(a, rng, strength=0.03):
    """Add Fourier mode perturbation."""
    n = len(a)
    x = np.arange(n)
    pert = np.zeros(n)
    for k in range(1, 5):
        amp = strength / k * rng.uniform(0.5, 1.5)
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(2 * np.pi * k * x / n + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 3:
        a = np.exp(-np.abs(np.arange(n) - n/2) / (n/4))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2*np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = 1.0
        for f, p in zip(freqs, phases):
            a = a + 0.3 * np.sin(2*np.pi*f*x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
```
