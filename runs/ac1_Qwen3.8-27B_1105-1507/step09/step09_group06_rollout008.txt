"""

Uses exact subgradient descent with momentum, aggressive LP polishing,
adaptive blending, and diverse multi-scale exploration.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu * rho > (cssv - 1.0)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_val = idx[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def exact_subgradient(a, conv, top_k=5):

def smoothed_gradient(a, tau):

    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    A = fft(a, N)

def build_conv_matrix(a):
    """Build matrix A_mat (2n-1 x n) such that A_mat @ g = convolve(a, g)."""
    n = len(a)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A_mat[j:j + r, j] = a[:r]
    return A_mat


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    if n > 3000:
        # Too large for LP, skip
        return None, None
    a_norm = a / np.sum(a)
    A_mat = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

        return a

    v = np.zeros(n)
    it = 0

def fast_coordinate_descent(a, time_budget=100.0, seed=0):
    """
    Fast coordinate descent: for each coordinate, find the best value
    by bisection on the 1D problem.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Random order
        order = np.arange(n)
        rng.shuffle(order)
        
        improved = False
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            
            old_val = a[i]
            if old_val < 1e-15:
                # Try setting it to small positive
                a[i] = 1e-3 / n
                s = float(np.sum(a))
                a = a / s
                obj = compute_obj(a)
                if obj > best_obj:
                    a[i] = 0
                    s = float(np.sum(a))
                    a = a / s
                else:
                    improved = True
                    best_obj = obj
                    best_a = a.copy()
                continue
            
            # Try multiple perturbations
            best_local = best_obj
            best_local_val = old_val
            base = 0.05 / np.sqrt(n)
            
            for delta in [base, -base, 2*base, -2*base, 0.5*base, -0.5*base,
                         4*base, -4*base, 8*base, -8*base]:
                new_val = old_val + delta
                if new_val < 0:
                    new_val = 0
                a[i] = new_val
                s = float(np.sum(a))
                if s < 1e-10:
                    a[i] = old_val
                    continue
                a = a / s
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_local_val = new_val
            
            a[i] = best_local_val
            s = float(np.sum(a))
            a = a / s
            if best_local < best_obj:
                best_obj = best_local
                best_a = a.copy()
                improved = True
        
        it += 1
        if not improved and it > 5:
            # Reset step size
            break
    
    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 8, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        c1 = rng.integers(0, n // 3)
        c2 = rng.integers(2 * n // 3, n)
        w = max(1, n // 12)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 8:
        a = rng.uniform(0, 1, n)
        k = max(2, n // 50)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 9:
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2
        a = a / a.sum()
    elif mode == 10:
        # Three peaks
        centers = [n//5, n//2, 4*n//5]
        w = n // 15
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 11:
        # Cosine window
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
