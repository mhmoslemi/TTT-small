"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Gerchberg-Saxton for fast initialization, multiplicative weights (mirror descent)
for main optimization, and LP polishing for final precision.

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = fft(a, N)
    return ifft(fa * fa).real[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def gs_iterate(a, target_mag, N, relax=1.0):
    """One Gerchberg-Saxton iteration."""
    n = len(a)
    A = fft(a, N)
    mag = np.abs(A)
    phase = A / (mag + 1e-30)
    A_new = target_mag * phase
    a_new = ifft(A_new).real[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    a_new = a_new / s
    return a * (1 - relax) + a_new * relax


def compute_target_mag(n, N):
    """Compute target magnitude for flat autocorrelation."""
    c = np.zeros(N)
    c[:2*n-1] = 1.0 / (2*n-1)
    C = fft(c)
    target_mag = np.sqrt(np.abs(C) + 1e-30)
    # Preserve the DC component (k=0) to maintain sum=1
    # Actually, we normalize after each iteration, so this is fine
    return target_mag


def gerchberg_saxton(a, iters=2000, relax=1.0, verbose=False):
    """Run Gerchberg-Saxton iterations."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    target_mag = compute_target_mag(n, N)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for i in range(iters):
        a = gs_iterate(a, target_mag, N, relax)
        if i % 100 == 0:

            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
            if verbose and i % 500 == 0:
                print(f"    GS it{i}: obj={obj:.8f}, best={best_obj:.8f}")

def multiplicative_weights(a, time_budget, eta_init=0.1, K=5, beta=5.0, verbose=False):

    N = 1
    while N < 2 * n - 1:

        return a

    best_obj = compute_obj(a)

    eta = eta_init

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)

    it = 0
    last_t = 0.5
    
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Try several blend factors
        candidates = [0.1, 0.2, 0.3, 0.5, 0.7, last_t, min(1.0, last_t * 1.5)]
        candidates = sorted(set([c for c in candidates if 0.01 <= c <= 1.0]))
        
        best_t_obj = best_obj
        best_t_val = last_t
        best_a_local = best_a.copy()
        
        for t in candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        it += 1
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            if verbose and it % 10 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={last_t:.3f}")
        else:
            # Perturb
            rng = np.random.default_rng(it)
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            if it > 100:

    return best_a

        # Build a "signal" from the peaks and correlate
        p = np.zeros(2*n-1)
        for wi, k in zip(w, top_idx):
            p[int(k)] = wi
        
        # cross-correlation: g[i] = sum_k p[k] * a[k-i] for valid k-i
        # This is the convolution of p_reversed with a
        # Actually, let's just do it directly for the top K
        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end+1] += wi * a[j_start:j_end+1][::-1]
        
        v = momentum * v + g
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        
        a_new = a - lr * v / v_norm
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            a = best_a
            if stall % 20 == 0:
                lr *= 0.7
                v *= 0.5
            if lr < 1e-10:
                break
    
    return best_a


def resample(a, new_n):
    """Resample sequence via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng):
    """Generate diverse initializations."""
    mode = rng.integers(0, 6)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Arcsine-like
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / (np.sqrt(1 - x**2) + 1e-6)
        a = a / a.sum()
    elif mode == 3:
        # Gaussian center
        x = np.arange(n)
        a = np.exp(-((x - n//2)**2) / (2 * (n//4)**2))
        a = a / a.sum()
    elif mode == 4:
        # Sine perturbation
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 5) * x + rng.uniform(0, 6.28))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = rng.paretian(rng.uniform(1.1, 3), n)
        a = a / a.sum()
    return a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Gerchberg-Saxton for fast spectral initialization, multiplicative weights
(mirror descent) for main optimization, truncated LP polishing, and multi-n
exploration. Key innovation: fast GS-based initialization + aggressive MW
with parameter sweeping + truncated LP for speed.
"""

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = fft(a, N)
    return ifft(fa * fa).real[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_target_mag(n, N):
    """Compute target magnitude spectrum for flat autocorrelation."""
    c = np.zeros(N)
    c[:2 * n - 1] = 1.0 / (2 * n - 1)
    C = fft(c)
    target_mag = np.sqrt(np.abs(C) + 1e-30)
    return target_mag


def gs_iterate(a, target_mag, N, relax):
    """One Gerchberg-Saxton iteration with relaxation."""
    n = len(a)
    A = fft(a, N)
    mag = np.abs(A)
    phase = A / (mag + 1e-30)
    A_new = target_mag * phase
    a_new = ifft(A_new).real[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    a_new = a_new / s
    return a * (1.0 - relax) + a_new * relax


def gerchberg_saxton(a, time_budget, relax=0.8, verbose=False):
    """Run Gerchberg-Saxton iterations for a given time budget."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    target_mag = compute_target_mag(n, N)

    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    check_interval = max(10, n // 5)

    while time.monotonic() - start < time_budget:
        a = gs_iterate(a, target_mag, N, relax)
        it += 1
        if it % check_interval == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
            if verbose and it % (check_interval * 5) == 0:
                print(f"    GS it{it}: obj={obj:.8f}, best={best_obj:.8f}")
    return best_a


def multiplicative_weights(a, time_budget, eta_init=0.1, K=5, beta=5.0, verbose=False):
    """
    Multiplicative weights (mirror descent) optimization.
    Update: a_i <- a_i * exp(-eta * p_i) where p_i aggregates pressure from top peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta_init
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        w = np.exp(beta * (peak_vals - max_conv))
        w /= w.sum()

        p = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                p[i_start:i_end + 1] += wi * a[j_start:j_end + 1][::-1]

        a_new = a * np.exp(-eta * p)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            it += 1
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall % 30 == 0:
                eta *= 0.93
            if stall > 400:
                eta *= 0.7
                stall = 0
                if eta < 1e-7:
                    break

        it += 1
        if verbose and it % 1000 == 0:
            print(f"    MW it{it}: obj={best_obj:.8f}, eta={eta:.6f}")

    return best_a


def subgradient_momentum(a, time_budget, K=10, verbose=False):
    """Subgradient method with momentum and adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 1.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        w = np.exp(3.0 * (peak_vals - max_pv) / (max_pv + 1e-15))
        w /= w.sum()

        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += wi * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break

        a_new = a - lr * v / v_norm
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            it += 1
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            a = best_a
            if stall % 20 == 0:
                lr *= 0.7
                v *= 0.5
            if lr < 1e-10:
                break

        it += 1
        if verbose and it % 500 == 0:
            print(f"    SG it{it}: obj={best_obj:.8f}, lr={lr:.6f}")

    return best_a


def solve_lp_truncated(a, top_K=50):
    """Solve LP with only top-K constraints for speed."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv = compute_conv(a_norm)
    top_idx = np.argsort(conv)[::-1][:top_K]

    A_red = np.zeros((top_K, n))
    for i, k in enumerate(top_idx):
        k = int(k)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start <= j_end:
            l_start = k - j_end
            l_end = k - j_start
            A_red[i, j_start:j_end + 1] = a_norm[l_start:l_end + 1][::-1]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_red @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def solve_lp_full(a):
    """Solve full LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))

    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def lp_polish(a, time_budget, full=False, verbose=False):
    """LP polishing with adaptive blending."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    last_t = 0.5
    solver_func = solve_lp_full if full else solve_lp_truncated

    while time.monotonic() - start < time_budget:
        g_lp = solver_func(a)
        if g_lp is None:
            break

        candidates = [0.1, 0.2, 0.3, 0.5, 0.7, last_t, min(1.0, last_t * 1.5)]
        candidates = sorted(set([c for c in candidates if 0.01 <= c <= 1.0]))

        best_t_obj = best_obj
        best_t_val = last_t
        best_a_local = best_a.copy()

        for t in candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={last_t:.3f}")
        else:
            # Perturb to escape
            rng = np.random.default_rng(1000 + it)
            scale = 0.01 + 0.005 * min(it, 10)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            if it > 80:
                break

    return best_a


def simulated_annealing(a, time_budget, n_perturb=10, T_init=0.01, T_end=0.001, verbose=False):
    """Simulated annealing: perturb random elements, accept if better (or with prob)."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        T = T_init * (T_end / T_init) ** progress

        # Perturb random elements
        a_try = a.copy()
        idx = rng.integers(0, n, size=n_perturb)
        perturbation = rng.normal(1.0, 0.3, size=n_perturb)
        a_try[idx] *= perturbation
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            it += 1
            continue
        a_try = a_try / s_try

        obj_try = compute_obj(a_try)
        delta = obj_try - best_obj
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-10)):
            a = a_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

        it += 1
        if verbose and it % 2000 == 0:
            print(f"    SA it{it}: obj={best_obj:.8f}, T={T:.6f}")

    return best_a


def resample(a, new_n):
    """Resample sequence via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng):
    """Generate diverse initializations."""
    mode = int(rng.integers(0, 8))
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / (np.sqrt(1 - x ** 2) + 1e-6)
        a = a / a.sum()
    elif mode == 3:
        x = np.arange(n)
        a = np.exp(-((x - n // 2) ** 2) / (2 * (n // 4) ** 2))
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 6.28)
        a = 1.0 + 0.4 * np.sin(2 * np.pi * freq * x + phase)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        a = rng.paretian(rng.uniform(1.2, 3.0), n)
        a = a / a.sum()
    elif mode == 6:
        # Two-peaked
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - n // 4) ** 2) / (2 * (n // 10) ** 2))
        a += np.exp(-((np.arange(n) - 3 * n // 4) ** 2) / (2 * (n // 10) ** 2))
        a = a / a.sum()
    else:
        # Random noise floor
        a = 0.5 + 0.5 * rng.random(n)
        a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: GS init + MW with param sweep + truncated LP + SA + multi-n.
    Time allocation: GS(10%), MW(35%), SG(10%), LP-trunc(15%), SA(8%), multi-n(12%), final(10%).
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Gerchberg-Saxton (10%) ===
    print("\n=== Phase 1: Gerchberg-Saxton ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.10, 100.0)
    if p1_budget > 10:
        # GS on parent
        alloc = min(p1_budget * 0.5, 50.0)
        a_gs = gerchberg_saxton(best_seq, time_budget=alloc, relax=0.8, verbose=True)
        obj_gs = compute_obj(a_gs)
        if obj_gs < best_score:
            best_score = obj_gs
            best_seq = a_gs.copy()
            print(f"  GS parent: obj={best_score:.8f} **")
        else:
            print(f"  GS parent: obj={obj_gs:.8f}")

        # GS on diverse inits
        remaining = deadline - time.monotonic()
        if remaining > 15:
            for init_i in range(2):
                remaining = deadline - time.monotonic()
                if remaining < 10:
                    break
                a_init = diverse_init(n_target, rng)
                alloc = min(remaining * 0.3, 30.0)
                a_gs2 = gerchberg_saxton(a_init, time_budget=alloc, relax=1.0)
                obj_gs2 = compute_obj(a_gs2)
                if obj_gs2 < best_score:
                    best_score = obj_gs2
                    best_seq = a_gs2.copy()
                    print(f"  GS init{init_i}: obj={best_score:.8f} **")
                else:
                    print(f"  GS init{init_i}: obj={obj_gs2:.8f}")

    # === Phase 2: Multiplicative Weights - parameter sweep (35%) ===
    print("\n=== Phase 2: Multiplicative Weights ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 400.0)
    if p2_budget > 20:
        configs = [
            {'eta_init': 0.05, 'K': 5, 'beta': 3.0},
            {'eta_init': 0.1, 'K': 10, 'beta': 5.0},
            {'eta_init': 0.03, 'K': 20, 'beta': 2.0},
            {'eta_init': 0.15, 'K': 3, 'beta': 10.0},
            {'eta_init': 0.08, 'K': 8, 'beta': 4.0},
        ]
        per_cfg = p2_budget / len(configs)
        for cfg_i, cfg in enumerate(configs):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(per_cfg, remaining * 0.6)
            a_mw = multiplicative_weights(best_seq, time_budget=alloc, **cfg, verbose=True)
            obj_mw = compute_obj(a_mw)
            if obj_mw < best_score:
                best_score = obj_mw
                best_seq = a_mw.copy()
                print(f"  MW cfg{cfg_i}({cfg}): obj={best_score:.8f} **")
            else:
                print(f"  MW cfg{cfg_i}({cfg}): obj={obj_mw:.8f}")

    # === Phase 3: Subgradient with momentum (10%) ===
    print("\n=== Phase 3: Subgradient ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.10, 120.0)
    if p3_budget > 10:
        a_sg = subgradient_momentum(best_seq, time_budget=p3_budget, K=10, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  SG: obj={best_score:.8f} **")
        else:
            print(f"  SG: obj={obj_sg:.8f}")

    # === Phase 4: Truncated LP polishing (15%) ===
    print("\n=== Phase 4: Truncated LP ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 180.0)
    if p4_budget > 20:
        # Multiple rounds
        num_rounds = 3
        per_round = p4_budget / num_rounds
        for rnd in range(num_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(per_round, remaining * 0.5)
            a_lp = lp_polish(best_seq, time_budget=alloc, full=False, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP-trunc r{rnd+1}: obj={best_score:.8f} **")
            else:
                print(f"  LP-trunc r{rnd+1}: obj={obj_lp:.8f}")
                if rnd < num_rounds - 1:
                    best_seq = best_seq * np.exp(rng.normal(0, 0.02, len(best_seq)))
                    best_seq = np.maximum(best_seq, 0)
                    best_seq = best_seq / best_seq.sum()

    # === Phase 5: Simulated Annealing (8%) ===
    print("\n=== Phase 5: Simulated Annealing ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.08, 100.0)
    if p5_budget > 10:
        a_sa = simulated_annealing(best_seq, time_budget=p5_budget, n_perturb=15,
                                   T_init=0.005, T_end=0.0005, verbose=True)
        obj_sa = compute_obj(a_sa)
        if obj_sa < best_score:
            best_score = obj_sa
            best_seq = a_sa.copy()
            print(f"  SA: obj={best_score:.8f} **")
        else:
            print(f"  SA: obj={obj_sa:.8f}")

    # === Phase 6: Multi-n Exploration (12%) ===
    print("\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 200),
        n_current + 100,
        n_current + 200,
        n_current + 400,
        600, 700, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500 and abs(n - n_current) > 100]

    for n_try in n_options[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample(best_seq, n_try)

        # Quick GS
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_try = gerchberg_saxton(a_try, time_budget=min(20, remaining * 0.2), relax=0.8)

        # Quick MW
        remaining = deadline - time.monotonic()
        if remaining > 25:
            a_try = multiplicative_weights(a_try, time_budget=min(25, remaining * 0.3),
                                           eta_init=0.1, K=5, beta=5.0)

        # Quick truncated LP
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_try = lp_polish(a_try, time_budget=min(25, remaining * 0.3), full=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final Polish (remaining) ===
    print("\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Full LP if time allows
    if remaining > 40:
        alloc = min(remaining * 0.3, 80.0)
        a_lp = lp_polish(best_seq, time_budget=alloc, full=True, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final full-LP: obj={best_score:.8f} **")

    # Final MW with small eta
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 50.0)
        a_mw = multiplicative_weights(best_seq, time_budget=alloc,
                                       eta_init=0.02, K=5, beta=3.0)
        obj_mw = compute_obj(a_mw)
        if obj_mw < best_score:
            best_score = obj_mw
            best_seq = a_mw.copy()
            print(f"  Final MW: obj={best_score:.8f} **")

    # Final subgradient
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_sg = subgradient_momentum(best_seq, time_budget=min(20, remaining * 0.3), K=5)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # One last truncated LP
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_lp = lp_polish(best_seq, time_budget=min(20, remaining * 0.4), full=False)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
