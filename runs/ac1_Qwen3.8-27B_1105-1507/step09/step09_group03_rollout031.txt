"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses accelerated LP fixed-point with warm starts, adaptive active sets,
Nesterov extrapolation, Nesterov smoothed gradient descent, and multi-scale
exploration. Key innovations: geometric mean blending, dual-based step sizing,
Fourier-mode perturbations for escape, and aggressive multi-n search.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_cols(a):
    """
    Build the matrix C (2n-1 x n) such that C @ g = conv(a, g).
    C[k, i] = a[k - i] with a[j] = 0 for j outside [0, n-1].
    """
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        # C[k, i] = a[k-i] for k where 0 <= k-i <= n-1
        # i.e., i <= k <= i + n - 1
        k_start = i
        k_end = min(2 * n - 2, i + n - 1)
        a_start = 0
        a_end = k_end - i + 1
        C[k_start:k_end + 1, i] = a[a_start:a_end]
    return C


def solve_lp_cross(a, warm_start=None, active_idx=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for k in active_idx (or all),
    sum(g) = 1, g >= 0.
    Returns (g_opt, M_opt).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    C = build_conv_cols(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    if active_idx is not None:
        constraints = [C[active_idx, :] @ g <= M, cp.sum(g) == 1]
    else:
        constraints = [C @ g <= M, cp.sum(g) == 1]
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if warm_start is not None:
        g.value = np.maximum(warm_start, 0)
        g.value = g.value / np.sum(g.value)
        M.value = np.max(C @ g.value)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(np.max(C @ g_val))
                    return g_val, M_val
        except Exception:
            continue
    return None, None


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_lp_iteration(a, time_budget=300.0, verbose=False, momentum=0.5):
    """
    Accelerated LP fixed-point iteration with Nesterov momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    it = 0
    stall = 0
    base_momentum = momentum
    prev_M = None

    while time.monotonic() - start < time_budget:
        # Solve LP
        g_lp, M_lp = solve_lp_cross(a, warm_start=best_a)
        if g_lp is None:
            break

        # Nesterov momentum update
        v = momentum * v + (g_lp - a)
        a_new = project_simplex(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Backtracking
            found = False
            for scale in [0.5, 0.25, 0.1, 0.05]:
                a_try = project_simplex(a + scale * v)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    v = scale * v
                    stall = 0
                    found = True
                    break
            if not found:
                stall += 1
                if stall > 30:
                    # Structured perturbation (Fourier mode)
                    k_mod = 1 + (it % 5)
                    x = np.linspace(0, 2 * np.pi * k_mod, n)
                    perturbation = np.cos(x)
                    scale = 0.05 + 0.02 * min(stall / 10, 1.0)
                    a = best_a * (1 + scale * perturbation)
                    a = np.maximum(a, 0)
                    s_a = float(np.sum(a))
                    if s_a > 1e-10:
                        a = a / s_a
                    v = np.zeros(n)
                    momentum = base_momentum
                    stall = 0
                    if it > 500:
                        break

        # Adaptive momentum
        it += 1
        if stall == 0:
            momentum = min(0.8, base_momentum + 0.002 * it)
        else:
            momentum = max(0.1, momentum * 0.9)

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    NLP it{it}: obj={best_obj:.8f}, M={M_lp:.6f}, mom={momentum:.3f}, t={elapsed:.0f}s")

    return best_a


def smoothed_gradient_fft(a, tau):
    """Compute smoothed gradient using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def nesterov_smoothed_descent(a, time_budget=200.0, tau_init=20.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-9, lr * 0.9999)

        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            beta = max(0.5, beta * 0.95)
            best_it = it

        if verbose and it % 2000 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={time.monotonic()-start:.0f}s")

    return best_a


def golden_section_line_search(a, direction, n_evals=8):
    """Golden section search for best blending parameter."""
    phi = (np.sqrt(5) - 1) / 2
    lo = 0.0
    hi = 1.0

    for _ in range(n_evals):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)

        a1 = (1 - t1) * a + t1 * direction
        a1 = np.maximum(a1, 0)
        s1 = float(np.sum(a1))
        f1 = compute_obj(a1 / s1) if s1 > 1e-10 else np.inf

        a2 = (1 - t2) * a + t2 * direction
        a2 = np.maximum(a2, 0)
        s2 = float(np.sum(a2))
        f2 = compute_obj(a2 / s2) if s2 > 1e-10 else np.inf

        if f1 < f2:
            hi = t2
        else:
            lo = t1

    t_best = (lo + hi) / 2
    a_best = (1 - t_best) * a + t_best * direction
    a_best = np.maximum(a_best, 0)
    s_best = float(np.sum(a_best))
    if s_best > 1e-10:
        a_best = a_best / s_best
    return a_best

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 6, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 6:
        # Triangular
        x = np.linspace(0, 1, n)
        a = 1 - 2 * np.abs(x - 0.5)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 7:
        # Sine-weighted
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses accelerated LP fixed-point with warm starts, adaptive active sets,
Nesterov extrapolation, Nesterov smoothed gradient descent, geometric-mean
blending, and multi-scale exploration. Key innovations: dual-gap-based step
sizing, Fourier-mode perturbations for escape, aggressive multi-n search,
and interleaved LP+GD polishing.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_cols(a):
    """
    Build matrix C (2n-1 x n) such that C @ g = cross_conv(a, g).
    C[k, j] = a[k-j] with a[m]=0 for m outside [0,n-1].
    """
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        a_idx = np.arange(k_start - j, k_end - j + 1)
        C[k_start:k_end + 1, j] = a[a_idx]
    return C


def solve_lp_cross(a, warm_start=None, top_k=None):
    """
    Solve LP: min M s.t. (C @ g)[k] <= M, sum(g)=1, g>=0.
    Optionally only constrain top_k positions.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_s = float(np.sum(a))
    if a_s < 1e-15:
        return None, None
    a_norm = a / a_s
    C = build_conv_cols(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()

    if top_k is not None and top_k < 2 * n - 1:
        # Find top-k positions by current cross-conv with a itself
        conv_a = compute_conv(a_norm)
        top_indices = np.argsort(conv_a)[::-1][:top_k]
        constraints = [C[top_indices, :] @ g <= M, cp.sum(g) == 1]
    else:
        constraints = [C @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        ws = np.maximum(warm_start, 0)
        ws_sum = float(np.sum(ws))
        if ws_sum > 1e-15:
            g.value = ws / ws_sum
            M.value = float(np.max(C @ g.value))

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(np.max(C @ g_val))
                    return g_val, M_val
        except Exception:
            continue
    return None, None


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_lp_iteration(a, time_budget=300.0, verbose=False, momentum=0.5, adaptive_k=True):
    """
    Accelerated LP fixed-point with Nesterov momentum and adaptive active set.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    it = 0
    stall = 0
    base_momentum = momentum
    rng = np.random.default_rng(12345)

    # Adaptive top_k: start with fewer constraints, grow over time
    current_k = min(n, 2 * n - 1) if not adaptive_k else max(n, int(0.5 * (2 * n - 1)))

    while time.monotonic() - start < time_budget:
        # Solve LP (with adaptive active set)
        g_lp, M_lp = solve_lp_cross(a, warm_start=best_a, top_k=current_k if adaptive_k else None)
        if g_lp is None:
            # Fallback: solve without active set restriction
            g_lp, M_lp = solve_lp_cross(a, warm_start=best_a)
            if g_lp is None:
                break

        # Nesterov momentum update
        v = momentum * v + (g_lp - a)
        a_new = project_simplex(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Backtracking
            found = False
            for scale in [0.5, 0.25, 0.1, 0.05, 0.02]:
                a_try = project_simplex(a + scale * v)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    v = scale * v
                    stall = 0
                    found = True
                    break
            if not found:
                stall += 1
                if stall > 25:
                    # Fourier-mode perturbation
                    k_mod = 1 + int(rng.integers(0, 7))
                    x = np.linspace(0, 2 * np.pi * k_mod, n)
                    phase = rng.uniform(0, 2 * np.pi)
                    scale_pert = 0.04 + 0.01 * min(stall / 10, 1.0)
                    a = best_a * (1.0 + scale_pert * np.cos(x + phase))
                    a = np.maximum(a, 0)
                    s_a = float(np.sum(a))
                    if s_a > 1e-10:
                        a = a / s_a
                    else:
                        a = best_a.copy()
                    v = np.zeros(n)
                    momentum = base_momentum
                    stall = 0
                    if it > 800:
                        break

        # Adaptive momentum and active set
        it += 1
        if stall == 0:
            momentum = min(0.85, base_momentum + 0.001 * it)
            # Grow active set as we progress
            if adaptive_k and it % 100 == 0:
                current_k = min(2 * n - 1, current_k + n // 10)
        else:
            momentum = max(0.1, momentum * 0.92)

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    NLP it{it}: obj={best_obj:.8f}, M={M_lp:.6f}, mom={momentum:.3f}, k={current_k}, t={elapsed:.0f}s")

    return best_a


def lp_with_line_search(a, time_budget=120.0, verbose=False):
    """LP with golden-section line search for optimal step sizing."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(54321)

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_cross(a, warm_start=best_a)
        if g_lp is None:
            break

        # Golden section line search
        phi = (np.sqrt(5) - 1) / 2
        lo = 0.0
        hi = 1.0
        for _ in range(7):
            t1 = hi - phi * (hi - lo)
            t2 = lo + phi * (hi - lo)
            a1 = np.maximum((1 - t1) * a + t1 * g_lp, 0)
            s1 = float(np.sum(a1))
            f1 = compute_obj(a1 / s1) if s1 > 1e-10 else np.inf
            a2 = np.maximum((1 - t2) * a + t2 * g_lp, 0)
            s2 = float(np.sum(a2))
            f2 = compute_obj(a2 / s2) if s2 > 1e-10 else np.inf
            if f1 < f2:
                hi = t2
            else:
                lo = t1
        t_best = (lo + hi) / 2
        a_new = np.maximum((1 - t_best) * a + t_best * g_lp, 0)
        s_new = float(np.sum(a_new))
        if s_new <= 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Try geometric mean blending
            with np.errstate(divide='ignore', invalid='ignore'):
                geo = np.where(a > 1e-15, np.sqrt(np.maximum(a, 0) * np.maximum(g_lp, 0)), 0.0)
            geos = float(np.sum(geo))
            if geos > 1e-10:
                geo = geo / geos
                obj_geo = compute_obj(geo)
                if obj_geo < best_obj - 1e-13:
                    best_obj = obj_geo
                    best_a = geo.copy()
                    a = geo
                    stall = 0
                    if verbose:
                        print(f"    LPS: geo blend improved to {best_obj:.8f}")
                    continue

            stall += 1
            if stall > 15:
                k_mod = 1 + int(rng.integers(0, 8))
                x = np.linspace(0, 2 * np.pi * k_mod, n)
                phase = rng.uniform(0, 2 * np.pi)
                scale_pert = 0.03 + 0.01 * min(stall / 10, 1.0)
                a = best_a * (1.0 + scale_pert * np.cos(x + phase))
                a = np.maximum(a, 0)
                s_a = float(np.sum(a))
                if s_a > 1e-10:
                    a = a / s_a
                stall = 0
                if stall > 40:
                    break

        if verbose:
            print(f"    LPS obj={best_obj:.8f}, M={M_lp:.6f}, t={time.monotonic()-start:.0f}s")

    return best_a


def smoothed_gradient_fft(a, tau):
    """Compute smoothed gradient of logsumexp(tau * conv) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def nesterov_smoothed_descent(a, time_budget=200.0, tau_init=20.0, tau_end=600.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    lr = 2.5 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    prev_a = a.copy()
    prev_g = None

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)

        # Barzilai-Borstep style adaptive step size
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.4 * lr + 0.6 * min(max(bb_lr, 1e-9), 8.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.99995)

        if it - best_it > 8000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.5 / np.sqrt(n) * 0.4
            beta = max(0.5, beta * 0.93)
            best_it = it

        if verbose and it % 3000 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={time.monotonic()-start:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * (w + 1) ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6) + 0.1)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 6, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = 1 - 2 * np.abs(x - 0.5)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 7:
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.01
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def coordinate_polish(a, time_budget=40.0, verbose=False):
    """
    Targeted coordinate descent: find the peak of conv, then try to reduce it
    by adjusting the specific coefficients that contribute most to it.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(77777)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_k = int(np.argmax(conv))

        # Find contributing indices for position max_k
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)

        contribs = []
        for i in range(i_start, i_end + 1):
            j = max_k - i
            if 0 <= j < n:
                contribs.append((a[i] * a[j], i, j))

        if not contribs:
            break

        contribs.sort(reverse=True)
        improved = False

        for contrib_val, src, dst in contribs[:5]:
            if src == dst:
                # Reduce this peak by reducing a[src]
                for frac in [0.05, 0.02, 0.01, 0.005, 0.002]:
                    delta = frac * a[src]
                    if delta >= a[src] * 0.5:
                        continue
                    # Redistribute to random positions away from the peak
                    a_try = a.copy()
                    a_try[src] -= delta
                    # Spread to positions that don't contribute to this peak
                    for _ in range(3):
                        tgt = int(rng.integers(0, n))
                        # Check if tgt would contribute to the peak
                        j_check = max_k - tgt
                        if 0 <= j_check < n:
                            # Would still contribute, pick another
                            tgt2 = int(rng.integers(0, n))
                            if not (0 <= max_k - tgt2 < n):
                                tgt = tgt2
                    a_try[tgt] += delta
                    if a_try[src] < -1e-15:
                        continue
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
            else:
                # Try to reduce the product a[src]*a[dst] by moving mass
                for frac in [0.03, 0.01, 0.005]:
                    delta = frac * min(a[src], a[dst])
                    if delta < 1e-15:
                        continue
                    a_try = a.copy()
                    a_try[src] -= delta * 0.5
                    a_try[dst] -= delta * 0.5
                    if a_try[src] < 0 or a_try[dst] < 0:
                        continue
                    # Add to positions far from peak
                    tgt1 = int(rng.integers(0, max(1, max_k - n + 1) if max_k >= n else n))
                    tgt2 = int(rng.integers(0, n))
                    a_try[tgt1 % n] += delta * 0.5
                    a_try[tgt2 % n] += delta * 0.5
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                    if obj_try < best_obj:
                        # Small improvement, accept anyway
                        pass
            if improved:
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization combining accelerated LP, smoothed GD,
    line-search LP, coordinate polish, multi-n exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent available: {e}")

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Nesterov LP (main workhorse) ===
    print(f"\n=== Phase 1: Nesterov LP ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 30:
        a1 = nesterov_lp_iteration(best_seq, time_budget=p1_budget, momentum=0.5, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  NLP: obj={best_score:.8f} **")
        else:
            print(f"  NLP: obj={obj1:.8f}")

    # === Phase 2: LP with line search ===
    print(f"\n=== Phase 2: LP + Line Search ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 200.0)
    if p2_budget > 30:
        a2 = lp_with_line_search(best_seq, time_budget=p2_budget, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  LP+LS: obj={best_score:.8f} **")
        else:
            print(f"  LP+LS: obj={obj2:.8f}")

    # === Phase 3: Smoothed GD ===
    print(f"\n=== Phase 3: Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 180.0)
    if p3_budget > 30:
        a3 = nesterov_smoothed_descent(best_seq, time_budget=p3_budget,
                                       tau_init=15.0, tau_end=500.0, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  GD: obj={best_score:.8f} **")
        else:
            print(f"  GD: obj={obj3:.8f}")

    # === Phase 4: Coordinate polish ===
    print(f"\n=== Phase 4: Coordinate Polish ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.08, 50.0)
    if p4_budget > 10:
        a4 = coordinate_polish(best_seq, time_budget=p4_budget, verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj4:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = [300, 400, 500, 600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600, 1800]
    n_options = [n for n in n_options if abs(n - n_current) > 30 and 200 <= n <= 2000]
    n_options_sorted = sorted(n_options, key=lambda n: abs(n - n_current))

    for n_try in n_options_sorted[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 70:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc_lp = min(remaining * 0.12, 45.0)
        alloc_gd = min(remaining * 0.08, 25.0)

        if alloc_lp > 12:
            a_try = nesterov_lp_iteration(a_try, time_budget=alloc_lp, momentum=0.4)

        remaining = deadline - time.monotonic()
        if alloc_gd > 10 and remaining > 25:
            a_try = nesterov_smoothed_descent(a_try, time_budget=alloc_gd,
                                              tau_init=30.0, tau_end=400.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = r % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.25 + 0.05 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r < 1e-10:
            continue
        a_r = a_r / s_r

        alloc_total = min(remaining * 0.18, 70.0)
        if alloc_total > 20:
            alloc_lp = alloc_total * 0.55
            alloc_gd = alloc_total * 0.4
            a_r = nesterov_lp_iteration(a_r, time_budget=alloc_lp, momentum=0.5)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_r = nesterov_smoothed_descent(a_r, time_budget=min(alloc_gd, remaining * 0.4),
                                                tau_init=40.0, tau_end=400.0)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 35:
        alloc = min(remaining * 0.3, 80.0)
        a_ts = lp_with_line_search(best_seq, time_budget=alloc, verbose=True)
        obj_ts = compute_obj(a_ts)
        if obj_ts < best_score:
            best_score = obj_ts
            best_seq = a_ts.copy()
            print(f"  Final LP+LS: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.3, 50.0)
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                          tau_init=80.0, tau_end=1200.0)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 40.0)
        a_alp = nesterov_lp_iteration(best_seq, time_budget=alloc, momentum=0.6)
        obj_alp = compute_obj(a_alp)
        if obj_alp < best_score:
            best_score = obj_alp
            best_seq = a_alp.copy()
            print(f"  Final NLP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
