def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""

                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

Strategy: LP-centric approach with subgradient using LP dual information,
adaptive blending, spectral initialization, and multi-n exploration.
Key innovation: uses the LP optimal solution g* in subgradient computation
(instead of current a), enabling more principled descent directions.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_cross_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP, cp.CBC, cp.GLPK]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def get_active_set(a, g, M, tolerance=1e-6):
    """Get the active set: indices k where conv(a,g)[k] is close to M."""
    conv_ag = compute_cross_conv(a, g)
    active = np.where(conv_ag >= M - tolerance * M)[0]
    return active


def subgradient_from_lp(a, g_lp, M_lp, K=30):
    """
    Compute subgradient of M*(a) using the LP solution g*.
    The subgradient is the average of g* shifted by active peak indices.
    """
    n = len(a)
    conv_ag = compute_cross_conv(a, g_lp)
    # Get top-K peaks
    top_idx = np.argsort(conv_ag)[::-1][:K]
    peak_vals = conv_ag[top_idx]
    max_pv = peak_vals[0]
    # Weight by proximity to max
    spread = max_pv - peak_vals[-1]
    if spread < 1e-15:
        weights = np.ones(K) / K
    else:
        weights = np.exp(-3.0 * (max_pv - peak_vals) / spread)
        weights /= weights.sum()

    # Subgradient: g_i = sum_k w_k * g*[k-i] (with bounds)
    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        # (a * g_lp)[k] = sum_i a_i * g_lp[k-i]
        # d/d a_i of (a*g_lp)[k] = g_lp[k-i] if 0 <= k-i < n
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            g[i_start:i_end + 1] += w * g_lp[j_start:j_end + 1][::-1]

    return g


def spectral_init(n, mode=0, rng=None):
    """Generate spectral-based initializations."""
    if rng is None:
        rng = np.random.default_rng(0)

    if mode == 0:
        # Uniform
        return np.ones(n) / n
    elif mode == 1:
        # Arcsine (flat top in frequency)
        x = np.linspace(-0.999, 0.999, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-8)
        return a / a.sum()
    elif mode == 2:
        # Bounded FFT: start uniform, flatten spectrum
        a = np.ones(n) / n
        A = fft(a, n)
        # Compress dynamic range
        A_mag = np.abs(A)
        A_phase = np.angle(A)
        A_flat = A_mag ** 0.3 * np.exp(1j * A_phase)
        a_new = ifft(A_flat).real
        a_new = np.maximum(a_new, 0)
        return a_new / a_new.sum()
    elif mode == 3:
        # Random smooth
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.arange(n) / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 4:
        # Two-lobe
        c1, c2 = n // 4, 3 * n // 4
        w = n // 10
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1.0, n)
        a = x ** rng.uniform(-1.5, 1.5)
        return a / a.sum()
    else:
        return np.ones(n) / n


def lp_polish_core(a, time_budget, verbose=False, seed=42):
    """
    Core LP polishing: repeatedly solve LP, blend with current,
    and use subgradient steps for escape.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    iter_count = 0
    consecutive_fails = 0
    last_improve_iter = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed

        # Solve LP
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        iter_count += 1

        # Strategy: alternate between blend and subgradient
        if iter_count % 3 == 0:
            # Subgradient step using LP solution
            subgrad = subgradient_from_lp(a, g_lp, M_lp, K=20)
            sg_norm = np.linalg.norm(subgrad)
            if sg_norm > 1e-15:
                # Adaptive learning rate
                lr = 0.05 / (sg_norm * np.sqrt(n))
                vg = a - lr * subgrad
                vg = project_simplex(vg)
                obj_vg = compute_obj(vg)
                if obj_vg < best_obj - 1e-14:
                    best_obj = obj_vg
                    best_a = vg.copy()
                    a = vg
                    consecutive_fails = 0
                    last_improve_iter = iter_count
                    if verbose and iter_count % 10 == 0:
                        print(f"    LP it{iter_count}: obj={best_obj:.8f} (subgrad), M={M_lp:.6f}")
                else:
                    # Try blend
                    pass
            else:
                consecutive_fails += 1
        else:
            # Blend step
            best_blend_obj = best_obj
            best_blend_a = best_a

            # Adaptive candidates: biased toward larger t when we're improving
            if consecutive_fails == 0:
                candidates = [0.5, 0.7, 0.8, 0.9, 1.0]
            else:
                candidates = [0.3, 0.5, 0.7, 0.9, 1.0, 0.4, 0.6, 0.8]

            for t in candidates:
                a_try = (1.0 - t) * a + t * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_blend_obj:
                    best_blend_obj = obj_try
                    best_blend_a = a_try

            if best_blend_obj < best_obj - 1e-14:
                best_obj = best_blend_obj
                best_a = best_blend_a.copy()
                a = best_a
                consecutive_fails = 0
                last_improve_iter = iter_count
                if verbose and iter_count % 10 == 0:
                    print(f"    LP it{iter_count}: obj={best_obj:.8f} (blend), M={M_lp:.6f}")
            else:
                consecutive_fails += 1

        # Stuck detection and perturbation
        if consecutive_fails > 8:
            scale = min(0.05, 0.005 * consecutive_fails)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            consecutive_fails = 0

        if consecutive_fails > 25:
            # Bigger jump
            scale = 0.03
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            consecutive_fails = 0

        # Check time for next LP
        if time.monotonic() - start > time_budget - 5:
            break

    return best_a


def fast_subgradient_polish(a, time_budget, K=15, verbose=False):
    """Fast subgradient polishing using top-K peaks of self-convolution."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.02
    momentum = 0.8
    v = np.zeros(n)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]

        if max_pv < 1e-15:
            break

        spread = max_pv - peak_vals[-1]
        if spread < 1e-15:
            weights = np.ones(K) / K
        else:
            weights = np.exp(-5.0 * (max_pv - peak_vals) / spread)
            weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break

        a_new = a - lr * v / v_norm
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.05, 0.1)
        else:
            stall += 1
            if stall % 50 == 0:
                lr *= 0.7
                v *= 0.5
            if stall > 500:
                break

        if verbose:
            elapsed = time.monotonic() - start
            if int(elapsed) % 10 == 0 and int(elapsed) != getattr(fast_subgradient_polish, '_last_print', -1):
                fast_subgradient_polish._last_print = int(elapsed)
                print(f"    SG t={elapsed:.0f}s: obj={best_obj:.8f}, lr={lr:.4f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP-centric with subgradient hybrid, spectral init, multi-n.
    Time allocation: init (5%), LP polish (65%), subgrad (20%), multi-n + final (10%).
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to use parent as starting point
    n_parent = 0
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_parent = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        n_parent = 800
        print(f"Default init (n=800): obj={best_score:.8f}")

    # === Phase 1: Multi-initialization exploration (5%) ===
    print(f"\n=== Phase 1: Initialization Exploration ===")
    p1_budget = min((deadline - time.monotonic()) * 0.05, 75.0)
    if p1_budget > 20:
        n_base = n_parent if n_parent > 0 else 800
        n_tries = [n_base]
        # Add some nearby n values
        for dn in [-100, -50, 50, 100, 150]:
            nt = n_base + dn
            if 400 <= nt <= 1200 and nt not in n_tries:
                n_tries.append(nt)

        for n_try in n_tries[:5]:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            per_init = min(p1_budget / max(1, len(n_tries) - 1), remaining * 0.3)
            if per_init < 10:
                break

            a_try = None
            # Try a few initializations
            for mode in [0, 1, 2, 3]:
                a_try = spectral_init(n_try, mode=mode, rng=rng)
                # Quick LP polish (short)
                a_try = lp_polish_core(a_try, time_budget=min(per_init / 2, 15.0), verbose=False)
                obj_try = compute_obj(a_try)
                if obj_try < best_score:
                    best_score = obj_try
                    best_seq = a_try.copy()
                    print(f"  Init n={n_try}, mode={mode}: obj={best_score:.8f} **")
                break  # Just use mode 0 (uniform) for speed in phase 1

            # Also try resampling the current best
            a_resample = resample_sequence(best_seq, n_try)
            a_resample = lp_polish_core(a_resample, time_budget=min(per_init, 20.0), verbose=False)
            obj_resample = compute_obj(a_resample)
            if obj_resample < best_score:
                best_score = obj_resample
                best_seq = a_resample.copy()
                print(f"  Resample n={n_try}: obj={best_score:.8f} **")

    # === Phase 2: Intensive LP Polishing (65%) ===
    print(f"\n=== Phase 2: Intensive LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.65

    # Split into rounds with perturbation
    n_rounds = 6
    per_round = p2_budget / n_rounds
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_round, remaining * 0.6)
        if alloc < 15:
            break

        # Use different seeds and turbulence levels for each round
        seed_rnd = seed + rnd * 100
        a_lp = lp_polish_core(best_seq, time_budget=alloc, verbose=(rnd < 2), seed=seed_rnd)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score - 1e-12:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP round {rnd+1}/{n_rounds}: obj={best_score:.8f} **")
        else:
            print(f"  LP round {rnd+1}/{n_rounds}: obj={obj_lp:.8f} (no improvement)")
            # Perturb for next round
            if rnd < n_rounds - 1:
                scale = 0.01 + 0.005 * rnd
                best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                best_seq = np.maximum(best_seq, 0)
                best_seq = best_seq / best_seq.sum()

    # === Phase 3: Subgradient Polish (20%) ===
    print(f"\n=== Phase 3: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.25
    if p3_budget > 15:
        a_sg = fast_subgradient_polish(best_seq, time_budget=p3_budget, K=15, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score - 1e-12:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj_sg:.8f}")

    # === Phase 4: Final Intensive LP + Multi-n (15%) ===
    print(f"\n=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()

    # One more LP round
    if remaining > 30:
        alloc = min(remaining * 0.4, 120.0)
        a_lp = lp_polish_core(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score - 1e-12:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_lp:.8f}")

    # Try a few nearby n values
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    n_options = []
    for dn in [-50, 50, -100, 100, 150]:
        nt = n_current + dn
        if 400 <= nt <= 1200:
            n_options.append(nt)

    for n_try in n_options[:3]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.2, 40.0)
        if alloc > 10:
            a_try = lp_polish_core(a_try, time_budget=alloc, verbose=False)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # Final subgradient on best
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_sg = fast_subgradient_polish(best_seq, time_budget=min(remaining * 0.5, 40.0), K=10)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
