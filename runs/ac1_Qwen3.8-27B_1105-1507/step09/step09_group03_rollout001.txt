    return irfft(fa * fg, size)[:2 * n - 1]


def quadratic_line_search(a, g, f0):
    """Find best t in [0,1] for (1-t)*a + t*g using 3-point quadratic fit."""
    a_mid = 0.5 * a + 0.5 * g
    a_mid = np.maximum(a_mid, 0)
    s_mid = float(np.sum(a_mid))
    if s_mid < 1e-10:
        f_mid = np.inf
    else:
        f_mid = compute_obj(a_mid / s_mid)

    f_end = compute_obj(g)

    if f_end < f0 and f_end <= f_mid:
        t = 0.85
    elif f_mid < f0 and f_mid < f_end:
        denom = f0 - 2.0 * f_mid + f_end
        if abs(denom) < 1e-15:
            t = 0.5
        else:
            t = 0.5 * (f0 - f_end) / denom
            t = max(0.05, min(0.95, t))
    elif f0 <= f_mid and f0 <= f_end:
        return a, f0, 0.0
    else:
        t = 0.3 if f_mid < f_end else 0.7

    a_new = (1.0 - t) * a + t * g
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        return a, f0, 0.0
    a_new = a_new / s_new
    f_new = compute_obj(a_new)
    return a_new, f_new, t


def frank_wolfe_lp(a, time_budget, verbose=False):
    """LP-based Frank-Wolfe with two-stage direction and quadratic line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        # Stage 1: min-max LP
        g1, M1 = solve_lp_minmax(a)
        if g1 is None:
            break

        # Find active set
        conv_ag = conv_ag(a, g1)
        max_val = float(np.max(conv_ag))
        tol = 0.02 * max_val
        active = np.where(conv_ag >= max_val - tol)[0]

        # Stage 2: min-sum at active
        if len(active) < (2 * n - 1):
            g2 = solve_lp_minsum_active(a, active)
            if g2 is not None:
                # Compare
                peak_g1 = float(np.max(conv_ag))
                conv_ag2 = conv_ag(a, g2)
                peak_g2 = float(np.max(conv_ag2))
                if peak_g2 < peak_g1 * 1.01:
                    g_dir = 0.5 * g1 + 0.5 * g2
                else:
                    g_dir = g1
            else:
                g_dir = g1
        else:
            g_dir = g1

        # Line search
        a_new, f_new, t_used = quadratic_line_search(a, g_dir, best_obj)

        it += 1
        if f_new < best_obj - 1e-13:
            best_obj = f_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    FW-LP it{it}: obj={best_obj:.8f}, M1={M1:.6f}, t={t_used:.3f}, |S|={len(active)}")
        else:
            stall += 1
            # Perturb to escape
            scale = 0.01 + 0.005 * min(stall, 20)
            a = best_a * np.exp(rng.normal(0, min(scale, 0.15), n))
            a = np.maximum(a, 0)
            s_a = float(np.sum(a))
            if s_a > 1e-10:
                a = a / s_a
            else:
                a = best_a.copy()
            if stall > 60:
                break

    return best_a


def smoothed_gradient_fft(a, tau):
    """Compute gradient of softmax-smoothed max of conv(a,a) components."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    # d/d a_i of sum_k p_k * conv(a,a)[k]
    # conv(a,a)[k] = sum_j a[j]*a[k-j]
    # d/da_i [k] = a[k-i] + a[i-k] (if valid)
    # So grad_i = sum_k p_k * (a[k-i] + a[i-k])
    # = 2 * sum_k p_k * a[k-i] (by symmetry)
    # = 2 * (p * a)[reversed] ... let me just do it via FFT

    N = 1
    while N < 4 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    # Cross-correlation: ifft(conj(P) * A) gives (p * a) in linear sense
    # Actually we need: grad_i = 2 * sum_k p_k * conv(a,a gradient wrt a_i at k)
    # = 2 * sum_k p_k * (a[k-i] + a[i-k]) for valid indices
    # This is 2 * (circular_crosscorr)(p, a)[i] approximately
    cross = ifft(np.conj(P) * A).real
    # But we need linear, not circular. For our purposes, if N is large enough,
    # the circular and linear are the same for the first n elements.
    grad = 2.0 * cross[:n]
    return grad, conv


def nesterov_smoothed_descent(a, time_budget, tau_init=30.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        if verbose and it % 500 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)

        # Normalized gradient
        g_norm = float(np.linalg.norm(g))
        if g_norm > 1e-15:
            g = g / g_norm * 2.0 * n  # Scale to match objective scale

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9995)

        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.7
            best_it = it

    return best_a


def peak_shaving(a, time_budget, verbose=False):
    """Directly attack the top peak by reducing large contributors."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_k = int(np.argmax(conv))

        # Find top contributors to peak at max_k
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)

        contribs = []
        for i in range(i_start, i_end + 1):
            j = max_k - i
            if 0 <= j < n:
                contribs.append((a[i] * a[j], i, j))

        if not contribs:
            break
        contribs.sort(reverse=True)

        improved = False
        for contrib_val, src, dst_contrib in contribs[:5]:
            if src == dst_contrib:
                continue
            # Try reducing src and increasing a random other position
            for delta_frac in [0.03, 0.015, 0.008]:
                delta = delta_frac * a[src]
                if delta <= 0:
                    continue
                # Target: spread to positions far from src
                targets = [rng.integers(0, n), rng.integers(0, n),
                           (src + n // 3) % n, (src + 2 * n // 3) % n]
                for tgt in targets:
                    tgt = int(tgt)
                    if tgt == src:
                        continue
                    a_try = a.copy()
                    a_try[src] -= delta
                    if a_try[src] < 0:
                        continue
                    a_try[tgt] += delta
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 80:
                # Perturb
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if it > 500:
                    break

        if verbose and it % 20 == 0:
            print(f"    PS it{it}: obj={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        # Sum of sinusoids
        freqs = rng.integers(1, 6, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Comb pattern
        period = max(2, n // rng.integers(5, 15))
        a = np.zeros(n)
        for i in range(0, n, period):
            a[i] = 1.0
        a = a + 0.1 * rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 7:
        # Sharp peaks
        n_peaks = rng.integers(3, 8)
        a = 0.01 * np.ones(n)
        peak_positions = rng.choice(n, size=n_peaks, replace=False)
        for p in peak_positions:
            a[p] = 1.0
        a = a / a.sum()
    else:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    return a


def get_parent_sequence():
    """Try to load parent sequence."""
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                return parent
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization combining LP Frank-Wolfe, smoothed GD,
    peak shaving, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    parent = get_parent_sequence()
    if parent is not None:
        best_seq = parent.copy()
        best_score = compute_obj(best_seq)
        print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    else:
        n_init = 600
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: LP-based Frank-Wolfe ===
    print(f"\n=== Phase 1: LP Frank-Wolfe ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 450.0)
    if p1_budget > 30:
        a_fw = frank_wolfe_lp(best_seq, time_budget=p1_budget, verbose=True)
        obj_fw = compute_obj(a_fw)
        if obj_fw < best_score:
            best_score = obj_fw
            best_seq = a_fw.copy()
            print(f"  FW-LP: obj={best_score:.8f} **")
        else:
            print(f"  FW-LP: obj={obj_fw:.8f}")

    # === Phase 2: Smoothed GD ===
    print(f"\n=== Phase 2: Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 200.0)
    if p2_budget > 30:
        a_gd = nesterov_smoothed_descent(best_seq, time_budget=p2_budget,
                                         tau_init=30.0, tau_end=600.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  GD: obj={best_score:.8f} **")
        else:
            print(f"  GD: obj={obj_gd:.8f}")

    # === Phase 3: Peak Shaving ===
    print(f"\n=== Phase 3: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 120.0)
    if p3_budget > 20:
        a_ps = peak_shaving(best_seq, time_budget=p3_budget, verbose=True)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  PS: obj={best_score:.8f} **")
        else:
            print(f"  PS: obj={obj_ps:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Generate candidate n values
    n_candidates = set()
    for offset in [-200, -100, -50, 50, 100, 200, 300, 400]:
        n_try = n_current + offset
        if 200 <= n_try <= 2000:
            n_candidates.add(n_try)
    # Add some absolute values
    for n_abs in [300, 400, 500, 700, 900, 1100, 1300]:
        if n_abs != n_current:
            n_candidates.add(n_abs)

    n_sorted = sorted(n_candidates)
    # Limit to 5 best candidates (close to current n first)
    n_sorted.sort(key=lambda x: abs(x - n_current))
    n_to_try = n_sorted[:5]

    for n_try in n_to_try:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick optimization: short FW + GD
        alloc = min(remaining * 0.2, 80.0)
        if alloc > 20:
            half = alloc / 2
            a_try = frank_wolfe_lp(a_try, time_budget=half)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_try = nesterov_smoothed_descent(a_try, time_budget=min(remaining * 0.3, 40.0),
                                                  tau_init=50.0, tau_end=400.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        mode = r % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with current best
        blend = 0.4
        a_r = (1.0 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.18, 60.0)
        if alloc > 15:
            third = alloc / 3
            a_r = frank_wolfe_lp(a_r, time_budget=third)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_r = nesterov_smoothed_descent(a_r, time_budget=min(remaining * 0.3, 30.0),
                                                tau_init=50.0, tau_end=500.0)
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_r = peak_shaving(a_r, time_budget=min(remaining * 0.3, 15.0))

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 40:
        alloc = min(remaining * 0.35, 120.0)
        a_final = frank_wolfe_lp(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final FW: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 80.0)
        a_final = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                            tau_init=100.0, tau_end=1200.0)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 50.0)
        a_final = peak_shaving(best_seq, time_budget=alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final PS: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 40.0)
        a_final = frank_wolfe_lp(best_seq, time_budget=alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final FW2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
