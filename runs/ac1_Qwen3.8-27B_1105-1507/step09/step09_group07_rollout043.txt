def softmax_from_z(z, n):
    z = np.asarray(z, dtype=np.float64)
    if n <= 1:
        return np.array([1.0])
    u = np.empty(n)
    u[0]=0.0
    zpart = z[:n-1]
    # sanitize
    zpart = np.where(np.isfinite(zpart), zpart, 0.0)
    u[1:]=zpart
    m = float(np.max(u))
    e = np.exp(u - m)
    e[~np.isfinite(e)] = 0.0
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0/n)
    a = e / s
    a = np.maximum(a, 0.0)
    ss = a.sum()
    if ss <= 0: return np.full(n,1.0/n)
    return a/ss
```

"""
Search for low-peak nonnegative autocorrelation sequences.

Combines:
- diverse analytic/random seed generation,
- fast KKT/active-set alternating heuristics,
- smooth log-sum-exp L-BFGS optimization on a softmax parameterization,
- exact LP alternating refinement,
- length sweep and final local polishing.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
from operator import itemgetter


class OptTimeout(Exception):
    pass


_LBFGS_STATE = {}


def _next_pow2(m):
    size = 1
    while size < m:
        size <<= 1
    return size


def normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-15:
        return np.full(a.size, 1.0 / a.size, dtype=np.float64)
    return a / s


def compute_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = _next_pow2(2 * n - 1)
    fa = rfft(a, size)
    c = irfft(fa * fa, size)[:2 * n - 1]
    return c


def compute_cross_conv(a, b):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n, m = a.size, b.size
    if n == 0 or m == 0:
        return np.array([], dtype=np.float64)
    size = _next_pow2(n + m - 1)
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def objective(a):
    a = normalize(a)
    n = a.size
    if n == 0:
        return float("inf")
    c = compute_conv(a)
    if c.size == 0:
        return float("inf")
    m = float(np.max(c))
    if m < 0.0 and m > -1e-15:
        m = 0.0
    return float(2.0 * n * m)


def resample_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    if new_n <= 0:
        return np.array([], dtype=np.float64)
    n = a.size
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    b = np.interp(x_new, x_old, a)
    return np.maximum(b, 0.0)


def softmax_from_z(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if n <= 1:
        return np.array([1.0], dtype=np.float64)
    u = np.empty(n, dtype=np.float64)
    u[0] = 0.0
    zp = np.where(np.isfinite(z[:n - 1]), z[:n - 1], 0.0)
    u[1:] = zp
    m = float(np.max(u))
    e = np.exp(u - m)
    e[~np.isfinite(e)] = 0.0
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    a = e / s
    a = np.maximum(a, 0.0)
    ss = float(np.sum(a))
    if ss <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / ss


def z_from_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n <= 1:
        return np.array([], dtype=np.float64)
    eps = 1e-12
    u = np.log(np.maximum(a, eps))
    z = u[1:] - u[0]
    z = np.where(np.isfinite(z), z, 0.0)
    return np.clip(z, -40.0, 40.0)


def softmax_weights(values, beta):
    v = np.asarray(values, dtype=np.float64).ravel()
    if v.size == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    m = float(np.max(v))
    tmp = float(beta) * (v - m)
    tmp = np.clip(tmp, -700.0, 0.0)
    e = np.exp(tmp)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.full(v.size, 1.0 / v.size, dtype=np.float64)
    w = e / s
    return np.where(np.isfinite(w), w, 0.0)


def convolution_grad_from_weights(a, w):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    w = np.asarray(w, dtype=np.float64).ravel()
    if w.size != L:
        tmp = np.zeros(L, dtype=np.float64)
        m = min(w.size, L)
        tmp[:m] = w[:m]
        w = tmp
    w2 = np.zeros(2 * n, dtype=np.float64)
    w2[:L] = w
    a_rev = a[::-1].copy()
    size = _next_pow2(3 * n - 1)
    fw = rfft(w2, size)
    fr = rfft(a_rev, size)
    out = irfft(fw * fr, size)[n - 1:n - 1 + n]
    return np.where(np.isfinite(out), out, 0.0)


def surrogate_value_grad_z(z, beta, n):
    a = softmax_from_z(z, n)
    c = compute_conv(a)
    y = float(n) * c
    if y.size == 0:
        return 0.0, np.zeros(max(0, n - 1), dtype=np.float64)
    y_max = float(np.max(y))
    if y_max < 0.0:
        y_max = 0.0
    tmp = float(beta) * (y - y_max)
    tmp = np.clip(tmp, -700.0, 0.0)
    e = np.exp(tmp)
    e_sum = float(np.sum(e))
    if not np.isfinite(e_sum) or e_sum <= 0.0:
        e = np.full(y.size, 1.0 / y.size, dtype=np.float64)
        e_sum = float(y.size)
    w = e / e_sum
    val = (y_max + float(np.log(e_sum))) / float(beta)
    obj_exact = float(2.0 * y_max)

    if _LBFGS_STATE.get("active", False):
        if obj_exact < _LBFGS_STATE.get("best_obj", np.inf):
            _LBFGS_STATE["best_obj"] = obj_exact
            _LBFGS_STATE["best_a"] = a.copy()

    h = convolution_grad_from_weights(a, w)
    grad_a = float(n) * (2.0 * h)
    dot = float(np.dot(a, grad_a))
    grad_z = a[1:] * (grad_a[1:] - dot)
    grad_z = np.where(np.isfinite(grad_z), grad_z, 0.0)
    return float(val), grad_z


def _lbfgs_fun(x):
    if not _LBFGS_STATE.get("active", False):
        return 0.0, np.zeros(0, dtype=np.float64)
    if time.monotonic() > _LBFGS_STATE.get("deadline", 0.0):
        raise OptTimeout()
    z = np.asarray(x, dtype=np.float64).ravel()
    n = int(_LBFGS_STATE["n"])
    beta = float(_LBFGS_STATE["beta"])
    return surrogate_value_grad_z(z, beta, n)


def lbfgs_optimize(a, time_budget, betas=None, verbose=False):
    a = normalize(a)
    n = a.size
    if time_budget is None or time_budget <= 0.0 or n <= 1:
        return a

    if betas is None:
        if time_budget > 120.0:
            betas = (10.0, 50.0, 200.0, 800.0)
        elif time_budget > 45.0:
            betas = (20.0, 80.0, 300.0)
        else:
            betas = (50.0, 200.0)

    betas = tuple(float(b) for b in betas)
    current = a.copy()
    current_obj = objective(current)
    start = time.monotonic()
    alloc = float(time_budget) / max(1, len(betas))

    for beta in betas:
        stage_deadline = min(start + float(time_budget), time.monotonic() + alloc)
        if time.monotonic() >= stage_deadline:
            break

        _LBFGS_STATE.update(
            {
                "active": True,
                "n": n,
                "beta": float(beta),
                "deadline": stage_deadline,
                "best_obj": current_obj,
                "best_a": current.copy(),
            }
        )

        z0 = z_from_a(current)
        z_end = None
        try:
            res = minimize(
                _lbfgs_fun,
                z0,
                jac=True,
                method="L-BFGS-B",
                options={
                    "maxiter": 20000,
                    "maxfun": 50000,
                    "gtol": 1e-9,
                    "ftol": 1e-15,
                    "maxls": 20,
                    "maxcor": 20,
                },
            )
            if hasattr(res, "x") and res.x is not None:
                z_end = res.x
        except OptTimeout:
            z_end = None
        except Exception:
            z_end = None

        if z_end is not None:
            a_end = softmax_from_z(z_end, n)
            obj_end = objective(a_end)
            if obj_end < current_obj:
                current = a_end
                current_obj = obj_end

        a_state = _LBFGS_STATE.get("best_a")
        obj_state = _LBFGS_STATE.get("best_obj", np.inf)
        if a_state is not None and obj_state < current_obj:
            current = normalize(a_state)
            current_obj = objective(current)

        if verbose:
            print(f"      LBFGS beta={beta:.0f}: obj={current_obj:.8f}")

    _LBFGS_STATE["active"] = False
    return current


def build_cross_dense(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    a = normalize(a)
    n = a.size
    if n <= 1:
        return a, 0.0

    A = build_cross_dense(a)
    A_ub = np.zeros((2 * n - 1, n + 1), dtype=np.float64)
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0

    b_ub = np.zeros(2 * n - 1, dtype=np.float64)
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    bounds = [(0.0, None)] * n + [(None, None)]

    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={
                "presolve": True,
                "dual_feasibility_tolerance": 1e-8,
                "primal_feasibility_tolerance": 1e-8,
            },
        )
        if res is not None and res.status == 0 and res.x is not None:
            g = np.maximum(res.x[:n], 0.0)
            s = float(np.sum(g))
            if s > 1e-12:
                g = g / s
                cross = compute_cross_conv(a, g)
                M = float(np.max(cross))
                return g, M
    except Exception:
        pass

    return None, None


def solve_subproblem_mirror(a, time_budget=1.0, beta=100.0):
    a = normalize(a)
    n = a.size
    if n <= 1:
        return a

    g = normalize(a)
    cross = compute_cross_conv(a, g)
    current_f = float(np.max(cross))
    best_f = current_f
    best_g = g.copy()

    log_g = np.log(np.maximum(g, 1e-12))
    step_scale = 0.3
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        y = float(n) * cross
        w = softmax_weights(y, beta)
        h = convolution_grad_from_weights(a, w)
        maxh = float(np.max(np.abs(h))) if h.size else 0.0
        if maxh <= 0.0:
            break

        base_step = step_scale / maxh
        old_log = log_g.copy()
        old_f = current_f
        accepted = False

        for factor in (1.0, 0.5, 0.25, 0.1):
            log_try = old_log - factor * base_step * h
            log_try -= float(np.max(log_try))
            log_try = np.clip(log_try, -60.0, 60.0)
            g_try = np.exp(log_try)
            s = float(np.sum(g_try))
            if s <= 1e-15:
                continue
            g_try /= s
            cross_try = compute_cross_conv(a, g_try)
            f_try = float(np.max(cross_try))
            if f_try <= old_f * (1.0 + 1e-10) + 1e-15:
                log_g = log_try
                g = g_try
                cross = cross_try
                current_f = f_try
                accepted = True
                if f_try < best_f:
                    best_f = f_try
                    best_g = g_try.copy()
                break

        if not accepted:
            step_scale *= 0.5
            if step_scale < 1e-6:
                break
        else:
            step_scale = min(1.0, step_scale * 1.05)

    return best_g


def array_close(a, b, tol=1e-6):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    if a.size != b.size:
        return False
    return float(np.linalg.norm(a - b)) <= tol


def lp_alternate(a, time_budget, verbose=False):
    a = normalize(a)
    best = a.copy()
    best_obj = objective(best)
    start = time.monotonic()

    stall = 0
    patience = 15
    last_t = 0.5
    iteration = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        g, _ = solve_lp(a)
        if g is None:
            g = solve_subproblem_mirror(a, time_budget=max(0.5, min(2.0, 0.05 * remaining)), beta=100.0)
        if g is None:
            break

        if array_close(g, a, 1e-8):
            break

        ts = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        if 0.0 < last_t < 1.0:
            ts.extend(
                [
                    max(0.02, last_t * 0.6),
                    max(0.02, last_t * 0.8),
                    min(0.99, last_t * 1.2),
                    min(0.99, last_t * 1.5),
                ]
            )
        ts = sorted(set(ts))

        local_best = best_obj
        local_a = best.copy()
        local_t = last_t

        for t in ts:
            cand = (1.0 - t) * a + t * g
            cand = np.maximum(cand, 0.0)
            s = float(np.sum(cand))
            if s <= 1e-15:
                continue
            cand = cand / s
            obj = objective(cand)
            if obj < local_best:
                local_best = obj
                local_a = cand.copy()
                local_t = float(t)

        iteration += 1
        if local_best < best_obj - 1e-12:
            best_obj = local_best
            best = local_a.copy()
            a = local_a.copy()
            last_t = local_t
            stall = 0
        else:
            stall += 1
            if stall > patience:
                rng = np.random.default_rng((iteration * 7919 + len(a)) % (2 ** 32))
                scale = 0.03 + 0.01 * (stall // patience)
                a = best * np.exp(rng.normal(0.0, scale, len(best)))
                a = np.maximum(a, 0.0)
                s = float(np.sum(a))
                if s > 1e-15:
                    a /= s
                else:
                    a = best.copy()
                obj = objective(a)
                if obj < best_obj:
                    best_obj = obj
                    best = a.copy()
                stall = 0
                last_t = 0.5

        if verbose and iteration % 20 == 0:
            print(f"      LP it{iteration}: obj={best_obj:.8f}")

    return best


def active_response(a, beta=100.0):
    a = normalize(a)
    n = a.size
    if n == 0:
        return a

    c = compute_conv(a)
    y = float(n) * c
    w = softmax_weights(y, beta)
    h = convolution_grad_from_weights(a, w)

    maxh = float(np.max(np.abs(h))) if h.size else 0.0
    if maxh <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)

    h_reg = np.maximum(h, 1e-7 * maxh + 1e-30)
    resp = 1.0 / h_reg
    resp = np.where(np.isfinite(resp), resp, 0.0)
    resp = np.maximum(resp, 0.0)
    s = float(np.sum(resp))
    if s <= 1e-15:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return resp / s

def random_local_search(a, time_budget, n_trials=5000, verbose=False):
    a = normalize(a)
    n = a.size
    if n <= 1 or time_budget <= 0.0:
        return a

    best = a.copy()
    best_obj = objective(best)
    start = time.monotonic()
    rng = np.random.default_rng(1234567)
    trials = 0

    while time.monotonic() - start < time_budget and trials < n_trials:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue

        frac = 0.003 + 0.06 * float(rng.random())
        delta = frac * a[i]
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta
        obj = objective(a)
        if obj < best_obj:
            best_obj = obj
            best = a.copy()
        else:
            a[i] += delta
            a[j] -= delta

        trials += 1
        if verbose and trials % 2000 == 0:
            print(f"      Local trials={trials}: obj={best_obj:.8f}")

    return best


def add_seed(seed_list, arr):
    arr = np.asarray(arr, dtype=np.float64).ravel()
    if arr.size == 0:
        return
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(np.sum(arr))
    if s <= 1e-15:
        return
    arr = arr / s
    seed_list.append(arr.copy())


def make_seeds(n, rng):
    seeds = []
    if n <= 0:
        return seeds

    x = np.linspace(1.0 / (2.0 * n), 1.0 - 1.0 / (2.0 * n), n)

    add_seed(seeds, np.ones(n, dtype=np.float64))

    exps = [0.0, 0.25, 0.5, 1.0, 2.0, 4.0]
    for p in exps:
        for q in exps:
            a = np.power(x, p) * np.power(1.0 - x, q)
            add_seed(seeds, a)
            add_seed(seeds, a[::-1])

    for lam in (-8.0, -4.0, -2.0, -1.0, 1.0, 2.0, 4.0, 8.0):
        add_seed(seeds, np.exp(lam * (x - 0.5)))

    for f in (1, 2, 3, 4, 6, 8, 12):
        for amp in (0.2, 0.4, 0.6, 0.9):
            for phase in (0.0, np.pi / 2, np.pi, 3.0 * np.pi / 2):
                a = 1.0 + amp * np.cos(2.0 * np.pi * f * x + phase)
                add_seed(seeds, a)

    for m in (1, 2, 4, 8):
        base = np.sin(np.pi * x) ** m
        add_seed(seeds, base)
        for amp in (0.2, 0.5, 0.8):
            add_seed(seeds, base * (1.0 + amp * np.cos(2.0 * np.pi * x)))
            add_seed(seeds, base * (1.0 + amp * np.sin(2.0 * np.pi * x)))

    for c0 in (0.15, 0.25, 0.35, 0.5, 0.65, 0.75, 0.85):
        for sigma in (0.05, 0.1, 0.2, 0.35):
            add_seed(seeds, np.exp(-((x - c0) ** 2) / (2.0 * sigma * sigma)))

    for c1 in (0.2, 0.3, 0.4):
        for c2 in (0.6, 0.7, 0.8):
            for sigma in (0.07, 0.15):
                for w2 in (0.5, 1.0, 2.0):
                    a = np.exp(-((x - c1) ** 2) / (2.0 * sigma * sigma))
                    a = a + w2 * np.exp(-((x - c2) ** 2) / (2.0 * sigma * sigma))
                    add_seed(seeds, a)

    for _ in range(12):
        a = np.ones(n, dtype=np.float64)
        nf = int(rng.integers(2, 5))
        for _f in range(nf):
            f = int(rng.integers(1, 12))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            amp = float(rng.uniform(0.2, 0.9))
            a += amp * np.sin(2.0 * np.pi * f * x + phase)
        add_seed(seeds, a)

    for alpha in (0.5, 1.0, 2.0):
        for _ in range(6):
            add_seed(seeds, rng.dirichlet(np.ones(n) * alpha))

    return seeds


def _candidate_key(item):
    return item[0]


def add_unique(starts, arr, max_count=8, tol=1e-5):
    if arr is None or len(arr) == 0:
        return starts
    arr = normalize(arr)
    for s in starts:
        if array_close(s, arr, tol):
            return starts
    if len(starts) < max_count:
        starts.append(arr.copy())
    return starts


def make_n_options(base):
    opts = set()
    for delta in (-150, -100, -60, -30, -15, 0, 15, 30, 60, 100, 150, 200, 250, 300):
        opts.add(base + delta)
    for m in (700, 750, 800, 900, 1000, 1100, 1200, 1400, 1600, 1800, 2000):
        opts.add(m)

    opts = [m for m in opts if 500 <= m <= 2000]
    decorated = []
    for m in opts:
        decorated.append((abs(m - base), m))
    decorated.sort(key=itemgetter(0))
    return [item[1] for item in decorated]


def propose_candidate(seed=42, budget_s=1500):
    """
    Returns a list of nonnegative heights with a low evaluation score.
    Uses the global height_sequence_1 as a parent when available.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 40.0
    rng = np.random.default_rng(seed)

    parent_seq = None
    best_seq = None
    best_score = np.inf

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.size > 0:
            s = float(np.sum(parent_arr))
            if s > 1e-15:
                parent_arr /= s
                parent_obj = objective(parent_arr)
                if np.isfinite(parent_obj):
                    parent_seq = parent_arr.copy()
                    best_seq = parent_arr.copy()
                    best_score = float(parent_obj)
    except Exception:
        parent_seq = None
        best_seq = None
        best_score = np.inf

    if best_seq is None:
        best_seq = np.ones(800, dtype=np.float64) / 800.0
        best_score = float(objective(best_seq))

    if best_seq.size > 2000:
        best_seq = resample_sequence(best_seq, 1600)
        best_score = float(objective(best_seq))
        if parent_seq is not None and parent_seq.size > 2000:
            parent_seq = resample_sequence(parent_seq, 1600)

    n = best_seq.size
    print(f"Initial: n={n}, score={best_score:.8f}")

    candidates = [(best_score, best_seq.copy())]
    if parent_seq is not None:
        candidates.append((float(objective(parent_seq)), parent_seq.copy()))

    if time.monotonic() < deadline:
        seeds = make_seeds(n, rng)
        for arr in seeds:
            if time.monotonic() >= deadline:
                break
            obj = float(objective(arr))
            if np.isfinite(obj):
                candidates.append((obj, arr))
                if obj < best_score:
                    best_score = obj
                    best_seq = arr.copy()

    candidates.sort(key=_candidate_key)

    starts = []
    if parent_seq is not None:
        starts = add_unique(starts, parent_seq, 12)
    starts = add_unique(starts, best_seq, 12)

    for obj, arr in candidates:
        if len(starts) >= 10:
            break
        starts = add_unique(starts, arr, 10)

    if len(starts) < 4:
        for obj, arr in candidates[:20]:
            starts = add_unique(starts, arr, 4)
            if len(starts) >= 4:
                break

    if not starts:
        starts = [best_seq.copy()]

    print(f"Selected {len(starts)} starts, best after seeds={best_score:.8f}")

    remaining = max(0.0, deadline - time.monotonic())
    phase1_budget = 0.35 * remaining
    phase2_budget = 0.22 * remaining
    phase3_budget = 0.28 * remaining
    phase4_budget = 0.15 * remaining

    # Phase 1: fast active-set alternating on diverse starts.
    if starts and phase1_budget > 0.0:
        n_starts = len(starts)
        base_alloc = phase1_budget / max(1, n_starts)
        for idx, st in enumerate(starts):
            remaining = deadline - time.monotonic()
            if remaining < 15.0:
                break

            if idx == 0:
                alloc = min(2.0 * base_alloc, 220.0)
            elif idx == 1:
                alloc = min(1.5 * base_alloc, 160.0)
            else:
                alloc = base_alloc

            alloc = max(0.0, min(alloc, remaining * 0.9))
            if alloc >= 12.0:
                a_opt = active_alternate(st, alloc, verbose=(idx < 2))
                obj = float(objective(a_opt))
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  Phase1 start[{idx}]: score={best_score:.8f} **")

    # Phase 2: heavier refinement of current best and a perturbed copy.
    remaining = max(0.0, deadline - time.monotonic())
    if phase2_budget > 0.0 and remaining > 30.0:
        if best_seq.size <= 1100:
            alloc = min(phase2_budget * 0.55, remaining * 0.55, 280.0)
            if alloc > 25.0:
                a_opt = lp_alternate(best_seq, alloc, verbose=True)
                obj = float(objective(a_opt))
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  Phase2 LP best: score={best_score:.8f} **")

        remaining = max(0.0, deadline - time.monotonic())
        if remaining > 30.0:
            alloc = min(phase2_budget, remaining * 0.65, 260.0)
            if alloc > 20.0:
                rng2 = np.random.default_rng(seed + 11)
                perturbed = best_seq * np.exp(rng2.normal(0.0, 0.045, best_seq.size))
                perturbed = normalize(perturbed)
                a_opt = active_alternate(perturbed, alloc, verbose=True)
                obj = float(objective(a_opt))
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  Phase2 active perturbed: score={best_score:.8f} **")

    # Phase 3: length sweep with resampling and quick local refinement.
    remaining = max(0.0, deadline - time.monotonic())
    if phase3_budget > 0.0 and remaining > 40.0:
        n_options = make_n_options(best_seq.size)[:10]
        for far in (1500, 1600, 1800, 2000):
            if far not in n_options:
                n_options.append(far)
        n_options = n_options[:12]

        per = phase3_budget / max(1, len(n_options))
        for m in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 20.0:
                break

            alloc = min(2.0 * per, remaining * 0.35, 95.0)
            if alloc < 12.0:
                alloc = 12.0

            if m == best_seq.size:
                a_try = best_seq.copy()
            else:
                a_try = resample_sequence(best_seq, m)
                a_try = normalize(a_try)

            if alloc > 18.0:
                a_try = active_alternate(a_try, min(0.45 * alloc, 35.0), verbose=False)

            remaining_stage = deadline - time.monotonic()
            if remaining_stage > 25.0 and alloc > 35.0 and m <= 1400:
                a_try = lbfgs_optimize(
                    a_try,
                    min(remaining_stage * 0.5, 0.55 * alloc, 55.0),
                    betas=(50.0, 200.0, 800.0),
                    verbose=False,
                )

            obj = float(objective(a_try))
            marker = " **" if obj < best_score else ""
            print(f"  n={m}: score={obj:.8f}{marker}")
            if obj < best_score:
                best_score = obj
                best_seq = normalize(a_try).copy()

    # Phase 4: final polish.
    remaining = max(0.0, deadline - time.monotonic())
    if phase4_budget > 0.0 and remaining > 25.0:
        alloc = min(0.45 * phase4_budget, remaining * 0.45, 140.0)
        if alloc > 15.0:
            a_opt = lbfgs_optimize(
                best_seq,
                alloc,
                betas=(20.0, 80.0, 300.0, 1000.0),
                verbose=True,
            )
            obj = float(objective(a_opt))
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Final LBFGS: score={best_score:.8f} **")

        remaining = max(0.0, deadline - time.monotonic())
        if remaining > 50.0 and best_seq.size <= 1000:
            alloc = min(remaining * 0.35, 130.0)
            if alloc > 20.0:
                a_opt = lp_alternate(best_seq, alloc, verbose=True)
                obj = float(objective(a_opt))
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  Final LP: score={best_score:.8f} **")

        remaining = max(0.0, deadline - time.monotonic())
        if remaining > 25.0:
            alloc = min(remaining * 0.4, 90.0)
            if alloc > 10.0:
                a_opt = active_alternate(best_seq, alloc, verbose=True)
                obj = float(objective(a_opt))
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  Final active: score={best_score:.8f} **")

        remaining = max(0.0, deadline - time.monotonic())
        if remaining > 15.0:
            alloc = min(remaining * 0.5, 60.0)
            if alloc > 8.0:
                a_opt = random_local_search(best_seq, alloc, n_trials=4000, verbose=False)
                obj = float(objective(a_opt))
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  Final local: score={best_score:.8f} **")

    result = [float(max(0.0, min(1000.0, float(x)))) for x in best_seq]

    try:
        val = evaluate_sequence(result)
        if np.isfinite(val):
            best_score = float(val)
    except Exception:
        pass

    final_obj = float(objective(np.asarray(result, dtype=np.float64)))
    if not np.isfinite(final_obj):
        result = [1.0] * 100
        final_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"TARGET: 1.5027, achieved={best_score:.8f}, gap={best_score - 1.5027:.8f}")

    return result
