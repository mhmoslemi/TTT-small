import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

        for i in range(n):

    return A.tocsr()

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Iterated LP (reduced constraints for speed)
- Golden section line search
- Simulated annealing
- Multiplicative weight updates
- Multi-length exploration
- Fast FFT-based convolution

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = np.asarray(a, dtype=np.float64) / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_sparse(a):
    """Build CSR matrix A (2n-1 x n) where A @ g = conv(a, g)."""
    n = len(a)
    total_nnz = 0
    for j in range(n):
        total_nnz += min(n, 2 * n - 1 - j)
    
    data = np.zeros(total_nnz)
    indices = np.zeros(total_nnz, dtype=np.int32)
    indptr = np.zeros(n + 1, dtype=np.int32)
    
    pos = 0
    for j in range(n):
        r_start = j
        r_end = min(2 * n - 1, j + n)
        length = r_end - r_start
        data[pos:pos + length] = a[:length]
        indices[pos:pos + length] = np.arange(r_start, r_end)
        pos += length
        indptr[j + 1] = pos
    
    return csr_matrix((data, indices, indptr), shape=(2 * n - 1, n))


def solve_lp(a, K=None, solver_list=None):
    """
    Solve LP: min M s.t. conv(a, g)[k] <= M for (top K) k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a = a / s
    
    if K is not None and K < 2 * n - 1:
        conv_a = compute_conv(a)
        top_idx = np.argsort(conv_a)[::-1][:K]
    else:
        top_idx = np.arange(2 * n - 1)
    
    A_full = build_conv_matrix_sparse(a)
    A = A_full[top_idx, :]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

        solver_list = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    return g_val / s_val, float(M.value)
        except Exception:
            continue
    return None, None


def golden_section_search(f, a_bound=0.0, b_bound=1.0, tol=1e-5, max_iter=30):
    """Golden section search for minimum of f on [a_bound, b_bound]."""
    GR = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    
    c = b_bound - GR * (b_bound - a_bound)
    d = a_bound + GR * (b_bound - a_bound)
    fc = f(c)
    fd = f(d)
    
    for _ in range(max_iter):
        if b_bound - a_bound < tol:
            break
        if fc < fd:
            b_bound = d
            d = c
            fd = fc
            c = b_bound - GR * (b_bound - a_bound)
            fc = f(c)
        else:
            a_bound = c
            c = d
            fc = fd
            d = a_bound + GR * (b_bound - a_bound)
            fd = f(d)
    
    t_opt = (a_bound + b_bound) / 2
    return t_opt, f(t_opt)


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)
