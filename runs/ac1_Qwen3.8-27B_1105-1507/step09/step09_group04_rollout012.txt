def efficient_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = efficient_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coord_descent_fast(a, time_budget, start_time=0, rng=None, verbose=False):
    """
    Fast coordinate descent with tight golden section brackets.
    Normalizes a to sum=1 and minimizes max(conv(a,a)).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
    a = a / S
    conv = efficient_conv(a)

    best_obj = float(np.max(conv))
    best_a = a.copy()
    gr = (np.sqrt(5) - 1) / 2.0
    total_sweeps = 0
    stagnant = 0
    cd_start = time.monotonic() if start_time == 0 else start_time

    for sweep in range(2000):
        if time.monotonic() - cd_start > time_budget:
            break

        obj_before = float(np.max(conv))

        if rng is not None and sweep % 5 == 4:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        improved = False
        for k in order:
            if time.monotonic() - cd_start > time_budget:
                break

            d = np.zeros(2 * n - 1)
            d[k:k + n] = a

            lo_delta = -a[k]
            cur_max = float(np.max(conv))
            # Tighter upper bound
            up = cur_max - conv[2 * k]
            if up > 0:
                hi_delta = np.sqrt(max(0.0, up)) * 0.8 + a[k] * 0.5
            else:
                hi_delta = 1.0 / n
            hi_delta = max(hi_delta, lo_delta + 1.0 / (2 * n))
            hi_delta = min(hi_delta, 3.0 / n)

            # Golden section search (reduced iterations)
            a_gs = lo_delta
            b_gs = hi_delta
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)

            def obj_delta(delta):
                val = conv + 2.0 * delta * d
                val[2 * k] += delta * delta
                return float(np.max(val))

            fc = obj_delta(c_gs)
            fd = obj_delta(d_gs)

            for _ in range(30):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = obj_delta(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = obj_delta(d_gs)

            delta_opt = (a_gs + b_gs) / 2.0
            new_obj_est = obj_delta(delta_opt)

            if new_obj_est < obj_before - 1e-14:
                a[k] += delta_opt
                a[k] = max(0.0, a[k])
                conv += 2.0 * delta_opt * d
                conv[2 * k] += delta_opt * delta_opt
                S_new = float(np.sum(a))
                if S_new < 1e-10:
                    a[:] = 1.0 / n
                    conv = efficient_conv(a)
                else:
                    a /= S_new
                    conv /= (S_new * S_new)
                improved = True

        total_sweeps += 1
        obj_after = float(np.max(conv))

        if obj_after < best_obj - 1e-13:
            best_obj = obj_after
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 8:
                if rng is not None:
                    scale = 0.005 + 0.003 * min(stagnant, 20)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    conv = efficient_conv(a)
                    best_obj = float(np.max(conv))
                    stagnant = 0
                else:
                    break

        if verbose and total_sweeps % 20 == 0:
            elapsed = time.monotonic() - cd_start
            print(f"    CD sweep {total_sweeps}: obj={2*n*best_obj:.8f}, time={elapsed:.0f}s")

    return best_a, compute_obj(best_a)


def build_conv_matrix_sparse(a):
    """Build the convolution matrix A such that A@g = conv(a,g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for i in range(n):
        # A[k, i] = a[k-i] if 0 <= k-i < n
        k_start = i
        k_end = min(2 * n - 1, i + n)
        for k in range(k_start, k_end):
            idx = k - i
            if 0 <= idx < n:
                rows.append(k)
                cols.append(i)
                vals.append(a[idx])
    A = cp.CompactableVar if False else None
    return rows, cols, vals, 2 * n - 1, n


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)

    rows = []
    cols = []
    vals = []
    for i in range(n):
        for k in range(i, min(2 * n - 1, i + n)):
            idx = k - i
            if 0 <= idx < n and a_norm[idx] > 1e-15:
                rows.append(k)
                cols.append(i)
                vals.append(a_norm[idx])

    nz = len(vals)
    A_data = np.array(vals, dtype=np.float64)
    A_rows = np.array(rows, dtype=np.int32)
    A_cols = np.array(cols, dtype=np.int32)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [cp.sum(g) == 1]

    # Use sparse matrix for the convolution constraint
    from scipy.sparse import coo_matrix
    A_sparse = coo_matrix((A_data, (A_rows, A_cols)), shape=(2 * n - 1, n))
    constraints.append(A_sparse @ g <= M)

    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None

        conv_aa = efficient_conv(a)

        def obj_at_t(t):
            val = conv_aa + 2.0 * t * conv_ag + t * t * conv_gg

                c_gs = d_gs
                fc = fd

def mirror_descent_phase(a, time_budget, rng=None, verbose=False):
    """Mirror descent (exponential gradient) on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 2.0 / np.sqrt(n)
    it = 0
    best_it = 0
    conv_cache = efficient_conv(a)

    while time.monotonic() - start < time_budget:
        # Compute subgradient
        conv = conv_cache
        max_val = float(np.max(conv))
        # Use soft active set for stability
        tol = 0.01 * max_val
        P = np.where(conv >= max_val - tol)[0]
        if len(P) == 0:
            P = np.array([np.argmax(conv)])

        g = np.zeros(n)
        for k in P:
            if k < n:
                g[:n - k] += a[k:]
            g[k:] += a[:n - k]
        g /= len(P)

        # Mirror descent step
        log_ratio = -eta * g
        log_ratio -= np.max(log_ratio)
        weights = np.exp(log_ratio)
        a_new = a * weights
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        conv_cache = efficient_conv(a)
        it += 1

        # Decay eta
        if it % 200 == 0:
            eta *= 0.92

        # Restart
        if it - best_it > 1500:
            if rng is not None:
                a = best_a * np.exp(rng.normal(0, 0.04, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
            else:
                a = best_a.copy()
            eta = 2.0 / np.sqrt(n)
            conv_cache = efficient_conv(a)
            best_it = it

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD it{it}: obj={2*n*best_obj/( (best_a.sum())**2 ):.8f}, eta={eta:.4f}, t={elapsed:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_init(n, rng, mode=0):
    """Generate structured initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        # Sqrt-Bartlett
        t = np.arange(n)
        center = (n - 1) / 2.0
        tri = np.maximum(0, 1 - np.abs(t - center) / center)
        a = np.sqrt(tri + 1e-10)
        a = a / a.sum()
    elif mode == 2:
        # Raised cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 3:
        # Hann window
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 4:
        # Flat top approximation
        a = 1.0 - 1.93 * np.cos(2 * np.pi * x) + 1.29 * np.cos(4 * np.pi * x) - 0.39 * np.cos(6 * np.pi * x)
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 5:
        # Two Gaussian bumps
        c1, c2 = n // 4, 3 * n // 4
        w = n // 10
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 6:
        # Power law
        alpha = rng.uniform(-1.5, 1.5)
        a = (x + 1e-3) ** alpha * ((1 - x) + 1e-3) ** alpha
        a = a / a.sum()
    elif mode == 7:
        # Cosine with random phase
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * x + phase)
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 8:
        # Super-Gaussian
        p = rng.uniform(2, 5)
        t = (np.arange(n) - n / 2) / (n / 2)
        a = np.exp(-np.abs(t) ** p)
        a = a / a.sum()
    else:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    return a

    if p2_budget > 40:
        for rnd in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            alloc = min(p2_budget / (6 - rnd), remaining * 0.6)
            if alloc < 25:
                break
            a_lp = lp_polish_optimal(best_seq, alloc, rng=rng, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
                # Perturb before next round
                scale = 0.01 * (rnd + 1)
                best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                best_seq = np.maximum(best_seq, 0)
                best_seq = best_seq / best_seq.sum()

    # === Phase 3: Mirror Descent ===
    print(f"\n=== Phase 3: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.18, 200.0)
    if p3_budget > 30:
        a_md = mirror_descent_phase(best_seq, p3_budget, rng=rng, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Phase3 MD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 MD: obj={obj_md:.8f}")

    # === Phase 4: Fine Coordinate Descent ===
    print(f"\n=== Phase 4: Fine CD ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.18, 200.0)
    if p4_budget > 30:
        a_cd2, obj_cd2 = coord_descent_fast(best_seq, p4_budget, rng=rng, verbose=True)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  Phase4 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 CD: obj={obj_cd2:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(300, n_current - 200), max(300, n_current - 100),
        n_current + 100, n_current + 200, n_current + 400,
        500, 700, 900, 1100, 1400, 1600, 2000
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000 and abs(n - n_current) > 50]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 120:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick CD
        alloc1 = min(remaining * 0.3, 60.0)
        if alloc1 > 20:
            a_try, _ = coord_descent_fast(a_try, alloc1, rng=rng)

        # Quick LP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 40.0)
        if alloc2 > 15:
            a_try = lp_polish_optimal(a_try, alloc2, rng=rng)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Structured Restarts ===
    print(f"\n=== Phase 6: Structured Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 120:
            break

        mode = [1, 4, 8][r]  # Sqrt-Bartlett, Flat-top, Super-Gaussian
        a_r = structured_init(n_best, rng, mode=mode)
        blend = 0.4 + 0.15 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc1 = min(remaining * 0.35, 70.0)
        if alloc1 > 20:
            a_r, _ = coord_descent_fast(a_r, alloc1, rng=rng)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.2, 40.0)
        if alloc2 > 15:
            a_r = lp_polish_optimal(a_r, alloc2, rng=rng)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    if remaining > 40:
        alloc = min(remaining * 0.4, 100.0)
        a_lp_f = lp_polish_optimal(best_seq, alloc, rng=rng, verbose=True)
        obj_lp_f = compute_obj(a_lp_f)
        if obj_lp_f < best_score:
            best_score = obj_lp_f
            best_seq = a_lp_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # CD
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 80.0)
        a_cd_f, obj_cd_f = coord_descent_fast(best_seq, alloc, rng=rng, verbose=True)
        if obj_cd_f < best_score:
            best_score = obj_cd_f
            best_seq = a_cd_f.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.5, 50.0)
        a_lp_f2 = lp_polish_optimal(best_seq, alloc, rng=rng)
        obj_lp_f2 = compute_obj(a_lp_f2)
        if obj_lp_f2 < best_score:
            best_score = obj_lp_f2
            best_seq = a_lp_f2.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final CD micro-polish
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.6, 30.0)
        a_cd_f2, obj_cd_f2 = coord_descent_fast(best_seq, alloc, rng=rng)
        if obj_cd_f2 < best_score:
            best_score = obj_cd_f2
            best_seq = a_cd_f2.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
