   def compute_obj(a):

   def project_simplex(v):
       n = len(v)
       if n == 0:
           return v
       u = np.sort(v)[::-1]
       cssv = np.cumsum(u)
       idx = np.arange(1, n + 1)
       cond = u * idx > (cssv - 1.0)
       rho = idx[cond]
       if len(rho) == 0:
           return np.ones(n) / n
       rho_val = rho[-1]
       theta = (cssv[rho_val - 1] - 1.0) / rho_val
       return np.maximum(v - theta, 0.0)

   def mirror_descent_subgrad(a, time_budget=60.0, top_k=1, verbose=False):
       """
       Mirror descent with exact subgradient for max conv.
       Uses exponential (multiplicative weights) update.
       """
       n = len(a)
       a = np.maximum(a, 0).astype(np.float64).copy()
       s = float(np.sum(a))
       if s < 1e-10:
           return a
       a = a / s

       best_a = a.copy()
       best_obj = compute_obj(a)
       start = time.monotonic()

       # Adaptive learning rate
       lr = 1.0
       patience = 0
       best_patience = 0

       while time.monotonic() - start < time_budget:
           conv = compute_conv(a)
           max_val = float(np.max(conv))
           k_star = int(np.argmax(conv))

           # Compute subgradient
           if top_k == 1:
               # Single peak subgradient
               # g[m] = 2 * a[k_star - m] if 0 <= k_star - m < n
               idx = k_star - np.arange(n)
               mask = (idx >= 0) & (idx < n)
               g = np.zeros(n)
               g[mask] = 2.0 * a[idx[mask]]
           else:
               # Multi-peak weighted subgradient
               top_indices = np.argsort(conv)[::-1][:top_k]
               weights = conv[top_indices] / float(np.sum(conv[top_indices]))
               g = np.zeros(n)
               for k in top_indices:
                   idx = k - np.arange(n)
                   mask = (idx >= 0) & (idx < n)
                   g[mask] += weights[list(top_indices).index(k)] * 2.0 * a[idx[mask]]

           # Mirror descent update: a_new = a * exp(-lr * g) / Z
           # Clip lr * g to avoid overflow
           logit = -lr * g
           logit -= np.max(logit)  # for numerical stability
           a_new = a * np.exp(logit)
           s_new = float(np.sum(a_new))
           if s_new < 1e-30:
               a_new = a.copy()
           else:
               a_new = a_new / s_new

           obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))

           if obj_new < best_obj - 1e-12:
               best_obj = obj_new
               best_a = a_new.copy()
               a = a_new
               patience = 0
               # Slightly increase lr
               lr = min(lr * 1.01, 10.0)
           else:
               patience += 1
               # Decrease lr
               lr = max(lr * 0.95, 0.001)
               if patience > 200:
                   # Perturb
                   rng = np.random.default_rng(int(time.monotonic() * 1000))
                   a = best_a * np.exp(rng.normal(0, 0.05, n))
                   a = np.maximum(a, 0)
                   s_a = float(np.sum(a))
                   if s_a > 1e-10:
                       a = a / s_a
                   else:
                       a = best_a.copy()
                   patience = 0
                   lr = 1.0
                   if patience > 0:
                       break

           if verbose and patience % 100 == 0:
               elapsed = time.monotonic() - start
               print(f"    MD: obj={best_obj:.8f}, lr={lr:.4f}, k*={k_star}, elapsed={elapsed:.1f}s")

       return best_a

   def mirror_descent_smoothed(a, time_budget=60.0, tau=50.0, verbose=False):
       """
       Mirror descent with smoothed (softmax-weighted) subgradient.
       """
       n = len(a)
       a = np.maximum(a, 0).astype(np.float64).copy()
       s = float(np.sum(a))
       if s < 1e-10:
           return a
       a = a / s

       best_a = a.copy()
       best_obj = compute_obj(a)
       start = time.monotonic()

       lr = 0.5
       patience = 0
       total_iter = 0

       while time.monotonic() - start < time_budget:
           # Progressively increase tau
           progress = (time.monotonic() - start) / time_budget
           tau_cur = tau * (1 + progress * 5)

           conv = compute_conv(a)
           max_conv = float(np.max(conv))
           # Softmax weights
           weights = np.exp(tau_cur * (conv - max_conv))
           w_sum = float(np.sum(weights))
           if w_sum < 1e-30:
               weights = np.zeros(len(conv))
               weights[np.argmax(conv)] = 1.0
           else:
               weights = weights / w_sum

           # Weighted subgradient
           g = np.zeros(n)
           # For efficiency, only use top Lags
           top_L = min(20, len(weights))
           top_idx = np.argsort(weights)[::-1][:top_L]
           w_top = weights[top_idx]
           w_top_sum = float(np.sum(w_top))
           if w_top_sum < 1e-30:
               break
           w_top = w_top / w_top_sum

           for k in top_idx:
               idx = k - np.arange(n)
               mask = (idx >= 0) & (idx < n)
               g[mask] += w_top[list(top_idx).index(k)] * 2.0 * a[idx[mask]]

           # Mirror descent update
           logit = -lr * g
           logit -= np.max(logit)
           a_new = a * np.exp(logit)
           s_new = float(np.sum(a_new))
           if s_new < 1e-30:
               a_new = a.copy()
           else:
               a_new = a_new / s_new

           obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
           total_iter += 1

           if obj_new < best_obj - 1e-12:
               best_obj = obj_new
               best_a = a_new.copy()
               a = a_new
               patience = 0
               lr = min(lr * 1.005, 5.0)
           else:
               patience += 1
               lr = max(lr * 0.97, 0.01)
               if patience > 500:
                   rng = np.random.default_rng(total_iter)
                   pert = rng.normal(0, 0.03, n)
                   a = best_a * np.exp(pert)
                   a = np.maximum(a, 0)
                   s_a = float(np.sum(a))
                   if s_a > 1e-10:
                       a = a / s_a
                   else:
                       a = best_a.copy()
                   patience = 0

               print(f"    MD-sm: obj={best_obj:.8f}, tau={tau_cur:.1f}, lr={lr:.4f}, it={total_iter}, elapsed={elapsed:.1f}s")

       return best_a

   def build_conv_matrix_sparse(a):
       """Build sparse convolution matrix."""
       n = len(a)
       rows = []
       cols = []
       vals = []
       for j in range(n):
           r = min(n, 2 * n - 1 - j)
           for i in range(r):
               rows.append(j + i)
               cols.append(j)
               vals.append(a[i])
       from scipy.sparse import csr_matrix
       A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
       return A

   def solve_lp_direction(a):
       """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns g."""
       n = len(a)
       a_norm = a / float(np.sum(a))
       A = build_conv_matrix_sparse(a_norm)

       g = cp.Variable(n, nonneg=True)
       M = cp.Variable()
       constraints = [A @ g <= M, cp.sum(g) == 1]
       prob = cp.Problem(cp.Minimize(M), constraints)

       for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
           try:
               prob.solve(solver=solver)
               if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                   g_val = np.maximum(g.value, 0)
                   s = float(np.sum(g_val))
                   if s > 1e-10:
                       return g_val / s, float(M.value)
           except Exception:
               continue
       return None, None

   def lp_polish(a, time_budget=60.0, verbose=False):
       """LP-based Frank-Wolfe polishing."""
       n = len(a)
       a = np.maximum(a, 0).astype(np.float64).copy()
       s = float(np.sum(a))
       if s < 1e-10:
           return a
       a = a / s

       best_a = a.copy()
       best_obj = compute_obj(a)
       start = time.monotonic()

       it = 0
       stall = 0

       while time.monotonic() - start < time_budget:
           g_lp, M_lp = solve_lp_direction(a)
           if g_lp is None:
               break

           # Try different blending factors
           best_t = 0
           best_obj_t = best_obj
           for t in [0.3, 0.5, 0.7, 1.0]:
               a_try = (1 - t) * a + t * g_lp
               a_try = np.maximum(a_try, 0)
               s_try = float(np.sum(a_try))
               if s_try < 1e-10:
                   continue
               a_try = a_try / s_try
               obj_try = compute_obj(a_try)
               if obj_try < best_obj_t:
                   best_obj_t = obj_try
                   best_t = t

           if best_t > 0 and best_obj_t < best_obj - 1e-13:
               best_obj = best_obj_t
               best_a = ((1 - best_t) * a + best_t * g_lp)
               best_a = np.maximum(best_a, 0)
               best_a = best_a / best_a.sum()
               a = best_a.copy()
               stall = 0
           else:
               stall += 1
               if stall > 20:
                   # Perturb
                   rng = np.random.default_rng(it)
                   a = best_a * np.exp(rng.normal(0, 0.03, n))
                   a = np.maximum(a, 0)
                   a = a / a.sum()
                   stall = 0
                   if it > 100:
                       break

           it += 1
           if verbose and it % 5 == 0:
               print(f"    LP it{it}: obj={best_obj:.8f}, M_lp={M_lp:.6f}, t={best_t}")

       return best_a

   def diverse_init(n, rng, mode=0):
       if mode == 0:
           a = np.ones(n) / n
       elif mode == 1:
           a = rng.uniform(0.1, 1, n)
           a = a / a.sum()
       elif mode == 2:
           # Two-bump
           x = np.arange(n)
           a = np.exp(-((x - n*0.25)**2) / (n*0.05)) + np.exp(-((x - n*0.75)**2) / (n*0.05))
           a = a / a.sum()
       elif mode == 3:
           # Arcsine
           x = np.linspace(-1, 1, n)
           a = 1.0 / np.sqrt(1 - x**2 + 1e-8)
           a = a / a.sum()
       elif mode == 4:
           # Cosine window
           x = np.linspace(0, np.pi, n)
           a = 0.5 + 0.5 * np.cos(x)
           a = a / a.sum()
       elif mode == 5:
           # Linear ramp up then down (tent)
           x = np.linspace(0, 1, n)
           a = 1 - np.abs(2*x - 1)
           a = a / a.sum()
       elif mode == 6:
           # Random smooth
           a = rng.uniform(0, 1, n)
           # Smooth
           from scipy.ndimage import gaussian_filter1d
           a = gaussian_filter1d(a, sigma=np.sqrt(n)/3)
           a = np.maximum(a, 0.001)
           a = a / a.sum()
       else:
           # Comb pattern
           a = np.ones(n)
           a[::2] = 0.3
           a = a / a.sum()
       return a

   def resample_sequence(a, new_n):
       n = len(a)
       if n == new_n:
           return a.copy()
       if n == 1:
           return np.full(new_n, a[0])
       x_old = np.linspace(0, 1, n)
       x_new = np.linspace(0, 1, new_n)
       a_new = np.interp(x_new, x_old, a)
       return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses exact-subgradient mirror descent (exponential/multiplicative updates),
LP-based Frank-Wolfe polishing, QP active-set minimization, and multi-n exploration.
Key innovations: O(n) subgradient computation via shifted-reflected indexing,
adaptive learning rate with patience-based restart, and systematic length search.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = idx[cond]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def subgradient_single(a, k_star):
    """Compute subgradient of conv(a,a)[k_star] w.r.t. a. O(n)."""
    n = len(a)
    g = np.zeros(n)
    idx = k_star - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    g[mask] = 2.0 * a[idx[mask]]
    return g


def subgradient_multi(a, top_indices, weights):
    """Compute weighted sum of subgradients for multiple peaks. O(n*T)."""
    n = len(a)
    g = np.zeros(n)
    for i, k in enumerate(top_indices):
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g[mask] += weights[i] * 2.0 * a[idx[mask]]
    return g
