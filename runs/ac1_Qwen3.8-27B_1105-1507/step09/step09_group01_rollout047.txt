"""

Uses a hybrid approach: spectral initialization with Gerchberg-Saxton style
flat-spectrum construction, symmetric alternating LP, peak-shaving subgradient,

convolution matrix construction for faster LP solves.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy import sparse
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_sparse(a):
    """Build sparse matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # conv(a,g)[k] = sum_i a[i]*g[k-i], so for fixed j (index of g):
        # contribution to position k is a[k-j] for valid k
        # k ranges from j to j+n-1, and a index is k-j from 0 to n-1
        # Also k must be in [0, 2n-2]
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        a_start = 0
        a_end = k_end - j + 1
        for k in range(k_start, k_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_sparse(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = solver_pref or [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue
    return None, None


def solve_lp_maxsum(a, M_bound):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M_bound, g>=0.
    Returns (g, M_bound) or (None, None) if infeasible."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_sparse(a_norm)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M_bound]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val, s
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue
    return None, None


def spectral_init_flat(n, num_iters=100, rng=None):
    """
    Gerchberg-Saxton style initialization: iteratively project between
    time domain (a>=0, sum=1) and frequency domain (flat |A|^2).
    Aims to create a sequence with flatter power spectrum.
    """
    if rng is None:
        rng = np.random.default_rng(0)

    # Start with random phase, flat amplitude
    size = 1
    while size < 2 * n:
        size <<= 1

    # Target: flat power spectrum (uniform |A|^2)
    target_amp = np.ones(size)
    # Zero out high frequencies to get a "bandlimited" flat spectrum
    band = int(size * 0.4)
    target_amp[band:-band] = 0.0
    target_amp /= target_amp.max()

    # Random initial phase
    phase = rng.uniform(0, 2 * np.pi, size)
    A = target_amp * np.exp(1j * phase)

    a = np.ones(n) / n
    for it in range(num_iters):
        # Frequency domain: enforce target amplitude
        A = fft(a, size)
        A = target_amp * np.exp(1j * np.angle(A))
        # Time domain: project to {a>=0, sum=1, length n}
        a_full = ifft(A).real[:n]
        a_full = np.maximum(a_full, 0)
        s = a_full.sum()
        if s < 1e-10:
            a_full = rng.dirichlet(np.ones(n))
        else:
            a_full = a_full / s
        a = a_full

    return a


def diverse_init_v2(n, rng, mode=0):
    """Generate diverse initializations with spectral and combinatorial ideas."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Flat spectrum via Gerchberg-Saxton
        a = spectral_init_flat(n, num_iters=50, rng=rng)
    elif mode == 3:
        # Sine-modulated
        x = np.linspace(0, 1, n)
        f1 = rng.integers(1, 5)
        f2 = rng.integers(5, 15)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * f1 * x) + 0.15 * np.cos(2 * np.pi * f2 * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Two-peak structure
        c1 = n // 4
        c2 = 3 * n // 4
        w = max(1, n // 10)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Cosine bump (optimal for some bandlimited problems)
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a / a.sum()
    elif mode == 6:
        # Exponential decay pair
        a = np.zeros(n)
        a[:n//2] = np.exp(-0.5 * np.arange(n//2) / (n//8))
        a[n//2:] = np.exp(-0.5 * np.arange(n//2) / (n//8))[::-1]
        a = a / a.sum()
    elif mode == 7:
        # Higher-order cosine (flattens convolution more)
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x) + 0.05 * np.cos(6 * np.pi * x)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 8:
        # Random Fourier features made positive
        freqs = rng.uniform(0, 3, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = 1.0
        for f, p in zip(freqs, phases):
            a += 0.2 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.005)
        a = a / a.sum()
    elif mode == 9:
        # Symmetric bump (optimal for symmetric convolution)
        center = n // 2
        a = np.exp(-((np.arange(n) - center) ** 2) / (2 * (n/6) ** 2))
        a = a / a.sum()
    else:
        # Multi-scale
        a = np.ones(n) / n
        x = np.arange(n) / n
        a += 0.1 * np.cos(2 * np.pi * x * rng.integers(1, 8))
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    return a


def peak_shaving_subgradient(a, time_budget=60.0, K=3, verbose=False, rng=None):
    """
    Aggressive peak-shaving subgradient method. Focuses on reducing
    the top-K peaks of the convolution using weighted subgradients.
    Uses adaptive step size (Polyak-like) and momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    if rng is None:
        rng = np.random.default_rng(0)

    lr = 3.0 / np.sqrt(n)
    momentum = 0.8
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break

        # Weight peaks exponentially (more weight on larger peaks)
        weights = np.exp((peak_vals - max_pv) * 8.0 / max_pv)
        weights /= weights.sum()

        # Compute subgradient
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += 2.0 * w * a[j_start:j_end + 1][::-1]

        # Update with momentum
        v = momentum * v + lr * g
        a_new = a - v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.01, 10.0 / np.sqrt(n))
        else:
            stall += 1
            a = best_a.copy()
            if stall % 20 == 0:
                lr *= 0.7
                v *= 0.5
            if stall > 500:
                # Perturb
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = project_simplex(a)
                v = np.zeros(n)
                lr = 2.0 / np.sqrt(n)
                stall = 0
                if stall > 1000:
                    break

        it += 1
        if verbose and it % 200 == 0:
            print(f"    PS it{it}: obj={best_obj:.8f}")

    return best_a


def nesterov_smoothed_descent(a, time_budget=100.0, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Smoothed gradient via FFT
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(y, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        # BB step size adaptation
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.6 * lr + 0.4 * min(max(bb_lr, 1e-8), 20.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.99995)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)
            best_it = it

    return best_a


def symmetric_alternating_lp(a, time_budget=300.0, verbose=False, seed=42):
    """
    Symmetric alternating LP: alternately optimize a (given g) and g (given a).
    This forms a symmetric cycle that may escape local optima better than
    the asymmetric approach.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Initialize g with current a
    g = a.copy()
    iter_count = 0
    no_improve = 0
    last_best_iter = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start

        # Step 1: Optimize g given a (min peak of conv(a, g))
        g_lp, M1 = solve_lp_minpeak_sparse(a)
        if g_lp is None:
            break
        g = g_lp

        # Step 2: Optimize a' given g (min peak of conv(a', g))
        a_lp, M2 = solve_lp_minpeak_sparse(g)
        if a_lp is None:
            break

        # Blend a with a_lp
        a_try = 0.3 * best_a + 0.7 * a_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            a_try = best_a.copy()
        else:
            a_try = a_try / s_try
        obj_try = compute_obj(a_try)

        iter_count += 1

        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()
            no_improve = 0
            if verbose and iter_count % 3 == 0:
                print(f"    SymLP it{iter_count}: obj={best_obj:.8f}, M1={M1:.6f}, M2={M2:.6f}")
        else:
            no_improve += 1
            if no_improve == 2:
                # Try different blend ratio
                a_try2 = 0.5 * best_a + 0.5 * a_lp
                a_try2 = np.maximum(a_try2, 0)
                a_try2 = a_try2 / a_try2.sum()
                obj_try2 = compute_obj(a_try2)
                if obj_try2 < best_obj - 1e-14:
                    best_obj = obj_try2
                    best_a = a_try2.copy()
                    a = a_try2.copy()
                    no_improve = 0
            elif no_improve >= 5:
                # Perturb and restart cycle
                scale = 0.01 + 0.005 * no_improve
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                g = a.copy()
                no_improve = 0
                if iter_count > 50:
                    break

        # Periodic: also try pure g optimization result
        if iter_count % 5 == 0:
            a_try3 = (1 - 0.5) * best_a + 0.5 * a_lp
            a_try3 = np.maximum(a_try3, 0)
            a_try3 = a_try3 / a_try3.sum()
            obj_try3 = compute_obj(a_try3)
            if obj_try3 < best_obj - 1e-14:
                best_obj = obj_try3
                best_a = a_try3.copy()
                a = a_try3.copy()
                no_improve = 0

    return best_a


def lp_polish_multit(a, time_budget=200.0, verbose=False, seed=42):
    """
    LP-based polishing with multiple blend ratios and constraint tightening.
    More aggressive than the previous version with periodic constraint sweeps.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start

        # Solve LP with current a
        g_lp, M_lp = solve_lp_minpeak_sparse(a)
        if g_lp is None:
            break

        # Try multiple blend ratios
        best_blend_obj = best_obj
        best_blend_a = best_a.copy()
        for t in [0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj - 1e-14:
                best_blend_obj = obj_try
                best_blend_a = a_try.copy()

        iter_count += 1

        if best_blend_obj < best_obj - 1e-14:
            best_obj = best_blend_obj
            best_a = best_blend_a.copy()
            a = best_a.copy()
            no_improve = 0
            if verbose and iter_count % 5 == 0:
                print(f"    LPM it{iter_count}: obj={best_obj:.8f}, M={M_lp:.6f}")
        else:
            no_improve += 1
            if no_improve == 3:
                # Constraint tightening: try to force lower peak
                for frac in [0.95, 0.90, 0.85]:
                    M_tight = M_lp * frac
                    g_tight, _ = solve_lp_minpeak_sparse(a, M_target=M_tight)
                    if g_tight is not None:
                        a_try = 0.5 * best_a + 0.5 * g_tight
                        a_try = np.maximum(a_try, 0)
                        a_try = a_try / a_try.sum()
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try.copy()
                            no_improve = 0
                            break
            elif no_improve >= 6:
                scale = 0.008 * min(no_improve, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                if iter_count > 80:
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    from scipy.interpolate import PchipInterpolator
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    try:
        interp = PchipInterpolator(x_old, a, fill_value=(a[0], a[-1]))
        a_new = interp(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def local_search(a, time_budget=60.0, rng=None):
    """
    Quick local search combining peak shaving and smoothed gradient.
    Used for evaluating diverse initializations.
    """
    if time_budget < 5:
        return a

    # Peak shaving
    alloc1 = time_budget * 0.5
    a = peak_shaving_subgradient(a, time_budget=alloc1, K=3, rng=rng)

    # Smoothed gradient
    alloc2 = time_budget * 0.5
    a = nesterov_smoothed_descent(a, time_budget=alloc2, tau_init=20.0, tau_end=300.0)

    return a

            if obj_ns < best_score:
                best_score = obj_ns
                best_seq = a_ns.copy()
                print(f"  NS: obj={best_score:.8f} **")
            else:
                print(f"  NS: obj={obj_ns:.8f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Generate candidate n values around current
    n_candidates = sorted(set([
        max(200, n_current - 300), max(200, n_current - 150),
        n_current + 100, n_current + 250, n_current + 500,
        max(200, int(n_current * 0.7)), max(200, int(n_current * 1.3)),
    ]))
    n_candidates = [n for n in n_candidates if 200 <= n <= 3000 and abs(n - n_current) > 50]

    for n_try in n_candidates[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)

        # Quick local search
        alloc1 = min(remaining * 0.2, 60.0)
        if alloc1 > 10:
            a_try = local_search(a_try, time_budget=alloc1, rng=rng)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.3, 150.0)
        if alloc2 > 20:
            a_try = lp_polish_multit(a_try, time_budget=alloc2, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Perturbation + re-optimization ===
    print(f"\n=== Phase 6: Perturbation Search ===")
    remaining = deadline - time.monotonic()

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        scale = 0.005 + 0.01 * r
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        # Alternating LP for perturbed
        alloc = min(remaining * 0.3, 120.0)
        if alloc > 20:
            a_pert = symmetric_alternating_lp(a_pert, time_budget=alloc, seed=200 + r)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Pert {r} (s={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Pert {r} (s={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        # Combine best methods
        alloc_lp = remaining * 0.5
        alloc_sg = remaining * 0.3
        alloc_ns = remaining * 0.2

        if alloc_lp > 20:
            a_final = lp_polish_multit(best_seq, time_budget=alloc_lp, verbose=True, seed=999)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_sg = peak_shaving_subgradient(best_seq, time_budget=min(remaining * 0.4, 40.0), K=8, rng=rng)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  Final PS: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_ns = nesterov_smoothed_descent(best_seq, time_budget=min(remaining * 0.5, 30.0),
                                             tau_init=100.0, tau_end=1000.0)
            obj_ns = compute_obj(a_ns)
            if obj_ns < best_score:
                best_score = obj_ns
                best_seq = a_ns.copy()
                print(f"  Final NS: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
