For fixed i (column), A[k, i] = a[k-i] for k in [i, i+n-1].
So column i has a[0] at row i, a[1] at row i+1, ..., a[n-1] at row i+n-1.

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based fictitious play with line search, momentum acceleration,
smoothed gradient descent with high tau, multi-length exploration,
and diverse restarts. Key improvements over previous: proper 1D line search
for blend factor, Nesterov momentum on LP iterations, adaptive constraint
focus, and more aggressive length sweeping.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize_scalar
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def solve_lp_minpeak(a, solver_order=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    if solver_order is None:
        solver_order = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]

    for solver in solver_order:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M_var.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a) via FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1

    P = fft(p, N)
    A_freq = fft(a, N)
    g_full = ifft(P * np.conj(A_freq)).real
    g = 2.0 * g_full[:n]
    return g, conv


def lp_iteration(a, time_budget, verbose=False, use_momentum=True, seed=42):
    """
    LP-based fictitious play with line search and optional momentum.
    Iteratively: solve LP for best response g, line search for best blend, update a.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    v = np.zeros(n) if use_momentum else None
    beta_mom = 0.7
    iter_count = 0
    stall_count = 0
    last_line_search_t = 0.3

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        iter_count += 1

        # Line search for best blend factor
        direction = g_lp - a

        def obj_at_t(t_val):
            a_try = a + t_val * direction
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return 1e10
            a_try = a_try / s_try
            return compute_obj(a_try)

        # Try a few t values around last good value and do golden section
        t_candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]
        if last_line_search_t > 0:
            t_candidates.extend([
                max(0.01, last_line_search_t * 0.5),
                max(0.01, last_line_search_t * 0.8),
                min(1.0, last_line_search_t * 1.2),
                min(1.0, last_line_search_t * 1.5),
            ])
        t_candidates = sorted(set(max(0.0, min(1.0, t)) for t in t_candidates))

        best_t = 0.0
        best_t_obj = obj_at_t(0.0)
        for t in t_candidates:
            obj_t = obj_at_t(t)
            if obj_t < best_t_obj:
                best_t_obj = obj_t
                best_t = t

        # If best t is interior, do a finer search
        if 0.05 < best_t < 0.95:
            result = minimize_scalar(obj_at_t, bounds=(max(0, best_t - 0.15), min(1.0, best_t + 0.15)),
                                     method='bounded', options={'xatol': 1e-4})
            if result.success and result.fun < best_t_obj:
                best_t = result.x
                best_t_obj = result.fun

        # Apply update with optional momentum
        if use_momentum:
            v = beta_mom * v + best_t * direction
            a_new = a + v
        else:
            a_new = a + best_t * direction

        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a
            v = np.zeros(n) if use_momentum else None
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            last_line_search_t = best_t
            if verbose and iter_count % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t:.4f}, t={elapsed:.0f}s")
        else:
            # Try without momentum
            a_new2 = a + best_t * direction
            a_new2 = np.maximum(a_new2, 0)
            s_new2 = float(np.sum(a_new2))
            if s_new2 > 1e-10:
                a_new2 = a_new2 / s_new2
                obj_new2 = compute_obj(a_new2)
                if obj_new2 < best_obj - 1e-12:
                    best_obj = obj_new2
                    best_a = a_new2.copy()
                    a = a_new2
                    stall_count = 0
                    last_line_search_t = best_t
                    if verbose and iter_count % 20 == 0:
                        print(f"    LP it{iter_count} (no mom): obj={best_obj:.8f}, t={best_t:.4f}")
                else:
                    stall_count += 1
            else:
                stall_count += 1

            if stall_count > 3:
                # Perturb to escape
                noise = rng.normal(0, 0.01, n)
                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                v = np.zeros(n) if use_momentum else None
                stall_count = 0
                last_line_search_t = 0.1

            if iter_count > 200 and stall_count > 5:
                break

        # Decay momentum
        if use_momentum and v is not None:
            v *= 0.999

    return best_a


def gradient_descent_smoothed(a, time_budget, tau_init=50.0, tau_end=2000.0, verbose=False):
    """
    Nesterov accelerated gradient descent on the log-sum-exp smoothed objective.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        # Cosine annealing for tau
        tau = tau_init + (tau_end - tau_init) * 0.5 * (1 - np.cos(np.pi * progress))

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)
        g_norm = float(np.linalg.norm(g))

        # Adaptive learning rate
        if g_norm > 1e-15:
            adaptive_lr = 1.0 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.05 * base_lr), 2.0 * base_lr)
            lr = 0.8 * lr + 0.2 * adaptive_lr

        # Barzilai-Borwein
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                bb_lr = min(max(bb_lr, 1e-8), 4.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb_lr

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-9, lr * 0.99995)

        # Reset if stuck
        if it - best_it > 50000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr
            best_it = it

        if verbose and it % 20000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a


def fourier_perturbation(a, rng, strength=0.03):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 4
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.2) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 3:
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Two bumps
        c1 = n // 4
        c2 = 3 * n // 4
        w = max(1, n // 10)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 7:
        # Arcsine-like (optimal for some flat spectrum problems)
        x = np.linspace(0, 1, n)
        a = 1.0 / np.sqrt(x * (1 - x) + 1e-8)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

            a_r = diverse_init(n_current, rng, mode=1)
            blend = 0.5
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.15, 80.0)
        if alloc > 20:
            # LP then GD
            half = alloc * 0.55
            a_r = lp_iteration(a_r, time_budget=half, verbose=False,
                               use_momentum=True, seed=2000 + r * 100)
            remaining = deadline - time.monotonic()
            gd_alloc = min(remaining * 0.08, 35.0)
            if gd_alloc > 10:
                a_r = gradient_descent_smoothed(a_r, time_budget=gd_alloc,
                                                tau_init=30.0, tau_end=800.0)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau GD
    if remaining > 30:
        alloc = min(remaining * 0.2, 100.0)
        a_final = gradient_descent_smoothed(best_seq, time_budget=alloc,
                                            tau_init=100.0, tau_end=5000.0, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final GD: obj={best_score:.8f} **")
        else:
            print(f"  Final GD: obj={obj_final:.8f}")

    # More LP rounds
    remaining = deadline - time.monotonic()
    for lp_rnd in range(4):
        if remaining <= 15:
            break
        alloc = min(remaining * 0.25, 80.0)
        if alloc < 10:
            break
        a_lp = lp_iteration(best_seq, time_budget=alloc, verbose=False,
                            use_momentum=(lp_rnd % 2 == 0), seed=5000 + lp_rnd * 200)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP {lp_rnd}: obj={best_score:.8f} **")
        else:
            if lp_rnd < 3:
                best_seq = fourier_perturbation(best_seq, rng, strength=0.005)
            else:
                break

    # Ultra-high tau GD
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.3, 60.0)
        a_ult = gradient_descent_smoothed(best_seq, time_budget=alloc,
                                          tau_init=200.0, tau_end=8000.0)
        obj_ult = compute_obj(a_ult)
        if obj_ult < best_score:
            best_score = obj_ult
            best_seq = a_ult.copy()
            print(f"  Ultra GD: obj={best_score:.8f} **")
        else:
            print(f"  Ultra GD: obj={obj_ult:.8f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
