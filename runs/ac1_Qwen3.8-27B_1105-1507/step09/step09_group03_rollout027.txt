```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov accelerated smoothed gradient descent with adaptive tau,
peak-targeted coordinate refinement, multi-n exploration, and diverse restarts.
Key: FFT-based gradient of log-sum-exp smoothed max convolution,
with backtracking line search and periodic restarts.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_grad(a, tau):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*conv(a,a)[k])) w.r.t. a.
    Returns (gradient, conv_array, weights_p).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv, p


def nesterov_smoothed_descent(a_init, time_budget, tau_start=30.0, tau_end=2000.0,
                              verbose=False, seed=42):
    """
    Nesterov accelerated gradient descent on smoothed max-conv objective over simplex.
    Uses adaptive tau, backtracking, and periodic restarts.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    y = a.copy()
    it = 0
    best_it = 0
    restart_count = 0
    max_iters_no_improve = 500

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.95))
        # Exponential tau schedule
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.6)

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute gradient at y
        g, conv, p = smoothed_grad(y, tau)

        # Backtracking line search
        step_lr = lr
        a_new = a + step_lr * (-g)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        backtrack_count = 0
        while obj_new >= best_obj - 1e-15 and backtrack_count < 8:
            step_lr *= 0.5
            a_new = a - step_lr * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
            else:
                a_new = best_a.copy()
                break
            obj_new = compute_obj(a_new)
            backtrack_count += 1

        if obj_new < best_obj - 1e-14:
            obj_new = max(obj_new, compute_obj(a_new))
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        # Update Nesterov velocity
        v = beta * v + (a - a_new) / max(step_lr, 1e-12) * step_lr
        # Simpler: v = beta * v - (a_new - a) / step_lr * step_lr
        # Actually standard Nesterov: v_new = beta * v + (a_old - a_new)
        v = beta * v + (a - a_new)
        a = a_new
        it += 1

        # Decay learning rate slowly
        lr = max(0.5 / np.sqrt(n), lr * 0.99995)

        # Restart if stuck
        if it - best_it > max_iters_no_improve:
            restart_count += 1
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)
            best_it = it
            # Perturb
            rng = np.random.default_rng(seed + restart_count * 1000)
            scale = 0.02 * min(restart_count, 5)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s_a = float(np.sum(a))
            if s_a > 1e-10:
                a = a / s_a
            else:
                a = best_a.copy()
            max_iters_no_improve = max(200, max_iters_no_improve // 2)

        if verbose and it % 500 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}, restarts={restart_count}")

    return best_a


def peak_surgery(a_init, time_budget, verbose=False):
    """
    Local refinement: identify top peak, try coordinate moves to reduce it.
    Uses greedy selection of (source, target) pairs.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_k = int(np.argmax(conv))

        # Find contributing indices to this peak
        i_min = max(0, peak_k - (n - 1))
        i_max = min(n - 1, peak_k)

        if i_min > i_max:
            break

        # Get contributions
        contribs = []
        for i in range(i_min, i_max + 1):
            j = peak_k - i
            if 0 <= j < n:
                c = a[i] * a[j]
                contribs.append((c, i, j))
        contribs.sort(reverse=True)

        improved = False
        # Try reducing top contributors
        for c_val, i, j in contribs[:5]:
            # Try moving mass from the larger of a[i], a[j] to a random other position
            src = i if a[i] >= a[j] else j
            for delta_frac in [0.03, 0.01, 0.005, 0.002]:
                delta = delta_frac * a[src]
                if delta <= 0 or delta >= a[src]:
                    continue
                # Try several targets
                targets = [src - max(1, n // 10) % n, (src + n // 10) % n,
                           rng.integers(0, n), n - 1 - src, (src + n // 3) % n]
                for tgt in targets:
                    tgt = int(tgt) % n
                    if tgt == src:
                        continue
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[tgt] += delta
                    if a_try[src] < 0:
                        continue
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                # Try larger random perturbation
                scale = 0.01 * min(stall, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = float(np.sum(a))
                if s_a > 1e-10:
                    a = a / s_a
                stall = 0
                if stall > 100:
                    break

        if verbose and (it if 'it' in dir() else 0) % 500 == 0:
            pass

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x) ** 2
        a = np.maximum(a, 1e-8)
        a = a / a.sum()
    elif mode == 3:
        # Triangle
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2 * x - 1)
        a = np.maximum(a, 1e-8)
        a = a / a.sum()
    elif mode == 4:
        # Bimodal
        centers = [0.25, 0.75]
        widths = [0.1, 0.1]
        x = np.linspace(0, 1, n)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Archimedean
        x = np.linspace(0.01, 1, n)
        a = 1.0 / np.sqrt(x + 0.1)
        a = a / a.sum()
    elif mode == 6:
        # sin shape
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 1.5
        a = np.maximum(a, 1e-8)
        a = a / a.sum()
    else:
        # Logarithmic
        x = np.linspace(0.01, 1, n)
        a = 1.0 / (1.0 + np.log(1 + 5 * x))
        a = a / a.sum()
    return a


def multi_scale_nesterov(a_init, time_budget, verbose=False):
    """
    Multi-scale Nesterov: start at low tau (coarse), increase to high tau (fine).
    Uses multiple restarts within each scale.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Schedule of (tau_start, tau_end, fraction_of_time)
    scales = [
        (20.0, 100.0, 0.2),
        (80.0, 400.0, 0.3),
        (300.0, 1500.0, 0.35),
        (800.0, 3000.0, 0.15),
    ]

    for tau_s, tau_e, frac in scales:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 20:
            break
        alloc = frac * time_budget
        alloc = min(alloc, remaining)

        # Run Nesterov for this scale
        a_cur = nesterov_smoothed_descent(best_a, time_budget=alloc,
                                          tau_start=tau_s, tau_end=tau_e,
                                          verbose=False)
        obj_cur = compute_obj(a_cur)
        if obj_cur < best_obj - 1e-14:
            best_obj = obj_cur
            best_a = a_cur.copy()
            if verbose:
                print(f"    MS thu tau=[{tau_s},{tau_e}]: obj={best_obj:.8f}")
        else:
            if verbose:
                print(f"    MS thu tau=[{tau_s},{tau_e}]: obj={obj_cur:.8f} (no improvement)")

    return best_a


def propose_candidate():
    """
    Multi-phase optimization combining Nesterov smoothed gradient descent,
    peak surgery, multi-n exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + 1450.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(2024)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_parent = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Parent not available: {e}")
        n_parent = 800

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        n_parent = 800
        print(f"Default init: obj={best_score:.8f}")

    # === Phase 1: Quick Nesterov from parent ===
    print(f"\n=== Phase 1: Quick Nesterov (parent n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.12, 170.0)
    if p1_budget > 20:
        a1 = nesterov_smoothed_descent(best_seq, time_budget=p1_budget,
                                       tau_start=30.0, tau_end=500.0, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj1:.8f}")

    # === Phase 2: Peak surgery ===
    print(f"\n=== Phase 2: Peak Surgery ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.06, 90.0)
    if p2_budget > 15:
        a2 = peak_surgery(best_seq, time_budget=p2_budget)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj2:.8f}")

    # === Phase 3: Multi-scale Nesterov ===
    print(f"\n=== Phase 3: Multi-Scale Nesterov ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.3, 430.0)
    if p3_budget > 30:
        a3 = multi_scale_nesterov(best_seq, time_budget=p3_budget, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj3:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(200, n_current - 100), max(300, n_current - 200),
        n_current + 100, n_current + 200, n_current + 300,
        500, 700, 900, 1100, 1300, 1500
    ]))
    # Filter to reasonable range and distinct from current
    n_options = [n for n in n_options if 300 <= n <= 2000 and abs(n - n_current) > 80]
    n_options = n_options[:5]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick Nesterov on this length
        alloc = min(remaining * 0.15, 80.0)
        if alloc > 15:
            a_try = nesterov_smoothed_descent(a_try, time_budget=alloc,
                                              tau_start=50.0, tau_end=800.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = r * 2  # Use even modes: 0, 2, 4, 6
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with current best
        blend = 0.4
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.2, 120.0)
        if alloc > 20:
            a_r = nesterov_smoothed_descent(a_r, time_budget=alloc,
                                            tau_start=30.0, tau_end=1000.0)
            # Then peak surgery
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20:
                a_r = peak_surgery(a_r, time_budget=min(remaining2 * 0.3, 30.0))

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final high-tau polish ===
    print(f"\n=== Phase 6: Final High-Tau Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 150.0)
        a_final = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                            tau_start=500.0, tau_end=5000.0, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final polish: obj={best_score:.8f} **")
        else:
            print(f"  Final polish: obj={obj_final:.8f}")

    # === Phase 7: Final peak surgery ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 50.0)
        a_ps = peak_surgery(best_seq, time_budget=alloc)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  Final peak surgery: obj={best_score:.8f} **")
        else:
            print(f"  Final peak surgery: obj={obj_ps:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
