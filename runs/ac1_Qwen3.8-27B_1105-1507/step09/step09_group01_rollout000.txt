"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a hybrid approach: aggressive LP-based direction finding with refined line
search, coordinate descent sweeps, FFT-accelerated gradient polishing, and
multi-length exploration with diverse restarts.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, M_target=None):
    """
    Solve LP: min M s.t. A@ g <= M, sum(g)=1, g>=0.
    Where A @ g = conv(a, g).
    """
    n = len(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCIP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue
    
    return None, None


def coordinate_descent_step(a, max_sweeps=1, verbose=False):
    """
    Perform coordinate descent on the objective 2n*max(conv)/(sum)^2.
    For each coordinate, find optimal value via golden section search.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s  # normalize to sum = 1
    
    best_obj = compute_obj(a)
    
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            # Compute the max of unaffected convolution entries
            # a[i] affects conv[k] for k in [i, i+n-1] intersect [0, 2n-2]
            k_start = max(0, i)
            k_end = min(2 * n - 2, i + n - 1)
            
            # Compute current conv
            conv = compute_conv(a)
            
            # Unaffected max
            if k_start > 0 or k_end < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_start:k_end+1] = False
                unaffected_max = float(np.max(conv[mask])) if mask.any() else 0.0
            else:
                unaffected_max = 0.0
            
            # For affected positions, C_k(t) = t * a[k-i] + d_k (or t^2 + d_{2i} for k=2i)
            # Precompute coefficients
            coeffs = []  # (slope, intercept) pairs for linear parts
            quad_intercept = None  # for the quadratic term at k=2i
            
            for k in range(k_start, k_end + 1):
                idx = k - i  # this is the index into a, so a[idx] is the coefficient of t
                if k == 2 * i:
                    # quadratic: t^2 + d
                    d = conv[k] - a[i] * a[i]
                    quad_intercept = d
                else:
                    # linear: t * a[idx] + d
                    d = conv[k] - a[i] * a[idx]
                    coeffs.append((a[idx], d))
            
            # Also include unaffected_max as a "flat" lower bound
            # The objective at t is:
            # 2n * max(max(coeffs linear), quad, unaffected_max) / (sum_new)^2
            # where sum_new = s_old - a[i] + t = 1 - a[i] + t (since s=1)
            
            ai = a[i]
            
            def obj_at_t(t):
                if t < 0:
                    return 1e10
                new_sum = 1.0 - ai + t
                if new_sum < 1e-10:
                    return 1e10
                # Compute max conv at this t
                max_val = unaffected_max
                for slope, intercept in coeffs:
                    val = t * slope + intercept
                    if val > max_val:
                        max_val = val
                if quad_intercept is not None:
                    val = t * t + quad_intercept
                    if val > max_val:
                        max_val = val
                return 2.0 * n * max_val / (new_sum * new_sum)
            
            # Golden section search for t in [0, t_max]
            # t_max: reasonable upper bound. Since a[i] is part of a probability vector,
            # t shouldn't exceed much more than 1 (or a few times a[i])
            t_max = max(2.0 * ai, 0.5 / n) * 5
            t_max = max(t_max, 0.1)
            
            # Quick check: is t=0 already optimal?
            f0 = obj_at_t(0)
            if f0 >= best_obj - 1e-15:
                # Try a small perturbation
                t_trial = min(t_max, ai * 0.5)
                if t_trial > 0:
                    ft = obj_at_t(t_trial)
                    if ft < best_obj:
                        # Do golden section
                        gr = (np.sqrt(5.0) - 1.0) / 2.0
                        lo, hi = 0.0, t_max
                        c = hi - gr * (hi - lo)
                        d = lo + gr * (hi - lo)
                        fc = obj_at_t(c)
                        fd = obj_at_t(d)
                        for _ in range(30):
                            if hi - lo < 1e-10:
                                break
                            if fc < fd:
                                hi = d
                                d = c
                                fd = fc
                                c = hi - gr * (hi - lo)
                                fc = obj_at_t(c)
                            else:
                                lo = c
                                c = d
                                fc = fd
                                d = lo + gr * (hi - lo)
                                fd = obj_at_t(d)
                        t_opt = (lo + hi) / 2.0
                        new_obj = obj_at_t(t_opt)
                        if new_obj < best_obj - 1e-14:
                            a[i] = t_opt
                            best_obj = new_obj
                            improved = True
                continue
            
            # Do golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo, hi = 0.0, t_max
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = obj_at_t(c)
            fd = obj_at_t(d)
            for _ in range(35):
                if hi - lo < 1e-10:
                    break
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = obj_at_t(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = obj_at_t(d)
            
            t_opt = (lo + hi) / 2.0
            new_obj = obj_at_t(t_opt)
            if new_obj < best_obj - 1e-14:
                a[i] = t_opt
                best_obj = new_obj
                improved = True
        
        if not improved and sweep > 0:
            break
    
    return a


def lp_polish(a, time_budget=100.0, verbose=False, seed=42):
    """
    LP-based polishing: find direction g that minimizes max(a*g),
    then do line search on (1-t)*a + t*g.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    iter_count = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed > time_budget * 0.95:
            break
        
        # Solve LP
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Try multiple blend factors
        best_t = 0
        best_blend_obj = best_obj
        
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_t = t
        
        # Refine with golden section around best_t
        if best_t > 0:
            lo = max(0, best_t - 0.1)
            hi = min(1.0, best_t + 0.1)
            if hi - lo > 0.01:
                def blend_obj(t):
                    a_try = (1.0 - t) * best_a + t * g_lp
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        return 1e10
                    a_try = a_try / s_try
                    return compute_obj(a_try)
                
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c = hi - gr * (hi - lo)
                d = lo + gr * (hi - lo)
                fc = blend_obj(c)
                fd = blend_obj(d)
                for _ in range(20):
                    if hi - lo < 0.005:
                        break
                    if fc < fd:
                        hi = d
                        d = c
                        fd = fc
                        c = hi - gr * (hi - lo)
                        fc = blend_obj(c)
                    else:
                        lo = c
                        c = d
                        fc = fd
                        d = lo + gr * (hi - lo)
                        fd = blend_obj(d)
                t_refined = (lo + hi) / 2.0
                obj_refined = blend_obj(t_refined)
                if obj_refined < best_blend_obj:
                    best_blend_obj = obj_refined
                    best_t = t_refined
        
        iter_count += 1
        
        if best_blend_obj < best_obj - 1e-14:
            a_try = (1.0 - best_t) * best_a + best_t * g_lp
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            best_obj = best_blend_obj
            best_a = a_try.copy()
            a = best_a.copy()
            no_improve = 0
            if verbose and iter_count % 3 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t:.3f}, M={M_lp:.6f}")
        else:
            no_improve += 1
            if no_improve >= 3:
                # Perturb
                scale = 0.01 * min(no_improve, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                if iter_count > 40:
                    break
    
    return best_a


def nesterov_gradient_descent(a, time_budget=60.0, tau_init=5.0, tau_end=200.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(1e-10, time_budget))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient via cross-correlation
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]
        
        v = beta * v - lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9998)
        
        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj

                n_target = parent.size

    except Exception:
        pass
    
    if best_seq is None:
        best_seq = np.ones(n_target) / n_target

        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    print(f"\n=== Phase 1: Quick GD Warmup ===")
