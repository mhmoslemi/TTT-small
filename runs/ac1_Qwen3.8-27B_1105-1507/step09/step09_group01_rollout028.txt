def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix_v2(a):
    """Faster build using numpy operations."""
    n = len(a)
    # A[k, m] = a[k-m] if 0 <= k-m < n
    # We can build this using the fact that each column m has a and shifted copies
    # Column m: A[k, m] = a[k-m] for k = m, m+1, ..., m+n-1
    A = np.zeros((2 * n - 1, n))
    for m in range(n):
        A[m:m+n, m] = a
    return A


def solve_lp(a, M_target=None, maximize_sum=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_value) or (None, None) if infeasible.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_v2(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    objective = cp.Minimize(M)
    prob = cp.Problem(objective, constraints)

    # Try solvers in order of preference
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, float('inf')
        except Exception:
            continue

    return None, None

def golden_section_search(f, a_lo, a_hi, tol=0.01, max_iter=30):
    """Golden section search for minimum of f on [a_lo, a_hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if abs(a_hi - a_lo) < tol:
            break
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = f(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = f(d)
    return (a_lo + a_hi) / 2.0

def local_search(a, time_budget=60.0, verbose=False, seed=42):

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        # Sparse: random subset
        k = max(1, n // 3)
        a = np.zeros(n)
        indices = rng.choice(n, size=k, replace=False)
        a[indices] = 1.0
        a = a / a.sum()
    elif mode == 5:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 6:
        # Periodic pattern
        period = max(2, n // 20)
        a = np.ones(n)
        a[period - 1::period] = 0.3
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based best-reply iteration with symmetric constraint exploitation,
multi-n exploration, golden-section blending, oversampling, and local search.
Key improvements: symmetric LP (4x faster), wider n range, more iterations,
convergence-aware stopping, and periodic perturbative restarts.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for m in range(n):
        A[m:m + n, m] = a
    return A


def build_sym_conv_matrix(a):
    """
    Build matrix for symmetric LP.
    a is symmetric of length n. Returns A_sym (n x half_n) where
    A_sym @ g_free = conv(a, g_sym)[:n] (first n entries of convolution).
    g_free are the free variables of the symmetric sequence g.
    """
    n = len(a)
    half = (n + 1) // 2  # number of free variables for symmetric g
    A_sym = np.zeros((n, half))
    # conv(a, g)[k] = sum_j a[j] * g[k-j] for k = 0..n-1
    # g is symmetric: g[j] = g[n-1-j]
    # g_free[i] = g[i] for i = 0..half-1
    # g[j] = g_free[j] if j < half, else g_free[n-1-j]
    for k in range(n):
        for j in range(n):
            if 0 <= k - j < n:
                coeff = a[k - j]
                if coeff == 0:
                    continue
                gj = j if j < half else (n - 1 - j)
                A_sym[k, gj] += coeff
    return A_sym


def make_symmetric(a):
    """Make a sequence symmetric: a[i] = a[n-1-i]."""
    n = len(a)
    result = np.zeros(n)
    half = (n + 1) // 2
    result[:half] = a[:half]
    if n % 2 == 0:
        result[half:] = result[half - 1::-1]
    else:
        result[half:] = result[half - 2::-1]
    return np.maximum(result, 0)


def solve_lp_full(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Full (non-symmetric)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, float('inf')
        except Exception:
            continue
    return None, None


def solve_lp_sym(a):
    """
    Solve symmetric LP: min M s.t. conv(a,g)[:n] <= M, sum(g)=1, g>=0, g symmetric.
    a must be symmetric. Returns full symmetric g and M.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A_sym = build_sym_conv_matrix(a_norm)
    half = (n + 1) // 2

    g_free = cp.Variable(half, nonneg=True)
    M = cp.Variable()

    # Build the sum constraint: sum of full g = 1
    if n % 2 == 0:
        # g = [g0, g1, ..., g_{half-1}, g_{half-1}, ..., g0]
        # sum = 2 * sum(g_free)
        sum_constraint = 2.0 * cp.sum(g_free) == 1
    else:
        # g = [g0, g1, ..., g_{half-1}, ..., g1, g0]
        # sum = 2 * sum(g_free) - g_free[half-1]
        sum_constraint = 2.0 * cp.sum(g_free) - g_free[half - 1] == 1

    constraints = [A_sym @ g_free <= M, sum_constraint]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_free.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gf = np.maximum(g_free.value, 0)
                # Reconstruct full symmetric g
                g_full = np.zeros(n)
                if n % 2 == 0:
                    g_full[:half] = gf
                    g_full[half:] = gf[::-1]
                else:
                    g_full[:half] = gf
                    g_full[half:] = gf[-2::-1]
                s = float(np.sum(g_full))
                if s > 1e-10:
                    return g_full / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, float('inf')
        except Exception:
            continue
    return None, None


def golden_section_search(f, a_lo, a_hi, tol=0.02, max_iter=25):
    """Golden section search for minimum of f on [a_lo, a_hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if abs(a_hi - a_lo) < tol:
            break
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = f(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = f(d)
    return (a_lo + a_hi) / 2.0


def lp_iteration(a, time_budget=100.0, verbose=False, seed=42, use_sym=True, max_iter=500):
    """
    LP-based best-reply iteration with golden-section blending.
    Optionally enforces symmetry for faster LP solves.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    # If using symmetry, start with symmetric version
    if use_sym:
        a_sym = make_symmetric(a)
        # Only use symmetric if it doesn't hurt too much
        obj_sym = compute_obj(a_sym)
        obj_orig = compute_obj(a)
        if obj_sym < obj_orig * 1.001:
            a = a_sym

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    no_improve = 0
    last_t = 0.5

    while time.monotonic() - start < time_budget and iter_count < max_iter:
        elapsed = time.monotonic() - start

        # Solve LP
        if use_sym:
            g_lp, M_lp = solve_lp_sym(best_a)
        else:
            g_lp, M_lp = solve_lp_full(best_a)

        if g_lp is None:
            # Fallback to full LP
            g_lp, M_lp = solve_lp_full(best_a)
            if g_lp is None:
                break

        # Line search for blend factor
        def blend_obj(t):
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return 1e10
            return compute_obj(a_try / s_try)

        t_opt = golden_section_search(blend_obj, 0.0, 1.0, tol=0.02, max_iter=20)

        a_try = (1.0 - t_opt) * best_a + t_opt * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            a_try = best_a.copy()
        else:
            a_try = a_try / s_try
        obj_try = compute_obj(a_try)

        iter_count += 1

        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            last_t = t_opt
            no_improve = 0
            if verbose and iter_count % 5 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={t_opt:.3f}, M={M_lp:.6f}, el={elapsed:.0f}s")
        else:
            no_improve += 1
            if no_improve == 1:
                # Try full replacement
                obj_full = compute_obj(g_lp)
                if obj_full < best_obj - 1e-14:
                    best_obj = obj_full
                    best_a = g_lp.copy()
                    no_improve = 0
            elif no_improve == 2:
                # Try smaller step
                t_small = max(0.05, last_t * 0.3)
                a_try2 = (1.0 - t_small) * best_a + t_small * g_lp
                a_try2 = np.maximum(a_try2, 0)
                a_try2 = a_try2 / a_try2.sum()
                obj_try2 = compute_obj(a_try2)
                if obj_try2 < best_obj - 1e-14:
                    best_obj = obj_try2
                    best_a = a_try2.copy()
                    no_improve = 0
            elif no_improve >= 3:
                # Perturb and retry
                scale = 0.004 + 0.002 * min(no_improve, 10)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                if use_sym:
                    a_pert = make_symmetric(a_pert)
                a_pert = a_pert / a_pert.sum()
                no_improve = 0
                if iter_count > 80:
                    break
                continue

        # Convergence check: if LP gives very small improvement, stop
        current_peak = float(np.max(compute_conv(best_a)))
        if M_lp < current_peak * 1.0005 and no_improve > 0:
            if verbose:
                print(f"    Converged it{iter_count}: obj={best_obj:.8f}, el={elapsed:.0f}s")
            break

    return best_a


def local_search(a, time_budget=60.0, seed=42):

    n = len(a)

        return a

            a_try[idx] = max(0.0, a_try[idx] + delta)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        k = max(1, n // 4)
        a = np.zeros(n)
        indices = rng.choice(n, size=k, replace=False)
        a[indices] = 1.0
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 6:
        # Alternating pattern
        a = np.ones(n)
        a[1::3] = 0.3
        a[2::3] = 0.3
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a
