"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Mirror descent (multiplicative weights) with KL divergence on the simplex,
combined with Nesterov accelerated gradient on log-sum-exp smoothed objective,
multiple diverse initializations (including tapered and structured), and 
coordinate descent polish. Key innovations: natural multiplicative updates for 
non-negativity, FFT-based gradient computation, and adaptive initialization.

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def mirror_descent(a, time_budget, tau_init=2.0, tau_end=200.0, verbose=False):
    """
    Mirror descent with KL divergence (multiplicative weights) on the simplex.
    Minimizes the LSE-smoothed max of conv(a,a).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    # Precompute FFT size
    N = 1
    while N < 3 * n:
        N <<= 1

    # Step size schedule: eta_t = C / sqrt(t+1)
    C = 2.0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.95))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)
        eta = C / np.sqrt(it + 1)

        if it % 5000 == 0 and verbose:
            print(f"  MD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, eta={eta:.4f}, t={elapsed:.0f}s")

        # Compute convolution
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # LSE gradient
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient: g_j = 2 * sum_k p_k * a_{k-j} (circular-ish, but we use linear)
        # g = 2 * convolve(p, a) truncated to n
        P = rfft(p, N)
        A = rfft(a, N)
        g_full = irfft(P * A, N)  # This gives conv(p, a)
        # Wait, this is p * a (convolution), but we need sum_k p_k * a_{k-j}
        # conv(p,a)[j] = sum_k p[k]*a[j-k], which is what we want (shifted)
        # Actually conv(p,a)[j] = sum_{k} p[k] * a[j-k]
        # We want g_j = 2 * sum_k p[k] * a[k-j] = 2 * conv(a, p)[j] = 2 * conv(p, a)[j]
        # (convolution is commutative for real sequences)
        # So g = 2 * conv(p, a)[:n]
        g = 2.0 * g_full[:n]
        
        # Mirror descent update: a_new = a * exp(-eta * g) / Z
        log_a_new = np.log(np.maximum(a, 1e-300)) - eta * g
        log_a_new -= np.max(log_a_new)  # for numerical stability
        a_new = np.exp(log_a_new)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

    return best_a


def nesterov_smoothed(a, time_budget, tau_init=5.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on LSE-smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 3 * n:
        N <<= 1

    beta = 0.85
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        if it % 5000 == 0 and verbose:
            print(f"  NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        # Nesterov extrapolation
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        P = rfft(p, N)
        A = rfft(y, N)
        g_full = irfft(P * A, N)
        g = 2.0 * g_full[:n]

        # Gradient step
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.99995)

        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            beta = 0.8
            best_it = it

    return best_a


def coordinate_descent(a, time_budget, verbose=False):
    """
    Coordinate descent: for each coordinate, find the optimal value
    given the others fixed. Uses the fact that the objective is the max
    of quadratic functions.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    improvement = True

    while time.monotonic() - start < time_budget and improvement:
        improvement = False
        # Try each coordinate (or a random subset for speed)
        if n > 200:
            indices = np.random.default_rng(it).permutation(n)[:min(n, 100)]
        else:
            indices = np.arange(n)

        for j in indices:
            # Current convolution
            conv = compute_conv(a)
            max_k = np.argmax(conv)
            max_val = conv[max_k]
            
            # If we change a_j to a new value x, how does conv change?
            # conv_new[k] = conv[k] + 2*x*a_{k-j} - 2*a_j*a_{k-j} for relevant k
            # Actually: conv_new[k] = conv[k] - a_j*(a_{k-j} + a_{k-j}) + x*(a_{k-j} + a_{k-j})
            # Wait, conv[k] = sum_i a_i * a_{k-i}
            # If we change a_j to x: conv_new[k] = conv[k] - a_j*a_{k-j} - a_{k-j}*a_j + x*a_{k-j} + a_{k-j}*x
            # = conv[k] - 2*a_j*a_{k-j} + 2*x*a_{k-j}
            # (when both j and k-j are in [0,n-1])
            
            # For the max to decrease, we want to reduce the peaks.
            # Simple heuristic: try a_j = 0, a_j = 2*current, a_j = current*0.5
            old_val = a[j]
            for factor in [0.0, 0.5, 0.75, 1.25, 1.5, 2.0]:
                a[j] = old_val * factor
                s_test = float(np.sum(a))
                if s_test < 1e-10:
                    a[j] = old_val
                    continue
                a_test = a / s_test
                obj_test = compute_obj(a_test)
                if obj_test < best_obj - 1e-12:
                    best_obj = obj_test
                    best_a = a_test.copy()
                    a = a_test.copy()
                    improvement = True
                else:
                    a[j] = old_val

        it += 1
        if verbose and it % 50 == 0:
            print(f"  CD it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Raised cosine (Hann-like)
        x = np.arange(n)
        a = 0.5 * (1 + np.cos(np.pi * x / (n - 1)))
        a = a / a.sum()
    elif mode == 3:
        # Triangular
        x = np.arange(n)
        mid = (n - 1) / 2
        a = 1 - np.abs(x - mid) / mid
        a = a / a.sum()
    elif mode == 4:
        # Gaussian
        x = np.arange(n)
        mid = (n - 1) / 2
        sigma = n / 4
        a = np.exp(-(x - mid)**2 / (2 * sigma**2))
        a = a / a.sum()
    elif mode == 5:
        # Two bumps
        x = np.arange(n)
        a = np.exp(-(x - n*0.3)**2 / (2*(n*0.1)**2)) + np.exp(-(x - n*0.7)**2 / (2*(n*0.1)**2))
        a = a / a.sum()
    elif mode == 6:
        # Power law (favor center)
        x = np.arange(n)
        mid = (n - 1) / 2
        a = (1 + (x - mid)**2 / (n**2)) ** (-1.5)
        a = a / a.sum()
    elif mode == 7:
        # Optimal taper (approximately)
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 1e-10)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Mirror descent (multiplicative weights) with KL divergence on the simplex
for fast, natural constraint handling; Nesterov accelerated gradient on LSE-smoothed
objective for fine refinement; multiple diverse structured initializations (tapered,
Gaussian, power-law) to avoid local minima; adaptive step-size with backtracking;
coordinate descent polish. Uses FFT-based gradient computation for O(n log n) per step.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Assumes a>=0."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lse_gradient(a, tau, N):
    """Compute gradient of LSE-smoothed max of conv(a,a) at temperature tau.
    Returns (gradient, conv, max_conv)."""
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    P = rfft(p, N)
    A = rfft(a, N)
    g_full = irfft(np.conj(P) * A, N)
    g = 2.0 * g_full[:len(a)]
    return g, conv, max_conv


def mirror_descent(a, time_budget, tau_init=2.0, tau_end=200.0, verbose=False):
    """
    Mirror descent with KL divergence (multiplicative weights) on the simplex.
    Minimizes the LSE-smoothed max of conv(a,a).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    N = 1
    while N < 4 * n:
        N <<= 1

    C = 3.0
    eta = C

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget * 0.95, 1e-6))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)
        eta = C / np.sqrt(it + 1)

        if it % 10000 == 0 and verbose:
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, eta={eta:.4f}, t={elapsed:.0f}s")

        g, conv, max_conv = lse_gradient(a, tau, N)

        # Mirror descent: a_new = a * exp(-eta * g) / Z
        log_scale = -eta * g
        log_scale -= np.max(log_scale)
        a_new = a * np.exp(log_scale)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10 or not np.isfinite(s_new):
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if np.isfinite(obj_new) and obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Backtrack: reduce eta
            eta *= 0.8

        it += 1

    return best_a


def nesterov_smoothed(a, time_budget, tau_init=5.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on LSE-smoothed objective with simplex projection."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 4 * n:
        N <<= 1

    beta = 0.85
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget * 0.9, 1e-6))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        if it % 10000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.4f}, t={elapsed:.0f}s")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv, max_conv = lse_gradient(y, tau, N)

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if np.isfinite(obj_new) and obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall_count = 0
        else:
            stall_count += 1

        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.99997)

        if stall_count > 500:
            lr *= 0.7
            v *= 0.5
            stall_count = 0

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            beta = 0.8
            best_it = it

    return best_a


def coordinate_descent(a, time_budget, verbose=False):
    """
    Coordinate descent: try multiplicative changes to individual coordinates.
    Efficient for final polish.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    rng = np.random.default_rng(12345)
    improvement_in_round = True

    factors = [0.0, 0.25, 0.5, 0.75, 1.25, 1.5, 2.0, 3.0]

    while time.monotonic() - start < time_budget and improvement_in_round:
        improvement_in_round = False
        # Random subset of coordinates for speed
        if n > 300:
            indices = rng.permutation(n)[:200]
        else:
            indices = np.arange(n)

        for j in indices:
            old_val = a[j]
            if old_val < 1e-15:
                # Try adding mass
                for f in [1e-4, 1e-3, 1e-2]:
                    a[j] = f * best_a.sum() / n
                    s_test = float(np.sum(a))
                    a_test = a / s_test
                    obj_test = compute_obj(a_test)
                    if obj_test < best_obj - 1e-12:
                        best_obj = obj_test
                        best_a = a_test.copy()
                        a = a_test.copy()
                        improvement_in_round = True
                    else:
                        a[j] = 0
            else:
                for f in factors:
                    a[j] = old_val * f
                    s_test = float(np.sum(a))
                    if s_test < 1e-10:
                        a[j] = old_val
                        continue
                    a_test = a / s_test
                    obj_test = compute_obj(a_test)
                    if obj_test < best_obj - 1e-12:
                        best_obj = obj_test
                        best_a = a_test.copy()
                        a = a_test.copy()
                        improvement_in_round = True
                        break
                a[j] = old_val if not improvement_in_round else a[j]
                # Restore if no improvement was found for this j
                if a[j] != best_a[j]:
                    a = best_a.copy()

        it += 1
        if verbose and it % 20 == 0:
            print(f"    CD it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
    elif mode == 2:
        # Raised cosine (Hann)
        a = 0.5 * (1 + np.cos(np.pi * x / max(n - 1, 1)))
    elif mode == 3:
        # Triangular
        mid = (n - 1) / 2.0
        a = 1.0 - np.abs(x - mid) / max(mid, 1.0)
    elif mode == 4:
        # Gaussian
        mid = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-(x - mid) ** 2 / (2 * sigma ** 2))
    elif mode == 5:
        # Two bumps
        a = np.exp(-(x - 0.3 * n) ** 2 / (2 * (0.1 * n) ** 2)) + \
            np.exp(-(x - 0.7 * n) ** 2 / (2 * (0.1 * n) ** 2))
    elif mode == 6:
        # Power law center-weighted
        mid = (n - 1) / 2.0
        a = (1.0 + (x - mid) ** 2 / (n ** 2)) ** (-2.0)
    elif mode == 7:
        # sin^2 taper
        a = np.sin(np.pi * x / max(n - 1, 1)) ** 2
    elif mode == 8:
        # Optimal-ish: minimize edge effects
        mid = (n - 1) / 2.0
        t = np.abs(x - mid) / max(mid, 1.0)
        a = (1.0 - t) ** 1.5 + 0.01
    elif mode == 9:
        # Band-limited random
        freqs = rng.integers(1, 4, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.sin(2 * np.pi * f * x / n + p)
        a = np.maximum(a, 0.05)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def simulated_annealing_jump(a, best_a, scale, rng):
    """Make a random perturbation of scale 'scale' to escape local minima."""
    n = len(a)
    a_new = a * np.exp(rng.normal(0, scale, n))
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return best_a.copy()
    a_new = a_new / s
    return a_new

            break

        # Mirror descent
        a_md = mirror_descent(best_seq, time_budget=alloc * 0.5,
                             tau_init=0.5 * (rnd + 1), tau_end=30.0 * (rnd + 1), verbose=(rnd == 0))
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Round {rnd+1} MD: obj={best_score:.8f} **")

        # Nesterov
        remaining = deadline - time.monotonic()
        alloc2 = min(alloc * 0.5, remaining * 0.4)
        if alloc2 > 10:
            a_ns = nesterov_smoothed(best_seq, time_budget=alloc2,
                                     tau_init=2.0, tau_end=150.0, verbose=(rnd == 0))
            obj_ns = compute_obj(a_ns)
            if obj_ns < best_score:
                best_score = obj_ns
                best_seq = a_ns.copy()
                print(f"  Round {rnd+1} NSD: obj={best_score:.8f} **")

        # Perturb for next round
        if rnd < num_rounds - 1:
            scale = 0.005 * (rnd + 1)
            best_seq = simulated_annealing_jump(best_seq, best_seq, scale, rng)

    # === Phase 4: Targeted peak reduction ===
    print(f"\n=== Phase 4: Targeted Peak Reduction ===")
    remaining = deadline - time.monotonic()
    phase4_budget = remaining * 0.20

    if phase4_budget > 20:
        # Focus on reducing the specific peak(s)
        a = best_seq.copy()
        conv = compute_conv(a)
        max_k = int(np.argmax(conv))
        print(f"  Peak at k={max_k}, value={conv[max_k]:.8f}")

        # Subgradient descent targeting the top peaks
        n = len(a)
        N = 1
        while N < 4 * n:
        N <<= 1
        top_K = 5
        lr = 0.05
        v = np.zeros(n)
        momentum = 0.7
        peak_best = compute_obj(a)
        peak_best_a = a.copy()
        it = 0

        t_end = time.monotonic() + phase4_budget
        while time.monotonic() < t_end:
            conv = compute_conv(a)
            top_idx = np.argsort(conv)[::-1][:top_K]
            peak_vals = conv[top_idx]
            max_pv = peak_vals[0]
            if max_pv < 1e-15:
                break
            weights = np.exp(3.0 * (peak_vals - max_pv) / (max_pv + 1e-15))
            weights /= weights.sum()

            g = np.zeros(n)
            for w, k in zip(weights, top_idx):
                k = int(k)
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                if i_start <= i_end:
                    j_start = k - i_end
                    j_end = k - i_start
                    g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

            v = momentum * v + g
            a_new = a - lr * v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)

            if obj_new < peak_best - 1e-12:
                peak_best = obj_new
                peak_best_a = a_new.copy()
                a = a_new
            else:
                lr *= 0.7
                v *= 0.3
                a = peak_best_a
                if lr < 1e-10:
                    break

            it += 1

        if peak_best < best_score:
            best_score = peak_best
            best_seq = peak_best_a.copy()
            print(f"  ** Peak reduction: obj={best_score:.8f} **")
        else:
            print(f"  Peak reduction: obj={peak_best:.8f} (no improvement)")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau Nesterov (very sharp)
    if remaining > 30:
        alloc = min(remaining * 0.35, 120.0)
        a_ns = nesterov_smoothed(best_seq, time_budget=alloc, tau_init=20.0, tau_end=600.0, verbose=True)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  ** High-tau NSD: obj={best_score:.8f} **")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 90.0)
        a_cd = coordinate_descent(best_seq, time_budget=alloc, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  ** CD: obj={best_score:.8f} **")

    # Final low-tau MD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 60.0)
        a_md = mirror_descent(best_seq, time_budget=alloc, tau_init=0.3, tau_end=10.0)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  ** Final MD: obj={best_score:.8f} **")

    # One more NSD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 50.0)
        a_ns = nesterov_smoothed(best_seq, time_budget=alloc, tau_init=10.0, tau_end=100.0)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  ** Final NSD: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
