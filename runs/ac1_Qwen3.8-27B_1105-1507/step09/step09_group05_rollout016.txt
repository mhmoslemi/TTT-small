"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses exact coordinate descent with golden section search, FISTA smoothed optimization,
LP-based polishing, and diverse multi-length exploration.
Key innovation: exact 1D coordinate minimization exploiting the structure of
the autocorrelation, combined with adaptive LP polishing.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_obj_given_conv(a, conv):
    """Compute objective given precomputed convolution (a should be normalized to sum=1)."""
    n = len(a)
    return 2.0 * n * float(np.max(conv))

    n = len(a)
    ai = a[i]
    S = 1.0 - ai + x  # new sum
    
    if S < 1e-15:
        return np.inf
    
    inv_S2 = 1.0 / (S * S)
    
    # conv_new(0) = (conv_old[0] - ai^2 + x^2) / S^2
    max_val = (conv_old[0] - ai * ai + x * x) * inv_S2
    
    # For k > 0: conv_new(k) = (conv_old[k] + (x - ai) * c_k) / S^2
    # where c_k = a[i+k]*I(i+k<n) + a[i-k]*I(i-k>=0)
    
    # c_k for k=1..n-1-i (only a[i+k] term)
    # c_k for k=1..i (both terms if valid)
    
    # Let's compute all c_k values
    # c_k = a[i+k] (if i+k < n) + a[i-k] (if i-k >= 0)
    
    # For efficiency, compute the relevant range
    # c_k is non-zero for k in 1..max(i, n-1-i)
    k_max = max(i, n - 1 - i)
    
    if k_max >= 1:
        c = np.zeros(k_max + 1)
        # For k=1..n-1-i: c_k includes a[i+k]
        upper = min(n - 1 - i, k_max)
        for k in range(1, upper + 1):
            c[k] += a[i + k]
        # For k=1..i: c_k includes a[i-k]
        lower = min(i, k_max)
        for k in range(1, lower + 1):
            c[k] += a[i - k]
        
        # conv_new(k) for k=1..k_max
        conv_part = conv_old[1:k_max+1] + (x - ai) * c[1:]
        max_val = max(max_val, float(np.max(conv_part)) * inv_S2)
    
    return 2.0 * n * max_val


def coord_descent_sweep(a, conv, order=None, n_iter=10):
    """
    Perform coordinate descent sweeps. a is normalized to sum=1.
    Returns improved a.
    """
    n = len(a)
    a = a.copy()
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    
    for it in range(n_iter):
        if order is None:
            indices = list(range(n))
        elif order == 'forward':
            indices = list(range(n))
        elif order == 'backward':
            indices = list(range(n-1, -1, -1))
        else:
            indices = list(order)
        
        for i in indices:
            ai = a[i]
            if ai < 1e-10:
                # Try a small positive value
                best_x = ai
                best_f = best_obj
                # Try x = mean of neighbors
                neighbors = []
                if i > 0:
                    neighbors.append(a[i-1])
                if i < n-1:
                    neighbors.append(a[i+1])
                if neighbors:
                    trial_x = sum(neighbors) / len(neighbors) * 0.5
                    f_trial = coord_obj(a, i, trial_x, conv)
                    if f_trial < best_f:
                        best_x = trial_x
                        best_f = f_trial
                if best_x != ai:
                    a[i] = best_x
                    s = float(np.sum(a))
                    a = a / s
                    conv = compute_conv(a)
                    obj = 2.0 * n * float(np.max(conv))
                    if obj < best_obj:
                        best_obj = obj
                continue
            
            # Golden section search for optimal x in [0, upper]
            # Upper bound: try 2*ai, 3*ai, etc. until objective increases
            upper = ai * 2.0
            f_upper = coord_obj(a, i, upper, conv)
            f_zero = coord_obj(a, i, 0.0, conv)
            f_cur = best_obj  # current objective (x = ai)
            
            # Find a bracket [lo, hi] where f(lo) > f(mid) and f(hi) > f(mid)
            # Start with [0, upper]
            lo = 0.0
            hi = upper
            
            # Make sure we have a valid bracket
            if f_zero < f_cur and f_upper < f_cur:
                # Both endpoints are better than current - search the whole range
                pass
            elif f_zero < f_cur:
                # Minimum is at or near 0
                hi = ai * 0.5
                if hi > 0:
                    hi_f = coord_obj(a, i, hi, conv)
                    if hi_f > f_zero:
                        lo = 0.0
                    else:
                        lo = 0.0
                        hi = ai
                # Actually let's just search [0, ai]
                hi = ai
            elif f_upper < f_cur:
                # Minimum is at or near upper
                lo = ai
                # Extend upper
                while f_upper < best_obj and hi < ai * 10:
                    lo = hi
                    hi = hi * 1.5
                    f_new = coord_obj(a, i, hi, conv)
                    if f_new > f_upper:
                        break
                    f_upper = f_new
                    lo = hi
            else:
                # Current is local min in [0, upper], try a narrower range
                lo = ai * 0.5
                hi = ai * 1.5
                f_lo = coord_obj(a, i, lo, conv)
                f_hi = coord_obj(a, i, hi, conv)
                if f_lo < f_cur and f_hi < f_cur:
                    pass  # search [lo, hi]
                elif f_lo < f_cur:
                    hi = ai
                elif f_hi < f_cur:
                    lo = ai
                else:
                    # No improvement possible
                    continue
            
            # Golden section search in [lo, hi]
            if hi - lo < 1e-14:
                continue
            
            gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            f_c = coord_obj(a, i, c, conv)
            f_d = coord_obj(a, i, d, conv)
            
            for _ in range(40):
                if hi - lo < 1e-12:
                    break
                if f_c <= f_d:
                    hi = d
                    d = c
                    f_d = f_c
                    c = hi - gr * (hi - lo)
                    f_c = coord_obj(a, i, c, conv)
                else:
                    lo = c
                    c = d
                    f_c = f_d
                    d = lo + gr * (hi - lo)
                    f_d = coord_obj(a, i, d, conv)
            
            opt_x = (lo + hi) / 2
            f_opt = coord_obj(a, i, opt_x, conv)
            
            if f_opt < best_obj - 1e-14:
                a[i] = opt_x
                s = float(np.sum(a))
                a = a / s
                conv = compute_conv(a)
                best_obj = 2.0 * n * float(np.max(conv))
    
    return a, best_obj

    if s < 1e-10:
        return a

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    x = a.copy()
    y = a.copy()

    k = 0
    
    while time.monotonic() - start < time_budget:

        progress = min(1.0, elapsed / time_budget)

        # Compute smoothed gradient at y
        conv_y = compute_conv(y)

def fista_smoothed_v2(a, time_budget=60.0, tau_init=5.0, tau_end=500.0, verbose=False):
    """FISTA on smoothed max of convolution, with efficient gradient via precomputed matrix."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    # Precompute the |i-j| index matrix (for p[|i-j|])
    idx = np.abs(np.subtract.outer(np.arange(n), np.arange(n)))
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    x = a.copy()
    y = a.copy()
    t_k = 1.0
    k = 0
    lr = 1.0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.95))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)
        
        if verbose and k % 200 == 0:
            obj_y = compute_obj(y)
            print(f"    FISTA k={k}: obj_best={best_obj:.8f}, obj_y={obj_y:.8f}, tau={tau:.1f}, lr={lr:.6f}")
        
        # Compute smoothed gradient at y
        conv_y = compute_conv(y)
        max_conv = float(np.max(conv_y))
        # Smooth the max: use softmax with temperature 1/tau
        exp_vals = np.exp(tau * (conv_y - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient: grad[i] = p[0]*a[i] + (B @ a)[i] where B[i,j] = p[|i-j|]
        # Actually: grad[i] = 2*p[0]*y[i] + sum_{j≠i} p[|i-j|]*y[j]
        # = p[0]*y[i] + sum_{j} p[|i-j|]*y[j]
        # = p[0]*y[i] + (B @ y)[i]
        
        grad = p[0] * y + p[idx] @ y
        
        # Projected gradient step (stay on simplex: a >= 0, sum(a) = 1)
        x_new = y - lr * grad
        # Project onto simplex
        x_new = project_simplex(x_new)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + (t_k - 1.0) / t_new * (x_new - x)
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = x_new.copy()
        else:
            y = y / s_y
        x = x_new
        t_k = t_new
        
        # Adaptive step size
        if k % 50 == 0:
            obj_x = compute_obj(x)
            if obj_x < best_obj:
                best_obj = obj_x
                best_a = x.copy()
            # Backtracking
            obj_try = compute_obj(np.maximum(x - 0.5*lr*grad, 0))
            # Just keep lr fixed for simplicity
            lr = max(0.1, lr * 0.999)
        
        k += 1
    
    return best_a


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    k = np.max(np.where(cond)[0]) if np.any(cond) else n-1
    theta = cssv[k] / (k + 1.0)
    w = np.maximum(v - theta, 0)
    s = float(np.sum(w))
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A
