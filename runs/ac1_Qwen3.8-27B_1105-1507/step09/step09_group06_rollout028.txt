    while size < 2 * n - 1:

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def solve_lp_fast(a):

    Returns g_opt (normalized) or None."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a_norm = a / s
    
    m = 2 * n - 1

        alpha = 2.0 / (it + 2)
        
        # Try multiple step sizes
        candidates = [alpha, alpha * 0.5, alpha * 2.0, 0.1, 0.2, 0.3, 0.5]
        candidates = sorted(set([max(0.01, min(1.0, c)) for c in candidates]))
        
        best_t = 0
        best_new_obj = best_obj
        
        for t in candidates:
            x_try = (1 - t) * y + t * g
            x_try = np.maximum(x_try, 0)
            st = x_try.sum()
            if st < 1e-10:
                continue
            x_try = x_try / st
            obj_try = compute_obj(x_try)
            if obj_try < best_new_obj:
                best_new_obj = obj_try
                best_t = t
        
        if best_t > 0:
            x_new = (1 - best_t) * y + best_t * g
            x_new = np.maximum(x_new, 0)
            x_new = x_new / x_new.sum()
            x_prev = x.copy()
            x = x_new
            if best_new_obj < best_obj:
                best_obj = best_new_obj
                best_a = x.copy()
                stuck_count = 0
            else:
                stuck_count += 1
        else:
            x_prev = x.copy()
            x = y.copy()
            stuck_count += 1
        
        it += 1
        
        # Reset if stuck for too long
        if stuck_count > 500:
            x = best_a.copy()
            x_prev = x.copy()
            stuck_count = 0
            
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"      NLP it{it}: obj={best_obj:.8f}, t={elapsed:.0f}s")
    
    return best_a

def smoothed_gradient_descent(a, time_budget, tau_init=20.0, tau_end=500.0, seed=0, verbose=False):
    """Smoothed gradient descent with log-sum-exp smoothing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    lr = 2.0 / np.sqrt(n)
    momentum = 0.8
    it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init + (tau_end - tau_init) * progress
        
        # Compute smoothed gradient
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / exp_vals.sum()
        
        # Gradient via FFT
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        grad = 2.0 * g_full[:n]
        
        # Nesterov step
        y = a + momentum * v
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-10:
            y = a.copy()
        else:
            y = y / sy
        
        # Recompute gradient at y
        conv_y = compute_conv(y)
        max_conv_y = float(np.max(conv_y))
        exp_y = np.exp(tau * (conv_y - max_conv_y))
        p_y = exp_y / exp_y.sum()
        P_y = fft(p_y, N)
        A_y = fft(y, N)
        g_y = ifft(P_y * np.conj(A_y)).real
        grad_y = 2.0 * g_y[:n]
        
        v = momentum * v - lr * grad_y
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9998)
        
        # Reset if stuck
        if it % 10000 == 0:
            obj_check = compute_obj(a)
            if obj_check > best_obj * 1.01:
                a = best_a.copy()
                v = np.zeros(n)
                lr = 2.0 / np.sqrt(n)
        
        if verbose and it % 5000 == 0:
            print(f"      SGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")
    
    return best_a

def coordinate_descent_polish(a, time_budget, seed=0, verbose=False):
    """Coordinate descent with golden section on individual coordinates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Precompute: for each coordinate i, changing a[i] by delta and 
    # adjusting a[0] to maintain sum=1 gives a new point.
    # But this is expensive. Instead, use a simpler approach:
    # For each i, try a few perturbations and keep the best.
    
    it = 0
    while time.monotonic() - start < time_budget:
        order = np.arange(n)
        rng.shuffle(order)
        
        improved = False
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            
            old_val = a[i]
            # Try relative perturbations
            for factor in [0.8, 0.9, 1.1, 1.2, 0.7, 1.3]:
                a_try = a.copy()
                new_val = old_val * factor
                if new_val < 0:
                    continue
                delta = new_val - old_val
                a_try[i] = new_val
                # Adjust another coordinate to maintain sum
                # Find the coordinate with the largest value (other than i)
                if i == 0:
                    adj = 1
                else:
                    adj = 0
                a_try[adj] = a_try[adj] - delta
                if a_try[adj] < 0:
                    continue
                st = a_try.sum()
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    old_val = new_val
                    improved = True
                    break
        
        it += 1
        if not improved and it > 3:
            break
    
    return best_a

def spectral_init(n, rng):
    """
    Initialize a sequence with a near-flat power spectrum.
    We want |A(f)| to be as flat as possible, which makes the autocorrelation flat.
    Strategy: Start with random phases, take IFFT, take absolute value, normalize.
    """
    # Generate random phases
    phases = rng.uniform(0, 2 * np.pi, n)
    # Create a spectrum with flat magnitude and random phases
    spectrum = np.exp(1j * phases)
    # Inverse FFT to get the sequence
    a = ifft(spectrum).real
    a = np.abs(a)  # Take absolute value to ensure non-negativity
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Raised cosine
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 3:
        # Gaussian centered
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 4:
        # Spectral init
        a = spectral_init(n, rng)
    elif mode == 5:
        # Tapered (Hann)
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2
        a = a / a.sum()
    elif mode == 6:
        # Smooth random
        a = rng.uniform(0, 1, n)
        k = max(2, n // 100)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 7:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** 0.5
        a = a / a.sum()
    elif mode == 8:
        # Two bumps
        c1, c2 = n // 3, 2 * n // 3
        w = max(1, n // 15)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 9:
        # Near-flat with perturbation
        a = np.ones(n) + rng.normal(0, 0.1, n)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
