"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses bidirectional LP alternating minimization as the primary optimizer,
with smoothed gradient descent for polishing and escape, multi-n exploration,
and diverse restarts. The key innovation is solving LPs in both directions
(a given g, and g given a) which provides a more powerful descent direction
than unidirectional LP + blending.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):
    """Compute cross-convolution a * b via FFT."""
    n = len(a)
    size = 1
    while size < n + len(b) - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + len(b) - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a_fixed):
    """Build matrix B (2n-1 x n) such that B @ x = convolve(a_fixed, x)."""
    n = len(a_fixed)
    B = np.zeros((2 * n - 1, n))
    for i in range(n):
        # x[i] contributes to positions k = i + j for j = 0, ..., n-1
        # with coefficient a_fixed[j]
        B[i:i+n, i] = a_fixed
    return B

def solve_lp_given_fixed(a_fixed):
    """
    Solve: min M s.t. conv(a_fixed, x)[k] <= M for all k, sum(x)=1, x>=0.
    Returns (x_opt, M_opt).
    """
    n = len(a_fixed)
    a_f = a_fixed / np.sum(a_fixed)  # normalize
    B = build_conv_matrix(a_f)
    
    x = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [B @ x <= M, cp.sum(x) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if x.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                x_val = np.maximum(x.value, 0)
                s = float(np.sum(x_val))
                if s > 1e-10:
                    return x_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def smoothed_gradient_fft(a, tau):
    """Compute gradient of smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu * rho > (cssv - 1.0)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_val = idx[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def gd_polish(a, time_budget=60.0, tau_init=10.0, tau_end=500.0, seed=0):
    """Nesterov smoothed gradient descent for polishing."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.8
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    
    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        tau = tau_init + (tau_end - tau_init) * (1.0 - np.cos(np.pi * progress)) / 2.0
        
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:

        else:

            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.7 * lr + 0.3 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))

            a_new = best_a.copy()
            v = np.zeros(n)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.99995)

def bidirectional_lp(a, time_budget=100.0, seed=0, verbose=False):
    """
    Bidirectional LP alternating minimization.
    alternately solves:
      - LP1: min M s.t. conv(a, x) <= M, sum(x)=1, x>=0  → x_opt
      - LP2: min M s.t. conv(x_opt, y) <= M, sum(y)=1, y>=0  → y_opt
    and updates a ← blend of x_opt and y_opt.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    it = 0
    consecutive_fails = 0
    
    while time.monotonic() - start < time_budget:
        # LP1: find best x given current a
        x_opt, M1 = solve_lp_given_fixed(a)
        if x_opt is None:
            break
        
        # Check improvement from LP1 alone
        obj_x = compute_obj(x_opt)
        
        # LP2: find best y given x_opt
        y_opt, M2 = solve_lp_given_fixed(x_opt)
        if y_opt is None:
            break
        
        # Try various blends
        best_local_obj = min(best_obj, obj_x)
        best_local_a = best_a.copy() if best_obj <= obj_x else x_opt.copy()
        
        if obj_x < best_local_obj:
            best_local_obj = obj_x
            best_local_a = x_opt.copy()
        
        # Blend x_opt and y_opt
        for t in [0.3, 0.5, 0.7]:
            blend = (1-t) * x_opt + t * y_opt
            s_b = float(np.sum(blend))
            if s_b < 1e-10:
                continue
            blend = blend / s_b
            obj_b = compute_obj(blend)
            if obj_b < best_local_obj:
                best_local_obj = obj_b
                best_local_a = blend.copy()
        
        # Also try blend with current best
        for t in [0.1, 0.2, 0.3, 0.5]:
            blend2 = (1-t) * best_a + t * y_opt
            s_b2 = float(np.sum(blend2))
            if s_b2 < 1e-10:
                continue
            blend2 = blend2 / s_b2
            obj_b2 = compute_obj(blend2)
            if obj_b2 < best_local_obj:
                best_local_obj = obj_b2
                best_local_a = blend2.copy()
        
        it += 1
        
        if best_local_obj < best_obj - 1e-12:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a.copy()
            consecutive_fails = 0
            if verbose and it % 5 == 0:
                print(f"      BiLP it{it}: obj={best_obj:.8f}, M1={M1:.6f}, M2={M2:.6f}")
        else:
            consecutive_fails += 1
            # Perturb to escape
            noise = rng.normal(0, 0.02, n)
            a = a * np.exp(noise)
            a = np.maximum(a, 0)
            a = a / a.sum()
            # Pull back toward best
            a = 0.8 * a + 0.2 * best_a
            a = a / a.sum()
            
            if consecutive_fails > 30:
                break
    
    return best_a

def subgradient_step(a, eta, n_sub=10):
    """
    Take a subgradient step on the max-norm of convolution.
    Uses multiple subgradients (from different argmax positions) for stability.
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = conv.max()
    
    # Find top positions
    top_idx = np.argsort(conv)[::-1][:n_sub]
    
    if max_val < 1e-15:
        return a
    
    # Average subgradient
    grad = np.zeros(n)
    weights = []
    for k in top_idx:
        k = int(k)
        w = conv[k] / max_val
        weights.append(w)
        # Subgradient of (a*a)[k] w.r.t. a is 2*a_shifted_by_k
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += w * a[j]
    
    w_sum = sum(weights)
    if w_sum > 0:
        grad = 2.0 * grad / w_sum
    
    # Projected gradient step on the simplex
    a_new = a - eta * grad
    a_new = project_simplex(a_new)
    return a_new

def diverse_init(n, rng, mode=0):

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses bidirectional LP alternating minimization (solving LPs in both directions:
a given g AND g given a) as the primary optimizer, combined with smoothed
gradient descent for polishing, subgradient steps for escape, multi-n exploration,
and diverse restarts. The bidirectional LP is more powerful than unidirectional
LP+blending because it provides a two-player game-theoretic descent direction.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a_fixed):
    """Build matrix B (2n-1 x n) such that B @ x = convolve(a_fixed, x)."""
    n = len(a_fixed)
    B = np.zeros((2 * n - 1, n))
    for i in range(n):
        B[i:i + n, i] = a_fixed
    return B


def solve_lp_given_fixed(a_fixed):
    """
    Solve: min M s.t. conv(a_fixed, x)[k] <= M for all k, sum(x)=1, x>=0.
    Returns (x_opt, M_opt) or (None, None).
    """
    n = len(a_fixed)
    a_f = a_fixed / np.sum(a_fixed)
    B = build_conv_matrix(a_f)

    x = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [B @ x <= M, cp.sum(x) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if x.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                x_val = np.maximum(x.value, 0)
                s = float(np.sum(x_val))
                if s > 1e-10:
                    return x_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu * rho > (cssv - 1.0)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_val = idx[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def gd_polish(a, time_budget=60.0, tau_init=10.0, tau_end=500.0, seed=0):
    """Nesterov accelerated smoothed gradient descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.8
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init + (tau_end - tau_init) * (1.0 - np.cos(np.pi * progress)) / 2.0

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)

        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.7 * lr + 0.3 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.99995)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n)
            best_it = it

    return best_a


def bidirectional_lp(a, time_budget=100.0, seed=0, verbose=False):
    """
    Bidirectional LP alternating minimization.
    Alternately solves:
      LP1: min M s.t. conv(a, x) <= M, sum(x)=1, x>=0  -> x_opt
      LP2: min M s.t. conv(x_opt, y) <= M, sum(y)=1, y>=0  -> y_opt
    and updates a using optimal blends of x_opt, y_opt, and current best.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    consecutive_fails = 0

    while time.monotonic() - start < time_budget:
        x_opt, M1 = solve_lp_given_fixed(a)
        if x_opt is None:
            break

        obj_x = compute_obj(x_opt)

        y_opt, M2 = solve_lp_given_fixed(x_opt)
        if y_opt is None:
            break

        best_local_obj = best_obj
        best_local_a = best_a.copy()

        if obj_x < best_local_obj:
            best_local_obj = obj_x
            best_local_a = x_opt.copy()

        obj_y = compute_obj(y_opt)
        if obj_y < best_local_obj:
            best_local_obj = obj_y
            best_local_a = y_opt.copy()

        # Try blends of x_opt and y_opt
        for t in [0.3, 0.5, 0.7]:
            blend = (1 - t) * x_opt + t * y_opt
            s_b = float(np.sum(blend))
            if s_b < 1e-10:
                continue
            blend = blend / s_b
            obj_b = compute_obj(blend)
            if obj_b < best_local_obj:
                best_local_obj = obj_b
                best_local_a = blend.copy()

        # Try blends with current best
        for t in [0.1, 0.2, 0.4]:
            blend2 = (1 - t) * best_a + t * y_opt
            s_b2 = float(np.sum(blend2))
            if s_b2 < 1e-10:
                continue
            blend2 = blend2 / s_b2
            obj_b2 = compute_obj(blend2)
            if obj_b2 < best_local_obj:
                best_local_obj = obj_b2
                best_local_a = blend2.copy()

        it += 1

        if best_local_obj < best_obj - 1e-12:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a.copy()
            consecutive_fails = 0
            if verbose and it % 10 == 0:
                print(f"      BiLP it{it}: obj={best_obj:.8f}, M1={M1:.6f}, M2={M2:.6f}")
        else:
            consecutive_fails += 1
            # Perturb to escape local min
            noise_scale = 0.015 + 0.003 * min(consecutive_fails, 8)
            noise = rng.normal(0, noise_scale, n)
            a = a * np.exp(noise)
            a = np.maximum(a, 0)
            a = 0.85 * a + 0.15 * best_a
            a = a / a.sum()

            if consecutive_fails > 40:
                break

    return best_a


def subgradient_loop(a, n_steps=300, seed=0):
    """
    Subgradient method with averaging for the max-convolution problem.
    Uses multiple subgradients from top positions for variance reduction.
    """
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(seed)

    a_avg = a.copy()
    a_curr = a.copy()

    for step_i in range(n_steps):
        eta = 1.0 / (np.sqrt(step_i + 1) * np.sqrt(n) * 0.5)
        conv = compute_conv(a_curr)
        max_val = conv.max()

        if max_val < 1e-15:
            break

        # Get top K positions for averaged subgradient
        top_k = min(10, len(conv))
        top_idx = np.argsort(conv)[::-1][:top_k]

        grad = np.zeros(n)
        total_w = 0.0
        for k in top_idx:
            k = int(k)
            w = (conv[k] / max_val) ** 2
            total_w += w
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += w * a_curr[j]

        if total_w > 0:
            grad = 2.0 * grad / total_w

        a_curr = a_curr - eta * grad
        a_curr = project_simplex(a_curr)

        # Running average
        a_avg = ((step_i + 1) / (step_i + 2)) * a_avg + (1.0 / (step_i + 2)) * a_curr

        obj_curr = compute_obj(a_curr)
        if obj_curr < best_obj:
            best_obj = obj_curr
            best_a = a_curr.copy()

        obj_avg = compute_obj(a_avg)
        if obj_avg < best_obj:
            best_obj = obj_avg
            best_a = a_avg.copy()

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 3:
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(np.pi * x) + 0.3 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = 1.0
        for j in range(1, 6):
            a += ((-1) ** (j + 1)) * np.cos(j * np.pi * x) / (j * j)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 7:
        # Wigner semicircle
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x**2, 0))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function using bidirectional LP alternating minimization
    with gradient polishing, subgradient averaging, multi-n exploration,
    and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Phase 0: Initialize from parent or fallback
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = diverse_init(n_target, rng, mode=2)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Bidirectional LP (primary workhorse) ===
    print(f"\n=== Phase 1: Bidirectional LP (main) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 450.0)
    if p1_budget > 30:
        a_bi = bidirectional_lp(best_seq, time_budget=p1_budget, verbose=True, seed=seed)
        obj_bi = compute_obj(a_bi)
        if obj_bi < best_score:
            best_score = obj_bi
            best_seq = a_bi.copy()
            print(f"  Phase1 BiLP: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 BiLP: obj={obj_bi:.8f}")

    # === Phase 2: GD Polish ===
    print(f"\n=== Phase 2: GD Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.12, 120.0)
    if p2_budget > 20:
        a_gd = gd_polish(best_seq, time_budget=p2_budget, tau_init=20.0, tau_end=800.0, seed=seed)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase2 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 GD: obj={obj_gd:.8f}")

    # === Phase 3: BiLP Round 2 with different starting perturbation ===
    print(f"\n=== Phase 3: BiLP Round 2 ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 200.0)
    if p3_budget > 30:
        noise = rng.normal(0, 0.015, len(best_seq))
        a_pert = best_seq * np.exp(noise)
        a_pert = np.maximum(a_pert, 0)
        a_pert = 0.92 * a_pert + 0.08 * (np.ones(len(best_seq)) / len(best_seq))
        a_pert = a_pert / a_pert.sum()

        a_bi2 = bidirectional_lp(a_pert, time_budget=p3_budget, verbose=True, seed=seed + 100)
        obj_bi2 = compute_obj(a_bi2)
        if obj_bi2 < best_score:
            best_score = obj_bi2
            best_seq = a_bi2.copy()
            print(f"  Phase3 BiLP2: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 BiLP2: obj={obj_bi2:.8f}")

    # === Phase 4: Subgradient with averaging + GD finish ===
    print(f"\n=== Phase 4: Subgradient + GD ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.1, 100.0)
    if p4_budget > 20:
        n_subs = int(p4_budget * 8)
        a_sub = subgradient_loop(best_seq, n_steps=min(n_subs, 400), seed=seed + 200)
        obj_sub = compute_obj(a_sub)
        if obj_sub < best_score:
            best_score = obj_sub
            best_seq = a_sub.copy()
            print(f"  Phase4 Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 Subgrad: obj={obj_sub:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_gd4 = gd_polish(best_seq, time_budget=min(remaining * 0.5, 30.0),
                              tau_init=50.0, tau_end=1000.0, seed=seed + 300)
            obj_g4 = compute_obj(a_gd4)
            if obj_g4 < best_score:
                best_score = obj_g4
                best_seq = a_gd4.copy()
                print(f"  Phase4 GD: obj={best_score:.8f} **")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        n_current + 150, n_current + 300,
        max(500, n_current - 150),
        n_current + 50, n_current + 400
    ]))
    n_options = [n for n in n_options if 500 <= n <= 2000 and abs(n - n_current) >= 50]
    n_options = n_options[:3]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        alloc = min(remaining * 0.15, 80.0)
        if alloc > 20:
            a_try = bidirectional_lp(a_try, time_budget=alloc, verbose=False, seed=seed + n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts (short) ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = (r * 2 + 1) % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend_frac = 0.15 + 0.05 * r
        a_r = (1 - blend_frac) * a_r + blend_frac * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.15, 60.0)
        if alloc > 15:
            a_r = bidirectional_lp(a_r, time_budget=alloc, verbose=False, seed=seed + 1000 + r)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        alloc = min(remaining * 0.3, 60.0)
        a_bi_final = bidirectional_lp(best_seq, time_budget=alloc, verbose=False, seed=seed + 500)
        obj_bf = compute_obj(a_bi_final)
        if obj_bf < best_score:
            best_score = obj_bf
            best_seq = a_bi_final.copy()
            print(f"  Final BiLP: obj={best_score:.8f} **")
        else:
            print(f"  Final BiLP: obj={obj_bf:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 50.0)
        a_gd_final = gd_polish(best_seq, time_budget=alloc, tau_init=100.0, tau_end=2000.0, seed=seed + 600)
        obj_gf = compute_obj(a_gd_final)
        if obj_gf < best_score:
            best_score = obj_gf
            best_seq = a_gd_final.copy()
            print(f"  Final GD: obj={best_score:.8f} **")
        else:
            print(f"  Final GD: obj={obj_gf:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val):
        print(f"  evaluate_sequence confirms: {val:.8f}")
        if val < best_score:
            best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
