import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    # Try cvxpy solvers
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    # Fallback to scipy linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None

def _m_seq_init(n, rng):
    """Generate M-sequence inspired initialization."""
    # Generate a pseudo-M-sequence using LFSR
    m = max(4, int(np.log2(n)) + 1)
    length = 2**m - 1
    # Simple LFSR with primitive polynomial
    reg = np.ones(m, dtype=int)
    taps = [m-1] + [i for i in range(m-1) if _is_prime_poly(m, i)]
    if len(taps) < 2:
        taps = [m-1, m-2]

    seq = []
    for _ in range(min(length, n * 2)):
        seq.append(reg[0])
        feedback = 0
        for t in taps:
            feedback ^= reg[t]
        reg = np.roll(reg, 1)
        reg[0] = feedback

    seq = np.array(seq[:n * 2], dtype=float)
    # Map to [0, 1] and resample to length n
    seq = seq[:n] if len(seq) == n else np.interp(
        np.linspace(0, 1, n), np.linspace(0, 1, len(seq)), seq
    )
    seq = 0.2 + 0.6 * seq  # Map to [0.2, 0.8] to avoid zeros
    return seq / seq.sum()


def _is_prime_poly(m, i):
    """Simple check (not truly checking primitivity, just for diversity)."""
    return (i * 7 + m) % 3 != 0

            quiet_candidates = np.arange(n)
        # Weight by inverse contribution

            if delta <= 0 or delta >= a[src] * 0.9:
                continue

            # Update FFT
            A_new = A_fft.copy()
            A_new[src] -= delta
            A_new[target] += delta
            conv_new = irfft(A_new * A_new, fsize)[:2 * n - 1]
            new_peak = float(np.max(conv_new))
            new_obj = 2.0 * n * new_peak

            if new_obj < best_obj - 1e-12:
                # Accept
                a[src] -= delta
                a[target] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_obj = new_obj
                best_a = a.copy()
                conv = conv_new
                peak = new_peak
                accepted += 1
                break
            else:
                rejected += 1
                # If rejected, try smaller fraction
                break

        if (accepted + rejected) % 1000 == 0 and verbose:
            elapsed = time.monotonic() - start
            print(f"      Local it{accepted+rejected}: obj={best_obj:.8f}, "
                  f"acc={accepted}, rej={rejected}, elapsed={elapsed:.1f}s")

    return best_a


def waterfall_reduction(a, time_budget, verbose=False):
    """
    Systematically reduce the top peaks by water-filling.
    Iteratively identify the highest peak and try to reduce it.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv = compute_conv(a)
    rng = np.random.default_rng(123)

    iteration = 0
    stall = 0
    patience = 50

    while time.monotonic() - start < time_budget:
        if time_budget - (time.monotonic() - start) < 1:
            break
        iteration += 1

        # Find the peak
        peak_val = float(np.max(conv))
        peak_lag = int(np.argmax(conv))

        # Find all pairs contributing to this peak
        i_start = max(0, peak_lag - n + 1)
        i_end = min(n - 1, peak_lag)

        pairs = []
        for i in range(i_start, i_end + 1):
            j = peak_lag - i
            c = a[i] * a[j]
            if c > 1e-8:
                pairs.append((c, i, j))

        if not pairs:
            break

        pairs.sort(reverse=True)

        improved = False
        for contrib, i, j in pairs[:20]:
            if time_budget - (time.monotonic() - start) < 1:
                break
            src = i if a[i] >= a[j] else j
            dst = j if a[i] >= a[i] else i

            # Try moving to various targets
            # Target: random position far from peak contributors
            for _ in range(5):
                target = rng.integers(0, n)
                if target == src or target == dst:
                    continue

                for frac in [0.001, 0.003, 0.005, 0.01]:
                    delta = frac * a[src]
                    if delta <= 0 or delta >= a[src]:
                        continue

                    a[src] -= delta
                    a[target] += delta
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s < 1e-10:
                        a[src] += delta
                        a[target] -= delta
                        continue
                    a = a / s

                    conv = compute_conv(a)
                    obj = 2.0 * n * float(np.max(conv))

                    if obj < best_obj - 1e-13:
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        break

                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > patience:
                # Perturb and restart
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv = compute_conv(a)
                stall = 0
                patience = max(20, patience // 2)

        if verbose and iteration % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"      Waterfall it{iteration}: obj={best_obj:.8f}, "
                  f"stall={stall}, elapsed={elapsed:.1f}s")

    return best_a


def structured_init(n, rng, mode=0):
    """Generate diverse structured initializations."""
    if mode == 0:
        # M-sequence inspired
        return _m_seq_init(n, rng)
    elif mode == 1:
        # Barker-like: few large values, rest small
        a = np.full(n, 0.01)
        n_ones = max(3, n // 10)
        positions = rng.choice(n, n_ones, replace=False)
        a[positions] = 1.0
        return a / a.sum()
    elif mode == 2:
        # Periodic structure
        period = rng.integers(3, 20)
        pattern = rng.uniform(0.1, 1.0, period)
        a = np.tile(pattern, n // period + 1)[:n]
        return a / a.sum()
    elif mode == 3:
        # Concentrated center
        x = np.linspace(-1, 1, n)
        a = np.exp(-x**2 / (2 * 0.2**2))
        return a / a.sum()
    elif mode == 4:
        # Two peaks
        x = np.linspace(-1, 1, n)
        c1 = rng.uniform(-0.5, -0.2)
        c2 = rng.uniform(0.2, 0.5)
        a = np.exp(-10*(x-c1)**2) + np.exp(-10*(x-c2)**2)
        return a / a.sum()
    elif mode == 5:
        # Ramp with noise
        a = np.linspace(0.1, 1.0, n) + rng.normal(0, 0.1, n)
        a = np.maximum(a, 0)
        return a / a.sum()
    elif mode == 6:
        # Sparse random
        a = np.zeros(n)
        n_nonzero = max(5, n // 5)
        positions = rng.choice(n, n_nonzero, replace=False)
        a[positions] = rng.uniform(0.1, 1.0, n_nonzero)
        a += 0.001
        return a / a.sum()
    else:
        # Random Dirichlet
        alpha = rng.uniform(0.1, 2.0)
        return rng.dirichlet(np.ones(n) * alpha)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def refine_with_lp_and_local(a, time_budget):
    """Combined LP alternating + local search for final refinement."""
    start = time.monotonic()
    n = len(a)

    # Phase A: LP alternating
    lp_budget = time_budget * 0.6
    a_lp = lp_alternate(a, time_budget=lp_budget, verbose=False)
    obj_lp = compute_obj(a_lp)

    # Phase B: Local search
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        a_loc = local_search_fft(a_lp, time_budget=remaining, verbose=False)
        obj_loc = compute_obj(a_loc)
        if obj_loc < obj_lp:
            return a_loc, obj_loc

    return a_lp, obj_lp


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: LP alternating + FFT local search + structured restarts
    + n-sweep + peak-targeting.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    print(f"\nInitial: n={len(best_seq)}, obj={best_score:.8f}, target=1.5027")
    print(f"Budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    n_current = len(best_seq)

    # === Phase 1: Quick LP polish on parent (10%) ===
    print(f"\n=== Phase 1: LP Polish (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.10, 120.0)
    if p1_budget > 20:
        a_p1 = lp_alternate(best_seq, time_budget=p1_budget, verbose=True)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_p1:.8f}")

    # === Phase 2: Local search on best so far (10%) ===
    print(f"\n=== Phase 2: Local Search (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.10, 100.0)
    if p2_budget > 15:
        a_ls = local_search_fft(best_seq, time_budget=p2_budget, verbose=True)
        obj_ls = compute_obj(a_ls)
        if obj_ls < best_score:
            best_score = obj_ls
            best_seq = a_ls.copy()
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj_ls:.8f}")

    # === Phase 3: n-sweep (20%) ===
    print(f"\n=== Phase 3: n-Sweep ===")
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.20

    n_options = sorted(set([
        max(500, n_current - 100),
        max(500, n_current - 50),
        max(500, n_current - 20),
        n_current,
        n_current + 20,
        n_current + 50,
        n_current + 100,
        n_current + 200,
        700, 750, 800, 850, 900, 950, 1000, 1050, 1100, 1200, 1300, 1400,
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1500]

    per_n_budget = p3_budget / max(1, len(n_options))
    n_results = []
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_n_budget, remaining * 0.12)
        if alloc < 10:
            alloc = 10

        a_try = resample_sequence(best_seq, n_try)
        a_opt = lp_alternate(a_try, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)
        n_results.append((obj_opt, n_try, a_opt))
        marker = " **" if obj_opt < best_score else ""
        print(f"  n={n_try}: obj={obj_opt:.8f}{marker}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try

    # === Phase 4: Intensive optimization on best n (30%) ===
    print(f"\n=== Phase 4: Intensive Optimization (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.30

    # Multiple rounds with different strategies
    n_rounds = 5
    per_round = p4_budget / n_rounds

    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        alloc = min(per_round, remaining * 0.25)
        if alloc < 20:
            break

        if rnd == 0:
            # LP alternating from current best
            a_start = best_seq.copy()
            method = "LP"
        elif rnd == 1:
            # LP + local search
            a_start = best_seq.copy()
            method = "LP+LS"
        elif rnd == 2:
            # Perturbed LP
            rng_r = np.random.default_rng(7919 * (rnd + 1) + n_current)
            a_start = best_seq * np.exp(rng_r.normal(0, 0.04, n_current))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
            method = "Perturb+LP"
        elif rnd == 3:
            # Waterfall reduction
            a_start = best_seq.copy()
            method = "Waterfall"
        else:
            # Structured init blended with best
            a_struct = structured_init(n_current, rng_r, mode=rnd)
            a_start = 0.4 * best_seq + 0.6 * a_struct
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
            a_start = lp_alternate(a_start, time_budget=alloc * 0.3, verbose=False)
            method = "Struct+LP"

        if method == "LP+LS":
            half = alloc / 2
            a_r = lp_alternate(a_start, time_budget=half, verbose=False)
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_r = local_search_fft(a_r, time_budget=min(alloc - half, remaining - 5), verbose=False)
        elif method == "Waterfall":
            a_r = waterfall_reduction(a_start, time_budget=alloc, verbose=False)
        else:
            a_r = lp_alternate(a_start, time_budget=alloc, verbose=False)

        obj_r = compute_obj(a_r)
        # Also try local search on the result
        remaining = deadline - time.monotonic()
        if remaining > 15 and obj_r < best_score + 0.01:
            a_r2 = local_search_fft(a_r, time_budget=min(30, remaining * 0.3), verbose=False)
            obj_r2 = compute_obj(a_r2)
            if obj_r2 < obj_r:
                a_r = a_r2
                obj_r = obj_r2

        marker = " **" if obj_r < best_score else ""
        print(f"  Round {rnd} ({method}): obj={obj_r:.8f}{marker}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()

    # === Phase 5: Structured restarts (15%) ===
    print(f"\n=== Phase 5: Structured Restarts ===")
    remaining = deadline - time.monotonic()
    p5_budget = remaining * 0.15
    n_use = len(best_seq)
    n_restarts = 4
    per_restart = p5_budget / max(1, n_restarts)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        alloc = min(per_restart, remaining * 0.2)
        if alloc < 15:
            break

        mode = r
        a_r = structured_init(n_use, rng, mode=mode)
        # Blend with best
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        # LP + local search
        lp_alloc = alloc * 0.6
        a_r = lp_alternate(a_r, time_budget=lp_alloc, verbose=False)
        remaining = deadline - time.monotonic()
        ls_alloc = min(alloc * 0.35, remaining * 0.2)
        if ls_alloc > 10:
            a_r = local_search_fft(a_r, time_budget=ls_alloc, verbose=False)

        obj_r = compute_obj(a_r)
        marker = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}{marker}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()

    # === Phase 6: Final polish (15%) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Combined LP + local search
    alloc = min(remaining * 0.5, 80.0)
    if alloc > 20:
        a_final, obj_final = refine_with_lp_and_local(best_seq, time_budget=alloc)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final refine: obj={best_score:.8f} **")

    # Waterfall reduction
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.3, 40.0)
    if alloc > 10:
        a_wf = waterfall_reduction(best_seq, time_budget=alloc, verbose=True)
        obj_wf = compute_obj(a_wf)
        if obj_wf < best_score:
            best_score = obj_wf
            best_seq = a_wf.copy()
            print(f"  Final waterfall: obj={best_score:.8f} **")

    # One final LP + local
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_last = lp_alternate(best_seq, time_budget=min(remaining * 0.4, 25.0), verbose=False)
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_last = local_search_fft(a_last, time_budget=min(remaining * 0.5, 15.0), verbose=False)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP+LS: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = compute_obj(np.array(result))

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"=== Target: 1.5027, Achieved: {best_score:.8f}, Gap: {best_score - 1.5027:.8f} ===")

    return result
