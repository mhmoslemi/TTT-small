Uses L-BFGS on log-sum-exp smoothed objective, LP-based Sjöberg iteration with

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    fa = rfft(a, fft_size)
    return irfft(fa * fa, fft_size)[:size]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS on log-sum-exp smoothed objective with softmax parameterization,
LP-based Sjöberg iteration with geometric blending, multi-n exploration,
and active-set refinement for final polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import cvxpy as cp
import time


def compute_conv(a, b=None):
    """Full linear convolution via FFT. If b is None, computes a*a."""
    n = len(a)
    if b is None:
        b = a
    m = len(b)
    size = n + m - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    fa = rfft(a, fft_size)
    fb = rfft(b, fft_size)
    return irfft(fa * fb, fft_size)[:size]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_obj_grad(x, tau):
    """
    Compute smoothed objective and gradient with respect to x (log-parameters).
    a = softmax(x), objective = (1/tau) * log(sum_k exp(tau*(a*a)[k]))
    """
    n = len(x)
    # Compute softmax
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    a = exp_x / np.sum(exp_x)
    
    # Compute convolution
    conv = compute_conv(a)
    
    # Log-sum-exp
    max_conv = np.max(conv)
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = np.sum(exp_vals)
    f = max_conv + np.log(sum_exp) / tau
    
    # Gradient with respect to a
    p = exp_vals / sum_exp
    # grad_a[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
    # Using convolution trick
    r = p[::-1]  # reverse p
    conv_ar = compute_conv(a, r)  # length 3n-2
    c = conv_ar[n-1:2*n-1]  # length n
    c = c[::-1]
    grad_a = 2.0 * c
    
    # Gradient with respect to x (softmax chain rule)
    # df/dx[i] = a[i] * (grad_a[i] - dot(grad_a, a))
    ga_dot_a = np.dot(grad_a, a)
    grad_x = a * (grad_a - ga_dot_a)
    
    return f, grad_x


def lbfgs_smoothed(a_init, time_budget=100.0, tau_init=5.0, tau_end=0.5, 
                    n_anneal=5, verbose=True):
    """
    Run L-BFGS on the smoothed objective with tau annealing.
    Starts from a_init (on the simplex) and returns the optimized a.
    """
    n = len(a_init)
    # Initialize x from a_init
    a = np.maximum(a_init, 1e-30)
    a = a / a.sum()
    x = np.log(a)  # log-parameters
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for level in range(n_anneal):
        elapsed = time.monotonic() - start
        progress = level / max(1, n_anneal - 1)
        tau = tau_init * (tau_end / tau_init) ** progress
        
        remaining = time_budget - elapsed
        if remaining < 5:
            break
        
        # Run L-BFGS at this tau
        result = minimize(
            softmax_obj_grad,
            x,
            args=(tau,),
            method='L-BFGS-B',
            jac=True,
            options={
                'maxiter': int(remaining * 20),  # adaptive
                'maxfun': int(remaining * 15),
                'ftol': 1e-12,
                'gtol': 1e-8,
                'maxls': 20
            }
        )
        
        x_new = result.x
        # Compute a from new x
        x_shifted = x_new - np.max(x_new)
        exp_x = np.exp(x_shifted)
        a_new = exp_x / np.sum(exp_x)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            x = x_new.copy()
        
        if verbose:
            print(f"  L-BFGS tau={tau:.4f}: obj={best_obj:.8f} (iters={result.nit})")
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, M_target=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue
    
    return None, None


def sjoberg_iteration(a, time_budget=100.0, verbose=True, seed=42):
    """
    Sjöberg iteration with geometric blending.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    iter_count = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Try both arithmetic and geometric blending
        best_blend_obj = best_obj
        best_blend_a = best_a.copy()
        
        # Arithmetic blending with golden section
        for t in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_blend_a = a_try.copy()
        
        # Geometric blending
        eps = 1e-15
        for t in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (best_a + eps) ** (1.0 - t) * (g_lp + eps) ** t
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_blend_a = a_try.copy()
        
        iter_count += 1
        
        if best_blend_obj < best_obj - 1e-12:
            best_obj = best_blend_obj
            best_a = best_blend_a.copy()
            a = best_a.copy()
            no_improve = 0
            if verbose and iter_count % 5 == 0:
                print(f"  Sjoberg it{iter_count}: obj={best_obj:.8f}, M={M_lp:.6f}")
        else:
            no_improve += 1
            if no_improve >= 5:
                # Perturb
                scale = 0.01 * min(no_improve, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                if iter_count > 50:
                    break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    # Use cubic spline for better quality
    from scipy.interpolate import CubicSpline
    cs = CubicSpline(x_old, a, bc_type='natural')
    a_new = cs(x_new)
    return np.maximum(a_new, 0)


def peak_flatten(a, n_peaks=5, strength=0.5):
    """
    Reduce the peaks of the autocorrelation by adjusting a.
    For each peak lag k, reduce the contribution of a at positions that contribute to it.
    """
    n = len(a)
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:n_peaks]
    
    a_new = a.copy()
    for k in top_idx:
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if 0 <= j < n:
                # Reduce a[i] and a[j] slightly
                a_new[i] *= (1.0 - strength * 0.01)
                a_new[j] *= (1.0 - strength * 0.01)
    
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5)**2) / (2 * 0.1**2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a / a.sum()
    elif mode == 5:
        # Two bumps
        c1, c2 = n // 4, 3 * n // 4
        w = n // 8
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1)**2) / (2*w**2))
        a += np.exp(-((np.arange(n) - c2)**2) / (2*w**2))
        a = a / a.sum()
    elif mode == 6:
        # Triangle
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2*x - 1)
        a = a / a.sum()
    elif mode == 7:
        # Hamming window
        a = 0.54 + 0.46 * np.cos(2 * np.pi * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 8:
        # Blackman window
        n_arr = np.arange(n)
        a = (0.42 - 0.5*np.cos(2*np.pi*n_arr/n) + 0.08*np.cos(4*np.pi*n_arr/n))
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 9:
        # Sinc-like
        x = np.linspace(-3, 3, n)
        a = np.sinc(x / 3)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        else:
            a = np.ones(n) / n
    else:
        a = np.ones(n) / n
    return a

            s = float(np.sum(parent_arr))

    print(f"\n=== Phase 1: L-BFGS Smoothed Optimization ===")
