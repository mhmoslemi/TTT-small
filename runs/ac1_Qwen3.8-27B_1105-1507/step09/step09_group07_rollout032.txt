Uses: fast LP (HiGHS), correct gradient-based smooth optimization, spectral flattening,

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g) for g of same length."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_fast(a):
    """Solve LP: min M s.t. conv(a,g)[k]<=M all k, sum(g)=1, g>=0. Uses HiGHS."""
    n = len(a)
    a_s = a / np.sum(a)
    A = build_conv_matrix(a_s)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None

    """Multiplicative weight update approximation to the LP. Faster for large n."""
    n = len(a)
    a_s = a / np.sum(a)
    g = np.ones(n) / n
    
    state_clip = np.log(1.0)
    for _ in range(iters):
        c = compute_conv(np.concatenate([a_s, g]))
        # We need conv(a_s, g), not conv of the concatenated vector

        c_ag = np.zeros(2 * n - 1)
        N = 1
        while N < 2 * n - 1:
            N <<= 1
        fa = rfft(a_s, N)
        fg = rfft(g, N)
        c_ag = irfft(fa * fg, N)[:2 * n - 1]
        
        # Find the most violated constraint
        k_star = int(np.argmax(c_ag))
        
        # Update: decrease g where a[k_star - j] is large
        # grad of c_ag[k_star] w.r.t. g[j] = a_s[k_star - j] (with a_s[m]=0 outside)

        log_g = np.log(g + 1e-15)
        log_g -= eta * weights
        log_g -= np.max(log_g)
        g = np.exp(log_g)
        g = g / g.sum()
    
    return g

                scale = 0.05 + 0.03 * (stall // patience)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                stall = 0
                last_good_t = 0.5

        if verbose and iteration % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"          LP-alt it{iteration}: obj={best_obj:.8f}, stall={stall}, elapsed={elapsed:.1f}s")

    return best_a


def smooth_gradient_descent(a, time_budget, beta_start=20.0, beta_end=500.0, verbose=False):
    """
    Smooth optimization using log-sum-exp surrogate with correct FFT-based gradient.
    Uses mirror descent (multiplicative updates) for the simplex constraint.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    # FFT size for gradient computation
    N = 1
    while N < 3 * n - 2:
        N <<= 1

    lr = 2.0 / n
    it = 0
    total_iters = max(1, int(time_budget * 5))

    while time.monotonic() - start < time_budget:
        frac = min(1.0, (it + 1) / total_iters)
        beta = beta_start * (beta_end / beta_start) ** frac

        conv = compute_conv(a)
        # Compute softmax weights w[k] = exp(beta * conv[k]) / sum
        max_c = float(np.max(conv))
        shifted = beta * (conv - max_c)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()

        # Gradient: grad[j] = 4*n * sum_m a[m] * w[m+j]
        # This is the cross-correlation of a and w
        fa = fft(a, N)
        fw = fft(w, N)
        grad_full = np.real(ifft(fa * np.conj(fw)))
        grad = 4.0 * n * grad_full[:n]

        # Mirror descent (multiplicative update)
        log_a = np.log(a)
        log_a_new = log_a - lr * grad
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = max(1e-6, lr * 0.999)
        else:
            # Try smaller step
            for shrink in [0.5, 0.25, 0.1]:
                log_a_new = log_a - lr * shrink * grad
                log_a_new -= np.max(log_a_new)
                a_try = np.exp(log_a_new)
                a_try = a_try / a_try.sum()
                obj_try = compute_obj_normalized(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    lr = max(1e-6, lr * shrink * 0.99)
                    break
            else:
                lr = max(1e-6, lr * 0.8)

        it += 1
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"          Smooth it{it}: obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")

    return best_a


def spectral_flatten(a, time_budget, verbose=False):
    """
    Iteratively flatten the convolution spectrum.
    In frequency domain: we want |A(f)|^2 to correspond to a flat time-domain signal.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for alpha in [0.3, 0.5, 0.7, 0.9]:
        if time.monotonic() - start > time_budget:
            break
        for it in range(30):
            if time.monotonic() - start > time_budget:
                break
            
            # Current convolution
            conv = compute_conv(a)
            # Target: flat convolution
            conv_flat = np.ones(2 * n - 1) / (2 * n - 1)
            
            # In frequency: current A^2 = FFT(conv_padded)
            A = fft(a, N)
            A2 = A * A  # This is FFT of zero-padded conv
            
            # Target FFT of flat conv
            target_conv = np.zeros(N)
            target_conv[:2*n-1] = conv_flat
            target_A2 = fft(target_conv)
            
            # Update: blend magnitudes, keep phase of A
            # A_new^2 should be closer to target_A2
            # A_new = A * (target_A2 / A^2)^(alpha/2) in some sense
            # More simply: A_new = A * (1-alpha) + alpha * sqrt(target_A2) (complex sqrt)
            
            # Compute target_A (square root of target_A2)
            # Handle the phase carefully
            mag_A2 = np.abs(A2)
            mag_target = np.abs(target_A2)
            
            # Blend the magnitude of A^2
            mag_A2_new = (1 - alpha) * mag_A2 + alpha * mag_target
            # Blended A^2
            A2_new = A2 * (mag_A2_new / (mag_A2 + 1e-30))
            
            # A_new = sqrt(A2_new) - take the branch that's closest to current A
            # For each frequency bin, A_new = sqrt(mag_A2_new) * exp(i*angle(A2_new)/2)
            # But we want A_new to be close to A, so:
            A_new = np.sqrt(mag_A2_new + 0j) * np.exp(0.5j * np.angle(A2_new))
            
            # Choose branch: for each bin, pick the sign that makes A_new closer to A
            for i in range(N):
                if abs(A_new[i]) > 1e-15:
                    # Two choices: A_new[i] or -A_new[i]
                    if abs(A_new[i] - A[i]) > abs(-A_new[i] - A[i]):
                        A_new[i] = -A_new[i]
            
            # Invert
            a_new = np.real(ifft(A_new, N))[:n]
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new

            obj_new = compute_obj_normalized(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
                if verbose and it % 5 == 0:
                    print(f"          Spectral alpha={alpha}, it{it}: obj={best_obj:.8f}")
            else:
                break

    return best_a


def targeted_peaks_reduction(a, time_budget, verbose=False):
    """
    Targeted reduction of convolution peaks by moving mass from contributors to quiet positions.
    Uses FFT-based convolution for speed.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    round_num = 0

    while time.monotonic() - start < time_budget:
        round_num += 1
        conv = compute_conv(a)
        peak_val = float(np.max(conv))
        
        # Find top 5 peak lags
        peak_lags = np.argsort(conv)[::-1][:5]
        
        # Find quiet positions: low a[i] that don't contribute to peaks
        # A position i contributes to lag k if a[i]*a[k-i] is large
        contrib_score = np.zeros(n)
        for k in peak_lags:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contrib_score[i] += a[i] * a[j]
        
        quiet_positions = np.argsort(contrib_score)[:10]
        
        improved = False
        for k_star in peak_lags:
            if time.monotonic() - start > time_budget:
                break
            k_star = int(k_star)
            
            # Find top contributors to this lag
            contribs = []
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    contribs.append((a[i] * a[j], i, j))
            contribs.sort(reverse=True)
            
            for contrib, i, j in contribs[:8]:
                if time.monotonic() - start > time_budget:
                    break
                if contrib < 1e-12:
                    continue
                
                src = i if a[i] >= a[j] else j
                for target in quiet_positions:
                    if target == src:
                        continue
                    for frac in [0.01, 0.02, 0.05]:
                        delta = frac * a[src]
                        if delta <= 0 or delta >= 0.9 * a[src]:
                            continue
                        a[src] -= delta
                        a[target] += delta
                        new_obj = compute_obj_normalized(a)
                        if new_obj < best_obj - 1e-13:
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[src] += delta
                            a[target] -= delta
                            break
                if improved:
                    break
            if improved:
                break
        
        if not improved and round_num > 20:
            break

        if verbose and round_num % 20 == 0:
            print(f"          Targeted round {round_num}: obj={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode % 10 == 0:
        a = np.ones(n) / n
    elif mode % 10 == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode % 10 == 2:
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 3:
        a = 1.0 + 0.3 * np.sin(4 * np.pi * x) + 0.2 * np.cos(6 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 4:
        a = np.exp(-1.5 * (x - 0.3) ** 2) + np.exp(-1.5 * (x - 0.7) ** 2) + 0.05
        a = a / a.sum()
    elif mode % 10 == 5:
        a = x**2 + (1 - x)**2 + 0.1
        a = a / a.sum()
    elif mode % 10 == 6:
        freqs = rng.integers(2, 10, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.25 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 7:
        # Monotonic (tries to minimize middle convolution)
        a = x + 0.1
        a = a / a.sum()
    elif mode % 10 == 8:
        # Two-peak
        a = np.exp(-10*(x-0.25)**2) + np.exp(-10*(x-0.75)**2) + 0.02
        a = a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n) * 0.3)
    return a
