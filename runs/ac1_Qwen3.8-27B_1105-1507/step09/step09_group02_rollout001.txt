"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiplicative (mirror descent) updates with peak-aware weighting

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def multiplicative_descent(a, time_budget, K=20, eta_init=0.5, eta_end=0.01, verbose=False):
    """
    Multiplicative (mirror) descent for minimizing max conv(a,a).
    Uses peak-aware weighting and annealed step size.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        eta = eta_init * (eta_end / eta_init) ** progress

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        mean_conv = 1.0 / (2 * n - 1)  # since sum(a)=1
        
        # Weight peaks by excess over mean
        excess = np.maximum(conv - mean_conv, 0)
        # Top K peaks
        top_idx = np.argsort(conv)[::-1][:K]
        top_excess = excess[top_idx]
        
        if top_excess[0] < 1e-15:
            break
        
        # Normalize weights
        w = top_excess ** 2  # quadratic weighting emphasizes largest peaks
        w_sum = float(np.sum(w))
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Compute pressure vector
        p = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k = int(k)
            # conv[k] = sum_j a[j] * a[k-j]
            # contribution to a[i] from peak k: a[k-i] (if valid)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                p[i_start:i_end + 1] += wi * a[j_start:j_end + 1][::-1]

        # Multiplicative update: a_new[i] = a[i] * (1 - eta * p[i] / mean(p))
        mean_p = float(np.mean(p))
        if mean_p < 1e-30:
            break
        
        a_new = a * (1.0 - eta * p / mean_p)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a = best_a.copy()
            stall += 50
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            a = best_a.copy()
            if stall > 200:
                # Add noise to escape
                noise = np.random.default_rng(it).normal(0, 0.01, n)
                a = best_a * np.exp(noise)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                eta = eta * 0.5

        it += 1
        if verbose and it % 500 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, eta={eta:.4f}, max_conv={max_conv:.6e}")

    return best_a

    it = 0

        # Build restricted problem
        A_rows = []
        for k in active_set:

        solved = False

            except Exception:
                continue

        g_val = np.maximum(g.value, 0)
        s_g = float(np.sum(g_val))

        if best_blend_obj < best_obj - 1e-14:

            new_active = list(np.argsort(conv_new)[::-1][:50])

            if verbose and it % 3 == 0:

                if violated:

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def simulated_annealing_final(a, time_budget, verbose=False):
    """
    Final polish using simulated annealing with multiplicative mutations.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))

        return a

        T = T_init * (T_end / T_init) ** progress

        # Generate candidate: multiplicative noise
        noise = rng.normal(0, T, n)
        a_new = a * np.exp(noise)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        delta = obj_new - best_obj
        
        if delta < 0 or rng.random() < np.exp(-delta / (T * 0.01)):
            a = a_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose and it % 100 == 0:
                    print(f"    SA it{it}: obj={best_obj:.8f}, T={T:.5f}")

def peak_surgical_lp(a, time_budget, verbose=False):
    """
    For the top peaks, solve targeted LPs that specifically try to reduce them.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        top_idx = np.argsort(conv)[::-1][:5]
        
        # For the worst peak, try to find a direction that reduces it
        k_worst = int(top_idx[0])
        
        # The peak at position k_worst is sum_j a[j]*a[k_worst-j]
        # We want to find delta such that:
        # 1. (a+delta)[i] >= 0 for all i
        # 2. sum(delta) = 0 (maintain normalization)
        # 3. conv(a+delta, a+delta)[k_worst] is minimized
        # Approximation: conv(a+d, a+d)[k] ≈ conv(a,a)[k] + 2*sum_j a[j]*d[k-j] + ...
        # So we want to minimize sum_j a[j]*d[k_worst-j] subject to sum(d)=0, a+d>=0
        
        # This is a linear program!
        d = cp.Variable(n)
        # Objective: minimize sum_j a[j] * d[k_worst - j]
        obj_terms = np.zeros(n)
        i_start = max(0, k_worst - (n-1))
        i_end = min(n-1, k_worst)
        j_start = k_worst - i_end
        j_end = k_worst - i_start
        obj_terms[i_start:i_end+1] = a[j_start:j_end+1][::-1]
        
        constraints = [
            cp.sum(d) == 0,
            a + d >= 0,
            d >= -a + 1e-10,  # ensure a+d >= 0 strictly
            d <= cp.norm(a, 1) / n * 0.1,  # limit step size
        ]
        
        # Also constrain other top peaks
        for k_other in top_idx[1:]:
            k_other = int(k_other)
            terms = np.zeros(n)
            i_s = max(0, k_other - (n-1))
            i_e = min(n-1, k_other)

            j_e = k_other - i_s
            terms[i_s:i_e+1] = a[j_s:j_e+1][::-1]
            # Don't increase other peaks by more than the decrease in worst
            constraints.append(cp.sum(terms * d) <= 0)

        solved = False
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if d.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
        
        if not solved:
            it += 1
            if it > 20:
                break
            continue
        
        d_val = d.value
        # Try different step sizes
        best_step_obj = best_obj
        best_step = 0
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = a + alpha * d_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_step_obj:
                best_step_obj = obj_try
                best_step = alpha
        
        it += 1
        if best_step_obj < best_obj - 1e-14:
            best_obj = best_step_obj
            a = a + best_step * d_val
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_a = a.copy()
            if verbose and it % 3 == 0:
                print(f"    PSLP it{it}: obj={best_obj:.8f}, step={best_step:.2f}")

                a = best_a * np.exp(rng.normal(0, 0.005, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                it = 0

    return best_a
