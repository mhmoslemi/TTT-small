import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def reweighted_lp_optimize(a, time_budget=300.0, verbose=False):
    """
    Main optimization loop using reweighted LP.
    Weights are adaptively chosen to emphasize current peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    it = 0
    stall = 0
    momentum = 0.3
    v = np.zeros(n)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        
        # Compute weights: emphasize peaks
        # Use softmax-like weighting with temperature
        max_conv = float(np.max(conv))
        # Temperature controls how concentrated the weights are
        tau_w = max_conv * 3.0  # weights spread over ~top 30% of peaks
        weights = np.exp((conv - max_conv) / tau_w)
        weights = weights / np.sum(weights)
        
        # Add small uniform component to avoid degeneracy
        weights = 0.9 * weights + 0.1 / (2 * n - 1)
        weights = weights / np.sum(weights)
        
        # Solve reweighted LP
        g_lp = solve_lp_reweighted(a, weights)
        if g_lp is None:
            g_lp = solve_lp_minmax(a)
            if g_lp is None or (isinstance(g_lp, tuple) and g_lp[0] is None):
                break
            if isinstance(g_lp, tuple):
                g_lp = g_lp[0]
        
        # Update with momentum
        v = momentum * v + (g_lp - a)
        a_new = project_simplex(a + v)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            momentum = min(0.6, momentum + 0.01)
        else:
            # Try smaller step
            found = False
            for scale in [0.5, 0.25, 0.1]:
                a_try = project_simplex(a + scale * v)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    v = scale * v
                    stall = 0
                    found = True
                    break
            if not found:
                stall += 1
                momentum = max(0.1, momentum * 0.9)
                v *= 0.5
                a = best_a.copy()
                if stall > 30:
                    # Restart with perturbation
                    a = best_a * np.exp(rng.normal(0, 0.02 * (1 + stall/50), n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    v = np.zeros(n)
                    momentum = 0.3
                    stall = 0
                    if it > 200:
                        break
        
        it += 1
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    RWLP it{it}: obj={best_obj:.8f}, time={elapsed:.1f}s")
    
    return best_a


def minmax_lp_polish(a, time_budget=100.0, verbose=False):
    """Polish using min-max LP (more expensive but potentially better directions)."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(54321)
    
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        result = solve_lp_minmax(a)
        if result[0] is None:
            break
        g_lp, M_lp = result
        
        a_new = project_simplex(a + 0.5 * (g_lp - a))
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Try golden section
            lo, hi = 0.0, 1.0
            for _ in range(5):
                t1 = hi - 0.618 * (hi - lo)
                t2 = lo + 0.618 * (hi - lo)
                a1 = np.maximum((1-t1)*a + t1*g_lp, 0)
                a1 = a1 / a1.sum()
                a2 = np.maximum((1-t2)*a + t2*g_lp, 0)
                a2 = a2 / a2.sum()
                f1 = compute_obj(a1)
                f2 = compute_obj(a2)
                if f1 < f2:
                    hi = t2
                else:
                    lo = t1
            t_best = (lo + hi) / 2
            a_new = np.maximum((1-t_best)*a + t_best*g_lp, 0)
            a_new = a_new / a_new.sum()
            obj_new = compute_obj(a_new)
            
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                a = best_a.copy()
                if stall > 20:
                    a = best_a * np.exp(rng.normal(0, 0.03, n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    stall = 0
                    if it > 100:
                        break
        
        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    MMLP it{it}: obj={best_obj:.8f}, M={M_lp:.6f}, time={elapsed:.1f}s")
    
    return best_a


def high_tau_gradient(a, time_budget=100.0, tau_init=200.0, tau_end=2000.0, verbose=False):
    """High-tau smoothed gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    it = 0
    lr = 2.0 / np.sqrt(n)
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** progress
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient of smoothed max
        # d/d a[j] of sum_k p[k] * conv(a,a)[k]
        # = 2 * sum_k p[k] * (cross-correlation of p and a at appropriate lag)
        
        # Use FFT:
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(np.conj(P) * A).real  # This gives one part of the gradient
        # Actually need to be more careful
        # grad_a[j] = 2 * (p * a_reversed)[j] ... 
        # Let me use the direct formula:
        # d conv(a,a)[k] / d a[j] = a[k-j] (if valid) + a[j-(k-n+1)] (if valid)
        # So grad[j] = 2 * sum_k p[k] * a[k-j] (first term) + 2 * sum_k p[k] * a[j+k-n+1] (second term)
        
        # First term: (p * a)[j] where * is correlation
        # sum_k p[k] * a[k-j] = (p correlate a)[j]
        # Second term: sum_k p[k] * a[j+k-n+1] = (p correlate a)[j+n-1] ... hmm
        
        # Let me just use the matrix approach for correctness:
        # grad[j] = 2 * sum_k p[k] * (d conv[k]/d a[j])
        # d conv[k]/d a[j] = a[k-j] * 1[0<=k-j<n] + a[j-(k-n+1)] * 1[0<=j-(k-n+1)<n]
        
        # For the first term: sum_k p[k] * a[k-j] = sum_i p[i+j] * a[i] (i from 0 to n-1, i+j from 0 to 2n-2)
        # This is a "sliding dot product" - can be computed via convolution
        
        # Actually, let me just use:
        # grad = 2 * (A_p @ a) where A_p is the matrix with A_p[k,j] = p[k] * delta(k-j in [0,n-1])
        # which means A_p @ a = [sum_j p[j]*a[0] (for k=0), sum_j p[j+1]*a[j] ... ]
        
        # This is getting complicated. Let me use a simpler (slower) approach:
        N_fft = 1
        while N_fft < 2 * n - 1:
            N_fft <<= 1
        
        # The gradient of sum_k p[k] * conv(a,a)[k] w.r.t. a is:
        # 2 * IFFT(FFT(p) * conj(FFT(a))) [:n]  ... no that's for circular convolution
        
        # Let me just compute it directly using the convolution matrix:
        # Actually, for the purpose of this code, let me use a simpler gradient:
        # The gradient of max(conv) is approximately the "autocorrelation" of p with a
        
        # Simple approach: use the fact that for the smoothed objective,
        # the gradient w.r.t. a[j] is approximately proportional to
        # (p * a)[j] where * denotes cross-correlation
        
        # Cross-correlation: (p ⊛ a)[j] = sum_k p[k] * a[k-j] (with appropriate bounds)
        # This can be computed as: ifft(fft(p) * conj(fft(a))) for circular, 
        # or use the convolution trick
        
        # Let me use numpy correlate:
        # Actually the cleanest way:
        # grad[j] = 2 * sum_{k} p[k] * (a[k-j] if 0<=k-j<n else 0) + (a[j-k+n-1] if valid)
        
        # I'll use the FFT-based approach from the previous code:
        P_fft = fft(p, N_fft)
        A_fft = fft(a, N_fft)
        # The "gradient" in the conv sense:
        g1 = ifft(P_fft * np.conj(A_fft), N_fft).real[:n]  # sum_k p[k]*a[k-j]
        g2 = ifft(np.conj(P_fft) * A_fft, N_fft).real[:n]  # sum_k p[k-j]*a[j]... 
        # Hmm, this isn't right either. Let me just use the direct computation for now.
        
        # Direct computation (O(n^2) but n is at most ~2000, so it's fine for n<=2000):
        # Actually for n=2000, O(n^2) = 4M operations per iteration, which is fine.
        # But we're doing many iterations, so let's use FFT.
        
        # The correct formula:
        # Let C[k] = conv(a,a)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i]*a[k-i]
        # dC[k]/da[j] = a[k-j] * 1[0<=k-j<n] + a[k-j] * 1[0<=k-j<n]
        # Wait, by the product rule: d/dj (a[i]*a[k-i]) where i=j or k-i=j
        # = a[k-j] (when i=j) + a[j] * 0 ... no
        # dC[k]/da[j] = a[k-j] (from the a[i] factor when i=j) + a[k-j] (from the a[k-i] factor when k-i=j, i.e., i=k-j)
        # = 2*a[k-j] if 0<=k-j<n, else 0
        # Wait, that's only true if both i=j and k-i=j are valid, which means k=2j.
        # Otherwise, only one term contributes.
        
        # Correctly: dC[k]/da[j] = a[k-j] * 1[0<=k-j<n] + 1[0<=k-j<n] * a[j] ... no
        # C[k] = sum_i a[i] * a[k-i]
        # dC[k]/da[j] = (d a[j]/da[j]) * a[k-j] + a[j] * (d a[k-j]/da[j])
        # = a[k-j] * 1[0<=k-j<n] + a[j] * 1[j==k-i for some i, i.e., i=k-j, 0<=k-j<n]
        # = a[k-j] * 1[0<=k-j<n] + a[j] * 1[0<=k-j<n]
        
        # Wait, that gives 2*a[j]*1[...] when k-j=j (i.e., k=2j), and a[k-j] otherwise.
        # Actually no: dC[k]/da[j] = a[k-j] (from differentiating the first factor) + a[k-j] (from differentiating the second factor, since d a[k-i]/da[j] = 1 when k-i=j, i.e., i=k-j)
        # So dC[k]/da[j] = a[k-j] + a[j]... no that's wrong.
        
        # Let me be very careful:
        # C[k] = sum_i a[i] * a[k-i]
        # Partial derivative w.r.t. a[j]:
        # = sum_i [delta(i,j) * a[k-i] + a[i] * delta(k-i, j)]
        # = a[k-j] (if 0<=k-j<n, from first term with i=j)
        #   + a[k-j] (if k-j is a valid index, from second term with i=k-j)
        #   ... wait, second term: a[i] * delta(k-i, j) means i = k-j, so a[k-j] * 1[0<=k-j<n]
        # = 2 * a[k-j] * 1[0<=k-j<n]
        
        # So dC[k]/da[j] = 2 * a[k-j] if 0 <= k-j < n, else 0.
        # This is simple!
        
        # Therefore: grad[j] = sum_k p[k] * 2 * a[k-j] * 1[0<=k-j<n]
        # = 2 * sum_{k: 0<=k-j<n} p[k] * a[k-j]
        # = 2 * sum_{i: 0<=i<n, 0<=j+i<2n-1} p[j+i] * a[i]
        # This is a "constrained cross-correlation" of p and a.
        
        # For the unrestricted case (no bounds), this would be (p * a)[j] (correlation).
        # With bounds, it's a truncated version.
        
        # For j in [0, n-1] and k in [0, 2n-2]:
        # k-j ranges from -j to 2n-2-j
        # Valid when 0 <= k-j < n, i.e., j <= k <= j+n-1
        # So grad[j] = 2 * sum_{k=j}^{min(j+n-1, 2n-2)} p[k] * a[k-j]
        
        # This is a "sliding window" dot product. Can be computed efficiently with FFT:
        # Let's define q[k] = p[k] for k in [0, 2n-2], 0 otherwise
        # Then grad[j] = 2 * sum_{i=0}^{n-1} q[j+i] * a[i]
        # = 2 * (q * a_rev)[j + n - 1] where a_rev is reversed... 
        # This is getting complicated. Let me just use the O(n^2) approach for n <= 2000.
        
        if n <= 1500:
            grad = np.zeros(n)
            for j in range(n):
                k_start = j
                k_end = min(j + n, 2 * n - 1)
                if k_start < k_end:
                    grad[j] = 2.0 * np.dot(p[k_start:k_end], a[:k_end - k_start])
        else:
            # FFT approach for larger n
            N_pad = 1
            while N_pad < 2 * n - 1:
                N_pad <<= 1
            # Pad a to length 2n-1
            a_pad = np.zeros(2 * n - 1)
            a_pad[:n] = a
            # grad[j] = 2 * sum_{i=0}^{n-1} p[j+i] * a[i]
            # This is the "anti-convolution" or "cross-correlation"
            # = 2 * ifft(fft(p, N_pad) * conj(fft(a_pad, N_pad))).real ... not exactly
            # Let me use: correlate(p, a_pad) but truncated
            # Actually: sum_i p[j+i] * a[i] = sum_i p_rev[2n-2-j-i] * a[i] ... 
            # I'll just use a simple loop for now
            grad = np.zeros(n)
            for j in range(n):
                k_start = j
                k_end = min(j + n, 2 * n - 1)
                if k_start < k_end:
                    grad[j] = 2.0 * np.dot(p[k_start:k_end], a[:k_end - k_start])
        
        # Gradient step with projection
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.999)
        
        if it - best_it > 5000:
            a = best_a.copy()
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    HTG it{it}: obj={best_obj:.8f}, tau={tau:.0f}, time={elapsed:.1f}s")
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    from scipy.interpolate import interp1d
    f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
    a_new = f(x_new)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Comb structure
        period = max(2, n // 30)
        a = np.zeros(n)
        for i in range(0, n, period):
            a[i] = 1.0
        # Spread a bit
        a = np.convolve(a, np.ones(3)/3, mode='same')
        a = np.maximum(a, 0) / a.sum()
    elif mode == 3:
        # Gaussian mixture
        n_components = rng.integers(3, 10)
        centers = rng.uniform(0, n, n_components)
        widths = rng.uniform(n/20, n/5, n_components)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 4:
        # Random Fourier features
        x = np.linspace(0, 1, n)
        n_freqs = rng.integers(5, 20)
        freqs = rng.uniform(1, 30, n_freqs)
        phases = rng.uniform(0, 2 * np.pi, n_freqs)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    else:
        # Sinc-like
        x = np.linspace(-1, 1, n)
        a = (np.sinc(x) + 0.5)
        a = np.maximum(a, 0) / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization using reweighted LP, min-max LP, and gradient methods.
    Explores multiple sequence lengths and uses adaptive restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent available: {e}")
    
    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")
    
    # === Phase 1: Reweighted LP (fast, many iterations) ===
    print(f"\n=== Phase 1: Reweighted LP ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 400.0)
    if p1_budget > 30:
        a_rwl = reweighted_lp_optimize(best_seq, time_budget=p1_budget, verbose=True)
        obj_rwl = compute_obj(a_rwl)
        if obj_rwl < best_score:
            best_score = obj_rwl
            best_seq = a_rwl.copy()
            print(f"  ** RWLP: obj={best_score:.8f} **")
        else:
            print(f"  RWLP: obj={obj_rwl:.8f}")
    
    # === Phase 2: Min-max LP polish ===
    print(f"\n=== Phase 2: Min-Max LP ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 250.0)
    if p2_budget > 30:
        a_mml = minmax_lp_polish(best_seq, time_budget=p2_budget, verbose=True)
        obj_mml = compute_obj(a_mml)
        if obj_mml < best_score:
            best_score = obj_mml
            best_seq = a_mml.copy()
            print(f"  ** MMLP: obj={best_score:.8f} **")
        else:
            print(f"  MMLP: obj={obj_mml:.8f}")
    
    # === Phase 3: High-tau gradient ===
    print(f"\n=== Phase 3: High-Tau Gradient ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 200.0)
    if p3_budget > 30:
        a_htg = high_tau_gradient(best_seq, time_budget=p3_budget, tau_init=200.0, tau_end=3000.0, verbose=True)
        obj_htg = compute_obj(a_htg)
        if obj_htg < best_score:
            best_score = obj_htg
            best_seq = a_htg.copy()
            print(f"  ** HTG: obj={best_score:.8f} **")
        else:
            print(f"  HTG: obj={obj_htg:.8f}")
    
    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    
    n_current = len(best_seq)
    n_options = []
    # Try nearby and distant sizes
    for n_try in [n_current - 100, n_current + 100, n_current + 200,
                  n_current + 400, 1000, 1200, 1500, 2000]:

The gradient of sum_k p[k] * conv(a,a)[k] w.r.t. a[j] is:
2 * sum_{k=j}^{min(j+n-1, 2n-2)} p[k] * a[k-j]

Let me define:
- p is of length 2n-1
- a is of length n

For each j in [0, n-1]:
grad[j] = 2 * sum_{i=0}^{min(n-1, 2n-2-j)} p[j+i] * a[i]

This is a "sliding inner product" which can be computed as a cross-correlation if we pad appropriately.

Let pad a to length 2n-1: a_pad = [a[0], ..., a[n-1], 0, ..., 0]
Then grad[j]/2 = sum_{i=0}^{2n-2-j} p[j+i] * a_pad[i]

This is the "anti-correlation" or "flip correlation": if we define p_flip[k] = p[2n-2-k], then:
