Given a support set $T$ (subset of $\{0, ..., n-1\}$), we want to solve:

One approach: parameterize $a_j = e^{b_j} / \sum_{l \in T} e^{b_l}$ for $j \in T$ (and $a_j = 0$ for $j \notin T$). Then optimize over $b \in \mathbb{R}^{|T|}$.

and compute the gradient with respect to $b$ via the chain rule.

$\frac{\partial f_\tau}{\partial b_j} = \frac{\partial f_\tau}{\partial \tilde{a}} \cdot \frac{\partial \tilde{a}}{\partial b_j}$

$\frac{\partial \tilde{a}_i}{\partial b_j} = \tilde{a}_i (\delta_{ij} - \tilde{a}_j)$ (softmax derivative)

So $\frac{\partial f_\tau}{\partial b_j} = \sum_i \frac{\partial f_\tau}{\partial \tilde{a}_i} \tilde{a}_i (\delta_{ij} - \tilde{a}_j)$

This is the "centered" gradient, which makes sense because the softmax parameterization is invariant to adding a constant to all $b_j$.

import cvxpy as cp

from scipy.optimize import minimize as scipy_minimize

    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):

    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_g=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M, dual_y) or (None, None, None)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    g_val = None
    M_val = None
    dual_y = None

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    # Extract dual variables
                    try:
                        dual_y = np.array(A.value[0]).astype(np.float64)
                    except:
                        dual_y = None
                break
        except Exception:
            continue

    return g_val, M_val, dual_y


def lp_polish_linesearch(a, time_budget, verbose=True):
    """LP polishing with line-search mixing for optimal step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.5

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, M_lp, _ = solve_lp(a)
        if g_lp is None:
            break

        # Line search over mixing parameter t
        t_candidates = np.concatenate([
            np.linspace(0.05, 1.0, 20),
            [max(0.03, last_t * 0.5), max(0.03, last_t * 0.7),
             max(0.03, last_t * 0.9), last_t,
             min(1.0, last_t * 1.1), min(1.0, last_t * 1.3), min(1.0, last_t * 1.5)]
        ])
        t_candidates = np.unique(np.clip(t_candidates, 0.01, 1.0))

        best_t_obj = best_obj
        best_t = last_t
        best_a_local = best_a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:

            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-12:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t
            stall = 0

def smoothed_obj_grad_log(b, tau, support):
    """Compute smoothed objective and gradient w.r.t. log-parameters b.
    b is the log of amplitude on support positions."""
    a_full = np.zeros(len(support) if isinstance(support, (list, np.ndarray)) and len(support) > 0 else 1, dtype=np.float64)
    # Actually, let's work with the full vector
    n = len(support) if not isinstance(support, np.ndarray) or support.ndim == 1 else len(support)
    # support is an array of indices into the full sequence
    full_len = support[-1] + 1 if len(support) > 0 else 1

    # Compute a from log-params
    b_shifted = b - b.max()  # for numerical stability
    a_supp = np.exp(b_shifted)
    a_supp = a_supp / a_supp.sum()

    a = np.zeros(full_len)
    a[support] = a_supp

    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    obj_val = max_conv + np.log(sum_exp) / tau

    # Gradient: (grad)_j = 2 * sum_k p_k * a_{k-j} for j in support
    N = 1
    while N < 3 * full_len:
        N <<= 1
    P = fft(p, N)
    A_full = fft(a, N)
    g_full = ifft(P * np.conj(A_full)).real
    grad_full = 2.0 * g_full[:full_len]

    # Chain rule for log-parameterization (softmax)
    grad_b = a_supp * (grad_full[support] - np.sum(a_supp * grad_full[support]))

    return obj_val, grad_b

    if len(support) < 2:
        return a

    b0 = np.log(np.maximum(a[support], 1e-30))
    b0 = b0 - b0.mean()  # center for numerical stability

    best_obj = compute_obj(a)
    start = time.monotonic()

    total_tau = tau_max - tau_min

    current_b = b0.copy()
    for stage_idx, tau in enumerate(tau_stages):
        stage_budget = (time_budget / n_stages) * (1.0 + 0.5 * stage_idx / n_stages)
        stage_deadline = time.monotonic() + stage_budget

                                 jac=True, options={'maxiter': 200, 'gtol': 1e-10,
                                                    'ftol': 1e-14, 'maxfun': 500})
            if res.x is not None:
                b_shifted = res.x - res.x.max()
                a_supp = np.exp(b_shifted)
                a_supp = a_supp / a_supp.sum()
                a_new = np.zeros(full_len)
                a_new[support] = a_supp
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    current_b = res.x.copy()
                    if verbose:
                        print(f"    LogOpt stage{stage_idx}: tau={tau:.0f}, obj={best_obj:.8f}")
        except Exception:
            pass

        if time.monotonic() > stage_deadline:
            break

    return best_a


def support_shrink_refine(a, time_budget, verbose=True):
    """Try zeroing out small entries and re-optimizing on reduced support."""
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Try different support thresholds
    sorted_vals = np.sort(a)[::-1]
    for frac in [0.3, 0.5, 0.7, 0.8, 0.9]:
        if time.monotonic() - start > time_budget:
            break
        k = max(2, int(len(a) * frac))
        threshold = sorted_vals[k - 1] if k <= len(sorted_vals) else 0
        support = np.where(a >= threshold)[0]
        if len(support) < 2:
            continue
        a_shrunk = np.zeros(len(a))
        a_shrunk[support] = a[support]
        a_shrunk = a_shrunk / a_shrunk.sum()

        # Quick optimization on reduced support
        alloc = min(30, time_budget - 5 - (time.monotonic() - start))
        if alloc > 5:
            a_shrunk = log_param_optimize(a_shrunk, time_budget=alloc,
                                          tau_min=100, tau_max=3000, verbose=False)
        obj_shrunk = compute_obj(a_shrunk)
        if obj_shrunk < best_obj - 1e-12:
            best_obj = obj_shrunk
            best_a = a_shrunk.copy()
            if verbose:
                print(f"    Shrink frac={frac}: obj={best_obj:.8f}, support={len(support)}")

    return best_a


def high_tau_gd(a, time_budget, tau_start=200, tau_end=8000, verbose=True):
    """High-precision smoothed gradient descent with aggressive tau."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Aggressive tau: stay high for longer
        tau = tau_start + (tau_end - tau_start) * progress ** 0.5

        # Nesterov
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_f = fft(y, N)
        g_full = ifft(P * np.conj(A_f)).real
        g = 2.0 * g_full[:n]

        # Adaptive step size
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        lr = min(1.0 / (g_norm + 1e-10), 2.0 / np.sqrt(n))
        lr = max(lr, 1e-8)

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        beta = min(0.95, beta + 0.0001)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it
            if it > 50000:
                break

        if verbose and it % 5000 == 0:
            print(f"    HTGD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Asymmetric ramp
        a = np.arange(1, n + 1, dtype=np.float64)
        a = a / a.sum()
    elif mode == 3:
        # Beta-like asymmetric
        x = np.linspace(0.01, 1, n)
        a = x ** 2 * (1 - x) ** 0.5
        a = a / a.sum()
    elif mode == 4:
        # Exponential decay
        a = np.exp(-np.arange(n) / (n * 0.3))
        a = a / a.sum()
    elif mode == 5:
        # Two exponential (asymmetric)
        a = np.exp(-np.arange(n) / (n * 0.2)) + 0.3 * np.exp(-np.arange(n) / (n * 0.8))
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = x ** 3 * (1 - x)
        a = a / a.sum()
    else:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP polishing with line search, high-precision GD,
    support reduction, log-param optimization, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 25.0  # safety margin

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = diverse_init(n_init, rng, mode=2)  # ramp init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: LP Polishing with Line Search (250s) ===
    print(f"\n=== Phase 1: LP Polishing with Line Search ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.2, 250.0)
    if p1_budget > 30:
        a_lp = lp_polish_linesearch(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: High-Tau GD Refinement (200s) ===
    print(f"\n=== Phase 2: High-Tau GD ===")

    if p2_budget > 20:
        a_gd = high_tau_gd(best_seq, time_budget=p2_budget,

    # === Phase 3: Support Shrink + Refine (100s) ===
    print(f"\n=== Phase 3: Support Shrink ===")

        a_shrunk = support_shrink_refine(best_seq, time_budget=p3_budget, verbose=True)

    # === Phase 4: Log-Param Optimization on Best (100s) ===
