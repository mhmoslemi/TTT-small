def smoothed_gradient_topk(a, k):
    """Compute gradient using top-k smoothing."""
    n = len(a)
    conv = compute_conv(a)
    # Get top-k indices
    top_k_idx = np.argsort(conv)[::-1][:k]
    p = np.zeros(2*n-1)
    p[top_k_idx] = 1.0 / k
    
    a_rev = a[::-1]
    N2 = 1
    while N2 < 3*n-2:
        N2 <<= 1
    g_full = irfft(rfft(a_rev, N2) * rfft(p, N2), N2)[:3*n-2]
    g = g_full[n-1 : 2*n-1]
    
    return 2.0 * g, conv

def nesterov_simplex(a, time_budget=200.0, tau_init=50.0, tau_end=500.0, use_topk=False, topk_k=10, verbose=False):
    """Nesterov accelerated gradient descending on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = 0.9
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    y = a.copy()
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        if it % 5000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")

        # Compute gradient at y
        if use_topk:
            g_y, _ = smoothed_gradient_topk(y, topk_k)
        else:
            g_y, _ = smoothed_gradient_lse(y, tau)

        # Projection-based Nesterov
        y_new = y - lr * g_y
        # The Nesterov update: 
        # y = a + momentum * (a - prev_a)  [this is the "lookahead"]
        # a_new = project(y - lr * grad(y))
        
        # Actually, let me use the standard projected Nesterov:
        # y_k = a_k + beta_k * (a_k - a_{k-1})
        # a_{k+1} = P_simplex(y_k - lr_k * grad(y_k))
        
        a_new = project_simplex(y - lr * g_y)
        
        # Update y for next iteration
        v = a_new - a  # This is the "momentum"
        y = a_new + momentum * v
        
        # Clip y to simplex (approximately)
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y > 1e-10:
            y = y / s_y
        
        # Evaluate
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        
        # Adaptive learning rate
        lr = max(1e-8, lr * 0.9995)
        
        # Reset if stuck
        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            y = a.copy()
            lr = 5.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a

def coord_descent_fast(a, time_budget=200.0, rng=None, verbose=False):
    """
    Fast coordinate descent with incremental convolution updates.
    For each coordinate, find optimal delta via bisection on subgradient.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        conv = compute_conv(a)
        return a
    a = a / S
    conv = compute_conv(a)

    best_obj = float(np.max(conv))
    best_a = a.copy()
    gr = (np.sqrt(5) - 1) / 2.0
    start = time.monotonic()
    stagnant = 0
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        obj_before = float(np.max(conv))
        
        # Random or cyclic order
        if rng is not None and sweep % 3 == 0:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        improved = False
        for k in order:
            # Precompute direction
            # d[m] = a[m-k] for m in [k, k+n-1]
            d = np.zeros(2 * n - 1)
            d[k:k+n] = a

            # The objective as function of delta: max_m(conv[m] + 2*delta*d[m] + delta^2 * I(m=2k))
            # This is convex in delta. Find minimum via golden section.
            # Lower bound: delta >= -a[k]
            lo = -a[k]
            
            # Upper bound estimate
            # At delta = a[k] (doubling), the new conv values are at most:
            # conv[m] + 2*a[k]*d[m] + a[k]^2
            # Max of this is bounded by max_conv + 2*a[k]*max(d) + a[k]^2
            # But a simpler bound: try a few values
            hi = 2.0 / n
            # Expand hi if needed
            for _ in range(10):
                test_val = max(conv + 2.0*hi*d, 0)
                test_val[2*k] += hi*hi
                if max(test_val[2*k], max(np.delete(test_val, 2*k))) <= best_obj + 1.0/n:
                    break
                hi *= 2
            hi = min(hi, 5.0 / n)  # Cap it
            
            # Golden section search
            a_gs = lo
            b_gs = hi
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)
            
            def obj_delta(delta):
                val = conv + 2.0 * delta * d
                if 2*k < len(val):
                    val[2*k] += delta * delta
                return float(np.max(val))
            
            fc = obj_delta(c_gs)
            fd = obj_delta(d_gs)
            
            for _ in range(40):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = obj_delta(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = obj_delta(d_gs)
            
            delta_opt = (a_gs + b_gs) / 2.0
            new_obj_est = obj_delta(delta_opt)
            
            if new_obj_est < obj_before - 1e-15:
                a[k] += delta_opt
                a[k] = max(0.0, a[k])
                conv += 2.0 * delta_opt * d
                conv[2*k] += delta_opt * delta_opt
                S_new = float(np.sum(a))
                if S_new < 1e-10:
                    a[:] = 1.0/n
                    conv = compute_conv(a)
                else:
                    a /= S_new
                    conv /= (S_new * S_new)
                improved = True
        
        obj_after = float(np.max(conv))
        if obj_after < best_obj - 1e-13:
            best_obj = obj_after
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 8:
                if rng is not None:
                    scale = 0.02 * (1 + stagnant * 0.3)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    s2 = a.sum()
                    if s2 > 1e-10:
                        a = a / s2
                        conv = compute_conv(a)
                        best_obj = float(np.max(conv))
                    stagnant = 0
                else:
                    break

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={2*n*obj_after:.8f}, time={elapsed:.0f}s")

    return best_a

import cvxpy as cp

def solve_lp_for_g(a, solver_list=None):
    """Solve LP: min M s.t. conv(a,g)[m] <= M for all m, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix: A @ g <= M
    # A[m, j] = a_norm[m-j] if 0 <= m-j < n, else 0
    A = np.zeros((2*n-1, n))
    for j in range(n):
        rows = np.arange(j, min(j + (2*n-1), 2*n-1))
        # A[m, j] = a_norm[m-j] for m in [j, j+n-1] (capped at 2n-2)
        actual_rows = np.arange(j, min(j + n, 2*n-1))
        A[actual_rows, j] = a_norm[:len(actual_rows)]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None

def alternating_lp(a, time_budget=100.0, verbose=False):
    """
    Alternating LP: alternate between optimizing a (given g) and g (given a).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s
    
    g = a.copy()  # Initialize g = a
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < time_budget:
        it += 1
        
        # Step 1: Given a, find best g
        g_new, M1 = solve_lp_for_g(a)
        if g_new is None:
            break
        g = g_new
        
        # Step 2: Given g, find best a
        a_new, M2 = solve_lp_for_g(g)
        if a_new is None:
            break
        a = a_new
        
        # Check convergence
        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
        
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    AltLP it{it}: obj={best_obj:.8f}, M1={M1:.6f}, M2={M2:.6f}, t={elapsed:.0f}s")
        
        # Check for convergence
        if it > 5 and abs(obj - best_obj) < 1e-12:
            # Try blending
            blend = 0.5
            a_blend = (1-blend)*a + blend*best_a
            a_blend = np.maximum(a_blend, 0)
            sb = a_blend.sum()
            if sb > 1e-10:
                a_blend = a_blend / sb
                obj_blend = compute_obj(a_blend)
                if obj_blend < best_obj:
                    best_a = a_blend
                    best_obj = obj_blend
                    a = a_blend

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussians
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        # Multi-freq
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        # Two offset Gaussians
        c1 = rng.integers(0, n // 3)
        c2 = rng.integers(2 * n // 3, n)
        w = max(1, n // 12)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 8:
        # Half-sine
        x = np.linspace(0, np.pi, n)
        a = np.sin(x)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 9:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 10:
        # Symmetric triangular
        mid = n // 2
        a = np.zeros(n)
        a[:mid+1] = np.linspace(1, 0, mid+1)
        a[mid:] = np.linspace(0, 1, n-mid)
        a = a / a.sum()
    elif mode == 11:
        # Flat top with smooth edges
        edge = n // 10
        a = np.ones(n)
        a[:edge] = np.linspace(0, 1, edge)
        a[-edge:] = np.linspace(1, 0, edge)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  Phase3 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 CD: obj={obj_cd2:.8f}")

    # === Phase 4: Alternating LP ===
    print(f"\n=== Phase 4: Alternating LP ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.20, 200.0)
    if p4_budget > 30:
        a_lp = alternating_lp(best_seq, time_budget=p4_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase4 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 LP: obj={obj_lp:.8f}")

    # === Phase 5: More CD ===
    print(f"\n=== Phase 5: Final CD ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.15, 150.0)
    if p5_budget > 20:
        a_cd3 = coord_descent_fast(best_seq, time_budget=p5_budget, rng=rng, verbose=True)
        obj_cd3 = compute_obj(a_cd3)
        if obj_cd3 < best_score:
            best_score = obj_cd3
            best_seq = a_cd3.copy()
            print(f"  Phase5 CD: obj={best_score:.8f} **")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    n_options = sorted(set([
        max(200, n_current - 200), max(200, n_current - 100),
        n_current + 100, n_current + 200, n_current + 400,
        400, 600, 800, 1000, 1200, 1600, 2000
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000 and abs(n - n_current) > 50]
    
    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)
        
        # CD
        alloc1 = min(remaining * 0.4, 60.0)
        if alloc1 > 20:
            a_try = coord_descent_fast(a_try, time_budget=alloc1, rng=rng)
        
        # LP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.2, 40.0)
        if alloc2 > 15:
            a_try = alternating_lp(a_try, time_budget=alloc2)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        mode = (r + 3) % 12
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        # CD
        alloc1 = min(remaining * 0.4, 60.0)
        if alloc1 > 20:
            a_r = coord_descent_fast(a_r, time_budget=alloc1, rng=rng)

        # Brief LP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 30.0)
        if alloc2 > 15:
            a_r = alternating_lp(a_r, time_budget=alloc2)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
