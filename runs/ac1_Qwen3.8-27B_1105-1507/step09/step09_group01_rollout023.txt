"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast LP polishing (HiGHS via scipy), coordinate descent, multiplicative weights,
and aggressive perturbation strategies. Key improvements: faster LP solver,
coordinate descent for local refinement, finer line search with parabolic
interpolation, and stronger diversification.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a[k-j] if 0 <= k-j < n
        k_start = j
        k_end = min(2 * n - 1, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_fast(a, M_target=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Uses scipy.optimize.linprog with HiGHS.
    Returns (g, M_value) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)  # (2n-1) x n

    # Variables: [g_0, ..., g_{n-1}, M]
    # Minimize: [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Inequality constraints: A @ g - M <= 0  =>  [A, -1] @ x <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Bounds: g >= 0, M >= 0
    bounds = [(0, None)] * n + [(0, None)]

    if M_target is not None:
        # Add M <= M_target as inequality
        A_ub = np.vstack([A_ub, np.zeros(n + 1)])
        A_ub[-1, n] = 1.0
        b_ub = np.concatenate([b_ub, [M_target]])

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-12:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def fine_line_search(a_base, a_dir, t_lo=0.01, t_hi=1.0, max_eval=25):
    """
    Fine line search: golden section + parabolic refinement.
    Evaluates compute_obj at various t values.
    Returns (best_t, best_obj).
    """
    evals = 0

    def f(t):
        nonlocal evals
        a_try = (1.0 - t) * a_base + t * a_dir
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-12:
            return 1e10
        a_try = a_try / s
        evals += 1
        return compute_obj(a_try)

    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_lo, a_hi = t_lo, t_hi
    c_pt = a_hi - gr * (a_hi - a_lo)
    d_pt = a_lo + gr * (a_hi - a_lo)
    fc = f(c_pt)
    fd = f(d_pt)

    # Track points for parabolic interpolation
    points = [(c_pt, fc), (d_pt, fd)]

    for _ in range(max_eval - 2):
        if abs(a_hi - a_lo) < 0.002:
            break
        if fc < fd:
            a_hi = d_pt
            d_pt = c_pt
            fd = fc
            points.append((c_pt, fc))
            c_pt = a_hi - gr * (a_hi - a_lo)
            fc = f(c_pt)
        else:
            a_lo = c_pt
            c_pt = d_pt
            fc = fd
            points.append((d_pt, fd))
            d_pt = a_lo + gr * (a_hi - a_lo)
            fd = f(d_pt)

    t_golden = (a_lo + a_hi) / 2.0
    f_golden = f(t_golden)
    points.append((t_golden, f_golden))

    # Parabolic refinement using last 3 points
    best_t = t_golden
    best_f = f_golden

    # Try a few more points around the golden result
    for dt in [0.005, -0.005, 0.01, -0.01, 0.02, -0.02]:
        t_try = t_golden + dt
        if t_try > t_lo and t_try < t_hi and evals < max_eval:
            ft = f(t_try)
            if ft < best_f:
                best_f = ft
                best_t = t_try
                evals += 1

    return best_t, best_f


def coordinate_descent(a, time_budget=100.0, verbose=False):

        sweep += 1
        improved = False
        perm = np.arange(n)  # cyclic order

        for j in perm:
            if time.monotonic() - start > time_budget:

            # Try a range of values for a[j]
            # The optimal is near old_val, so try a geometric range

        # Actually this O(n^2) loop is too slow. Let me use FFT properly.
        # conv[k] = sum_i a[i] * a[k-i]
        # d/d(a[j]) conv[k] = a[k-j] (if 0<=k-j<n) + a[k-j] (if 0<=k-j<n, same term unless k-j=j)
        # = 2*a[k-j] if k-j != j, or a[j] if k-j = j (i.e., k=2j)
        # Wait no: conv[k] = sum_i a[i]*a[k-i]. The terms involving a[j] are:
        #   i=j: a[j]*a[k-j]
        #   k-i=j, i.e., i=k-j: a[k-j]*a[j]
        # If j != k-j: both terms present, giving 2*a[j]*a[k-j], derivative w.r.t. a[j] is 2*a[k-j]
        # If j == k-j (k=2j): one term a[j]^2, derivative is 2*a[j]
        # So d(conv[k])/d(a[j]) = 2*a[k-j] if 0<=k-j<n and k-j != j
        #                        = 2*a[j] if k=2j (i.e., k-j=j)
        #                        = 0 if k-j < 0 or k-j >= n

        # Using cross-correlation:
        # sum_k p[k] * 2 * a[k-j] (with bounds) = 2 * (p circ-cross-correlated with a) at position j
        # Let's define: h[j] = sum_k p[k] * a[k-j] (with 0<=k-j<n, 0<=k<2n-1)
        # This is the cross-correlation of p and a.

        # Using FFT: cross_corr(j) = sum_k p[k] * a[k-j]
        # = ifft(fft(p) * conj(fft(a)))[j] (with appropriate padding)

        N2 = 1
        while N2 < 3 * n:
            N2 <<= 1
        P2 = fft(p, N2)
        A2 = fft(a, N2)
        # cross-correlation: ifft(conj(FFT(a)) * FFT(p))
        cc = ifft(np.conj(A2) * P2).real
        # cc[j] = sum_k a[k] * p[k-j] ... need to be careful with indexing
        # Actually: ifft(conj(A) * P)[n] = sum_k A[k] * P[k-n] (circular)
        # Let me just use the direct O(n^2) approach since n=800 and we skip small p[k]
        g = np.zeros(n)
        nz = np.where(p > 1e-8)[0]
        for k in nz:
            pk = 2.0 * p[k]
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            others = k - np.arange(j_min, j_max + 1)
            g[j_min:j_max + 1] += pk * a[others]

        # Add entropy gradient: d/d(a[j]) sum(a[j]*log(a[j])) = log(a[j]) + 1
        g += np.log(a + 1e-30) + 1.0

        v = momentum * v + g
        a_new = a * np.exp(-eta * v)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            eta *= 0.8
            if stall > 50:
                eta = 0.3
                v *= 0.1
                stall = 0
                if it > 200:
                    break

        it += 1
        if verbose and it % 20 == 0:
            print(f"    MW it{it}: obj={best_obj:.8f}, eta={eta:.4f}, el={time.monotonic()-start:.1f}s")

    return best_a


def nesterov_smoothed_descent(a, time_budget=60.0, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        y = a + beta * v
        y = np.maximum(y, 1e-30)
        y = y / y.sum()

        # Smoothed gradient via FFT
        conv = compute_conv(y)
        max_c = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_c) / max_c)
        p = exp_vals / float(np.sum(exp_vals))

        N_fft = 1
        while N_fft < 3 * n:
            N_fft <<= 1
        P = rfft(p, N_fft)
        A_f = rfft(y, N_fft)
        g_full = irfft(P * A_f, N_fft)
        g = g_full[:n]
        # Add the flip contribution
        if N_fft >= 2 * n:
            g += g_full[N_fft - n + 1:N_fft - 1][::-1]

        # Proper gradient: d/d(a[j]) sum_k p[k]*conv[k]
        # = sum_k p[k] * 2*a[k-j] (with bounds)
        # Use cross-correlation
        from scipy.fft import fft, ifft
        P_full = fft(p, N_fft)
        A_full = fft(y, N_fft)
        cc = ifft(np.conj(A_full) * P_full).real
        g = 2.0 * cc[:n]

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-12:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-6, lr * 0.9998)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def lp_polish_intensive(a, time_budget=300.0, verbose=False, seed=42):
    """
    Intensive LP-based polishing with fine line search, multiple M targets,
    and stronger perturbations.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    no_improve = 0
    last_t = 0.2

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed

        if remaining < 2:
            break

        # Solve LP
        g_lp, M_lp = solve_lp_fast(a)
        if g_lp is None:
            # Try with a looser formulation
            no_improve += 5
            if no_improve > 10:
                break
            scale = 0.03 * min(no_improve, 5)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            continue

        # Fine line search
        t_opt, obj_try = fine_line_search(best_a, g_lp)
        iter_count += 1

        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            a_new = (1.0 - t_opt) * best_a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            best_a = a_new.copy()
            a = a_new.copy()
            last_t = t_opt
            no_improve = 0
            if verbose and iter_count % 10 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={t_opt:.3f}, M={M_lp:.6f}, el={elapsed:.1f}s")
        else:
            no_improve += 1
            if no_improve == 3:
                # Try constrained LP with tighter M
                if M_lp is not None:
                    for frac in [0.90, 0.95, 0.98]:
                        M_tight = float(np.max(compute_conv(best_a / best_a.sum()))) * frac
                        g_tight, _ = solve_lp_fast(a, M_target=M_tight)
                        if g_tight is not None:
                            t_try, obj_tight = fine_line_search(best_a, g_tight, max_eval=15)
                            if obj_tight < best_obj - 1e-14:
                                best_obj = obj_tight
                                a_new = (1.0 - t_try) * best_a + t_try * g_tight
                                a_new = np.maximum(a_new, 0)
                                a_new = a_new / a_new.sum()
                                best_a = a_new.copy()
                                a = a_new.copy()
                                no_improve = 0
                                break

            if no_improve >= 6:
                # Stronger perturbation
                scale = 0.02 + 0.015 * min(no_improve, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0

                if iter_count > 100 and remaining < 30:
                    break

        # Every 50 iterations, try a "jump"
        if iter_count % 50 == 0 and no_improve > 2:
            # Partial randomization
            frac_rand = 0.3
            mask = rng.random(n) < frac_rand
            a[mask] = best_a[mask] * np.exp(rng.normal(0, 0.3, mask.sum()))
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a, kind='cubic')
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Cosine window (high in middle)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = a / a.sum()
    elif mode == 3:
        # U-shape (high at edges)
        a = 0.5 * (1 - np.cos(2 * np.pi * x))
        a = np.maximum(a + 0.05, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Asymmetric: high on left
        a = np.exp(-3 * x)
        a = a / a.sum()
    elif mode == 5:
        # Asymmetric: high on right
        a = np.exp(-3 * (1 - x))
        a = a / a.sum()
    elif mode == 6:
        # Two bumps
        c1, c2 = 0.25, 0.75
        w = 0.1
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 7:
        # Sine power
        a = np.sin(np.pi * x) ** 4
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 8:
        # Skewed Gaussian
        a = np.exp(-((x - 0.3) ** 2) / 0.1) * (1 + 2 * x)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: fast LP polishing with HiGHS, coordinate descent,
    multiplicative weights, Nesterov smoothed GD, multi-n exploration,
    and aggressive perturbation restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Quick Nesterov Smoothed GD ===
    print(f"\n=== Phase 1: Nesterov Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.06, 60.0)
    if p1_budget > 15:
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=p1_budget,
                                          tau_init=5.0, tau_end=300.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase1 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 GD: obj={obj_sgd:.8f}")

    # === Phase 2: Intensive LP Polishing ===
    print(f"\n=== Phase 2: Intensive LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.30, 350.0)
    if p2_budget > 30:
        a_lp = lp_polish_intensive(best_seq, time_budget=p2_budget, verbose=True, seed=42)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f}")

    # === Phase 3: Coordinate Descent ===
    print(f"\n=== Phase 3: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.10, 100.0)
    if p3_budget > 20:
        a_cd = coordinate_descent(best_seq, time_budget=p3_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase3 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 CD: obj={obj_cd:.8f}")

    # === Phase 4: Multiplicative Weights ===
    print(f"\n=== Phase 4: Multiplicative Weights ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.06, 60.0)
    if p4_budget > 15:
        a_mw = multiplicative_weights(best_seq, time_budget=p4_budget, verbose=True)
        obj_mw = compute_obj(a_mw)
        if obj_mw < best_score:
            best_score = obj_mw
            best_seq = a_mw.copy()
            print(f"  Phase4 MW: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 MW: obj={obj_mw:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(300, n_current - 200), max(300, n_current - 100),
        n_current + 100, n_current + 200, n_current + 400,
        600, 700, 900, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000 and abs(n - n_current) > 50]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)

        # Quick LP polish
        alloc = min(remaining * 0.15, 80.0)
        if alloc > 20:
            a_try = lp_polish_intensive(a_try, time_budget=alloc, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Initialization + Optimization ===
    print(f"\n=== Phase 6: Diverse Inits ===")
    remaining = deadline - time.monotonic()
    n_use = len(best_seq)

    for mode in range(9):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_div = diverse_init(n_use, rng, mode)
        # Quick LP
        alloc = min(remaining * 0.1, 50.0)
        if alloc > 15:
            a_div = lp_polish_intensive(a_div, time_budget=alloc, seed=1000 + mode)

        obj_div = compute_obj(a_div)
        if obj_div < best_score:
            best_score = obj_div
            best_seq = a_div.copy()
            print(f"  Diverse mode={mode}: obj={best_score:.8f} **")
        else:
            print(f"  Diverse mode={mode}: obj={obj_div:.8f}")

    # === Phase 7: Perturbation + Re-LP ===
    print(f"\n=== Phase 7: Perturbation + Re-LP ===")
    remaining = deadline - time.monotonic()

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        scale = 0.015 + 0.012 * r
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc = min(remaining * 0.2, 100.0)
        if alloc > 20:
            a_pert = lp_polish_intensive(a_pert, time_budget=alloc, seed=2000 + r)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Perturb {r} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Perturb {r} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 8: Final Intensive LP ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 150.0)
        a_final = lp_polish_intensive(best_seq, time_budget=alloc, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

        # Quick coordinate descent on result
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc2 = min(remaining * 0.3, 40.0)
            a_cd2 = coordinate_descent(best_seq, time_budget=alloc2)
            obj_cd2 = compute_obj(a_cd2)
            if obj_cd2 < best_score:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
