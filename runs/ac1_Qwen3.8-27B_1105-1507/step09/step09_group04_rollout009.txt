import numpy as np

import cvxpy as cp
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def coord_descent_enhanced(a, time_budget=100.0, rng=None, verbose=False):
    """
    Enhanced coordinate descent with both single and pair coordinate updates.
    Uses golden section search with incremental convolution updates.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S
    conv = compute_conv(a)

    best_obj = float(np.max(conv))
    best_a = a.copy()
    gr = (np.sqrt(5) - 1) / 2.0
    start = time.monotonic()
    stagnant = 0
    sweep_count = 0

    def obj_delta_single(conv_arr, d, k, delta):
        """Objective when changing coordinate k by delta."""
        val = conv_arr + 2.0 * delta * d
        val[2 * k] += delta * delta
        return float(np.max(val))

    def obj_delta_pair(conv_arr, d1, d2, k1, k2, delta1, delta2):
        """Objective when changing coordinates k1,k2 by delta1,delta2."""
        val = conv_arr + 2.0 * delta1 * d1 + 2.0 * delta2 * d2
        val[2 * k1] += delta1 * delta1
        val[2 * k2] += delta2 * delta2
        # Cross term at position 2k1+2k2-k1-k2 = k1+k2... actually at position k1+k2
        # conv[2k] gets delta^2, but the cross term is at position k1+k2 (if in range)
        # Actually: conv[j] += 2*delta1*a[k2] if j = k1+k2 (from the 2*delta*d part)
        # The delta1*delta2 cross term appears at position k1+k2 in the convolution
        cross_pos = k1 + k2
        if 0 <= cross_pos <= 2 * n - 2:
            val[cross_pos] += 2.0 * delta1 * delta2
        return float(np.max(val))

    while time.monotonic() - start < time_budget:
        sweep_count += 1
        obj_before = float(np.max(conv))
        improvements = 0

        order = np.arange(n)
        if rng is not None and sweep_count % 3 == 1:
            order = rng.permutation(n)

        for k in order:
            if time.monotonic() - start > time_budget:
                break

            # Direction for coordinate k
            d = np.zeros(2 * n - 1)
            d[k:k + n] = a

            # Bounds for delta
            lo = -a[k]
            cur_max = float(np.max(conv))
            # Estimate upper bound
            hi_est = np.sqrt(max(0.0, cur_max - conv[2 * k] + a[k] ** 2 + 0.1 * cur_max)) - a[k]
            hi = max(hi_est, 2.0 / n, lo + 0.5 / n)

            # Golden section search
            a_gs, b_gs = lo, hi
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)
            fc = obj_delta_single(conv, d, k, c_gs)
            fd = obj_delta_single(conv, d, k, d_gs)

            for _ in range(40):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = obj_delta_single(conv, d, k, c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = obj_delta_single(conv, d, k, d_gs)

            delta_opt = (a_gs + b_gs) / 2.0
            new_obj = obj_delta_single(conv, d, k, delta_opt)

            if new_obj < obj_before - 1e-14:
                a[k] += delta_opt
                a[k] = max(0.0, a[k])
                conv += 2.0 * delta_opt * d
                conv[2 * k] += delta_opt * delta_opt
                S_new = float(np.sum(a))
                if S_new < 1e-10:
                    a[:] = 1.0 / n
                    conv = compute_conv(a)
                else:
                    a /= S_new
                    conv /= (S_new * S_new)
                improvements += 1

        # Try pair updates for a few pairs per sweep
        if rng is not None:
            n_pairs = min(10, n // 2)
            for _ in range(n_pairs):
                if time.monotonic() - start > time_budget:
                    break
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue

                d1 = np.zeros(2 * n - 1)
                d1[i:i + n] = a
                d2 = np.zeros(2 * n - 1)
                d2[j:j + n] = a

                # Simplified: just try small perturbations
                for try_delta1 in [0.5 / n, -0.5 / n, 1.0 / n]:
                    for try_delta2 in [0.5 / n, -0.5 / n, 1.0 / n]:
                        new_obj = obj_delta_pair(conv, d1, d2, i, j, try_delta1, try_delta2)
                        if new_obj < float(np.max(conv)) - 1e-14:
                            a[i] = max(0.0, a[i] + try_delta1)
                            a[j] = max(0.0, a[j] + try_delta2)
                            conv += 2.0 * try_delta1 * d1 + 2.0 * try_delta2 * d2
                            conv[2 * i] += try_delta1 * try_delta1
                            conv[2 * j] += try_delta2 * try_delta2
                            cross_pos = i + j
                            if 0 <= cross_pos <= 2 * n - 2:
                                conv[cross_pos] += 2.0 * try_delta1 * try_delta2
                            S_new = float(np.sum(a))
                            if S_new > 1e-10:
                                a /= S_new
                                conv /= (S_new * S_new)
                            improvements += 1
                            break
                    else:
                        continue
                    break

        obj_after = float(np.max(conv))
        if obj_after < best_obj - 1e-13:
            best_obj = obj_after
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 3 and rng is not None:
                scale = 0.005 * (1 + stagnant * 0.3)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv = compute_conv(a)
                best_obj = float(np.max(conv))
                stagnant = 0
            elif stagnant > 8:
                break

        if verbose and sweep_count % 20 == 0:
            print(f"    CD+ sweep {sweep_count}: obj={2.0*n*float(np.max(conv)):.8f}, t={time.monotonic()-start:.0f}s")

    return best_a, compute_conv(best_a)


def solve_lp_minpeak(a, timeout=60):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)

    # Build the convolution matrix A (2n-1) x n
    # A[k, j] = a_norm[j] if k-j is in [0, n-1] and j is in [0, n-1]
    # conv(a, g)[k] = sum_j a[j] * g[k-j] (valid when 0<=k-j<n and 0<=j<n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] contributes to conv positions j, j+1, ..., j+n-1 with weights a[0], a[1], ..., a[n-1]
        lo = 0
        hi = min(n, 2 * n - 1 - j)
        A[j:j + hi, j] = a_norm[:hi]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, timeout=timeout)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_iteration_with_linesearch(a, time_budget=200.0, verbose=False):
    """
    LP-based iteration with line search on blending parameter t.
    At each step: solve LP for g, then find optimal t in (0,1] via line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    it = 0
    last_improve = 0

    while time.monotonic() - start < time_budget:
        it += 1
        g_lp, M_lp = solve_lp_minpeak(a, timeout=min(60, time_budget - (time.monotonic() - start)))
        if g_lp is None:
            break

        # Line search on t
        gr = (np.sqrt(5) - 1) / 2.0
        a_lo, b_lo = 0.01, 1.0
        c_lo = b_lo - gr * (b_lo - a_lo)
        d_lo = a_lo + gr * (b_lo - a_lo)

        def obj_at_t(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return np.inf
            a_try = a_try / s_try
            return compute_obj(a_try)

        fc = obj_at_t(c_lo)
        fd = obj_at_t(d_lo)
        for _ in range(30):
            if fc < fd:
                b_lo = d_lo
                d_lo = c_lo
                fd = fc
                c_lo = b_lo - gr * (b_lo - a_lo)
                fc = obj_at_t(c_lo)
            else:
                a_lo = c_lo
                c_lo = d_lo
                fc = fd
                d_lo = a_lo + gr * (b_lo - a_lo)
                fd = obj_at_t(d_lo)

        t_opt = (a_lo + b_lo) / 2.0
        a_new = (1.0 - t_opt) * a + t_opt * g_lp
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            last_improve = it
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.3f}, t={time.monotonic()-start:.0f}s")
        else:
            # Try smaller t values
            for t_try in [0.05, 0.1, 0.02, 0.01]:
                a_try = (1.0 - t_try) * a + t_try * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        last_improve = it
                        break
            if it - last_improve > 15:
                # Perturb and retry
                scale = 0.01 + 0.005 * (it - last_improve)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                last_improve = it
                if it > 80:
                    break

    return best_a


def peak_targeting(a, time_budget=60.0, rng=None):
    """
    Identify positions where convolution is near its maximum and try to
    reduce them by adjusting the coordinates that most contribute.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = compute_conv(a)

    best_obj = float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    gr = (np.sqrt(5) - 1) / 2.0

    while time.monotonic() - start < time_budget:
        # Find top positions in convolution
        top_frac = max(1, n // 20)
        top_positions = np.argsort(conv)[-top_frac:]
        top_positions = np.sort(top_positions)

        improved = False
        for pos in top_positions:
            if time.monotonic() - start > time_budget:
                break
            # Which coordinate k most affects conv[pos]?
            # conv[pos] = sum_j a[j] * a[pos-j] (valid indices)
            # d(conv[pos])/d(a[k]) = a[pos-k] (if valid)
            # So reducing a[k] where a[pos-k] is large helps most

            # Find the coordinate k that has the largest contribution to this position
            # and try to reduce it
            contribs = np.zeros(n)
            for j in range(n):
                ij = pos - j
                if 0 <= ij < n:
                    contribs[j] = a[ij]  # contribution of a[j] to conv[pos]

            # Try reducing the most contributing coordinate
            k_red = np.argmax(contribs)
            if a[k_red] < 1e-6:
                continue

            # Direction: reduce a[k_red], increase a[k_inc] where it doesn't hurt
            # Find k_inc that minimizes the maximum increase in convolution
            d_red = np.zeros(2 * n - 1)
            d_red[k_red:k_red + n] = a

            # Try a few candidate increase positions (edges of the distribution)
            candidates_inc = [0, n - 1, rng.integers(0, n // 4) if rng else n // 4,
                              rng.integers(3 * n // 4, n) if rng else 3 * n // 4]
            candidates_inc = [c for c in candidates_inc if c != k_red and 0 <= c < n]

            for k_inc in candidates_inc:
                d_inc = np.zeros(2 * n - 1)
                d_inc[k_inc:k_inc + n] = a

                # We want to find delta such that:
                # a[k_red] decreases by delta, a[k_inc] increases by delta
                # Conv changes by: -delta*d_red + delta*d_inc + delta^2 adjustment
                # Actually: new a[k_red] = a[k_red] - delta, new a[k_inc] = a[k_inc] + delta
                # conv change: -delta*d_red + delta*d_inc (from cross terms)
                # plus: -delta^2 at position 2*k_red, +delta^2 at 2*k_inc
                # keep sum constant (no renormalization needed)

                lo_d = 0.0
                hi_d = a[k_red] * 0.5

                a_gs, b_gs = lo_d, hi_d
                c_gs = b_gs - gr * (b_gs - a_gs)
                d_gs = a_gs + gr * (b_gs - a_gs)

                def obj_rebalance(delta):
                    val = conv - delta * d_red + delta * d_inc
                    val[2 * k_red] -= delta * delta
                    val[2 * k_inc] += delta * delta
                    # Cross terms between k_red and k_inc:
                    cross = k_red + k_inc
                    if 0 <= cross <= 2 * n - 2:
                        val[cross] += 2.0 * delta * delta  # approximate
                    return float(np.max(val))

                fc = obj_rebalance(c_gs)
                fd = obj_rebalance(d_gs)
                for _ in range(30):
                    if fc < fd:
                        b_gs = d_gs
                        d_gs = c_gs
                        fd = fc
                        c_gs = b_gs - gr * (b_gs - a_gs)
                        fc = obj_rebalance(c_gs)
                    else:
                        a_gs = c_gs
                        c_gs = d_gs
                        fc = fd
                        d_gs = a_gs + gr * (b_gs - a_gs)
                        fd = obj_rebalance(d_gs)

                delta_opt = (a_gs + b_gs) / 2.0
                new_obj = obj_rebalance(delta_opt)
                if new_obj < float(np.max(conv)) - 1e-13:
                    a[k_red] -= delta_opt
                    a[k_inc] += delta_opt
                    a[k_red] = max(0.0, a[k_red])
                    a[k_inc] = max(0.0, a[k_inc])
                    conv -= delta_opt * d_red
                    conv += delta_opt * d_inc
                    conv[2 * k_red] -= delta_opt * delta_opt
                    conv[2 * k_inc] += delta_opt * delta_opt
                    cross = k_red + k_inc
                    if 0 <= cross <= 2 * n - 2:
                        conv[cross] += 2.0 * delta_opt * delta_opt
                    # Renormalize (should be close to 1 already)
                    s_chk = float(np.sum(a))
                    if abs(s_chk - 1.0) > 1e-6:
                        a /= s_chk
                        conv /= (s_chk * s_chk)
                    improved = True
                    break

        obj_now = float(np.max(conv))
        if obj_now < best_obj - 1e-12:
            best_obj = obj_now
            best_a = a.copy()
        if not improved:
            break

    return best_a


def try_symmetrize(a):
    """Check if symmetrizing the sequence improves the objective."""
    a_sym = (a + a[::-1].copy()) / 2.0
    a_sym = np.maximum(a_sym, 0)
    s = float(np.sum(a_sym))
    if s < 1e-10:
        return a
    a_sym = a_sym / s
    obj_orig = compute_obj(a)
    obj_sym = compute_obj(a_sym)
    if obj_sym < obj_orig:
        return a_sym
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Enhanced coordinate descent + LP iteration with line search +
    peak targeting + multi-n exploration + symmetrization.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_target = 800
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    n_cur = len(best_seq)

    # Try symmetrization first
    sym_seq = try_symmetrize(best_seq)
    sym_score = compute_obj(sym_seq)
    if sym_score < best_score:
        best_score = sym_score
        best_seq = sym_seq
        print(f"After symmetrization: obj={best_score:.8f}")

    # === Phase 1: Enhanced Coordinate Descent ===
    print(f"\n=== Phase 1: Enhanced CD (budget 350s) ===")
    p1_budget = min(350.0, deadline - time.monotonic())
    if p1_budget > 50:
        a_cd, _ = coord_descent_enhanced(best_seq, time_budget=p1_budget,
                                         rng=rng, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase1 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 CD: obj={obj_cd:.8f}")

    # === Phase 2: LP Iteration with Line Search ===
    print(f"\n=== Phase 2: LP + Line Search (budget 350s) ===")
    p2_budget = min(350.0, deadline - time.monotonic())
    if p2_budget > 50:
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            alloc = min(p2_budget / max(1, 3 - rnd), remaining * 0.6)
            if alloc < 30:
                break
            a_lp = lp_iteration_with_linesearch(best_seq, time_budget=alloc, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
                # Perturb for next round
                if rnd < 2:
                    scale = 0.01 * (rnd + 1)
                    best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                    best_seq = np.maximum(best_seq, 0)
                    best_seq = best_seq / best_seq.sum()

    # === Phase 3: Peak Targeting ===
    print(f"\n=== Phase 3: Peak Targeting (budget 80s) ===")
    p3_budget = min(80.0, deadline - time.monotonic())
    if p3_budget > 20:
        a_pt = peak_targeting(best_seq, time_budget=p3_budget, rng=rng)
        obj_pt = compute_obj(a_pt)
        if obj_pt < best_score:
            best_score = obj_pt
            best_seq = a_pt.copy()
            print(f"  Peak targeting: obj={best_score:.8f} **")
        else:
            print(f"  Peak targeting: obj={obj_pt:.8f}")

    # === Phase 4: Fine CD ===
    print(f"\n=== Phase 4: Fine CD (budget 150s) ===")
    p4_budget = min(150.0, deadline - time.monotonic())
    if p4_budget > 30:
        a_cd2, _ = coord_descent_enhanced(best_seq, time_budget=p4_budget,
                                          rng=rng, verbose=False)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  Fine CD: obj={best_score:.8f} **")
        else:
            print(f"  Fine CD: obj={obj_cd2:.8f}")

    # === Phase 5: Symmetrize and re-optimize ===
    sym_seq2 = try_symmetrize(best_seq)
    sym_score2 = compute_obj(sym_seq2)
    if sym_score2 < best_score:
        best_score = sym_score2
        best_seq = sym_seq2
        print(f"After re-symmetrization: obj={best_score:.8f} **")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    n_options = sorted(set([
        max(400, n_current - 100), n_current + 100, n_current + 200,
        900, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500 and abs(n - n_current) >= 50]

    for n_try in n_options[:3]:
        remaining = deadline - time.monotonic()
        if remaining < 120:
            break

        a_try = resample_sequence(best_seq, n_try)

        # CD
        alloc1 = min(remaining * 0.4, 60.0)
        if alloc1 > 20:
            a_try, _ = coord_descent_enhanced(a_try, time_budget=alloc1, rng=rng)

        # LP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.3, 50.0)
        if alloc2 > 15:
            a_try = lp_iteration_with_linesearch(a_try, time_budget=alloc2)

        # Symmetrize
        a_try = try_symmetrize(a_try)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # CD
    if remaining > 30:
        alloc = min(remaining * 0.3, 60.0)
        a_f, _ = coord_descent_enhanced(best_seq, time_budget=alloc, rng=rng)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # LP
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 50.0)
        a_f = lp_iteration_with_linesearch(best_seq, time_budget=alloc)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # One more peak targeting
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_f = peak_targeting(best_seq, time_budget=min(remaining * 0.3, 30.0), rng=rng)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final PT: obj={best_score:.8f} **")

    # Final symmetrization
    final_sym = try_symmetrize(best_seq)
    final_sym_score = compute_obj(final_sym)
    if final_sym_score < best_score:
        best_score = final_sym_score
        best_seq = final_sym
        print(f"  Final sym: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure values are in valid range for evaluate_sequence
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    # Scale so we're well within bounds (use sum=100 for safety)
    s_result = sum(result)
    if s_result > 0:
        result = [x * 100.0 / s_result for x in result]

    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
