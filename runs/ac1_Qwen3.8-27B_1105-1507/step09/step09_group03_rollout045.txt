"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses iterative active-set LP with top-K peak equalization, randomized direction
sampling, Nesterov-accelerated blending, and multi-length exploration.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_active_lp(a, active_indices, max_iters=1):
    """
    Solve LP: min T s.t. conv(a,g)[k] <= T for k in active_indices, sum(g)=1, g>=0.
    Returns (g, T) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    A_active = A[active_indices, :]
    
    g = cp.Variable(n, nonneg=True)
    T = cp.Variable()
    constraints = [A_active @ g <= T, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(T), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(T.value)
        except Exception:
            continue
    return None, None


def solve_full_lp(a):
    """
    Solve LP: min T s.t. conv(a,g)[k] <= T for all k, sum(g)=1, g>=0.
    Returns (g, T) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    T = cp.Variable()
    constraints = [A @ g <= T, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(T), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(T.value)
        except Exception:
            continue
    return None, None


def active_set_iteration(a, time_budget=100.0, K_top=10, K_expand=50, 
                          verbose=False, seed=42):
    """
    Iterative active-set LP:
    1. Identify top K peaks in conv(a,a)
    2. Solve LP with only those K constraints
    3. Blend with line search
    4. Periodically expand to K_expand constraints
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    it = 0
    stall = 0
    v = np.zeros(n)  # momentum
    momentum = 0.5
    
    while time.monotonic() - start < time_budget:
        # Identify top K peaks
        conv_aa = compute_conv(a / np.sum(a))
        top_K = np.argsort(conv_aa)[-K_top:][::-1]
        
        # Solve LP with active constraints
        g_lp, T_lp = solve_active_lp(a / np.sum(a), top_K)
        
        if g_lp is None:
            # Try full LP
            g_lp, T_lp = solve_full_lp(a / np.sum(a))
            if g_lp is None:
                break
        
        # Momentum update
        v = momentum * v + (g_lp - a)
        a_cand = project_simplex(a + v)
        
        # Line search
        best_t = 0.0
        best_f = best_obj
        for t in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            f_try = compute_obj(a_try)
            if f_try < best_f:
                best_f = f_try
                best_t = t
        
        if best_t > 0:
            a_new = (1 - best_t) * a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)
            
            if best_f < best_obj - 1e-12:
                best_obj = best_f
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                a = best_a.copy()
                stall += 1
        else:
            stall += 1
            a = best_a.copy()
        
        # Every 20 iterations, try with expanded constraints
        if it > 0 and it % 20 == 0:
            conv_aa = compute_conv(best_a / np.sum(best_a))
            top_K_ext = np.argsort(conv_aa)[-K_expand:][::-1]
            g_ext, _ = solve_active_lp(best_a / np.sum(best_a), top_K_ext)
            if g_ext is not None:
                for t in [1.0, 0.5, 0.25]:
                    a_try = (1 - t) * best_a + t * g_ext
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    f_try = compute_obj(a_try)
                    if f_try < best_obj - 1e-12:
                        best_obj = f_try
                        best_a = a_try.copy()
                        a = a_try
                        stall = 0
                        break
        
        # Perturb if stalled
        if stall > 30:
            scale = 0.01 + 0.005 * min(stall / 10, 3.0)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s_a = float(np.sum(a))
            if s_a > 1e-10:
                a = a / s_a
            v = np.zeros(n)
            stall = 0
            if it > 200:
                break
        
        # Decay momentum
        momentum = max(0.2, momentum * 0.999)
        
        it += 1
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    ASI it{it}: obj={best_obj:.8f}, T_lp={T_lp:.6f}, stall={stall}, t={elapsed:.1f}s")
    
    return best_a


def randomized_direction_lp(a, time_budget=100.0, verbose=False, seed=123):
    """
    Solve LPs with multiple random directions and pick the best one.
    At each step, pick a random direction d, solve LP with d fixed,
    and blend the result.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    it = 0
    K = min(20, 2 * n - 1)
    
    while time.monotonic() - start < time_budget:
        # Try multiple random directions
        best_dir = None
        best_dir_obj = best_obj
        
        for attempt in range(3):
            if attempt == 0:
                # Current direction
                d = a.copy()
            elif attempt == 1:
                # Random direction blended with current
                r = rng.dirichlet(np.ones(n) * 0.5)
                d = 0.7 * a + 0.3 * r
                d = d / d.sum()
            else:
                # Pure random
                d = rng.dirichlet(np.ones(n) * 0.3)
            
            # Get top K peaks of conv(d, a)
            conv_da = build_conv_matrix(d / d.sum()) @ a
            top_K = np.argsort(conv_da)[-K:][::-1]
            
            # Solve LP: min T s.t. conv(d, g)[k] <= T for k in top_K
            g_lp, _ = solve_active_lp(d / d.sum(), top_K)
            
            if g_lp is not None:
                # Evaluate blend
                for t in [1.0, 0.6, 0.3]:
                    a_try = (1 - t) * best_a + t * g_lp
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    f_try = compute_obj(a_try)
                    if f_try < best_dir_obj:
                        best_dir_obj = f_try
                        best_dir = (a_try, t)
        
        if best_dir is not None:
            a_new = best_dir[0]
            if best_dir_obj < best_obj - 1e-12:
                best_obj = best_dir_obj
                best_a = a_new.copy()
                a = a_new
        
        it += 1
        if verbose and it % 5 == 0:
            print(f"    RDL it{it}: obj={best_obj:.8f}")
        
        if time.monotonic() - start < time_budget and it % 50 == 0:
            # Periodic perturbation
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a


def smoothed_lasso_descent(a, time_budget=100.0, tau_init=20.0, tau_end=500.0, 
                            verbose=False, seed=555):
    """
    Gradient descent on the log-sum-exp smoothed max of conv(a,a).
    Uses Nesterov acceleration and adaptive step size.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9  # momentum
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        # Compute smoothed gradient
        conv_y = compute_conv(y)
        max_c = float(np.max(conv_y))
        # Softmax weights
        exp_vals = np.exp(tau * (conv_y - max_c))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient of smooth max wrt a: 2 * sum_k p[k] * (dy/d a) [conv at k]
        # conv_y[k] = sum_i y[i] * y[k-i], so d(conv_y[k])/dy[j] = y[k-j] + y[j-k] (if in range)
        # = 2 * (cross_correlation of y with p)
        N = 1
        while N < 3 * n:
            N <<= 1
        Y = rfft(y, N)
        P = rfft(p, N)
        # grad[j] = 2 * sum_k p[k] * (y[k-j] if k>=j else y[j-k])... 
        # Actually: d/dy_j sum_k p[k] * conv_y[k] = 2 * sum_k p[k] * y[|k-j|] (circular approx)
        # Using FFT: grad = 2 * ifft(P * conj(Y)) * 2 (for the full convolution gradient)
        grad_full = irfft(P * np.conj(Y), N)
        grad = 2.0 * grad_full[:n]
        
        # Update
        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)
        
        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it
        
        if verbose and it % 1000 == 0:
            print(f"    SLD it{it}: obj={best_obj:.8f}, tau={tau:.1f}")
    
    return best_a


def spectral_init(n, mode=0):
    """Generate spectral/structured initializations."""
    x = np.arange(n)
    if mode == 0:
        # Uniform
        a = np.ones(n)
    elif mode == 1:
        # Chirp-like (quadratic phase envelope)
        a = np.exp(-0.5 * ((x - n//2) / (n//4))**2)
    elif mode == 2:
        # Two-bump
        a = np.exp(-((x - n//4)**2) / (2*(n//10)**2)) + np.exp(-((x - 3*n//4)**2) / (2*(n//10)**2))
    elif mode == 3:
        # Cosine window
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    elif mode == 4:
        # Triangle
        a = 1.0 - 2.0 * np.abs(x - n//2) / n
    elif mode == 5:
        # Random smooth
        from scipy.interpolate import interp1d
        k = max(3, n // 20)
        pts = np.sort(rng_random(k))
        vals = np.abs(np.random.randn(k)) + 0.1
        f = interp1d(pts, vals, kind='cubic', fill_value='extrapolate')
        a = f(np.linspace(0, 1, n))
        a = np.maximum(a, 0)
    elif mode == 6:
        # Geometric from both ends
        r = 0.95
        a = np.maximum(r**x, r**(n-1-x))
    elif mode == 7:
        # Block pattern
        block = max(2, n // 20)
        a = np.zeros(n)
        for i in range(0, n, block * 2):
            a[i:i+block] = 1.0
    else:
        # sinc-like
        z = (x - n//2) / (n//6)
        a = np.sinc(z / np.pi)
        a = np.maximum(a, 0)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def rng_random(k):
    """Simple random sorted points."""
    import random
    random.seed(42)
    return sorted([random.random() for _ in range(k)])


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
