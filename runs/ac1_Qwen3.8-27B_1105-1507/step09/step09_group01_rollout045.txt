    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS on a softmax-parameterized smoothed objective with progressive
tau annealing, peak-equalization subgradient steps, symmetry exploitation,
multi-length exploration, and final LP polishing. Key differences from previous
approach: second-order L-BFGS optimization (vs first-order Nesterov), softmax
reparameterization (vs simplex projection), and iterative tau-annealing schedule.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_objective_grad(z, n, tau):
    """
    Objective and gradient for softmax-parameterized smoothed problem.
    a = softmax(z), minimize n * LSE(tau * conv(a)) / tau.
    """
    z_shift = z - np.max(z)
    a = np.exp(z_shift)
    a = a / a.sum()

    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    exp_v = np.exp(shifted)
    p = exp_v / float(np.sum(exp_v))

    obj_val = n * (max_c + float(np.log(np.sum(exp_v))) / tau)

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_a = 2.0 * ifft(P * np.conj(A)).real[:n]

    sg = float(np.dot(a, g_a))
    grad_z = n * a * (g_a - sg)

    return obj_val, grad_z


def lbfgs_run(a_init, tau, max_iter=500, time_limit=5.0):
    """Run one L-BFGS optimization at fixed tau with time limit."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-12)
    a_init = a_init / a_init.sum()
    z0 = np.log(a_init)

    best_obj = compute_obj(a_init)
    best_a = a_init.copy()
    start = time.monotonic()
    stop_flag = [False]

    def obj_wrapped(z):
        if time.monotonic() - start > time_limit:
            stop_flag[0] = True
            return np.inf, np.zeros(n)
        if stop_flag[0]:
            return np.inf, np.zeros(n)
        val, grad = lse_objective_grad(z, n, tau)
        if np.isfinite(val) and val < np.inf:
            obj_wrapped.current_obj = val
        return val, grad

    try:
        result = minimize(obj_wrapped, z0, jac=True, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'gtol': 1e-10,
                                   'ftol': 1e-14, 'maxfun': max_iter * 3})
        if result.x is not None and np.all(np.isfinite(result.x)):

            obj_final = compute_obj(a_final)
            if obj_final < best_obj:
                best_a = a_final
                best_obj = obj_final
    except Exception:
        pass

def progressive_lbfgs(a_init, time_budget=200.0, verbose=False):
    """
    Progressive tau-annealing L-BFGS optimization.
    Starts with smooth (small tau) and progressively sharpens (large tau).
    """

    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau_schedule = [5, 15, 40, 100, 250, 500, 1000, 2000]

    for i, tau in enumerate(tau_schedule):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        time_alloc = min(remaining * 0.3, 30.0)
        if time_alloc < 2:
            time_alloc = min(remaining, 5.0)

        a_new, obj_new = lbfgs_run(a, tau, max_iter=300, time_limit=time_alloc)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            if verbose:
                print(f"    Progressive BF it{tau}: obj={best_obj:.8f}")
        else:
            # Try with fewer iterations

                best_a = a_new2.copy()
                a = a_new2.copy()

def peak_subgradient_step(a, k_peaks=5, alpha=None):
    """
    Single subgradient step targeting the top k_peaks of the convolution.
    Returns updated a on simplex.
    """
    n = len(a)
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:k_peaks]

    # Compute subgradient from top peaks using FFT
    # Weight peaks by their values
    peak_vals = conv[top_idx]
    max_pv = peak_vals[0]
    if max_pv < 1e-15:
        return a.copy()
    weights = np.exp(3.0 * (peak_vals - max_pv) / max_pv)
    weights /= weights.sum()

    # g_j = sum_k w_k * (a_{k-j} + a_{j-k})
    # Use FFT: create weighted impulse at peak positions
    N = 1
    while N < 3 * n:
        N <<= 1
    p_weighted = np.zeros(N)
    for w, k in zip(weights, top_idx):
        p_weighted[int(k)] = w

    P = fft(p_weighted)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]

    g_norm = float(np.linalg.norm(g))
    if g_norm < 1e-15:
        return a.copy()

    if alpha is None:
        alpha = 1.0 / (n ** 0.7 * g_norm)

    a_new = a - alpha * g
    a_new = project_simplex(a_new)
    return a_new


def subgradient_loop(a, time_budget=30.0, k_peaks=5, verbose=False):
    """Iterative subgradient descent targeting top peaks."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    alpha = 2.0 / np.sqrt(n)
    stall = 0

    while time.monotonic() - start < time_budget:
        a_new = peak_subgradient_step(a, k_peaks=k_peaks, alpha=alpha)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            alpha *= 0.95
            if stall > 100:
                # Restart with larger step
                alpha = 2.0 / np.sqrt(n)
                stall = 0
                a = best_a.copy()
                # Add small noise
                noise = np.random.default_rng(int(time.monotonic()*1000)).normal(0, 0.01, n)
                a = a * np.exp(noise)
                a = np.maximum(a, 0)
                a = a / a.sum()

    return best_a, best_obj


def build_conv_matrix(a):
    """Build dense matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, M_target=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue

    return None, None


def lp_polish(a, time_budget=60.0, verbose=False):
    """LP-based polishing with blending."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for iter_idx in range(50):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Try blending with different t values
        best_t = 0
        best_blend_obj = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_t = t

        if best_t > 0 and best_blend_obj < best_obj - 1e-14:
            best_obj = best_blend_obj
            best_a = (1.0 - best_t) * best_a + best_t * g_lp
            best_a = np.maximum(best_a, 0)
            best_a = best_a / best_a.sum()
            a = best_a.copy()
            if verbose:
                print(f"    LP it{iter_idx}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            # Try constraint tightening
            if M_lp is not None:
                for frac in [0.9, 0.95]:
                    g_t, _ = solve_lp_minpeak(a, M_target=M_lp * frac)
                    if g_t is not None:
                        for t in [0.3, 0.5, 0.7]:
                            a_try = (1.0 - t) * best_a + t * g_t
                            a_try = np.maximum(a_try, 0)
                            a_try = a_try / a_try.sum()
                            obj_try = compute_obj(a_try)
                            if obj_try < best_obj - 1e-14:
                                best_obj = obj_try
                                best_a = a_try.copy()
                                a = a_try.copy()
                                break
                        if best_obj < best_obj:
                            break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def optimize_fixed_length(a_init, time_budget=100.0, verbose=False):
    """Full optimization pipeline for a fixed-length sequence."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase A: Progressive L-BFGS
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        a_bf, obj_bf = progressive_lbfgs(a, time_budget=min(remaining * 0.5, 120.0), verbose=verbose)
        if obj_bf < best_obj:
            best_obj = obj_bf
            best_a = a_bf.copy()
            a = a_bf.copy()
            if verbose:
                print(f"  L-BFGS: obj={best_obj:.8f}")

    # Phase B: Subgradient refinement
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        a_sg, obj_sg = subgradient_loop(a, time_budget=min(remaining * 0.3, 40.0), k_peaks=5)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()
            a = a_sg.copy()
            if verbose:
                print(f"  Subgrad: obj={best_obj:.8f}")

    # Phase C: More L-BFGS with higher tau
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        a_bf2, obj_bf2 = progressive_lbfgs(a, time_budget=min(remaining * 0.4, 60.0), verbose=verbose)
        if obj_bf2 < best_obj:
            best_obj = obj_bf2
            best_a = a_bf2.copy()
            a = a_bf2.copy()
            if verbose:
                print(f"  L-BFGS2: obj={best_obj:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: multi-phase optimization combining L-BFGS,
    subgradient methods, LP polishing, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Primary optimization on parent length ===
    print(f"\n=== Phase 1: Primary Optimization (n={n_target}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 400.0)
    if p1_budget > 30:
        a_opt, obj_opt = optimize_fixed_length(best_seq, time_budget=p1_budget, verbose=True)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase 1 result: obj={best_score:.8f}")

    # === Phase 2: Multi-length exploration ===
    print(f"\n=== Phase 2: Multi-Length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Generate candidate lengths around current best
    n_candidates = []
    for delta in [-200, -100, -50, 50, 100, 200, 400, 600]:
        n_try = n_current + delta
        if 300 <= n_try <= 2500:
            n_candidates.append(n_try)
    # Add some absolute values
    for n_abs in [500, 1000, 1500, 2000]:
        if n_abs not in n_candidates and abs(n_abs - n_current) > 100:
            n_candidates.append(n_abs)

    n_candidates = sorted(set(n_candidates))
    # Sort by priority: closer to current first, then by absolute proximity to 800-1200 range
    n_candidates = [n for n in n_candidates if n != n_current]

    # Limit to top 8 candidates by distance from sweet spot
    def n_priority(n):
        return abs(n - 1000) + 0.1 * abs(n - n_current)
    n_candidates.sort(key=n_priority)
    n_candidates = n_candidates[:8]

    for idx, n_try in enumerate(n_candidates):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        print(f"  Trying n={n_try} (idx={idx}, remaining={remaining:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)

        # Optimize at this length
        alloc = min(remaining * 0.25, 150.0)
        if alloc < 20:
            alloc = min(remaining * 0.4, 60.0)
        if alloc > 15:
            a_opt, obj_opt = optimize_fixed_length(a_try, time_budget=alloc, verbose=False)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                n_current = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 3: Re-optimize best length with more time ===
    print(f"\n=== Phase 3: Re-Optimization (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = min(remaining * 0.3, 300.0)
        a_opt, obj_opt = optimize_fixed_length(best_seq, time_budget=p3_budget, verbose=True)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase 3 result: obj={best_score:.8f}")

    # === Phase 4: Perturbation + re-optimize ===
    print(f"\n=== Phase 4: Perturbation + Re-Optimize ===")
    remaining = deadline - time.monotonic()

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        scale = 0.015 + 0.01 * r
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 1e-10)
        a_pert = a_pert / a_pert.sum()

        alloc = min(remaining * 0.25, 100.0)
        if alloc > 15:
            a_opt, obj_opt = optimize_fixed_length(a_pert, time_budget=alloc, verbose=False)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Perturb {r} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Perturb {r}: obj={obj_opt:.8f}")

    # === Phase 5: LP Polish ===
    print(f"\n=== Phase 5: LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 120.0)
        a_lp, obj_lp = lp_polish(best_seq, time_budget=alloc, verbose=True)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP result: obj={best_score:.8f}")

    # === Phase 6: Final L-BFGS ===
    print(f"\n=== Phase 6: Final L-BFGS Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 100.0)
        a_final, obj_final = progressive_lbfgs(best_seq, time_budget=alloc, verbose=True)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final BF: obj={best_score:.8f}")

    # Quick final subgradient
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.3, 30.0)
        a_sg, obj_sg = subgradient_loop(best_seq, time_budget=alloc, k_peaks=8)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
