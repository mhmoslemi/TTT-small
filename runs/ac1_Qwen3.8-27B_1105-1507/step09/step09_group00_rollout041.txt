"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based direction finding with quadratic subproblem solving, spectral shaping
for diversification, L-BFGS smoothing, and multi-n exploration.
Key innovation: exploits the sparse anti-banded structure of the weighted Hessian
to solve QP subproblems efficiently, combined with dual-weighted refinement.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (optimal_g, optimal_M, dual_weights) or (None, None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    g_val = None
    M_val = None
    status = None

    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            if warm_start is not None and solver == cp.MOSEK:
                g.value = warm_start
                prob.solve(solver=solver)
            else:
                prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                M_val = float(M.value)
                status = prob.status
                break
        except Exception:
            continue

    if g_val is None:
        return None, None, None

    s = float(np.sum(g_val))
    if s < 1e-10:
        return None, None, None
    g_val = g_val / s

    # Extract dual variables (lambda for A@g <= M constraints)
    dual_weights = None
    try:
        # The dual variables for the inequality constraints A@g <= M
        # In cvxpy, these are accessible via .dual_value
        for constraint in constraints:
            if constraint.dual_value is not None:
                dv = np.asarray(constraint.dual_value).flatten()
                if len(dv) == 2 * n - 1:
                    dual_weights = np.maximum(dv, 0)
                    dw_sum = float(np.sum(dual_weights))
                    if dw_sum > 1e-10:
                        dual_weights = dual_weights / dw_sum
                    break
    except Exception:
        pass

    if dual_weights is None:
        # Fallback: use softmax of convolution as weights
        conv_a = compute_conv(a_norm)
        max_conv = float(np.max(conv_a))
        dual_weights = np.exp(50.0 * (conv_a - max_conv))
        ds = float(np.sum(dual_weights))
        if ds > 1e-30:
            dual_weights = dual_weights / ds
        else:
            dual_weights = np.zeros(2 * n - 1)
            dual_weights[np.argmax(conv_a)] = 1.0

    return g_val, M_val, dual_weights


def solve_qp_weighted(a_current, dual_weights, max_iters=200, tol=1e-10):
    """
    Solve min a^T H a s.t. a >= 0, sum(a) = 1,
    where H = 2 * sum_k dual_weights[k] * M_k.
    Uses active-set method exploiting sparse structure.
    Returns the solution or None if it fails.
    """
    n = len(a_current)
    w = dual_weights
    # Only use lags with significant weight to keep H sparse
    thresh = max(float(np.max(w)) * 0.05, 1e-6)
    active_lags = np.where(w > thresh)[0]
    if len(active_lags) == 0:
        return a_current.copy()

    # Precompute the weighted anti-diagonals
    # H[i,j] = 2 * sum_k w[k] * (1 if i+j==k else 0)
    # So (H @ a)[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
    # = 2 * sum_{j: i+j in active_lags} w[i+j] * a[j]

    def H_matvec(v):
        """Compute H @ v efficiently using the sparse anti-banded structure."""
        result = np.zeros(n)
        for k in active_lags:
            wk = 2.0 * w[k]
            if k == 0:
                result[0] += wk * v[0]
            elif k == 2 * n - 2:
                result[n - 1] += wk * v[n - 1]
            elif 0 < k < n:
                # For j in [0, k]: result[j] += wk * v[k-j]
                idx = np.arange(0, min(k, n - 1) + 1)
                result[idx] += wk * v[k - idx]
            elif k == n - 1:
                idx = np.arange(0, n)
                result[idx] += wk * v[k - idx]
            elif n <= k < 2 * n - 2:
                # For j in [k-n+1, n-1]: result[j] += wk * v[k-j]
                j_start = k - n + 1
                j_end = n - 1
                idx = np.arange(j_start, j_end + 1)
                result[idx] += wk * v[k - idx]
        return result

    # Active set method
    S = set(range(n))  # Start with all variables active
    a_sol = a_current.copy()

    for it in range(max_iters):
        S_list = sorted(S)
        n_S = len(S_list)
        if n_S == 0:
            return None
        if n_S == 1:
            return np.zeros(n) + np.array([1.0 if i == S_list[0] else 0.0 for i in range(n)])

        # Build H_SS (submatrix for active set)
        # H_SS is symmetric with H_SS[i,j] = 2*w[i_S + j_S] for valid sums
        # Use the matvec to build it: H_SS @ e_j = H @ e_j restricted to S
        # For efficiency, build directly
        H_SS = np.zeros((n_S, n_S))
        idx_map = {i: pos for pos, i in enumerate(S_list)}
        for pos_i, i in enumerate(S_list):
            for pos_j, j in enumerate(S_list):
                k = i + j
                if 0 <= k < 2 * n - 1:
                    val = 2.0 * w[k]
                    if pos_j != pos_i:
                        H_SS[pos_i, pos_j] = val
                    else:
                        H_SS[pos_i, pos_j] = val  # diagonal: a[i]^2 coefficient

        # Solve H_SS @ a_S = (nu/2) * 1
        # a_S = H_SS^{-1} * 1 / (1^T * H_SS^{-1} * 1)
        try:
            ones = np.ones(n_S)
            # Check if H_SS is PSD (eigenvalues)
            # For large n_S, use Cholesky
            try:
                L = np.linalg.cholesky(H_SS + 1e-12 * np.eye(n_S))
                # H_SS is PSD
                H_SS_inv_ones = np.linalg.solve(L.T, np.linalg.solve(L, ones))
                denom = float(ones @ H_SS_inv_ones)
                if denom > 1e-12:
                    a_S = H_SS_inv_ones / denom
                else:
                    a_S = ones / n_S
            except np.linalg.LinAlgError:
                # H_SS not PSD, use pseudoinverse with regularization
                eigvals, eigvecs = np.linalg.eigh(H_SS)
                eigvals_reg = np.maximum(eigvals, 1e-8)
                H_SS_inv = (eigvecs / eigvals_reg) @ eigvecs.T
                H_SS_inv_ones = H_SS_inv @ ones
                denom = float(ones @ H_SS_inv_ones)
                if abs(denom) > 1e-12:
                    a_S = H_SS_inv_ones / denom
                else:
                    a_S = ones / n_S
        except Exception:
            a_S = ones / n_S

        # Assemble full solution
        a_new = np.zeros(n)
        for pos, i in enumerate(S_list):
            a_new[i] = max(0.0, a_S[pos])

        # Renormalize
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            return None
        a_new = a_new / s_new

        # Check KKT: for i not in S, gradient should be >= 0
        # grad_i = 2 * sum_k w[k] * a[k-i] (for valid k-i)
        grad = H_matvec(a_new)
        # The KKT condition: grad_i - nu >= 0 for i not in S
        # nu = gradient at active coordinates (should be equal for all active)
        nu = float(np.mean([grad[i] for i in S_list]))

        min_grad_inactive = np.inf
        worst_i = -1
        for i in range(n):
            if i not in S:
                slack = grad[i] - nu
                if slack < min_grad_inactive:
                    min_grad_inactive = slack
                    worst_i = i

        # Remove worst from S (if a_S[i] <= 0) or add best to S (if slack < 0)
        removed = False
        added = False

        # Remove variables with a_S <= 0
        for pos, i in enumerate(S_list):
            if a_S[pos] <= 1e-12:
                S.discard(i)
                removed = True
                break

        if not removed and min_grad_inactive < -1e-8 and worst_i >= 0:
            S.add(worst_i)
            added = True

        if not removed and not added:
            # Converged
            return a_new

        a_sol = a_new

    return a_sol


def spectral_shape(a, alpha=0.3, rng=None):
    """
    Apply spectral shaping: modify the power spectrum to be flatter,
    then project back to the simplex.
    """
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    F = fft(a, size)
    power = np.abs(F) ** 2

    # Compute current autocorrelation
    auto = ifft(power).real[:2 * n - 1]

    # Target: flatten the power spectrum
    # Modify power to reduce peaks in autocorrelation
    # p_new = 1 / (1 + beta * auto) gives "inverse filter"
    beta = alpha
    p_target = 1.0 / (1.0 + beta * np.maximum(auto, 0))
    p_target[0] = 1.0  # Keep DC

    # Blend original and target power
    p_new = (1 - alpha) * power + alpha * p_target * (float(np.sum(power)) / max(float(np.sum(p_target)), 1e-10))

    # Reconstruct with original phase
    phase = np.angle(F)
    F_new = np.sqrt(np.maximum(p_new, 0)) * np.exp(1j * phase)
    F_new[0] = np.abs(F_new[0])  # Keep DC real positive

    a_new = ifft(F_new, size).real[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lp_qp_iteration(a, time_limit, verbose=False):
    """
    Main LP + QP iteration loop.
    Alternates between LP direction finding and QP subproblem solving.
    """
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    warm_g = None
    stall = 0

    iter_count = 0
    last_blend = 0.5

    while time.monotonic() - start < time_limit:
        remaining = time_limit - (time.monotonic() - start)
        if remaining < 2:
            break

        # Step 1: Solve LP
        g_lp, M_lp, dual_w = solve_lp(a, warm_start=warm_g)
        if g_lp is None:
            break

        warm_g = g_lp.copy()

        # Step 2: Try QP subproblem
        a_qp = solve_qp_weighted(a, dual_w, max_iters=100)

        # Step 3: Line search / blending
        candidates = [g_lp]
        if a_qp is not None:
            candidates.append(a_qp)
        # Also try blend of LP and QP
        if a_qp is not None:
            candidates.append(0.5 * g_lp + 0.5 * a_qp)

        best_trial_obj = best_obj
        best_trial_a = best_a

        for trial in candidates:
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0,
                      max(0.05, last_blend * 0.5), min(1.0, last_blend * 1.5)]:
                a_try = (1.0 - t) * a + t * trial
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_trial_obj:
                    best_trial_obj = obj_try
                    best_trial_a = a_try.copy()
                    last_blend = t

        iter_count += 1

        if best_trial_obj < best_obj - 1e-14:
            best_obj = best_trial_obj
            best_a = best_trial_a.copy()
            # Move towards the better solution (aggressive update)
            t_move = min(1.0, last_blend * 1.2)
            a = best_a.copy()
            stall = 0
            if verbose and iter_count % 10 == 0:
                print(f"    LP+QP it{iter_count}: obj={best_obj:.8f}")
        else:
            stall += 1
            if stall > 3:
                # Perturb to escape
                if stall % 3 == 0:
                    a = spectral_shape(best_a, alpha=0.2 + 0.05 * (stall % 5))
                else:
                    # Small random perturbation
                    rng_local = np.random.default_rng(iter_count * 7919)
                    a = best_a * (1.0 + 0.02 * rng_local.standard_normal(n))
                    a = np.maximum(a, 0)
                    s_a = float(np.sum(a))
                    if s_a > 1e-10:
                        a = a / s_a
                    else:
                        a = best_a.copy()
                if stall > 20:
                    break

    return best_a, best_obj


def smoothed_gd_lbfgs(a, time_budget=60.0, tau_init=10.0, tau_end=500.0):
    """
    L-BFGS optimization of the smoothed objective using scipy.
    """
    from scipy.optimize import minimize

    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a_work = a / s

    best_obj = compute_obj(a_work)
    best_a = a_work.copy()
    start = time.monotonic()

    def smoothed_obj_and_grad(y):
        """Compute smoothed objective and gradient."""
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            return 0.0, np.zeros(len(y))
        y = y / s_y

        conv = compute_conv(y)
        max_c = float(np.max(conv))
        exp_vals = np.exp(tau_end * (conv - max_c))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        obj = len(y) * max_c * (1.0 + (1.0 / tau_end) * np.log(sum_exp))

        # Gradient via FFT
        N = 1
        while N < 3 * len(y):
            N <<= 1
        P = fft(p, N)
        A_f = fft(y, N)
        g_full = ifft(P * np.conj(A_f)).real
        g = 2.0 * g_full[:len(y)]

        return obj, g

    # Use L-BFGS-B with a few different tau values
    for tau_frac in [0.3, 0.6, 1.0]:
        if time.monotonic() - start >= time_budget:
            break
        # Define objective with current tau
        tau_current = tau_init + (tau_end - tau_init) * tau_frac

        def obj_fn(y):
            y = np.maximum(y, 0)
            s_y = float(np.sum(y))
            if s_y < 1e-10:
                return 0.0
            y_n = y / s_y
            conv = compute_conv(y_n)
            max_c = float(np.max(conv))
            exp_vals = np.exp(tau_current * (conv - max_c))
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                return len(y) * max_c
            return len(y) * (max_c + (1.0 / tau_current) * np.log(sum_exp))

        def obj_grad_fn(y):
            y = np.maximum(y, 0)
            s_y = float(np.sum(y))
            if s_y < 1e-10:
                return 0.0, np.zeros(len(y))
            y_n = y / s_y
            conv = compute_conv(y_n)
            max_c = float(np.max(conv))
            exp_vals = np.exp(tau_current * (conv - max_c))
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = exp_vals / sum_exp
            N = 1
            while N < 3 * len(y_n):
                N <<= 1
            P = fft(p, N)
            A_f = fft(y_n, N)
            g_full = ifft(P * np.conj(A_f)).real
            g = 2.0 * g_full[:len(y_n)]
            f = len(y_n) * (max_c + (1.0 / tau_current) * np.log(sum_exp))
            return f, g

        try:
            result = minimize(
                obj_grad_fn, a_work, method='L-BFGS-B',
                jac=True, options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-8,
                                   'maxfun': 2000, 'iprint': -1}
            )
            if result.success or result.fun < best_obj:
                a_candidate = np.maximum(result.x, 0)
                s_c = float(np.sum(a_candidate))
                if s_c > 1e-10:
                    a_candidate = a_candidate / s_c
                    obj_c = compute_obj(a_candidate)
                    if obj_c < best_obj:
                        best_obj = obj_c
                        best_a = a_candidate.copy()
        except Exception:
            pass

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 / (np.sqrt(1 - (2*x-1)**2 + 0.01))
        a = a / a.sum()
    elif mode == 3:
        # Skewed: power law
        a = np.arange(1, n+1) ** 0.5
        a = a / a.sum()
    elif mode == 4:
        # Anti-skewed
        a = np.arange(n, 0, -1) ** 0.5
        a = a / a.sum()
    elif mode == 5:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 6:
        # Sine-squared
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2
        a = a / a.sum()
    elif mode == 7:
        # Two peaks
        c = n // 3
        w = n // 10
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c)**2) / (2*w**2))
        a += 0.7 * np.exp(-((np.arange(n) - 2*c)**2) / (2*w**2))
        a = a / a.sum()
    elif mode == 8:
        # Tilted cosine (breaks symmetry)
        x = np.arange(n) / n
        a = (1 + 0.5*x) * (1 + 0.3*np.cos(2*np.pi*x*3))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 9:
        # Random smooth
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2*np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2*np.pi*f*x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0) / np.maximum(a.sum(), 1e-10)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP+QP optimization, L-BFGS smoothing, spectral shaping,
    multi-n exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_seq = parent.copy()
                best_score = compute_obj(parent)
                n_current = parent.size
                print(f"Parent (n={n_current}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(n_current) / n_current
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")

    # === Phase 1: LP+QP main optimization ===
    print(f"\n=== Phase 1: LP+QP Optimization (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.45, 600.0)
    if p1_budget > 30:
        a_lp, obj_lp = lp_qp_iteration(best_seq, time_budget=p1_budget, verbose=True)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj_lp:.8f}")

    # === Phase 2: L-BFGS smoothing ===
    print(f"\n=== Phase 2: L-BFGS Smoothing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 200.0)
    if p2_budget > 30:
        a_lbfgs = smoothed_gd_lbfgs(best_seq, time_budget=p2_budget,
                                     tau_init=5.0, tau_end=400.0)
        obj_lbfgs = compute_obj(a_lbfgs)
        if obj_lbfgs < best_score:
            best_score = obj_lbfgs
            best_seq = a_lbfgs.copy()
            print(f"  Phase2: obj={best_score:.8f} **")
        else:
            print(f"  Phase2: obj={obj_lbfgs:.8f}")

    # === Phase 3: Spectral shaping + re-optimization ===
    print(f"\n=== Phase 3: Spectral Shaping ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    if p3_budget > 20:
        for alpha in [0.1, 0.2, 0.4, 0.6]:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            a_spec = spectral_shape(best_seq, alpha=alpha, rng=rng)
            alloc = min(remaining * 0.3, 60.0)
            if alloc > 10:
                a_spec, obj_spec = lp_qp_iteration(a_spec, time_budget=alloc, verbose=False)
            else:
                obj_spec = compute_obj(a_spec)
            if obj_spec < best_score:
                best_score = obj_spec
                best_seq = a_spec.copy()
                print(f"  Spectral (alpha={alpha}): obj={best_score:.8f} **")
            else:
                print(f"  Spectral (alpha={alpha}): obj={obj_spec:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        n_best - 100, n_best - 200,
        n_best + 100, n_best + 200, n_best + 400, n_best + 800,
        1000, 1200, 1500, 2000, 2500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 3000 and abs(n - n_best) > 50]
    n_options = n_options[:8]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        # Resample
        n_old = len(best_seq)
        if n_old == 1:
            a_try = np.full(n_try, 1.0 / n_try)
        else:
            x_old = np.linspace(0, 1, n_old)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try

        # Quick LP+QP
        alloc = min(remaining * 0.15, 80.0)
        if alloc > 15:
            a_try, obj_try = lp_qp_iteration(a_try, time_budget=alloc, verbose=False)
        else:
            obj_try = compute_obj(a_try)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = 1 + r % 9
        a_r = diverse_init(n_cur, rng, mode=mode)
        # Blend with current best
        blend = 0.3 + 0.15 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        alloc = min(remaining * 0.2, 80.0)
        if alloc > 20:
            a_r, obj_r = lp_qp_iteration(a_r, time_budget=alloc, verbose=False)
        else:
            obj_r = compute_obj(a_r)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Multiple LP+QP rounds with perturbations
    for rnd in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * 0.25, 80.0)
        a_final, obj_final = lp_qp_iteration(best_seq, time_budget=alloc, verbose=False)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP+QP {rnd}: obj={best_score:.8f} **")
        else:
            # Small perturbation
            best_seq = spectral_shape(best_seq, alpha=0.05 + 0.02 * rnd)
            print(f"  Final LP+QP {rnd}: obj={obj_final:.8f} (perturbing)")

    # One last L-BFGS pass
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 60.0)
        a_final = smoothed_gd_lbfgs(best_seq, time_budget=alloc,
                                     tau_init=50.0, tau_end=800.0)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final L-BFGS: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
