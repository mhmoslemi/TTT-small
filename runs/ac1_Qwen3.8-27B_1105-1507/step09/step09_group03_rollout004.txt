def build_conv_matrix_fast(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))

        r = min(n, 2 * n - 1 - j)

def solve_lp_direction(a, time_budget=60.0):
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_fast(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]
    start = time.monotonic()
    for solver in solvers:
        if time.monotonic() - start > time_budget:
            break
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def golden_line_search(a, direction, n_evals=10):
    phi = (np.sqrt(5) - 1) / 2
    lo = 0.0
    hi = 1.0
    for _ in range(n_evals):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)
        a1 = np.maximum((1 - t1) * a + t1 * direction, 0)
        s1 = float(np.sum(a1))
        f1 = compute_obj(a1 / s1) if s1 > 1e-10 else np.inf
        a2 = np.maximum((1 - t2) * a + t2 * direction, 0)
        s2 = float(np.sum(a2))
        f2 = compute_obj(a2 / s2) if s2 > 1e-10 else np.inf
        if f1 < f2:
            hi = t2
        else:
            lo = t1
    t_best = (lo + hi) / 2
    a_best = np.maximum((1 - t_best) * a + t_best * direction, 0)
    s_best = float(np.sum(a_best))
    if s_best > 1e-10:
        a_best = a_best / s_best
    return a_best

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc_sa = min(remaining * 0.15, 120.0)
        a_sa = simulated_annealing(best_seq, time_budget=alloc_sa, n_restarts=3)
        s_orig = float(np.sum(best_seq))
        a_sa = a_sa * s_orig
        obj_sa = compute_obj(a_sa)
        if obj_sa < best_score:
            best_score = obj_sa
            best_seq = a_sa.copy()
            print(f"  SA: obj={best_score:.8f} **")
        else:
            print(f"  SA: obj={obj_sa:.8f}")

    # Phase 7: Final intensive polish
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.4, 150.0)
        a_final = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                            tau_start=500.0, tau_end=20000.0)
        s_orig = float(np.sum(best_seq))
        a_final = a_final * s_orig
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc_lp3 = min(remaining * 0.4, 100.0)
        a_lp3 = lp_polish(best_seq, time_budget=alloc_lp3)
        s_orig = float(np.sum(best_seq))
        a_lp3 = a_lp3 * s_orig
        obj_lp3 = compute_obj(a_lp3)
        if obj_lp3 < best_score:
            best_score = obj_lp3
            best_seq = a_lp3.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Validate and return
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
