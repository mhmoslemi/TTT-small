def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def coordinate_descent(a, time_budget=60.0, verbose=False):
    """
    Coordinate descent: try moving mass from high-contributing positions to low ones.
    """

    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_k = int(np.argmax(conv))
        
        # Find contributing pairs
        i_start = max(0, peak_k - (n - 1))
        i_end = min(n - 1, peak_k)
        
        contributions = []
        for i in range(i_start, i_end + 1):
            j = peak_k - i
            if 0 <= j < n:
                contributions.append((a[i] * a[j], i, j))
        
        contributions.sort(reverse=True)
        
        improved = False
        for contrib, src, dst in contributions[:5]:
            if src == dst:
                continue
            
            for delta_frac in [0.03, 0.01, 0.005, 0.002]:
                delta = delta_frac * a[src]
                if delta <= 0 or delta > a[src]:
                    continue
                
                # Try random targets
                for _ in range(5):
                    tgt = int(rng.integers(0, n))
                    if tgt == src:
                        continue
                    
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[tgt] += delta
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue

                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-15:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
            # Perturb
            idx = int(rng.integers(0, n))
            a_try = a.copy()
            a_try[idx] *= (1 + rng.uniform(-0.1, 0.1))

    return best_a

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a_norm
    return A


def solve_lp_minmax(a_norm, A):
    """
    Solve: min t s.t. A @ g <= t, sum(g) = 1, g >= 0.
    Returns (g, t) or (None, None).
    """
    n = len(a_norm)
    g = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    constraints = [A @ g <= t, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(t), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    t_val = float(np.max(A @ g_val))
                    return g_val, t_val
        except Exception:
            continue
    return None, None


def solve_lp_active_min(a_norm, A, active):
    """
    Solve: min sum(A[active] @ g) s.t. sum(g) = 1, g >= 0.
    Returns g or None.
    """
    n = len(a_norm)
    if len(active) == 0:
        return None
    c = A[active, :].sum(axis=0)
    g = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Minimize(cp.sum(c * g)), [cp.sum(g) == 1])

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def solve_lp_peak(a_norm, A, target_k):
    """
    Solve: min A[target_k] @ g s.t. sum(g) = 1, g >= 0.
    Returns g or None.
    """
    n = len(a_norm)
    if 0 <= target_k < A.shape[0]:
        c = A[target_k, :]
    else:
        c = A[int(np.argmax(np.abs(A).sum(axis=0))), :]
    g = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Minimize(cp.sum(c * g)), [cp.sum(g) == 1])

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def line_search(a, direction, max_iter=10):
    """Golden section search for best t in [0,1]: a_new = (1-t)*a + t*direction."""
    phi = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    best_a = a.copy()
    best_f = compute_obj(a)

    for _ in range(max_iter):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)

        a1 = np.maximum((1 - t1) * a + t1 * direction, 0)
        s1 = float(np.sum(a1))
        f1 = compute_obj(a1 / s1) if s1 > 1e-10 else np.inf

        a2 = np.maximum((1 - t2) * a + t2 * direction, 0)
        s2 = float(np.sum(a2))
        f2 = compute_obj(a2 / s2) if s2 > 1e-10 else np.inf

        if f1 < f2:
            hi = t2
        else:
            lo = t1

        t_mid = (lo + hi) / 2
        a_mid = np.maximum((1 - t_mid) * a + t_mid * direction, 0)
        s_mid = float(np.sum(a_mid))
        f_mid = compute_obj(a_mid / s_mid) if s_mid > 1e-10 else np.inf
        if f_mid < best_f:
            best_f = f_mid
            best_a = a_mid / s_mid

    return best_a


def multi_lp_polish(a, time_budget=400.0, verbose=False):
    """
    Multi-direction LP polishing:
    - Full min-max LP
    - Active-set min-sum LP
    - Peak-specific LP
    - Combined direction with line search
    - Dual-based perturbation when stalled
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(31415)

    stall = 0
    it = 0
    momentum = 0.3
    v = np.zeros(n)

    while time.monotonic() - start < time_budget:
        a_norm = a / float(np.sum(a))
        A = build_conv_matrix(a_norm)

        # Direction 1: Full min-max LP
        g1, M1 = solve_lp_minmax(a_norm, A)
        if g1 is None:
            break

        # Find active set
        conv_ag = A @ g1
        max_conv_ag = float(np.max(conv_ag))
        tol = 0.02 * max_conv_ag + 1e-12
        active = np.where(conv_ag >= max_conv_ag - tol)[0]

        # Direction 2: Active-set min-sum LP
        g2 = None
        if 0 < len(active) < 2 * n - 1:
            g2 = solve_lp_active_min(a_norm, A, active)

        # Direction 3: Peak-specific LP
        conv_aa = compute_conv(a_norm)
        peak_k = int(np.argmax(conv_aa))
        g3 = solve_lp_peak(a_norm, A, peak_k)

        # Combine directions
        direction = np.zeros(n)
        total_w = 0.0
        direction += 1.0 * g1
        total_w += 1.0
        if g2 is not None:
            direction += 0.6 * g2
            total_w += 0.6
        if g3 is not None:
            direction += 0.4 * g3
            total_w += 0.4

        direction = np.maximum(direction / total_w, 0)
        s_d = float(np.sum(direction))
        if s_d > 1e-10:
            direction = direction / s_d
        else:
            direction = g1

        # Momentum-assisted update
        v = momentum * v + (direction - a)
        a_mom = project_simplex(a + 0.3 * v)

        # Line search on both the raw direction and momentum direction
        a_ls1 = line_search(a, direction, max_iter=8)
        obj_ls1 = compute_obj(a_ls1)

        a_ls2 = line_search(a, a_mom, max_iter=5)
        obj_ls2 = compute_obj(a_ls2)

        if obj_ls1 <= obj_ls2:
            a_new = a_ls1
            obj_new = obj_ls1
        else:
            a_new = a_ls2
            obj_new = obj_ls2

        it += 1
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            momentum = min(0.6, 0.3 + 0.003 * min(it, 60))

            if verbose and it % 5 == 0:
                print(f"    MLP it{it}: obj={best_obj:.8f}, M1={M1:.6f}, |S|={len(active)}")
        else:
            stall += 1
            momentum = max(0.15, momentum * 0.92)
            v *= 0.4

            if stall > 20:
                # Structured perturbation targeting active peaks
                a = best_a.copy()
                conv_aa = compute_conv(a)
                top_lags = np.argsort(conv_aa)[::-1][:6]
                for pk in top_lags:
                    i_start = max(0, pk - (n - 1))
                    i_end = min(n - 1, pk)
                    for i in range(i_start, i_end + 1):
                        j = pk - i
                        if 0 <= j < n and a[i] > 0.05 / n:
                            a[i] *= (1.0 - rng.uniform(0.02, 0.08))
                # Add small mass to random low-impact positions
                for _ in range(min(10, n // 10)):
                    pos = int(rng.integers(0, n))
                    a[pos] += rng.uniform(0.001, 0.01) / n
                a = np.maximum(a, 0)
                s_a = float(np.sum(a))
                if s_a > 1e-10:
                    a = a / s_a
                else:
                    a = best_a.copy()
                v = np.zeros(n)
                momentum = 0.3
                stall = 0

                if it > 250:
                    break

                if verbose:
                    print(f"    MLP it{it}: perturbed (stall={stall}, it={it})")

    return best_a
