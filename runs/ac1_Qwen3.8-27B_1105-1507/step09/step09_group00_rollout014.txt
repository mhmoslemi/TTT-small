import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    """Compute 2n * max(conv) / sum^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build (2n-1) x n matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        r = min(n, 2*n-1-j)
        A[j:j+r, j] = a[:r]
    return A

def solve_lp(a, verbose=False):
    """Solve LP: min M s.t. A@ g <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 3*n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]

def subgradient_exact(a):
    """Compute exact subgradient of max(conv) at a."""
    n = len(a)
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    g = np.zeros(n)
    for m in range(n):
        val = 0.0
        if 0 <= k_star - m < n:
            val += a[k_star - m]
        if 0 <= m - k_star < n:
            val += a[m - k_star]

        if obj_best_n < best_score:
            best_score = obj_best_n
            best_seq = a_best_n.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_best_n:.8f}")
    
    # === Phase 4: Diverse restarts ===
    print(f"\n=== Phase 4: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        n_cur = len(best_seq)
        for r in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 80:
                break
            
            # Create diverse init
            if r == 0:
                a_r = random_perturbation(best_seq, rng, strength=0.15)
            elif r == 1:
                # Frequency-domain random
                size = 1
                while size < 2*n_cur:
                    size <<= 1
                mag = np.exp(-np.fft.rfftfreq(size) * n_cur * 0.5)
                phases = rng.uniform(0, 2*np.pi, size=size//2+1)
                F = mag * np.exp(1j*phases)
                F[0] = abs(F[0])
                a_r = np.fft.irfft(F, size)[:n_cur]
                a_r = np.maximum(a_r, 0)
                s = float(np.sum(a_r))
                a_r = a_r / s if s > 1e-10 else best_seq.copy()
            else:
                # Blend with uniform
                a_r = 0.5 * best_seq + 0.5 * np.ones(n_cur)/n_cur
                a_r = a_r / a_r.sum()
            
            # Optimize
            alloc = min(remaining * 0.3, 120.0)
            a_r = adaptive_gd(a_r, time_budget=alloc*0.5, tau_init=10, tau_end=400)
            remaining = deadline - time.monotonic()
            lp_alloc = min(remaining * 0.2, 60.0)
            if lp_alloc > 15:
                a_r2 = a_r.copy()
                stall_r = 0
                t_start_r = time.monotonic()
                while time.monotonic() - t_start_r < lp_alloc:
                    g_lp, _ = solve_lp(a_r2)
                    if g_lp is None: break
                    bt, obj_t = lp_line_search(a_r2, g_lp)
                    if obj_t < compute_obj(a_r2) - 1e-14:
                        a_r2 = (1-bt)*a_r2 + bt*g_lp
                        a_r2 = np.maximum(a_r2, 0) / np.sum(a_r2)
                        stall_r = 0
                    else:
                        stall_r += 1
                        if stall_r > 3:
                            a_r2 = fourier_perturbation(a_r2, rng, 0.03)
                            stall_r = 0
                        elif stall_r > 8: break
                a_r = a_r2
            
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")
    
    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    
    # Intensive LP
    a = best_seq.copy()
    stall_f = 0
    t_start_f = time.monotonic()
    lp_budget_f = min(remaining * 0.5, 200.0)
    
    while time.monotonic() - t_start_f < lp_budget_f:
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break
        best_t, obj_t = lp_line_search(a, g_lp)
        if obj_t < best_score - 1e-14:
            a = (1-best_t)*a + best_t*g_lp
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            a = a / s
            best_score = obj_t
            best_seq = a.copy()
            stall_f = 0
            if int(time.monotonic() - t_start_f) % 30 == 0:
                print(f"  Final LP t={time.monotonic()-t_start_f:.0f}s: obj={best_score:.8f}")
        else:
            stall_f += 1
            if stall_f <= 3:
                a = fourier_perturbation(best_seq, rng, 0.01 * stall_f)
            elif stall_f <= 6:
                a = random_perturbation(best_seq, rng, 0.03)
            else:
                a = best_seq.copy()
                # Try subgradient
                a = subgradient_polish(a, time_budget=min(20, remaining*0.1))
                stall_f = 0
            a = a / a.sum()
            if stall_f > 10:
                break
    
    # Final subgradient polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_sg = subgradient_polish(best_seq, time_budget=min(60, remaining*0.5), verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Subgrad: obj={best_score:.8f} **")
    
    # One more LP round if time
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a = best_seq.copy()
        for _ in range(10):
            if time.monotonic() > deadline:
                break
            g_lp, _ = solve_lp(a)
            if g_lp is None:
                break
            bt, obj_t = lp_line_search(a, g_lp)
            if obj_t < best_score - 1e-14:
                a = (1-bt)*a + bt*g_lp
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = obj_t
                best_seq = a.copy()
            else:
                break
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
