"""
Optimizes step function coefficients using mirror descent, block LP,
and spectral shaping to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    # Clip to avoid overflow
    tau_clipped = min(tau, 500.0)
    exp_vals = np.exp(tau_clipped * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    # Gradient: g[i] = 2 * sum_k p[k] * a[k-i]
    # This is 2 * cross_correlation(p, a)
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def mirror_descent(a, time_budget, tau_init=10.0, tau_end=300.0, verbose=False):
    """Mirror descent (multiplicative weights) on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    lr = 1.0 / np.sqrt(n)
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        # Gradually increase tau (sharpen the approximation)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)
        
        g = smoothed_gradient(a, tau)
        
        # Multiplicative update (mirror descent with negative entropy)
        # a_new[i] = a[i] * exp(-lr * g[i])
        log_update = -lr * g
        # Clip to avoid overflow
        log_update = np.clip(log_update, -10, 10)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

def block_lp_optimize(a, num_blocks, time_budget, verbose=False):
    """
    Block coordinate descent where each block is optimized via LP.
    For block b, fix all other elements and solve:
    min M s.t. (a_with_block)*a <= M, sum(block) free, block >= 0
    """
    n = len(a)
    block_size = n // num_blocks
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Precompute the contribution of "other" blocks to the convolution
    # conv(a, a) = conv(block_b, a) + conv(others, a)
    # For fixed others, conv(others, a) is fixed, and we need conv(block_b, a) <= M - conv(others, a)
    
    rng = np.random.default_rng(123)
    block_order = list(range(num_blocks))
    iteration = 0
    
    while time.monotonic() - start < time_budget:
        rng.shuffle(block_order)
        
        for b in block_order:
            if time.monotonic() - start > time_budget:
                break
            
            start_idx = b * block_size
            end_idx = min(n, (b + 1) * block_size)
            bs = end_idx - start_idx
            
            if bs <= 1:
                continue
            
            # The convolution at position k is:
            # (a*a)[k] = sum_i a[i]*a[k-i]
            # = sum_{i in block} a[i]*a[k-i] + sum_{i not in block} a[i]*a[k-i]
            
            # For the LP, we parameterize the block as x (length bs)
            # The constraint is: for each k, sum_{i in block} x[i]*a[k-(start_idx+i)] + fixed_part[k] <= M
            # where fixed_part[k] = sum_{i not in block} a[i]*a[k-i]
            
            # This is: A_block @ x + fixed <= M
            # where A_block[k, i] = a[k - start_idx - i] (if valid)
            
            # Build A_block (sparse)
            # For efficiency, only include rows where the block actually contributes
            
            active_rows = []
            A_rows = []
            fixed_vals = []
            
            full_conv = compute_conv(a)
            
            for k in range(2 * n - 1):
                # Check if any element of the block contributes to position k
                # Block element at position start_idx+i contributes to k when k-i-start_idx is in [0, n-1]
                # i.e., start_idx+i <= k <= start_idx+i+n-1
                # So k ranges from start_idx to start_idx + (bs-1) + n - 1
                # And also k from 0 to start_idx + bs - 1 (when k-i-start_idx >= 0 means i <= k-start_idx)
                
                i_min = max(0, k - (start_idx + bs - 1) - (n - 1))  # k - i - start_idx >= 0 => i <= k - start_idx, and k-i-start_idx <= n-1 => i >= k-start_idx-(n-1)
                i_max = min(bs - 1, k - start_idx)
                
                if i_min > i_max:
                    continue
                
                # This row is active
                # A_row[i] = a[k - start_idx - i] for i in [i_min, i_max]
                # Fixed part = sum_{j not in block} a[j] * a[k-j]
                
                # For the full row sum:
                # (a*a)[k] = sum_all_j a[j]*a[k-j]
                # Block contribution = sum_{j in block} a[j]*a[k-j]
                # Fixed = (a*a)[k] - block_contribution
                
                block_contrib = 0
                for i in range(i_min, i_max + 1):
                    j = start_idx + i
                    if 0 <= k - j < n:
                        block_contrib += a[j] * a[k - j]
                
                fixed = full_conv[k] - block_contrib
                
                # Build the row
                row = np.zeros(bs)
                for i in range(i_min, i_max + 1):
                    j = start_idx + i
                    if 0 <= k - j < n:
                        row[i] = a[k - j]
                
                active_rows.append(k)
                A_rows.append(row)
                fixed_vals.append(fixed)
            
            if len(active_rows) == 0:
                continue
            
            A_block = np.array(A_rows)
            fixed = np.array(fixed_vals)
            M_current = float(np.max(full_conv))
            
            # LP: min M s.t. A_block @ x + fixed <= M, x >= 0, sum(x) = sum(block)
            # Actually, we should allow the sum to change too for more flexibility
            # But let's fix the sum to maintain normalization
            
            block_sum = float(np.sum(a[start_idx:end_idx]))
            
            x = cp.Variable(bs, nonneg=True)
            M = cp.Variable()
            constraints = [
                A_block @ x + fixed <= M,
                cp.sum(x) == block_sum
            ]
            prob = cp.Problem(cp.Minimize(M), constraints)
            
            solved = False
            for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
                try:
                    prob.solve(solver=solver)
                    if x.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                        x_val = np.maximum(x.value, 0)
                        s_x = x_val.sum()
                        if s_x > 1e-10:
                            x_val = x_val * block_sum / s_x
                            a[start_idx:end_idx] = x_val
                            solved = True
                            break
                except Exception:
                    continue
            
            if solved:
                a = np.maximum(a, 0)
                a = a / a.sum()
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a.copy()
            
            iteration += 1
            if verbose and iteration % 10 == 0:
                print(f"    BlockLP it{iteration}: obj={best_obj:.8f}")
        
        # Check if we should stop
        if iteration > num_blocks * 20:
            # Check for convergence
            break

def spectral_shaping(a, time_budget, verbose=False):
    """
    Use frequency domain insights to guide optimization.
    The idea: flatten the power spectrum to reduce autocorrelation peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    lr = 0.1
    
    while time.monotonic() - start < time_budget:
        # Compute power spectrum
        size = 1
        while size < 2 * n:
            size <<= 1
        A = rfft(a, size)
        power = np.abs(A) ** 2
        
        # Target: flat spectrum (uniform power)
        target = np.ones_like(power) * (power.sum() / len(power))
        
        # Gradient w.r.t. a: d(power)/d(a) relates to 2*Re(A * conj(dA/da))
        # dA/da: A is linear in a, so dA[k]/da[i] = exp(-2πi*k*i/size)
        # This gives us the gradient of sum(power - target)^2 w.r.t. a
        # But this is a different objective, so let's use it as a regularizer
        
        # Simpler approach: compute the "excess" at each frequency and
        # use it to guide how to modify a
        excess = power - target
        excess = excess / (np.max(np.abs(excess)) + 1e-15)
        
        # Map frequency-domain excess back to time domain
        # This gives a direction in the frequency domain
        # Convert to time domain: the "correction" signal
        A_corrected = A * (1.0 - 0.1 * excess / (power + 1e-15))
        a_corrected = irfft(A_corrected, size)[:n]
        
        # Blend
        t = lr * (1.0 - 0.5 * (it / max(1, int(time_budget / 2))))
        a_new = (1 - t) * a + t * np.maximum(a_corrected, 0)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            lr *= 1.05
        else:
            lr *= 0.9
            a = best_a
            if lr < 0.001:
                break
        
        a = a_new
        it += 1
        
        if verbose and it % 100 == 0:
            print(f"    Spectral it{it}: obj={best_obj:.8f}, lr={lr:.4f}")
    
    return best_a

def build_conv_matrix(a):
    """Build matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # a[j] * g contributes to positions j, j+1, ..., j+n-1
        # conv(a,g)[k] = sum_j a[j]*g[k-j]
        # So a[j]*g[i] contributes to position j+i
        # For fixed j, as i varies over [0,n-1], position = j+i ranges over [j, j+n-1]
        start = j
        end = min(m, j + n)
        A[start:end, j] = a[:end-start]
    return A

def diversity_mutate(a, rng, magnitude=0.1):
    """Apply a structured mutation to escape local optima."""
    n = len(a)
    mode = rng.integers(0, 5)
    
    if mode == 0:
        # Gaussian perturbation
        a_new = a * np.exp(rng.normal(0, magnitude, n))
    elif mode == 1:
        # Block swap
        block_size = rng.integers(10, n // 4)
        i = rng.integers(0, n - block_size)
        j = rng.integers(0, n - block_size)
        if i != j:
            a_new = a.copy()
            a_new[i:i+block_size], a_new[j:j+block_size] = a[j:j+block_size].copy(), a[i:i+block_size].copy()
        else:
            a_new = a.copy()
    elif mode == 2:
        # Frequency-domain perturbation
        size = 1
        while size < 2 * n:
            size <<= 1
        A = rfft(a, size)
        A *= np.exp(rng.normal(0, magnitude/2, size//2+1) + 1j*rng.normal(0, magnitude/4, size//2+1))
        a_new = irfft(A, size)[:n]
    elif mode == 3:
        # Gaussian kernel smoothing + noise
        from scipy.ndimage import gaussian_filter1d
        a_smooth = gaussian_filter1d(a, sigma=rng.uniform(1, n/10))
        a_new = (1-magnitude) * a + magnitude * a_smooth + magnitude*0.5*rng.normal(0, 1, n)
    else:
        # Random restart with memory
        a_new = np.maximum(a * rng.uniform(0.5, 1.5, n), 0)
    
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    
    return a_new

def nesterov_smoothed_descent(a, time_budget, tau_init=10.0, tau_end=400.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)
        
        if it % 2000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")
        
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        g = smoothed_gradient(y, tau)
        
        # Backtracking line search for adaptive lr
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 20.0 / np.sqrt(n))
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)
        
        if it - best_it > 10000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 5.0 / np.sqrt(n) * 0.5
            best_it = it
    
    return best_a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Active-set LP (fast, iterative) combined with Nesterov smoothed GD,
mirror descent for diversity, and adaptive multi-n search.
Key innovation: active-set LP that only uses top-K peak constraints for speed,
enabling far more LP iterations than the full-constraint approach.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    tau_c = min(tau, 500.0)
    exp_vals = np.exp(tau_c * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def nesterov_gd(a, time_budget, tau_init=10.0, tau_end=400.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        if it % 3000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)

        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 20.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)

        if it - best_it > 10000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 5.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def active_set_lp(a, K, time_budget, verbose=False):
    """
    Active set LP: solve min M s.t. A_topK @ g <= M, sum(g)=1, g>=0
    where A_topK uses only the top-K peak constraints.
    Much faster than full LP, enabling many more iterations.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    last_improvement = 0
    rng = np.random.default_rng(99)

    # Initialize active set
    conv = compute_conv(a)
    active = np.argsort(conv)[::-1][:K].astype(int)

    while time.monotonic() - start < time_budget:
        # Build active constraint matrix
        A_rows = []
        for k in active:
            k = int(k)
            row = np.zeros(n)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                for i in range(i_start, i_end + 1):
                    j = k - i
                    if 0 <= j < n:
                        row[i] = a[j]
            A_rows.append(row)

        if len(A_rows) == 0:
            break
        A_active = np.array(A_rows)

        # Solve LP
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A_active @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        solved = False
        for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s_g = g_val.sum()
                    if s_g > 1e-10:
                        g_val /= s_g
                        solved = True
                        break
            except Exception:
                continue

        if not solved:
            it += 1
            if it - last_improvement > 30:
                break
            continue

        # Check feasibility against ALL constraints
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        A_fft = rfft(a, size)
        G_fft = rfft(g_val, size)
        conv_ag = irfft(A_fft * G_fft, size)[:2 * n - 1]
        M_actual = float(np.max(conv_ag))

        if M_actual <= float(M.value) * (1 + 1e-5):
            # Feasible! Try blending
            best_t = 0
            best_t_obj = best_obj
            for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
                a_try = (1 - t) * a + t * g_val
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try /= s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_t_obj:
                    best_t_obj = obj_try
                    best_t = t

            if best_t > 0 and best_t_obj < best_obj - 1e-13:
                best_obj = best_t_obj
                best_a = ((1 - best_t) * a + best_t * g_val)
                best_a = np.maximum(best_a, 0)
                best_a /= best_a.sum()
                a = best_a.copy()
                last_improvement = it
                # Update active set
                conv = compute_conv(a)
                active = np.argsort(conv)[::-1][:K].astype(int)
                if verbose and it % 10 == 0:
                    print(f"    AS-LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            # Infeasible: update active set with violated constraint
            violated_k = int(np.argmax(conv_ag))
            if violated_k not in active.tolist():
                # Replace weakest active constraint
                active_vals = conv_ag[active]
                min_idx = int(np.argmin(active_vals))
                active[min_idx] = violated_k
                active = np.unique(active)[:K]

        it += 1
        if it - last_improvement > 50:
            # Stuck: perturb and restart
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            a /= a.sum()
            conv = compute_conv(a)
            active = np.argsort(conv)[::-1][:K].astype(int)
            last_improvement = it
            if it > 500:
                break

    return best_a


def mirror_descent(a, time_budget, tau_init=20.0, tau_end=250.0, verbose=False):
    """Mirror descent (multiplicative weights) on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    lr = 1.5 / np.sqrt(n)
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)

        g = smoothed_gradient(a, tau)
        log_update = np.clip(-lr * g, -8, 8)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            lr = min(lr * 1.002, 10.0 / np.sqrt(n))
        else:
            lr *= 0.93

        a = a_new
        it += 1

        if it - best_it > 3000:
            a = best_a.copy()
            lr = 1.5 / np.sqrt(n)
            best_it = it

        if verbose and it % 500 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")

    return best_a


def build_full_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = a[:end - j]
    return A

def diversity_mutate(a, rng, magnitude=0.08):
    """Apply a structured mutation to escape local optima."""
    n = len(a)
    mode = rng.integers(0, 6)

    if mode == 0:
        a_new = a * np.exp(rng.normal(0, magnitude, n))
    elif mode == 1:
        # Block rotation
        block_size = rng.integers(max(10, n // 20), max(11, n // 5))
        i = rng.integers(0, n - block_size)
        a_new = np.concatenate([a[i:], a[:i]])
        a_new = a * rng.uniform(0.8, 1.2, n)
    elif mode == 2:
        # Frequency-domain perturbation
        size = 1
        while size < 2 * n:
            size <<= 1
        A = rfft(a, size)
        phase_noise = rng.normal(0, magnitude / 3, size // 2 + 1)
        A *= np.exp(1j * phase_noise)
        a_new = irfft(A, size)[:n]
    elif mode == 3:
        # Smooth + re-noise
        from scipy.ndimage import gaussian_filter1d
        sigma = rng.uniform(2, n / 8)
        a_smooth = gaussian_filter1d(a, sigma=sigma)
        a_new = (1 - magnitude) * a + magnitude * a_smooth + 0.01 * rng.standard_normal(n)
    elif mode == 4:
        # Inverse (peaks become valleys)
        a_max = a.max()
        a_new = a_max - a + 0.01
    else:
        # Random restart with memory
        a_new = a * rng.uniform(0.3, 1.5, n)

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

                    print(f"  n={n_try}: obj={best_score:.8f} **")

    # One final Nesterov pass
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 40.0)
        a_fin = nesterov_gd(best_seq, time_budget=alloc,
                            tau_init=50.0, tau_end=500.0)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Final NSD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
