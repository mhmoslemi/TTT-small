"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Multi-strategy approach: multi-length exploration, dual-informed two-stage LP,
adaptive Nesterov smooth descent, systematic peak-targeted coordinate descent,
and multi-scale upsample-refine cycle.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a / s)))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, active_indices=None, get_dual=False):
    """Solve LP: min M s.t. conv(a_norm, g) <= M (or active only), sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    
    if active_indices is not None:
        active_A = A[active_indices, :]
        obj = cp.sum(active_A @ g)
        constraints = [cp.sum(g) == 1]
    else:
        M = cp.Variable()

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2 with sum-normalization."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a / s)))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, active_indices=None, get_dual=False):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k (or active_indices), sum(g)=1, g>=0.
    Returns (g, M_value, dual_values) or (None, None, None).
    """
    n = len(a)
    a_sum = float(np.sum(a))
    if a_sum < 1e-10:
        return None, None, None
    a_norm = a / a_sum
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)

    if active_indices is not None:
        active_A = A[active_indices, :]
        obj = cp.sum(active_A @ g)
        constraints = [cp.sum(g) == 1]
        M_var = None
    else:
        M_var = cp.Variable()
        obj = M_var
        constraints = [A @ g <= M_var, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(obj), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0).astype(np.float64)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    if get_dual and active_indices is None:
                        dual_vals = None
                        try:
                            dual_vals = constraints[0].dual_value
                        except Exception:
                            dual_vals = None
                        return g_val, float(M_var.value), dual_vals
                    return g_val, None, None
        except Exception:
            continue
    return None, None, None


def golden_section_line_search(a, g, n_evals=8):
    """Find optimal t in [0,1] that minimizes compute_obj(project((1-t)*a + t*g))."""
    phi = (np.sqrt(5) - 1) / 2
    lo = 0.0
    hi = 1.0

    for _ in range(n_evals):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)

        a1 = project_simplex((1 - t1) * a + t1 * g)
        f1 = compute_obj(a1)
        a2 = project_simplex((1 - t2) * a + t2 * g)
        f2 = compute_obj(a2)

        if f1 < f2:
            hi = t2
        else:
            lo = t1

    t_best = (lo + hi) / 2
    a_best = project_simplex((1 - t_best) * a + t_best * g)
    return a_best

def smoothed_gradient_fft(a, tau):
    """Compute subgradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv

def peak_targeted_cd(a, time_budget=60.0, verbose=False):

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)

    elif mode == 6:

    else:

        a = x ** 0.5 * (1 - x) ** 0.5

            print(f"  ** n={n_try}: obj={best_score:.8f} IMPROVED")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 2: Intensive Two-Stage LP ===
    print(f"\n=== Phase 2: Intensive 2SLP ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.3, 400.0)
    if p2_budget > 30:
        a_ts = two_stage_lp_polish(best_seq, time_budget=p2_budget, verbose=True)
        obj_ts = compute_obj(a_ts)
        if obj_ts < best_score:
            best_score = obj_ts
            best_seq = a_ts.copy()
            print(f"  2SLP: obj={best_score:.8f} **")
        else:
            print(f"  2SLP: obj={obj_ts:.8f}")

    # === Phase 3: Accelerated LP ===
    print(f"\n=== Phase 3: Accelerated LP ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 250.0)
    if p3_budget > 30:
        a_alp = accelerated_lp(best_seq, time_budget=p3_budget, momentum=0.5, verbose=True)
        obj_alp = compute_obj(a_alp)
        if obj_alp < best_score:
            best_score = obj_alp
            best_seq = a_alp.copy()
            print(f"  ALP: obj={best_score:.8f} **")
        else:
            print(f"  ALP: obj={obj_alp:.8f}")

    # === Phase 4: Nesterov Smoothed GD ===
    print(f"\n=== Phase 4: Nesterov Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.2, 250.0)
    if p4_budget > 30:
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=p4_budget,
                                          tau_init=20.0, tau_end=600.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  NSD: obj={best_score:.8f} **")
        else:
            print(f"  NSD: obj={obj_sgd:.8f}")

    # === Phase 5: Peak-Targeted CD ===
    print(f"\n=== Phase 5: Peak-Targeted CD ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.1, 100.0)
    if p5_budget > 15:
        a_cd = peak_targeted_cd(best_seq, time_budget=p5_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  PCD: obj={best_score:.8f} **")
        else:
            print(f"  PCD: obj={obj_cd:.8f}")

    # === Phase 6: Multi-scale refine ===
    print(f"\n=== Phase 6: Multi-Scale Refine ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Try going to slightly different lengths
    ms_options = [n_current - 50, n_current + 50, n_current + 100, n_current - 100]
    ms_options = [n for n in ms_options if 200 <= n <= 1200 and n != n_current]

    for n_ms in ms_options[:3]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        alloc = min(remaining * 0.15, 60.0)
        if alloc > 15:
            a_ms = resample_sequence(best_seq, n_ms)
            a_ms = two_stage_lp_polish(a_ms, time_budget=alloc)
            obj_ms = compute_obj(a_ms)
            if obj_ms < best_score:
                best_score = obj_ms
                best_seq = a_ms.copy()
                n_current = n_ms
                print(f"  ** MS n={n_ms}: obj={best_score:.8f} **")
            else:
                print(f"  MS n={n_ms}: obj={obj_ms:.8f}")

    # === Phase 7: Diverse restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        mode = (r * 3 + seed) % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.4
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.2, 100.0)
        if alloc > 20:
            half = alloc / 2
            a_r = two_stage_lp_polish(a_r, time_budget=half)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_r = accelerated_lp(a_r, time_budget=min(remaining * 0.3, 40.0), momentum=0.6)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Final 2SLP
    if remaining > 40:
        alloc = min(remaining * 0.3, 120.0)
        a_ts = two_stage_lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_ts = compute_obj(a_ts)
        if obj_ts < best_score:
            best_score = obj_ts
            best_seq = a_ts.copy()
            print(f"  Final 2SLP: obj={best_score:.8f} **")

    # Final NSD with high tau
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 80.0)
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                          tau_init=50.0, tau_end=1000.0)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Final NSD: obj={best_score:.8f} **")

    # Final ALP
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.25, 60.0)
        a_alp = accelerated_lp(best_seq, time_budget=alloc, momentum=0.7)
        obj_alp = compute_obj(a_alp)
        if obj_alp < best_score:
            best_score = obj_alp
            best_seq = a_alp.copy()
            print(f"  Final ALP: obj={best_score:.8f} **")

    # Final PCD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 40.0)
        a_cd = peak_targeted_cd(best_seq, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final PCD: obj={best_score:.8f} **")

    # One more 2SLP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 40.0)
        a_ts = two_stage_lp_polish(best_seq, time_budget=alloc)
        obj_ts = compute_obj(a_ts)
        if obj_ts < best_score:
            best_score = obj_ts
            best_seq = a_ts.copy()
            print(f"  Extra 2SLP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
