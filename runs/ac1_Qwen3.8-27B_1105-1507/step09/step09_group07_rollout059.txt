        # Add constraints: ag[k] <= ag[k] - level_gap for top peaks
        # Actually, let's just try the basic LP again with slightly different approach
        break

    return g, M, ag


def lp_alternate_annealed(a, time_budget, verbose=False, seed=42):
    """
    LP alternating with simulated annealing, adaptive step sizes,
    and periodic perturbation for escaping local minima.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    current_a = a.copy()
    current_obj = best_obj
    start = time.monotonic()

    rng = np.random.default_rng(seed)
    iteration = 0
    stall = 0
    patience = 25
    temperature = 0.05
    initial_temp = 0.05
    last_improve_iter = 0
    last_good_t = 0.3

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp(current_a)
        if g_lp is None:
            break

        iteration += 1
        elapsed = time.monotonic() - start

        # Line search: try multiple step sizes
        candidate_ts = np.concatenate([
            np.linspace(0.02, 0.15, 6),
            np.linspace(0.15, 0.5, 6),
            np.linspace(0.5, 1.0, 5),
            [last_good_t * 0.5, last_good_t * 0.75, last_good_t * 1.25, last_good_t * 1.5],
        ])
        candidate_ts = np.clip(candidate_ts, 0.01, 1.0)
        candidate_ts = np.unique(candidate_ts)

        best_t_obj = np.inf
        best_t_val = last_good_t
        best_candidate = current_a.copy()

        for t in candidate_ts:
            a_try = (1.0 - t) * current_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_candidate = a_try.copy()

        # Simulated annealing acceptance
        delta = best_t_obj - current_obj
        if delta < -1e-12:
            # Improvement: always accept
            current_a = best_candidate.copy()
            current_obj = best_t_obj
            last_good_t = best_t_val
            stall = 0

            if best_t_obj < best_obj - 1e-12:
                best_obj = best_t_obj
                best_a = best_candidate.copy()
                last_improve_iter = iteration
                temperature = max(0.002, initial_temp * (0.97 ** (iteration - last_improve_iter)))
        else:
            # No improvement: accept with probability exp(-delta / temperature)
            if temperature > 0.001 and rng.random() < np.exp(-max(0, delta) / temperature):
                current_a = best_candidate.copy()
                current_obj = best_t_obj
                last_good_t = best_t_val
            else:
                stall += 1
                temperature = max(0.001, temperature * 0.95)

        # Stall handling: perturb if stuck
        if stall > patience:
            perturbation_scale = 0.02 + 0.03 * min(stall // patience, 3)
            noise = rng.normal(0, perturbation_scale, n)
            current_a = best_a * np.exp(noise)
            current_a = np.maximum(current_a, 0)
            s = float(np.sum(current_a))
            if s > 1e-10:
                current_a = current_a / s
            else:
                current_a = best_a.copy()
            current_obj = compute_obj(current_a)
            stall = 0
            temperature = initial_temp * 0.8
            last_good_t = 0.5
            if verbose:
                print(f"        Perturbed at iter {iteration}: obj={current_obj:.8f}")

        if verbose and iteration % 30 == 0:
            elapsed = time.monotonic() - start
            print(f"        LP-anneal it{iteration}: best={best_obj:.8f}, "
                  f"curr={current_obj:.8f}, t={last_good_t:.3f}, "
                  f"T={temperature:.4f}, stall={stall}, elapsed={elapsed:.1f}s")

    return best_a


def subgradient_optimize(a, time_budget, verbose=False):
    """
    Projected subgradient descent with Nesterov momentum.
    f(a) = max_k (a*a)[k], minimize subject to a>=0, sum(a)=1.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov momentum
    y = a.copy()
    momentum = 0.9
    lr = 0.1 / np.sqrt(n)

    it = 0
    while time.monotonic() - start < time_budget:
        # Compute convolution and find peak
        conv = compute_conv(y)
        k_star = int(np.argmax(conv))

        # Subgradient: (∇h_k)(i) = a[k-i] (with out-of-range = 0)
        # This is the reversed and shifted version of y
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = y[j]

        # Nesterov step
        momentum = 0.9 + 0.05 * (it % 100) / 100  # slowly increase
        y_new = y - lr * grad
        # Project onto simplex {x >= 0, sum(x) = 1}
        y_new = project_simplex(y_new)

        # Update
        y = y_new
        lr *= 0.999  # slow decay

        obj = compute_obj(y)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = y.copy()

        it += 1
        if verbose and it % 200 == 0:
            print(f"          Subgrad it{it}: obj={best_obj:.8f}, lr={lr:.6f}")

    return best_a


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def targeted_peak_reduction(a, time_budget, verbose=False):
    """
    Identify peak lags and do targeted mass transfers to reduce them.
    Uses the structure: (a*a)[k] = sum_i a[i]*a[k-i].
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    round_num = 0
    while time.monotonic() - start < time_budget:
        round_num += 1
        conv = compute_conv(a)
        peak_val = float(np.max(conv))
        peak_lags = np.argsort(conv)[::-1][:5]

        improved = False
        for k_star in peak_lags:
            if time.monotonic() - start > time_budget:
                break
            k_star = int(k_star)
            if conv[k_star] < peak_val * 0.9:
                continue

            # Find top contributors to this lag
            contributors = []
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    contributors.append((a[i] * a[j], i, j))
            contributors.sort(reverse=True)

            for contrib, i, j in contributors[:8]:
                if time.monotonic() - start > time_budget:
                    break
                if contrib < 1e-10:
                    continue

                src = i if a[i] >= a[j] else j
                # Find target: a position that minimizes impact on peaks
                # Try positions far from peak lags
                target_score = np.zeros(n)
                for k in peak_lags[:3]:
                    k = int(k)
                    for pos in range(n):
                        other = k - pos
                        if 0 <= other < n:
                            target_score[pos] += a[other] ** 2
                # Also penalize positions near src
                for d in range(-3, 4):
                    src_d = src + d
                    if 0 <= src_d < n:
                        target_score[src_d] += 0.1

                good_targets = np.argsort(target_score)[:10]
                for target in good_targets:
                    if target == src:
                        continue
                    for frac in [0.005, 0.01, 0.02, 0.03]:
                        delta = frac * a[src]
                        if delta <= 0 or delta >= 0.9 * a[src]:
                            continue
                        a[src] -= delta
                        a[target] += delta
                        new_conv = compute_conv(a)
                        new_obj = 2.0 * n * float(np.max(new_conv))
                        if new_obj < best_obj - 1e-13:
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[src] += delta
                            a[target] -= delta
                            break
                    if improved:
                        break

        if not improved and round_num > 20:
            break
        if not improved:
            # Try random perturbations
            idx = rng.integers(0, n)
            for target in rng.permutation(n)[:5]:
                if target == idx:
                    continue
                for frac in [0.01, 0.02, 0.05]:
                    delta = frac * a[idx]
                    if delta <= 0 or delta >= a[idx]:
                        continue
                    a[idx] -= delta
                    a[target] += delta
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[idx] += delta
                        a[target] -= delta
                        break
                if improved:
                    break
            if not improved and round_num > 50:
                break

        if verbose and round_num % 20 == 0:
            print(f"          Targeted r{round_num}: obj={best_obj:.8f}")

    return best_a


def coordinate_descent_refined(a, time_budget, verbose=False):
    """
    Refined coordinate descent: for each coordinate, find optimal value
    via golden section search.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)

    for rnd in range(100):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        # Process coordinates in random order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            if a[i] < 1e-8:
                continue

            # Golden section search for optimal a[i]
            # Fix all other coordinates, vary a[i]
            a_save = a[i]
            lo = 0.0
            hi = min(0.5, a[i] * 3)  # don't go too far

            # First, check if decreasing helps
            for frac in [0.3, 0.5, 0.7]:
                new_val = a[i] * (1 - frac)
                a[i] = new_val
                s_new = float(np.sum(a))
                if s_new < 1e-10:
                    a[i] = a_save
                    continue
                # Don't renormalize (we're searching over the raw value)
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-13:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                    break
                elif obj_new < 2.0:  # Acceptable, keep exploring
                    a[i] = new_val
                    break
                else:
                    a[i] = a_save
                    break
            else:
                # Try increasing
                for frac in [0.3, 0.5]:
                    new_val = a[i] * (1 + frac)
                    a[i] = new_val
                    obj_new = compute_obj(a)
                    if obj_new < best_obj - 1e-13:
                        best_obj = obj_new
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] = a_save
                        break

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.ones(new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode % 10 == 0:
        # Exponential decay (non-symmetric)
        a = np.exp(-np.linspace(0, 3, n))
    elif mode % 10 == 1:
        # Reverse exponential
        a = np.exp(np.linspace(0, 3, n))
    elif mode % 10 == 2:
        # Ramp up then flat
        x = np.linspace(0, 1, n)
        a = np.minimum(x * 3, 1.0)
    elif mode % 10 == 3:
        # Flat then ramp down
        x = np.linspace(0, 1, n)
        a = np.minimum((1 - x) * 3, 1.0)
    elif mode % 10 == 4:
        # Two bumps (asymmetric)
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.2) ** 2) + 0.5 * np.exp(-8 * (x - 0.7) ** 2)
    elif mode % 10 == 5:
        # Sine wave (non-negative part)
        x = np.linspace(0, 1, n)
        a = np.maximum(0, np.sin(np.pi * x * 1.5) + 0.3)
    elif mode % 10 == 6:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** 0.5
    elif mode % 10 == 7:
        # Random smooth
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
    elif mode % 10 == 8:
        # Cosine window (asymmetric)
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * x * 0.8)
    else:
        a = rng.dirichlet(np.ones(n) * 0.3)

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a

    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.3, 40.0)
    if alloc > 10:
        a_c = coordinate_descent_refined(best_seq, time_budget=alloc, verbose=False)
        obj_c = compute_obj(a_c)
        if obj_c < best_score:
            best_score = obj_c
            best_seq = a_c.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # One more quick LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = lp_alternate_annealed(best_seq, time_budget=min(remaining * 0.5, 30.0),
                                        verbose=False, seed=9999)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"=== Target: 1.5027, Achieved: {best_score:.8f}, Gap: {best_score - 1.5027:.8f} ===")

    return result
