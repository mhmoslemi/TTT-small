def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

        return np.inf
    s = float(np.sum(a))

        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)

    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses combined spectral flatness (IWRAP), two-stage LP with extrapolation,
multiplicative subgradient updates, and adaptive top-K peak averaging.
Key innovations: spectral initialization for global structure, LP with extrapolation
beyond the LP solution, and annealed multiplicative updates for fine-tuning.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        i_start = 0
        i_end = min(n, 2 * n - 1 - j)
        A[i_start:i_end, j] = a[:i_end]
    return A


def solve_lp_minpeak_full(a):
    """Solve LP: min M s.t. conv(a,g) <= M for all k, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_active(a, active_indices):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for k in active_indices, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    active_A = A[active_indices, :]
    constraints = [active_A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def golden_section_line_search(a, g, n_evals=8):
    """Find optimal t in [0, 1.5] that minimizes compute_obj((1-t)*a + t*g)."""
    phi = (np.sqrt(5) - 1) / 2
    lo = 0.0
    hi = 1.5

    for _ in range(n_evals):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)

        a1 = (1 - t1) * a + t1 * g
        a1 = np.maximum(a1, 0)
        s1 = float(np.sum(a1))
        if s1 > 1e-10:
            a1 = a1 / s1
            f1 = compute_obj(a1)
        else:
            f1 = np.inf

        a2 = (1 - t2) * a + t2 * g
        a2 = np.maximum(a2, 0)
        s2 = float(np.sum(a2))
        if s2 > 1e-10:
            a2 = a2 / s2
            f2 = compute_obj(a2)
        else:
            f2 = np.inf

        if f1 < f2:
            hi = t2
        else:
            lo = t1

    a_best = (1 - t_best) * a + t_best * g
    a_best = np.maximum(a_best, 0)
    s_best = float(np.sum(a_best))

    a = np.maximum(a, 0).astype(np.float64).copy()

        return a

    size = 1
    while size < 4 * n:
        size <<= 1

        # Time domain: enforce non-negativity and normalize
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            a = best_a.copy()
        else:
            a = a / s

        # Frequency domain: compress the magnitude toward flat
        A = fft(a, size)
        magnitude = np.abs(A)
        phase = A / (magnitude + 1e-30)

        # Power compression: alpha < 1 compresses dynamic range
        new_mag = magnitude ** alpha
        A_new = new_mag * phase

        # Back to time domain
        a = ifft(A_new).real[:n]

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a


def subgradient_topk(a, k_peaks):
    """Compute averaged subgradient over top-K peaks of a*a."""
    n = len(a)
    conv = compute_conv(a)
    peak_indices = np.argsort(conv)[-k_peaks:]

    s_grad = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            s_grad[j] += 2.0 * a[k - j]

    s_grad /= len(peak_indices)
    return s_grad, conv


def multiplicative_anneal(a, time_budget=120.0, verbose=False):
    """
    Annealed multiplicative subgradient updates.
    a_i <- a_i * exp(-eta * s_i) with decreasing eta (annealing).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(4321)

    eta_high = 0.15
    eta_low = 0.005
    it = 0
    stall = 0
    max_it = 500

    while time.monotonic() - start < time_budget and it < max_it:
        # Anneal eta
        progress = min(1.0, it / max_it)
        eta = eta_high * (eta_low / eta_high) ** progress

        # Adaptive top-K: fewer peaks early (more aggressive), more later (smoother)
        k_top = max(3, int(n * 0.02 * (0.5 + 0.5 * progress)))
        k_top = min(k_top, n)

        s_grad, conv = subgradient_topk(a, k_top)
        sg_max = float(np.max(s_grad))
        if sg_max > 1e-15:
            s_grad = s_grad / sg_max

        # Multiplicative update with floor
        a_new = a * np.exp(-eta * s_grad)
        floor = 1e-12
        a_new = np.maximum(a_new, floor)
        a_new = a_new / float(np.sum(a_new))

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = best_a.copy()
            stall += 1
            if stall > 20:
                # Perturb to escape
                noise_scale = max(0.01, 0.05 * (1.0 - progress))
                a = best_a * np.exp(rng.normal(0, noise_scale, n))
                a = np.maximum(a, floor)
                a = a / a.sum()
                stall = 0

        it += 1
        if verbose and it % 30 == 0:
            print(f"    MA it{it}: obj={best_obj:.8f}, eta={eta:.5f}, K={k_top}")

    return best_a


def lp_extrapolate(a, time_budget=300.0, verbose=False):
    """
    LP-based polishing with extrapolation.
    Solves min M s.t. (a*g)[k] <= M, then blends or extrapolates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(7777)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        g, M_lp = solve_lp_minpeak_full(a)
        if g is None:
            break

        # Find active set and solve second LP
        A_mat = build_conv_matrix(a)
        conv_ag = A_mat @ g
        max_conv_ag = float(np.max(conv_ag))
        tol = 0.03 * max(max_conv_ag, 1e-10)
        active = np.where(conv_ag >= max_conv_ag - tol)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv_ag))])
        if len(active) < 2 * n - 1:
            g2 = solve_lp_active(a, active)
            if g2 is not None:
                g = (g + g2) / 2.0

        # Try multiple step sizes including extrapolation
        best_t = 0.0
        best_f = best_obj
        candidates = [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 1.2, 1.5, 0.5]
        for t_try in candidates:
            a_try = (1.0 - t_try) * a + t_try * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                f_try = compute_obj(a_try)
                if f_try < best_f:
                    best_f = f_try
                    best_t = t_try

        if best_f < best_obj - 1e-14:
            best_obj = best_f
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            a = best_a.copy()
            if stall > 20:
                scale = min(0.12, 0.02 + 0.004 * (stall // 10))
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = float(np.sum(a))
                if s_a > 1e-10:
                    a = a / s_a
                stall = 0
            if stall > 80:
                break

        it += 1
        if verbose and it % 5 == 0:
            print(f"    LPE it{it}: obj={best_obj:.8f}, M={M_lp:.6f}, |S|={len(active)}")

    return best_a


def two_stage_lp(a, time_budget=300.0, verbose=False):
    """Two-stage LP: min-max then min over active set, with golden section line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(8888)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        g1, M1 = solve_lp_minpeak_full(a)
        if g1 is None:
            break

        # Active set
        A_mat = build_conv_matrix(a)
        conv_ag = A_mat @ g1
        max_val = float(np.max(conv_ag))
        tol = 0.05 * max(max_val, 1e-10)
        active = np.where(conv_ag >= max_val - tol)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv_ag))])

        g2 = solve_lp_active(a, active)
        if g2 is not None:
            g_blend = (g1 + g2) / 2.0
        else:
            g_blend = g1

        a_new = golden_section_line_search(a, g_blend, n_evals=7)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            a = best_a.copy()
            if stall > 25:
                scale = min(0.1, 0.02 + 0.005 * (stall // 10))
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = float(np.sum(a))
                if s_a > 1e-10:
                    a = a / s_a
                stall = 0
            if stall > 80:
                break

        it += 1
        if verbose and it % 3 == 0:
            print(f"    2SLP it{it}: obj={best_obj:.8f}, M1={M1:.6f}, |S|={len(active)}")

    return best_a


def frank_wolfe_peaks(a, time_budget=80.0, verbose=False):
    """
    Frank-Wolfe with top-K peak subgradient.
    LW step: move mass to the coordinate with minimum subgradient.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(5555)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget and it < 3000:
        k_top = max(5, n // 20)
        s_grad, conv = subgradient_topk(a, k_top)

        # FW step: g = e_{argmin(s_grad)}
        min_idx = int(np.argmin(s_grad))
        g = np.zeros(n)
        g[min_idx] = 1.0

        # Step size: 2/(it+3) with backtracking
        t_fw = 2.0 / (it + 3.0)
        best_t = 0.0
        best_f = best_obj
        for t_try in [t_fw, t_fw * 0.5, min(1.0, t_fw * 2.0), 0.1]:
            a_try = (1.0 - t_try) * a + t_try * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                f_try = compute_obj(a_try)
                if f_try < best_f:
                    best_f = f_try
                    best_t = t_try

        if best_f < best_obj - 1e-14:
            best_obj = best_f
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / float(np.sum(a_new))
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 100:
                # Multiple random FW steps
                for _ in range(10):
                    idx = int(rng.integers(0, n))
                    g_r = np.zeros(n)
                    g_r[idx] = 1.0
                    a_try = 0.95 * a + 0.05 * g_r
                    a_try = np.maximum(a_try, 0)
                    a_try = a_try / a_try.sum()
                    f_try = compute_obj(a_try)
                    if f_try < best_obj:
                        best_obj = f_try
                        best_a = a_try.copy()
                        a = a_try
                        break
                else:
                    break
            it = 0

        it += 1
        if verbose and it % 100 == 0:
            print(f"    FW it{it}: obj={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    """
    Multi-phase optimization:
    1. Spectral flatness (global structure)
    2. Two-stage LP (direction finding)
    3. LP with extrapolation (aggressive improvement)
    4. Annealed multiplicative updates (fine-tuning)
    5. Multi-n exploration
    6. Frank-Wolfe coordinate moves
    7. Final joint polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    n_target = 800

            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent available: {e}")

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target

        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    print(f"\n=== Phase 1: Spectral Flatness ===")
    remaining = deadline - time.monotonic()
