"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses subgradient descent with Nesterov acceleration, LP-based direction finding,
log-sum-exp smoothing, and structured initializations (monotone, symmetric).
Key improvements: multi-active-set subgradient, adaptive restarts, and
monotone-sequence insight for better initializations.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_at(a, k, n):
    """
    Compute subgradient of conv(a,a)[k] with respect to a.
    Result: grad[j] = 2*a[k-j] if 0 <= k-j < n, else 0.
    """
    grad = np.zeros(n)
    if 0 <= k < n:
        # k-j in [0, n-1] means j in [k-(n-1), k], intersect [0, n-1]
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        grad[j_min:j_max+1] = 2.0 * a[k - j_max:k - j_min + 1][::-1]
    elif k >= n:
        # k-j in [0, n-1] means j in [k-(n-1), k], intersect [0, n-1]
        j_min = k - (n - 1)
        j_max = n - 1
        if j_min <= j_max:
            grad[j_min:j_max+1] = 2.0 * a[k - j_max:k - j_min + 1][::-1]
    return grad

def max_subgradient(a, n, top_k=5):
    """
    Compute a subgradient of max_k conv(a,a)[k].
    Uses a convex combination of gradients at the top-k active lags.
    """
    conv = compute_conv(a)
    # Find top-k lags
    sorted_idx = np.argsort(conv)[::-1][:top_k]
    max_val = conv[sorted_idx[0]]
    # Weights: softmax over top-k
    weights = np.exp(10 * (conv[sorted_idx] - max_val))
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, sorted_idx):
        grad += w * subgradient_at(a, int(k), n)
    return grad, conv

def nesterov_subgradient_descent(a, time_budget=200.0, top_k=5, verbose=False):
    """
    Nesterov accelerated subgradient descent on the simplex.
    Minimizes max(conv(p,p)) subject to p >= 0, sum(p) = 1.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 1.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    patience = 5000

    while time.monotonic() - start < time_budget:
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            v = np.zeros(n)
        else:
            y = y / s_y

        # Subgradient
        grad, conv = max_subgradient(y, n, top_k)
        
        # Gradient step
        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        
        # Adaptive learning rate
        if it % 100 == 0:
            lr *= 0.995
        
        # Restart condition
        if it - best_it > patience:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 1.0 / np.sqrt(n) * 0.5
            best_it = it
            patience = max(1000, patience // 2)

        if verbose and it % 5000 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, lr={lr:.6f}")

    return best_a

def lut_descent(a, time_budget=200.0, verbose=False):
    """
    Log-sum-exp smoothed descent. Uses softmax weighting over all lags
    for smoother gradients.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    patience = 10000

    # Temperature schedule
    tau_start = 5.0
    tau_end = 200.0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.95))
        tau = tau_start * (tau_end / tau_start) ** progress

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            v = np.zeros(n)
        else:
            y = y / s_y

        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        # Softmax weights over lags
        weights = np.exp(tau * (conv - max_conv))
        weights = weights / float(np.sum(weights))

        # Gradient: sum_k weights[k] * grad_k
        # grad_k[j] = 2*y[k-j] if 0 <= k-j < n
        # This can be computed as 2 * (weights * y reversed/shifted)
        # Specifically: grad[j] = 2 * sum_k weights[k] * y[k-j] (if valid)
        # = 2 * (correlation of weights and y_shifted)[j]
        
        # Use FFT for efficiency
        # grad[j] = 2 * sum_{k: 0<=k-j<n} weights[k]*y[k-j]
        # Let m = k-j, then k = j+m, so:
        # grad[j] = 2 * sum_{m: 0<=m<n, 0<=j+m<2n-1} weights[j+m]*y[m]
        # This is a cross-correlation!
        
        N = 1
        while N < 3 * n:
            N <<= 1
        W = fft(weights, N)
        Y = fft(y, N)
        # cross-correlation: ifft(conj(W) * Y) gives sum_k W[k+m]*Y[m]... 
        # Actually: ifft(conj(A) * B)[n] = sum_k A[k]*B[k+n] (circular)
        # We want: sum_m weights[j+m]*y[m] for each j
        # = (cross-corr of weights and y)[j]
        # cross-corr[x,y][j] = sum_m x[m+j]*y[m] = ifft(conj(X) * Y)[j] (with proper padding)
        
        grad_full = ifft(np.conj(W) * Y).real
        grad = 2.0 * grad_full[:n]
        
        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9998)

        if it - best_it > patience:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it
            patience = max(2000, patience // 2)

        if verbose and it % 5000 == 0:
            print(f"    LUT it{it}: obj={best_obj:.8f}, tau={tau:.1f}")

    return best_a

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_direction(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns the optimal g direction.
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None
    
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None

def ipm_polish(a, time_budget=100.0, verbose=False):
    """
    Interior-point method polish using LP direction finding.
    Iteratively: solve LP for direction, line search, update.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stall = 0
    momentum = 0.5
    v = np.zeros(n)

    while time.monotonic() - start < time_budget:
        g_lp = solve_lp_direction(a)
        if g_lp is None:
            break

        # Momentum update
        v = momentum * v + (g_lp - a)
        a_trial = project_simplex(a + v)
        obj_trial = compute_obj(a_trial)

        if obj_trial < best_obj - 1e-12:
            best_obj = obj_trial
            best_a = a_trial.copy()
            a = a_trial
            stall = 0
        else:
            # Try smaller steps
            found = False
            for scale in [0.5, 0.25, 0.1, 0.05]:
                a_try = project_simplex(a + scale * v)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    v *= scale
                    stall = 0
                    found = True
                    break
            if not found:
                stall += 1
                momentum = max(0.1, momentum * 0.8)
                v *= 0.3
                a = best_a.copy()
                if stall > 30:
                    # Perturb and restart
                    a = best_a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 0)
                    s_a = float(np.sum(a))
                    if s_a > 1e-10:
                        a = a / s_a
                    v = np.zeros(n)
                    momentum = 0.5
                    stall = 0
                    if stall > 60:  # Won't reach this, but safe
                        break
                    # Actually let's use a counter
                    if stall > 40:
                        break

        # Gradually increase momentum
        momentum = min(0.8, 0.5 + 0.0005 * (best_obj * 1000))

        if verbose:
            remaining = time_budget - (time.monotonic() - start)
            if remaining % 10 < 1:

def monotone_init(n, rng, mode=0):
    """Generate a monotone (increasing) initialization."""
    if mode == 0:
        # Linearly increasing
        a = np.arange(1, n + 1, dtype=np.float64)
    elif mode == 1:
        # Exponentially increasing
        a = np.exp(np.linspace(0, 1, n))
    elif mode == 2:
        # Quadratically increasing
        x = np.linspace(0, 1, n)
        a = x ** 2
    elif mode == 3:
        # Two-level: first half low, second half high
        a = np.concatenate([np.full(n//2, 0.3), np.full(n - n//2, 1.0)])
    elif mode == 4:
        # Smooth S-curve (sigmoid)
        x = np.linspace(-3, 3, n)
        a = 1.0 / (1.0 + np.exp(-x))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def symmetric_init(n, rng, mode=0):
    """Generate a symmetric initialization."""
    if mode == 0:
        # Inverted parabola (peak in middle)
        x = np.linspace(-1, 1, n)
        a = 1 - x**2
    elif mode == 1:
        # Triangle (peak in middle)
        x = np.linspace(-1, 1, n)
        a = np.maximum(0, 1 - np.abs(x))
    elif mode == 2:
        # U-shaped (valley in middle)
        x = np.linspace(-1, 1, n)
        a = x**2 + 0.1
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Monotone
        a = monotone_init(n, rng, mode=rng.integers(0, 5))
    elif mode == 3:
        # Symmetric
        a = symmetric_init(n, rng, mode=rng.integers(0, 4))
    elif mode == 4:
        # Two Gaussian bumps
        c1, c2 = rng.integers(0, n//3), rng.integers(2*n//3, n)
        w = max(1, n // 10)
        x = np.arange(n)
        a = np.exp(-((x-c1)**2)/(2*w**2)) + np.exp(-((x-c2)**2)/(2*w**2))
        a = a / a.sum()
    elif mode == 5:
        # Periodic modulation
        x = np.linspace(0, 1, n)
        f = rng.integers(2, 8)
        a = 1 + 0.5 * np.sin(2*np.pi*f*x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        # Random walk
        a = np.cumsum(rng.normal(0, 1, n))
        a = np.maximum(a - a.min(), 0) + 0.01
        a = a / a.sum()
    return a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def golden_section_line_search(a, g, n_evals=6):
    """Find optimal t in [0,1] minimizing compute_obj((1-t)*a + t*g)."""
    phi = (np.sqrt(5) - 1) / 2
    lo = 0.0
    hi = 1.0

    for _ in range(n_evals):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)

        a1 = (1 - t1) * a + t1 * g
        a1 = np.maximum(a1, 0)
        s1 = float(np.sum(a1))
        f1 = compute_obj(a1 / s1) if s1 > 1e-10 else np.inf

        a2 = (1 - t2) * a + t2 * g
        a2 = np.maximum(a2, 0)
        s2 = float(np.sum(a2))
        f2 = compute_obj(a2 / s2) if s2 > 1e-10 else np.inf

        if f1 < f2:
            hi = t2
        else:
            lo = t1

    t_best = (lo + hi) / 2
    a_best = (1 - t_best) * a + t_best * g
    a_best = np.maximum(a_best, 0)
    s_best = float(np.sum(a_best))
    if s_best > 1e-10:
        a_best = a_best / s_best
    return a_best

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based direction finding with momentum, Nesterov subgradient descent,
peak-shaving coordinate descent, log-sum-exp smoothing, and multi-length exploration.
Key innovations: structured monotone initializations, adaptive active-set LP,
and interleaved peak-shaving between LP iterations for finer local optimization.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    x_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(x_sorted)
    rho = np.arange(1, n + 1)
    cond = x_sorted * rho > (cssv - 1.0)
    rho_max = np.where(cond)[0]
    if len(rho_max) == 0:
        return np.ones(n) / n
    rho_val = rho_max[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_at(a, k, n):
    """Subgradient of conv(a,a)[k] w.r.t. a: grad[j] = 2*a[k-j] if valid."""
    grad = np.zeros(n)
    j_min = max(0, k - (n - 1))
    j_max = min(n - 1, k)
    if j_min <= j_max:
        indices = np.arange(j_min, j_max + 1)
        grad[indices] = 2.0 * a[k - indices]
    return grad


def nesterov_subgradient_descent(a, time_budget=200.0, top_k=8, verbose=False):
    """Nesterov accelerated subgradient descent on simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.8
    lr = 1.5 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    patience = 8000

    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            v = np.zeros(n)
        else:
            y = y / s_y

        conv = compute_conv(y)
        sorted_idx = np.argsort(conv)[::-1][:top_k]
        max_val = conv[sorted_idx[0]]
        weights = np.exp(5.0 * (conv[sorted_idx] - max_val))
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, sorted_idx):
            grad += w * subgradient_at(y, int(k), n)

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        if it % 500 == 0:
            lr *= 0.99

        if it - best_it > patience:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 1.5 / np.sqrt(n) * 0.5
            best_it = it
            patience = max(2000, patience // 2)

        if verbose and it % 5000 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, lr={lr:.6f}")

    return best_a

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a, max_time=30.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    try:
        import cvxpy as cp
    except ImportError:
        return None

    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        t0 = time.monotonic()
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
        if time.monotonic() - t0 > max_time:
            break
    return None


def two_stage_lp_direction(a):
    """Two-stage LP: min-max then min-sum-of-active."""
    try:
        import cvxpy as cp
    except ImportError:
        return None

    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    # Stage 1: min M
    g1 = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g1 <= M, cp.sum(g1) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    g1_val = None
    M_val = None
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g1.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g1_val = np.maximum(g1.value, 0)
                s = float(np.sum(g1_val))
                if s > 1e-10:
                    g1_val = g1_val / s
                    M_val = float(M.value)
                    break
        except Exception:
            continue

    if g1_val is None:
        return None

    # Find active constraints
    conv_a_g1 = A @ g1_val
    tol = 0.02 * max(float(np.max(conv_a_g1)), 1e-10)
    active = np.where(conv_a_g1 >= float(np.max(conv_a_g1)) - tol)[0]

    if len(active) == 0:
        return g1_val

    # Stage 2: min sum of active conv values
    g2 = cp.Variable(n, nonneg=True)
    active_A = A[active, :]
    obj = cp.sum(active_A @ g2)
    constraints2 = [cp.sum(g2) == 1]
    prob2 = cp.Problem(cp.Minimize(obj), constraints2)

    g2_val = None
    for solver in solvers:
        try:
            prob2.solve(solver=solver)
            if g2.value is not None and prob2.status in ('optimal', 'optimal_inaccurate'):
                g2_val = np.maximum(g2.value, 0)
                s = float(np.sum(g2_val))
                if s > 1e-10:
                    g2_val = g2_val / s
                    break
        except Exception:
            continue

    if g2_val is not None:
        # Compare and blend
        peak1 = float(np.max(A @ g1_val))
        peak2 = float(np.max(A @ g2_val))
        if peak2 < peak1:
            return g2_val
        else:
            return (g1_val + g2_val) / 2

    return g1_val


def ipm_polish(a, time_budget=200.0, use_two_stage=False, verbose=False):
    """LP-based direction finding with momentum and adaptive step."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(54321)

    v = np.zeros(n)
    momentum = 0.5
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        if use_two_stage:
            g_lp = two_stage_lp_direction(a)
        else:
            g_lp = solve_lp_direction(a)

        if g_lp is None:
            break

        v = momentum * v + (g_lp - a)
        a_trial = project_simplex(a + v)
        obj_trial = compute_obj(a_trial)

        if obj_trial < best_obj - 1e-12:
            best_obj = obj_trial
            best_a = a_trial.copy()
            a = a_trial
            stall = 0
        else:
            found = False
            for scale in [0.5, 0.25, 0.1, 0.05, 0.02]:
                a_try = project_simplex(a + scale * v)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    v *= scale
                    stall = 0
                    found = True
                    break
            if not found:
                stall += 1
                momentum = max(0.2, momentum * 0.85)
                v *= 0.3
                a = best_a.copy()
                if stall > 25:
                    a = best_a * np.exp(rng.normal(0, 0.04, n))
                    a = np.maximum(a, 0)
                    s_a = float(np.sum(a))
                    if s_a > 1e-10:
                        a = a / s_a
                    v = np.zeros(n)
                    momentum = min(0.8, 0.5 + 0.02 * stall)
                    stall = 0
                    if it > 200:
                        break
                if stall > 60:
                    break

        momentum = min(0.85, 0.5 + 0.001 * it)
        it += 1

        if verbose and it % 5 == 0:
            rem = time_budget - (time.monotonic() - start)
            print(f"    IPM it{it}: obj={best_obj:.8f}, stall={stall}, mom={momentum:.3f}, rem={rem:.0f}s")

    return best_a


def peak_shaving(a, time_budget=60.0, verbose=False):
    """Coordinate descent targeting the dominant peak."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(8888)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        j_min = max(0, k_star - (n - 1))
        j_max = min(n - 1, k_star)
        if j_min > j_max:
            break

        contribs = []
        for j in range(j_min, j_max + 1):
            m = k_star - j
            if 0 <= m < n:
                contribs.append((a[j] * a[m], j, m))
        contribs.sort(reverse=True)

        improved = False
        for c, j, m in contribs[:5]:
            if j == m:
                continue
            src = j if a[j] > a[m] else m
            dst = m if src == j else j

            targets = list(set([
                (src + n // 3) % n,
                (src + 2 * n // 3) % n,
                (n - 1 - src) % n,
                int(rng.integers(0, n)),
                int(rng.integers(0, n)),
                dst
            ]))

            for tgt in targets:
                tgt = int(tgt) % n
                if tgt == src:
                    continue
                for frac in [0.08, 0.04, 0.02, 0.01, 0.005]:
                    delta = frac * a[src]
                    if delta >= a[src] * 0.9:
                        continue
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[tgt] += delta
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        stall += 0 if improved else 1
        if stall > 80:
            break

        if verbose:
            remaining = time_budget - (time.monotonic() - start)
            if int(remaining) % 10 == 0:
                print(f"    PS: obj={best_obj:.8f}, stall={stall}")

    return best_a


def monotone_init(n, mode=0):
    """Generate a monotone (increasing) initialization."""
    if mode == 0:
        a = np.arange(1, n + 1, dtype=np.float64)
    elif mode == 1:
        a = np.exp(np.linspace(0, 0.8, n))
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = x ** 2 + 0.1
    elif mode == 3:
        a = np.concatenate([np.full(n // 2, 0.4), np.full(n - n // 2, 1.0)])
    elif mode == 4:
        x = np.linspace(-3, 3, n)
        a = 1.0 / (1.0 + np.exp(-1.5 * x))
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = x ** 0.5 + 0.1
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
