def solve_lp_with_top_k(a, K=100):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for top-K positions, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    conv_a = compute_conv(a_norm)
    
    # Select top-K positions
    top_k = np.argsort(conv_a)[::-1][:K]
    
    # Build constraint matrix for only these positions
    A = np.zeros((K, n))
    for idx, k in enumerate(top_k):
        k = int(k)
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[idx, j] = a_norm[i]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    # Try solvers
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None:
                g_val = np.maximum(g.value, 0)
                g_val = g_val / g_val.sum()
                return g_val, float(M.value)
        except:
            continue
    return None, None

Actually, I realize there might be an even better approach. Instead of the LP, we can directly optimize the blended objective using a different method.

**Idea: Direct optimization of the blend**

Given current a, we want to find g (on the simplex) that minimizes:
max_k ((1-t)*a + t*g)^2 * 2n

where the square is elementwise (i.e., we're computing the convolution of the blend with itself).

This is equivalent to:
max_k [(1-t)^2 (a*a)[k] + 2t(1-t) (a*g)[k] + t^2 (g*g)[k]]

For a fixed t, this is:
max_k [c_k + 2t(1-t) (a*g)[k] + t^2 (g*g)[k]]

where c_k = (1-t)^2 (a*a)[k].

This is still non-convex in g (because of the (g*g) term). But for small t, the t^2 term is negligible, and we can approximate:
max_k [c_k + 2t(1-t) (a*g)[k]]

which is linear in g! So for small t, the problem becomes:
min_g max_k [c_k + 2t(1-t) (a*g)[k]] s.t. sum(g)=1, g>=0

This is a LP! And it's the same as the LP we were solving before (with M absorbing the c_k term).

So the LP approach is already the right first-order approximation. The issue is that for larger t, the second-order term matters.

To handle this, we can use a "trust region" approach:
1. Solve the LP to get g_lp (the first-order optimal direction)
2. Use a line search to find the best t along the ray a -> a + t*(g_lp - a)
3. At each line search point, compute the full objective (including second-order terms)
4. Move to the best point

This is exactly what the golden section search does in the previous code! So the approach is correct.

The question is: can we make it faster or more effective?

**Key improvement: Adaptive K for the LP**
Start with a small K (e.g., K=20) for fast LP solves, and gradually increase K as we converge. This gives fast initial progress and precise final convergence.

**Another improvement: Multiple LP directions**
Solve the LP with different K values (or different subsets of constraints) and try blending with each direction. Pick the best one.

**Yet another: Dual information**
The LP has dual variables that tell us which constraints are binding. We can use this to:
- Focus the next LP on the binding constraints
- Detect convergence (when all dual variables are small)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based local search with adaptive constraint selection, structured
initializations, and multi-length exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_rows(a_norm, indices):
    """Build rows of convolution matrix for specified output positions.
    Returns matrix A such that A @ g = [conv(a,g)[k] for k in indices]."""
    n = len(a_norm)
    K = len(indices)
    A = np.zeros((K, n))
    for idx, k in enumerate(indices):
        k = int(k)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # A[idx, j] = a_norm[k - j] for j in [j_start, j_end]
            A[idx, j_start:j_end+1] = a_norm[k - j_end:k - j_start+1][::-1]
    return A


def solve_lp_topk(a, K=100, solver_list=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for top-K k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    conv_a = compute_conv(a_norm)
    
    # Select top-K positions
    if K < len(conv_a):
        top_k = np.argsort(conv_a)[::-1][:K]
    else:
        top_k = np.arange(len(conv_a))
    
    A = build_conv_matrix_rows(a_norm, top_k)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue
    
    return None, None


def golden_section_search(f, a_lo, a_hi, max_iter=30):
    """Golden section search for minimum of f on [a_lo, a_hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if abs(a_hi - a_lo) < 1e-4:
            break
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = f(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = f(d)
    return (a_lo + a_hi) / 2.0
