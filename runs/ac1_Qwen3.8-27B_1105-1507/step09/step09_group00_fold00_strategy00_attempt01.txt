def compute_conv(a):
    """Compute the full linear convolution a * a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2*n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]
