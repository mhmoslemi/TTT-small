import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def convolve_with_fixed(a_fixed, g):
    """Compute conv(a_fixed, g) efficiently. a_fixed is fixed, g varies."""
    n = len(a_fixed)

        size <<= 1
    fa = rfft(a_fixed, size)
    fg = rfft(g, size)
    return irfft(fa * fg, size)[:2 * n - 1]

        if obj_after < best_obj - 1e-12:
            best_obj = obj_after
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 8:
                if rng is not None:
                    scale = 0.01 + 0.005 * min(stagnant, 20)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    conv = compute_conv(a)
                    best_obj = float(np.max(conv))
                    stagnant = 0
                else:
                    break

        if verbose and sweep_count % 50 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    CD sweep {sweep_count}: obj={cur_global:.8f}, elapsed={elapsed:.0f}s")

        # Check time more finely
        if time.monotonic() - start_time > time_budget * 0.98:
            break

    best_a = np.maximum(best_a, 0)
    best_a = best_a / best_a.sum()
    return best_a


def build_conv_matrix_columns(a_fixed):
    """For LP: return the matrix such that M @ g = conv(a_fixed, g).
    We build it implicitly for speed. Returns a_fixed for use with convolve_with_fixed."""
    return a_fixed


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. conv(a, g)[k] <= M for all k, sum(g) = 1, g >= 0.
    Returns optimal g and M.
    """
    n = len(a)
    a_norm = a / np.sum(a)

    # Build constraint matrix A: (2n-1) x n where A[k,j] = a[k-j] if valid
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        col_start = j
        col_end = min(2 * n - 1, j + n)
        A[col_start:col_end, j] = a_norm[:col_end - col_start]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_with_linesearch(a, time_budget, start_time, rng, verbose=False):
    """LP-based direction finding with proper 1D line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()
    conv = compute_conv(a)
    best_a = a.copy()
    best_obj = float(np.max(conv))
    start = time.monotonic()
    iter_count = 0
    stagnant = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break

        # Line search on t in [0, 1]: minimize f(t) = max_k ((1-t)*a + t*g) * ((1-t)*a + t*g)[k]
        # f(t) = max_k [(1-t)^2*conv_a[k] + 2t(1-t)*conv_ag[k] + t^2*conv_g[k]]
        conv_ag = convolve_with_fixed(a, g_lp)
        conv_g = compute_conv(g_lp)

        lo_t = 0.0
        hi_t = 1.0
        for _ in range(40):
            mid_t = 0.5 * (lo_t + hi_t)
            a_t = (1 - mid_t) * a + mid_t * g_lp
            # Derivative at mid_t for the active k
            # d/dt (a_t*a_t)[k] = 2 * conv(a_t, g-a)[k]
            diff = g_lp - a
            conv_adiff = convolve_with_fixed(a_t, diff)
            val_t = (1 - mid_t) ** 2 * conv + 2 * mid_t * (1 - mid_t) * conv_ag + mid_t ** 2 * conv_g
            max_val = float(np.max(val_t))
            active = val_t >= max_val - 1e-12
            grad = 2.0 * conv_adiff[active]
            min_grad = float(np.min(grad))
            max_grad = float(np.max(grad))

            if min_grad > 1e-12:
                hi_t = mid_t
            elif max_grad < -1e-12:
                lo_t = mid_t
            else:
                break

        t_opt = 0.5 * (lo_t + hi_t)
        t_opt = max(0.0, min(1.0, t_opt))

        a_new = (1 - t_opt) * a + t_opt * g_lp
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            conv = compute_conv(a)
            stagnant = 0
            iter_count += 1
            if verbose and iter_count % 5 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={t_opt:.4f}, elapsed={time.monotonic()-start:.0f}s")
        else:
            stagnant += 1
            if stagnant > 6:
                scale = 0.02 + 0.01 * min(stagnant, 15)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv = compute_conv(a)
                stagnant = 0
            elif stagnant > 12:
                break

    return best_a


def mirror_descent(a, time_budget, start_time, rng, verbose=False):
    """
    Mirror descent (exponentiated gradient) with backtracking line search.
    Minimizes max_k (a*a)[k] over the simplex using KL-proximal updates.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = float(np.max(compute_conv(a)))
    start = time.monotonic()
    eta = 1.0 / np.sqrt(n)
    iter_count = 0
    stagnant = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        # Find active set
        active = conv >= max_val - 1e-10
        active_idx = np.where(active)[0]

        # Subgradient: average of gradients at active positions
        # grad[j] = 2 * sum_{k in active} (1/|active|) * a[k-j] (if 0 <= k-j < n)
        grad = np.zeros(n)
        for k in active_idx:
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * a[idx]
        grad /= len(active_idx)

        # Mirror descent update: a_new = a * exp(-eta * grad) / Z
        log_update = -eta * grad
        # For numerical stability
        log_update -= np.max(log_update)
        weights = np.exp(log_update)
        Z = float(np.sum(a * weights))
        if Z < 1e-30:
            Z = 1.0
        a_new = a * weights / Z
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj_new = float(np.max(compute_conv(a_new)))

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Increase eta slightly
            eta = min(eta * 1.05, 10.0)
            stagnant = 0
        else:
            # Backtrack: reduce eta
            eta = max(eta * 0.7, 0.01)
            stagnant += 1
            if stagnant > 50:
                # Perturb
                scale = 0.03
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                eta = 1.0 / np.sqrt(n)
                stagnant = 0
            if stagnant > 200:
                break

        iter_count += 1
        if verbose and iter_count % 200 == 0:
            print(f"    MD it{iter_count}: obj={2*n*best_obj:.8f}, eta={eta:.4f}, elapsed={time.monotonic()-start:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 7:
        c1 = rng.integers(0, n // 3)
        c2 = rng.integers(2 * n // 3, n)
        w = max(1, n // 12)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 8:
        x = np.linspace(0, np.pi, n)
        a = np.sin(x)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 9:
        # Open cosine window (Hamming)
        x = np.arange(n)
        a = 0.54 + 0.46 * np.cos(2 * np.pi * x / (n - 1))
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 10:
        # Exponential decay from center
        center = n // 2
        a = np.exp(-np.abs(np.arange(n) - center) / (n * 0.3))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Multi-phase optimization combining CD, LP, and mirror descent.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Coordinate Descent (300s) ===
    print(f"\n=== Phase 1: CD with subgradient bisection ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.22, 300.0)
    if p1_budget > 30:
        a_cd = cd_subgradient_bisection(best_seq, None, time_budget=p1_budget,
                                        start_time=time.monotonic(), rng=rng, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase1 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 CD: obj={obj_cd:.8f}")

    # === Phase 2: LP with line search (400s) ===
    print(f"\n=== Phase 2: LP + line search ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.28, 400.0)
    if p2_budget > 30:
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = p2_budget / (3 - rnd) * 0.8
            alloc = min(alloc, remaining * 0.7)
            if alloc < 20:
                break
            a_lp = lp_with_linesearch(best_seq, time_budget=alloc,
                                      start_time=time.monotonic(), rng=rng, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
                if rnd < 2:
                    scale = 0.02 * (rnd + 1)
                    best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                    best_seq = np.maximum(best_seq, 0)
                    best_seq = best_seq / best_seq.sum()

    # === Phase 3: Mirror Descent (250s) ===
    print(f"\n=== Phase 3: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.18, 250.0)
    if p3_budget > 30:
        a_md = mirror_descent(best_seq, time_budget=p3_budget,
                              start_time=time.monotonic(), rng=rng, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Phase3 MD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 MD: obj={obj_md:.8f}")

    # === Phase 4: Fine CD (200s) ===
    print(f"\n=== Phase 4: Fine CD ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 200.0)
    if p4_budget > 20:
        a_cd2 = cd_subgradient_bisection(best_seq, None, time_budget=p4_budget,
                                         start_time=time.monotonic(), rng=rng, verbose=False)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  Phase4 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 CD: obj={obj_cd2:.8f}")

    # === Phase 5: Multi-n exploration (200s) ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 100), n_current + 100, n_current + 200,
        600, 700, 900, 1000, 1100, 1200, 1300, 1500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500 and abs(n - n_current) > 50]
    n_options = n_options[:5]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        alloc1 = min(remaining * 0.4, 60.0)
        if alloc1 > 20:
            a_try = cd_subgradient_bisection(a_try, None, time_budget=alloc1,
                                             start_time=time.monotonic(), rng=rng)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.2, 40.0)
        if alloc2 > 15:
            a_try = lp_with_linesearch(a_try, time_budget=alloc2,
                                       start_time=time.monotonic(), rng=rng)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts (150s) ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    restart_modes = [3, 8, 9, 10, 5]
    for r, mode in enumerate(restart_modes[:4]):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.4 + 0.05 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc1 = min(remaining * 0.4, 50.0)
        if alloc1 > 15:
            a_r = cd_subgradient_bisection(a_r, None, time_budget=alloc1,
                                           start_time=time.monotonic(), rng=rng)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.2, 30.0)
        if alloc2 > 10:
            a_r = lp_with_linesearch(a_r, time_budget=alloc2,
                                     start_time=time.monotonic(), rng=rng)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final polish (remaining) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 40:
        alloc = min(remaining * 0.3, 80.0)
        a_f = cd_subgradient_bisection(best_seq, None, time_budget=alloc,
                                       start_time=time.monotonic(), rng=rng)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 60.0)
        a_f = lp_with_linesearch(best_seq, time_budget=alloc,
                                 start_time=time.monotonic(), rng=rng)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.4, 60.0)
        a_f = mirror_descent(best_seq, time_budget=alloc,
                             start_time=time.monotonic(), rng=rng)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # One more CD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 40.0)
        a_f = cd_subgradient_bisection(best_seq, None, time_budget=alloc,
                                       start_time=time.monotonic(), rng=rng)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0)
    best_seq = best_seq / best_seq.sum()

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
