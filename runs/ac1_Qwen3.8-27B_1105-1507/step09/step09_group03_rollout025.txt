    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based smoothed gradient descent with Nesterov acceleration,
adaptive tau scheduling, and multi-phase optimization.
Key innovations: efficient O(n log n) per-iteration cost via FFT-based
cross-correlation for gradient computation, softmax parameterization,
and active-set-aware peak reduction.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft

TOTAL_BUDGET = 1480.0  # Leave 20s margin

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def compute_obj(a):
    """Compute full objective."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    return compute_obj_normalized(a_norm)

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad_smoothed(a, tau):
    """
    Compute gradient of smoothed objective f_tau(a) = (1/tau)*log(sum exp(tau*conv_k(a))).
    Returns (grad, conv, f_tau_value).
    grad[m] = 2 * sum_j a[j] * p[j+m] where p = softmax(tau * conv)
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    # Safe softmax
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 0)  # Prevent underflow
    exp_vals = np.exp(shifted)
    p_sum = float(np.sum(exp_vals))
    if p_sum < 1e-300:
        p = np.zeros_like(conv)
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / p_sum
    
    f_tau = max_conv + np.log(p_sum) / tau
    
    # Compute gradient via FFT cross-correlation
    # grad[m] = 2 * sum_j a[j] * p[j+m]
    # = 2 * ifft(conj(FFT(a)) * FFT(p))[m]
    N = 1
    while N < 3 * n:
        N <<= 1
    A = fft(a, N)
    P = fft(p, N)
    cross = ifft(np.conj(A) * P).real
    grad = 2.0 * cross[:n]
    
    return grad, conv, f_tau

def fft_smoothed_descent(a, time_budget, tau_init=5.0, tau_end=200.0, verbose=False):
    """
    Nesterov-accelerated projected gradient descent on the smoothed objective.
    a is on the simplex (sum=1, a>=0).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    
    # Nesterov momentum
    beta = 0.9
    v = np.zeros(n)
    it = 0
    best_it = 0
    
    # Learning rate schedule
    lr_init = 5.0 / np.sqrt(n)
    lr = lr_init
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        # Compute gradient
        grad, conv, f_tau = compute_grad_smoothed(y, tau)
        
        # Gradient step with adaptive learning rate
        # Use the Lipschitz constant estimate
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break
        
        # Step size: use 2/L where L is estimated Lipschitz constant
        # L ≈ 2 * tau * max_conv * 2 (rough upper bound)
        L_est = max(4.0 * tau * float(np.max(conv)) * n, 1.0)
        step = lr * min(1.0, 2.0 / L_est)
        
        a_new = y - step * grad
        a_new = project_simplex(a_new)
        
        # Evaluate
        obj_new = compute_obj_normalized(a_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        # Update velocity (Nesterov)
        v = beta * v + (a - a_new)  # Note: this is -beta*v - step*grad effectively
        a = a_new
        
        # Decay learning rate
        lr *= 0.9999
        
        it += 1
        
        if verbose and it % 500 == 0:
            print(f"      FGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.4f}")
        
        # Stagnation check
        if it - best_it > 5000 and it > 1000:
            # Perturb and restart
            rng = np.random.default_rng(it)
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            beta = 0.8
            lr = lr_init
            best_it = it
        
        if it - best_it > 15000:
            break
    
    return best_a

def peak_reduction_cd(a, time_budget, verbose=False):
    """
    Coordinate descent that specifically targets the highest peaks.
    For each top peak, try to redistribute mass to reduce it.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stall = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        
        # Find top 5 peaks
        top_k = np.argsort(conv)[::-1][:5]
        
        improved = False
        for k in top_k:
            # For peak at position k, the contributing pairs are (i, k-i) for valid i
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            
            # Find the top contributing pair
            best_reduction = best_obj
            best_move = None
            
            # Try moving mass from high contributors to low areas
            for i in range(i_start, i_end + 1):
                j = k - i
                if i == j:
                    continue
                if a[i] < 1e-10:
                    continue
                
                # Try reducing a[i] and increasing a[j] (or vice versa)
                for frac in [0.1, 0.05, 0.02]:
                    delta = frac * a[i]
                    if delta > a[i]:
                        continue
                    
                    # Move from i to some other position
                    # Choose target: far from current peak contributors
                    targets = [j if j != i else (i + 1) % n,
                               (i + n // 3) % n, (i + 2 * n // 3) % n,
                               int(rng.integers(0, n))]
                    
                    for tgt in targets:
                        tgt = int(tgt) % n
                        if tgt == i:
                            continue
                        a_try = a.copy()
                        a_try[i] -= delta
                        a_try[tgt] += delta
                        if a_try[i] < 0:
                            continue
                        a_try = a_try / a_try.sum()
                        obj_try = compute_obj_normalized(a_try)
                        if obj_try < best_reduction:
                            best_reduction = obj_try
                            best_move = (a_try.copy(), obj_try)
            
            if best_move is not None:
                a = best_move[0]
                best_obj = best_move[1]
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
    
    return best_a

def lp_direction(a):
    """
    Solve LP: min M s.t. conv(a, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns g (direction on simplex).
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None
    
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build the convolution matrix A (2n-1 x n) where A[k, j] = a[k-j] if valid
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(max(0, j), min(2 * n - 1, j + n)):
            A[k, j] = a_norm[k - j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None

def lp_polish(a, time_budget, verbose=False):
    """LP-based polishing: repeatedly find LP direction and blend."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    
    it = 0
    while time.monotonic() - start < time_budget:
        g = lp_direction(a)
        if g is None:
            break
        
        # Line search over t in [0, 1]
        best_t = 0
        best_obj_t = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t
        
        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            if best_obj_t < best_obj:
                best_obj = best_obj_t
                best_a = a.copy()
            it += 1
            if verbose and it % 5 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            break
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 0.01))
        a = a / a.sum()
    elif mode == 3:
        freqs = rng.integers(1, 6, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.5)**2) + 0.1
        a = a / a.sum()
    elif mode == 5:
        a = rng.pareto(1.5, n) + 0.1
        a = a / a.sum()
    else:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    return a
