    fa = rfft(a, size)

    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

        A[j:j + r, j] = a[:r]
    return A

    # where g_j = 2*a_{k-j} if 0<=k-j<n else 0
    # = conv[k] + sum_j g_j * x_j - sum_j g_j * a_j
    # Note: sum_j g_j * a_j = sum_j 2*a_{k-j}*a_j = 2*(a*a)[k] = 2*conv[k]
    # So: (x*x)[k] ≈ conv[k] + sum_j g_j * x_j - 2*conv[k] = sum_j g_j * x_j - conv[k]
    # Constraint: sum_j g_j * x_j - conv[k] <= M
    # i.e., sum_j 2*a_{k-j}*x_j <= M + conv[k]
    
    # Let's verify: if x = a, then LHS = sum_j 2*a_{k-j}*a_j = 2*conv[k], RHS = M + conv[k]
    # So constraint is 2*conv[k] <= M + conv[k], i.e., conv[k] <= M. ✓
    
    A_lin = np.zeros((K_peaks, n))
    rhs_offset = np.zeros(K_peaks)
    
    for i, k in enumerate(top_k_idx):
        k = int(k)
        for m in range(n):
            idx = k - m
            if 0 <= idx < n:
                A_lin[i, m] = 2.0 * a[idx]
        rhs_offset[i] = conv[int(k)]
    
    # LP: min M
    # s.t. A_lin[i] @ x <= M + rhs_offset[i] for i = 0..K-1
    #      sum(x) = 1
    #      x >= 0
    #      |x - a|_inf <= trust_radius (trust region)
    
    x = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    constraints = [cp.sum(x) == 1]
    for i in range(K_peaks):
        constraints.append(A_lin[i] @ x <= M + rhs_offset[i])
    
    # Trust region (box constraint)
    constraints.append(x <= a + trust_radius)
    constraints.append(x >= np.maximum(a - trust_radius, 0))
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if x.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                x_val = np.maximum(x.value, 0)
                s_val = float(np.sum(x_val))
                if s_val > 1e-10:
                    x_val = x_val / s_val
                    return x_val, float(M.value)
                break
            elif prob.status == 'infeasible':
                return a, M_current
        except Exception:
            continue
    
    return a, M_current


def slp_optimize(a, K_peaks=20, time_budget=300.0, verbose=False, seed=42):
    """
    Full SLP optimization loop with adaptive trust region and damping.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    trust_radius = 0.3
    damping = 1.0
    no_improve = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if verbose and it % 5 == 0:
            print(f"    SLP it{it}: obj={best_obj:.8f}, trust={trust_radius:.4f}, damp={damping:.2f}, t={elapsed:.0f}s")
        
        x_sol, M_lp = slp_step(a, K_peaks=K_peaks, trust_radius=trust_radius)
        
        if x_sol is None or np.allclose(x_sol, a):
            # Try smaller trust region
            trust_radius *= 0.5
            if trust_radius < 1e-6:
                break
            continue
        
        # Damped step
        a_try = (1 - damping) * a + damping * x_sol
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            a_try = a.copy()
        else:
            a_try = a_try / s_try
        
        obj_try = compute_obj(a_try)
        
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()
            no_improve = 0
            # Can increase trust region slightly
            trust_radius = min(0.5, trust_radius * 1.1)
            damping = min(1.0, damping * 1.05)
        else:
            no_improve += 1
            if no_improve == 1:
                damping *= 0.5
            elif no_improve == 3:
                trust_radius *= 0.5
                damping = 1.0
            elif no_improve >= 6:
                # Perturb
                scale = 0.01 * min(no_improve, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                trust_radius = 0.3
                damping = 1.0
                if it > 200:
                    break
        
        it += 1
    
    return best_a

def subgradient_refine(a, time_budget=80.0, K=5, verbose=False):
    """Subgradient method with top-K peaks."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 2.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        weights = np.exp((peak_vals - max_pv) * 5.0 / max_pv)
        weights /= weights.sum()
        
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            for m in range(n):
                idx = k - m
                if 0 <= idx < n:
                    g[m] += w * 2.0 * a[idx]
        
        v = momentum * v + g
        a_new = a - lr * v
        
        # Project onto simplex
        u = np.sort(a_new)[::-1]
        cssv = np.cumsum(u)
        idx_arr = np.arange(1, n + 1)
        cond = u * idx_arr > (cssv - 1.0)
        rho_arr = np.where(cond)[0]
        if len(rho_arr) == 0:
            a_new = np.ones(n) / n
        else:
            rho = rho_arr[-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_new - theta, 0.0)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-10:
                    break
            else:
                lr *= 0.995
        it += 1
    
    return best_a


def spectral_init(n, rng):
    """Initialize with flat spectral magnitude (approximate)."""
    # Start with random phases in frequency domain
    # FFT(a) should have roughly constant magnitude
    target_mag = np.ones(n) / np.sqrt(n)
    phases = rng.uniform(0, 2 * np.pi, n)
    F = target_mag * np.exp(1j * phases)
    a = ifft(F).real
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
SLP-based optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Successive Linear Programming with trust region as the primary optimizer,
combined with LP-blending, subgradient, multi-n exploration, spectral
initialization, and perturbative restarts. Key new ideas: direct SLP
linearization of the full quadratic objective, adaptive trust region with
peak weighting, and diverse initialization strategies.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def slp_solve_core(a, K_peaks, trust_radius):
    """
    Core SLP step: linearize at current a, solve LP, return (x_sol, M_val).
    Returns (None, None) on failure.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a = a / s

    conv = compute_conv(a)
    M_current = float(np.max(conv))

    # Identify top-K peaks with weighting
    top_k_idx = np.argsort(conv)[::-1][:K_peaks]

    # Build linearized constraint matrix
    # sum_j 2*a_{k-m} * x_j <= M + conv[k]
    A_lin = np.zeros((K_peaks, n))
    rhs_extra = np.zeros(K_peaks)

    for i, k in enumerate(top_k_idx):
        k = int(k)
        for m in range(n):
            idx = k - m
            if 0 <= idx < n:
                A_lin[i, m] = 2.0 * a[idx]
        rhs_extra[i] = conv[k]

    x = cp.Variable(n, nonneg=True)
    M = cp.Variable()

    constraints = [cp.sum(x) == 1]
    for i in range(K_peaks):
        constraints.append(A_lin[i] @ x <= M + rhs_extra[i])

    # Trust region
    lower = np.maximum(a - trust_radius, 0)
    upper = a + trust_radius
    constraints.append(x <= upper)
    constraints.append(x >= lower)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if x.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                x_val = np.maximum(x.value, 0)
                s_val = float(np.sum(x_val))
                if s_val > 1e-10:
                    x_val = x_val / s_val
                    return x_val, float(M.value)
                break
            elif prob.status == 'infeasible':
                return a, M_current
        except Exception:
            continue

    return None, None

    trust_radius = 0.3

    no_improve = 0
    it = 0

        elapsed = time.monotonic() - start

            trust_radius *= 0.7
            if trust_radius < 1e-6:
                break
            continue

    return best_a


def lp_blend_solve(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_g = float(np.sum(g_val))
                if s_g > 1e-10:
                    return g_val / s_g, float(M.value)
                break
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue
    return None, None


def golden_section_top(f, a_lo, a_hi, tol, max_iter):
    """Golden section search - top-level to avoid closure issues."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if abs(a_hi - a_lo) < tol:
            break
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = f(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = f(d)
    return (a_lo + a_hi) / 2.0


def blend_eval(t, best_a, g_val):

    a_try = (1.0 - t) * best_a + t * g_val

    s_try = float(np.sum(a_try))

def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx_arr = np.arange(1, n + 1)
    cond = u * idx_arr > (cssv - 1.0)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_refine(a, time_budget, K, verbose):
    """Subgradient method with top-K peaks and momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        weights = np.exp((peak_vals - max_pv) * 5.0 / max_pv)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            for m in range(n):
                idx = k - m
                if 0 <= idx < n:
                    g[m] += w * 2.0 * a[idx]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-10:
                    break
            else:
                lr *= 0.995
        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def spectral_init(n, rng):
    """Initialize with approximately flat spectral magnitude."""
    target_mag = np.ones(n) / np.sqrt(n)
    phases = rng.uniform(0, 2 * np.pi, n)
    F = target_mag * np.exp(1j * phases)
    a = ifft(F).real
    a = np.maximum(a, 0.001)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a

        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)

        alloc1 = min(remaining * 0.25, 50.0)
        if alloc1 > 15:
            a_try = slp_optimize(a_try, K_peaks=20, time_budget=alloc1, verbose=False, seed=n_try)

        alloc2 = min(remaining * 0.15, 40.0)
        if alloc2 > 10:
            a_try = lp_polish_blend(a_try, time_budget=alloc2, verbose=False, seed=n_try + 1)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Perturbation + SLP restarts ===
    print(f"\n=== Phase 7: Perturbation Restart ===")
    remaining = deadline - time.monotonic()

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        scale = 0.004 + 0.004 * r
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc = min(remaining * 0.25, 80.0)
        if alloc > 15:
            a_pert = slp_optimize(a_pert, K_peaks=20, time_budget=alloc, verbose=False, seed=300 + r)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  ** Perturb {r} (s={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Perturb {r} (s={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 8: Final intensive SLP ===
    print(f"\n=== Phase 8: Final SLP ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.7, 200.0)
        a_final = slp_optimize(best_seq, K_peaks=30, time_budget=alloc, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Final SLP: obj={best_score:.8f} **")
        else:
            print(f"  Final SLP: obj={obj_final:.8f}")

    # Quick final subgradient
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.4, 30.0)
        a_sg = subgradient_refine(best_seq, time_budget=alloc, K=10, verbose=False)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  ** Final SG: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
