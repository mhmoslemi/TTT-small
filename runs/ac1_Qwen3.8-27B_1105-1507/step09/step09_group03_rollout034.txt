Uses FFT-based smoothed gradient descent with Nesterov acceleration,
LP polishing, and asymmetric initialization for faster convergence.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    # First project onto sum=1 simplex, then scale
    v_scaled = v / target_sum if target_sum > 0 else v
    u = np.sort(v_scaled)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v_scaled - theta, 0.0) * target_sum
    return result

def build_conv_matrix_rows(a, rows):
    """Build selected rows of the convolution matrix.
    A @ g = conv(a, g) for the specified rows only.
    """
    n = len(a)
    A = np.zeros((len(rows), n))
    for ii, k in enumerate(rows):
        # conv(a, g)[k] = sum_i a[i] * g[k-i]
        # i ranges from max(0, k-n+1) to min(k, n-1)
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i  # index in g
            A[ii, j] = a[i]
    return A


def solve_lp_direction(a, active_rows=None, warm_start=None):
    """
    Solve LP: minimize M subject to conv(a, g)[k] <= M for k in active_rows (or all),
    sum(g) = 1, g >= 0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    if active_rows is None:
        all_rows = range(2 * n - 1)
    else:
        all_rows = active_rows
    
    A = build_conv_matrix_rows(a_norm, all_rows)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(M.value)
        except Exception:
            continue
    return None, None

    """

    s = float(np.sum(a))

        return a
    a = a / s

    best_obj = compute_obj(a)
    start = time.monotonic()
    
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget and it < max_iters:
        # Find active constraints (top peaks)
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        threshold = max_val * 0.95
        active = np.where(conv >= threshold)[0]
        
        if len(active) < 5:
            # Always include the top 5
            top_idx = np.argsort(conv)[::-1][:5]
            active = np.unique(np.concatenate([active, top_idx]))
        
        # Solve LP with active constraints
        g_lp, M_lp = solve_lp_direction(a, active_rows=active)
        
        if g_lp is None:
            break
        
        # Line search: find best blend
        best_blend_obj = best_obj
        best_blend_a = a.copy()
        
        for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:

            a_try = a_try / s_try
            obj_try = compute_obj(a_try)

            best_obj = best_blend_obj
            best_a = best_blend_a.copy()
            a = best_blend_a
            stall = 0
        else:
            stall += 1
            if stall > 10:
                break
        
        it += 1
    
    return best_a

            best_it = it

        v = a_new - a
        a = a_new
        
        # Adaptive learning rate
        if obj_new > best_obj * 1.01:

                v = np.zeros(n)

                momentum = 0.9
                best_it = it

        if verbose and it % 500 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")

def diverse_init(n, rng, mode=0):
    """Generate diverse asymmetric initializations."""
    if mode == 0:
        # Linear ramp
        a = np.arange(1, n + 1, dtype=np.float64)
        a = a / a.sum()
    elif mode == 1:
        # Quadratic bump (asymmetric)
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.3) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 2:
        # Uniform
        a = np.ones(n) / n
    elif mode == 3:
        # Sine-weighted
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
        a = a / a.sum()
    elif mode == 4:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** 2.0
        a = a / a.sum()
    elif mode == 5:
        # Random smooth
        freqs = rng.integers(1, 4, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Two bumps (asymmetric)
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.2) ** 2) / (2 * 0.1 ** 2)) + 0.5 * np.exp(-((x - 0.7) ** 2) / (2 * 0.15 ** 2))
        a = a / a.sum()
    else:
        # Exponential decay
        a = np.exp(-np.linspace(0, 3, n))
        a = a / a.sum()
    return a


def try_different_n(a_best, n_try, time_budget=30.0):
    """
    Try a different sequence length by resampling and quick optimization.
    """
    n_cur = len(a_best)
    if n_cur == n_try:
        return a_best.copy(), compute_obj(a_best)
    
    # Resample
    x_old = np.linspace(0, 1, n_cur)
    x_new = np.linspace(0, 1, n_try)
    a_try = np.interp(x_new, x_old, a_best)
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 1e-10:
        return a_best.copy(), compute_obj(a_best)
    a_try = a_try / s
    
    # Quick Nesterov
    if time_budget > 10:
        a_try = nesterov_smoothed_descent(a_try, time_budget=time_budget,
                                           tau_init=20.0, tau_end=200.0)
    
    return a_try, compute_obj(a_try)

The key identity: `sum_m w[j+m] * a[m]` is the convolution of `w` and `a` at a specific index.

Define `w_pad` (length 2n-1, zero-padded to N) and `a_pad` (length n, zero-padded to N, but REVERSED).

`a_rev[m] = a[n-1-m]` (length n).

Then `(w * a_rev)[k] = sum_i w[i] * a_rev[k-i] = sum_i w[i] * a[n-1-(k-i)] = sum_i w[i] * a[n-1-k+i]`.

Set `n-1-k+i = m`, so `i = k-n+1+m`. Then:
`(w * a_rev)[k] = sum_m w[k-n+1+m] * a[m]` (where both indices are valid).

So `(w * a_rev)[k] = sum_m w[(k-n+1)+m] * a[m] = C[k-n+1]` (by definition of C).

Therefore: `C[j] = (w * a_rev)[j+n-1]`.

So I need to compute the linear convolution of `w` (length 2n-1) and `a_rev` (length n), and take indices `n-1, n, ..., 2n-2`.
