def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def golden_section_line_search(a, g, n_evals=8):
    """
    Find optimal t in [0,1] that minimizes compute_obj((1-t)*a + t*g).
    Uses golden section search.
    """
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ~0.618

    lo = 0.0
    hi = 1.0

    for _ in range(n_evals):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)

        a1 = (1 - t1) * a + t1 * g
        a1 = np.maximum(a1, 0)
        s1 = float(np.sum(a1))
        if s1 > 1e-10:
            a1 = a1 / s1
            f1 = compute_obj(a1)
        else:
            f1 = np.inf

        a2 = (1 - t2) * a + t2 * g
        a2 = np.maximum(a2, 0)
        s2 = float(np.sum(a2))
        if s2 > 1e-10:
            a2 = a2 / s2
            f2 = compute_obj(a2)
        else:
            f2 = np.inf

        if f1 < f2:
            hi = t2
        else:
            lo = t1

    t_best = (lo + hi) / 2
    a_best = (1 - t_best) * a + t_best * g
    a_best = np.maximum(a_best, 0)
    s_best = float(np.sum(a_best))
    if s_best > 1e-10:
        a_best = a_best / s_best
    return a_best

def smoothed_gradient_fft(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of Q_k(a) using FFT cross-correlation.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv

def peak_targeted_cd(a, time_budget=60.0, verbose=False):
    """
    Coordinate descent targeting the top peaks.
    For the dominant peak, find the best (source, target) pair to move mass.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(9999)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_k = int(np.argmax(conv))

        # Find the dominant contributing pair
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)

        if i_start > i_end:
            break

        # Compute contributions
        contribs = []
        for i in range(i_start, i_end + 1):
            j = max_k - i
            if 0 <= j < n:
                contribs.append((a[i] * a[j], i, j))

        if not contribs:
            break

        # Sort by contribution
        contribs.sort(reverse=True)

        improved = False
        # Try moving mass from top contributors to random targets
        for contrib_val, src, dst in contribs[:3]:
            if src == dst:
                continue

            # Try several deltas
            for delta_frac in [0.05, 0.02, 0.01, 0.005]:
                delta = delta_frac * max(a[src], 1e-10)
                if delta > a[src]:
                    continue

                # Try several targets
                targets = [src - max(1, n // 20), src + max(1, n // 20),
                           rng.integers(0, n), rng.integers(0, n), n - 1 - src]

                for tgt in targets:
                    tgt = int(tgt) % n
                    if tgt == src:
                        continue
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[tgt] += delta
                    if a_try[src] < 0:
                        continue
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break

        if verbose and it % 20 == 0:
            print(f"    PCD it{it}: obj={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    else:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    return a
