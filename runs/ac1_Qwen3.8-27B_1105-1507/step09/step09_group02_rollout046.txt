import numpy as np

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, top_k=None, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k (or top_k k's), sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Get current convolution peaks to identify top_k
    conv_current = compute_conv(a_norm)
    if top_k is not None and top_k < 2 * n - 1:
        top_indices = np.argsort(conv_current)[::-1][:top_k]
        A = build_conv_matrix(a_norm)
        A_reduced = A[top_indices, :]
    else:
        A = build_conv_matrix(a_norm)
        A_reduced = A

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_reduced @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g, obj_current, num_points=20):

    Find best blending parameter t in [0,1] by evaluating objective at multiple points.
    Uses golden section search after initial grid.

    best_obj = obj_current

    ts = np.linspace(0, 1, num_points)
    obj_vals = np.zeros(num_points)
    for i, t in enumerate(ts):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            obj_vals[i] = np.inf
        else:
            obj_vals[i] = compute_obj(a_try)
    
    # Find best in grid
    min_idx = np.argmin(obj_vals)
    best_t = ts[min_idx]
    best_obj = obj_vals[min_idx]
    
    # Refine with golden section in neighborhood
    if 0 < min_idx < num_points - 1:
        lo = ts[max(0, min_idx - 1)]
        hi = ts[min(num_points - 1, min_idx + 1)]
        gr = (np.sqrt(5) - 1) / 2
        a1 = hi - gr * (hi - lo)
        b1 = lo + gr * (hi - lo)
        f_a1 = _blend_obj(a, g, a1)
        f_b1 = _blend_obj(a, g, b1)
        
        for _ in range(30):
            if f_a1 < f_b1:
                hi = b1
                b1 = a1
                f_b1 = f_a1
                a1 = hi - gr * (hi - lo)
                f_a1 = _blend_obj(a, g, a1)
            else:
                lo = a1
                a1 = b1
                f_a1 = f_b1
                b1 = lo + gr * (hi - lo)
                f_b1 = _blend_obj(a, g, b1)
        
        mid = (lo + hi) / 2
        f_mid = _blend_obj(a, g, mid)
        if f_mid < best_obj:
            best_obj = f_mid
            best_t = mid
    
    return best_t, best_obj


def _blend_obj(a, g, t):
    """Compute objective of blended sequence."""
    a_try = (1 - t) * a + t * g
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 1e-10:
        return np.inf
    return compute_obj(a_try)

def spectral_smooth(a, bandwidth):
    """
    Apply spectral smoothing: multiply the FFT of a by a Gaussian window
    in frequency domain to smooth out high-frequency oscillations.
    """
    n = len(a)
    N = 1
    while N < 2 * n:
        N <<= 1
    A = fft(a, N)
    freqs = np.fft.fftfreq(N)
    # Gaussian window in frequency
    window = np.exp(-2 * (np.pi * bandwidth * freqs) ** 2)
    A_smooth = A * window
    a_smooth = ifft(A_smooth, N).real[:n]
    a_smooth = np.maximum(a_smooth, 0)
    s = float(np.sum(a_smooth))
    if s < 1e-10:
        return a.copy()
    return a_smooth / s


def peak_gradient_descent(a, time_budget, K=5, verbose=False):
    """
    Direct peak reduction: compute subgradient from top K peaks and
    use projected gradient descent with backtracking line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 2.0 / n
    momentum = 0.8
    v = np.zeros(n)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Use softmax weighting over top peaks
        sorted_idx = np.argsort(conv)[::-1]
        top_idx = sorted_idx[:K]
        top_vals = conv[top_idx]
        max_val = top_vals[0]
        
        # Exponential weighting
        weights = np.exp(5.0 * (top_vals - max_val) / (max_val + 1e-15))
        weights /= weights.sum()

        # Compute gradient
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        # Momentum update
        v = momentum * v + g
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break

        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            if lr < 1e-8:
                break
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.01, 5.0 / n)
        else:
            stall += 1
            lr *= 0.7
            a = best_a
            if stall > 50:
                momentum = max(0.3, momentum * 0.9)
                stall = 0
            if lr < 1e-10:
                break

        if verbose:
            elapsed = time.monotonic() - start
            if int(elapsed * 2) % 10 == 0:
                print(f"    PGPD: obj={best_obj:.8f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a


def mirror_descent(a, time_budget, verbose=False):
    """
    Mirror descent (exponential weights) on the smoothed objective.
    Use entropic regularization for natural simplex projection.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Use log-sum-exp smoothing
    tau = 50.0
    lr = 1.0 / np.sqrt(n)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        # Adaptive tau
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        tau = 20.0 + 300.0 * progress ** 1.5

        conv = compute_conv(a)
        max_c = float(np.max(conv))
        
        # Softmax weights over convolution
        p = np.exp(tau * (conv - max_c))
        p_sum = float(np.sum(p))
        if p_sum < 1e-15:
            p = np.full(len(conv), 1.0 / len(conv))
        else:
            p = p / p_sum

        # Gradient: d/d_a_i [sum_k p_k * conv_k]
        # conv_k = sum_j a_j * a[k-j]
        # d(conv_k)/d(a_i) = a[k-i] + a[i-k] (for valid indices)
        # So grad_i = sum_k p_k * (a[k-i] + a[i-k])
        # This is 2 * (p * a)_i (circular convolution approx)
        
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        # Mirror descent update: a_i <- a_i * exp(-lr * g_i) / Z
        log_a = np.log(a + 1e-300)
        log_a_new = log_a - lr * g
        # Subtract max for numerical stability
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                lr *= 0.5
                stall = 0
            if lr < 1e-10:
                break

        it += 1
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = (1 - x**2) ** 0.5 + 0.1
        a = a / a.sum()
    elif mode == 3:
        # Two-bump
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * (w * 1.5) ** 2))
        a = a / a.sum()
    elif mode == 4:
        # Cosine-modulated
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.cos(2 * np.pi * 2 * x)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** 0.3
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

"""

Strategy: Enhanced LP polishing with golden-section line search, peak-targeted
constraints, mirror descent (entropic), spectral smoothing, peak gradient descent,

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, full_A, top_idx=None, timeout=10.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for k in top_idx (or all k),
    sum(g)=1, g>=0. Returns (g, M) or (None, None).
    """
    n = len(a)
    if top_idx is not None:
        A_reduced = full_A[top_idx, :]
    else:
        A_reduced = full_A

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_reduced @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, timeout=timeout)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def blend_obj(a, g, t):
    """Compute objective of (1-t)*a + t*g."""
    a_try = (1.0 - t) * a + t * g
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 1e-10:
        return np.inf
    return compute_obj(a_try)


def golden_section_search(a, g, f_obj, lo=0.0, hi=1.0, tol=1e-6, max_iter=40):
    """Golden section search for minimum of f(t) = obj((1-t)*a + t*g) on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f_obj(c)
    fd = f_obj(d)

    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f_obj(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f_obj(d)

    t_opt = (lo + hi) / 2.0
    f_opt = f_obj(t_opt)
    return t_opt, f_opt

def mirror_descent_v2(a, time_budget, verbose=False):
    """
    Mirror descent with entropic regularization (exponential weights).
    Uses log-sum-exp smoothed max of convolution peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    it = 0
    stall = 0
    tau_base = 30.0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_base + 400.0 * progress ** 2.0

        conv = compute_conv(a)
        max_c = float(np.max(conv))

        # Softmax weights
        exp_vals = np.exp(tau * (conv - max_c))
        p_sum = float(np.sum(exp_vals))
        if p_sum < 1e-300:
            p = np.full(len(conv), 1.0 / len(conv))
        else:
            p = exp_vals / p_sum

        # Gradient via FFT: grad_i = 2 * sum_k p[k] * a[|k-i|] (circular approx)
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        # Entropic mirror descent: a_i <- a_i * exp(-lr * g_i)
        log_a = np.log(a + 1e-300)
        log_a_new = log_a - lr * g
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.005, 10.0 / np.sqrt(n))
        else:
            stall += 1
            if stall > 30:
                lr *= 0.5
                stall = 0
            if lr < 1e-10:
                break

        it += 1
        if verbose and it % 200 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")

def peak_gradient_descent(a, time_budget, K=5, verbose=False):
    """
    Direct peak reduction using subgradient from top K peaks with momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 1.0 / n
    momentum = 0.85
    v = np.zeros(n)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        sorted_idx = np.argsort(conv)[::-1]
        top_idx = sorted_idx[:K]
        top_vals = conv[top_idx]
        max_val = top_vals[0] + 1e-15

        # Softmax weighting over top peaks
        weights = np.exp(3.0 * (top_vals - max_val))
        weights /= weights.sum()

        # Subgradient
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break

        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            if lr < 1e-10:
                break
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.01, 5.0 / n)
        else:
            stall += 1
            lr *= 0.7
            a = best_a
            if stall > 40:
                momentum = max(0.3, momentum * 0.85)
                stall = 0
            if lr < 1e-10:
                break

        if verbose:
            elapsed = time.monotonic() - start
            if int(elapsed) % 10 == 0:
                print(f"    PGD: obj={best_obj:.8f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a


def spectral_smooth(a, bandwidth):
    """Apply Gaussian spectral smoothing."""
    n = len(a)
    N = 1
    while N < 2 * n:
        N <<= 1
    A = fft(a, N)
    freqs = np.fft.fftfreq(N)
    window = np.exp(-2.0 * (np.pi * bandwidth * freqs) ** 2)
    A_smooth = A * window
    a_smooth = ifft(A_smooth, N).real[:n]
    a_smooth = np.maximum(a_smooth, 0)
    s = float(np.sum(a_smooth))
    if s < 1e-10:
        return a.copy()
    return a_smooth / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def peak_equalization(a, n_iter=50, verbose=False):
    """
    Try to equalize the top peaks by iteratively solving for adjustments
    that bring the highest peak down while keeping others balanced.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(n_iter):
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        # Find peaks above 95% of max
        threshold = 0.95 * max_val
        peak_indices = np.where(conv >= threshold)[0]
        
        if len(peak_indices) == 0:
            break

        # Compute gradient pointing to reduce peaks
        g = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += a[j_start:j_end + 1][::-1]

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Small step
        step = 0.01 / g_norm
        a_new = a - step * g
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step *= 0.5
            if step < 1e-8:
                break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search orchestrator.
    Time allocation:
    - Phase 1: LP polishing (30%)
    - Phase 2: Mirror descent (12%)
    - Phase 3: Peak gradient descent (8%)
    - Phase 4: Peak equalization (5%)
    - Phase 5: Multi-n exploration (20%)
    - Phase 6: Final intensive polish (25%)
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_target = 800
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: LP Polishing (30%) ===
    print(f"\n=== Phase 1: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 420.0)
    if p1_budget > 30:
        # Two sub-rounds with perturbation between
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(p1_budget / (3 - rnd) * 0.6, remaining * 0.35)
            if alloc < 20:
                break
            a_lp = lp_polish_v2(best_seq, time_budget=alloc, verbose=True, rng=rng)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP r{rnd+1}: obj={best_score:.8f} **")
            else:
                print(f"  LP r{rnd+1}: obj={obj_lp:.8f}")
                if rnd < 2:
                    scale = 0.008 * (rnd + 1)
                    best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                    best_seq = np.maximum(best_seq, 0)
                    best_seq = best_seq / best_seq.sum()

    # === Phase 2: Mirror Descent (12%) ===
    print(f"\n=== Phase 2: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.12, 170.0)
    if p2_budget > 20:
        a_md = mirror_descent_v2(best_seq, time_budget=p2_budget, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  MD: obj={best_score:.8f} **")
        else:
            print(f"  MD: obj={obj_md:.8f}")
            # Try with spectral pre-smoothing
            for bw in [0.05, 0.1, 0.15]:
                remaining = deadline - time.monotonic()
                if remaining < 20:
                    break
                a_sm = spectral_smooth(best_seq, bw)
                a_md2 = mirror_descent_v2(a_sm, time_budget=min(40, remaining * 0.3))
                obj_md2 = compute_obj(a_md2)
                if obj_md2 < best_score:
                    best_score = obj_md2
                    best_seq = a_md2.copy()
                    print(f"  MD+smooth({bw}): obj={best_score:.8f} **")
                    break

    # === Phase 3: Peak Gradient Descent (8%) ===
    print(f"\n=== Phase 3: Peak Gradient Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.08, 110.0)
    if p3_budget > 15:
        for K in [3, 7, 15]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(p3_budget / 3, remaining * 0.4)
            a_pg = peak_gradient_descent(best_seq, time_budget=alloc, K=K)
            obj_pg = compute_obj(a_pg)
            if obj_pg < best_score:
                best_score = obj_pg
                best_seq = a_pg.copy()
                print(f"  PGD(K={K}): obj={best_score:.8f} **")
            else:
                print(f"  PGD(K={K}): obj={obj_pg:.8f}")

    # === Phase 4: Peak Equalization (5%) ===
    print(f"\n=== Phase 4: Peak Equalization ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.05, 70.0)
    if p4_budget > 10:
        a_pe = peak_equalization(best_seq, n_iter=200)
        obj_pe = compute_obj(a_pe)
        if obj_pe < best_score:
            best_score = obj_pe
            best_seq = a_pe.copy()
            print(f"  PeakEq: obj={best_score:.8f} **")
        else:
            print(f"  PeakEq: obj={obj_pe:.8f}")

    # === Phase 5: Multi-n Exploration (20%) ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Generate candidate lengths around current
    n_candidates = set()
    for delta in [-100, -50, 50, 100, 150, 200]:
        n_try = n_current + delta
        if 400 <= n_try <= 1200:
            n_candidates.add(n_try)
    # Also try some absolute values
    for n_abs in [600, 700, 750, 850, 900, 950, 1000, 1100]:
        if abs(n_abs - n_current) > 30:
            n_candidates.add(n_abs)
    
    n_options = sorted(n_candidates)

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.18, 50.0)
        if alloc > 10:
            a_try = lp_polish_v2(a_try, time_budget=alloc, rng=rng)
            # Quick mirror descent
            remaining = deadline - time.monotonic()
            alloc2 = min(remaining * 0.1, 30.0)
            if alloc2 > 10:
                a_try = mirror_descent_v2(a_try, time_budget=alloc2)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final Intensive Polish (25%) ===
    print(f"\n=== Phase 6: Final Intensive Polish ===")
    remaining = deadline - time.monotonic()

    # LP final
    if remaining > 25:
        alloc = min(remaining * 0.35, 150.0)
        a_lp = lp_polish_v2(best_seq, time_budget=alloc, verbose=True, rng=rng)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # MD final
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.25, 70.0)
        a_md = mirror_descent_v2(best_seq, time_budget=alloc)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # PGD final
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.15, 40.0)
        a_pg = peak_gradient_descent(best_seq, time_budget=alloc, K=5)
        obj_pg = compute_obj(a_pg)
        if obj_pg < best_score:
            best_score = obj_pg
            best_seq = a_pg.copy()
            print(f"  Final PGD: obj={best_score:.8f} **")

    # One last short LP
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.4, 50.0)
        a_lp = lp_polish_v2(best_seq, time_budget=alloc, rng=rng)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
