"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses intensive LP polishing with adaptive blending, high-tau smoothed gradient descent
with Armijo line search, coordinate descent fine-tuning, Frank-Wolfe diversification,
and systematic multi-n exploration.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (optimal_g, optimal_M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def armijo_line_search(a, g, direction, f0, c1=1e-4, rho=0.5, max_iter=50):
    """Armijo backtracking line search."""
    n = len(a)
    step = min(1.0, 5.0 / np.sqrt(n))
    for _ in range(max_iter):
        a_new = a + step * direction
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            return a, 0.0
        a_new = a_new / s
        f_new = compute_obj(a_new)
        if f_new <= f0 + c1 * step * float(np.dot(g, direction)):
            return a_new, step
        step *= rho
    return a, 0.0


def gd_armijo(a, time_budget=200.0, tau=100.0, verbose=False):
    """Gradient descent with Armijo line search and Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    v = np.zeros(n)
    it = 0
    best_it = 0
    tau_current = tau

    while time.monotonic() - start < time_budget:
        # Slowly increase tau
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        tau_current = tau + 500.0 * progress * progress

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau_current)

        # Nesterov update with Armijo
        direction = v - g / (np.linalg.norm(g) + 1e-10) * 0.1
        a_new, step = armijo_line_search(y, g, -g / (np.linalg.norm(g) + 1e-10), compute_obj(y))

        if step > 0:
            v = beta * v + (a_new - y)
            a = a_new
            obj_new = compute_obj(a)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a.copy()
                best_it = it
        else:
            # Decrease tau and retry
            v = np.zeros(n)
            tau_current = max(10.0, tau_current * 0.5)

        it += 1
        if it % 2000 == 0 and verbose:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau_current:.1f}")

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it

    return best_a


def coordinate_descent(a, time_budget=100.0, n_trials=5, verbose=False):
    """Coordinate descent: try modifying individual elements."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    multipliers = [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 0.3, 3.0]
    it = 0

    while time.monotonic() - start < time_budget:
        improved = False
        # Random order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            a_save = a[i]
            for mult in multipliers:
                a[i] = a_save * mult
                s = float(np.sum(a))
                if s < 1e-10:
                    a[i] = a_save
                    continue
                a = a / s
                obj = compute_obj(a)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break
                elif obj < compute_obj(a_save) - 1e-12:
                    improved = True
                    break
                else:
                    a[i] = a_save
                    s = float(np.sum(a))
                    a = a / s
            it += 1
        if not improved and verbose:
            if it > 2 * n:
                break
    return best_a


def frank_wolfe_step(a, tau=50.0):
    """Single Frank-Wolfe step: move toward the vertex with most negative gradient."""
    n = len(a)
    g, _ = smoothed_gradient_fft(a, tau)
    i_star = int(np.argmin(g))
    return i_star, g

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            a_lp = intensive_lp_polish(best_seq, time_budget=alloc, verbose=True,
                                       seed=1000 + rnd * 7)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
                # Perturb before next round
                noise = rng.normal(0, 0.005 * (rnd + 1), len(best_seq))
                best_seq = best_seq * np.exp(noise)
                best_seq = np.maximum(best_seq, 0)
                best_seq = best_seq / best_seq.sum()

    # === Phase 4: Coordinate descent fine-tuning (100s) ===
    print(f"\n=== Phase 4: Coordinate Descent (budget=100s) ===")
    p4_budget = min(100.0, (deadline - time.monotonic()) * 0.1)
    if p4_budget > 20:
        a_cd = coordinate_descent(best_seq, time_budget=p4_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CoordDesc: obj={best_score:.8f} **")
        else:
            print(f"  CoordDesc: obj={obj_cd:.8f}")

    # === Phase 5: Multi-n exploration (250s) ===
    print(f"\n=== Phase 5: Multi-n Exploration (budget=250s) ===")
    p5_budget = min(250.0, (deadline - time.monotonic()) * 0.2)
    n_best = len(best_seq)

    n_options = sorted(set([
        max(200, n_best - 200), max(200, n_best - 100),
        n_best + 50, n_best + 100, n_best + 200,
        max(300, n_best + 400),
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000]
    n_options = n_options[:5]
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick GD
        alloc1 = min(remaining * 0.3, 40.0)
        if alloc1 > 15:
            a_try = gd_armijo(a_try, time_budget=alloc1, tau=50.0)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.2, 35.0)
        if alloc2 > 10:
            a_try = intensive_lp_polish(a_try, time_budget=alloc2, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts (150s) ===
    print(f"\n=== Phase 6: Diverse Restarts (budget=150s) ===")
    p6_budget = min(150.0, (deadline - time.monotonic()) * 0.12)
    n_cur = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = 2 + r % 8
        a_r = diverse_init(n_cur, rng, mode=mode)
        # Blend with current best
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.25, 50.0)
        if alloc > 20:
            half = alloc * 0.5
            a_r = gd_armijo(a_r, time_budget=half, tau=60.0)
            remaining = deadline - time.monotonic()
            lp_alloc = min(remaining * 0.15, 25.0)
            if lp_alloc > 10:
                a_r = intensive_lp_polish(a_r, time_budget=lp_alloc, seed=500 + r)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau GD
    if remaining > 40:
        alloc = min(remaining * 0.3, 120.0)
        a_gd = gd_armijo(best_seq, time_budget=alloc, tau=200.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Final GD: obj={best_score:.8f} **")
        else:
            print(f"  Final GD: obj={obj_gd:.8f}")

    # Multiple LP passes
    remaining = deadline - time.monotonic()
    for lp_rnd in range(3):
        if remaining <= 15:
            break
        alloc = min(remaining * 0.35, 80.0)
        a_lp = intensive_lp_polish(best_seq, time_budget=alloc, verbose=False,
                                   seed=9000 + lp_rnd * 100)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP {lp_rnd}: obj={best_score:.8f} **")
        else:
            if lp_rnd < 2:
                noise = rng.normal(0, 0.003, len(best_seq))
                best_seq = best_seq * np.exp(noise)
                best_seq = np.maximum(best_seq, 0)
                best_seq = best_seq / best_seq.sum()
            break

    # Final coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_cd = coordinate_descent(best_seq, time_budget=min(remaining * 0.4, 60.0))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # One last LP if time
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_lp = intensive_lp_polish(best_seq, time_budget=min(remaining * 0.5, 40.0),
                                   seed=99999)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Ultra-final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
