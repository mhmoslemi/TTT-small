import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.sparse import csr_matrix
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_sparse_conv_matrix(a):
    """Build sparse matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows, cols, vals = [], [], []
    for j in range(n):
        k_start = max(0, j - (n-1))  # min k such that k-j >= 0 -> k >= j, but also k <= 2n-2
        k_end = min(2*n-2, j + (n-1))
        # A[k,j] = a[k-j] for k where 0 <= k-j <= n-1
        for offset in range(n):
            k = j + offset
            if k < 2*n-1:
                rows.append(k)
                cols.append(j)
                vals.append(a[offset])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def solve_lp(a, warm_start=None):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_sparse_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

                if s > 1e-10:

```python
def proj_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0: return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

    """
    Mirror descent for minimax: min_a max_k conv(a,a)[k] subject to a in simplex.
    Uses exponential weights on k to guide the descent direction.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    p = np.ones(2*n-1) / (2*n-1)  # weights on convolution indices
    
    start = time.monotonic()
    lr = 1.0 / np.sqrt(n)
    it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        eta = eta_init * (eta_end / eta_init) ** (progress ** 0.5)
        
        # Compute gradient: 2 * W * a
        grad = 2.0 * hankel_matvec(p, a)

        a = a_new

    """LP-based fixed point with adaptive relaxation."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    alpha = 0.5  # relaxation parameter
    start = time.monotonic()
    
    for it in range(max_iter):
        if time.monotonic() - start > time_limit:
            break
        
        g, M = solve_lp(a)
        if g is None:
            break
        
        # Try different alpha values
        best_local = best_obj
        best_local_a = best_a.copy()
        
        for alpha_try in [alpha * 0.5, alpha * 0.75, alpha, alpha * 1.25, alpha * 1.5, 1.0]:
            a_try = (1 - alpha_try) * a + alpha_try * g
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10: continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)

                best_local_a = a_try.copy()

            best_a = best_local_a.copy()
            a = best_a
            # Increase alpha if we made good progress
            if best_local < best_obj - 1e-6:
                alpha = min(1.0, alpha * 1.1)
        else:
            a = best_a
            alpha = max(0.1, alpha * 0.8)
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated mirror descent for minimax (fast, many iterations)
2. LP-based fixed-point with adaptive relaxation and warm-starting
3. Multi-length exploration with resampling
4. Intensive final polishing with targeted perturbations
Key: Hankel matrix-vector products via FFT for O(n log n) gradients.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.sparse import csr_matrix
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def hankel_matvec(p, a):
    """
    Compute W*a where W[j,l] = p[j+l] (Hankel matrix-vector product via FFT).
    p has length 2n-1, a has length n. Returns array of length n.
    """
    n = len(a)
    b = a[::-1]
    N = 1
    while N < len(p) + n - 1:
        N <<= 1
    full_conv = irfft(rfft(p, N) * rfft(b, N), N)
    return full_conv[n - 1:n - 1 + n]


def proj_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def mirror_descent_minimax(a, time_budget=150.0, T_init=None, T_end=None, verbose=False):
    """
    Mirror descent for minimax: min_a max_k conv(a,a)[k] s.t. a in simplex.
    Uses softmax weights on convolution indices with decreasing temperature.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    if T_init is None:
        T_init = 1.0 / n
    if T_end is None:
        T_end = 1.0 / (n * n)

    best_a = a.copy()
    best_obj = compute_obj(a)

    start = time.monotonic()
    lr = 1.5 / np.sqrt(n)
    v = np.zeros(n)
    beta = 0.85
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        T = T_init * (T_end / T_init) ** (progress ** 0.6)

        # Compute softmax weights p[k] = exp(conv[k]/T) / Z
        conv = compute_conv(a)
        max_c = np.max(conv)
        p = np.exp((conv - max_c) / max(T, 1e-12))
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = p / p_sum

        # Gradient: 2 * W * a where W[j,l] = p[j+l]
        grad = 2.0 * hankel_matvec(p, a)

        # Nesterov momentum step
        y = a + beta * v
        y = proj_simplex(y)
        grad_y = 2.0 * hankel_matvec(p, y)
        v = beta * v - lr * grad_y
        a_new = y + v
        a_new = proj_simplex(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        lr = max(1e-6, lr * 0.9997)

        if stall > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 1.5 / np.sqrt(n) * 0.5
            beta = 0.7
            stall = 0

        it += 1
        if verbose and it % 10000 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, T={T:.6f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a


def build_sparse_conv_matrix(a):
    """Build sparse matrix A (2n-1 x n) where A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        max_offset = min(n, 2 * n - 1 - j)
        for offset in range(max_offset):
            k = j + offset
            rows.append(k)
            cols.append(j)
            vals.append(a[offset])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp(a, timeout_s=10.0):
    """Solve LP: min M s.t. conv(a,g) <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    A = build_sparse_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish_round(a, max_iter=100, time_limit=80.0, verbose=False):
    """LP polishing with adaptive relaxation."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf, 0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    alpha = 0.5
    no_improve = 0
    start = time.monotonic()
    it = 0

    while it < max_iter and (time.monotonic() - start) < time_limit:
        g, M_lp = solve_lp(a)
        if g is None:
            break

        candidates = [0.1, 0.3, 0.5, 0.7, 0.9, 1.0,
                      max(0.05, alpha * 0.5), max(0.05, alpha * 0.8),
                      min(1.0, alpha * 1.2), min(1.0, alpha * 1.5)]

        best_t_obj = np.inf
        best_t_val = alpha
        best_a_local = best_a.copy()

        for t in candidates:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            alpha = best_t_val
            no_improve = 0
            if verbose and it % 5 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t_val:.2f}, M={M_lp:.6f}")
        else:
            no_improve += 1
            alpha = max(0.1, alpha * 0.7)
            if no_improve >= 4:
                break

    return best_a, best_obj, it


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Hamming window
        x = np.arange(n)
        a = 0.54 + 0.46 * np.cos(2 * np.pi * x / max(1, n - 1))
    elif mode == 3:
        # Hann window
        x = np.arange(n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x / max(1, n - 1))
    elif mode == 4:
        # Raised cosine with different params
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) ** 1.5
    elif mode == 5:
        # Gaussian bump
        x = np.arange(n)
        a = np.exp(-((x - n / 2) ** 2) / (2 * (n / 4) ** 2))
    elif mode == 6:
        # Two-peak structure
        x = np.arange(n)
        a = np.exp(-((x - n * 0.25) ** 2) / (2 * (n / 6) ** 2)) + \
            np.exp(-((x - n * 0.75) ** 2) / (2 * (n / 6) ** 2))
    elif mode == 7:
        # Flat top with rolled edges
        a = np.ones(n)
        fade = n // 10
        for i in range(fade):
            a[i] *= 0.5 + 0.5 * (i / fade)
            a[n - 1 - i] *= 0.5 + 0.5 * (i / fade)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 1e-8)
    a = a / a.sum()
    return a


def perturb_sequence(a, rng, method=0, scale=0.03):
    """Apply a perturbation to escape local minima."""
    n = len(a)
    a = a.copy()
    if method == 0:
        a = a * np.exp(rng.normal(0, scale, n))
    elif method == 1:
        noise = rng.uniform(1 - scale, 1 + scale, n)
        a = a * noise
    elif method == 2:
        mask = rng.random(n) > scale
        a = a * mask
    elif method == 3:
        shift = int(rng.integers(1, max(2, n // 10)))
        a = np.roll(a, shift)
    elif method == 4:
        a = (1 - scale) * a + scale / n
    elif method == 5:
        # Frequency domain perturbation
        A = fft(a, 2 * n)
        phase_noise = np.exp(1j * rng.normal(0, scale, len(A)))
        A = A * phase_noise
        a = ifft(A).real[:n]
        a = np.maximum(a, 0)
    elif method == 6:
        # Targeted: reduce peaks in convolution
        conv = compute_conv(a / a.sum())
        top_k = np.argsort(conv)[::-1][:10]
        deltas = np.zeros(n)
        for k in top_k:
            k = int(k)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    deltas[j] += a[idx]
        max_d = np.max(np.abs(deltas))
        if max_d > 1e-10:
            a = a - scale * 0.3 * deltas / max_d
        a = np.maximum(a, 0)
    elif method == 7:
        # Swap random pairs
        for _ in range(n // 20):
            i, j = rng.integers(0, n, size=2)
            a[i], a[j] = a[j], a[i]
    else:
        a = a * np.exp(rng.normal(0, scale, n))

    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Final MD: obj={best_score:.8f} **")
        else:
            print(f"  Final MD: obj={obj_sgd:.8f}")

    # Multiple LP polish rounds
    for lp_rnd in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining * 0.25, 60.0)
        if alloc < 8:
            break
        a_current = best_seq.copy()

        if lp_rnd > 0:
            a_current = perturb_sequence(a_current, rng, method=lp_rnd % 6, scale=0.015)

        a_lp, obj_lp, iters = lp_polish_round(a_current, max_iter=60, time_limit=alloc)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP {lp_rnd}: obj={best_score:.8f} ({iters} iters) **")
        else:
            print(f"  Final LP {lp_rnd}: obj={obj_lp:.8f} ({iters} iters)")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
