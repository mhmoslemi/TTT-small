"""
Optimized search for minimizing peak autocorrelation of non-negative step functions.
Uses smooth optimization (log-sum-exp + mirror descent), LP alternating with cutting planes,
exponential ramp initialization, n-sweep, and adaptive restarts.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def build_conv_matrix_sparse(n):
    """Build sparse convolution matrix A such that A@g = conv(a_ref, g) for a_ref of length n.
    Actually, we need A such that A @ g gives conv(a, g) for a FIXED a.
    So the matrix depends on a. Let's build it differently.
    
    For fixed a of length n, conv(a, g)[k] = sum_i a[i] * g[k-i] for valid indices.
    As a linear function of g: conv(a, g)[k] = sum_j a[k-j] * g[j] (for valid k-j).
    So A[k, j] = a[k-j] if 0 <= k-j < n, else 0.
    """
    # Returns a dense matrix for simplicity (n up to ~2000 should be OK)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a[k-j] for valid k-j
        # k-j is in [0, n-1] means k is in [j, j+n-1]
        a_vals = a  # a[0], a[1], ..., a[n-1]
        # For k = j, a_val = a[0]; for k = j+1, a_val = a[1]; etc.
        A[j:j+n, j] = a
    return A

def solve_lp_cutting_plane(a, max_iters=50, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0
    Using cutting plane: start with active constraints, add violated ones.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A_full = build_conv_matrix_sparse(n)  # (2n-1) x n
    M_current = float(np.max(A_full @ a_norm))
    
    # Start with constraints that are "near-active"
    conv_current = A_full @ a_norm
    threshold = M_current * 0.99
    active_idx = np.where(conv_current >= threshold)[0]
    if len(active_idx) == 0:
        active_idx = np.argsort(conv_current)[-n:]  # top n constraints
    
    # Current constraint set
    constraint_set = set(active_idx.tolist())
    
    for it in range(max_iters):
        k_constraints = len(constraint_set)
        idx = np.array(sorted(constraint_set))
        A_sub = A_full[idx]  # (k_constraints, n)
        
        # LP: min M s.t. A_sub @ g <= M, sum(g) = 1, g >= 0
        # Variables: g[0], ..., g[n-1], M
        # Total: n+1 variables
        c = np.zeros(n + 1)
        c[-1] = 1.0  # minimize M
        
        # Inequality: A_sub @ g - M <= 0
        A_ub = np.hstack([A_sub, -np.ones((k_constraints, 1))])
        b_ub = np.zeros(k_constraints)
        
        # Equality: sum(g) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(0, None)]
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status != 0:
                if verbose:
                    print(f"    LP failed at it{it}: {res.message}")
                break
        except Exception as e:
            if verbose:
                print(f"    LP error: {e}")
            break
        
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s < 1e-10:
            break
        g_val = g_val / s
        M_val = float(res.x[n])
        
        # Check if all constraints are satisfied
        conv_g = A_full @ g_val
        violations = np.where(conv_g > M_val + 1e-8)[0]
        
        if len(violations) == 0:
            if verbose:
                print(f"    LP converged in {it+1} iterations, M={M_val:.8f}")
            return g_val, M_val
        
        # Add violated constraints
        # Add top violators to avoid too many constraints at once
        viol_vals = conv_g[violations] - M_val
        top_viol = violations[np.argsort(viol_vals)[::-1][:min(50, len(violations))]]
        constraint_set.update(top_viol.tolist())
        
        if verbose:
            print(f"    LP it{it}: {len(violations)} violations, added {len(top_viol)}, constraints={len(constraint_set)}")
    
    return None, None

def lp_alternate_fast(a, time_budget, verbose=False):
    """LP alternating with cutting plane and adaptive step."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    stall = 0
    iteration = 0
    
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_cutting_plane(a, max_iters=30)
        if g_lp is None:
            break
        
        # Check similarity
        sim = 1.0 - np.linalg.norm(a - g_lp) / (np.linalg.norm(a) + np.linalg.norm(g_lp) + 1e-15)
        if sim > 0.9999:
            stall += 5
        else:
            stall = max(0, stall - 2)
        
        # Try full step first
        a_try = g_lp.copy()
        obj_try = compute_obj(a_try)
        
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()
            iteration += 1
            if verbose and iteration % 10 == 0:
                print(f"      LP-alt it{iteration}: obj={best_obj:.8f} (full step)")
        else:
            # Try fractional steps
            best_t_obj = best_obj
            best_t = 0
            for t in [0.5, 0.3, 0.2, 0.1, 0.7, 0.9]:
                a_t = (1 - t) * a + t * g_lp
                a_t = np.maximum(a_t, 0)
                s_t = float(np.sum(a_t))
                if s_t < 1e-10:
                    continue
                a_t = a_t / s_t
                obj_t = compute_obj(a_t)
                if obj_t < best_t_obj:
                    best_t_obj = obj_t
                    best_t = t
            
            if best_t > 0 and best_t_obj < best_obj - 1e-12:
                a = (1 - best_t) * a + best_t * g_lp
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_obj = best_t_obj
                best_a = a.copy()
                a = a.copy()
                iteration += 1
                if verbose and iteration % 10 == 0:
                    print(f"      LP-alt it{iteration}: obj={best_obj:.8f} (t={best_t:.2f})")
            else:
                stall += 1
        
        if stall > 20:
            # Perturbation to escape
            rng = np.random.default_rng(iteration * 12345 + n)
            scale = 0.05 + 0.02 * (stall // 20)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0
            
    return best_a

def mirror_descent(a, time_budget, beta_start=10, beta_end=500, verbose=False):
    """Mirror descent (multiplicative weights) for smooth approximation."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 0.1
    it = 0
    
    while time.monotonic() - start < time_budget:
        frac = min(1.0, it / max(1, int(time_budget * 2)))
        beta = beta_start * (beta_end / beta_start) ** frac
        
        conv = compute_conv(a)
        # Log-sum-exp for numerical stability
        conv_shifted = conv - conv.max()
        log_weights = beta * conv_shifted
        log_weights -= log_weights.max()
        weights = np.exp(log_weights)
        weights /= weights.sum()
        
        # Gradient: grad[m] = 2 * sum_j a[j] * weights[m+j] for valid m+j
        # Compute via cross-correlation
        # grad[m] = 2 * sum_{j=0}^{n-1} a[j] * weights[m+j] * 1_{m+j < 2n-1}
        # This is the cross-correlation of a and weights
        
        # Use FFT: cross_corr[m] = sum_j a[j] * w[m+j]
        # = (a * reverse(w))[m + n - 1] in convolution terms
        # Or: cross_corr = ifft(fft(a, 2n-1) * conj(fft(w, 2n-1)))
        
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        A_fft = rfft(a, size)
        W_fft = rfft(weights, size)
        cross = irfft(A_fft * np.conj(W_fft), size)[:n]
        # cross[m] = sum_j a[j] * conj(weights[m-j]) ... hmm this isn't quite right
        
        # Let me just do it directly for clarity (n up to ~2000, so O(n^2) might be OK)
        # Actually for n=2000, O(n^2) = 4M operations, which is fine.
        # But let's use the FFT properly.
        
        # cross_corr[m] = sum_{j=0}^{n-1} a[j] * weights[m+j]
        # Let's define: for m=0: sum_j a[j]*weights[j] = dot(a, weights)
        # for m=1: sum_j a[j]*weights[j+1]
        # This is a "sliding dot product" = cross-correlation
        
        # Using FFT: define a_ext = [a, 0, ..., 0] (length 2n-1)
        # w_ext = [0, ..., 0, weights] (shifted?)
        # Actually: sum_j a[j]*w[m+j] = (a * flip(w))[n-1+m] where * is convolution
        # flip(w) = w[::-1]
        # conv(a, flip(w))[n-1+m] = sum_i a[i] * flip(w)[n-1+m-i] = sum_i a[i] * w[i-m]
        # That's not what we want either.
        
        # Let me just use: sum_j a[j] * w[m+j]
        # = sum_j a[j] * w[m+j]
        # Define x = a, y = w. We want (x ⋆ y)[m] = sum_j x[j] * y[m+j]
        # In terms of correlation: R_xy[m] = sum_j x[j] * y[j+m]
        # This is the cross-correlation of x and y at lag m (with x "leading").
        # R_xy[m] = (x * flip(y))[m + len(y) - 1] ... I keep getting confused.
        
        # Let me just verify: for m=0, R = sum_j a[j]*w[j]
        # For m=1, R = sum_j a[j]*w[j+1]
        # This is equivalent to: take a, shift w left by m, take dot product.
        # In FFT terms: ifft(fft(a) * conj(fft(w))) gives circular cross-correlation.
        # For linear: pad to length >= 2n-1.
        
        # Actually, the standard result is:
        # ifft(fft(a, N) * conj(fft(w, N))) gives the linear cross-correlation
        # where the result at index m is sum_j a[j] * conj(w[j-m]) (circular)
        # Hmm, this gives R[m] = sum_j a[j+m] * conj(w[j]) for the circular case.
        
        # I think the cleanest way is:
        # sum_j a[j] * w[m+j] = sum_j a[j] * w[m+j]
        # Let k = j+m, then j = k-m:
        # = sum_k a[k-m] * w[k] (for valid indices)
        # This is the convolution of a[::-1] (reversed) with w, evaluated at position m+n-1!
        # (reversed_a * w)[p] = sum_i reversed_a[i] * w[p-i] = sum_i a[n-1-i] * w[p-i]
        # Let j = n-1-i: = sum_j a[j] * w[p-(n-1-j)] = sum_j a[j] * w[p-n+1+j]
        # We want this to equal sum_j a[j] * w[m+j], so p-n+1 = m, i.e., p = m+n-1.
        
        # So: grad[m]/2 = (reversed_a * w)[m+n-1]
        
        a_rev = a[::-1]
        size2 = 1
        while size2 < 2 * n - 1:
            size2 <<= 1
        cr = irfft(rfft(a_rev, size2) * rfft(weights, size2), size2)
        cross = cr[n-1:n-1+n]  # positions m+n-1 for m=0,...,n-1
        
        grad = 2.0 * cross
        
        # Mirror descent update
        log_a = np.log(a)
        log_a_new = log_a - lr * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = min(1.0, lr * 1.02)
        else:
            lr = max(0.001, lr * 0.7)
        
        it += 1
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"      Mirror it{it}: obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.4f}, t={elapsed:.1f}s")
    
    return best_a

def exponential_ramp(n, c=2.0):
    """Generate exponential ramp sequence."""
    x = np.arange(n, dtype=np.float64)
    a = np.exp(c * x / n)
    return a / a.sum()

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode % 10 == 0:
        return exponential_ramp(n, 1.5)
    elif mode % 10 == 1:
        return exponential_ramp(n, 2.0)
    elif mode % 10 == 2:
        return exponential_ramp(n, 2.5)
    elif mode % 10 == 3:
        return exponential_ramp(n, 3.0)
    elif mode % 10 == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(2.0 * x) * (1 + 0.2 * np.sin(4 * np.pi * x))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode % 10 == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(2.5 * x) * (1 + 0.1 * np.cos(6 * np.pi * x))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode % 10 == 6:
        # Power law
        x = np.arange(1, n + 1, dtype=np.float64)
        a = x ** 0.5
        return a / a.sum()
    elif mode % 10 == 7:
        # Power law 2
        x = np.arange(1, n + 1, dtype=np.float64)
        a = x ** 1.0
        return a / a.sum()
    elif mode % 10 == 8:
        # Gaussian centered off-center
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.7) ** 2) + 0.1
        return a / a.sum()
    else:
        # Random
        a = rng.uniform(0.1, 1.0, n)
        return a / a.sum()

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
