"""
Hybrid optimization for minimizing peak autocorrelation of non-negative step functions.
Combines fast FFT-based subgradient descent with reduced LP alternating, frequency-domain
shaping, and adaptive n-sweep. Uses analytic gradient structure of the convolution.
Target: score < 1.5027.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix_topK(a, K):
    """Build convolution matrix for top K constraints only."""
    n = len(a)
    conv = compute_conv(a)
    top_k_indices = np.argsort(conv)[::-1][:K]
    A = np.zeros((K, n))
    for idx, k in enumerate(top_k_indices):
        k = int(k)
        for j in range(n):
            val = k - j
            if 0 <= val < n:
                A[idx, j] = a[val]
    return A, top_k_indices


def solve_lp_reduced(a, K_ineq):
    """
    Solve reduced LP: min M s.t. conv(a,g)[k] <= M for top K lags, sum(g)=1, g>=0.
    Uses scipy linprog with HiGHS solver.
    """
    n = len(a)
    if n > 2000:
        return None
    a_norm = a / np.sum(a)
    conv = compute_conv(a_norm)
    top_k = np.argsort(conv)[::-1][:K_ineq]
    
    A = np.zeros((K_ineq, n))
    for idx, k in enumerate(top_k):
        k = int(k)
        for j in range(n):
            val = k - j
            if 0 <= val < n:
                A[idx, j] = a_norm[val]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K_ineq, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(K_ineq)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def solve_lp_full_cvxpy(a):
    """Solve full LP using cvxpy (for smaller n or when reduced LP fails)."""
    n = len(a)
    if n > 1500:
        return None
    a_norm = a / np.sum(a)
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        for k_idx in range(k_start, k_end):
            A[k_idx, j] = a_norm[k_idx - j]
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def subgradient_optimize(a, time_budget, verbose=False):
    """
    Fast subgradient method using FFT-based convolution and analytic gradients.
    Minimizes max_k conv(a,a)[k] over the simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 0.1 / np.sqrt(n)
    momentum = np.zeros(n)
    beta_mom = 0.7
    
    it = 0
    stall = 0
    patience = 50
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        current_obj = 2.0 * n * max_c
        
        # Find active set: lags where conv is within 5% of max
        threshold = max_c * 0.95
        active = np.where(conv >= threshold)[0]
        
        # Compute subgradient: g[m] = (2/|K|) * sum_{k in active} a[k-m] (with bounds)
        g = np.zeros(n)
        for k in active:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                g[j_start:j_end + 1] += a[k - j_end:k - j_start + 1][::-1]
        g *= 2.0 / len(active)
        
        # Momentum update
        momentum = beta_mom * momentum + (1 - beta_mom) * g
        direction = momentum
        
        # Line search: try decreasing step sizes
        found = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = lr * factor
            a_try = a - step * direction
            a_try = proj_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < current_obj:
                found = True
                break
        
        if found:
            new_obj = compute_obj(a_try)
            if new_obj < best_obj - 1e-12:
                best_obj = new_obj
                best_a = a_try.copy()
                a = a_try.copy()
                stall = 0
                lr = min(lr * 1.02, 1.0)
            else:
                a = a_try.copy()
                stall += 1
        else:
            stall += 1
            lr *= 0.8
        
        it += 1
        
        if stall > patience:
            # Restart with perturbation
            rng = np.random.default_rng(it * 12345 + n)
            scale = 0.02 + 0.01 * min(stall // patience, 5)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            else:
                a = best_a.copy()
            best_a = a.copy()
            best_obj = compute_obj(a)
            stall = 0
            lr = 0.1 / np.sqrt(n)
            momentum = np.zeros(n)
            patience = min(patience + 20, 200)
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"          Subgrad it{it}: obj={best_obj:.8f}, lr={lr:.5f}, "
                  f"stall={stall}, elapsed={elapsed:.1f}s")
    
    return best_a

def frequency_shaping(a, time_budget, verbose=False):
    """
    Frequency-domain shaping: iteratively flatten the power spectrum
    to reduce autocorrelation peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for gamma in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5]:
        if time.monotonic() - start > time_budget:
            break
        for it in range(30):
            if time.monotonic() - start > time_budget:
                break
            A = rfft(a, size)
            P = np.abs(A) ** 2 + 1e-30
            
            # Target: flatten P towards its geometric mean
            log_P = np.log(P)
            log_target = np.mean(log_P)
            # Blend in log domain
            log_P_new = (1 - gamma * 0.3) * log_P + gamma * 0.3 * log_target
            P_new = np.exp(log_P_new)
            
            # Preserve phase
            phase = A / np.abs(A + 1e-30)
            A_new = np.sqrt(P_new) * phase
            a_new = irfft(A_new, size)[:n]
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
            else:
                break
    
    return best_a


def targeted_refinement(a, time_budget, verbose=False):
    """
    Targeted refinement: identify peak autocorrelation lags and
    perform local 2-swap moves to reduce them.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    round_num = 0
    while time.monotonic() - start < time_budget:
        round_num += 1
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        
        # Find top peak lags
        sorted_lags = np.argsort(conv)[::-1]
        peak_lags = sorted_lags[:5]
        
        # Find positions that most contribute to each peak
        improved = False
        for k_star in peak_lags:
            k_star = int(k_star)
            # Contributors to this lag
            j_start = max(0, k_star - n + 1)
            j_end = min(n - 1, k_star)
            if j_start > j_end:
                continue
            
            # Get contribution magnitudes
            contribs = []
            for i in range(j_start, j_end + 1):
                j_val = k_star - i
                contribs.append((a[i] * a[j_val], i, j_val))
            contribs.sort(reverse=True)
            
            for contrib, i_src, j_src in contribs[:8]:
                if contrib < 1e-10:
                    continue
                # Try moving mass from the larger of (i_src, j_src) to a quiet position
                src = i_src if a[i_src] >= a[j_src] else j_src
                # Quiet positions: where a is small
                quiet_idx = np.argsort(a)[:10]
                
                for target in quiet_idx:
                    if target == src:
                        continue
                    for frac in [0.005, 0.01, 0.02, 0.03]:
                        delta = frac * a[src]
                        if delta <= 0 or delta >= 0.9 * a[src]:
                            continue
                        a[src] -= delta
                        a[target] += delta
                        new_conv = compute_conv(a)
                        new_obj = 2.0 * n * float(np.max(new_conv))
                        if new_obj < best_obj - 1e-12:
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[src] += delta
                            a[target] -= delta
                            break
                    if improved:
                        break
            
            if improved:
                conv = compute_conv(a)
        
        if not improved and round_num > 20:
            break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode % 10 == 0:
        a = np.ones(n) / n
    elif mode % 10 == 1:
        a = rng.uniform(0.05, 1, n)
        a = a / a.sum()
    elif mode % 10 == 2:
        # Linear ramp (asymmetric)
        a = np.linspace(0.3, 1.0, n)
        a = a / a.sum()
    elif mode % 10 == 3:
        # Quadratic ramp
        x = np.linspace(0, 1, n)
        a = (x ** 0.5 + 0.3)
        a = a / a.sum()
    elif mode % 10 == 4:
        # Cosine taper
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 5:
        # Two-hump asymmetric
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.3) ** 2) + 0.6 * np.exp(-8 * (x - 0.7) ** 2)
        a = a / a.sum()
    elif mode % 10 == 6:
        # Random Fourier features
        freqs = rng.integers(1, 10, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        amp = rng.uniform(0.1, 0.4, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p, am in zip(freqs, phases, amp):
            a += am * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 7:
        # Power-law decay
        a = np.power(np.linspace(1, 0.1, n), 1.5)
        a = a / a.sum()
    elif mode % 10 == 8:
        # Dirichlet with concentration
        a = rng.dirichlet(np.ones(n) * 0.3)
    else:
        # Sigmoid-like
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + np.exp(-5 * (x - 0.4)))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    return a

            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Subgrad: obj={best_score:.8f} **")
        
        # LP polish
        remaining = deadline - time.monotonic()
        lp_budget = min(remaining * 0.06, 60.0)
        if lp_budget > 20:
            a_lp = lp_alternate_improved(best_seq, time_budget=lp_budget, verbose=False)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP: obj={best_score:.8f} **")
            else:
                print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 2: n-sweep (18%) ===
    print(f"\n=== Phase 2: n-Sweep ===")
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.18
    n_options = sorted(set([
        max(400, n_current - 80),
        max(400, n_current - 40),
        max(400, n_current - 20),
        max(400, n_current - 10),
        n_current,
        n_current + 10,
        n_current + 20,
        n_current + 40,
        n_current + 80,
        n_current + 120,
        750, 768, 780, 800, 820, 840, 860, 880, 900, 920, 960, 1000, 1024,
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1200]

    per_n_budget = p2_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(per_n_budget, remaining * 0.12)
        if alloc < 10:
            alloc = 10

        a_try = resample_sequence(best_seq, n_try)
        # Quick subgradient then LP
        half = alloc / 2
        a_sg = subgradient_optimize(a_try, time_budget=half, verbose=False)
        obj_sg = compute_obj(a_sg)
        
        remaining = deadline - time.monotonic()
        lp_alloc = min(remaining * 0.06, 30.0)
        if lp_alloc > 8:
            a_lp = lp_alternate_improved(a_sg, time_budget=lp_alloc, verbose=False)
            obj_lp = compute_obj(a_lp)
            if obj_lp < obj_sg:
                a_try = a_lp
                obj_try = obj_lp
            else:
                a_try = a_sg
                obj_try = obj_sg
        else:
            a_try = a_sg
            obj_try = obj_sg
        
        marker = " **" if obj_try < best_score else ""
        print(f"  n={n_try}: obj={obj_try:.8f}{marker}")
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try

    # === Phase 3: Intensive LP alternating (35%) ===
    print(f"\n=== Phase 3: Intensive LP (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.35, 500.0)

    n_rounds = 5
    per_round = p3_budget / n_rounds
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        alloc = min(per_round, remaining * 0.2)
        if alloc < 25:
            break
        
        if rnd == 0:
            a_start = best_seq.copy()
        elif rnd == 1:
            # Slight asymmetric perturbation
            rng_r = np.random.default_rng(1000 + rnd)
            x = np.linspace(0, 1, n_current)
            bias = 1.0 + 0.05 * np.sin(np.pi * x)
            a_start = best_seq * bias
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        else:
            rng_r = np.random.default_rng(rnd * 7919 + n_current)
            scale = 0.02 + 0.015 * rnd
            a_start = best_seq * np.exp(rng_r.normal(0, scale, n_current))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()

        # Interleave subgradient and LP
        sg_alloc = alloc * 0.3
        lp_alloc = alloc * 0.7
        
        a_cur = a_start
        if sg_alloc > 5:
            a_cur = subgradient_optimize(a_cur, time_budget=sg_budget if False else sg_alloc, verbose=False)
        
        remaining = deadline - time.monotonic()
        lp_alloc2 = min(lp_alloc, remaining * 0.15)
        if lp_alloc2 > 15:
            a_cur = lp_alternate_improved(a_cur, time_budget=lp_alloc2, verbose=True)

        obj_cur = compute_obj(a_cur)
        marker = " **" if obj_cur < best_score else ""
        print(f"  Round {rnd}: obj={obj_cur:.8f}{marker}")
        if obj_cur < best_score:
            best_score = obj_cur
            best_seq = a_cur.copy()

    # === Phase 4: Frequency shaping + targeted (15%) ===
    print(f"\n=== Phase 4: Freq + Targeted ===")
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.15

    alloc_f = min(p4_budget * 0.4, 50.0)
    if alloc_f > 10:
        a_f = frequency_shaping(best_seq, time_budget=alloc_f, verbose=False)
        obj_f = compute_obj(a_f)
        marker = " **" if obj_f < best_score else ""
        print(f"  Freq: obj={obj_f:.8f}{marker}")
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()

    remaining = deadline - time.monotonic()
    alloc_t = min(remaining * 0.08, 50.0)
    if alloc_t > 10:
        a_t = targeted_refinement(best_seq, time_budget=alloc_t, verbose=False)
        obj_t = compute_obj(a_t)
        marker = " **" if obj_t < best_score else ""
        print(f"  Targeted: obj={obj_t:.8f}{marker}")
        if obj_t < best_score:
            best_score = obj_t
            best_seq = a_t.copy()

    # === Phase 5: Diverse restarts (12%) ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p5_budget = remaining * 0.12
    n_use = len(best_seq)
    n_restarts = 5
    per_restart = p5_budget / max(1, n_restarts)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        alloc = min(per_restart, remaining * 0.15)
        if alloc < 15:
            break

        mode = (r * 7 + 2) % 10
        a_r = diverse_init(n_use, rng, mode=mode)
        # Blend with best
        blend = 0.4 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        # Subgradient then LP
        half = alloc / 2
        if half > 5:
            a_r = subgradient_optimize(a_r, time_budget=half, verbose=False)
        remaining = deadline - time.monotonic()
        lp_alloc = min(remaining * 0.05, 30.0)
        if lp_alloc > 10:
            a_r = lp_alternate_improved(a_r, time_budget=lp_alloc, verbose=False)

        obj_r = compute_obj(a_r)
        marker = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}{marker}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()

    # === Phase 6: Final polish (8%) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    alloc = min(remaining * 0.4, 60.0)
    if alloc > 15:
        a_final = lp_alternate_improved(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Subgradient polish
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.3, 30.0)
    if alloc > 10:
        a_sg = subgradient_optimize(best_seq, time_budget=alloc, verbose=False)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # One last LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = lp_alternate_improved(best_seq, time_budget=min(remaining * 0.5, 20.0), verbose=False)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # Targeted refinement
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_targ = targeted_refinement(best_seq, time_budget=min(remaining * 0.5, 20.0), verbose=False)
        obj_targ = compute_obj(a_targ)
        if obj_targ < best_score:
            best_score = obj_targ
            best_seq = a_targ.copy()
            print(f"  Last Targeted: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"=== Target: 1.5027, Achieved: {best_score:.8f}, Gap: {best_score - 1.5027:.8f} ===")

    return result
