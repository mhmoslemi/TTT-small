import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a):
    """Solve: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None

def solve_lp_dual(a):
    """Solve dual: max sum(y) s.t. A^T y <= 1, y >= 0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    y = cp.Variable(2 * n - 1, nonneg=True)
    constraints = [A.T @ y <= 1]
    prob = cp.Problem(cp.Maximize(cp.sum(y)), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:

            if y.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

    return None

LP polishing with bisection line search, multi-length exploration, and
adaptive restart strategies. Key innovation: O(n log n) subgradient steps
using FFT-based convolution and vectorized subgradient computation, combined
with LP direction finding for final polishing.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    result = irfft(fa * fa, size)[:2 * n - 1]
    return np.maximum(result, 0.0)


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_vector(a, k_star):
    """
    Compute subgradient of max_k conv(a,a) at position k_star.
    d/d a[j] C_{k_star} = a[k_star - j] if 0 <= k_star - j < n
    """
    n = len(a)
    g = np.zeros(n)
    j_start = max(0, k_star - (n - 1))
    j_end = min(n - 1, k_star)
    if j_start <= j_end:
        indices = k_star - np.arange(j_start, j_end + 1)
        g[j_start:j_end + 1] = a[indices]
    return g


def multi_peak_subgradient(a, conv, top_k=5):
    """Compute a smoothed subgradient using the top-k peaks of the convolution."""
    n = len(a)
    # Get top-k indices
    if top_k >= len(conv):
        top_idx = np.arange(len(conv))
    else:
        top_idx = np.argsort(conv)[::-1][:top_k]
    
    # Weight by relative peak height (softmax-like)
    peak_vals = conv[top_idx]
    weights = peak_vals - peak_vals.min() + 1e-10
    weights = weights / weights.sum()
    
    # Accumulate subgradient vectors weighted
    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        j_start = max(0, int(k) - (n - 1))
        j_end = min(n - 1, int(k))
        if j_start <= j_end:
            indices = int(k) - np.arange(j_start, j_end + 1)
            g[j_start:j_end + 1] += w * a[indices]
    return g


def nesterov_subgradient(a, time_budget=150.0, top_k=5, verbose=False):
    """
    Nesterov-accelerated projected subgradient descent on the simplex.
    Uses multi-peak subgradient for smoother directions.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    f_lower = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound
    start = time.monotonic()

    it = 0
    a_prev = a.copy()
    v = np.zeros(n)
    beta = 0.9
    lr = 0.01
    best_it = 0
    patience = 50000
    last_improve_it = 0

    while time.monotonic() - start < time_budget:
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute convolution and subgradient
        conv = compute_conv(y)
        g = multi_peak_subgradient(y, conv, top_k=top_k)

        # Polyak-style adaptive step size
        f_y = 2.0 * n * float(np.max(conv))
        g_norm_sq = float(np.dot(g, g))
        if g_norm_sq < 1e-15:
            g_norm_sq = 1e-15
        # Adaptive lr: balance exploration and convergence
        polyak_lr = max(0.0, (f_y - f_lower - 0.01)) / (4.0 * n * g_norm_sq)
        lr = 0.7 * lr + 0.3 * min(max(polyak_lr, 1e-6), 0.05)
        lr = max(lr, 1e-8)

        # Update
        v = beta * v
        a_new = y - lr * g
        a_new = project_simplex(a_new)

        # Check improvement
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            last_improve_it = it

        a = a_new
        it += 1

        # Adaptive top_k: increase as we converge
        progress = min(1.0, it / 100000)
        top_k = 3 + int(7 * progress)

        # Patience-based restart
        if it - last_improve_it > patience:
            a = best_a.copy()
            v = np.zeros(n)
            lr = max(lr * 0.5, 1e-8)
            last_improve_it = it
            # Small perturbation
            rng = np.random.default_rng(it)
            a = best_a * np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 0)
            sa = float(np.sum(a))
            if sa > 1e-10:
                a = a / sa
            else:
                a = best_a.copy()
            last_improve_it = it

        if verbose and it % 50000 == 0:
            elapsed = time.monotonic() - start
            print(f"    NS it{it}: obj={best_obj:.8f}, lr={lr:.6f}, top_k={top_k}, time={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns the optimal g.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g_var = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g_var <= M, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_dual(a):
    """
    Solve dual LP: max sum(y) s.t. A^T y <= 1, y >= 0.
    Returns dual solution y.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    y_var = cp.Variable(2 * n - 1, nonneg=True)
    constraints = [A.T @ y_var <= 1]
    prob = cp.Problem(cp.Maximize(cp.sum(y_var)), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if y_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.maximum(y_var.value, 0)
        except Exception:
            continue
    return None


def bisection_line_search(a, g, n_evals=16):
    """
    Bisection line search for t in [0,1] minimizing obj((1-t)*a + t*g).
    Since obj is convex in t, bisection on the subgradient is effective.
    """
    best_obj = compute_obj(a)
    best_t = 0.0

    # Quick evaluations at key points
    candidates = [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    for t in candidates:
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t

    # Refine around best_t
    if best_t > 0 and best_t < 1:
        lo = max(0, best_t - 0.1)
        hi = min(1, best_t + 0.1)
        for _ in range(6):
            t1 = lo + (hi - lo) / 3
            t2 = hi - (hi - lo) / 3
            a1 = (1 - t1) * a + t1 * g
            a1 = np.maximum(a1, 0)
            s1 = float(np.sum(a1))
            f1 = compute_obj(a1 / s1) if s1 > 1e-10 else np.inf
            a2 = (1 - t2) * a + t2 * g
            a2 = np.maximum(a2, 0)
            s2 = float(np.sum(a2))
            f2 = compute_obj(a2 / s2) if s2 > 1e-10 else np.inf
            if f1 < f2:
                hi = t2
            else:
                lo = t1
            if f1 < best_obj:
                best_obj = f1
                best_t = t1
            if f2 < best_obj:
                best_obj = f2
                best_t = t2

    a_best = (1 - best_t) * a + best_t * g
    a_best = np.maximum(a_best, 0)
    s_best = float(np.sum(a_best))
    if s_best > 1e-10:
        a_best = a_best / s_best
    return a_best, best_obj


def lp_polish(a, time_budget=200.0, verbose=False):
    """
    LP-based polishing: solve LP for direction, do line search, iterate.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break

        # Line search
        a_new, obj_new = bisection_line_search(a, g_lp)

        it += 1
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 5 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it{it}: obj={best_obj:.8f}, M_lp={M_lp:.6f}, time={elapsed:.1f}s")
        else:
            # Try mixing LP direction with current
            for alpha in [0.3, 0.1, 0.05]:
                g_mix = (1 - alpha) * a + alpha * g_lp
                g_mix = np.maximum(g_mix, 0)
                sm = float(np.sum(g_mix))
                if sm > 1e-10:
                    g_mix = g_mix / sm
                    a_try, obj_try = bisection_line_search(a, g_mix, n_evals=8)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        stall = 0
                        break
            else:
                stall += 1
                # Perturb to escape
                if stall > 3:
                    a = best_a * np.exp(rng.normal(0, 0.02 + 0.005 * stall, n))
                    a = np.maximum(a, 0)
                    sa = float(np.sum(a))
                    if sa > 1e-10:
                        a = a / sa
                    else:
                        a = best_a.copy()
                    stall = 0
                if stall > 20:
                    break

    return best_a


def dual_guided_update(a, time_budget=100.0, verbose=False):
    """
    Use dual LP solution to guide multiplicative updates.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        y_dual = solve_lp_dual(a)
        if y_dual is None:
            break

        # Compute gradient from dual: grad[j] = sum_k y[k] * A[k,j]
        # A[k,j] = a_norm[k-j] if 0 <= k-j < n
        a_norm = a / np.sum(a)
        grad = np.zeros(n)
        for j in range(n):
            k_start = j
            k_end = min(2 * n - 2, j + n - 1)
            if k_start <= k_end:
                indices = np.arange(k_start, k_end + 1)
                a_indices = a_norm[indices - j]
                grad[j] = float(np.dot(y_dual[k_start:k_end + 1], a_indices))

        # Multiplicative update: a[j] <- a[j] * (1 - eta * grad[j] / max(grad))
        grad_max = float(np.max(grad))
        if grad_max < 1e-15:
            break
        eta = 0.1
        a_new = a * (1.0 - eta * grad / grad_max)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            it += 1
            if verbose and it % 3 == 0:
                elapsed = time.monotonic() - start
                print(f"    Dual it{it}: obj={best_obj:.8f}, time={elapsed:.1f}s")
        else:
            eta *= 0.5
            if eta < 1e-4:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Walk-like: accumulate random signs
        steps = rng.normal(0, 1, n)
        a = np.cumsum(steps) + abs(np.max(np.cumsum(steps))) + 1
        a = a / a.sum()
    else:
        # Periodic with noise
        periods = rng.integers(n // 10, n // 3, size=3)
        x = np.arange(n)
        a = np.ones(n)
        for p in periods:
            a += 0.5 * np.cos(2 * np.pi * x / p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    return a


def combined_polish(a, time_budget=100.0, verbose=False):
    """
    Alternate between subgradient and LP for maximum progress.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Subgradient burst (70% of remaining, capped at 60s)
        sg_budget = min(remaining * 0.7, 60.0)
        if sg_budget > 3:
            a_sg = nesterov_subgradient(a, time_budget=sg_budget, top_k=7)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_obj - 1e-12:
                best_obj = obj_sg
                best_a = a_sg.copy()
                a = a_sg
                if verbose:
                    print(f"    Combined SG: obj={best_obj:.8f}")
            a = best_a

        # LP step (30% of remaining, capped at 30s)
        lp_budget = min(remaining * 0.3, 30.0)
        if lp_budget > 3:
            a_lp = lp_polish(a, time_budget=lp_budget)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_obj - 1e-12:
                best_obj = obj_lp
                best_a = a_lp.copy()
                a = a_lp
                if verbose:
                    print(f"    Combined LP: obj={best_obj:.8f}")
            a = best_a

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization combining fast subgradient methods with LP polishing:
    1. Quick Nesterov subgradient on parent
    2. LP polishing with bisection line search
    3. Multi-length exploration with combined polishing
    4. Dual-guided updates
    5. Final intensive combined polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent available: {e}")

    if best_seq is None:
        n_target = 500
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Fast Nesterov Subgradient ===
    print(f"\n=== Phase 1: Nesterov Subgradient ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.2, 200.0)
    if p1_budget > 10:
        a_sg = nesterov_subgradient(best_seq, time_budget=p1_budget, top_k=5, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score - 1e-10:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  SG: obj={best_score:.8f} **")
        else:
            print(f"  SG: obj={obj_sg:.8f} (no improvement)")

    # === Phase 2: LP Polishing ===
    print(f"\n=== Phase 2: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 250.0)
    if p2_budget > 20:
        a_lp = lp_polish(best_seq, time_budget=p2_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score - 1e-10:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Multi-n Exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Try various lengths
    n_options = sorted(set([
        200, 300, 400, 500, 600, n_current,
        n_current + 100, n_current + 200, 1000
    ]))
    # Filter: reasonable range, different from current
    n_options = [n for n in n_options if 150 <= n <= 1500]
    # Prioritize lengths close to current and some far ones
    n_options.sort(key=lambda x: (abs(x - n_current), x))
    # Take a mix: nearest 2, and 2 far ones
    n_options_final = []
    if n_options:
        n_options_final.append(n_options[0])
    for n in n_options:
        if n not in n_options_final and abs(n - n_current) > 100:
            n_options_final.append(n)
        if len(n_options_final) >= 4:
            break

    for n_try in n_options_final[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        if n_try == n_current:
            continue

        a_try = resample_sequence(best_seq, n_try)

        # Quick combined polish
        alloc = min(remaining * 0.2, 80.0)
        if alloc > 10:
            a_try = combined_polish(a_try, time_budget=alloc)

        obj_try = compute_obj(a_try)
        if obj_try < best_score - 1e-10:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Diverse Restarts ===
    print(f"\n=== Phase 4: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = r % 7
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with current best
        for blend in [0.2, 0.4]:
            a_try = (1 - blend) * a_r + blend * best_seq
            a_try = np.maximum(a_try, 0)
            sa = float(np.sum(a_try))
            if sa > 1e-10:
                a_try = a_try / sa
                # Quick subgradient + short LP
                alloc = min(remaining * 0.15, 50.0)
                if alloc > 10:
                    a_try = nesterov_subgradient(a_try, time_budget=alloc * 0.6, top_k=5)
                    remaining = deadline - time.monotonic()
                    if remaining > 15:
                        a_try = lp_polish(a_try, time_budget=min(remaining * 0.2, 25.0))

                    obj_try = compute_obj(a_try)
                    if obj_try < best_score - 1e-10:
                        best_score = obj_try
                        best_seq = a_try.copy()
                        print(f"  Restart {r} (mode={mode}, blend={blend}): obj={best_score:.8f} **")
                        break
                    else:
                        print(f"  Restart {r} (mode={mode}, blend={blend}): obj={obj_try:.8f}")

    # === Phase 5: Dual-Guided Update ===
    print(f"\n=== Phase 5: Dual-Guided Update ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.2, 80.0)
        a_dual = dual_guided_update(best_seq, time_budget=alloc, verbose=True)
        obj_dual = compute_obj(a_dual)
        if obj_dual < best_score - 1e-10:
            best_score = obj_dual
            best_seq = a_dual.copy()
            print(f"  Dual: obj={best_score:.8f} **")
        else:
            print(f"  Dual: obj={obj_dual:.8f} (no improvement)")

    # === Phase 6: Final Intensive Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = combined_polish(best_seq, time_budget=min(remaining * 0.9, 200.0), verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score - 1e-10:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final: obj={best_score:.8f} **")
        else:
            print(f"  Final: obj={obj_final:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
