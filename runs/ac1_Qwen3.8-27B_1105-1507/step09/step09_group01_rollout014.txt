        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

    s = float(np.sum(a))

        return np.inf

    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_obj_and_grad(a, tau):
    """Compute smoothed objective and gradient using softmax over peaks."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    # Gradient of sum_k p_k * (a*a)[k] w.r.t. a
    # = sum_k p_k * 2 * flip_shift(a, k)
    # Using FFT: this is 2 * ifft(FFT(p) * conj(FFT(a))) restricted to first n
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    grad = 2.0 * g_full[:n]
    smoothed_val = max_conv + np.log(float(np.sum(exp_vals))) / tau
    return smoothed_val, grad, conv

def lp_polish_geometric(a, time_budget=100.0, seed=42, verbose=False):
    """LP polishing with geometric mean updates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    no_improve = 0
    alpha = 0.5  # Geometric blend parameter
    
    for it in range(200):
        if time.monotonic() - start >= time_budget:
            break
        
        M_current = float(np.max(compute_conv(best_a)))
        g_lp, M_lp = solve_lp_minpeak(best_a)
        if g_lp is None:
            break
        
        # Try geometric blend with different alpha values
        best_try_obj = best_obj
        best_try_a = best_a
        
        for alpha_try in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = geometric_blend(best_a, g_lp, 1.0 - alpha_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_try_obj:
                best_try_obj = obj_try
                best_try_a = a_try
        
        if best_try_obj < best_obj - 1e-14:
            best_obj = best_try_obj
            best_a = best_try_a.copy()
            no_improve = 0
            if verbose and it % 5 == 0:
                print(f"    LP-geo it{it}: obj={best_obj:.8f}, alpha={alpha_try:.2f}, t={time.monotonic()-start:.1f}s")
        else:
            no_improve += 1
            if no_improve >= 5:
                # Perturb
                a = best_a * np.exp(rng.normal(0, 0.01, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                if it > 50:
                    break
    
    return best_a

def peak_shave_iterate(a, time_budget=50.0, lr_init=0.05, verbose=False):
    """Iterative peak shaving with adaptive learning rate."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    stall = 0
    
    for it in range(5000):
        if time.monotonic() - start >= time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Compute gradient at peak
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        a_new = a - lr * grad
        a_new = project_simplex(a_new)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = best_a.copy()
            stall += 1
            if stall % 10 == 0:
                lr *= 0.7
                if lr < 1e-8:
                    break
    
    return best_a

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Arc-sine-like
        x = np.linspace(0.01, 0.99, n)
        a = 1.0 / np.sqrt(x * (1 - x) + 1e-6)
        a = a / a.sum()
    elif mode == 3:
        # Cosine
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 4:
        # U-shaped
        x = np.linspace(0, 1, n)
        a = x**2 + (1-x)**2
        a = a / a.sum()
    elif mode == 5:
        # W-shaped
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
