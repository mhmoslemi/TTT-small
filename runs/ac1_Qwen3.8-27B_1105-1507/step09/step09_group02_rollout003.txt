def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def variance_grad_fft(a):
    """Gradient of sum_k ((a*a)[k] - mean)^2, normalized to sum(a)=1."""
    n = len(a)
    conv = compute_conv(a)
    c = np.mean(conv)
    excess = conv - c
    # grad[m] = 4 * sum_k excess[k] * a[k-m]
    # = 4 * (excess * flip(a))[n-1+m]
    b = a[::-1]
    N = 1
    while N < len(excess) + n - 1:
        N <<= 1
    E = fft(excess, N)
    B = fft(b, N)
    cross_full = ifft(E * B).real
    cross = cross_full[n-1:n-1+n]
    return 4.0 * cross

def lse_grad_fft(a, tau):
    """gradient of (1/tau)*log(sum exp(tau*conv))"""

    P = fft(p, N)
    B = fft(b, N)
    cross_full = ifft(P * B).real
    cross = cross_full[n-1:n-1+n]
    return 2.0 * cross, conv

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def solve_direction_lp(a):
    """

    n = len(a)
    a_norm = a / np.sum(a)
    conv_aa = compute_conv(a_norm)
    A = build_conv_matrix(a_norm)
    
    d = cp.Variable(n)
    M = cp.Variable()
    constraints = [
        A @ d + conv_aa <= M,
        cp.sum(d) == 0,
        d >= -0.5,
        d <= 0.5
    ]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]

            if d.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
