```python
import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
FFT-based warmup, adaptive blending, multi-length exploration,
and perturbative restarts.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = int(rho[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(j + n, 2 * n - 1)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp(a, M_target=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_value) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.PDP, cp.MOSEK, cp.CLARABEL, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue
    
    return None, None


def smoothed_gd(a, time_budget=30.0, tau_init=5.0, tau_end=200.0):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)
        
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient via FFT cross-correlation
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(y, N)
        g_full = ifft(P * np.conj(A)).real
        grad = 2.0 * g_full[:n]
        
        # Line search (Barzilai-Borwein)
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = grad - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))
        
        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        prev_a = a.copy()
        prev_g = grad.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)
        
        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            best_it = it
    
    return best_a


def lp_polish(a, time_budget=100.0, seed=42, verbose=False):
    """
    LP-based polishing with adaptive blend factor and perturbative restarts.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    iter_count = 0
    no_improve = 0
    t_blend = 0.3
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        
        # Solve LP
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break
        
        # Try blend
        a_try = (1.0 - t_blend) * best_a + t_blend * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            t_blend = max(0.05, t_blend * 0.5)
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        
        iter_count += 1
        
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a.copy()
            no_improve = 0
            # Increase blend factor slightly
            t_blend = min(0.8, t_blend * 1.05)
            if verbose and iter_count % 10 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={t_blend:.3f}, M={M_lp:.6f}, el={elapsed:.0f}s")
        else:
            no_improve += 1
            # Try smaller step
            t_small = t_blend * 0.3
            a_try2 = (1.0 - t_small) * best_a + t_small * g_lp
            a_try2 = np.maximum(a_try2, 0)
            if a_try2.sum() > 1e-10:
                a_try2 = a_try2 / a_try2.sum()
                obj_try2 = compute_obj(a_try2)
                if obj_try2 < best_obj - 1e-14:
                    best_obj = obj_try2
                    best_a = a_try2.copy()
                    a = best_a.copy()
                    t_blend = t_small
                    no_improve = 0
                    continue
            
            if no_improve >= 3:
                # Perturb
                scale = 0.01 * min(no_improve, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                t_blend = 0.3
                no_improve = 0
                if iter_count > 50:
                    break
            else:
                t_blend = max(0.05, t_blend * 0.7)
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 4) * x + rng.uniform(0, 6.28))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = 1.0 / (x ** rng.uniform(0.2, 0.8) + 1.0 / ((1-x) ** rng.uniform(0.2, 0.8) + 1e-10))
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
    elif mode == 7:
        # Bell-shaped
        c = n // 2
        w = n // 6
        a = np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a
```

```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses intensive LP-based alternating minimization with PDP/MOSEK solvers,
FFT-based smoothed gradient warmup, adaptive blend scheduling, multi-n
exploration with diverse initializations, and perturbative LP restarts.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = int(rho[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(j + n, 2 * n - 1)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp(a, M_target=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_value) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.PDP, cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            dt = time.monotonic() - t0
            if dt > 60:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue

    return None, None


def smoothed_gd(a, time_budget=30.0, tau_init=5.0, tau_end=200.0):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget * 0.9, 1e-6))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient via log-sum-exp
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        # Gradient via FFT cross-correlation
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(y, N)
        g_full = ifft(P * np.conj(A)).real
        grad = 2.0 * g_full[:n]

        # Barzilai-Borwein step size
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = grad - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = grad.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def lp_polish(a, time_budget=100.0, seed=42, verbose=False):
    """
    LP-based polishing with adaptive blend factor and perturbative restarts.
    Uses the LP solution as a direction to move towards, with adaptive step.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    no_improve = 0
    t_blend = 0.4
    last_improve_iter = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 2:
            break

        # Solve LP
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        # Adaptive blend: try increasing if making progress, decreasing if not
        if no_improve == 0:
            t_blend = min(0.9, t_blend * 1.08)
        elif no_improve == 1:
            t_blend = max(0.1, t_blend * 0.6)
        else:
            t_blend = max(0.05, t_blend * 0.5)

        a_try = (1.0 - t_blend) * best_a + t_blend * g_lp
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            t_blend = max(0.05, t_blend * 0.5)
            no_improve += 1
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)

        iter_count += 1

        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a.copy()
            no_improve = 0
            last_improve_iter = iter_count
            if verbose and iter_count % 5 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={t_blend:.3f}, M={M_lp:.6f}, el={elapsed:.0f}s")
        else:
            no_improve += 1
            # Try a smaller step
            t_small = max(0.02, t_blend * 0.3)
            a_try2 = (1.0 - t_small) * best_a + t_small * g_lp
            a_try2 = np.maximum(a_try2, 0)
            if a_try2.sum() > 1e-10:
                a_try2 = a_try2 / a_try2.sum()
                obj_try2 = compute_obj(a_try2)
                if obj_try2 < best_obj - 1e-14:
                    best_obj = obj_try2
                    best_a = a_try2.copy()
                    a = best_a.copy()
                    t_blend = t_small
                    no_improve = 0
                    continue

            # Try the LP solution directly (t=1) occasionally
            if no_improve == 2 and iter_count % 10 == 0:
                obj_direct = compute_obj(g_lp)
                if obj_direct < best_obj - 1e-14:
                    best_obj = obj_direct
                    best_a = g_lp.copy()
                    a = best_a.copy()
                    t_blend = 0.5
                    no_improve = 0
                    continue

            if no_improve >= 4:
                # Perturb to escape local optimum
                scale = 0.008 * min(no_improve, 6)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                t_blend = 0.4
                no_improve = 0
                if iter_count - last_improve_iter > 80:
                    break
            elif no_improve >= 2:
                t_blend = max(0.05, t_blend * 0.7)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x * x + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        f = rng.uniform(1.5, 3.5)
        p = rng.uniform(0, 6.28)
        a = 1.0 + 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        alpha = rng.uniform(0.3, 1.2)
        a = x ** alpha
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
    elif mode == 7:
        c = n // 2
        w = max(1, n // 6)
        a = np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 8:
        # Triangular (inverted) - mass at edges
        x = np.linspace(0, 1, n)
        a = np.abs(x - 0.5) * 2.0 + 0.1
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def n_dist_key(n_val, current_n):
    """Helper for sorting n options by distance from current."""
    return abs(n_val - current_n)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Intensive LP-based alternating minimization with FFT warmup,
    adaptive blending, multi-n exploration, diverse initializations, and
    perturbative restarts. Targets the minimum of 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Quick warmup ===
    print(f"\n=== Phase 1: Warmup ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 60.0)

    if p1_budget > 10:
        a_warm = smoothed_gd(best_seq, time_budget=p1_budget)
        obj_warm = compute_obj(a_warm)
        if obj_warm < best_score:
            best_score = obj_warm
            best_seq = a_warm.copy()
            print(f"  Warmup: obj={best_score:.8f} **")
        else:
            print(f"  Warmup: obj={obj_warm:.8f}")

    # Try a couple diverse inits at same n
    remaining = deadline - time.monotonic()
    if remaining > 30:
        per_init = min(remaining * 0.08, 15.0)
        for mode in [3, 7]:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            a_try = diverse_init(n_current, rng, mode=mode)
            a_try = smoothed_gd(a_try, time_budget=per_init)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Init mode {mode}: obj={best_score:.8f} **")
            else:
                print(f"  Init mode {mode}: obj={obj_try:.8f}")

    # === Phase 2: Intensive LP Polishing (main workhorse) ===
    print(f"\n=== Phase 2: Intensive LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.45, 550.0)
    if p2_budget > 30:
        a_lp = lp_polish(best_seq, time_budget=p2_budget, seed=42, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            n_current = len(best_seq)
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f}")

    # === Phase 3: Multi-n Exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()

    n_candidates = set()
    n_candidates.add(n_current)
    for delta in [-400, -200, -100, 100, 200, 400, 600]:
        n_cand = n_current + delta
        if 200 <= n_cand <= 2500:
            n_candidates.add(n_cand)
    for n_fixed in [500, 700, 900, 1100, 1300, 1500, 1800]:
        if 200 <= n_fixed <= 2500:
            n_candidates.add(n_fixed)

    n_candidates = [n for n in n_candidates if abs(n - n_current) > 80]
    # Sort by distance from current
    n_sorted = sorted(n_candidates, key=lambda x: abs(x - n_current))
    # Replace lambda with explicit approach
    n_dist_pairs = [(abs(n - n_current), n) for n in n_candidates]
    n_dist_pairs.sort()
    n_options = [n for _, n in n_dist_pairs]

    explored = 0
    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 70:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)

        # Quick warmup
        alloc1 = min(remaining * 0.12, 25.0)
        if alloc1 > 5:
            a_try = smoothed_gd(a_try, time_budget=alloc1)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.30, 130.0)
        if alloc2 > 20:
            a_try = lp_polish(a_try, time_budget=alloc2, seed=n_try + 100)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")
        explored += 1

    # === Phase 4: Perturbation + Re-LP ===
    print(f"\n=== Phase 4: Perturbation + Re-LP ===")
    remaining = deadline - time.monotonic()

    for r in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        # Vary perturbation scale
        scale = 0.006 + 0.005 * r
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc = min(remaining * 0.28, 110.0)
        if alloc > 20:
            a_pert = lp_polish(a_pert, time_budget=alloc, seed=300 + r)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Perturb {r} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Perturb {r} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 5: Final intensive LP ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.75, 300.0)
        a_final = lp_polish(best_seq, time_budget=alloc, seed=999, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # Final quick gradient polish
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.4, 30.0)
        a_sg = smoothed_gd(best_seq, time_budget=alloc, tau_init=50.0, tau_end=500.0)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
