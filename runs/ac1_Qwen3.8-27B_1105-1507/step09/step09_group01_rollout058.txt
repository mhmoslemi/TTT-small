"""
Multi-strategy optimizer for step function coefficient minimization.
Uses: (1) Fienup-style frequency domain iteration for exploration,
(2) Fast O(n) subgradient method with BB step size and multi-peak weighting,
(3) Coordinate descent polishing, (4) LP-based final polish,
(5) Multi-length and multi-seed exploration.
"""

import numpy as np
import time
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fienup_iterate(a, N, alpha, max_iter, time_budget):
    """Fienup/Gerchberg-Saxton style iteration."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = a.sum()
    if s < 1e-10: return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break
        
        A = fft(a, N)
        P = np.abs(A)**2
        P_max = P.max()
        if P_max < 1e-15:
            break
        P_norm = P / P_max
        P_comp = P_norm ** alpha
        # Scale to preserve DC (sum)
        if P_comp.sum() > 0:
            P_new = P_comp * (P.sum() / P_comp.sum())
        else:
            P_new = P.copy()
        
        # Reconstruct with same phase
        phase = np.angle(A)
        magnitude = np.sqrt(np.maximum(P_new, 0))
        A_new = magnitude * np.exp(1j * phase)
        
        a_new = ifft(A_new, N).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
    
    return best_a

    it = 0

    prev_dir = None

        m = float(conv.max())

def lp_polish(a, time_budget, verbose=False):
    """LP-based polishing."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(77)
    
    iter_count = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        a_norm = best_a  # already normalized to sum=1
        A = build_conv_matrix_dense(a_norm)
        
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        solved = False
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        g_val = g_val / s_g
                        solved = True
                        break
            except Exception:
                continue
        
        if not solved:
            break
        
        # Try blending
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1-t) * best_a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                no_improve = 0
                break
        else:
            no_improve += 1
            if no_improve >= 5:
                a_pert = best_a * np.exp(rng.normal(0, 0.01, n))
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                # Check if perturbation helps
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                no_improve = 0
                if iter_count > 30:
                    break
        
        iter_count += 1
        if verbose and iter_count % 3 == 0:
            print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

Actually, I just realized something important. Let me reconsider the problem.

We're minimizing 2n * max_k (a*a)_k / (sum a)^2.

For a sequence of length n, the convolution has 2n-1 elements. The maximum is achieved at some lag k*.

If the sequence is symmetric (a[i] = a[n-1-i]), then the convolution is also symmetric: (a*a)_k = (a*a)_{2n-2-k}. And the center (a*a)_{n-1} = sum a[i]^2.

For a symmetric sequence, we need sum a[i]^2 to be small relative to (sum a)^2 / n. By Cauchy-Schwarz, sum a[i]^2 >= (sum a)^2 / n, with equality iff a is constant.

So for symmetric sequences, the minimum of 2n * max_k (a*a)_k / (sum a)^2 is at least 2, achieved by the constant sequence.

But we can do better with asymmetric sequences! The key is that for asymmetric sequences, the center of the convolution (a*a)_{n-1} = sum a[i]*a[n-1-i] is NOT sum a[i]^2. It can be much smaller.

For example, if a is concentrated at the beginning (a[0] ~ 1, rest ~ 0), then (a*a)_{n-1} ~ 0, but (a*a)_0 ~ 1, so the max is at the edge.

The optimal strategy is to spread the "energy" in a way that all convolution values are small. This is the "low peak to average ratio" (PAPR) problem.

For the continuous case with non-negativity, the optimal solution is believed to have a specific structure related to the "Proakis" or "WKS" sequence designs, but I don't have a closed form.

"""
Multi-strategy optimizer minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines: (1) Fienup-style frequency domain iteration, (2) Fast O(n) subgradient
with Barzilai-Borwein step size and multi-peak weighting, (3) Stochastic
coordinate descent, (4) LP-based final polish, (5) Multi-length exploration
with resampling. The O(n) subgradient exploits the fact that the gradient of
max_k conv(a,a)_k only involves 2*a[k*-i] at the active peak lags.
"""

import numpy as np

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
