"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Fast LP polishing with analytical optimal blending, active set reduction,
coordinate descent fine-tuning, and multi-length exploration.
Key innovations: analytical t computation from LP solution, active set tracking
to reduce LP size, and coordinate descent for sub-LP-scale improvements.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix_sparse(a, active_indices=None):
    """
    Build the convolution matrix for active constraints only.
    Returns a sparse-ish representation for the LP.
    If active_indices is None, use all 2n-1 constraints.
    """
    n = len(a)
    if active_indices is None:
        active_indices = np.arange(2 * n - 1)
    
    K = len(active_indices)
    # For each active constraint k, the coefficients are a[k-j] for valid j
    # A[k_idx, j] = a[k - j] if 0 <= k-j < n, else 0
    A = np.zeros((K, n))
    for idx, k in enumerate(active_indices):
        k = int(k)
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[idx, j_min:j_max+1] = a[k-j_max:k-j_min+1][::-1]
    return A, active_indices


def solve_lp(a, active_indices=None, solver=cp.MOSEK):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for active k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    A, active_idx = build_conv_matrix_sparse(a_norm, active_indices)
    K = A.shape[0]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    try:
        prob.solve(solver=solver)
        if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
            g_val = np.maximum(g.value, 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(M.value)
    except Exception:
        pass
    
    # Try fallback solvers
    for s2 in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=s2)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def optimal_t(P_a, M_ag, P_g):
    """
    Compute optimal blending parameter t analytically.
    f(t) = P_a*(1-t)^2 + 2*M_ag*t*(1-t) + P_g*t^2
    (Note: M_ag is max of conv(a,g), P_g is max of conv(g,g))
    Returns t in [0, 1].
    """
    # f(t) = P_a + t*(2*M_ag - 2*P_a) + t^2*(P_a - 2*M_ag + P_g)
    A_coeff = P_a
    B_coeff = 2 * (M_ag - P_a)
    C_coeff = P_a - 2*M_ag + P_g
    
    if C_coeff <= 1e-15:
        # Linear or concave: minimum at boundary
        if B_coeff >= 0:
            return 0.0
        else:
            return 1.0
    
    t_star = -B_coeff / (2.0 * C_coeff)
    t_star = max(0.0, min(1.0, t_star))
    return t_star

def coordinate_descent(a, time_budget, verbose=False):
    """
    Coordinate descent: for each coordinate, find optimal step.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_count = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        current_max = float(np.max(conv))
        
        # Process coordinates in random order
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
                
            # Try modifying coordinate i
            # conv(a + delta*e_i, a + delta*e_i)[k] = c[k] + 2*delta*a[k-i] + delta^2*1_{k=2i}
            # We want to find delta that minimizes the new max
            
            # The new max is max over k of (c[k] + 2*delta*a[|k-i| mod something])
            # Actually: c_new[k] = c[k] + 2*delta*a[k-i] (if 0<=k-i<n) + delta^2 (if k==2i)
            
            # For small delta, the dominant effect is 2*delta*a[k-i]
            # If a[k-i] > 0 and c[k] is near the max, increasing delta increases c[k]
            # So we should try negative delta (decrease a[i])
            
            # Try a small decrease
            alpha = a[i]
            for frac in [0.1, 0.25, 0.5]:
                delta = -frac * alpha
                if a[i] + delta < -1e-10:
                    continue
                a_new = a.copy()
                a_new[i] = max(0, a[i] + delta)
                
                # Quick objective check using the formula
                # c_new[k] = c[k] + 2*delta*a[k-i] for valid k, + delta^2 for k=2i
                # But this is O(n) to compute, which is fine for n=800
                
                # Actually let's just compute the max efficiently
                # The change in c[k] is 2*delta*a[k-i] for k in [i, i+n-1] (clamped to [0, 2n-2])
                # and delta^2 for k=2i (if 2i < 2n-1)
                
                k_lo = max(0, i)
                k_hi = min(2*n-2, i+n-1)
                j_lo = k_lo - i  # = 0
                j_hi = k_hi - i  # = n-1 or less
                delta_c = 2 * delta * a[j_lo:j_hi+1]
                c_new_slice = conv[k_lo:k_hi+1] + delta_c
                new_max = max(float(np.max(c_new_slice)), 
                              float(np.max(conv[:k_lo])) if k_lo > 0 else 0,
                              float(np.max(conv[k_hi+1:])) if k_hi+1 < 2*n-1 else 0)
                if 2*i < 2*n-1:
                    new_max = max(new_max, conv[2*i] + delta_c[2*i-i] + delta*delta)
                
                s_new = s + delta
                if s_new < 0.01:
                    continue
                obj_new = 2.0 * n * new_max / (s_new * s_new)
                
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    s = s_new
                    conv = compute_conv(a)  # Update for next coordinate
                    improved = True
                    break  # Restart frac loop for this i
        
        round_count += 1
        if not improved:
            break
        if verbose and round_count % 5 == 0:
            print(f"    CoordDesc round {round_count}: obj={best_obj:.10f}")
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def smoothed_gd(a, time_budget, tau=100.0, verbose=False):
    """Smoothed gradient descent with log-sum-exp approximation of max."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 5.0 / np.sqrt(n)
    momentum = 0.9
    v = np.zeros(n)
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        # Soft max
        shifted = tau * (conv - max_c)
        exp_vals = np.exp(shifted)
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient of soft max: grad_i = 2 * sum_k p[k] * a[k-i mod stuff]
        # More precisely: d/d a[i] of sum_k p[k]*c[k] where c = a*a
        # = 2 * sum_k p[k] * (dp/d a[i] term + c gradient)
        # Approximate: grad_i ≈ 2 * sum_k p[k] * a[|k-i|] (for the parts where it's valid)
        # Using FFT:
        from scipy.fft import fft, ifft
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]
        
        # Projected update
        v = momentum * v - lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        # Projected gradient actually should project onto simplex
        # For now, the normalize step is a rough projection
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                lr *= 0.7
                v *= 0.5
                stall = 0
                if lr < 1e-12:
                    break
    
    return best_a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Fast LP polishing with analytical optimal blending parameter, active set
reduction for speed, coordinate descent for sub-LP-scale improvements, and
multi-length exploration. Key innovations: closed-form t from LP solution,
adaptive active set tracking, and structured perturbation for escaping local minima.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a, active_indices=None):
    """
    Build convolution matrix rows for given active constraint indices.
    A[idx, j] = a[k - j] if 0 <= k-j < n, else 0, where k = active_indices[idx].
    """
    n = len(a)
    if active_indices is None:
        active_indices = np.arange(2 * n - 1)
    K = len(active_indices)
    A = np.zeros((K, n))
    for idx, k in enumerate(active_indices):
        k = int(k)
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[idx, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A


def solve_lp(a, active_indices=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for active k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm, active_indices)
    K = A.shape[0]
    if K == 0:
        return None, None

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def optimal_t(P_a, M_ag, P_g):
    """
    Analytically compute optimal blending t.
    The peak of conv((1-t)*a + t*g, (1-t)*a + t*g) is bounded by:
    f(t) = P_a*(1-t)^2 + 2*M_ag*t*(1-t) + P_g*t^2
    Minimize this quadratic in t over [0,1].
    """
    A_c = P_a
    B_c = 2.0 * (M_ag - P_a)
    C_c = P_a - 2.0 * M_ag + P_g

    if abs(C_c) < 1e-18:
        if B_c >= 0:
            return 0.5
        return 1.0

    t_star = -B_c / (2.0 * C_c)
    t_star = max(0.01, min(0.99, t_star))
    return t_star


def lp_polish_fast(a, time_budget, verbose=False, top_k=50):
    """
    Fast LP polishing with analytical optimal t and active set reduction.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    rng = np.random.default_rng(12345 + n)
    consecutive_fails = 0
    last_t = 0.5
    lp_time_total = 0.0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed > time_budget * 0.95:
            break

        # Determine active set: use top peaks after first few iterations
        if it < 3:
            active_idx = None
        else:
            conv_curr = compute_conv(a)
            active_idx = np.argsort(conv_curr)[::-1][:top_k]

        t0 = time.monotonic()
        g, M_lp = solve_lp(a, active_idx)
        lp_time_total += time.monotonic() - t0

        if g is None:
            consecutive_fails += 5
            if consecutive_fails > 30:
                break
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            continue

        # Compute values for optimal t
        conv_a = compute_conv(a)
        P_a = float(np.max(conv_a))
        conv_g = compute_conv(g)
        P_g = float(np.max(conv_g))

        t_opt = optimal_t(P_a, M_lp, P_g)

        # Build candidate t values
        t_candidates = [t_opt]
        t_candidates.append(max(0.02, t_opt * 0.6))
        t_candidates.append(min(0.98, t_opt * 1.4))
        t_candidates.append(last_t)
        t_candidates.append(1.0)
        t_candidates = sorted(set([max(0.01, min(0.99, round(t, 4))) for t in t_candidates]))

        best_t_obj = best_obj + 1e-13
        best_t_val = last_t
        best_a_local = best_a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            consecutive_fails = 0
            if verbose and (it % 20 == 0 or it <= 5):
                print(f"    LP it{it}: obj={best_obj:.10f}, t={last_t:.4f}, M={M_lp:.8f}, lp_avg={lp_time_total/it:.4f}s")
        else:
            consecutive_fails += 1
            scale = 0.003 + 0.001 * min(consecutive_fails, 50)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            if consecutive_fails > 20:
                scale = 0.02 + 0.005 * min(consecutive_fails - 20, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                consecutive_fails = 0
            if consecutive_fails > 80:
                if verbose:
                    print(f"    LP: breaking after {consecutive_fails} stalls")
                break

    return best_a


def coordinate_descent(a, time_budget, verbose=False):
    """
    Coordinate descent: for each coordinate, try decreasing it by a fraction.
    Uses the analytical formula for how conv changes with a single coordinate.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_count = 0
    rng = np.random.default_rng(777)
    no_improve_rounds = 0

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 2:
            break
        conv = compute_conv(a)
        current_max = float(np.max(conv))
        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - start > time_budget - 1:
                break
            if a[i] < 1e-10:
                continue

            alpha = a[i]
            for frac in [0.05, 0.15, 0.3, 0.5]:
                delta = -frac * alpha
                new_val = a[i] + delta
                if new_val < -1e-12:
                    continue

                # Compute new max using the formula
                # c_new[k] = c[k] + 2*delta*a[k-i] for k in [i, i+n-1]
                # plus delta^2 at k=2i
                k_lo = max(0, i)
                k_hi = min(2 * n - 2, i + n - 1)
                j_lo = k_lo - i
                j_hi = k_hi - i
                delta_c = 2.0 * delta * a[j_lo:j_hi + 1]
                c_new_slice = conv[k_lo:k_hi + 1] + delta_c

                # Handle k=2i extra term
                if 2 * i < 2 * n - 1 and 2 * i >= k_lo and 2 * i <= k_hi:
                    idx_2i = 2 * i - k_lo
                    c_new_slice[idx_2i] += delta * delta

                new_max = float(np.max(c_new_slice))
                # Check parts outside [k_lo, k_hi] (unchanged)
                if k_lo > 0:
                    new_max = max(new_max, float(np.max(conv[:k_lo])))
                if k_hi + 1 < 2 * n - 1:
                    new_max = max(new_max, float(np.max(conv[k_hi + 1:])))

                s_new = s + delta
                if s_new < 0.01:
                    continue
                obj_new = 2.0 * n * new_max / (s_new * s_new)

                if obj_new < best_obj - 1e-15:
                    a[i] = new_val
                    s = s_new
                    best_obj = obj_new
                    best_a = a.copy()
                    conv = compute_conv(a)
                    improved = True
                    break

        round_count += 1
        if improved:
            no_improve_rounds = 0
        else:
            no_improve_rounds += 1
            if no_improve_rounds >= 3:
                break
        if verbose and round_count % 10 == 0:
            print(f"    CD round {round_count}: obj={best_obj:.10f}")

    return best_a


def smoothed_gd(a, time_budget, tau_init=50.0, tau_end=500.0, verbose=False):
    """Smoothed gradient descent with log-sum-exp approximation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 4.0 / np.sqrt(n)
    momentum = 0.9
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        conv = compute_conv(a)
        max_c = float(np.max(conv))
        shifted = tau * (conv - max_c)
        exp_vals = np.exp(shifted)
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        v = momentum * v - lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall % 30 == 0:
                lr *= 0.8
                v *= 0.5
            if stall > 300:
                break

        it += 1
        if verbose and it % 200 == 0:
            print(f"    GD it{it}: obj={best_obj:.10f}, tau={tau:.1f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * 3 * x + rng.uniform(0, 6))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 3:
        a = np.zeros(n)
        for _ in range(3):
            c = rng.integers(0, n)
            w = rng.integers(n // 8, n // 3)
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w * w))
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1.5, 1.5)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP polishing with analytical t + coordinate descent + smoothed GD
    + multi-n exploration. Time allocation: LP 50%, CD 15%, GD 10%, multi-n 15%,
    final polish 10%.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            par_s = float(np.sum(parent_arr))
            if par_s > 0.01:
                parent = parent_arr / par_s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception:
        pass

    if best_seq is None:
        n_target = 800
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.10f}")

    # Also try a few diverse inits to establish a baseline
    for mode in [1, 2, 3]:
        a_try = diverse_init(len(best_seq), rng, mode)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  Diverse init mode {mode}: obj={best_score:.10f} **")

    # === Phase 1: Intensive LP Polishing (50%) ===
    print(f"\n=== Phase 1: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.5, 650.0)
    if p1_budget > 60:
        num_rounds = 5
        per_round = p1_budget / num_rounds
        for rnd in range(num_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(per_round, remaining * 0.5)
            if alloc < 30:
                break
            a_lp = lp_polish_fast(best_seq, time_budget=alloc, verbose=True, top_k=60)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score - 1e-14:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd + 1}: obj={best_score:.10f} **")
            else:
                print(f"  LP round {rnd + 1}: obj={obj_lp:.10f}")
                # Structured perturbation for next round
                if rnd < num_rounds - 1:
                    scale = 0.008 * (rnd + 1)
                    noise = rng.normal(0, scale, len(best_seq))
                    # Add a gentle sinusoidal component
                    x = np.linspace(0, 4 * np.pi, len(best_seq))
                    noise += 0.005 * np.sin(x + rng.uniform(0, 6))
                    best_seq = best_seq * np.exp(noise)
                    best_seq = np.maximum(best_seq, 0)
                    best_seq = best_seq / best_seq.sum()

    # === Phase 2: Coordinate Descent (15%) ===
    print(f"\n=== Phase 2: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 180.0)
    if p2_budget > 20:
        a_cd = coordinate_descent(best_seq, time_budget=p2_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score - 1e-14:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CoordDesc: obj={best_score:.10f} **")
        else:
            print(f"  CoordDesc: obj={obj_cd:.10f}")

    # === Phase 3: Smoothed GD (10%) ===
    print(f"\n=== Phase 3: Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 120.0)
    if p3_budget > 15:
        a_gd = smoothed_gd(best_seq, time_budget=p3_budget, tau_init=50.0, tau_end=500.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score - 1e-14:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  SmoothedGD: obj={best_score:.10f} **")
        else:
            print(f"  SmoothedGD: obj={obj_gd:.10f}")

    # === Phase 4: Multi-n Exploration (15%) ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_candidates = []
    for offset in [-200, -100, 100, 200, 300, 500]:
        n_try = n_current + offset
        if 400 <= n_try <= 1500 and n_try != n_current:
            n_candidates.append(n_try)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        a_try = resample_sequence(best_seq, n_try)
        # Quick LP polish
        alloc = min(remaining * 0.3, 50.0)
        if alloc > 20:
            a_try = lp_polish_fast(a_try, time_budget=alloc, top_k=40)
        obj_try = compute_obj(a_try)
        if obj_try < best_score - 1e-14:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 5: Final Intensive Polish (remaining) ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP with smaller active set for finer tuning
    if remaining > 30:
        alloc = min(remaining * 0.35, 100.0)
        a_lp = lp_polish_fast(best_seq, time_budget=alloc, verbose=True, top_k=30)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score - 1e-14:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP (top30): obj={best_score:.10f} **")

    # Coordinate descent again
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 50.0)
        a_cd = coordinate_descent(best_seq, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score - 1e-14:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.10f} **")

    # One more LP pass
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 40.0)
        a_lp = lp_polish_fast(best_seq, time_budget=alloc, top_k=25)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score - 1e-14:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP2: obj={best_score:.10f} **")

    # Final CD
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.7, 20.0)
        a_cd = coordinate_descent(best_seq, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score - 1e-14:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD2: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
