def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # conv(a, g)[k] = sum_i a[i] * g[k-i]
        # For fixed g[j], it contributes to conv[k] where k = i + j, i.e., k - j = i
        # So for g[j], the coefficient in conv[k] is a[k-j] (when 0 <= k-j < n)
        r_start = j
        r_end = min(2 * n - 1, j + n)
        for k in range(r_start, r_end):
            A[k, j] = a[k - j]
    return A


def solve_lp_minmax(a, solver_list=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    import cvxpy as cp
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build the convolution matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        A[k_start:k_end, j] = a_norm[:k_end - k_start]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(M.value)
        except Exception:

    return None, None


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(a, solver_list=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    import cvxpy as cp
    n = len(a)
    a_norm = a / np.sum(a)

    # Build convolution matrix: (A @ g)[k] = sum_j a_norm[k-j] * g[j] (valid j)
    # = conv(a_norm, g)[k]
    # Matrix A has shape (2n-1, n), A[k,j] = a_norm[k-j] if 0<=k-j<n else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_lo = j
        k_hi = min(2 * n - 2, j + n - 1)
        A[k_lo:k_hi + 1, j] = a_norm[:k_hi - k_lo + 1]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]

    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(M.value)
        except Exception:
            continue
    return None, None


def nesterov_smoothed_descent(a, time_budget=200.0, tau_init=20.0, tau_end=500.0,
                               beta=0.9, lr_scale=1.0, verbose=False):
    """
    Nesterov accelerated gradient descent on smoothed objective over simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = lr_scale / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    patience = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget * 0.95, 1e-6))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            v = np.zeros(n)
        else:
            y = y / s_y

        grad, conv = smoothed_gradient(y, tau)

        # Barzilai-Borwein adaptive step size
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = grad - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                bb_lr = min(max(bb_lr, 1e-8), 5.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb_lr

        # Gradient step
        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
            lr = max(1e-8, lr * 0.5)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            patience = 0
        else:
            patience += 1
            if patience > 500:
                # Restart from best with smaller lr
                a = best_a.copy()
                v = np.zeros(n)
                lr = max(1e-8, lr * 0.5)
                patience = 0

        prev_a = a.copy()
        prev_g = grad.copy()
        a = a_new
        it += 1

        if verbose and it % 2000 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")

    return best_a


def subgradient_descent(a, time_budget=100.0, lr_scale=0.5, verbose=False):
    """
    Subgradient descent on the non-smooth max objective, with averaging.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Average iterate
    a_avg = np.zeros(n)
    num_avg = 0
    lr = lr_scale / np.sqrt(n)
    it = 0

    while time.monotonic() - start < time_budget:
        grad, k_star = subgradient_at_peak(a)

        # Subgradient step
        a_new = a - lr * grad
        a_new = project_simplex(a_new)

        # Track average
        a_avg += a_new
        num_avg += 1
        a_check = a_avg / num_avg
        obj_check = compute_obj(a_check)
        if obj_check < best_obj:
            best_obj = obj_check
            best_a = a_check.copy()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr = max(1e-10, lr * 0.999)
        it += 1

        if verbose and it % 1000 == 0:
            print(f"    SGD it{it}: obj={best_obj:.8f}, lr={lr:.6f}")

    return best_a


def lp_polish(a, time_budget=200.0, verbose=False):
    """
    LP-based polishing: repeatedly solve min-max LP and take steps toward the solution.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break

        # Try several step sizes
        improved = False
        for t in [0.3, 0.2, 0.1, 0.5, 0.15, 0.05]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break

        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, M_lp={M_lp:.6f}")

    return best_a

        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / (a.sum() + 1e-10)
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * (w + 1) ** 2))
        a = a / (a.sum() + 1e-10)
    elif mode == 3:
        # Archimedes / semi-circle
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-4))
        a = a / (a.sum() + 1e-10)
    elif mode == 4:
        # Comb pattern
        a = np.ones(n)
        a[::2] = 2.0
        a = a / (a.sum() + 1e-10)
    elif mode == 5:
        # Periodic
        freq = rng.integers(2, 5)
        x = np.linspace(0, freq, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / (a.sum() + 1e-10)
    elif mode == 6:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1, 2)
        a = a / (a.sum() + 1e-10)
    elif mode == 7:
        # Triangular
        x = np.linspace(0, 1, n)
        a = 1 - 2 * np.abs(x - 0.5)
        a = np.maximum(a, 0.01)
        a = a / (a.sum() + 1e-10)
    else:
        # Cosine taper
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / (a.sum() + 1e-10)
    return a


def local_coord_search(a, time_budget=60.0, verbose=False):
    """
    Coordinate descent: try to reduce the peak by adjusting individual elements.
    For the peak at k*, try decreasing elements that contribute most to it.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find contributing indices
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)

        # Compute contributions
        contribs = []
        for j in range(lo, hi + 1):
            c = a[j] * a[k_star - j]
            contribs.append((c, j, k_star - j))

        contribs.sort(reverse=True)

        improved = False
        # Try reducing top contributors
        for c, j1, j2 in contribs[:5]:
            if j1 == j2:
                # Can't reduce diagonal without loss
                for frac in [0.1, 0.05, 0.02]:
                    delta = frac * a[j1]
                    if delta <= 0:
                        continue
                    a_try = a.copy()
                    a_try[j1] -= delta
                    # Redistribute to random positions
                    targets = rng.choice(n, size=min(3, n), replace=False)
                    for t_idx in targets:
                        a_try[t_idx] += delta / len(targets)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = np.maximum(a_try, 0)
                    a_try = a_try / a_try.sum()
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
            else:
                # Try moving mass from the larger of j1, j2
                for frac in [0.1, 0.05, 0.03, 0.01]:
                    # Reduce j1, add to j2 (or some other position)
                    delta = frac * a[j1]
                    if delta > a[j1]:
                        continue
                    a_try = a.copy()
                    a_try[j1] -= delta
                    # Add to a position far from the peak
                    target = rng.integers(0, n)
                    a_try[target] += delta
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = np.maximum(a_try, 0)
                    a_try = a_try / a_try.sum()
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                    # Also try symmetric move: reduce j2, add to j1
                    a_try2 = a.copy()
                    a_try2[j2] -= frac * a[j2]
                    a_try2[j1] += frac * a[j2]
                    if a_try2[j2] < 0:
                        continue
                    a_try2 = np.maximum(a_try2, 0)
                    a_try2 = a_try2 / a_try2.sum()
                    obj_try2 = compute_obj(a_try2)
                    if obj_try2 < best_obj - 1e-13:
                        best_obj = obj_try2
                        best_a = a_try2.copy()
                        a = a_try2
                        improved = True
                        break
                if improved:
                    break

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break

        if verbose and it % 20 == 0:
            print(f"    LCS it{it}: obj={best_obj:.8f}")

    return best_a


def random_perturbation(a, time_budget=60.0, sigma=0.05, verbose=False):
    """
    Random perturbations with acceptance criterion.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99999)
    it = 0

    while time.monotonic() - start < time_budget:
        # Multiplicative perturbation
        a_try = a * np.exp(rng.normal(0, sigma, n))
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            it += 1
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            # Reduce perturbation slightly
            sigma = max(0.01, sigma * 0.98)
        else:
            # Increase slightly to explore more
            sigma = min(0.15, sigma * 1.01)

        it += 1

        # Occasionally try a larger structural change
        if it % 50 == 0:
            n_change = max(1, n // 20)
            a_try = best_a.copy()
            indices = rng.choice(n, size=n_change, replace=False)
            a_try[indices] *= np.exp(rng.normal(0, 0.3, n_change))
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try

        if verbose and it % 100 == 0:
            print(f"    RP it{it}: obj={best_obj:.8f}, sigma={sigma:.4f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization pipeline:
    1. LP polishing (fixed bug)
    2. Nesterov smoothed descent (multiple tau ranges)
    3. Subgradient descent
    4. Local coordinate search
    5. Multi-n exploration with resampling
    6. Random perturbation / diversification
    7. Final combined polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # --- Load parent ---
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent available: {e}")

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: LP Polish ===
    print(f"\n=== Phase 1: LP Polish ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.2, 250.0)
    if p1_budget > 20:
        a_lp = lp_polish(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 2: Nesterov Smoothed Descent (low tau) ===
    print(f"\n=== Phase 2: Nesterov GD (tau 20-300) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 300.0)
    if p2_budget > 20:
        a_ns = nesterov_smoothed_descent(best_seq, time_budget=p2_budget,
                                         tau_init=20.0, tau_end=300.0,
                                         beta=0.9, lr_scale=1.0, verbose=True)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  NSD: obj={best_score:.8f} **")
        else:
            print(f"  NSD: obj={obj_ns:.8f}")

    # === Phase 3: Subgradient Descent ===
    print(f"\n=== Phase 3: Subgradient ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 120.0)
    if p3_budget > 15:
        a_sg = subgradient_descent(best_seq, time_budget=p3_budget,
                                   lr_scale=0.5, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  SGD: obj={best_score:.8f} **")
        else:
            print(f"  SGD: obj={obj_sg:.8f}")

    # === Phase 4: Nesterov (high tau) ===
    print(f"\n=== Phase 4: Nesterov GD (tau 100-1000) ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 180.0)
    if p4_budget > 20:
        a_ns2 = nesterov_smoothed_descent(best_seq, time_budget=p4_budget,
                                          tau_init=100.0, tau_end=1000.0,
                                          beta=0.85, lr_scale=0.5, verbose=True)
        obj_ns2 = compute_obj(a_ns2)
        if obj_ns2 < best_score:
            best_score = obj_ns2
            best_seq = a_ns2.copy()
            print(f"  NSD2: obj={best_score:.8f} **")
        else:
            print(f"  NSD2: obj={obj_ns2:.8f}")

    # === Phase 5: Local Coordinate Search ===
    print(f"\n=== Phase 5: Local Coord Search ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.08, 80.0)
    if p5_budget > 10:
        a_cd = local_coord_search(best_seq, time_budget=p5_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  LCS: obj={best_score:.8f} **")
        else:
            print(f"  LCS: obj={obj_cd:.8f}")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Generate candidate n values
    n_candidates = []
    for n_try in [200, 300, 400, 500, 600, 700, 800, 900, 1000, 1200, 1500]:
        if n_try != n_current and (n_try < n_current * 0.7 or n_try > n_current * 1.3):
            n_candidates.append(n_try)

    # Sort by distance from current
    n_candidates.sort(key=lambda x: abs(x - n_current))

    for n_try in n_candidates[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # Quick optimization: Nesterov + LP
        alloc = min(remaining * 0.15, 60.0)
        if alloc > 15:
            a_try = nesterov_smoothed_descent(a_try, time_budget=alloc * 0.7,
                                              tau_init=30.0, tau_end=400.0)
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_try = lp_polish(a_try, time_budget=min(remaining * 0.3, 20.0))

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse Restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = (r * 3) % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with current best
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r < 1e-10:
            continue
        a_r = a_r / s_r

        alloc = min(remaining * 0.2, 80.0)
        if alloc > 15:
            a_r = nesterov_smoothed_descent(a_r, time_budget=alloc * 0.6,
                                            tau_init=30.0, tau_end=500.0)
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_r = lp_polish(a_r, time_budget=min(remaining * 0.3, 25.0))

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Random Perturbation ===
    print(f"\n=== Phase 8: Random Perturbation ===")
    remaining = deadline - time.monotonic()
    p8_budget = min(remaining * 0.1, 80.0)
    if p8_budget > 15:
        a_rp = random_perturbation(best_seq, time_budget=p8_budget,
                                   sigma=0.03, verbose=True)
        obj_rp = compute_obj(a_rp)
        if obj_rp < best_score:
            best_score = obj_rp
            best_seq = a_rp.copy()
            print(f"  RP: obj={best_score:.8f} **")
        else:
            print(f"  RP: obj={obj_rp:.8f}")

    # === Phase 9: Final Polish ===
    print(f"\n=== Phase 9: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP
    if remaining > 20:
        alloc = min(remaining * 0.25, 60.0)
        a_f = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Nesterov high tau
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.25, 50.0)
        a_f = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                        tau_init=200.0, tau_end=2000.0,
                                        beta=0.8, lr_scale=0.3, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final NSD: obj={best_score:.8f} **")

    # Local coord
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.2, 40.0)
        a_f = local_coord_search(best_seq, time_budget=alloc, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LCS: obj={best_score:.8f} **")

    # Perturbation
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.3, 30.0)
        a_f = random_perturbation(best_seq, time_budget=alloc, sigma=0.01)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final RP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
