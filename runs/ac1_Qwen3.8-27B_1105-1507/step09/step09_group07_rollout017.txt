```python
"""
Optimized search for minimizing peak autocorrelation of non-negative step functions.
Uses accelerated gradient methods (FISTA) with log-sum-exp surrogate,
combined with LP polishing, n-sweep, and diverse restarts.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def cross_corr(w, a):
    """Cross-correlation of w and a, returning array of length n."""
    # cross_corr[m] = sum_k w[k] * a[k-m] (with appropriate bounds)
    # This is equivalent to conv(w, a[::-1])[n-1:n-1+n]
    # Or we can use FFT
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    # Cross-correlation: ifft(conj(W) * A)
    cc = irfft(np.conj(W) * A, size)[:n]
    return cc

    a = a / s
    a = project_simplex(a)

    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    it = 0

        w = exp_vals / float(np.sum(exp_vals))

        grad = 2.0 * cross_corr(w, y)
        
        # FISTA gradient step with backtracking
        # Project onto simplex
        max_backtracks = 5
        l = lr
        for bt in range(max_backtracks):
            z = y - l * grad
            x = project_simplex(z)
            # Check descent condition (approximate)
            if bt < max_backtracks - 1:
                # Simple backtracking: check if function decreased
                conv_x = compute_conv(x)
                f_new = float(np.max(conv_x))
                f_old = float(np.max(conv_y))
                # Lipschitz check: f_new <= f_old + grad·(x-y) + (1/(2l))||x-y||^2
                # Simplified: just check if f didn't increase too much
                if f_new <= f_old + 1e-10:
                    break
                l *= 0.5
            else:
                break
        
        lr = l
        
        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x + (t_k - 1.0) / t_new * (x - a)
        y = np.maximum(y, 0)
        y = y / max(1e-15, float(np.sum(y)))
        a = x
        t_k = t_new
        
        it += 1
        
        # Periodic objective check
        if it % 5 == 0 or it < 10:
            obj = compute_obj(a)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
        
        if verbose and it % 100 == 0:
            obj = compute_obj(a)
            print(f"    FISTA it{it}: obj={obj:.8f}, beta={beta:.0f}, lr={lr:.6f}, elapsed={time.monotonic()-start:.1f}s")
    
    # Final check
    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()
    
    return best_a

def mirror_descent(a0, time_budget=30, beta_start=20, beta_end=2000, verbose=False):
    """
    Mirror descent (exponential family) for the smoothed objective.
    """
    n = len(a0)
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 0.1
    it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / time_budget)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        conv = compute_conv(a)
        m = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - m))
        w = exp_vals / float(np.sum(exp_vals))
        
        # Gradient
        grad = 2.0 * cross_corr(w, a)
        
        # Mirror descent update
        log_a = np.log(a)
        log_a -= lr * grad
        log_a -= np.max(log_a)  # numerical stability
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = max(0.001, lr * 0.995)
        else:
            lr = max(0.001, lr * 0.9)
        
        it += 1
        if verbose and it % 50 == 0:
            print(f"    Mirror it{it}: obj={best_obj:.8f}, beta={beta:.0f}, lr={lr:.4f}")
    
    return best_a

def lp_polish(a, time_budget=30):
    """Quick LP-based polishing (simplified version)."""
    import cvxpy as cp
    
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Build convolution matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    
    iteration = 0
    while time.monotonic() - start < time_budget:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=cp.CLARABEL)
            
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    g_val = g_val / s_val
                    
                    # Try different step sizes
                    improved = False
                    for t in [0.3, 0.5, 0.7, 1.0]:
                        a_try = (1 - t) * best_a + t * g_val
                        a_try = project_simplex(a_try)
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-12:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            improved = True
                    
                    if not improved:
                        break
                    a = best_a.copy()
                    # Rebuild A with new a
                    A[:] = 0
                    for j in range(n):
                        r = min(n, 2 * n - 1 - j)
                        A[j:j + r, j] = a[:r]
            else:
                break
        except Exception:
            break
        
        iteration += 1
    
    return best_a

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode % 10 == 0:
        a = np.ones(n) / n
    elif mode % 10 == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode % 10 == 2:
        # U-shaped
        x = np.linspace(0, 1, n)
        a = (x - 0.5)**2 + 0.05
        a = a / a.sum()
    elif mode % 10 == 3:
        # Inverted U (bell)
        x = np.linspace(0, 1, n)
        a = np.exp(-4*(x-0.5)**2) + 0.05
        a = a / a.sum()
    elif mode % 10 == 4:
        # Cosine
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.4*np.cos(2*np.pi*x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 5:
        # Sine
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3*np.sin(2*np.pi*2*x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 6:
        # Two peaks
        x = np.linspace(0, 1, n)
        a = np.exp(-50*(x-0.25)**2) + np.exp(-50*(x-0.75)**2) + 0.01
        a = a / a.sum()
    elif mode % 10 == 7:
        # Linear ramp
        x = np.linspace(0, 1, n)
        a = x + 0.1
        a = a / a.sum()
    elif mode % 10 == 8:
        # Quantized
        a = np.round(rng.uniform(0.2, 1, n) * 4) / 4
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n) * 0.3)
    return a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def coordinate_refine(a, time_budget=30):
    """Coordinate descent refinement."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for rnd in range(50):
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        # Try moving mass between pairs
        for _ in range(100):
            if time.monotonic() - start > time_budget:
                break
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j or a[i] < 1e-10:
                continue
            for frac in [0.01, 0.05, 0.1]:
                delta = frac * a[i]
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s < 1e-10:
                    a[i] += delta
                    a[j] -= delta
                    continue
                a = a / s
                obj = compute_obj(a)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
                    a = np.maximum(a, 0)
                    a = a / max(s, 1e-10)
                    break
        
        if not improved:
            break
    
    return best_a
```

```python
def cross_corr(w, a):
    """Compute result[m] = sum_k w[k] * a[k-m] for m = 0,...,n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    # The cross-correlation via FFT:
    # ifft(conj(W) * A)[m] = sum_k w[k] * a[(k+m) mod size]
    # With zero-padding to size >= 2n, this gives the linear cross-correlation
    # shifted. Specifically:
    # result[m] = sum_k w[k] * a[k-m] for k-m >= 0
    # = sum_j w[j+m] * a[j] for j+m < n
    # This is ifft(conj(W) * A)[-m] or ifft(conj(W) * A)[size-m]
    cc = irfft(np.conj(W) * A, size)
    # cc[m] = sum_k w[k] * a[(k+m) mod size]
    # For m >= 0 and zero-padded: cc[m] = sum_{k=0}^{n-1} w[k] * a[(k+m) if k+m < n else 0]
    # But we want: result[m] = sum_k w[k] * a[k-m] (when k-m >= 0)
    # = sum_{k=m}^{n-1} w[k] * a[k-m]
    # Let j = k-m: = sum_{j=0}^{n-1-m} w[j+m] * a[j]
    # This is cc[-m] (circular) or cc[size-m]
    result = np.zeros(n)
    for m in range(n):
        if m == 0:
            result[m] = cc[0]
        else:
            result[m] = cc[size - m]
```
