import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A such that (A@g)[k] = sum_j a[k-j]*g[j] = conv(a,g)[k]."""
    n = len(a)
    # A has shape (2n-1, n)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt).

    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

                s = float(np.sum(g_val))
                if s > 1e-10:

            stall = max(0, stall - 1)

        # Try step sizes
        candidate_ts = [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]
        if 0.05 < last_good_t < 1.0:
            candidate_ts = [last_good_t, last_good_t * 0.7, last_good_t * 1.3, 0.5, 0.3, 0.1]
        candidate_ts = [t for t in candidate_ts if 0 < t <= 1.0]
        candidate_ts = sorted(set(candidate_ts))

        best_t_obj = np.inf
        best_t_val = 0.5
        best_a_local = a.copy()

        for t in candidate_ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        iteration += 1
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local.copy()
            last_good_t = best_t_val
            stall = 0
        else:
            stall += 1
            if stall > patience:
                # Perturb and restart
                rng = np.random.default_rng(iteration * 7919 + n)
                scale = 0.05
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                last_good_t = 0.5

        if verbose and iteration % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"          LP it{iteration}: obj={best_obj:.8f}, t={last_good_t:.3f}, stall={stall}, elapsed={elapsed:.1f}s")

    return best_a


def gradient_refine(a, time_budget, verbose=False):
    """
    Gradient-based refinement using log-sum-exp smoothing.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.1
    beta = 50.0
    it = 0
    stuck = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = conv.max()
        
        # Log-sum-exp weights
        shifted = beta * (conv - max_c)
        w = np.exp(shifted)
        w = w / w.sum()
        
        # Gradient: d(max conv)/d(a[i]) ≈ 2 * sum_k w[k] * a[k-i]
        # This is 2 * cross-correlation of w and a
        # cross_corr[i] = sum_k w[k] * a[k-i] (with a[j]=0 if j not in [0,n-1])
        # We can compute this via FFT:
        # cross_corr = ifft(conj(fft(w)) * fft(a)) ... but we need linear cross-correlation
        
        # Linear cross-correlation of w (length 2n-1) and a (length n):
        # result[i] = sum_k w[k] * a[k-i] for i = 0, ..., n-1
        # This is the same as: for each i, sum over k where 0 <= k-i < n
        
        # Using convolution: cross_corr = conv(w, flip(a)) shifted appropriately
        # Or just: pad a to length 2n-1, compute irfft(conj(fft(w)) * fft(a_padded))
        
        size = 2 * n - 1
        size_fft = 1
        while size_fft < 2 * (2 * n - 1) - 1:
            size_fft <<= 1
        
        W = rfft(w, size_fft)
        A_k = rfft(a, size_fft)
        # cross_corr[i] = sum_k w[k] * a[k-i]
        # This is the cross-correlation, which is ifft(conj(FFT(w)) * FFT(a)) for circular
        # For linear, we need to be careful about the length
        
        # Actually, let's just compute it directly using the structure:
        # cross_corr[i] = sum_{k=max(0,i)}^{min(2n-2, i+n-1)} w[k] * a[k-i]
        # = sum_{j=0}^{n-1} w[i+j] * a[j] (where w[i+j] is 0 if i+j >= 2n-1)
        # This is a "sliding dot product" which can be done with convolution:
        # cross_corr = conv(w, a_reversed) ... hmm, let me think.
        
        # Actually, the simplest way:
        # grad[i] = 2 * sum_{k: 0<=k-i<n} w[k] * a[k-i]
        # = 2 * sum_{j: 0<=j<n} w[i+j] * a[j]
        # This is the (i-th) element of the "autocorrelation-like" quantity
        
        # We can compute this as: for the vector w of length 2n-1 and a of length n,
        # the quantity sum_j w[i+j] * a[j] for i = 0,...,n-1
        # is the same as the cross-correlation of w and a at specific lags.
        
        # Let's use the FFT approach:
        # Pad a to length 2n-1 (zero-pad on the right)
        a_padded = np.zeros(2 * n - 1)
        a_padded[:n] = a
        # Now cross_corr[i] = sum_j w[i+j] * a_padded[j] = sum_j a_padded[j] * w[i+j]
        # This is the i-th element of the "anti-convolution" or...
        # Actually, this is cross-correlation: (w ⋆ a_padded)[i] = sum_j w[i+j] * a_padded[j]
        # = sum_j a_padded[j] * w[i+j]
        # In terms of convolution: if we define b = flip(w) (i.e., b[k] = w[2n-2-k]),
        # then conv(b, a_padded)[2n-2+i] = sum_j b[2n-2+i-j] * a_padded[j] = sum_j w[j-i] * a_padded[j]
        # Hmm, that's not quite right either.
        
        # Let me just use the direct FFT computation:
        # cross_corr[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
        # Define the vector w_ext of length 2n-1
        # This is the same as: cross_corr = ifft(FFT(a_extended) * conj(FFT(w))) ... 
        # Actually, for cross-correlation: C[i] = sum_j a[j] * w[i+j]
        # This is the same as: C = ifft(FFT(w) * conj(FFT(a_padded))) ... no.
        
        # Let me just do it the simple way for n=800 (it's fast enough):
        grad = np.zeros(n)
        for i in range(n):
            # sum_{j=0}^{n-1} a[j] * w[i+j] where w[i+j] = 0 if i+j >= 2n-1
            j_max = min(n - 1, 2 * n - 2 - i)
            if j_max >= 0:
                grad[i] = 2.0 * np.dot(a[:j_max + 1], w[i:i + j_max + 1])
        
        # Mirror descent (multiplicative update)
        log_a = np.log(a)
        log_a_new = log_a - lr * grad
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stuck = 0
            lr = min(lr * 1.01, 1.0)
        else:
            stuck += 1
            lr = max(0.001, lr * 0.9)
            if stuck > 100:
                break
        
        it += 1
        if verbose and it % 50 == 0:
            print(f"          Grad it{it}: obj={best_obj:.8f}, lr={lr:.4f}, elapsed={time.monotonic()-start:.1f}s")

    return best_a
