"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a hybrid approach: LP-based direction finding with exact peak-focused
subgradient descent, adaptive line search, and structured perturbations.
Key differences from previous: top-k peak focus, direct peak manipulation,

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def topk_subgradient(a, k_frac=0.1):
    """
    Compute subgradient focusing on top-k peaks of conv(a,a).
    Returns (gradient, conv, peak_indices).
    """
    n = len(a)
    conv = compute_conv(a)
    n_top = max(1, int(k_frac * len(conv)))
    top_idx = np.argsort(conv)[::-1][:n_top]
    # Subgradient of max is uniform over the argmax set
    # For top-k, use uniform weights over top-k
    weights = np.zeros(len(conv))
    weights[top_idx] = 1.0 / n_top
    
    # Gradient of (a*a)[k] wrt a[j] = 2*a[k-j] if 0 <= k-j < n
    # So grad[j] = 2 * sum_k weights[k] * a[k-j] for valid k-j
    # This is a cross-correlation of weights with a
    N = 1
    while N < 3 * n:
        N <<= 1
    W = fft(weights, N)
    A = fft(a, N)
    g_full = ifft(W * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv, top_idx

def projected_subgradient_descent(a, time_budget=100.0, k_frac=0.05, verbose=False):
    """
    Projected subgradient descent on the exact (non-smoothed) objective.
    Uses top-k peaks for subgradient computation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = 1.0 / np.sqrt(n)
    rho = 1.0
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        g, conv, top_idx = topk_subgradient(a, k_frac)
        
        # Project gradient onto subspace sum=0
        g = g - np.mean(g)
        
        # Step
        a_new = a - eta * rho * g
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
            rho *= 0.5
            it += 1
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            rho = min(rho * 1.1, 5.0)
        else:
            no_improve += 1
            rho *= 0.7
            if no_improve > 20:
                break
        
        eta /= np.sqrt(1 + it * eta**2)
        it += 1
        
        if it % 100 == 0 and verbose:
            elapsed = time.monotonic() - start
            print(f"      PSD it{it}: obj={best_obj:.8f}, rho={rho:.4f}, eta={eta:.6f}, t={elapsed:.0f}s")
    
    return best_a

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a, time_limit=10.0):
    """
    Solve LP: min M s.t. conv(a_norm, g) <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def ternary_line_search(a, direction, t_range=(0.0, 1.0), iters=20):
    """
    Ternary search for optimal blend t in a_new = (1-t)*a + t*direction.
    Returns (best_a, best_obj, best_t).
    """
    lo, hi = t_range
    best_t = 0.0
    best_obj = np.inf
    
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        
        a1 = (1 - m1) * a + m1 * direction
        a1 = np.maximum(a1, 0)
        s1 = float(np.sum(a1))
        if s1 > 1e-10:
            a1 = a1 / s1
            obj1 = compute_obj(a1)
        else:
            obj1 = np.inf
        
        a2 = (1 - m2) * a + m2 * direction
        a2 = np.maximum(a2, 0)
        s2 = float(np.sum(a2))
        if s2 > 1e-10:
            a2 = a2 / s2
            obj2 = compute_obj(a2)
        else:
            obj2 = np.inf
        
        if obj1 < obj2:
            hi = m2
        else:
            lo = m1
        
        # Track best
        if obj1 < best_obj:
            best_obj = obj1
            best_t = m1
        if obj2 < best_obj:
            best_obj = obj2
            best_t = m2
    
    a_best = (1 - best_t) * a + best_t * direction
    a_best = np.maximum(a_best, 0)
    s_best = float(np.sum(a_best))
    if s_best > 1e-10:
        a_best = a_best / s_best
        best_obj = compute_obj(a_best)
    
    return a_best, best_obj, best_t

    # Line search

def peak_cancellation(a, n_steps=10, verbose=False):
    """
    Direct peak cancellation: find the peak, identify contributing elements,
    and redistribute mass.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for step in range(n_steps):
        conv = compute_conv(a)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]
        
        # Find which j values contribute most to conv[peak_k]
        # conv[peak_k] = sum_j a[j] * a[peak_k - j] for valid indices
        contributions = np.zeros(n)
        for j in range(n):
            idx = peak_k - j
            if 0 <= idx < n:
                contributions[j] = a[j] * a[idx]
        
        # Top contributors
        top_m = max(5, n // 20)
        top_contrib_idx = np.argsort(contributions)[::-1][:top_m]
        
        # Reduce top contributors, increase others
        total_reduce = float(contributions[top_contrib_idx].sum())
        if total_reduce < 1e-15:
            break
        
        # Reduce proportionally
        reduce_frac = 0.05
        for j in top_contrib_idx:
            a[j] *= (1 - reduce_frac)
        
        # Renormalize
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            break
        a = a / s
        
        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
        # Undo if it got worse
        else:
            a = best_a.copy()
            break
    
    return best_a, best_obj

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

Uses LP-based direction finding with ternary line search, top-k peak-focused

descent for final polish, and focused multi-n exploration.
Key differences from previous: ternary line search (not discrete t), top-k exact

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize as scipy_minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def topk_subgradient(a, k_frac=0.05):
    """
    Compute subgradient focusing on top-k peaks of conv(a,a).
    Returns (gradient_array, conv_array, peak_indices).
    """
    n = len(a)
    conv = compute_conv(a)
    n_top = max(1, int(k_frac * len(conv)))
    top_idx = np.argsort(conv)[::-1][:n_top]
    weights = np.zeros(len(conv))
    weights[top_idx] = 1.0 / n_top

    N = 1
    while N < 3 * n:
        N <<= 1
    W = fft(weights, N)
    A = fft(a, N)
    g_full = ifft(W * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv, top_idx

    Projected subgradient descent on exact objective using top-k peaks.
    Uses Polyak step size with running best.

    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a

    best_a = a.copy()
    best_obj = compute_obj(a)
    lb = 2.0 * n / (2 * n - 1)  # theoretical lower bound
    start = time.monotonic()

    gamma = 1.0 / np.sqrt(n)
    it = 0
    no_improve = 0

        g, conv, top_idx = topk_subgradient(a, k_frac)
        g = g - np.mean(g)  # project to sum=0 hyperplane

        gnorm_sq = float(np.dot(g, g))
        if gnorm_sq < 1e-20:

        eta = min(3.0 * (best_obj - lb) / gnorm_sq, 10.0 / np.sqrt(n))
        eta = max(eta, 1e-10)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

    """
    Solve LP: min M s.t. conv(a_norm, g) <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def ternary_line_search(a, direction, n_iters=25):
    """
    Ternary search for optimal blend t in a_new = (1-t)*a + t*direction.
    Returns (best_a, best_obj, best_t).
    """
    lo, hi = 0.0, 1.0
    best_t = 0.0
    best_obj = compute_obj(a)

    for _ in range(n_iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0

        a1 = (1 - m1) * a + m1 * direction
        a1 = np.maximum(a1, 0)
        s1 = float(np.sum(a1))
        obj1 = compute_obj(a1 / s1) if s1 > 1e-10 else np.inf

        a2 = (1 - m2) * a + m2 * direction
        a2 = np.maximum(a2, 0)
        s2 = float(np.sum(a2))
        obj2 = compute_obj(a2 / s2) if s2 > 1e-10 else np.inf

        if obj1 < obj2:
            hi = m2
        else:
            lo = m1

        if obj1 < best_obj:
            best_obj = obj1
            best_t = m1
        if obj2 < best_obj:
            best_obj = obj2
            best_t = m2

    a_best = (1 - best_t) * a + best_t * direction
    a_best = np.maximum(a_best, 0)
    s_best = float(np.sum(a_best))
    if s_best > 1e-10:
        a_best = a_best / s_best
        best_obj = compute_obj(a_best)

    return a_best, best_obj, best_t


def lp_line_search_step(a, line_search_iters=25):
    """
    One iteration: solve LP for direction, then ternary line search.
    Returns (a_new, obj_new, t_used, M_lp).
    """
    a_norm = a / float(np.sum(a))
    g_lp, M_lp = solve_lp_minpeak(a_norm)
    if g_lp is None:
        return a, compute_obj(a), 0.0, np.inf

    a_new, obj_new, t_best = ternary_line_search(a_norm, g_lp, n_iters=line_search_iters)
    return a_new, obj_new, t_best, M_lp


def peak_aware_perturb(a, rng, strength=0.03):
    """
    Perturb a in a way that's aware of the current peak structure.
    Reduces elements that contribute most to peaks, increases others.
    """
    n = len(a)
    a = a.copy()
    conv = compute_conv(a)
    peak_k = int(np.argmax(conv))

    # Compute contribution of each element to the peak
    contrib = np.zeros(n)
    for j in range(n):
        idx = peak_k - j
        if 0 <= idx < n:
            contrib[j] = a[j] * a[idx]
        idx2 = j - peak_k
        if 0 <= idx2 < n:
            contrib[j] += a[j] * a[idx2]

    max_c = np.max(contrib)
    if max_c < 1e-15:
        # Fallback: random perturbation
        a *= np.exp(rng.normal(0, strength, n))
    else:
        # Reduce high-contributing elements, increase low-contributing ones
        reduction = strength * contrib / max_c
        a = a * (1.0 - reduction)
        a = np.maximum(a, 1e-8)

    # Add small random noise for diversity
    a *= np.exp(rng.normal(0, strength * 0.3, n))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def block_coordinate_descent(a, block_size=20, n_sweeps=3, time_budget=60.0):
    """
    Block coordinate descent: optimize blocks of elements using scipy.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for sweep in range(n_sweeps):
        if time.monotonic() - start > time_budget:
            break
        for start_idx in range(0, n, block_size):
            if time.monotonic() - start > time_budget:
                break
            end_idx = min(start_idx + block_size, n)
            block = a[start_idx:end_idx].copy()
            fixed = a.copy()

            def block_obj(x):
                trial = fixed.copy()
                trial[start_idx:end_idx] = x
                trial = np.maximum(trial, 0)
                s = float(np.sum(trial))
                if s < 1e-10:
                    return 1e10
                return compute_obj(trial / s)

            try:
                res = scipy_minimize(
                    block_obj, block,
                    method='Nelder-Mead',
                    options={'maxiter': 500, 'xatol': 1e-8, 'fatol': 1e-10}
                )
                if res.fun < best_obj:
                    a[start_idx:end_idx] = np.maximum(res.x, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                        obj = compute_obj(a)
                        if obj < best_obj:
                            best_obj = obj
                            best_a = a.copy()
            except Exception:
                pass

    return best_a, best_obj


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 6, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        # Prolate-like: sinc-squared window
        x = np.arange(n) / n - 0.5
        a = np.sinc(2 * x) ** 2
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 8:
        # Tukey window
        x = np.arange(n) / n - 0.5
        r = 0.3
        a = np.ones(n)
        mask = np.abs(x) > (0.5 - r)
        x_tukey = (np.abs(x[mask]) - (0.5 - r)) / r
        a[mask] = 0.5 * (1 + np.cos(np.pi * x_tukey))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def optimize_at_n(a, time_budget=120.0, verbose=False):
    """
    Full optimization pipeline for a given length n:
    LP + line search loop with perturbations, then PSD, then BCD.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)

    # Main LP + line search loop
    no_improve = 0
    it = 0
    max_iters = 200

    while it < max_iters:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 8:
            break

        # LP + line search
        a_new, obj_new, t_used, M_lp = lp_line_search_step(a, line_search_iters=20)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
            if verbose and it % 3 == 0:
                print(f"      LP-LS it{it}: obj={best_obj:.8f}, t={t_used:.3f}, M={M_lp:.6f}")
        else:
            no_improve += 1
            if no_improve >= 4:
                # Perturb and reset
                a = peak_aware_perturb(best_a, rng, strength=0.02 + 0.01 * min(no_improve, 5))
                no_improve = 0

        # Every 5 iterations, do a quick PSD step
        if (it + 1) % 5 == 0:
            elapsed = time.monotonic() - start
            if time_budget - elapsed > 5:
                a_psd = psd_simplex(a, time_budget=min(5.0, (time_budget - elapsed) * 0.3), k_frac=0.03)
                obj_psd = compute_obj(a_psd)
                if obj_psd < best_obj - 1e-13:
                    best_obj = obj_psd
                    best_a = a_psd.copy()
                    a = a_psd.copy()

        it += 1

    # Final PSD
    elapsed = time.monotonic() - start
    if time_budget - elapsed > 10:
        a_psd = psd_simplex(best_a, time_budget=min(20.0, (time_budget - elapsed) * 0.4), k_frac=0.02)
        obj_psd = compute_obj(a_psd)
        if obj_psd < best_obj:
            best_obj = obj_psd
            best_a = a_psd.copy()

    # Final BCD
    elapsed = time.monotonic() - start
    if time_budget - elapsed > 10:
        a_bcd, obj_bcd = block_coordinate_descent(best_a, block_size=15, n_sweeps=2,
                                                  time_budget=min(30.0, time_budget - elapsed - 2))
        if obj_bcd < best_obj:
            best_obj = obj_bcd
            best_a = a_bcd.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses LP + ternary line search as primary optimizer,
    top-k subgradient descent and block coordinate descent as refinements,
    peak-aware perturbations for escaping local minima, and focused multi-n
    exploration. Returns list of non-negative floats.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Primary optimization at parent n ===
    print(f"\n=== Phase 1: Primary optimization (n={n_target}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 450.0)
    if p1_budget > 30:
        a_opt, obj_opt = optimize_at_n(best_seq, time_budget=p1_budget, verbose=True)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase 1 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 result: obj={obj_opt:.8f}")

    # === Phase 2: Multi-n exploration ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Choose n values to explore: nearby and some diverse
    n_candidates = set()
    for delta in [-200, -100, -50, 50, 100, 200, 300, 400, 500, 600]:
        n_try = n_current + delta
        if 200 <= n_try <= 2500:
            n_candidates.add(n_try)
    for n_abs in [400, 500, 600, 700, 900, 1000, 1200, 1500, 1800, 2000]:
        if 200 <= n_abs <= 2500:
            n_candidates.add(n_abs)
    n_candidates.discard(n_current)

    # Sort by distance to current, but ensure diversity
    n_list = sorted(n_candidates, key=lambda x: abs(x - n_current))
    # Pick at most 4 diverse n values
    selected_n = []
    for n_try in n_list:
        if len(selected_n) >= 4:
            break
        if all(abs(n_try - sn) > 100 for sn in selected_n):
            selected_n.append(n_try)

    print(f"  Exploring n values: {selected_n}")

    for i, n_try in enumerate(selected_n):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        # Start from resampled best
        a_try = resample_sequence(best_seq, n_try)

        # Also try a diverse init blended with best
        mode = i % 9
        a_div = diverse_init(n_try, rng, mode=mode)
        a_try = 0.7 * a_try + 0.3 * a_div
        a_try = np.maximum(a_try, 0)
        a_try = a_try / a_try.sum()

        alloc = min(remaining * 0.2, 150.0)
        if alloc < 20:
            continue

        a_opt, obj_opt = optimize_at_n(a_try, time_budget=alloc, verbose=(i == 0))

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try
            print(f"  n={n_try} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  n={n_try} (mode={mode}): obj={obj_opt:.8f}")

    # === Phase 3: Diverse restarts at best n ===
    print(f"\n=== Phase 3: Diverse Restarts (n={n_current}) ===")
    remaining = deadline - time.monotonic()

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        mode = (r + 2) % 9
        a_r = diverse_init(n_current, rng, mode=mode)
        # Blend with current best (less blend for later restarts)
        blend = 0.4 - 0.05 * r
        blend = max(0.1, blend)
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.15, 100.0)
        if alloc < 15:
            continue

        a_opt, obj_opt = optimize_at_n(a_r, time_budget=alloc)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}, blend={blend:.1f}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}, blend={blend:.1f}): obj={obj_opt:.8f}")

    # === Phase 4: Final intensive polish ===
    print(f"\n=== Phase 4: Final Intensive Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish loop
    if remaining > 20:
        p4_budget = min(remaining * 0.5, 200.0)
        a_opt, obj_opt = optimize_at_n(best_seq, time_budget=p4_budget, verbose=True)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Polish round 1: obj={best_score:.8f} **")

    # BCD on the best
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_bcd, obj_bcd = block_coordinate_descent(best_seq, block_size=10, n_sweeps=3,
                                                  time_budget=min(60.0, remaining * 0.4))
        if obj_bcd < best_score:
            best_score = obj_bcd
            best_seq = a_bcd.copy()
            print(f"  BCD: obj={best_score:.8f} **")
        else:
            print(f"  BCD: obj={obj_bcd:.8f}")

    # One more LP polish
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_opt, obj_opt = optimize_at_n(best_seq, time_budget=min(60.0, remaining * 0.5))
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
