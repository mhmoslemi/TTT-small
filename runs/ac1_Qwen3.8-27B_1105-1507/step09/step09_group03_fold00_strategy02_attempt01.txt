import numpy as np

def evaluate_sequence(sequence):
    # Security checks
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    # Convert to float and ensure non-negative
    sequence_float = [float(x) for x in sequence]
    sequence_float = [max(0, x) for x in sequence_float]

# Main search function
def propose_candidate(time_budget=1000.0):
    start_time = time.time()
    deadline = start_time + time_budget
    
    # Initialization: If parent is available, use it, else fallback to uniform
    try:
        parent_arr = np.asarray(height_sequence_1)  # Assuming global parent variable exists
        if parent_arr.size == 0:
            raise ValueError("Parent sequence appears empty")
        init_seq = parent_arr.copy()
    except NameError:
        print("Parent sequence not available, using fallback initial conditions")
        init_seq = np.ones(800) * 0.01
        init_seq_normalized = init_seq / np.sum(init_seq)
        init_seq = init_seq_normalized
    
    # Normalization procedure
    def normalize(seq, tol=1e-10):
        total = np.sum(seq)
        if total < tol:
            return seq * (0.1 / total) if total != 0 else np.ones_like(seq) * 0.01
        return seq / total
    
    init_seq = normalize(init_seq)
    
    # Compute initial objective
    initial_score = evaluate_sequence(init_seq)
    best_seq = init_seq
    best_score = initial_score
    
    # Time tracking variables
    elapsed = 0
    remaining_time = deadline - start_time
    
    # PCD procedure
    def peak_targeted_cd(seq, time_limit=60):
        start_peak = time.time()
        time_available = start_peak - elapsed
        if time_available > time_limit:
            time.sleep(time_limit - time_available)
        conv_seq = np.convolve(seq, seq)
        max_pos = np.argmax(conv_seq)
        return max_pos, conv_seq[max_pos]  # Return peak index and value
    
    # Optimization loop
    iteration = 0
    total_iterations = 50
    while True:
        current_time = time.time()
        elapsed = current_time - start_time
        if elapsed > deadline:
            break
        
        peak_idx, peak_val = peak_targeted_cd(best_seq, deadline - elapsed)
        
        # Candidate generation
        if peak_val > 0.01 * best_score:
            neighbors = []
            for i in range(max(0, peak_idx-10), min(2100, len(seq))):
                neighbors.append(i)
            neighbors = list(set(neighbors))
            
            for target in neighbors:
                probability = np.exp(-(abs(target-peak_idx)/10)**2)
                candidate_seq = np.zeros(800)
                if target < len(seq):
                    candidate_seq[target] += 0.5 * probability * best_score
                else:
                    candidate_seq[target % 800] += 0.5 * probability * best_score
            new_seq = normalize(candidate_seq)
            new_score = evaluate_sequence(new_seq)
            
            if new_score < best_score - 1e-6:
                best_seq = new_seq
                best_score = new_score
        iteration += 1
        if iteration > total_iterations or elapsed > deadline - 20:
            break
    
    # Final sequence check and expensive optimization avoided by budget
    final_seq = best_seq  # Final is stored here for export
    
    return final_seq.tolist()

# Execute with budget
