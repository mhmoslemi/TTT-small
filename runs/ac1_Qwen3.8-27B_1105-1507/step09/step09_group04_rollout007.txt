            d[k:k + n] = a
            
            lo_delta = -a[k]
            upper_est = np.sqrt(max(0.0, float(np.max(conv)) - conv[2 * k] + a[k] ** 2 + 0.05 * float(np.max(conv)))) - a[k]
            hi_delta = max(upper_est, 2.0 / n)
            hi_delta = max(hi_delta, lo_delta + 1.0 / n)
            
            # Golden section
            a_gs = lo_delta
            b_gs = hi_delta
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)
            
            def obj_at_delta(delta):
                val = conv + 2.0 * delta * d
                val[2 * k] += delta * delta
                return float(np.max(val))
            
            fc = obj_at_delta(c_gs)
            fd = obj_at_delta(d_gs)
            
            for _ in range(40):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = obj_at_delta(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = obj_at_delta(d_gs)
            
            delta_opt = (a_gs + b_gs) / 2.0
            new_obj_est = obj_at_delta(delta_opt)
            
            if new_obj_est >= obj_before - 1e-15:
                continue
            
            a[k] += delta_opt
            a[k] = max(0.0, a[k])
            conv += 2.0 * delta_opt * d
            conv[2 * k] += delta_opt * delta_opt
            
            S_new = float(np.sum(a))
            if S_new < 1e-10:
                a[:] = 1.0 / n
                conv = compute_conv(a)
                break
            else:
                a /= S_new
                conv /= (S_new * S_new)
        
        obj_after = float(np.max(conv))
        if obj_after < best_obj - 1e-13:
            best_obj = obj_after
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 3 and rng is not None:
                scale = 0.01 * (1 + stagnant * 0.5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv = compute_conv(a)
                best_obj = float(np.max(conv))
                stagnant = 0
            elif stagnant > 8:
                break
    
    return best_a


def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def init_seq(n, rng, mode=0):
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 0.01))
        return a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 4:
        # Beta(2,2) shape
        x = np.linspace(0, 1, n)
        a = 4 * x * (1 - x)
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif mode == 5:
        # Arc-sine (U-shaped)
        x = np.linspace(0.01, 0.99, n)
        a = 1.0 / (np.pi * np.sqrt(x * (1 - x)))
        return a / a.sum()
    elif mode == 6:
        # Two peaks
        c = n // 2
        w = n // 6
        a = np.exp(-((np.arange(n) - c * 0.5) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c * 1.5) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif mode == 7:
        # Linear ramp
        a = np.linspace(1, 0.1, n)
        return a / a.sum()
    else:
        a = rng.uniform(0.1, 1, n)
        return a / a.sum()

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses iterative LP polishing (best-response dynamics) as the primary optimizer,
combined with fast coordinate descent, constraint-subset LP acceleration,
multi-length exploration, and diverse restarts.
Key improvements: adaptive constraint generation for LP speed, gap-based
convergence detection, targeted perturbation, and systematic n sweep.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    a = np.asarray(a, dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def fast_lp(a, g_init=None, subset_size=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Optionally use only a subset of the tightest constraints for speed.
    Returns (g_val, M_val, active_indices) or (None, None, None).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    A_full = build_conv_matrix(a)
    
    # Determine which constraints to use
    if subset_size is not None and g_init is not None:
        # Use the tightest constraints based on g_init
        conv_test = A_full @ g_init
        M_test = float(np.max(conv_test))
        # Pick the top subset_size tightest + a margin
        thresholds = np.sort(conv_test)[::-1]
        if subset_size < len(thresholds):
            thresh = max(thresholds[subset_size] * 1.05, M_test * 0.5)
        else:
            thresh = -np.inf
        active = np.where(conv_test >= thresh)[0]
        if len(active) < 10:
            active = np.argsort(conv_test)[::-1][:max(10, subset_size)]
        A = A_full[active, :]
    else:
        A = A_full
        active = np.arange(A.shape[0])
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    g_val = None
    M_val = None
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                M_val = float(M.value) if M.value is not None else None
                break
        except Exception:
            continue
    
    return g_val, M_val, active


def lp_polish(a, time_budget=60.0, rng=None, verbose=False):
    """
    Iterative LP polishing with adaptive constraint subsets and blending.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if rng is None:
        rng = np.random.default_rng(777)
    
    last_t = 0.5
    iter_count = 0
    stagnant = 0

    while time.monotonic() - start < time_budget:
        # Use constraint subset for speed after first few iterations
        if iter_count < 2:
            subset = None
        elif iter_count < 5:
            subset = max(n, 2 * n // 3)
        else:
            subset = n  # Use n constraints (roughly half of 2n-1)
        
        g_val, M_val, active = fast_lp(a, g_init=g_last, subset_size=subset)
        
        if g_val is None:
            # Retry with full LP
            g_val, M_val, active = fast_lp(a, g_init=None, subset_size=None)
            if g_val is None:
                break
        
        sg = float(np.sum(g_val))
        if sg < 1e-10:
            break
        g_val = g_val / sg
        g_last = g_val.copy()

        conv_aa = compute_conv(a)

def coord_descent(a, time_budget=60.0, rng=None, verbose=False):
    """Coordinate descent with golden section search on normalized self-conv peak."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
