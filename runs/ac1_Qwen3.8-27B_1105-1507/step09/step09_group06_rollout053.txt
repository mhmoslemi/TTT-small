"""
Optimizes step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses aggressive LP-based Frank-Wolfe with active set reduction, line search,
over-relaxation, multi-length exploration, and diversified restarts.
Key improvements over previous: proper line search, active set reduction for speed,
over-relaxation, more LP iterations, wider n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy import sparse
from scipy.optimize import minimize_scalar
import cvxpy as cp
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix_sparse(a):
    n = len(a)
    j_idx = np.repeat(np.arange(n), n)
    i_idx = np.tile(np.arange(n), n)
    row_idx = j_idx + i_idx
    vals = a[i_idx]
    return sparse.csr_matrix((vals, (row_idx, j_idx)), shape=(2 * n - 1, n))


def solve_lp(a, constraint_rows=None, time_limit=10.0):
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a = a / s

    A_full = build_conv_matrix_sparse(a)
    if constraint_rows is not None and len(constraint_rows) < 2 * n - 1:
        A = A_full[list(constraint_rows), :]
    else:
        A = A_full

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

            g_val = np.maximum(g.value, 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(M.value)

        pass

        prob.solve(solver=cp.MOSEK)
        if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
            g_val = np.maximum(g.value, 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(M.value)
    except Exception:
        pass
