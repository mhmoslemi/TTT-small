"""
Adaptive Smooth Mirror Descent for minimizing peak autocorrelation of non-negative sequences.
Uses log-sum-exp smoothing of the max-autocorrelation objective, with FFT-based gradient
computation for efficiency. Combines with LP polishing for final refinement.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_smooth_gradient(a, w):
    """
    Compute grad[j] = 2 * sum_k w[k] * a[k-j] for j=0..n-1
    where a has length n and w has length 2n-1.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    N = 1
    while N < 2 * (2 * n - 1):  # Need N >= len(w) + len(a) - 1 = 3n-2
        N <<= 1
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(W * np.conj(A), N)
    return 2.0 * R[:n]

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode % 6 == 0:
        a = np.ones(n) / n
    elif mode % 6 == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode % 6 == 2:
        x = np.linspace(0, 1, n)
        # Try a chirp-like shape
        a = np.abs(np.cos(np.pi * x * x * 3)) + 0.1
        a = a / a.sum()
    elif mode % 6 == 3:
        x = np.linspace(0, 1, n)
        # Multi-frequency
        freqs = rng.integers(2, 10, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 6 == 4:
        # Gaussian mixture
        centers = rng.uniform(0.1, 0.9, size=3)
        widths = rng.uniform(0.05, 0.2, size=3)
        x = np.linspace(0, 1, n)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-0.5 * ((x - c) / w) ** 2)
        a = a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n) * 0.3)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-6)

"""
Smooth Mirror Descent + LP Polish for minimizing peak autocorrelation.
Uses log-sum-exp smoothing with FFT-based gradients for rapid exploration,
structured initializations (chirps, multi-frequency), n-sweep, and LP polish.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_smooth_grad(a, w):
    """
    Compute gradient of log-sum-exp objective.
    grad[j] = 2 * sum_k w[k] * a[k-j] for j=0..n-1.
    a has length n, w has length 2n-1.
    Uses FFT cross-correlation: O(n log n).
    """
    n = len(a)
    N = 1
    while N < 4 * n:
        N <<= 1
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(W * np.conj(A), N)
    return 2.0 * R[:n]


def smooth_mirror_descent(a_init, time_budget, beta_start=10.0, beta_end=3000.0,
                          lr_scale=1.0, verbose=False, tag=""):
    """
    Smooth mirror descent on the simplex with adaptive beta and learning rate.
    Minimizes log-sum-exp of autocorrelation entries as smooth proxy for max.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = beta_start
    lr = lr_scale * 2.0 / np.sqrt(n)
    momentum = 0.5
    velocity = np.zeros(n)
    it = 0
    no_improve = 0
    max_no_improve = 100

    estimated_iters = max(50, int(time_budget * 30))

    while time.monotonic() - start < time_budget:
        frac = min(1.0, it / estimated_iters)
        beta = beta_start * (beta_end / beta_start) ** frac

        # Compute autocorrelation
        b = compute_conv(a)
        b_max = float(b.max())

        # Softmax weights (numerically stable)
        shifted = beta * (b - b_max)
        shifted = np.clip(shifted, -60.0, 0.0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(b))
            w[int(np.argmax(b))] = 1.0
        else:
            w /= w_sum

        # Gradient via FFT
        grad = compute_smooth_grad(a, w)

        # Momentum
        velocity = momentum * velocity + (1.0 - momentum) * grad

        # Mirror descent (exponential weights)
        log_a = np.log(a) - lr * velocity
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-300:
            break
        a_new = a_new / s_new

        # Evaluate
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Still accept the step for exploration but track best
            a = a_new.copy()
            if no_improve > max_no_improve:
                lr *= 0.85
                momentum = max(0.3, momentum * 0.9)
                no_improve = 0
            elif no_improve > 50:
                lr *= 0.95

        it += 1
        if verbose and it % 300 == 0:
            elapsed = time.monotonic() - start
            print(f"    SMD{tag} it{it}: obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.5f}, elapsed={elapsed:.1f}s")

    return best_a


def projected_gradient_descent(a_init, time_budget, beta_start=50.0, beta_end=5000.0, verbose=False, tag=""):
    """
    Projected gradient descent onto simplex with log-sum-exp smoothing.
    Different geometry than mirror descent - useful for final polishing.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = beta_start
    lr = 5.0 / (n * beta)
    it = 0
    no_improve = 0
    estimated_iters = max(50, int(time_budget * 20))

    while time.monotonic() - start < time_budget:
        frac = min(1.0, it / estimated_iters)
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = 5.0 / (n * beta * (1.0 + 0.01 * it))

        b = compute_conv(a)
        b_max = float(b.max())
        shifted = beta * (b - b_max)
        shifted = np.clip(shifted, -60.0, 0.0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(b))
            w[int(np.argmax(b))] = 1.0
        else:
            w /= w_sum

        grad = compute_smooth_grad(a, w)

        # Gradient step
        a_new = a - lr * grad

        # Project onto simplex {x: sum(x)=1, x>=0}
        a_new = proj_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                lr *= 0.8
                no_improve = 0

        it += 1
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    PGD{tag} it{it}: obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")

    return best_a


def proj_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    guid = int(np.max(ind[cond])) if np.any(cond) else 1
    theta = float(cssv[guid - 1] / guid)
    return np.maximum(v - theta, 0.0)


def lp_polish(a_init, time_budget, verbose=False, tag=""):
    """
    LP-based polishing. At each step, solve LP to find optimal g for current a,
    then interpolate. Different from SMD in that it takes optimal (greedy) steps.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        a_norm = a / a.sum()

        # Build convolution matrix: A[k,j] = a_norm[k-j] if 0<=k-j<n else 0
        A_mat = np.zeros((2 * n - 1, n))
        for j in range(n):
            k_lo = max(0, j)
            k_hi = min(2 * n - 2, j + n - 1)
            if k_lo <= k_hi:
                A_mat[k_lo:k_hi + 1, j] = a_norm[0:k_hi - k_lo + 1[::-1] if False else k_hi - k_lo + 1]
            # Actually simpler:
            for k in range(k_lo, k_hi + 1):
                A_mat[k, j] = a_norm[k - j]

        # Solve LP: min M s.t. A_mat @ g <= M, sum(g)=1, g>=0
        try:
            c_vec = np.zeros(n + 1)
            c_vec[-1] = 1.0
            A_ub = np.zeros((2 * n - 1, n + 1))
            A_ub[:, :n] = A_mat
            A_ub[:, n] = -1.0
            b_ub = np.zeros(2 * n - 1)
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.array([1.0])
            bounds = [(0, None)] * n + [(None, None)]

            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'presolve': True})
            if res.status == 0:
                g_val = np.maximum(res.x[:n], 0.0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:
                    g_val = g_val / g_sum
                    # Try multiple step sizes
                    for t in [1.0, 0.7, 0.5, 0.3, 0.8]:
                        a_try = (1.0 - t) * a + t * g_val
                        a_try = np.maximum(a_try, 0.0)
                        s_try = float(a_try.sum())
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-13:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try.copy()
                            break
        except Exception:
            pass

        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP{tag} it{it}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_a


def chirp_init(n, mode=0, rng=None):
    """Generate chirp-like initializations with flat power spectrum tendency."""
    x = np.arange(n) / n
    if mode == 0:
        # Quadratic chirp magnitude
        a = np.abs(np.cos(np.pi * x * x * 5)) + 0.1
    elif mode == 1:
        # Linear chirp with envelope
        a = np.abs(np.sin(2 * np.pi * x * x * 7)) + 0.2
    elif mode == 2:
        # Multi-frequency
        a = np.ones(n)
        for f in [3, 5, 8, 13]:
            a += 0.4 * np.cos(2 * np.pi * f * x + 0.5 * f)
        a = np.maximum(a, 0.05)
    elif mode == 3:
        # Gaussian chirp
        env = np.exp(-4 * (x - 0.5) ** 2)
        a = env * (0.5 + 0.5 * np.cos(2 * np.pi * x * x * 10)) + 0.05
    elif mode == 4:
        # Random-like with smooth envelope
        if rng is None:
            rng = np.random.default_rng(0)
        a = rng.uniform(0.3, 1.0, n)
        # Smooth it
        k = max(3, n // 50)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.05)
    elif mode == 5:
        # Beta distribution shaped
        if rng is None:
            rng = np.random.default_rng(1)
        a = 1.0 + 0.5 * np.sin(np.pi * x) ** 2 + 0.3 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.05)
    else:
        a = np.ones(n) / n
    a = a / a.sum()
    return a


def diverse_init(n, rng, mode=0):
    """Diverse initializations."""
    if mode % 8 == 0:
        a = np.ones(n) / n
    elif mode % 8 == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode % 8 == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.sin(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 8 == 3:
        freqs = rng.integers(2, 12, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    elif mode % 8 == 4:
        centers = rng.uniform(0.1, 0.9, size=3)
        widths = rng.uniform(0.05, 0.25, size=3)
        x = np.linspace(0, 1, n)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-0.5 * ((x - c) / w) ** 2)
        a = a / a.sum()
    elif mode % 8 == 5:
        a = rng.dirichlet(np.ones(n) * 0.5)
    elif mode % 8 == 6:
        # Tapered
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 0.5 + 0.1
        a = a / a.sum()
    else:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** 0.3 + (1 - x) ** 0.3 + 0.1
        a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-6)


def multiple_peak_reduction(a, time_budget, verbose=False):
    """
    Targeted approach: identify top peak lags, try to reduce them by redistributing mass.
    Uses gradient information to guide moves.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        b = compute_conv(a)
        b_max = float(b.max())

        # Find top 5 peak lags
        peak_lags = np.argsort(b)[::-1][:5]

        # For each peak, compute gradient and try to reduce
        improved_this_round = False
        for k_star in peak_lags:
            k_star = int(k_star)
            # Gradient at this specific lag: grad[j] = 2*a[k_star-j] (if valid)
            grad = np.zeros(n)
            k_lo = max(0, k_star - n + 1)
            k_hi = min(n - 1, k_star)
            for j in range(k_lo, k_hi + 1):
                grad[j] = 2.0 * a[k_star - j]

            # Move mass from high-grad positions to low-grad positions
            # (reduce contribution to this peak)
            src_candidates = np.argsort(grad)[::-1][:10]
            dst_candidates = np.argsort(grad)[:10]

            for src in src_candidates[:5]:
                for dst in dst_candidates[:5]:
                    if src == dst:
                        continue
                    if a[src] < 1e-8:
                        continue
                    for frac in [0.005, 0.01, 0.02, 0.03]:
                        delta = frac * a[src]
                        if delta >= a[src] * 0.5:
                            continue
                        a[src] -= delta
                        a[dst] += delta
                        a = np.maximum(a, 0)
                        s = a.sum()
                        if s < 1e-10:
                            a[src] += delta
                            a[dst] -= delta
                            continue
                        a = a / s
                        obj = compute_obj(a)
                        if obj < best_obj - 1e-13:
                            best_obj = obj
                            best_a = a.copy()
                            improved_this_round = True
                            break
                        else:
                            a[src] += delta
                            a[dst] -= delta
                            break
                if improved_this_round:
                    break
            if improved_this_round:
                break

        if not improved_this_round and it > 20:
            break

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    PeakRed it{it}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_a
