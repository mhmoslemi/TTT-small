"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated smoothed sub-gradient descent with Barzilai-Borwein step size,
progressive-constraint LP polishing, and multi-length exploration.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_idx = np.where(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_max_grad(a, tau):
    """
    Compute gradient of (1/tau) * log(sum_k exp(tau * conv_k)) w.r.t. a.
    This is a smooth approximation of max_k conv_k.
    
    Returns (gradient, conv_values, smooth_max_value).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    smooth_val = max_conv + np.log(float(np.sum(exp_vals))) / tau
    
    # Gradient: d/d a_m of smooth_val
    # = sum_k p_k * d(conv_k)/d(a_m)
    # = sum_k p_k * 2 * a_{k-m} (if 0 <= k-m < n)
    # This is a cross-correlation of p and a!
    
    # (cross_corr_p_a)[m] = sum_k p_k * a_{k-m}
    # = (p * a_flipped reversed)[m] in some sense
    
    # More precisely: we want sum_k p_k * a_{k-m} for k = 0..2n-2, with a_{k-m} = 0 if k-m not in [0, n-1]
    # This is the cross-correlation of p (length 2n-1) and a (length n), evaluated at lag m.
    # = ifft(fft(p, N) * conj(fft(a, N)))[:n] ... let me verify.
    
    # Actually: sum_k p_k * a_{k-m} = (p \star a)[m] where \star is cross-correlation.
    # In FFT: cross_correlation = ifft(fft(p) * conj(fft(a)))
    
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    xc = ifft(P * np.conj(A)).real
    # xc[m] = sum_k p_k * a_{k-m} (approximately, need to check indices)
    
    # Actually, ifft(fft(p) * conj(fft(a)))[m] = sum_k p_k * a_{(k-m) mod N}
    # For our purposes, since p has length 2n-1 and a has length n, and N >= 3n,
    # the circular wrap-around doesn't affect the first n elements.
    
    grad = 2.0 * xc[:n]
    return grad, conv, smooth_val


def nesterov_smoothed(a, time_budget=200.0, tau_start=5.0, tau_end=500.0, verbose=False):
    """
    Nesterov accelerated gradient descent on the smooth approximation.
    Uses Barzilai-Borwein step size adaptation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = 0.9
    lr = 1.0 / math.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        # Exponential tau schedule: from tau_start to tau_end
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.5)

        if it % 5000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}")

        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smooth gradient
        g, conv, smooth_val = smoothed_max_grad(y, tau)

        # Barzilai-Borwein step size
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / max(abs(sy), 1e-12)
                lr = 0.3 * lr + 0.7 * min(max(bb_lr, 1e-10), 50.0 / n)

        # Update
        v = momentum * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
            lr = 1.0 / math.sqrt(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall_count = 0
        else:
            stall_count += 1

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1

        # Adaptive momentum
        if stall_count > 5000:
            # Restart with perturbation
            a = best_a.copy()
            v = np.zeros(n)
            lr = 1.0 / math.sqrt(n)
            momentum = min(0.95, momentum + 0.05)
            stall_count = 0
            best_it = it

        # Decay learning rate slowly
        lr = max(1e-10, lr * 0.9999)

    return best_a


def solve_lp_reduced(a, top_k=30, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for top-k peaks, sum(g)=1, g>=0.
    Returns optimal g.
    """
    n = len(a)
    a = a / float(np.sum(a))
    conv_a = compute_conv(a)
    
    # Get top-k indices
    top_idx = np.argsort(conv_a)[::-1][:top_k]
    
    # Build constraint matrix for only top-k rows
    # (A @ g)[k] = sum_i a[i] * g[k-i] (valid range)
    # For each k in top_idx, the constraint is: sum_{i: 0<=k-i<n} a[i] * g[k-i] <= M
    
    # Build the matrix efficiently
    A_rows = []
    for k in top_idx:
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            if 0 <= j < n:
                row[i] = a[i]  # Wait, this isn't right
        A_rows.append(row)
    
    # Actually, (a*g)[k] = sum_i a[i] * g[k-i]
    # So the row for constraint k is: [a[k-(n-1)], a[k-(n-2)], ..., a[k]] (padded with 0s)
    # i.e., row[i] = a[i] if (k-i) is in [0, n-1], else 0
    # Wait no: (a*g)[k] = sum_{i=0}^{n-1} a[i] * g[k-i] where g[j] = 0 if j not in [0,n-1]
    # So the coefficient of g[j] in constraint k is a[k-j] if 0 <= k-j < n, else 0.
    # Row k: for each j in 0..n-1, coefficient = a[k-j] if 0 <= k-j < n else 0
    
    A_reduced = np.zeros((len(top_idx), n))
    for idx, k in enumerate(top_idx):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_reduced[idx, j] = a[i]
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [
        A_reduced @ g_var <= M_var,
        cp.sum(g_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M_var.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget=300.0, top_k_start=20, top_k_end=100, verbose=False):
    """
    LP polishing with progressive constraints.
    Start with few constraints (fast), add more as solution stabilizes.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        # Determine current top_k (progressive)
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        top_k = top_k_start + int(progress * (top_k_end - top_k_start))
        top_k = min(top_k, 2 * n - 1)

        g_lp, M_lp = solve_lp_reduced(a, top_k=top_k)
        if g_lp is None:
            # Fallback: use sub-gradient direction
            g, conv, _ = smoothed_max_grad(a, 50.0)
            g_lp = project_simplex(-g)
            M_lp = None

        # Line search
        best_t = 0.0
        best_f = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                f_try = compute_obj(a_try)
                if f_try < best_f:
                    best_f = f_try
                    best_t = t

        if best_t > 0:
            a_new = (1 - best_t) * a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 5 == 0:
            print(f"    LPP it{it}: obj={best_obj:.8f}, top_k={top_k}, stall={stall}")

        if stall > 30:
            # Perturb and restart
            scale = 0.03
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s_a = float(np.sum(a))
            a = a / s_a
            stall = 0
            if it > 100:
                break

    return best_a


def coord_descent_peaks(a, time_budget=100.0, verbose=False):
    """
    Coordinate descent targeting the top peaks of the convolution.
    For each peak, try to reduce it by adjusting the contributing elements.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(77777)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Get top few peaks
        top_peaks = np.argsort(conv)[::-1][:5]
        
        improved = False
        for k in top_peaks:
            # Find contributing pairs
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            
            # Try reducing the largest contributor and distributing to non-contributors
            max_contrib = -1
            max_i = -1
            for i in range(i_min, i_max + 1):
                j = k - i
                if 0 <= j < n:
                    c = a[i] * a[j]
                    if c > max_contrib:
                        max_contrib = c
                        max_i = i
            
            if max_i < 0:
                continue
            
            # Try several targets (indices that don't contribute to this peak)
            targets = []
            for _ in range(10):
                t = int(rng.integers(0, n))
                if not (i_min <= t <= i_max and i_min <= k - t <= i_max):
                    targets.append(t)
            
            for tgt in targets[:5]:
                for frac in [0.1, 0.05, 0.02]:
                    delta = frac * a[max_i]
                    if delta > a[max_i]:
                        continue
                    a_try = a.copy()
                    a_try[max_i] -= delta
                    a_try[tgt] += delta
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-15:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
        
        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break

        if verbose and it % 20 == 0:
            print(f"    CD it{it}: obj={best_obj:.8f}")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two-hump
        c1, c2 = n // 4, 3 * n // 4
        w = n // 10
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Cosine taper
        x = np.arange(n) / (n - 1)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a / a.sum()
    elif mode == 4:
        # Power law
        x = np.arange(1, n + 1)
        a = x ** rng.uniform(-1.5, 1.5)
        a = a / a.sum()
    elif mode == 5:
        # Triangular
        a = np.minimum(np.arange(n), n - 1 - np.arange(n)) + 1
        a = a / a.sum()
    elif mode == 6:
        # Sine-squared
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated smoothed sub-gradient descent with Barzilai-Borwein step size,
progressive-constraint LP polishing, peak-targeted coordinate descent, and multi-length exploration.
Key innovations: FFT-based cross-correlation gradient, adaptive tau schedule, progressive LP constraints,
and smart restart with dual-guided perturbation.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_idx = np.where(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_max_grad(a, tau):
    """
    Compute gradient of smooth max of convolution using FFT cross-correlation.
    Returns (gradient, conv_values, smooth_max_value).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    smooth_val = max_conv + np.log(float(np.sum(exp_vals))) / tau

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    xc = ifft(P * np.conj(A)).real
    grad = 2.0 * xc[:n]
    return grad, conv, smooth_val

    Nesterov accelerated gradient descent on smooth max-conv objective.
    Uses Barzilai-Borwein step size adaptation and adaptive momentum.
    """

    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = 0.85
    lr = 2.0 / math.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    stall_count = 0
    rng = np.random.default_rng(54321)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.6)

        if it % 5000 == 0 and verbose:

        y = np.maximum(y, 0)
        s_y = float(np.sum(y))

def build_conv_constraint_matrix(a, top_indices):
    """Build constraint matrix rows for conv(a, g) at specified indices."""
    n = len(a)
    A = np.zeros((len(top_indices), n))

            if 0 <= i < n:
                A[idx, j] = a[i]
    return A


def solve_lp_reduced(a, top_k=30):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for top-k peaks, sum(g)=1, g>=0.
    Returns (g, M) or (None, None) on failure.
    """
    n = len(a)
    a = a / float(np.sum(a))
    conv_a = compute_conv(a)
    top_idx = np.argsort(conv_a)[::-1][:top_k]

    A = build_conv_constraint_matrix(a, top_idx)

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [
        A @ g_var <= M_var,
        cp.sum(g_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M_var.value)
        except Exception:
            continue
    return None, None


def golden_section_line_search(a, direction, n_evals=10):
    """Golden section search for optimal blend t in [0,1]."""
    phi = (np.sqrt(5) - 1) / 2
    lo = 0.0
    hi = 1.0
    f_lo = np.inf
    f_hi = np.inf

    # Evaluate endpoints
    s_lo = float(np.sum(a))
    if s_lo > 1e-10:
        f_lo = compute_obj(a / s_lo)
    s_hi = float(np.sum(direction))
    if s_hi > 1e-10:
        f_hi = compute_obj(direction / s_hi)

    for _ in range(n_evals):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)

        a1 = (1 - t1) * a + t1 * direction
        a1 = np.maximum(a1, 0)
        s1 = float(np.sum(a1))
        f1 = compute_obj(a1 / s1) if s1 > 1e-10 else np.inf

        a2 = (1 - t2) * a + t2 * direction
        a2 = np.maximum(a2, 0)
        s2 = float(np.sum(a2))
        f2 = compute_obj(a2 / s2) if s2 > 1e-10 else np.inf

        if f1 < f2:
            hi = t2
            f_hi = f2
        else:
            lo = t1
            f_lo = f1

    t_best = (lo + hi) / 2
    a_best = (1 - t_best) * a + t_best * direction
    a_best = np.maximum(a_best, 0)
    s_best = float(np.sum(a_best))
    if s_best > 1e-10:
        a_best = a_best / s_best
    return a_best


def lp_polish(a, time_budget=300.0, top_k_start=15, top_k_end=120, verbose=False):
    """
    LP polishing with progressive constraints and golden section line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(99999)

    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        top_k = top_k_start + int(progress * (top_k_end - top_k_start))
        top_k = min(top_k, 2 * n - 1)

        g_lp, M_lp = solve_lp_reduced(a, top_k=top_k)
        if g_lp is None:
            # Fallback: use negative sub-gradient projected to simplex
            g, conv, _ = smoothed_max_grad(a, 100.0)
            g_lp = project_simplex(-g)

        # Golden section line search
        a_new = golden_section_line_search(a, g_lp, n_evals=8)
        obj_new = compute_obj(a_new)

        it += 1
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            # Try smaller steps
            found = False
            for t in [0.1, 0.05, 0.02, 0.01]:
                a_try = (1 - t) * a + t * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    f_try = compute_obj(a_try)
                    if f_try < best_obj - 1e-15:
                        best_obj = f_try
                        best_a = a_try.copy()
                        a = a_try
                        stall = 0
                        found = True
                        break
            if not found:
                if stall > 20:
                    # Perturb and restart
                    scale = min(0.05, 0.01 + 0.005 * (stall // 10))
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    s_a = float(np.sum(a))
                    a = a / s_a
                    stall = 0
                    if it > 80:
                        break

        if verbose and it % 5 == 0:
            print(f"    LPP it{it}: obj={best_obj:.8f}, top_k={top_k}, stall={stall}")

    return best_a


def coord_descent_peaks(a, time_budget=100.0, verbose=False):
    """
    Coordinate descent targeting the top peaks.
    Tries moving mass from peak contributors to non-contributors.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(31415)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_peaks = np.argsort(conv)[::-1][:5]

        improved = False
        for k in top_peaks:
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)

            # Find largest contributor
            max_contrib = -1
            max_i = -1
            for i in range(i_min, i_max + 1):
                j = k - i
                if 0 <= j < n:
                    c = a[i] * a[j]
                    if c > max_contrib:
                        max_contrib = c
                        max_i = i

            if max_i < 0:
                continue

            # Try targets that don't contribute to this peak
            for _ in range(8):
                tgt = int(rng.integers(0, n))
                tgt_j = k - tgt
                if 0 <= tgt_j < n and i_min <= tgt <= i_max:
                    continue  # This target also contributes
                for frac in [0.08, 0.04, 0.02, 0.01]:
                    delta = frac * a[max_i]
                    if delta > a[max_i] * 0.9:
                        continue
                    a_try = a.copy()
                    a_try[max_i] -= delta
                    a_try[tgt] += delta
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-15:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break

        if verbose and it % 20 == 0:
            print(f"    CD it{it}: obj={best_obj:.8f}")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 10)
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.arange(n) / max(1, n - 1)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a / a.sum()
    elif mode == 4:
        x = np.arange(1, n + 1, dtype=np.float64)
        a = x ** rng.uniform(-1.5, 1.5)
        a = a / a.sum()
    elif mode == 5:
        a = np.minimum(np.arange(n, dtype=np.float64), n - 1 - np.arange(n, dtype=np.float64)) + 1.0
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2
        a = a / a.sum()
    else:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 0.001))
        a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization:
    1. Fast Nesterov smoothed descent (BB step size, adaptive tau)
    2. Progressive-constraint LP polishing with golden section
    3. Higher-tau Nesterov for fine-tuning
    4. More LP refinement
    5. Multi-length exploration
    6. Coordinate descent
    7. Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        print("No parent available, using default init")

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Fast Nesterov smoothed descent ===
    print(f"\n=== Phase 1: Nesterov Smoothed Descent (fast) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.18, 250.0)
    if p1_budget > 30:
        a1 = nesterov_smoothed(best_seq, time_budget=p1_budget,
                               tau_start=5.0, tau_end=300.0, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj1:.8f}")

    # === Phase 2: LP polishing with progressive constraints ===
    print(f"\n=== Phase 2: LP Polishing (progressive) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.28, 400.0)
    if p2_budget > 50:
        a2 = lp_polish(best_seq, time_budget=p2_budget,
                       top_k_start=15, top_k_end=100, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj2:.8f}")

    # === Phase 3: Higher-tau Nesterov for fine-tuning ===
    print(f"\n=== Phase 3: High-Tau Nesterov ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.18, 200.0)
    if p3_budget > 30:
        a3 = nesterov_smoothed(best_seq, time_budget=p3_budget,
                               tau_start=50.0, tau_end=1500.0, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj3:.8f}")

    # === Phase 4: LP refinement with more constraints ===
    print(f"\n=== Phase 4: LP Refinement ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.22, 300.0)
    if p4_budget > 30:
        a4 = lp_polish(best_seq, time_budget=p4_budget,
                       top_k_start=30, top_k_end=200, verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase 4: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4: obj={obj4:.8f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(200, n_current - 200),
        max(200, n_current - 100),
        n_current + 100,
        n_current + 200,
        600, 700, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1500 and abs(n - n_current) > 50]

    for n_try in n_options[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        alloc_ns = min(remaining * 0.15, 70.0)
        alloc_lp = min(remaining * 0.15, 50.0)

        if alloc_ns > 20:
            a_try = nesterov_smoothed(a_try, time_budget=alloc_ns,
                                       tau_start=10.0, tau_end=200.0)
        remaining = deadline - time.monotonic()
        if alloc_lp > 20 and remaining > 30:
            a_try = lp_polish(a_try, time_budget=min(alloc_lp, remaining * 0.4),
                              top_k_start=15, top_k_end=60)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p6_budget = min(remaining * 0.12, 80.0)
        a6 = coord_descent_peaks(best_seq, time_budget=p6_budget, verbose=False)
        obj6 = compute_obj(a6)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Phase 6: obj={best_score:.8f} **")
        else:
            print(f"  Phase 6: obj={obj6:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Very high tau Nesterov
    if remaining > 50:
        alloc = min(remaining * 0.25, 100.0)
        a_f = nesterov_smoothed(best_seq, time_budget=alloc,
                                tau_start=200.0, tau_end=5000.0)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final NSD: obj={best_score:.8f} **")

    # LP with many constraints
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(remaining * 0.35, 150.0)
        a_f = lp_polish(best_seq, time_budget=alloc,
                        top_k_start=50, top_k_end=300)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # One more Nesterov pass
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.5, 100.0)
        a_f = nesterov_smoothed(best_seq, time_budget=alloc,
                                tau_start=1000.0, tau_end=10000.0)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final2 NSD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
