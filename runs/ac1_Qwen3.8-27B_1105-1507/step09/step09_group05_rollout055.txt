"""

Multi-phase approach:
1. Parametric + parent initialization with evaluation
2. BFGS on log-sum-exp smoothed objective with softmax parameterization
3. Mirror descent (exponential/multiplicative updates) with tempered peak distribution
4. KKT-based polishing exploiting Hankel structure of autocorrelation gradients

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_crosscorr(a, p):
    """
    Compute q_j = sum_m a_m * p_{j+m} for j=0..n-1.
    This is the cross-correlation needed for gradient computation.
    a has length n, p has length 2n-1.
    """
    n = len(a)
    N = 1
    while N < n + (2 * n - 1):
        N <<= 1
    A = fft(a, N)
    P = fft(p, N)
    q_full = ifft(np.conj(A) * P).real
    return q_full[:n]


def smoothed_obj_and_grad(a, tau):
    """
    Compute log-sum-exp smoothed objective and gradient.
    a should be normalized (sum=1), non-negative.
    Returns (obj, grad) where grad is the gradient w.r.t. a (before simplex projection).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    # Avoid overflow
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = 2.0 * n * (max_conv + log_sum / tau)
    
    p = np.exp(shifted) / np.sum(np.exp(shifted))
    q = compute_crosscorr(a, p)
    grad = 4.0 * n * q
    return obj, grad, conv


def bfgs_softmax(a_init, time_budget=200.0, tau_init=50.0, tau_end=500.0):
    """
    BFGS optimization using softmax parameterization.
    Optimize over x in R^n where a = softmax(x).
    Uses log-sum-exp smoothed objective with annealed tau.
    """

    a_init = a_init / a_init.sum()
    x = np.log(a_init)  # Initial softmax parameter
    
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()
    
    def objective_and_grad(x_vec):
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)
        
        # Softmax (numerically stable)
        x_shifted = x_vec - np.max(x_vec)
        a = np.exp(x_shifted)
        a = a / a.sum()
        
        obj, grad_a, _ = smoothed_obj_and_grad(a, tau)
        
        # Chain rule: d(obj)/d(x_j) = a_j * (grad_a_j - sum_k a_k * grad_a_k)
        mean_grad = float(np.dot(a, grad_a))
        grad_x = a * (grad_a - mean_grad)
        
        return obj, grad_x
    
    # Use scipy BFGS with our custom objective
    from scipy.optimize import minimize as sp_minimize
    
    result = sp_minimize(
        objective_and_grad, x, method='BFGS', jac=True,
        options={'maxiter': 50000, 'gtol': 1e-10}
    )
    
    x_opt = result.x
    a_opt = np.exp(x_opt - np.max(x_opt))
    a_opt = a_opt / a_opt.sum()
    a_opt = np.maximum(a_opt, 0)
    
    obj_opt = compute_obj(a_opt)
    if obj_opt < best_obj:
        best_obj = obj_opt
        best_a = a_opt.copy()
    
    return best_a

def kkt_solve(a, p, support):
    """
    Solve the KKT linear system for given support and peak distribution p.
    System: sum_{m in S} p_{j+m} * a_m = lambda for j in S, sum(a_S) = 1.
    Returns (a_full, lambda) or None if no valid solution.
    """
    n = len(a)
    S = np.array(support, dtype=int)
    k = len(S)
    if k == 0:
        return None
    
    # Build the (k+1) x (k+1) system
    # First k rows: sum_m p_{j+m} * a_m - lambda = 0
    # Last row: sum_m a_m = 1
    H = np.zeros((k + 1, k + 1))
    rhs = np.zeros(k + 1)
    
    for i, j in enumerate(S):
        for m_idx, m in enumerate(S):
            idx = j + m
            if idx < len(p):
                H[i, m_idx] = p[idx]
        H[i, k] = -1.0  # coefficient of lambda
    
    # Normalization row
    H[k, :k] = 1.0
    H[k, k] = 0.0
    rhs[k] = 1.0
    
    try:
        sol = np.linalg.solve(H, rhs)
    except np.linalg.LinAlgError:
        # Try least squares
        try:
            sol, residuals, rank, sv = np.linalg.lstsq(H, rhs, rcond=None)
        except Exception:
            return None
    
    a_S = sol[:k]
    lam = sol[k]
    
    # Check non-negativity
    if np.any(a_S < -1e-10):
        return None
    
    a_full = np.zeros(n)
    a_full[S] = np.maximum(a_S, 0)
    s = a_full.sum()
    if s < 1e-10:
        return None
    a_full = a_full / s
    
    return a_full, lam

    KKT-based polishing: iteratively solve the KKT system with adaptive support.

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    
    while it < max_iter and (time.monotonic() - start) < time_limit:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Tempered peak distribution
        shifted = beta * (conv - max_conv)
        p = np.exp(shifted)
        p = p / p.sum()
        
        # Determine support: indices where a is "active"
        threshold = max(1e-8, 0.01 * np.mean(a))
        support = list(np.where(a > threshold)[0])
        
        if len(support) < 2:
            break
        
        # Solve KKT system
        result = kkt_solve(a, p, support)
        if result is None:
            # Try removing the smallest elements from support
            sorted_idx = np.argsort(a)
            for remove_count in range(min(5, len(support) // 10 + 1)):
                new_support = [i for i in support if i not in set(sorted_idx[:remove_count].tolist())]
                if len(new_support) < 2:
                    break
                result = kkt_solve(a, p, new_support)
                if result is not None:
                    break
            if result is None:
                # Try with all indices
                result = kkt_solve(a, p, list(range(n)))
                if result is None:
                    break
        
        a_kkt, lam = result
        obj_kkt = compute_obj(a_kkt)
        
        if obj_kkt < best_obj - 1e-14:
            best_obj = obj_kkt
            best_a = a_kkt.copy()
            a = a_kkt.copy()
            it += 1
        else:

                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-10:

                a_try = a_try / s_try

                    best_a = a_try.copy()
                    a = a_try.copy()
                    break

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):

    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish_round(a, max_iter=50, time_limit=80.0):
    """LP polishing: repeatedly solve LP for optimal g given a, then blend."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf, 0
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    no_improve = 0
    last_t = 0.5
    start = time.monotonic()
    it = 0
    
    while it < max_iter and (time.monotonic() - start) < time_limit:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0,
                      max(0.05, last_t * 0.5), max(0.05, last_t * 0.8),
                      min(1.0, last_t * 1.5)]
        
        best_t_obj = np.inf
        best_t_val = last_t
        best_a_local = best_a.copy()
        
        for t in candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        it += 1
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                break
    
    return best_a, best_obj, it


def perturb_sequence(a, rng, method=0, scale=0.05):
    """Apply a perturbation to escape local minima."""
    n = len(a)
    a = a.copy()
    if method == 0:
        a = a * np.exp(rng.normal(0, scale, n))
    elif method == 1:
        noise = rng.uniform(1 - scale, 1 + scale, n)
        a = a * noise
    elif method == 2:
        mask = rng.random(n) > scale * 2
        a = a * mask
        if a.sum() < 1e-10:
            a = np.ones(n) / n
    elif method == 3:
        shift = int(rng.integers(1, max(2, n // 10)))
        a = np.roll(a, shift)
    elif method == 4:
        a = (1 - scale) * a + scale / n
    elif method == 5:
        A = fft(a, 2 * n)
        A[1:-1] *= np.exp(1j * rng.normal(0, scale, len(A[1:-1])))
        a = ifft(A).real[:n]
        a = np.maximum(a, 0)
    elif method == 6:
        conv = compute_conv(a / a.sum())
        top_k = np.argsort(conv)[::-1][:5]
        deltas = np.zeros(n)
        for k in top_k:
            k = int(k)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    deltas[j] += a[idx] / a.sum()
        max_d = np.max(np.abs(deltas))
        if max_d > 1e-10:
            a = a - scale * 0.5 * deltas / max_d
        a = np.maximum(a, 0)
    elif method == 7:
        # Focus perturbation on elements contributing to peaks
        conv = compute_conv(a / a.sum())
        peak_idx = np.argmax(conv)
        contrib = np.zeros(n)
        for j in range(n):
            idx = peak_idx - j
            if 0 <= idx < n:
                contrib[j] = a[idx]
        contrib = contrib / (contrib.sum() + 1e-15)
        a = a * (1 + scale * (rng.random(n) - 0.5) * (1 + 3 * contrib / (contrib.max() + 1e-15)))
        a = np.maximum(a, 0)
    else:
        a = a * np.exp(rng.normal(0, scale, n))
    
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def parametric_init(n, mode=0):
    """Generate parametric initializations."""
    x = np.arange(n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        # Raised cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x / (n - 1))
    elif mode == 2:
        # Hann window
        a = 0.5 - 0.5 * np.cos(2 * np.pi * x / (n - 1))
    elif mode == 3:
        # Tukey window (t=0.5)
        t = 0.5
        a = np.ones(n)
        m = int(t * n // 2)
        if m > 0:
            a[:m] = 0.5 + 0.5 * np.cos(np.pi * (2 * np.arange(m) / (t * n) - 1))
            a[-m:] = 0.5 + 0.5 * np.cos(np.pi * (2 * np.arange(m) / (t * n) - 1))[::-1]
    elif mode == 4:
        # Parabolic (Quadratic) window
        center = (n - 1) / 2.0
        a = 1.0 - ((x - center) / center) ** 2
    elif mode == 5:
        # Gaussian window
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-((x - center) ** 2) / (2 * sigma ** 2))
    elif mode == 6:
        # Two-peak
        center = (n - 1) / 2.0
        a = np.exp(-((x - center * 0.3) ** 2) / (n * 0.02)) + np.exp(-((x - center * 0.7) ** 2) / (n * 0.02))
    elif mode == 7:
        # Flat top
        flat = 0.3
        a = np.ones(n)
        edge = int(flat * n)
        if edge > 1:
            a[:edge] = 0.5 + 0.5 * np.cos(np.pi * np.arange(edge) / (2 * edge) - np.pi / 2)
            a[-edge:] = a[:edge][::-1]
        a = 0.8 + 0.2 * a
    elif mode == 8:
        # Sinc-like (abs)
        center = (n - 1) / 2.0
        a = np.abs(np.sinc((x - center) / (n * 0.3))) + 0.1
    else:
        a = np.ones(n) / n
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def diverse_init(n, rng, mode=0):
    """Generate diverse random initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)


def refine_sequence(a, time_budget=120.0):
    """
    Full refinement pipeline for a single sequence:
    1. BFGS smoothed
    2. Mirror descent
    3. KKT polish
    4. LP polish
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Step 1: BFGS
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 30:
        t_bfgs = min(remaining * 0.3, 60.0)
        a_bfgs = bfgs_softmax(a, time_budget=t_bfgs, tau_init=30.0, tau_end=300.0)
        obj_bfgs = compute_obj(a_bfgs)
        if obj_bfgs < best_obj:
            best_obj = obj_bfgs
            a = a_bfgs
            print(f"      BFGS: obj={best_obj:.8f}")
    
    # Step 2: Mirror descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        t_md = min(remaining * 0.25, 40.0)
        a_md = mirror_descent(a, time_budget=t_md, beta_init=20.0, beta_end=200.0)
        obj_md = compute_obj(a_md)
        if obj_md < best_obj:
            best_obj = obj_md
            a = a_md
            print(f"      Mirror: obj={best_obj:.8f}")
    
    # Step 3: KKT polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        t_kkt = min(remaining * 0.2, 30.0)
        a_kkt, obj_kkt = kkt_polish_round(a, time_limit=t_kkt, beta=50.0)
        if obj_kkt < best_obj:
            best_obj = obj_kkt
            a = a_kkt
            print(f"      KKT: obj={best_obj:.8f}")
    
    # Step 4: LP polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t_lp = min(remaining * 0.35, 60.0)
        a_lp, obj_lp, _ = lp_polish_round(a, max_iter=40, time_limit=t_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            a = a_lp
            print(f"      LP: obj={best_obj:.8f}")
    
    # Step 5: Final KKT + LP combo
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t_final = remaining * 0.4
        a_kkt2, obj_kkt2 = kkt_polish_round(a, time_limit=t_final * 0.4, beta=100.0)
        if obj_kkt2 < best_obj:
            best_obj = obj_kkt2
            a = a_kkt2
        remaining2 = time_budget - (time.monotonic() - start)
        if remaining2 > 5:
            a_lp2, obj_lp2, _ = lp_polish_round(a, max_iter=20, time_limit=min(remaining2 * 0.5, 20.0))
            if obj_lp2 < best_obj:
                best_obj = obj_lp2
                a = a_lp2
    
    return a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Returns list of non-negative floats minimizing the evaluation objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # === Load parent ===
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")
    
    # === Phase 1: Try parametric inits and pick best ===
    print(f"\n=== Phase 1: Parametric Init Selection ===")
    n_current = len(best_seq)
    for mode in range(9):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        a_p = parametric_init(n_current, mode=mode)
        obj_p = compute_obj(a_p)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Parametric mode {mode}: obj={best_score:.8f} **")
    
    # Also try blending parent with parametric
    for mode in range(9):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        a_p = parametric_init(n_current, mode=mode)
        for blend in [0.3, 0.5, 0.7]:
            a_blend = (1 - blend) * best_seq + blend * a_p
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()
            obj_blend = compute_obj(a_blend)
            if obj_blend < best_score:
                best_score = obj_blend
                best_seq = a_blend.copy()
                print(f"  Blend mode {mode} t={blend}: obj={best_score:.8f} **")
    
    # === Phase 2: Full refinement on best sequence ===
    print(f"\n=== Phase 2: Full Refinement ===")
    remaining = deadline - time.monotonic()
    refine_budget = min(remaining * 0.25, 250.0)
    if refine_budget > 40:
        a_ref, obj_ref = refine_sequence(best_seq, time_budget=refine_budget)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"  After refinement: obj={best_score:.8f} **")
    
    # === Phase 3: KKT polish (intensive) ===
    print(f"\n=== Phase 3: Intensive KKT Polish ===")
    remaining = deadline - time.monotonic()
    kkt_budget = min(remaining * 0.2, 200.0)
    if kkt_budget > 20:
        # Multiple KKT rounds with different betas
        for kkt_round in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            beta_kkt = 30.0 + 20.0 * kkt_round
            alloc = min(kkt_budget / 5.0, remaining * 0.4)
            if alloc < 10:
                break
            a_kkt, obj_kkt = kkt_polish_round(best_seq, time_limit=alloc, beta=beta_kkt, max_iter=200)
            if obj_kkt < best_score:
                best_score = obj_kkt
                best_seq = a_kkt.copy()
                print(f"  KKT round {kkt_round} (beta={beta_kkt:.0f}): obj={best_score:.8f} **")
            else:
                print(f"  KKT round {kkt_round} (beta={beta_kkt:.0f}): obj={obj_kkt:.8f}")
                # Perturb and retry
                a_pert = perturb_sequence(best_seq, rng, method=kkt_round % 8, scale=0.03)
                a_kkt2, obj_kkt2 = kkt_polish_round(a_pert, time_limit=alloc * 0.5, beta=beta_kkt, max_iter=100)
                if obj_kkt2 < best_score:
                    best_score = obj_kkt2
                    best_seq = a_kkt2.copy()
                    print(f"  KKT round {kkt_round} (perturbed): obj={best_score:.8f} **")
    
    # === Phase 4: LP Polish (intensive) ===
    print(f"\n=== Phase 4: Intensive LP Polish ===")
    remaining = deadline - time.monotonic()
    lp_budget = min(remaining * 0.2, 250.0)
    if lp_budget > 30:
        for lp_round in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(lp_budget / 5.0, remaining * 0.5)
            if alloc < 15:
                break
            
            a_start = best_seq.copy()
            if lp_round > 0:
                a_start = perturb_sequence(a_start, rng, method=lp_round % 8, scale=0.02 + 0.01 * lp_round)
            
            a_lp, obj_lp, iters = lp_polish_round(a_start, max_iter=80, time_limit=alloc)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {lp_round}: obj={best_score:.8f} ({iters} iters) **")
            else:
                print(f"  LP round {lp_round}: obj={obj_lp:.8f} ({iters} iters)")
                
                # After LP, do KKT to clean up
                remaining = deadline - time.monotonic()
                if remaining > 10:
                    a_kkt3, obj_kkt3 = kkt_polish_round(a_lp, time_limit=min(remaining * 0.2, 20.0), beta=80.0)
                    if obj_kkt3 < best_score:
                        best_score = obj_kkt3
                        best_seq = a_kkt3.copy()
                        print(f"  LP+KKT round {lp_round}: obj={best_score:.8f} **")
    
    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    n_options = sorted(set([
        max(300, n_best - 200), max(300, n_best - 100), max(300, n_best - 50),
        n_best + 50, n_best + 100, n_best + 200, n_best + 400,
        400, 500, 600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1800 and n != n_best]
    n_options.sort(key=lambda x: abs(x - n_best))
    
    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        
        a_try = resample_sequence(best_seq, n_try)
        # Also try parametric init at this length
        a_param = parametric_init(n_try, mode=n_try % 9)
        a_blend = 0.6 * a_try + 0.4 * a_param
        a_blend = np.maximum(a_blend, 0)
        a_blend = a_blend / a_blend.sum()
        
        if compute_obj(a_blend) < compute_obj(a_try):
            a_try = a_blend
        
        alloc = min(remaining * 0.15, 80.0)
        if alloc > 20:
            a_ref2, obj_ref2 = refine_sequence(a_try, time_budget=alloc)
        else:
            a_ref2 = a_try
            obj_ref2 = compute_obj(a_ref2)
        
        if obj_ref2 < best_score:
            best_score = obj_ref2
            best_seq = a_ref2.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_ref2:.8f}")
    
    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        
        mode = r % 9
        if mode < 8:
            a_r = diverse_init(n_best, rng, mode=mode)
        else:
            a_r = parametric_init(n_best, mode=mode % 9)
        
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()
        
        alloc = min(remaining * 0.15, 60.0)
        if alloc > 25:
            a_ref3, obj_ref3 = refine_sequence(a_r, time_budget=alloc)
            if obj_ref3 < best_score:
                best_score = obj_ref3
                best_seq = a_ref3.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_ref3:.8f}")
    
    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    
    # Alternate KKT and LP several times
    for final_round in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        
        # KKT step
        alloc_kkt = min(remaining * 0.2, 30.0)
        if alloc_kkt > 8:
            a_kkt_f, obj_kkt_f = kkt_polish_round(best_seq, time_limit=alloc_kkt,
                                                    beta=50.0 + 20.0 * final_round, max_iter=150)
            if obj_kkt_f < best_score:
                best_score = obj_kkt_f
                best_seq = a_kkt_f.copy()
        
        # LP step
        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.2, 40.0)
        if alloc_lp > 8:
            a_lp_f, obj_lp_f, _ = lp_polish_round(best_seq, max_iter=30, time_limit=alloc_lp)
            if obj_lp_f < best_score:
                best_score = obj_lp_f
                best_seq = a_lp_f.copy()
        
        # Mirror descent step
        remaining = deadline - time.monotonic()
        alloc_md = min(remaining * 0.1, 20.0)
        if alloc_md > 8:
            a_md_f = mirror_descent(best_seq, time_budget=alloc_md, beta_init=50.0, beta_end=300.0)
            obj_md_f = compute_obj(a_md_f)
            if obj_md_f < best_score:
                best_score = obj_md_f
                best_seq = a_md_f.copy()
        
        if final_round % 2 == 0:
            print(f"  Final round {final_round}: obj={best_score:.8f}")
    
    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
