def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return float(2.0 * n * np.max(conv))


def compute_conv_and_max(a):
    """Compute convolution and return (conv, max_val, argmax)."""
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    argmax = int(np.argmax(conv))
    return conv, max_val, argmax

def active_set_lp(a, n_constraints=30, verbose=False):
    """
    Solve LP: minimize M s.t. conv(a, g)[k] <= M for top-k positions,
    sum(g) = 1, g >= 0.
    Returns optimal g.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv_a = compute_conv(a_norm)
    
    # Get top positions
    top_idx = np.argsort(conv_a)[::-1][:n_constraints]
    
    # Build constraint matrix for these positions only
    # For position k: (A@g)[k] = sum_i a_norm[i] * g[k-i] (valid range)
    # We need a matrix B of shape (n_constraints, n)
    B = np.zeros((n_constraints, n))
    for idx, k in enumerate(top_idx):
        # g[j] contributes to position k when a_norm[k-j] is valid
        # i.e., k-j in [0, n-1] => j in [k-n+1, k]
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            B[idx, j_lo:j_hi+1] = a_norm[k - (j_lo:j_hi+1)]  # a_norm[k-j] for j in range
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [B @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    
    return None

def full_lp(a):
    """Full LP: minimize M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    # Build full matrix
    A = np.zeros((2*n-1, n))
    for j in range(n):
        r = min(n, 2*n-1-j)
        A[j:j+r, j] = a_norm[:r]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    
    return None

            cur_a = new_a
            cur_obj = new_obj
            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_a = cur_a.copy()
                best_it = it
        
        it += 1
        
        if verbose and it % 5000 == 0:
            print(f"    SA it{it}: best={best_obj:.8f}, cur={cur_obj:.8f}, T={T:.6f}, time={elapsed:.1f}s")
        
        if it - best_it > 50000:
            cur_a = best_a.copy()
            cur_obj = best_obj
            best_it = it
    
    return best_a


def gradient_descent_smoothed(a, time_budget=100.0, verbose=False):
    """
    Gradient descent on smoothed max of convolution using FFT.
    a is normalized to sum=1.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    tau = 20.0
    lr = 5.0 / n
    it = 0
    best_it = 0
    v = np.zeros(n)  # momentum
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.95))
        tau = 20.0 * (500.0 / 20.0) ** (progress ** 0.5)
        
        # Compute smoothed gradient
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient of sum_k p_k * conv_k(a) w.r.t. a
        # = 2 * (p * a) convolved appropriately
        # Actually: d/d a[i] of conv_k(a) = a[k-i] + a[k-i] ... 
        # = 2 * sum_k p_k * a[k-i] (for the symmetric case)
        # More precisely: conv_k = sum_j a[j] * a[k-j]
        # d/d a[i] conv_k = a[k-i] + (if 2i != k) a[k-i] ... 
        # Actually d/d a[i] of sum_j a[j]*a[k-j] = a[k-i] (from j=i term) + a[k-i] (from k-j=i term) 
        # = a[k-i] + a[k-i] if k-i is valid and i != k-i
        # = 2*a[k-i] if i != k-i, a[k] if 2i = k
        
        # Using FFT: gradient[i] = 2 * sum_k p[k] * a[k-i] (with appropriate boundary handling)
        # This is 2 * (p * a)[i] where * is the "full correlation"
        
        N_fft = 1
        while N_fft < 3 * n:
            N_fft <<= 1
        
        P = fft(p, N_fft)
        A = fft(a, N_fft)
        # corr[i] = sum_k p[k] * a[k-i] = (p cross-correlated with a)[i]
        # This is ifft(fft(p) * conj(fft(a)))... let me think
        # Actually: (p * a)[i] in the correlation sense = sum_k p[k] a[k-i]
        # = ifft(FFT(p) * conj(FFT(a)))[i] (circular, with padding)
        
        corr = ifft(P * np.conj(A)).real[:n]
        grad = 2.0 * corr
        
        # Project gradient to tangent space of simplex
        grad -= np.mean(grad)
        
        # Update with momentum
        v = 0.9 * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        lr = max(lr * 0.9995, 0.1 / n)
        
        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 5.0 / n
            best_it = it
        
        if verbose and it % 2000 == 0:
            print(f"    GD it{it}: best={best_obj:.8f}, tau={tau:.1f}, time={elapsed:.1f}s")
    
    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations, emphasizing asymmetry."""
    if mode == 0:
        # Ramp (highly asymmetric)
        a = np.linspace(0.1, 1.0, n)
    elif mode == 1:
        # Exponential decay
        a = np.exp(-np.linspace(0, 3, n))
    elif mode == 2:
        # Two peaks asymmetric
        c1, c2 = n // 5, 4 * n // 5
        w1, w2 = n // 15, n // 10
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w1 ** 2)) + 0.5 * np.exp(-((np.arange(n) - c2) ** 2) / (2 * w2 ** 2))
    elif mode == 3:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(0.5, 3.0)
    elif mode == 4:
        # Sine wave (asymmetric phase)
        x = np.linspace(0, 2 * np.pi, n)
        a = 1.0 + 0.5 * np.sin(x + rng.uniform(0, np.pi))
        a = np.maximum(a, 0.01)
    elif mode == 5:
        # Random asymmetric
        a = rng.uniform(0, 1, n)
        # Apply random ramp
        ramp = np.linspace(0.5, 1.5, n)
        if rng.random() < 0.5:
            ramp = ramp[::-1]
        a = a * ramp
        a = np.maximum(a, 0)
    elif mode == 6:
        # Cosine window (symmetric but good starting point)
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif mode == 7:
        # Bump on one side
        c = rng.integers(n//4, n//2)
        w = max(1, n // 8)
        a = np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a += 0.1  # small baseline
    elif mode == 8:
        # Polynomial
        x = np.linspace(0, 1, n)
        p = rng.integers(1, 5)
        a = x ** p + 0.1
    else:
        a = np.ones(n) / n
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
