import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a, mode='minmax'):
    """
    Solve LP with a fixed:
    - mode='minmax': min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0
    - mode='minvar': min max - min s.t. conv(a,g)[k] in [m, M], sum(g)=1, g>=0
    - mode='minactive': min sum conv(a,g)[k] for k in active_set
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    
    if mode == 'minmax':
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
    elif mode == 'minvar':
        M = cp.Variable()
        m = cp.Variable()
        constraints = [A @ g <= M, A @ g >= m, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M - m), constraints)
    else:
        # shouldn't reach here
        return None
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:

            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:

    return None

def backtracking_line_search(a, direction, max_iter=20):
    """Find step size using backtracking."""
    n = len(a)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:

    a = a / s
    f0 = compute_obj(a)

    d = direction - a
    
    t = 1.0
    best_a = a.copy()
    best_f = f0
    
    for _ in range(max_iter):
        a_try = a + t * d
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:

        a_try = a_try / s_try

            best_a = a_try.copy()
        if f_try <= f0 * (1 - 1e-4 * t):

def spectral_smooth(a, alpha=0.1):

    while N < 2 * n:
        N <<= 1
    A = rfft(a, N)

            A[i] *= (1 - alpha)
    a_smooth = irfft(A, N)[:n]
    return np.maximum(a_smooth, 0)

def peak_shave(a, n_peaks=5, n_trials=10, verbose=False):
    """
    Directly shave peaks by moving mass from contributing elements.
    For each peak, identify the top contributing pairs and try moving mass.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(12345)
    
    for trial in range(n_trials):
        conv = compute_conv(best_a)
        # Get top peaks
        peak_indices = np.argsort(conv)[::-1][:n_peaks]
        improved = False
        
        for k in peak_indices:
            # Find contributing pairs
            i_range = range(max(0, k - n + 1), min(n, k + 1))
            contribs = []
            for i in i_range:
                j = k - i
                if 0 <= j < n:
                    contribs.append((best_a[i] * best_a[j], i, j))
            contribs.sort(reverse=True)
            
            # Try moving mass from top contributors
            for contrib_val, i, j in contribs[:5]:
                if i == j:
                    continue
                # Try reducing the larger of a[i], a[j]
                src = i if best_a[i] >= best_a[j] else j
                # Try several targets
                targets = sorted(rng.choice(n, size=min(20, n), replace=False).tolist())
                
                for tgt in targets[:5]:
                    if tgt == src:
                        continue
                    for frac in [0.01, 0.005, 0.002]:
                        delta = frac * best_a[src]
                        if delta <= 0 or best_a[src] - delta < 0:
                            continue
                        a_try = best_a.copy()
                        a_try[src] -= delta
                        a_try[tgt] += delta
                        a_try = a_try / a_try.sum()
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-12:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            
            if not improved:
                # Small random perturbation
                a_try = best_a * np.exp(rng.normal(0, 0.001, n))
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
        
        if verbose:
            print(f"    PeakShave trial {trial}: obj={best_obj:.8f}")
    
    return best_a

def bilinear_alternate(a, time_budget=200.0, verbose=False):
    """
    Bilinear alternating LP with multiple directions and backtracking.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)

    if s < 1e-10:
        return a

    start = time.monotonic()

    stall = 0
    
    while time.monotonic() - start < time_budget:
        # Direction 1: min-max LP
        g1 = solve_lp(a, mode='minmax')
        
        # Direction 2: min-variance LP (flatter convolution)
        g2 = solve_lp(a, mode='minvar')
        
        # Combine directions
        if g1 is not None and g2 is not None:
            # Weight by which gives better immediate improvement
            a1 = project_simplex(a + 0.1 * (g1 - a))
            a2 = project_simplex(a + 0.1 * (g2 - a))
            f1 = compute_obj(a1)
            f2 = compute_obj(a2)
            if f1 < f2:
                direction = g1
            elif f2 < f1:
                direction = g2
            else:
                direction = 0.5 * g1 + 0.5 * g2
        elif g1 is not None:
            direction = g1
        elif g2 is not None:
            direction = g2
        else:
            break
        
        # Backtracking line search
        a_new = backtracking_line_search(a, direction)
        obj_new = compute_obj(a_new)
        
        it += 1
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            # Perturb
            scale = 0.01 + 0.005 * min(stall, 10)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            
            if stall > 30:
                break
        
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    BA it{it}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

def smooth_descent(a, time_budget=200.0, verbose=False):
    """
    Smoothed gradient descent with adaptive tau and Nesterov momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    momentum = 0.7
    lr = 1.0 / np.sqrt(n)
    tau = 20.0
    it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = (time.monotonic() - start) / time_budget
        tau = 20.0 + 300.0 * elapsed  # increase smoothness over time
        
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_c))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient: d/da[i] sum_k p[k] conv[k] = 2 * sum_k p[k] a[k-i] (for valid k-i)
        # Using FFT: gradient = 2 * ifft(fft(p, 2n) * conj(fft(a, 2n)))[:n]
        N = 1
        while N < 2 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        grad_full = ifft(P * np.conj(A)).real
        grad = 2.0 * grad_full[:n]
        
        # Nesterov step
        y = a + momentum * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y > 1e-10:
            y = y / s_y
            # Recompute gradient at y
            conv_y = compute_conv(y)
            max_cy = float(np.max(conv_y))
            exp_y = np.exp(tau * (conv_y - max_cy))
            p_y = exp_y / float(np.sum(exp_y))
            P_y = fft(p_y, N)
            A_y = fft(y, N)
            grad_y_full = ifft(P_y * np.conj(A_y)).real
            grad_y = 2.0 * grad_y_full[:n]
        else:
            y = a.copy()
            grad_y = grad.copy()
        
        v = momentum * v - lr * grad_y
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        it += 1
        lr = max(1e-6, lr * 0.999)
        momentum = min(0.9, 0.7 + 0.1 * elapsed)
        
        if it % 3000 > 2999:
            a = best_a.copy()
            v = np.zeros(n)
        
        if verbose and it % 1000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}")
    
    return best_a

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses enhanced bilinear LP alternation with min-variance direction, adaptive smoothed
gradient descent, peak shaving with pairwise moves, and focused multi-n exploration.
Key improvements: min-variance LP for flatter convolution, Nesterov-accelerated
smoothed descent with adaptive tau, gradient-based peak shaving, and fine n search.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minmax(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def solve_lp_minvar(a):
    """Solve LP: min (M-m) s.t. m <= conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    m = cp.Variable()
    constraints = [A @ g <= M, A @ g >= m, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M - m), constraints)
    for solver in [cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def solve_lp_wactive(a, active_indices):
    """Solve LP: min sum conv(a,g)[k] for k in active, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    active_A = A[active_indices, :]
    g = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Minimize(cp.sum(active_A @ g)), [cp.sum(g) == 1])
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def parabolic_line_search(a, direction, max_iter=12):
    """Line search with parabolic interpolation for step size."""
    n = len(a)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    f0 = compute_obj(a)
    d = direction - a
    t = 0.5
    best_a = a.copy()
    best_f = f0

    t1, f1 = t, f0
    t2, f2 = 0.0, f0

    for _ in range(max_iter):
        a_try = a + t * d
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            t *= 0.5
            continue
        a_try = a_try / s_try
        f_try = compute_obj(a_try)
        if f_try < best_f:
            best_f = f_try
            best_a = a_try.copy()

        if f_try <= f0 * (1 - 1e-5 * t):
            return a_try

        # Parabolic interpolation
        if abs(f_try - f1) > 1e-15 and abs(f_try - f2) > 1e-15:
            num = (f2 - f1) * t**2 + (f_try - f2) * t1**2 + (f1 - f_try) * t2**2
            den = (f2 - f1) * t + (f_try - f2) * t1 + (f1 - f_try) * t2
            if abs(den) > 1e-15:
                t_par = num / den
                if 0.01 < t_par < 2.0:
                    t = t_par
                    continue

        t *= 0.6
        t2, f2 = t1, f1
        t1, f1 = t, f_try

    return best_a


def bilinear_alternate(a, time_budget=200.0, verbose=False):
    """Bilinear LP alternation with multiple directions and adaptive step."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    it = 0
    stall = 0
    use_minvar = True

    while time.monotonic() - start < time_budget:
        elapsed = (time.monotonic() - start) / time_budget
        if elapsed > 0.85:
            break

        # Direction 1: min-max LP
        g1 = solve_lp_minmax(a)

        # Direction 2: min-variance LP (every other iteration to save time)
        g2 = None
        if use_minvar and it % 2 == 0:
            g2 = solve_lp_minvar(a)

        # Direction 3: active-set LP
        g3 = None
        if it % 3 == 0:
            conv_a = compute_conv(a)
            mean_c = float(np.mean(conv_a))
            active = np.where(conv_a > mean_c * 1.05)[0]
            if len(active) > 0 and len(active) < n:
                g3 = solve_lp_wactive(a, active)

        # Combine available directions
        directions = [d for d in [g1, g2, g3] if d is not None]
        if not directions:
            break

        if len(directions) == 1:
            direction = directions[0]
        else:
            # Use immediate evaluation to weight directions
            weights = []
            for d in directions:
                a_test = project_simplex(a + 0.05 * (d - a))
                f_test = compute_obj(a_test)
                weights.append(1.0 / max(f_test, 1e-10))
            w_sum = sum(weights)
            direction = np.zeros(n)
            for d, w in zip(directions, weights):
                direction += (w / w_sum) * d

        # Line search
        a_new = parabolic_line_search(a, direction)
        obj_new = compute_obj(a_new)

        it += 1
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Try smaller step
            d_small = 0.5 * (direction - a)
            a_try = np.maximum(a + d_small, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                f_try = compute_obj(a_try)
                if f_try < best_obj - 1e-12:
                    best_obj = f_try
                    best_a = a_try.copy()
                    a = a_try
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1

            if stall > 0:
                scale = 0.005 + 0.003 * min(stall, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()

            if stall > 40:
                break

        if verbose and it % 5 == 0:
            print(f"    BLP it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

    return best_a


def smoothed_descent(a, time_budget=200.0, verbose=False):
    """Nesterov accelerated smoothed gradient descent with adaptive tau."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n:
        N <<= 1

    v = np.zeros(n)
    momentum = 0.7
    lr = 2.0 / np.sqrt(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed_frac = min(1.0, (time.monotonic() - start) / time_budget)
        tau = 15.0 + 500.0 * (elapsed_frac ** 1.5)

        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient at y
        conv_y = compute_conv(y)
        max_c = float(np.max(conv_y))
        # Clip to avoid overflow
        scaled = np.clip(tau * (conv_y - max_c), -50, 50)
        exp_vals = np.exp(scaled)
        p = exp_vals / float(np.sum(exp_vals))

        # Gradient via FFT
        P = fft(p, N)
        A_y = fft(y, N)
        grad_full = ifft(P * np.conj(A_y)).real
        grad = 2.0 * grad_full[:n]

        # Update
        v = momentum * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
            momentum = 0.5
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-7, lr * 0.9995)
        momentum = min(0.92, 0.7 + 0.2 * elapsed_frac)

        # Reset if stuck
        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            momentum = 0.7
            best_it = it

        if verbose and it % 5000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}")

    return best_a


def gradient_peak_shave(a, time_budget=80.0, verbose=False):
    """
    Peak shaving using gradient information.
    For the top peaks, compute the gradient and move mass from high-gradient
    positions to low-gradient positions.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(9999)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(best_a)
        # Top 3 peaks
        peak_indices = np.argsort(conv)[::-1][:3]

        # Compute gradient of max peak with respect to a
        k = int(peak_indices[0])
        grad = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] = 2.0 * best_a[j]

        # Also consider contributions to all top peaks
        for pk in peak_indices[1:]:
            for i in range(n):
                j = pk - i
                if 0 <= j < n:
                    grad[i] += 0.5 * best_a[j]

        # Move mass from high-grad to low-grad positions
        sorted_idx = np.argsort(grad)
        improved = False

        # Try several pairs
        for src_rank in range(min(5, n)):
            for tgt_rank in range(min(5, n)):
                src = int(sorted_idx[n - 1 - src_rank])
                tgt = int(sorted_idx[tgt_rank])
                if src == tgt:
                    continue
                if best_a[src] < 1e-10:
                    continue

                for frac in [0.01, 0.005, 0.002]:
                    delta = frac * best_a[src]
                    if delta <= 0:
                        continue
                    a_try = best_a.copy()
                    a_try[src] -= delta
                    a_try[tgt] += delta
                    a_try = a_try / a_try.sum()
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            # Random pair move
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i != j and best_a[i] > 1e-10:
                for frac in [0.005, 0.002]:
                    delta = frac * best_a[i]
                    a_try = best_a.copy()
                    a_try[i] -= delta
                    a_try[j] += delta
                    a_try = a_try / a_try.sum()
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        stall = 0
                        break

            if stall > 50:
                break

        if verbose and it % 20 == 0:
            print(f"    GPS it{it}: obj={best_obj:.8f}")

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two-bump
        centers = [n // 3, 2 * n // 3]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / (np.sqrt(1 - x**2) + 0.1)
        a = a / a.sum()
    elif mode == 4:
        # Random Fourier
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1.5, 1.5)
        a = a / a.sum()
    else:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.3 + 0.7 * np.cos(x) ** 2
        a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization combining bilinear LP, smoothed GD,
    gradient peak shaving, and focused multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent: {e}")

    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default (n={n_init}): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Bilinear LP (main workhorse) ===
    print(f"\n=== Phase 1: Bilinear LP (budget limited) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 360.0)
    if p1_budget > 30:
        a1 = bilinear_alternate(best_seq, time_budget=p1_budget, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  BLP: obj={best_score:.8f} **")
        else:
            print(f"  BLP: obj={obj1:.8f}")

    # === Phase 2: Smoothed GD ===
    print(f"\n=== Phase 2: Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 250.0)
    if p2_budget > 30:
        a2 = smoothed_descent(best_seq, time_budget=p2_budget, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  GD: obj={best_score:.8f} **")
        else:
            print(f"  GD: obj={obj2:.8f}")

    # === Phase 3: Gradient Peak Shaving ===
    print(f"\n=== Phase 3: Gradient Peak Shaving ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 80.0)
    if p3_budget > 10:
        a3 = gradient_peak_shave(best_seq, time_budget=p3_budget, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  GPS: obj={best_score:.8f} **")
        else:
            print(f"  GPS: obj={obj3:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Focused n options around current best
    n_options = []
    for dn in [-200, -150, -100, -50, 50, 100, 150, 200, 300, 400]:
        nt = n_best + dn
        if 400 <= nt <= 2000 and nt != n_best:
            n_options.append(nt)
    # Add some absolute options
    for nt in [600, 700, 900, 1000, 1200]:
        if 400 <= nt <= 2000 and nt != n_best and nt not in n_options:
            n_options.append(nt)

    # Sort by distance from current best
    n_options.sort(key=lambda x: abs(x - n_best))

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_seq(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try > 0.01:
            a_try = a_try / s_try

        # Quick optimization cycle
        alloc = min(remaining * 0.18, 100.0)
        if alloc > 20:
            a_try = bilinear_alternate(a_try, time_budget=alloc * 0.5)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_try = smoothed_descent(a_try, time_budget=min(remaining * 0.25, 40.0))
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_try = gradient_peak_shave(a_try, time_budget=min(remaining * 0.3, 20.0))

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = r % 7
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with current best
        blend = 0.4
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.2, 100.0)
        if alloc > 20:
            a_r = bilinear_alternate(a_r, time_budget=alloc * 0.5)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_r = smoothed_descent(a_r, time_budget=min(remaining * 0.3, 40.0))

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 25:
        alloc = min(remaining * 0.35, 60.0)
        a = bilinear_alternate(best_seq, time_budget=alloc)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  Final BLP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 40.0)
        a = smoothed_descent(best_seq, time_budget=alloc)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 30.0)
        a = gradient_peak_shave(best_seq, time_budget=alloc)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  Final GPS: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
