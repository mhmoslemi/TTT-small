def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_sparse(a):
    """Build the convolution matrix using a sparse-like approach for the LP.
    Returns (A_ub, b_ub) for constraint A_ub @ [g, M] <= b_ub."""
    n = len(a)
    n_rows = 2 * n - 1
    # We'll build A as a dense matrix for simplicity (n up to ~2000 is OK)
    A = np.zeros((n_rows, n))
    for j in range(n):
        # A[k, j] = a[k-j] if 0 <= k-j < n
        k_start = j
        k_end = min(n_rows, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_fast(a):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses scipy HiGHS directly for speed.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix: conv(a_norm, g)[k] - M <= 0 for all k
    # conv(a_norm, g)[k] = sum_j a_norm[k-j] * g[j]
    n_rows = 2 * n - 1
    A = np.zeros((n_rows, n))
    for j in range(n):
        k_start = j
        k_end = min(n_rows, j + n)
        A[k_start:k_end, j] = a_norm[:k_end - k_start]
    
    # LP: min M
    # Variables: g[0..n-1], M
    # Constraints: A @ g - M <= 0 (i.e., A @ g + [-1]*M <= 0)
    #              sum(g) = 1
    #              g >= 0, M free
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A_ub = np.zeros((n_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_rows)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None

        threshold = max_val * 0.95

        if len(weights) == 0:
            break
        weights = weights / weights.sum()

        # Compute subgradient: s[i] = sum over peak lags k of weights * a[k-i]
        subgrad = np.zeros(n)
        for idx, k in enumerate(peak_lags):
            k = int(k)
            w = weights[idx]

def spectral_riemann(a, time_budget, verbose=False):
    """
    Riemann-style spectral method: iteratively reduce the peak of the convolution
    by modifying the power spectrum.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 2 * n - 1  # padded length
    # Next power of 2 for FFT
    fft_size = 1
    while fft_size < N:
        fft_size <<= 1

    for alpha in [0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
        if time.monotonic() - start > time_budget:
            break
        for it in range(50):
            if time.monotonic() - start > time_budget:
                break
            
            # Compute the convolution (time domain peak)
            conv = compute_conv(a)
            max_val = float(np.max(conv))
            peak_idx = int(np.argmax(conv))
            
            # The peak at lag peak_idx corresponds to a "spike" in time domain
            # In frequency domain, this spike has a flat spectrum (all frequencies equal)
            # So we subtract a constant from the power spectrum, weighted by alpha
            # This reduces all time-domain values equally in the "wrong" phase
            
            # Better approach: compute the spectrum, find where it's large, reduce it
            a_pad = np.zeros(fft_size)
            a_pad[:n] = a
            A = rfft(a_pad, fft_size)
            P = np.abs(A) ** 2
            
            # Find peak frequency and reduce it
            P_real = np.zeros(fft_size)
            P_real[:len(P)] = P
            # Make full spectrum (symmetric for real signal)
            P_full = np.zeros(fft_size)
            P_full[:len(P)] = P
            for k in range(1, len(P)):
                P_full[fft_size - k] = P[k]
            
            # Subtract a "bump" centered at the frequency that contributes most to the peak
            # The peak at lag k* in time domain gets contribution cos(2*pi*k*omega/fft_size) from freq omega
            # To reduce the peak, reduce P where cos(2*pi*peak_idx*omega/fft_size) is positive
            omega = np.arange(fft_size) * 2 * np.pi / fft_size
            bump = np.cos(peak_idx * omega)
            bump = np.maximum(bump, 0)  # only reduce where it contributes positively
            
            P_full_new = P_full - alpha * bump * P_full.max()
            P_full_new = np.maximum(P_full_new, 0)
            
            # Recover from modified spectrum
            # Take square root (amplitude) and preserve phase
            A_new = np.zeros(fft_size, dtype=complex)
            A_new[:len(P)] = np.sqrt(P_full_new[:len(P)]) * (A / (np.abs(A) + 1e-30))
            for k in range(1, len(P)):
                A_new[fft_size - k] = np.conj(A_new[k])
            
            a_new = irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
            else:
                break

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""

    elif m == 2:

        a = a / a.sum()
    elif m == 3:
        # Ramp with noise

        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif m == 4:
        # Anti-ramp

    elif m == 5:
        # Multi-frequency

        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif m == 6:
        # W-shaped
        a = 1.0 + 0.5 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif m == 7:
        # M-shaped
        a = 1.0 - 0.3 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif m == 8:

    elif m == 9:

        a = 1.0 - 0.5 * np.exp(-20 * (x - 0.5) ** 2)

    elif m == 10:

        a = 1.0 + 0.4 * np.cos(6 * np.pi * x + 0.5)
        a = np.maximum(a, 0.01)

    else:

        a = rng.dirichlet(np.ones(n) * 0.3)

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        a_opt = lp_alternate(a_try, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)
        marker = " **" if obj_opt < best_score else ""
        print(f"  n={n_try}: obj={obj_opt:.8f}{marker}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try

    # === Phase 3: Intensive optimization on best n (35%) ===
    print(f"\n=== Phase 3: Intensive (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.35

    # Round 1: LP alternating
    alloc = min(p3_budget * 0.35, 200.0)
    if alloc > 20:
        a_lp = lp_alternate(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        marker = " **" if obj_lp < best_score else ""
        print(f"  LP round 1: obj={obj_lp:.8f}{marker}")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()

    # Round 2: KKT-guided search
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.12, 100.0)
    if alloc > 15:
        a_kkt = kkt_guided_search(best_seq, time_budget=alloc, verbose=True)
        obj_kkt = compute_obj(a_kkt)
        marker = " **" if obj_kkt < best_score else ""
        print(f"  KKT search: obj={obj_kkt:.8f}{marker}")
        if obj_kkt < best_score:
            best_score = obj_kkt
            best_seq = a_kkt.copy()

    # Round 3: LP alternating with perturbation
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.15, 120.0)
    if alloc > 15:
        rng_r = np.random.default_rng(7919 + n_current)
        a_pert = best_seq * np.exp(rng_r.normal(0, 0.03, n_current))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        a_lp2 = lp_alternate(a_pert, time_budget=alloc, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        marker = " **" if obj_lp2 < best_score else ""
        print(f"  LP round 2: obj={obj_lp2:.8f}{marker}")
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()

    # Round 4: Pairwise local search
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.10, 60.0)
    if alloc > 10:
        a_pair = pairwise_local_search(best_seq, time_budget=alloc, verbose=True)
        obj_pair = compute_obj(a_pair)
        marker = " **" if obj_pair < best_score else ""
        print(f"  Pairwise: obj={obj_pair:.8f}{marker}")
        if obj_pair < best_score:
            best_score = obj_pair
            best_seq = a_pair.copy()

    # Round 5: Spectral Riemann
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.10, 50.0)
    if alloc > 10:
        a_spec = spectral_riemann(best_seq, time_budget=alloc, verbose=True)
        obj_spec = compute_obj(a_spec)
        marker = " **" if obj_spec < best_score else ""
        print(f"  Spectral: obj={obj_spec:.8f}{marker}")
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()

    # === Phase 4: Diverse restarts (20%) ===
    print(f"\n=== Phase 4: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.20
    n_use = len(best_seq)
    n_restarts = 6
    per_restart = p4_budget / max(1, n_restarts)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        alloc = min(per_restart, remaining * 0.15)
        if alloc < 15:
            break

        mode = (r * 5 + 2) % 12
        a_r = diverse_init(n_use, rng, mode=mode)
        # Blend with best
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        # LP alternating
        lp_alloc = min(alloc * 0.7, 60.0)
        if lp_alloc > 10:
            a_r = lp_alternate(a_r, time_budget=lp_alloc, verbose=False)

        # Quick KKT
        remaining = deadline - time.monotonic()
        kkt_alloc = min(remaining * 0.03, 20.0)
        if kkt_alloc > 5:
            a_r = kkt_guided_search(a_r, time_budget=kkt_alloc, verbose=False)

        obj_r = compute_obj(a_r)
        marker = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}{marker}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            n_use = n_current

    # === Phase 5: Final polish (15%) ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    alloc = min(remaining * 0.4, 80.0)
    if alloc > 15:
        a_final = lp_alternate(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # KKT polish
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.3, 40.0)
    if alloc > 10:
        a_kkt2 = kkt_guided_search(best_seq, time_budget=alloc, verbose=False)
        obj_kkt2 = compute_obj(a_kkt2)
        if obj_kkt2 < best_score:
            best_score = obj_kkt2
            best_seq = a_kkt2.copy()
            print(f"  Final KKT: obj={best_score:.8f} **")

    # Pairwise polish
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.2, 30.0)
    if alloc > 8:
        a_pair2 = pairwise_local_search(best_seq, time_budget=alloc, verbose=False)
        obj_pair2 = compute_obj(a_pair2)
        if obj_pair2 < best_score:
            best_score = obj_pair2
            best_seq = a_pair2.copy()
            print(f"  Final Pairwise: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"=== Target: 1.5027, Achieved: {best_score:.8f}, Gap: {best_score - 1.5027:.8f} ===")

    return result
