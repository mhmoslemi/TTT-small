def compute_conv(a):
    n = len(a)
    if n == 0: return np.array([])
    size = 1
    while size < 2*n-1: size <<= 1
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

def compute_conv_cross(a, b):
    """Compute conv(a, b) via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1: size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa*fb, size)[:len(a)+len(b)-1]

def compute_obj(a):
    n = len(a)

def project_simplex(v):

    n = len(a)
    conv = compute_conv(a)
    top_k = np.argsort(conv)[::-1][:K]
    peak_vals = conv[top_k]
    max_pv = peak_vals[0]
    if max_pv < 1e-15: return a.copy()
    weights = np.exp((peak_vals - max_pv) * 5.0 / max_pv)
    weights /= weights.sum()
    
    g = np.zeros(n)
    for w, k in zip(weights, top_k):
        g += w * get_conv_row(a, int(k))
    
    # Normalize gradient
    g_norm = np.linalg.norm(g)
    if g_norm < 1e-15: return a.copy()
    g = g / g_norm
    
    return g
