    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    # grad_i = 2 * sum_{j=0}^{n-1} a[j] * p[i+j]
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(a, p[i:i + n])
    return grad


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_iteration(a, t, verbose=False):
    """One LP iteration: solve LP, blend with t, return new a and obj."""
    g_lp, M_lp = solve_lp(a, verbose=verbose)
    if g_lp is None:
        return a, compute_obj(a), None
    a_new = (1.0 - t) * a + t * g_lp
    a_new = np.maximum(a_new, 0)
    s = float(a_new.sum())
    if s < 1e-10:
        return a, compute_obj(a), None
    a_new = a_new / s
    return a_new, compute_obj(a_new), M_lp


def lp_polish(a, time_budget, verbose=False):
    """LP-based polishing with adaptive line search and periodic restarts."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(123)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        # Line search for optimal t
        t_best, obj_best, a_best_ls = line_search(a, g_lp, best_obj)

        if a_best_ls is not None and obj_best < best_obj - 1e-14:
            best_obj = obj_best
            best_a = a_best_ls.copy()
            a = a_best_ls.copy()
            stall = 0
        else:
            # Try direct replacement (t=1)
            a_try = g_lp.copy()
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                stall = 0
            else:
                stall += 1
                if stall > 8:
                    # Perturb to escape
                    scale = min(0.05, 0.01 * stall * 0.5)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    stall = 0
                elif stall > 20:
                    break

        it += 1
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, M_lp={M_lp:.6f}, t={t_best:.3f}, t={elapsed:.0f}s")

    return best_a

def smoothed_gd(a, time_budget, tau_init=20.0, tau_end=500.0, verbose=False):
    """Nesterov smoothed gradient descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.95))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.8)

        g = smoothed_gradient(a, tau)

        v = beta * v - lr * g
        a_new = a + v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                lr *= 0.7
                v = np.zeros(n)
                stall = 0
                if lr < 1e-10:
                    break

        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9997)

        if verbose and it % 2000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a


def peak_shave(a, time_budget, K=15, verbose=False):
    """Peak shaving: reduce top convolution peaks via subgradient descent."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.5 / np.sqrt(n)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        if max_conv < 1e-15:
            break

        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        weights = np.exp(3.0 * (peak_vals - max_pv) / (max_pv + 1e-15))
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        a_new = a - (lr / g_norm) * g
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            lr *= 0.85
            if stall > 200:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Boundary-enhanced
        a = np.ones(n)
        a[0] = 2.0
        a[-1] = 2.0
        a = a / a.sum()
    elif mode == 3:
        # Cosine shape
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
        a = a / a.sum()
    elif mode == 4:
        # Triangular
        a = 1.0 - 2.0 * np.abs(np.arange(n) - (n - 1) / 2) / (n / 2)
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 5:
        # Inverse cosine (peaked at boundaries)
        x = np.linspace(0.01, 0.99, n)
        a = 1.0 / np.cos(np.pi * x / 2 + 0.1)
        a = a / a.sum()
    else:
        a = rng.uniform(0.5, 1.5, n)
        a = a / a.sum()
    return a

            s = float(np.sum(parent_arr))
