def compute_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    rho_mask = np.where(cond)[0]

        return np.ones(n) / n
    rho_val = rho_mask[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a hybrid approach: efficient coordinate descent, subgradient descent with
momentum, LP polishing with adaptive peak weighting, and multi-length exploration.
Key improvements over previous: O(n) subgradient computation, reduced-constraint
LP for speed, better perturbation strategy, and adaptive time allocation.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_indices = np.where(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, k_star):
    """
    Compute subgradient of max_k conv(a,a)[k] at the active index k_star.
    grad[j] = 2 * a[k_star - j] if 0 <= k_star - j < n, else 0.
    """
    n = len(a)
    sg = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        # sg[j] = 2*a[k_star - j] for j in [lo, hi]
        indices = np.arange(lo, hi + 1)
        sg[indices] = 2.0 * a[k_star - indices]
    return sg


def coord_descent(a, time_budget=60.0, verbose=False, rng=None):
    """
    Coordinate descent with golden section search on normalized self-convolution peak.
    Uses incremental convolution updates for efficiency.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
    a = a / S
    conv = compute_conv(a)

    best_obj = float(np.max(conv))
    best_a = a.copy()
    gr = (np.sqrt(5) - 1) / 2.0
    start = time.monotonic()
    stagnant = 0
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        obj_before = float(np.max(conv))

        if rng is not None and sweep % 4 == 3:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for k in order:
            if time.monotonic() - start > time_budget:
                break

            # Direction: d[m] = a[m-k] for m in [k, k+n-1]
            d = np.zeros(2 * n - 1)
            k_lo = max(0, k)
            k_hi = min(2 * n - 1, k + n - 1)
            src_lo = max(0, 0)
            src_hi = min(n, k_hi - k + 1)
            d[k_lo:k_hi + 1] = a[src_lo:src_hi]

            lo_delta = -a[k]
            max_conv_cur = float(np.max(conv))
            upper_est = np.sqrt(max(0.0, max_conv_cur - conv[2 * k] + a[k] ** 2 + 0.1 * max_conv_cur)) + a[k]
            hi_delta = max(upper_est - a[k], 3.0 / n)
            hi_delta = max(hi_delta, lo_delta + 2.0 / n)

            # Golden section search (30 iterations)
            a_gs = lo_delta
            b_gs = hi_delta
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)

            def obj_at_delta(delta):
                val = conv + 2.0 * delta * d
                val[2 * k] += delta * delta
                return float(np.max(val))

            fc = obj_at_delta(c_gs)
            fd = obj_at_delta(d_gs)

            for _ in range(35):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = obj_at_delta(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = obj_at_delta(d_gs)

            delta_opt = (a_gs + b_gs) / 2.0
            new_obj_est = obj_at_delta(delta_opt)

            if new_obj_est >= obj_before - 1e-14:
                continue

            # Apply update
            a[k] = max(0.0, a[k] + delta_opt)
            conv += 2.0 * delta_opt * d
            conv[2 * k] += delta_opt * delta_opt

            # Re-normalize
            S_new = float(np.sum(a))
            if S_new < 1e-10:
                a[:] = 1.0 / n
                conv = compute_conv(a)
            else:
                a /= S_new
                conv /= (S_new * S_new)

        obj_after = float(np.max(conv))
        if obj_after < best_obj - 1e-12:
            best_obj = obj_after
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 3:
                if rng is not None:
                    scale = 0.015 * (1 + 0.3 * stagnant)
                    a = best_a * np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    S = a.sum()
                    if S > 1e-10:
                        a = a / S
                        conv = compute_conv(a)
                        best_obj = float(np.max(conv))
                    stagnant = 0
                else:
                    break

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={2.0*n*obj_after:.8f}, t={elapsed:.0f}s")

    return best_a


def subgradient_momentum(a, time_budget=100.0, verbose=False, rng=None):
    """
    Subgradient descent with momentum on the simplex.
    Uses O(n) subgradient computation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_f = float(np.max(compute_conv(a)))
    start = time.monotonic()

    momentum = 0.7
    v = np.zeros(n)
    lr_base = 1.0 / np.sqrt(n)
    it = 0
    best_it = 0
    stagnant = 0

    while time.monotonic() - start < time_budget:
        conv_a = compute_conv(a)
        k_star = int(np.argmax(conv_a))
        f_a = float(conv_a[k_star])

        sg = compute_subgradient(a, k_star)
        sg_norm_sq = float(np.dot(sg, sg))

        # Polyak step size
        lr_p = max((f_a - best_f + 1e-10), 1e-8) / (sg_norm_sq + 1e-10)
        lr = min(lr_p, lr_base)

        # Momentum update
        v = momentum * v - lr * sg
        a_new = a + v
        a_new = simplex_project(a_new)

        # Evaluate
        conv_new = compute_conv(a_new)
        f_new = float(np.max(conv_new))

        if f_new < best_f - 1e-13:
            best_f = f_new
            best_a = a_new.copy()
            best_it = it
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 3000:
                a_new = best_a * np.exp(rng.normal(0, 0.02, n)) if rng is not None else best_a.copy()
                a_new = np.maximum(a_new, 0)
                S_n = a_new.sum()
                if S_n > 1e-10:
                    a_new = a_new / S_n
                v = np.zeros(n)
                stagnant = 0
                momentum = min(0.95, momentum * 1.02)

        a = a_new
        it += 1
        lr_base = max(1e-6, lr_base * 0.9999)

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    SGD it{it}: obj={2.0*n*best_f:.8f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a

        grad_g = grad_full[:n] + grad_full[n:2*n]  # wrap around for circular
        # Actually this isn't quite right for linear convolution. Let me just do it properly.
        # grad_g[j] = sum_{k: 0<=k-j<n} w[k] * a[k-j]
        # Let m = k-j, so k = j+m, m in [0, n-1], k in [j, j+n-1]
        # grad_g[j] = sum_m w[j+m] * a[m] for m in [0, min(n-1, 2n-2-j)]

        # This is a cross-correlation. Let's use FFT properly.
        # grad_g = correlate(w, a) of appropriate length
        # For now, use direct computation for correctness (O(n^2) but n~800 is ok for a few iters)
        # Actually let's just use the matrix approach but with the weighted sum as objective

        # Simpler: just minimize the weighted sum
        # obj(g) = sum_k w[k] * (conv(a,g))[k] = g^T B where B[j] = sum_k w[k]*a[k-j]
        # This is linear in g! So the LP is: min B^T g s.t. sum(g)=1, g>=0
        # Solution: g[j] = 1/n for all j in argmin B, 0 otherwise (vertex solution)
        # Or more usefully: g is the uniform distribution on the set of j minimizing B[j]

        # Compute B[j] = sum_k w[k] * a[k-j]
        # This is cross-correlation of w and a
        B = np.zeros(n)
        for j in range(n):
            # B[j] = sum_k w[k] * a[k-j] where 0 <= k-j < n and 0 <= k < 2n-1
            m_lo = max(0, -j)  # k-j >= 0 means k >= j
            m_hi = min(2 * n - 1, j + n - 1)
            valid = np.arange(max(j, 0), m_hi + 1)
            B[j] = np.dot(weights[valid], a[valid - j])

        # The LP solution puts all mass where B is minimized
        # But this gives a sparse solution. Let's instead use a blending approach:
        # g = (1-alpha) * uniform + alpha * delta_argmin(B)
        # Or: g proportional to exp(-beta * B)

        beta = 5.0
        g_val = np.exp(-beta * (B - B.min()))
        s_g = g_val.sum()
        if s_g < 1e-10:
            g_val = np.ones(n) / n
        else:
            g_val = g_val / s_g

        # Also solve the min-max LP for comparison
        # Use the reduced constraint LP
        top_k = min(40, 2 * n - 1)
        top_idx = np.argsort(conv_a)[::-1][:top_k]

        rows = []
        col_data = []
        for m in top_idx:
            j_lo = max(0, m - n + 1)
            j_hi = min(n - 1, m)
            for j in range(j_lo, j_hi + 1):
                rows.append(len(col_data))
                col_data.append((j, a[m - j]))

        if len(col_data) > 0:
            from scipy.sparse import coo_matrix
            rows_arr = np.array(rows, dtype=np.int32)
            cols_arr = np.array([x[0] for x in col_data], dtype=np.int32)
            vals_arr = np.array([x[1] for x in col_data], dtype=np.float64)
            A_mat = coo_matrix((vals_arr, (rows_arr, cols_arr)),
                               shape=(len(top_idx), n)).tocsc()

            g_lp = cp.Variable(n, nonneg=True)
            M_lp = cp.Variable()
            constraints = [A_mat @ g_lp <= M_lp, cp.sum(g_lp) == 1]
            prob = cp.Problem(cp.Minimize(M_lp), constraints)

            g_lp_val = None
            for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
                try:
                    prob.solve(solver=solver, **{'verbose': False})
                    if g_lp.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                        g_lp_val = np.maximum(g_lp.value, 0)
                        s_lp = float(np.sum(g_lp_val))
                        if s_lp > 1e-10:
                            g_lp_val = g_lp_val / s_lp
                        break
                except Exception:
                    continue

            if g_lp_val is not None:
                # Combine the two directions
                g_combined = 0.5 * g_val + 0.5 * g_lp_val
                g_combined = g_combined / g_combined.sum()
            else:
                g_combined = g_val
        else:
            g_combined = g_val

        # Try blending
        for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
            a_try = (1.0 - t) * a + t * g_combined
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try

        iter_count += 1
        if iter_count > 30:
            # Perturb
            if rng is not None:
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                S_a = a.sum()
                if S_a > 1e-10:
                    a = a / S_a
                best_a = a.copy()
                best_obj = compute_obj(a)
            else:
                break

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    LPW it{iter_count}: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-6)
        a = a / a.sum()
    elif mode == 3:
        # Half-sine
        x = np.linspace(0, np.pi, n)
        a = np.sin(x)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 4:
        # Two bumps
        c1 = rng.integers(0, n // 4)
        c2 = rng.integers(3 * n // 4, n)
        w = max(1, n // 10)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Smooth noise
        a = np.ones(n) + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 4) * np.linspace(0, 1, n))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 7:
        # Triangular
        a = 1.0 - np.abs(np.linspace(-1, 1, n))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def nesterov_smoothed(a, time_budget=100.0, verbose=False):
    """
    Nesterov accelerated gradient on smoothed (log-sum-exp) objective.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    best_it = 0
    lr = 2.0 / np.sqrt(n)

    def smoothed_grad(x, tau):
        """Compute gradient of log-sum-exp smoothed max of conv(x,x)."""
        conv = compute_conv(x)
        max_c = float(np.max(conv))
        # softmax weights
        p = np.exp(tau * (conv - max_c))
        p = p / p.sum()

        # grad of sum_k p[k] * conv_k w.r.t. x
        # = 2 * sum_k p[k] * a[|k - j|] type thing
        # Use FFT: grad = 2 * ifft(fft(p) * conj(fft(x)))[:n]
        size = 1
        while size < 3 * n:
            size <<= 1
        P = rfft(p, size)
        X = rfft(x, size)
        g_full = irfft(P * np.conj(X), size)
        grad = 2.0 * (g_full[:n] + g_full[n:2*n])
        # Actually need to be more careful. Let me use direct computation via the subgradient formula.
        # grad[j] = 2 * sum_k p[k] * x[k-j] (with x=0 outside [0,n-1])
        # This is 2 * (p circularly convolved with reversed x)
        # Let's just use the FFT approach carefully:
        # sum_k p[k] * x[k-j] = (p x x_rev)[padding] ... this is getting complicated.
        # Use the fact that it's a convolution:
        # result[j] = sum_k p[k] * x[k-j] for j in [0, n-1]
        # = sum_m p[j+m] * x[m] for m in [0, n-1] (with bounds)
        # This is the cross-correlation of p and x.
        # cross_corr[j] = sum_m p[j+m] * x[m]
        # In FFT: cross_correlate(p, x) = ifft(fft(p) * conj(fft(x)))
        # But we need to be careful about the indexing.

        # Let me just compute it directly for correctness (O(n^2) but ok for moderate n)
        # Actually for n~1000 that's 1M ops which is fine in numpy
        grad_direct = np.zeros(n)
        for j in range(n):
            k_range = np.arange(max(0, j), min(2*n-1, j+n))
            m_range = k_range - j
            grad_direct[j] = np.dot(p[k_range], x[m_range])
        grad = 2.0 * grad_direct
        return grad, conv

    # Use simpler approach: subgradient with Nesterov momentum
    v = np.zeros(n)
    momentum = 0.8
    y = a.copy()

    while time.monotonic() - start < time_budget:
        # Compute smoothed gradient
        tau = 30.0 + 50.0 * min(1.0, (time.monotonic() - start) / time_budget)
        grad, conv = smoothed_grad(y, tau)

        # Nesterov step
        t_k = 2.0 / (it + 2.0)
        y = a + ((it + 4) / (it + 6)) * (a - y + v)  # not standard, let me simplify
        # Actually let's just use momentum:
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = simplex_project(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr *= 0.9999

        if it - best_it > 10000:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    NSD it{it}: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Multi-phase optimization combining coordinate descent,
    subgradient descent, LP polishing (reduced and weighted), and
    multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Coordinate Descent (initial cleanup) ===
    print(f"\n=== Phase 1: Coordinate Descent (initial) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.20, 250.0)
    if p1_budget > 30:
        a_cd = coord_descent(best_seq, time_budget=p1_budget, verbose=True, rng=rng)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase1 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 CD: obj={obj_cd:.8f}")

    # === Phase 2: Subgradient with Momentum ===
    print(f"\n=== Phase 2: Subgradient Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.12, 120.0)
    if p2_budget > 20:
        a_sg = subgradient_momentum(best_seq, time_budget=p2_budget, verbose=True, rng=rng)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Phase2 SGD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 SGD: obj={obj_sg:.8f}")

    # === Phase 3: LP Polishing (reduced constraints) ===
    print(f"\n=== Phase 3: LP Polishing (reduced) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.25, 300.0)
    if p3_budget > 30:
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(p3_budget / (3 - rnd), remaining * 0.6)
            if alloc < 20:
                break
            a_lp = lp_polish_reduced(best_seq, time_budget=alloc, verbose=True, rng=rng, top_k=50)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
                if rnd < 2:
                    scale = 0.015 * (rnd + 1)
                    best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                    best_seq = np.maximum(best_seq, 0)
                    best_seq = best_seq / best_seq.sum()

    # === Phase 4: Weighted LP ===
    print(f"\n=== Phase 4: Weighted LP ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.12, 120.0)
    if p4_budget > 20:
        a_wlp = lp_polish_weighted(best_seq, time_budget=p4_budget, verbose=True, rng=rng)
        obj_wlp = compute_obj(a_wlp)
        if obj_wlp < best_score:
            best_score = obj_wlp
            best_seq = a_wlp.copy()
            print(f"  Phase4 WLp: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 WLp: obj={obj_wlp:.8f}")

    # === Phase 5: Fine Coordinate Descent ===
    print(f"\n=== Phase 5: Fine Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.15, 180.0)
    if p5_budget > 20:
        a_cd2 = coord_descent(best_seq, time_budget=p5_budget, verbose=True, rng=rng)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
