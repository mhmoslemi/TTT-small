"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses frequency-domain flattening, Nesterov smoothed gradient descent,
pairwise coordinate swaps, LP polishing, and multi-length exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def freq_flatten_init(n, rng, n_iter=200):
    """Initialize via frequency-domain flattening (Griffin-Lim style)."""
    a = np.ones(n) / n
    for i in range(n_iter):
        X = fft(a)
        mag = np.abs(X)
        # Flatten: replace magnitudes with geometric mean
        target = np.sqrt(np.max(mag) * np.min(mag[mag > 1e-10]))
        phase = X / (mag + 1e-30)
        X_new = target * phase
        a_new = np.real(ifft(X_new))
        a = np.maximum(a_new, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        else:
            a = np.ones(n) / n
            break
    return a

def smoothed_gradient(a, tau):
    """Compute gradient of smooth approximation of max(conv)."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_v = np.exp(tau * (conv - max_c))
    p = exp_v / float(np.sum(exp_v))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv

def nesterov_descent(a, time_budget=100.0, tau_init=5.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective (simplex-constrained)."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    it = 0
    best_it = 0
    lr = 2.0 / np.sqrt(n)
    momentum = 0.9
    
    v = np.zeros(n)
    y = a.copy()
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)
        
        # Nesterov step
        y = a + momentum * v
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        g, conv = smoothed_gradient(y, tau)
        
        # Barzilai-Borwein step size (every 2 iterations)
        if it > 0 and it % 2 == 0:
            # Simple BB: lr = ||s||^2 / |s^T y|
            pass  # Skip BB for simplicity, use fixed decay
        
        # Gradient step
        a_new = y - lr * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        v = a_new - a
        a = a_new
        it += 1
        
        # Decay learning rate
        lr = lr * 0.9999
        lr = max(lr, 0.5 / np.sqrt(n))
        
        # Reset if stuck
        if it - best_it > 50000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)
            best_it = it
        
        if verbose and it % 5000 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")
    
    return best_a

        # Wait, that's not right. Let me be more careful.
        # The first sum: sum_k δ(k,i) * a[m-k] = a[m-i] (only nonzero when k=i)
        # The second sum: sum_k δ(k,j) * a[m-k] = a[m-j]
        # The third sum: sum_k a[k] * δ(m-k,i) = a[m-i] (only nonzero when m-k=i, i.e., k=m-i)
        # The fourth sum: sum_k a[k] * δ(m-k,j) = a[m-j]
        
        # BUT: in a valid convolution, we need 0 <= k < n and 0 <= m-k < n
        # So a[m-i] is valid only if 0 <= m-i < n
        # And a[m-j] is valid only if 0 <= m-j < n
        
        # Linear part: 2t * (a[m-i] * 1[valid] - a[m-j] * 1[valid])
        
        # Quadratic terms:
        # t^2 * δ(i, m-i) = t^2 if m = 2i
        # -t^2 * δ(i, m-j) = -t^2 if m = i+j
        # -t^2 * δ(j, m-i) = -t^2 if m = i+j
        # +t^2 * δ(j, m-j) = t^2 if m = 2j
        
        # So total delta_conv[m] = 2t*(a[m-i]*v_i - a[m-j]*v_j) + t^2*(δ(m,2i) - 2*δ(m,i+j) + δ(m,2j))
        # where v_i = 1 if 0<=m-i<n, v_j = 1 if 0<=m-j<n
        
        # OK this is getting complex. For the 1D optimization, let me just use numerical evaluation.
        
        def obj_at_t(t):
            a_try = a.copy()
            a_try[i] += t
            a_try[j] -= t
            if a_try[i] < 0 or a_try[j] < 0:
                return np.inf
            s_try = a_try.sum()
            if s_try < 1e-10:
                return np.inf
            a_try = a_try / s_try
            c = compute_conv(a_try)
            return float(np.max(c))
        
        # Golden section search
        gr = (np.sqrt(5) - 1) / 2.0
        a_gs = lo
        b_gs = hi
        c_gs = b_gs - gr * (b_gs - a_gs)
        d_gs = a_gs + gr * (b_gs - a_gs)
        fc = obj_at_t(c_gs)
        fd = obj_at_t(d_gs)
        
        improved = False
        for _ in range(30):
            if fc < fd:
                b_gs = d_gs
                d_gs = c_gs
                fd = fc
                c_gs = b_gs - gr * (b_gs - a_gs)
                fc = obj_at_t(c_gs)
            else:
                a_gs = c_gs
                c_gs = d_gs
                fc = fd
                d_gs = a_gs + gr * (b_gs - a_gs)
                fd = obj_at_t(d_gs)
        
        t_opt = (a_gs + b_gs) / 2.0
        obj_opt = obj_at_t(t_opt)
        
        if obj_opt < best_obj - 1e-13:
            a[i] += t_opt
            a[j] -= t_opt
            a = np.maximum(a, 0)
            s2 = a.sum()
            if s2 > 1e-10:
                a = a / s2
            conv = compute_conv(a)
            best_obj = float(np.max(conv))
            best_a = a.copy()
            no_improve = 0
            improved = True
        else:
            no_improve += 1
        
        iter_count += 1
        if no_improve > 200:
            # Perturb
            scale = 0.01 * (1 + no_improve // 100)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            conv = compute_conv(a)
            best_obj = float(np.max(conv))
            no_improve = 0
        
        if verbose and iter_count % 500 == 0:
            print(f"    PWD it{iter_count}: obj={2*n*best_obj:.8f}, t={time.monotonic()-start:.0f}s")
    
    return best_a

def lp_polish(a, time_budget=50.0, verbose=False):
    """LP-based polishing: fix a, find g minimizing max(conv(a,g)), blend."""
    import cvxpy as cp
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    # Build convolution matrix
    A = np.zeros((2*n-1, n))
    for j in range(n):
        for k in range(n):
            if j + k < 2*n-1:
                A[j+k, j] = a[k]
    
    it = 0
    last_t = 0.3
    
    while time.monotonic() - start < time_budget:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        solved = False
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    g_val = g_val / g_val.sum()
                    solved = True
                    break
            except:
                continue
        
        if not solved:
            break
        
        # Try different blending factors
        candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0,
                     max(0.05, last_t*0.5), min(1.0, last_t*1.5)]
        best_t_obj = best_obj
        best_t = last_t
        best_a_local = best_a.copy()
        
        for t in candidates:
            a_try = (1-t)*a + t*g_val
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()
        
        it += 1
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local
            a = best_a
            last_t = best_t
            if verbose and it % 2 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            scale = 0.02
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            if it > 20:
                break
    
    return best_a

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # U-shaped
        x = np.arange(n)
        center = n / 2
        a = 1.0 / (np.abs(x - center) + 1.0)
        a = a / a.sum()
    elif mode == 3:
        # Two peaks at edges
        w = n // 6
        a = np.zeros(n)
        a[:w] = np.exp(-np.linspace(0, 3, w))
        a[-w:] = np.exp(np.linspace(-3, 0, w))
        a = a / a.sum()
    elif mode == 4:
        # Sparse-like: mass at pseudo-random positions
        n_sparse = max(3, n // 10)
        positions = rng.choice(n, size=n_sparse, replace=False)
        a = np.zeros(n)
        a[positions] = 1.0
        # Smooth
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=n/20)
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 5:
        # Frequency-domain flattened
        a = freq_flatten_init(n, rng, n_iter=100)
    elif mode == 6:
        # Periodic with perturbation
        period = max(2, n // 8)
        a = np.zeros(n)
        for k in range(0, n, period):
            a[k] = 1.0
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=period/3)
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 7:
        # Cosine window (inverted - U shaped)
        x = np.linspace(0, np.pi, n)
        a = 1.0 - 0.5*np.cos(x)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

Wait, I have a concern about the pairwise swap descent being too slow. Each evaluation requires a full FFT convolution, which is O(n log n). With 30 golden section iterations per swap, and each swap requiring 30 evaluations, that's 30 * O(n log n) per swap. For n=800, that's about 30 * 8000 = 240K operations per swap. In Python, this might take a few milliseconds per swap. So we might get a few hundred swaps per second.

Actually, the bigger issue is that I'm computing the full convolution from scratch each time. Let me optimize the pairwise swap by using the analytical formula I derived.

For a swap a[i] += t, a[j] -= t:
delta_conv[m] = 2t*(a[m-i]*v_i - a[m-j]*v_j) + t^2*(δ(m,2i) - 2*δ(m,i+j) + δ(m,2j))

where v_i = 1 if 0 <= m-i < n, else 0.

This allows O(1) evaluation of the new max (well, O(n) to find the max, but no FFT needed).

Let me rewrite the pairwise swap with this optimization:

```python
def pairwise_swap_fast(a, time_budget=100.0, rng=None, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)
    best_obj = float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    
    if rng is None:
        rng = np.random.default_rng(42)
    
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - start < time_budget:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        
        # Precompute direction vectors
        # dir[m] = 2*(a[m-i]*v_i(m) - a[m-j]*v_j(m))
        # where v_i(m) = 1 if 0 <= m-i < n
        dir_vec = np.zeros(2*n-1)
        lo_i = max(0, i)
        hi_i = min(2*n-1, i+n)
        dir_vec[lo_i:hi_i] += a[lo_i - i : hi_i -
        i]
        
        lo_j = max(0, j)
        hi_j = min(2*n-1, j+n
        dir_vec[lo_j:hi_j] -= a[lo_j - j : hi_j - j]
        dir_vec *= 2.0
        
        # Quadratic correction at specific points
        q_2i = 1.0 if 2*i < 2*n-1 else 0.0
        q_ij = -2.0 if i+j < 2*n-
        q_2j = 1.0 if 2*j < 2*n-1 else 0.0
        
        def eval_at_t(t):
            # new_conv = conv + t*dir_vec + t^2 * quadratic
            # Find max
            vals = conv + t * dir_vec
            if 2*i < 2*n-1:
                vals[2*i] += t*t
            if i+j < 2*n-1:
                vals[i+j] -= 2*t*t
            if 2*j < 2*n-1:
                vals[2*j] += t*t
            return float(np.max(vals))
        
        # Bounds
        lo_t = -a[j] + 1e-15
        hi_t = a[i] - 1e-15
        if hi_t - lo_t < 1e-15:
            continue
        
        # Golden section
        gr = (np.sqrt(5) - 1) / 2.0
        a_gs = lo_t
        b_gs = hi_t
        c_gs = b_gs - gr * (b_gs - a_gs)
        d_gs = a_gs + gr * (b_gs - a_gs)
        fc = eval_at_t(c_gs)
        fd = eval_at_t(d_gs)
        
        for _ in range(40):
            if fc < fd:
                b_gs = d_gs
                d_gs = c_gs
                fd = fc
                c_gs = b_gs - gr * (b_gs - a_gs)
                fc = eval_at_t(c_gs)
            else:
                a_gs = c_gs
                c_gs = d_gs
                fc = fd
                d_gs = a_gs + gr * (b_gs - a_gs)
                fd = eval_at_t(d_gs)
        
        t_opt = (a_gs + b_gs) / 2.0
        obj_opt = eval_at_t(t_opt)
        
        # Apply normalization effect (since we're on the simplex)
        # After swap, sum is still 1 (we added t to one and subtracted from another)
        # So no renormalization needed!
        
        if obj_opt < best_obj - 1e-13:
            a[i] += t_opt
            a[j] -= t_opt
            # Update convolution incrementally
            conv += t_opt * dir_vec
            if 2*i < 2*n-1:
                conv[2*i] += t_opt * t_opt
            if i+j < 2*n-1:
                conv[i+j] -= 2*t_opt * t_opt
            if 2*j < 2*n-1:
                conv[2*j] += t_opt * t_opt
            best_obj = float(np.max(conv))
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        iter_count += 1
        if no_improve > 500:
            scale = 0.005 * (1 + no_improve // 200)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            conv = compute_conv(a)
            best_obj = float(np.max(conv))
            no_improve = 0
        
        if verbose and iter_count % 1000 == 0:
            print(f"    PWD it{iter_count}: obj={2*n*best_obj:.8f}, t={time.monotonic()-start:.0f}s")
    
    return best_a
```

This is much faster since each evaluation is O(n) (just a max over an array) rather than O(n log n) (FFT).
