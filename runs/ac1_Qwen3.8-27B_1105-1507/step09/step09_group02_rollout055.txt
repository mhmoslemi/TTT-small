import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_active(a, top_k=None):
    """Solve LP with only top-k peak constraints."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    if top_k is None:
        top_k = 2 * n - 1
    
    conv = compute_conv(a_norm)
    top_idx = np.argsort(conv)[::-1][:top_k]
    A_full = build_conv_matrix(a_norm)
    A = A_full[top_idx, :]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def fourier_init(n, rng, mode=0):
    """Fourier-based structured initialization."""
    if mode == 0:
        # Uniform
        a = np.ones(n)
    elif mode == 1:
        # Single cosine ripple
        x = np.arange(n)
        freq = rng.integers(1, 5)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x / n + rng.uniform(0, 2*np.pi))
    elif mode == 2:
        # Multiple cosines
        x = np.arange(n)
        a = np.ones(n)
        for _ in range(rng.integers(2, 5)):
            f = rng.integers(1, 10)
            phi = rng.uniform(0, 2*np.pi)
            a += 0.2 * np.cos(2*np.pi*f*x/n + phi)
    elif mode == 3:
        # Chirp-based
        x = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2*np.pi * (x**2) / (4*n) + rng.uniform(0, 2*np.pi))
    elif mode == 4:
        # Random Fourier
        x = np.arange(n)
        a = np.ones(n)
        for _ in range(rng.integers(3, 8)):
            f = rng.uniform(0.5, 5.0)
            phi = rng.uniform(0, 2*np.pi)
            amp = rng.uniform(0.1, 0.3)
            a += amp * np.cos(2*np.pi*f*x/n + phi)
    elif mode == 5:
        # Sinc-windowed (flat spectrum approximation)
        x = np.arange(n) - n/2
        a = np.sinc(x / (n * 0.3))
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def nesterov_smoothed_descent_v2(a, time_budget, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)
        
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(y, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]
        
        # Barzilai-Borwein step size
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9998)
        
        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            best_it = it
    
    return best_a

For large `tau`, the first term dominates and `H ≈ 2 * Q_{k*}` where `k*` is the argmax. This is a rank-1 (or rank-2) matrix, which is easy to invert.

For small `tau`, the Hessian is more complex but the problem is smoother.

**Newton's method on the smoothed objective:**

At each step:
1. Compute `p = softmax(tau * conv(a,a))`
2. Compute gradient: `g = 2 * sum_k p_k * (Q_k a)`
3. Compute Hessian approximation: `H ≈ 2 * sum_k p_k * Q_k` (drop the second term for simplicity)
4. Solve `H * d = -g` for the Newton step
5. Line search: find `t` such that `F(a + t*d)` decreases
6. Update: `a = project_simplex(a + t*d)`

The key advantage: Newton's method converges much faster than gradient descent (quadratic convergence vs. linear).

But the Hessian `H = 2 * sum_k p_k * Q_k` is an `n x n` dense matrix. For `n=800`, this is `800*800 = 640,000` entries. Computing it explicitly is feasible. Solving `H * d = -g` is O(n^3) = 512 million operations, which is fast.

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: High-iteration LP with reduced active sets, structured Fourier perturbations,
coordinate descent fine-tuning, coarse-to-fine multi-n exploration, and Nesterov SGD.
Key innovations: (1) only top-30 peak constraints in LP for speed, (2) Fourier-structured

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_rows(a, idx_list):
    """Build only the specified rows of the convolution matrix."""
    n = len(a)
    m = len(idx_list)
    A = np.zeros((m, n))
    for row_i, k in enumerate(idx_list):
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            A[row_i, i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return A


def solve_lp_fast(a, top_k=30):
    """Fast LP: minimize max of top-k convolution entries of a*g, s.t. g in simplex."""
    n = len(a)
    a_norm = a / np.sum(a)
    conv = compute_conv(a_norm)
    top_idx = np.argsort(conv)[::-1][:top_k]
    A = build_conv_matrix_rows(a_norm, top_idx)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def coordinate_descent(a, time_budget, verbose=False):
    """
    Cyclic coordinate descent: optimize each coordinate one at a time.
    For coordinate i, the objective as a function of a[i] is quadratic:
    conv(a,a)[k] involves a[i]^2 (when 2i=k) and a[i]*a[j] (when i+j=k).
    So F(a) = max_k (quadratic in a[i]) is piecewise quadratic.
    We minimize it by finding the best a[i] via ternary search or by
    solving the KKT conditions.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    full_sweeps = 0

    while time.monotonic() - start < time_budget:
        improved = False
        for i in range(n):
            if time.monotonic() - start < time_budget:
                # Save old value
                old_ai = a[i]
                rest_sum = s - old_ai

                # Try different values of a[i]
                # The optimal a[i] should be near old_ai
                # Try: 0, old_ai*0.5, old_ai*0.7, old_ai, old_ai*1.3, old_ai*1.6, old_ai*2
                candidates = [0.0, old_ai * 0.3, old_ai * 0.6, old_ai * 0.8,
                              old_ai, old_ai * 1.2, old_ai * 1.5, old_ai * 2.0,
                              old_ai + 1.0 / n, old_ai - 0.5 / n]
                candidates = [max(0.0, c) for c in candidates]

                best_val = best_obj + 1  # We want to improve
                best_ai = old_ai

                for ai_try in candidates:
                    a_test = a.copy()
                    a_test[i] = ai_try
                    s_test = float(np.sum(a_test))
                    if s_test < 1e-10:
                        continue
                    a_test = a_test / s_test
                    ov = compute_obj(a_test)
                    if ov < best_val:
                        best_val = ov
                        best_ai = ai_try

                if best_ai != old_ai and best_val < best_obj - 1e-14:
                    a[i] = best_ai
                    s = float(np.sum(a))
                    a = a / s
                    best_obj = best_val
                    best_a = a.copy()
                    improved = True

        full_sweeps += 1
        if not improved and full_sweeps > 2:
            break
        if verbose and full_sweeps % 5 == 0:
            print(f"    CD sweep {full_sweeps}: obj={best_obj:.10f}")

    return best_a


def nesterov_sgd(a, time_budget, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent with FFT-based gradients."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(1, time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        y = a + beta * v
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy < 1e-10:
            y = best_a.copy()
        else:
            y = y / sy

        conv = compute_conv(y)
        max_c = float(np.max(conv))
        exp_v = np.exp(tau * (conv - max_c))
        p = exp_v / float(np.sum(exp_v))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_f = fft(y, N)
        g_full = ifft(P * np.conj(A_f)).real
        g = 2.0 * g_full[:n]

        # Barzilai-Borwein
        if prev_g is not None and it % 3 == 0:
            sv = y - prev_a
            yv = g - prev_g
            ss = float(np.dot(sv, sv))
            sy2 = float(np.dot(sv, yv))
            if ss > 1e-15 and abs(sy2) > 1e-12:
                bb = ss / abs(sy2)
                lr = 0.4 * lr + 0.6 * min(max(bb, 1e-9), 10.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / sn

        ov = compute_obj(a_new)
        if ov < best_obj - 1e-13:
            best_obj = ov
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-9, lr * 0.9999)

        if it - best_it > 12000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.4
            best_it = it

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)


def diverse_init(n, rng, mode=0):
    """Generate diverse structured initializations."""
    x = np.arange(n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        # Two bumps
        c1, c2 = n // 3, 2 * n // 3
        w = n // 6
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
    elif mode == 2:
        # Cosine ripple
        f = rng.integers(2, 6)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f * x / n)
    elif mode == 3:
        # Chirp
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x ** 2 / (4 * n))
    elif mode == 4:
        # Random multi-frequency
        a = np.ones(n)
        for _ in range(rng.integers(2, 5)):
            f = rng.uniform(0.5, 5.0)
            phi = rng.uniform(0, 2 * np.pi)
            a += 0.2 * np.cos(2 * np.pi * f * x / n + phi)
    elif mode == 5:
        # Arcsine distribution (flat in frequency)
        xx = np.linspace(-0.999, 0.999, n)
        a = 1.0 / np.sqrt(1 - xx ** 2 + 1e-6)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()

    Main search function.
    Phase 1: Intensive LP with small active sets (50% time)
    Phase 2: Multi-n exploration with resampling (20% time)
    Phase 3: Coordinate descent fine-tuning (10% time)
    Phase 4: Final combined polish (20% time)
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init n=800: obj={best_score:.10f}")

    n_cur = len(best_seq)

    # === Phase 1: Intensive LP (50% time) ===
    print(f"\n=== Phase 1: Intensive LP (n={n_cur}) ===")
    remaining = deadline - time.monotonic()
    p1 = min(remaining * 0.50, 650.0)

    if p1 > 60:
        # Split into rounds with different active set sizes
        rounds = [
            (p1 * 0.35, 25, "k=25"),
            (p1 * 0.30, 40, "k=40"),
            (p1 * 0.20, 20, "k=20"),
            (p1 * 0.15, 50, "k=50"),
        ]
        for alloc, k, label in rounds:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(alloc, remaining * 0.55)
            if alloc < 15:
                break
            print(f"  LP {label}, budget={alloc:.0f}s")
            a_res = lp_polish_intensive(best_seq, time_budget=alloc, top_k=k, verbose=(k == 25))
            ov = compute_obj(a_res)
            if ov < best_score - 1e-12:
                old = best_score
                best_score = ov
                best_seq = a_res.copy()
                print(f"    -> {old:.10f} -> {best_score:.10f} **")
            else:
                print(f"    -> {ov:.10f} (no improvement)")
                # Perturb between rounds
                scale = 0.015
                best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                best_seq = np.maximum(best_seq, 1e-8)
                best_seq = best_seq / best_seq.sum()

    # === Phase 2: Multi-n exploration (20% time) ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    p2 = min(remaining * 0.20, 250.0)

    n_cur = len(best_seq)
    n_options = []
    for delta in [-200, -100, 100, 200, 300, 500]:
        nt = n_cur + delta
        if 300 <= nt <= 2000:
            n_options.append(nt)
    for fixed in [500, 700, 1000, 1200, 1500]:
        if abs(fixed - n_cur) > 80:
            n_options.append(fixed)
    n_options = sorted(set(n_options))

    for nt in n_options[:7]:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        a_try = resample_sequence(best_seq, nt)
        # Quick LP
        alloc = min(remaining * 0.18, 50.0)
        if alloc > 12:
            a_try = lp_polish_intensive(a_try, time_budget=alloc, top_k=25)

        ov = compute_obj(a_try)
        if ov < best_score:
            best_score = ov
            best_seq = a_try.copy()
            n_cur = nt
            print(f"  ** n={nt}: {best_score:.10f} **")
        else:
            print(f"  n={nt}: {ov:.10f}")

    # === Phase 3: Coordinate Descent (10% time) ===
    print(f"\n=== Phase 3: Coordinate Descent (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p3 = min(remaining * 0.10, 100.0)

    if p3 > 15 and len(best_seq) <= 1200:
        a_cd = coordinate_descent(best_seq, time_budget=p3, verbose=True)
        ov = compute_obj(a_cd)
        if ov < best_score:
            best_score = ov
            best_seq = a_cd.copy()
            print(f"  CD: {best_score:.10f} **")
        else:
            print(f"  CD: {ov:.10f}")

    # === Phase 4: Final Polish (remaining) ===
    print(f"\n=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Nesterov SGD
    if remaining > 25:
        alloc = min(remaining * 0.30, 80.0)
        a_sgd = nesterov_sgd(best_seq, time_budget=alloc,
                             tau_init=20.0, tau_end=800.0, verbose=True)
        ov = compute_obj(a_sgd)
        if ov < best_score:
            best_score = ov
            best_seq = a_sgd.copy()
            print(f"  SGD: {best_score:.10f} **")
        else:
            print(f"  SGD: {ov:.10f}")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.40, 100.0)
        a_lp = lp_polish_intensive(best_seq, time_budget=alloc, top_k=30, verbose=True)
        ov = compute_obj(a_lp)
        if ov < best_score:
            best_score = ov
            best_seq = a_lp.copy()
            print(f"  Final LP: {best_score:.10f} **")

    # One more coordinate descent pass
    remaining = deadline - time.monotonic()
    if remaining > 15 and len(best_seq) <= 1000:
        alloc = min(remaining * 0.30, 40.0)
        a_cd = coordinate_descent(best_seq, time_budget=alloc)
        ov = compute_obj(a_cd)
        if ov < best_score:
            best_score = ov
            best_seq = a_cd.copy()
            print(f"  Final CD: {best_score:.10f} **")

    # Final LP micro-polish
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.50, 30.0)
        a_lp = lp_polish_intensive(best_seq, time_budget=alloc, top_k=20)
        ov = compute_obj(a_lp)
        if ov < best_score:
            best_score = ov
            best_seq = a_lp.copy()
            print(f"  Micro LP: {best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
