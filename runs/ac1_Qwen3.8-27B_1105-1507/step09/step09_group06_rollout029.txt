Let me also think about the "spectral perturbation" more carefully:

The idea: 
1. Compute C = conv(a, a)
2. In frequency domain: A = fft(a, N), so A^2 ≈ fft(C, N) (approximately, for zero-padded)
3. Modify C to reduce peaks: C_new = min(C, t) or C_new = C / (1 + eps * (C - t)_+ / t)
4. Recover a from C_new: a_new = ifft(sqrt(fft(C_new))).real[:n]
5. Project to simplex

The issue in step 4 is that sqrt of a complex number is complex, and the result might not be real or non-negative. But if C_new is a valid autocorrelation (PSD in frequency), then fft(C_new) should be non-negative real, and its square root is real non-negative.

The problem is that C_new = min(C, t) is NOT generally a valid autocorrelation. But in practice, if t is close to the actual max, the clipping is mild and the result is approximately valid.

A better approach: use an iterative "spectral projection":
1. Start with a
2. Compute C = conv(a,a)
3. Compute C_target = min(C, t) where t is slightly above the desired peak
4. Compute a_candidate = ifft(sqrt(max(re(fft(C_target)), 0))).real[:n]
5. Blend: a_new = (1-alpha)*a + alpha*a_candidate
6. Project to simplex
7. Repeat

This should gradually push the peaks down.

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu * rho > (cssv - 1.0)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_val = idx[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_perturb(a, n_iters=100, alpha=0.3, seed=0):
    """
    Spectral error-reduction perturbation.
    Iteratively clips the convolution peaks and recovers the sequence
    via spectral factorization.
    """
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    a_work = a.copy()
    rng = np.random.default_rng(seed)
    
    for i in range(n_iters):
        # Compute convolution
        fa = rfft(a_work, size)
        conv = irfft(fa * fa, size)[:2*n-1]
        max_c = float(np.max(conv))
        
        if max_c < 1e-10:
            break
        
        # Target: reduce peaks by a factor
        t = max_c * 0.95  # Clip to 95% of current max
        conv_clipped = np.minimum(conv, t)
        
        # Pad to full size for FFT
        conv_full = np.zeros(size)
        conv_full[:2*n-1] = conv_clipped
        
        # Spectral factorization: A = sqrt(FFT(conv_full))
        fc = rfft(conv_full)
        # Handle negative values (shouldn't happen much if clipping is mild)
        fc_real = np.maximum(fc, 0)
        A_new = np.sqrt(fc_real)
        
        a_rec = irfft(A_new, size)[:n]
        a_rec = np.maximum(a_rec, 0)
        s_rec = float(np.sum(a_rec))
        if s_rec < 1e-10:
            continue
        a_rec = a_rec / s_rec
        
        # Blend
        a_work = (1 - alpha) * a_work + alpha * a_rec
    
    return a_work

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
