"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses iterated LP best-response with adaptive line search, FFT-accelerated gradient
polishing, and multi-strategy exploration. Key innovations: exact line search on
LP blend direction, subgradient-based active set detection, and periodic
frequency-domain perturbations to escape local minima.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, time_budget=30.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (optimal_g, optimal_M, dual_vars) or (None, None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCIP, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, time_limit=time_budget)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Get dual variables for constraint A@g <= M
                    dual = None
                    try:
                        # The first constraint block has duals
                        if hasattr(prob, 'value') and prob.constraints[0].dual_value is not None:
                            dual = prob.constraints[0].dual_value.copy()
                    except:
                        dual = None
                    return g_val, float(M.value), dual
        except Exception:
            continue
    return None, None, None


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv, p


def line_search_blend(a, g, obj_current, max_iters=20):
    """
    Find best t in [0,1] to minimize obj((1-t)*a + t*g).
    Uses golden section search.
    Returns (best_t, best_obj, best_point).
    """
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ~0.618
    
    a_sum = np.sum(a)
    g_sum = np.sum(g)
    
    def obj_at_t(t):
        blend = (1 - t) * a + t * g
        blend = np.maximum(blend, 0)
        s = float(np.sum(blend))
        if s < 1e-10:
            return np.inf
        blend = blend / s
        return compute_obj(blend)
    
    # First check endpoints and midpoint
    obj_0 = obj_current
    obj_1 = obj_at_t(1.0)
    obj_half = obj_at_t(0.5)
    
    # Find bracket
    if obj_1 < obj_0 and obj_1 <= obj_half:
        # Objective decreasing, try to find where it starts increasing
        lo, hi = 0.5, 1.0
    elif obj_half < obj_0:
        lo, hi = 0.0, 1.0
    else:
        # Objective increasing from start, optimal is near 0
        lo, hi = 0.0, 0.5
    
    best_t = 0.0
    best_obj = obj_0
    best_point = a.copy()
    
    if obj_half < best_obj:
        best_obj = obj_half
        best_t = 0.5
        s = float(np.sum(np.maximum((1-0.5)*a + 0.5*g, 0)))
        best_point = np.maximum((1-0.5)*a + 0.5*g, 0) / s
    
    if obj_1 < best_obj:
        best_obj = obj_1
        best_t = 1.0
        best_point = g.copy()
    
    # Golden section in [lo, hi]
    if hi - lo > 0.01:
        c = hi - phi * (hi - lo)
        d = lo + phi * (hi - lo)
        fc = obj_at_t(c)
        fd = obj_at_t(d)
        
        for _ in range(max_iters - 3):
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = hi - phi * (hi - lo)
                fc = obj_at_t(c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + phi * (hi - lo)
                fd = obj_at_t(d)
            
            # Track best
            if fc < best_obj:
                best_obj = fc
                best_t = c
                s = float(np.sum(np.maximum((1-c)*a + c*g, 0)))
                if s > 1e-10:
                    best_point = np.maximum((1-c)*a + c*g, 0) / s
            if fd < best_obj:
                best_obj = fd
                best_t = d
                s = float(np.sum(np.maximum((1-d)*a + d*g, 0)))
                if s > 1e-10:
                    best_point = np.maximum((1-d)*a + d*g, 0) / s
            if hi - lo < 0.01:
                break
    
    return best_t, best_obj, best_point


def gradient_polish(a, time_budget=30.0, tau_start=50.0, tau_end=800.0):
    """
    Gradient descent polishing with adaptive tau (log-sum-exp smoothing parameter).
    Uses Nesterov acceleration.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    it = 0
    best_it = it
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 1.5
        
        g, conv, p = smoothed_gradient_fft(a, tau)
        
        # Nesterov step
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
            g, conv, p = smoothed_gradient_fft(y, tau)
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.99995)
        
        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it
    
    return best_a

    it = 0

        if remaining < 10:

        it += 1
        if obj_best < best_obj - 1e-14:
            best_obj = obj_best
            best_a = a_best_try.copy()
            a = a_best_try
            stall = 0
            if verbose and it % 10 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={t_best:.4f}, M_lp={M_lp:.6f}")
        else:
            stall += 1
            if verbose and it % 10 == 0:
                print(f"    LP it{it}: no improvement (stall={stall})")
            if stall > 3:
                # Perturb and try again
                rng = np.random.default_rng(it)
                n_pert = max(10, n // 50)
                mask = rng.choice(n, size=n_pert, replace=False)
                a[mask] *= (1 + 0.05 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0
                if stall > 10:
                    break
    
    return best_a


def frequency_perturb(a, rng, strength=0.03):
    """Perturb in frequency domain for diversity."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    A = fft(a, size)
    # Perturb low frequencies
    n_freq = max(5, size // 20)
    idx = np.arange(1, n_freq + 1)
    phase_pert = rng.standard_normal(len(idx)) * strength
    mag_pert = 1 + rng.standard_normal(len(idx)) * strength * 0.5
    A[idx] *= mag_pert * np.exp(1j * phase_pert)
    A[size - idx] = np.conj(A[idx])  # maintain real symmetry
    a_new = ifft(A).real[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses iterated LP best-response with golden-section line search as the primary
method, complemented by FFT-accelerated smoothed gradient polishing,
frequency-domain perturbation for diversity, and multi-length exploration.
Key innovation: proper line search on the LP blend direction (vs. fixed grid),
and aggressive LP iteration count enabled by efficient sparse LP solves.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build sparse-friendly matrix A (2n-1 x n) such that A @ g gives conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, time_budget=20.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (optimal_g, optimal_M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, time_limit=time_budget)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def line_search_blend(a, g, obj_current, max_evals=25):
    """
    Golden-section search for best t in [0,1] minimizing obj((1-t)*a + t*g).
    Returns (best_t, best_obj, best_point).
    """
    golden = (np.sqrt(5.0) - 1.0) / 2.0
    eval_count = 0

    def obj_at_t(t):
        nonlocal eval_count
        blend = (1.0 - t) * a + t * g
        blend = np.maximum(blend, 0.0)
        s = float(np.sum(blend))
        if s < 1e-10:
            return np.inf
        blend = blend / s
        eval_count += 1
        return compute_obj(blend)

    # Evaluate at key points
    obj_0 = obj_current
    obj_mid = obj_at_t(0.5)
    obj_1 = obj_at_t(1.0)

    best_t = 0.0
    best_obj = obj_0
    best_point = a.copy()

    if obj_mid < best_obj:
        best_obj = obj_mid
        best_t = 0.5
        s = float(np.sum(np.maximum(0.5 * a + 0.5 * g, 0)))
        if s > 0:
            best_point = np.maximum(0.5 * a + 0.5 * g, 0) / s
    if obj_1 < best_obj:
        best_obj = obj_1
        best_t = 1.0
        best_point = g.copy()

    # Determine bracket
    if obj_1 < obj_0 and obj_1 <= obj_mid:
        lo, hi = 0.5, 1.0
    elif obj_mid < obj_0 and obj_mid <= obj_1:
        lo, hi = 0.0, 1.0
    else:
        lo, hi = 0.0, 0.5

    # Golden section
    while hi - lo > 0.02 and eval_count < max_evals:
        c = hi - golden * (hi - lo)
        d = lo + golden * (hi - lo)
        fc = obj_at_t(c)
        fd = obj_at_t(d)

        if fc < fd:
            hi = d
        else:
            lo = c

        if fc < best_obj:
            best_obj = fc
            best_t = c
            s = float(np.sum(np.maximum((1.0 - c) * a + c * g, 0)))
            if s > 0:
                best_point = np.maximum((1.0 - c) * a + c * g, 0) / s
        if fd < best_obj:
            best_obj = fd
            best_t = d
            s = float(np.sum(np.maximum((1.0 - d) * a + d * g, 0)))
            if s > 0:
                best_point = np.maximum((1.0 - d) * a + d * g, 0) / s

    return best_t, best_obj, best_point

    lr = base_lr
    v = np.zeros(n)
    it = 0
    best_it = it

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * (progress ** 1.3)

        g, conv = smoothed_gradient_fft(a, tau)

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))

            y = best_a.copy()

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))

            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it

                s = float(np.sum(a))
                if s > 1e-10:

                else:

def frequency_perturb(a, rng, strength=0.03):
    """Perturb in frequency domain for diversity."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    A = fft(a, size)
    n_freq = max(5, size // 15)
    idx = np.arange(1, n_freq + 1)
    phase_pert = rng.standard_normal(len(idx)) * strength
    mag_pert = 1.0 + rng.standard_normal(len(idx)) * strength * 0.3
    A[idx] *= mag_pert * np.exp(1j * phase_pert)
    A[size - idx] = np.conj(A[idx])
    a_new = ifft(A).real[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_to_length(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.9f}")
    except Exception:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.9f}")

    n_curr = len(best_seq)

    # === Phase 1: Intensive LP iteration (main workhorse) ===
    print(f"\n=== Phase 1: Intensive LP Iteration (n={n_curr}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.45, 600.0)
    if p1_budget > 30:
        a_out = lp_iterate(best_seq, time_budget=p1_budget, verbose=True)
        obj_out = compute_obj(a_out)
        if obj_out < best_score:
            best_score = obj_out
            best_seq = a_out.copy()
            print(f"  Phase1 result: obj={best_score:.9f} **")
        else:
            print(f"  Phase1 result: obj={obj_out:.9f} (no improvement)")

    # === Phase 2: Gradient polishing ===
    print(f"\n=== Phase 2: Gradient Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.12, 120.0)
    if p2_budget > 20:
        a_grad = gradient_polish(best_seq, time_budget=p2_budget, tau_start=30, tau_end=1200)
        obj_grad = compute_obj(a_grad)
        if obj_grad < best_score:
            best_score = obj_grad
            best_seq = a_grad.copy()
            print(f"  Phase2: obj={best_score:.9f} **")
        else:
            print(f"  Phase2: obj={obj_grad:.9f}")

    # === Phase 3: LP with frequency perturbation (escape local minima) ===
    print(f"\n=== Phase 3: LP + Perturbation Rounds ===")
    for rnd in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(remaining * 0.15, 100.0)

        # Apply perturbation
        strength = 0.015 + 0.008 * rnd
        a_pert = frequency_perturb(best_seq, rng, strength=strength)
        a_out = lp_iterate(a_pert, time_budget=alloc, verbose=(rnd < 2))
        obj_out = compute_obj(a_out)
        if obj_out < best_score:
            best_score = obj_out
            best_seq = a_out.copy()
            print(f"  Phase3 r{rnd}: obj={best_score:.9f} **")
        else:
            print(f"  Phase3 r{rnd}: obj={obj_out:.9f}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length Exploration ===")
    n_best = len(best_seq)
    n_candidates = set()
    for delta in [-100, -50, 0, 50, 100, 200, 300]:
        n_try = n_best + delta
        if 400 <= n_try <= 2000:
            n_candidates.add(n_try)
    for n_abs in [500, 600, 700, 900, 1000, 1200, 1400]:
        if abs(n_abs - n_best) > 50:
            n_candidates.add(n_abs)
    n_list = sorted(n_candidates)[:6]
    print(f"  n_best={n_best}, trying: {n_list}")

    for n_try in n_list:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_to_length(best_seq, n_try)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s

        alloc = min(remaining * 0.1, 70.0)
        if alloc > 20:
            a_try = lp_iterate(a_try, time_budget=alloc, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.9f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.9f}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Final LP
    if remaining > 30:
        alloc = min(remaining * 0.35, 200.0)
        a_out = lp_iterate(best_seq, time_budget=alloc, verbose=True)
        obj_out = compute_obj(a_out)
        if obj_out < best_score:
            best_score = obj_out
            best_seq = a_out.copy()
            print(f"  Final LP: obj={best_score:.9f} **")
        else:
            print(f"  Final LP: obj={obj_out:.9f}")

    # Final gradient
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.35, 100.0)
        a_grad = gradient_polish(best_seq, time_budget=alloc, tau_start=200, tau_end=3000)
        obj_grad = compute_obj(a_grad)
        if obj_grad < best_score:
            best_score = obj_grad
            best_seq = a_grad.copy()
            print(f"  Final Grad: obj={best_score:.9f} **")
        else:
            print(f"  Final Grad: obj={obj_grad:.9f}")

    # One more short LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 60.0)
        a_out = lp_iterate(best_seq, time_budget=alloc, verbose=False)
        obj_out = compute_obj(a_out)
        if obj_out < best_score:
            best_score = obj_out
            best_seq = a_out.copy()
            print(f"  Ultra-final LP: obj={best_score:.9f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.9f}, time={elapsed:.1f}s ===")
    return result
