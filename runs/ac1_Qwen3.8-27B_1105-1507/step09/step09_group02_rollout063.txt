import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        r = min(n, 2*n-1-j)
        A[j:j+r, j] = a[:r]
    return A

def solve_lp(a, warm_start=None):
    """Solve LP: min M s.t. (a_norm * g)_k <= M, sum g = 1, g >= 0"""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    if warm_start is not None:
        g.value = warm_start
    M = cp.Variable()
    
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

                s = float(np.sum(g_val))
                if s > 1e-10:

        except:

        g.value = np.maximum(g_warm, 0)
    M = cp.Variable()

    c1 = A @ g <= M
    c2 = cp.sum(g) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    duals = None
                    try:
                        duals = c1.dual_value
                        if duals is not None:
                            duals = np.asarray(duals, dtype=np.float64)
                    except Exception:
                        pass
                    return g_val / s, float(M.value), duals
        except Exception:
            continue
    return None, None, None


def solve_qp_energy(a):
    """Solve QP: min ||A @ g||_2^2 s.t. sum(g)=1, g>=0.
    Gives a different direction than LP (minimizes total energy not peak)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Minimize(cp.sum_squares(A @ g)), [cp.sum(g) == 1])

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def golden_section_t(a, g, f_obj, tol=1e-6, max_iter=30):
    """Find optimal t in (0,1] minimizing f_obj((1-t)*a + t*g) via golden section.
    Since f(t) is convex in t (max of quadratics), this converges fast."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618

    a_low = 1e-4
    a_high = 1.0

    c = a_high - gr * (a_high - a_low)
    d = a_low + gr * (a_high - a_low)

    a_c = np.maximum((1 - c) * a + c * g, 0)
    sc = float(np.sum(a_c))
    f_c = f_obj(a_c / sc) if sc > 1e-10 else np.inf

    a_d = np.maximum((1 - d) * a + d * g, 0)
    sd = float(np.sum(a_d))
    f_d = f_obj(a_d / sd) if sd > 1e-10 else np.inf

    for _ in range(max_iter):
        if a_high - a_low < tol:
            break
        if f_c < f_d:
            a_high = d
            d = c
            f_d = f_c
            c = a_high - gr * (a_high - a_low)
            a_c = np.maximum((1 - c) * a + c * g, 0)
            sc = float(np.sum(a_c))
            f_c = f_obj(a_c / sc) if sc > 1e-10 else np.inf
        else:
            a_low = c
            c = d
            f_c = f_d
            d = a_low + gr * (a_high - a_low)
            a_d = np.maximum((1 - d) * a + d * g, 0)
            sd = float(np.sum(a_d))
            f_d = f_obj(a_d / sd) if sd > 1e-10 else np.inf

    t_best = (a_low + a_high) / 2
    return t_best


def geometric_blend(a, g, t):
    """Geometric blend: a_new = a * g^t / sum(a * g^t)."""
    ln_a = np.log(np.maximum(a, 1e-30))
    ln_g = np.log(np.maximum(g, 1e-30))
    log_mix = (1 - t) * ln_a + t * ln_g
    log_mix -= log_mix.max()  # numerical stability
    a_new = np.exp(log_mix)
    s = float(np.sum(a_new))
    if s < 1e-15:
        return a.copy()
    return a_new / s


def lp_polish_main(a, time_budget, verbose=False, use_qp=False, use_geo=False,
                   use_golden=True, seed=42):
    """
    Main LP polishing loop with:
    - Warm-starting
    - Golden section line search on t (or discrete candidates)
    - Optional QP hybrid direction
    - Optional geometric blending
    - Dual-guided perturbations for restart
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    iter_count = 0
    g_prev = None
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if time_budget - elapsed < 5:
            break

        # Solve LP with warm start
        g_lp, M_lp, duals = solve_lp_warm(a, g_warm=g_prev)
        if g_lp is None:
            break
        g_prev = g_lp.copy()

        # Get candidate directions
        directions = [g_lp]
        if use_qp and iter_count % 4 == 3:
            g_qp = solve_qp_energy(a)
            if g_qp is not None:
                directions.append(0.7 * g_lp + 0.3 * g_qp)

        best_trial_obj = best_obj
        best_trial_a = None

        for g_dir in directions:
            if use_golden:
                # Golden section search on t
                t_best = golden_section_t(a, g_dir, compute_obj, tol=0.01, max_iter=12)
                a_try = (1 - t_best) * a + t_best * g_dir
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st > 1e-10:
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_trial_obj:
                        best_trial_obj = obj_try
                        best_trial_a = a_try.copy()
            else:
                # Discrete t values
                for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
                    a_try = (1 - t) * a + t * g_dir
                    a_try = np.maximum(a_try, 0)
                    st = float(np.sum(a_try))
                    if st > 1e-10:
                        a_try = a_try / st
                        obj_try = compute_obj(a_try)
                        if obj_try < best_trial_obj:
                            best_trial_obj = obj_try
                            best_trial_a = a_try.copy()

            # Geometric blending
            if use_geo:
                for t in [0.1, 0.3, 0.5, 0.8, 1.0]:
                    a_try = geometric_blend(a, g_dir, t)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_trial_obj:
                        best_trial_obj = obj_try
                        best_trial_a = a_try.copy()

        iter_count += 1

        if best_trial_a is not None and best_trial_obj < best_obj - 1e-14:
            best_obj = best_trial_obj
            best_a = best_trial_a.copy()
            a = best_a
            stall = 0
            if verbose and iter_count % 5 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, M={M_lp:.6f}")
        else:
            stall += 1
            # Structured perturbation using dual info
            if duals is not None and len(duals) > 0 and np.max(np.abs(duals)) > 1e-8:
                # Identify active peaks
                active_mask = np.abs(duals) > 0.3 * np.max(np.abs(duals))
                n_active = int(np.sum(active_mask))
                # Create perturbation that targets active peaks
                scale = 0.005 + 0.003 * min(stall, 15)
                pert = rng.normal(0, scale, n)
                # Extra push on indices that contribute to active peaks
                for k_idx in np.where(active_mask)[0][:5]:
                    k = int(k_idx)
                    i_s = max(0, k - (n - 1))
                    i_e = min(n - 1, k)
                    pert[i_s:i_e + 1] += scale * 0.5
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
            else:
                scale = 0.008 + 0.004 * min(stall, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()

            if stall > 25:
                # Bigger restart
                scale = 0.04
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                g_prev = None
                stall = 0

            if stall > 60:
                break

    return best_a


def subgradient_refine(a, time_budget, K=15, verbose=False):
    """Subgradient method with softmax-weighted top-K peaks and momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.3 / np.sqrt(n)
    momentum = 0.8
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        if max_conv < 1e-15:
            break

        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]

        # Softmax weighting (sharper = more focused on top peaks)
        beta = 10.0 / (max_conv + 1e-15)
        weights = np.exp(beta * (peak_vals - max_conv))
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_s = max(0, k - (n - 1))
            i_e = min(n - 1, k)
            if i_s <= i_e:
                j_s = k - i_e
                j_e = k - i_s
                g[i_s:i_e + 1] += w * a[j_s:j_e + 1][::-1]

        v = momentum * v + g
        vn = np.linalg.norm(v)
        if vn < 1e-14:
            break

        a_new = a - lr * v / vn
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall % 30 == 0:
                lr *= 0.85
                v *= 0.4
                a = best_a
            if lr < 1e-12:
                break

        it += 1
        if verbose and it % 200 == 0:
            print(f"    SG it{it}: obj={best_obj:.8f}, lr={lr:.6f}")

    return best_a


def smoothed_gradient_descent(a, time_budget, tau_init=5.0, tau_end=500.0,
                               verbose=False):
    """Nesterov smoothed gradient descent on log-sum-exp objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta_mom = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        # Nesterov lookahead
        y = a + beta_mom * v
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy < 1e-10:
            y = best_a.copy()
        else:
            y = y / sy

        # Compute smoothed gradient via FFT
        conv = compute_conv(y)
        max_c = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_c))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_f = fft(y, N)
        g_raw = ifft(P * np.conj(A_f)).real
        g = 2.0 * g_raw[:n]

        # Barzilai-Borwein step size
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy_dot = float(np.dot(s_vec, y_vec))
            if ss > 1e-14 and abs(sy_dot) > 1e-12:
                bb_lr = ss / abs(sy_dot)
                lr = 0.5 * lr + 0.5 * np.clip(bb_lr, 1e-10, 5.0 / np.sqrt(n))

        v = beta_mom * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9997)

        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it

        if verbose and it % 3000 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}")

    return best_a


def skewed_init(n, rng, mode=0):
    """Generate diverse skewed initializations (asymmetric shapes)."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        lam = rng.uniform(2, 6)
        a = np.exp(-lam * x)
    elif mode == 1:
        p = rng.uniform(1, 5)
        a = (1 - x + 1e-4) ** p
    elif mode == 2:
        center = rng.uniform(0.2, 0.6)
        width = rng.uniform(0.08, 0.3)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    elif mode == 3:
        c = rng.uniform(0.3, 0.7)
        k = rng.uniform(10, 40)
        a = 1.0 / (1.0 + np.exp(-k * (x - c)))
    elif mode == 4:
        c1 = rng.uniform(0.1, 0.3)
        c2 = rng.uniform(0.6, 0.9)
        w1 = rng.uniform(0.05, 0.15)
        w2 = rng.uniform(0.05, 0.15)
        a = 3 * np.exp(-((x - c1) ** 2) / (2 * w1 ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w2 ** 2))
    elif mode == 5:
        a = 1 + rng.uniform(2, 8) * x
    elif mode == 6:
        p = rng.uniform(1.5, 4)
        a = (1 - x + 0.01) ** p * np.maximum(1 + 0.15 * rng.normal(0, 1, n), 0.1)
    elif mode == 7:
        # Power law with peak
        p = rng.uniform(1, 3)
        a = (1 - x + 1e-4) ** p + 0.5 * np.exp(-((x - 0.5) ** 2) / 0.05)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if n == 2:
        x_new = np.linspace(0, 1, new_n)
        return np.maximum(a[0] + (a[1] - a[0]) * x_new, 0)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Multi-skew init -> LP polish (golden section, warm-start)
    -> QP hybrid -> subgradient -> GD -> multi-n -> final polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    n_current = 800

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}", flush=True)
    except Exception as e:
        print(f"Parent load failed: {e}", flush=True)

    if best_seq is None:
        best_seq = skewed_init(800, rng, mode=1)
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}", flush=True)

    # === Phase 1: Multi-skew initialization (10%) ===
    print(f"\n=== Phase 1: Multi-skew init (budget: {min(120, (deadline-start_time)*0.1):.0f}s) ===", flush=True)
    remaining = deadline - time.monotonic()
    p1_total = min(remaining * 0.1, 120.0)
    if p1_total > 15:
        per_init = p1_total / 8
        for mode in range(8):
            remaining = deadline - time.monotonic()
            if remaining < per_init * 0.5:
                break
            a_try = skewed_init(n_current, rng, mode=mode)
            # Quick LP to refine
            a_try = lp_polish_main(a_try, time_budget=min(per_init, remaining * 0.4),
                                   use_golden=True, seed=seed + mode)
            obj_try = compute_obj(a_try)
            status = " **" if obj_try < best_score else ""
            print(f"  Mode {mode}: obj={obj_try:.8f}{status}", flush=True)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()

    # === Phase 2: Heavy LP polishing with golden section (40%) ===
    print(f"\n=== Phase 2: LP Polish (golden section) ===", flush=True)
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.4, 450.0)

    # Round 1: Pure LP with golden section
    alloc1 = p2_budget * 0.35
    remaining = deadline - time.monotonic()
    alloc1 = min(alloc1, remaining * 0.4)
    if alloc1 > 20:
        print(f"  Round 1 (LP+golden): {alloc1:.0f}s", flush=True)
        a_r = lp_polish_main(best_seq, time_budget=alloc1, use_golden=True,
                             verbose=True, seed=seed)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  ** Round 1: {best_score:.8f}", flush=True)
        else:
            print(f"  Round 1: {obj_r:.8f}", flush=True)

    # Round 2: LP + QP hybrid
    alloc2 = p2_budget * 0.25
    remaining = deadline - time.monotonic()
    alloc2 = min(alloc2, remaining * 0.4)
    if alloc2 > 20:
        print(f"  Round 2 (LP+QP): {alloc2:.0f}s", flush=True)
        a_r = lp_polish_main(best_seq, time_budget=alloc2, use_golden=True,
                             use_qp=True, verbose=True, seed=seed + 1)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  ** Round 2: {best_score:.8f}", flush=True)
        else:
            print(f"  Round 2: {obj_r:.8f}", flush=True)
            # Perturb
            best_seq = best_seq * np.exp(rng.normal(0, 0.012, len(best_seq)))
            best_seq = np.maximum(best_seq, 0)
            best_seq = best_seq / best_seq.sum()

    # Round 3: LP + Geometric blend
    alloc3 = p2_budget * 0.2
    remaining = deadline - time.monotonic()
    alloc3 = min(alloc3, remaining * 0.4)
    if alloc3 > 15:
        print(f"  Round 3 (LP+geo): {alloc3:.0f}s", flush=True)
        a_r = lp_polish_main(best_seq, time_budget=alloc3, use_golden=True,
                             use_geo=True, verbose=True, seed=seed + 2)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  ** Round 3: {best_score:.8f}", flush=True)
        else:
            print(f"  Round 3: {obj_r:.8f}", flush=True)

    # Round 4: Full hybrid
    alloc4 = p2_budget * 0.2
    remaining = deadline - time.monotonic()
    alloc4 = min(alloc4, remaining * 0.4)
    if alloc4 > 15:
        print(f"  Round 4 (LP+QP+geo): {alloc4:.0f}s", flush=True)
        a_r = lp_polish_main(best_seq, time_budget=alloc4, use_golden=True,
                             use_qp=True, use_geo=True, verbose=True, seed=seed + 3)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  ** Round 4: {best_score:.8f}", flush=True)
        else:
            print(f"  Round 4: {obj_r:.8f}", flush=True)

    # === Phase 3: Subgradient refinement (10%) ===
    print(f"\n=== Phase 3: Subgradient ===", flush=True)
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 100.0)
    if p3_budget > 15:
        a_sg = subgradient_refine(best_seq, time_budget=p3_budget, K=15)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  ** SG: {best_score:.8f}", flush=True)
        else:
            print(f"  SG: {obj_sg:.8f}", flush=True)

    # === Phase 4: Smoothed GD (10%) ===
    print(f"\n=== Phase 4: Smoothed GD ===", flush=True)
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.1, 100.0)
    if p4_budget > 15:
        a_gd = smoothed_gradient_descent(best_seq, time_budget=p4_budget,
                                         tau_init=10.0, tau_end=600.0)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  ** GD: {best_score:.8f}", flush=True)
        else:
            print(f"  GD: {obj_gd:.8f}", flush=True)

    # === Phase 5: Multi-length exploration (15%) ===
    print(f"\n=== Phase 5: Multi-n exploration ===", flush=True)
    remaining = deadline - time.monotonic()
    n_opts = sorted(set([
        n_current - 100, n_current + 100, n_current + 200,
        n_current + 400, 700, 900, 1000, 1200
    ]))
    n_opts = [n for n in n_opts if 400 <= n <= 2000 and abs(n - n_current) >= 50]

    for n_try in n_opts[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.2, 60.0)
        a_try = lp_polish_main(a_try, time_budget=alloc, use_golden=True, seed=seed + n_try)
        obj_try = compute_obj(a_try)
        status = " **" if obj_try < best_score else ""
        print(f"  n={n_try}: obj={obj_try:.8f}{status}", flush=True)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try

    # === Phase 6: Final intensive polish (remaining) ===
    print(f"\n=== Phase 6: Final Polish ===", flush=True)
    remaining = deadline - time.monotonic()

    # Final LP with all options
    if remaining > 30:
        alloc = min(remaining * 0.35, 120.0)
        a_r = lp_polish_main(best_seq, time_budget=alloc, use_golden=True,
                             use_qp=True, use_geo=True, verbose=True, seed=seed + 100)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  ** Final LP: {best_score:.8f}", flush=True)
        else:
            print(f"  Final LP: {obj_r:.8f}", flush=True)

    # Final short SG
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.25, 60.0)
        a_sg = subgradient_refine(best_seq, time_budget=alloc, K=20)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  ** Final SG: {best_score:.8f}", flush=True)

    # Final micro-LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 40.0)
        a_r = lp_polish_main(best_seq, time_budget=alloc, use_golden=True, seed=seed + 200)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  ** Final micro-LP: {best_score:.8f}", flush=True)

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===", flush=True)

    return result
