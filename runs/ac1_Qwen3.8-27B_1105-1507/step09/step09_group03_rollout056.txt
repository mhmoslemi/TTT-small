    A = np.fft.fft(a, N)
    # We want grad[i] = 2 * sum_k p[k] * a[k-i]
    # = 2 * (p ⊛ a)[i] where ⊛ is cross-correlation
    # In FFT terms: cross-correlation = IFFT(conj(FFT(p)) * FFT(a))
    # Wait: (p ⊛ a)[i] = sum_k p[k+i] * conj(a[k]) for complex... 
    # For real: (p ⊛ a)[i] = sum_k p[k] * a[k-i] = sum_j p[j+i] * a[j]
    # = (p_shifted * a) where p_shifted[m] = p[m+i]
    # This is a convolution: sum_j a[j] * p[j+i]
    # = convolve(a, reverse(p))[n-1-i] ... let me just use the direct approach
    
    # Actually, the simplest: use np.convolve
    # grad[i] = 2 * sum_k p[k] * a[k-i]
    # = 2 * sum_j a[j] * p[j+i]  (substituting k=j+i)
    # = 2 * (a * p)[n-1-i] ... no
    
    # Let me just compute it directly with a matrix or FFT
    # Using the fact that this is a Toeplitz-like operation:
    # grad = 2 * T(a) where T is the cross-correlation operator
    
    # FFT approach:
    # Define b = reverse(a) = a[n-1], a[n-2], ..., a[0]
    # Then sum_k p[k]*a[k-i] = sum_k p[k]*b[n-1-(k-i)] = sum_k p[k]*b[n-1-k+i]
    # = conv(p, b)[n-1+i] (if we define conv appropriately)
    
    # Let me just use the direct formula with FFT:
    # We want c[i] = sum_k p[k] * a[k-i] for i = 0, ..., n-1
    # This is the cross-correlation (p ⊛ a)[i]
    # In FFT: IFFT(FFT(p) * conj(FFT(a))) gives the circular cross-correlation
    # But we need linear, so pad
    
    # Actually let me just use: grad_full = np.convolve(p[::-1], a, 'full')[:n] * 2
    # Wait: np.convolve(x, y, 'full')[k] = sum_j x[j]*y[k-j]
    # We want c[i] = sum_k p[k]*a[k-i]
    # Let j = k-i, then k = j+i: c[i] = sum_j p[j+i]*a[j]
    # = sum_j a[j]*p[j+i]
    # np.convolve(a, p, 'full')[i] = sum_j a[j]*p[i-j]
    # That's not quite what we want.
    
    # Let me use: c[i] = sum_j a[j]*p[j+i]
    # = sum_j a[j]*p_reversed[n-1-j-i] (if p has length 2n-1)
    # Hmm this is getting complicated. Let me just use the direct O(n*(2n-1)) approach for n=800.
    
    # Direct computation:
    grad = np.zeros(n)
    for i in range(n):
        k_min = i  # k-i >= 0 => k >= i
        k_max = (2*n - 2) - (n - 1 - i)  # k-i <= n-1 => k <= n-1+i
        k_max = min(2*n - 2, n - 1 + i)
        if k_min <= k_max:
            grad[i] = 2.0 * float(np.dot(p[k_min:k_max+1], a[0:k_max-k_min+1]))
    
    return grad, conv

# Actually the O(n*(2n-1)) direct computation is fine for n=800 (about 1.6M ops)
# But let me use a smarter FFT approach:
# c[i] = sum_{j=0}^{n-1} a[j] * p[j+i]
# This is a "sliding dot product" which is a convolution!
# c[i] = sum_j a[j] * p[j+i] = (a * reverse(p))[n-1-i] (with p reversed)
# Wait: (a * q)[k] = sum_j a[j] * q[k-j]
# If q = reverse(p) (length 2n-1), then q[m] = p[2n-2-m]
# (a * q)[k] = sum_j a[j] * q[k-j] = sum_j a[j] * p[2n-2-k+j]
# We want: sum_j a[j] * p[j+i]
# Set 2n-2-k+j = j+i => k = 2n-2-i
# So c[i] = (a * reverse(p))[2n-2-i]

# Even simpler: just use numpy's correlate or a manual FFT.
# For n=800, the direct O(n^2) approach takes about 0.01s in numpy, which is fine.

def nesterov_smoothed_descent(a, time_budget, tau_start=50.0, tau_end=1000.0):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.5)
        
        # Nesterov lookahead
        y = a + beta * v
        y = project_simplex(y)
        
        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] (with valid indices)
        # Efficient computation using FFT
        N = 1
        while N < 3 * n:
            N <<= 1
        P = np.fft.fft(p, N)
        A = np.fft.fft(y, N)
        # cross-correlation: IFFT(conj(P) * A) gives (p ⊛ y)
        # where (p ⊛ y)[i] = sum_k p[k] * conj(y[k-i]) = sum_k p[k]*y[k-i] for real
        corr = np.fft.ifft(np.conj(P) * A).real
        grad = 2.0 * corr[:n]
        
        # Barzilai-Borwein step size (simplified)
        a_new = project_simplex(y - lr * grad)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        # Update Nesterov velocity
        v = beta * (a_new - y)  # simplified momentum
        a = a_new
        it += 1
        
        # Adaptive learning rate
        lr = max(1e-8, lr * 0.9995)
        
        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 5.0 / np.sqrt(n) * 0.5
            best_it = it
        
        if it % 2000 == 0 and verbose:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}")
    
    return best_a

def coordinate_descent(a, time_budget):
    """Coordinate descent: try to reduce the dominant peak by adjusting one coefficient."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_k = int(np.argmax(conv))
        
        # Find top contributing pairs to the peak
        i_start = max(0, peak_k - (n - 1))
        i_end = min(n - 1, peak_k)
        
        if i_start > i_end:
            break
        
        # Compute contributions
        best_improvement = 0
        best_move = None
        
        # Try reducing the largest contributor
        for i in range(i_start, i_end + 1):
            j = peak_k - i
            if i == j:
                # Reducing a[i] reduces the contribution a[i]^2
                # Try reducing a[i] by trying to move mass to other positions
                for delta_frac in [0.1, 0.05, 0.02]:
                    delta = delta_frac * a[i]
                    for tgt in [rng.integers(0, n), (i + n//2) % n, (i + n//3) % n]:
                        a_try = a.copy()
                        a_try[i] -= delta
                        a_try[tgt] += delta
                        if a_try[i] < 0 or a_try[tgt] < 0:
                            continue
                        a_try = a_try / a_try.sum()
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-13:
                            improvement = best_obj - obj_try
                            if improvement > best_improvement:
                                best_improvement = improvement
                                best_move = a_try.copy()
                if best_move is not None:
                    break
            else:
                # Try reducing either a[i] or a[j]
                for src in [i, j]:
                    for delta_frac in [0.05, 0.02, 0.01]:
                        delta = delta_frac * a[src]
                        if delta > a[src]:
                            continue
                        for tgt in [rng.integers(0, n), (src + n//2) % n]:
                            a_try = a.copy()
                            a_try[src] -= delta
                            a_try[tgt] += delta
                            if a_try[src] < 0:
                                continue
                            a_try = a_try / a_try.sum()
                            obj_try = compute_obj(a_try)
                            if obj_try < best_obj - 1e-13:
                                improvement = best_obj - obj_try
                                if improvement > best_improvement:
                                    best_improvement = improvement
                                    best_move = a_try.copy()
                    if best_move is not None:
                        break
            if best_move is not None:
                break
        
        if best_move is not None:
            best_obj = compute_obj(best_move)
            best_a = best_move
            a = best_move
            stall = 0
        else:
            stall += 1
            if stall > 30:
                # Perturb
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                if it > 200:
                    break
        
        it += 1
    
    return best_a

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Comb pattern
        a = np.zeros(n)
        a[::2] = 1.0
        a = a / a.sum()
    elif mode == 3:
        # Two bumps
        x = np.arange(n)
        a = np.exp(-((x - n//4)**2) / (2*(n//8)**2)) + np.exp(-((x - 3*n//4)**2) / (2*(n//8)**2))
        a = a / a.sum()
    elif mode == 4:
        # Triangular
        x = np.linspace(-1, 1, n)
        a = 1 - np.abs(x)
        a = a / a.sum()
    elif mode == 5:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 6:
        # Power law decay
        x = np.arange(1, n+1, dtype=float)
        a = x ** (-rng.uniform(0.5, 2.0))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)
