def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_toeplitz_matrix(a):
    """Build (2n-1) x n matrix A where A[k,i] = a[k-i] if valid, else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        # A[k,i] = a[k-i] for k-i in [0, n-1], i.e., k in [i, i+n-1]
        k_start = i
        k_end = min(i + n, 2 * n - 1)
        A[k_start:k_end, i] = a[:k_end - k_start]
    return A

def solve_lp_cross(a, solver_list=None):
    """

    A = build_toeplitz_matrix(a_norm)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based optimization (cross-convolution LP) as the primary driver,
combined with coordinate descent, multi-n exploration, and diverse initializations.
Key improvements: aggressive LP iterations, wider n exploration, and
line-search blending for LP step selection.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_toeplitz(a):
    """Build (2n-1) x n matrix A where A[k,i] = a[k-i] if 0<=k-i<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(i + n, 2 * n - 1)
        A[k_start:k_end, i] = a[:k_end - k_start]
    return A


def solve_lp_cross(a, solver_list=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_toeplitz(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]

    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_step(a, solver_list=None, n_line_evals=16):
    """
    One LP step: solve cross-conv LP, line-search blend factor, return improved point.
    """
    n = len(a)
    a = a / float(np.sum(a))
    g_lp, M_lp = solve_lp_cross(a, solver_list)
    if g_lp is None:
        return a.copy(), compute_obj(a)

    # Line search on t in [0, 1]: a_t = (1-t)*a + t*g_lp
    gr = (np.sqrt(5) - 1) / 2.0
    lo_t, hi_t = 0.0, 1.0

    def blend_obj(t):
        a_t = (1.0 - t) * a + t * g_lp
        a_t = np.maximum(a_t, 0.0)
        s = float(np.sum(a_t))
        if s < 1e-10:
            return np.inf
        a_t = a_t / s
        conv = compute_conv(a_t)
        return float(np.max(conv))

    f_lo = float(np.max(compute_conv(a)))
    f_hi = blend_obj(1.0)

    best_t = 0.0
    best_f = f_lo
    if f_hi < best_f:
        best_t = 1.0
        best_f = f_hi

    c_t = hi_t - gr * (hi_t - lo_t)
    d_t = lo_t + gr * (hi_t - lo_t)
    f_c = blend_obj(c_t)
    f_d = blend_obj(d_t)

    if f_c < best_f:
        best_t = c_t
        best_f = f_c
    if f_d < best_f:
        best_t = d_t
        best_f = f_d

    for _ in range(n_line_evals):
        if f_c < f_d:
            hi_t = d_t
            d_t = c_t
            f_d = f_c
            c_t = hi_t - gr * (hi_t - lo_t)
            f_c = blend_obj(c_t)
        else:
            lo_t = c_t
            c_t = d_t
            f_c = f_d
            d_t = lo_t + gr * (hi_t - lo_t)
            f_d = blend_obj(d_t)

        if f_c < best_f:
            best_t = c_t
            best_f = f_c
        if f_d < best_f:
            best_t = d_t
            best_f = f_d

    a_new = (1.0 - best_t) * a + best_t * g_lp
    a_new = np.maximum(a_new, 0.0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        return a.copy(), f_lo
    a_new = a_new / s_new
    return a_new, compute_obj(a_new)


def coord_descent(a, max_sweeps=50, time_budget=None, start_time=0, rng=None):
    """Coordinate descent on normalized self-convolution peak with incremental updates."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
    a = a / S
    conv = compute_conv(a)

    best_obj = float(np.max(conv))
    best_a = a.copy()
    stagnant = 0

    for sweep in range(max_sweeps):
        if time_budget is not None and time.monotonic() - start_time > time_budget:
            break

        obj_before = float(np.max(conv))
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for k in order:
            d = np.zeros(2 * n - 1)
            d[k:k + n] = a

            lo = -a[k]
            hi = max(2.0 / n, lo + 2.0 / n)

            # Bisection search for optimal delta
            for _ in range(22):
                mid = (lo + hi) / 2.0
                val = conv + 2.0 * mid * d
                val[2 * k] += mid * mid
                f_mid = float(np.max(val))

                mid2 = (lo + mid) / 2.0
                val2 = conv + 2.0 * mid2 * d
                val2[2 * k] += mid2 * mid2
                f_mid2 = float(np.max(val2))

                if f_mid2 < f_mid:
                    hi = mid
                else:
                    lo = mid

            delta = (lo + hi) / 2.0
            val_new = conv + 2.0 * delta * d
            val_new[2 * k] += delta * delta

            if float(np.max(val_new)) >= obj_before - 1e-14:
                continue

            a[k] = max(0.0, a[k] + delta)
            conv += 2.0 * delta * d
            conv[2 * k] += delta * delta

            S_new = float(np.sum(a))
            if S_new < 1e-10:
                a[:] = 1.0 / n
                conv = compute_conv(a)
            else:
                a /= S_new
                conv /= (S_new * S_new)

        obj_after = float(np.max(conv))
        if obj_after < best_obj - 1e-12:
            best_obj = obj_after
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 4 and rng is not None:
                scale = 0.015 * (1 + stagnant * 0.5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv = compute_conv(a)
                best_obj = float(np.max(conv))
                stagnant = 0

    return best_a, compute_obj(best_a)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(0, np.pi, n)
        a = np.sin(x)
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    elif mode == 4:
        x = np.arange(n, dtype=np.float64)
        mid = (n - 1) / 2.0
        a = 1.0 - np.abs(x - mid) / (mid + 1.0)
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    elif mode == 5:
        x = np.arange(n, dtype=np.float64)
        mid = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-(x - mid) ** 2 / (2.0 * sigma ** 2))
        return a / a.sum()
    elif mode == 6:
        a = np.ones(n)
        taper = max(2, n // 8)
        t = np.linspace(0, 1, taper)
        a[:taper] = np.sin(t * np.pi / 2)
        a[-taper:] = np.sin((1.0 - t) * np.pi / 2)
        return a / a.sum()
    elif mode == 7:
        x = np.linspace(0.01, 1.0, n)
        a = x ** 0.3 * (1.0 - x) ** 0.3

    elif mode == 8:

def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)

        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        mode = (r + 2) % 9
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.25 + 0.08 * r
        a_r = (1.0 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        # LP
        for _ in range(3):
            if time.monotonic() > deadline - 40:
                break
            a_r, _ = lp_step(a_r, solver_list=solver_list, n_line_evals=8)

        # CD
        remaining = deadline - time.monotonic()
        if remaining > 25:
            a_r, obj_r = coord_descent(a_r, max_sweeps=10,
                                        time_budget=min(remaining * 0.3, 25.0),
                                        start_time=time.monotonic(), rng=rng)
        else:
            obj_r = compute_obj(a_r)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final Intensive Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Final LP with more line search evaluations
    if remaining > 30:
        consecutive_fails = 0
        for it in range(20):
            if time.monotonic() > deadline - 15:
                break
            current = best_seq.copy()
            current, obj = lp_step(current, solver_list=solver_list, n_line_evals=18)
            if obj < best_score - 1e-10:
                best_score = obj
                best_seq = current.copy()
                consecutive_fails = 0
                print(f"  Final LP {it}: obj={best_score:.8f} **")
            else:
                consecutive_fails += 1
                if consecutive_fails >= 3:
                    break

    # Final CD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final, obj_final = coord_descent(best_seq, max_sweeps=40,
                                            time_budget=min(remaining * 0.6, 50.0),
                                            start_time=time.monotonic(), rng=rng)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
