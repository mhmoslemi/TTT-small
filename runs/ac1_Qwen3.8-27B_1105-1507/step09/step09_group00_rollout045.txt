            best_a = a_t.copy()

    return best_t, best_a, best_obj


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)

    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1

    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]

    return g, conv


def adaptive_smoothed_descent(a, time_budget=100.0, tau_init=5.0, tau_end=500.0, verbose=False):
    """
    Nesterov accelerated gradient descent with adaptive cosine annealing.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    base_lr = 3.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Cosine anneal tau
        if progress < 0.1:
            tau = tau_init
        else:
            frac = (progress - 0.1) / 0.9
            cos_val = 0.5 * (1 + np.cos(np.pi * frac))
            tau = tau_init + (tau_end - tau_init) * (1 - cos_val)

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute gradient at y
        g, conv = smoothed_gradient_fft(y, tau)

        # Adaptive learning rate
        g_norm = float(np.linalg.norm(g))
        if g_norm > 1e-15:
            adaptive_lr = 2.0 / (g_norm + 1e-10)
            adaptive_lr = min(max(adaptive_lr, 0.1 * base_lr), 4.0 * base_lr)
            lr = 0.8 * lr + 0.2 * adaptive_lr

        # Update
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1

        lr = max(1e-9, lr * 0.9998)

        # Reset if stuck
        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it

        if verbose and it % 2000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

    return best_a


def spectral_perturbation(a, rng, strength=0.03):
    """Add structured spectral perturbation to diversify search."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1

    A = rfft(a, size)

    # Perturb the phase of higher frequency components
    max_k = min(size // 2, n)
    for k in range(2, max_k):
        if rng.random() < 0.3:
            phase_shift = rng.uniform(-strength, strength)
            A[k] *= np.exp(1j * phase_shift)
            A[size - k if k < size else k] = np.conj(A[k]) if k < size // 2 else A[k]

    # Also perturbation in amplitude
    for k in range(1, max_k):
        if rng.random() < 0.2:
            amp_change = 1.0 + rng.uniform(-strength, strength)
            A[k] *= amp_change

    a_new = irfft(A, size)[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lowfreq_perturbation(a, rng, strength=0.05):
    """Perturb only low-frequency Fourier modes."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = 4
    x = np.arange(n)
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.0) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * x + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lp_polish_intensive(a, time_budget=200.0, verbose=False, seed=42):
    """
    Intensive LP polishing with line search and adaptive step decay.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    stall_count = 0
    step_scale = 1.0  # Adaptive scaling for perturbation strength

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, M_lp, dual = solve_lp(a)
        if g_lp is None:
            break

        iter_count += 1

        # Line search for optimal blend
        t_best, a_candidate, obj_candidate = line_search_lp_step(a, g_lp, max_iter=20)

        if obj_candidate < best_obj - 1e-14:
            best_obj = obj_candidate
            best_a = a_candidate.copy()
            a = a_candidate
            stall_count = 0
            if verbose and iter_count % 10 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={t_best:.4f}, M={M_lp:.6f}")
        else:
            # No improvement from LP blend alone
            stall_count += 1
            if stall_count >= 3:
                # Try perturbation + LP
                strength = 0.01 * min(stall_count, 5) + 0.005
                a_pert = lowfreq_perturbation(best_a, rng, strength=strength)
                g_lp2, M_lp2, _ = solve_lp(a_pert)
                if g_lp2 is not None:
                    t_best2, a_cand2, obj_cand2 = line_search_lp_step(a_pert, g_lp2, max_iter=15)
                    if obj_cand2 < best_obj - 1e-14:
                        best_obj = obj_cand2
                        best_a = a_cand2.copy()
                        a = a_cand2
                        stall_count = 0
                        if verbose:
                            print(f"    LP+pert it{iter_count}: obj={best_obj:.8f}")
                        continue
                # If perturbation didn't help, try spectral perturbation
                if stall_count >= 5:
                    a_pert2 = spectral_perturbation(best_a, rng, strength=0.02)
                    g_lp3, M_lp3, _ = solve_lp(a_pert2)
                    if g_lp3 is not None:
                        t_best3, a_cand3, obj_cand3 = line_search_lp_step(a_pert2, g_lp3, max_iter=15)
                        if obj_cand3 < best_obj - 1e-14:
                            best_obj = obj_cand3
                            best_a = a_cand3.copy()
                            a = a_cand3
                            stall_count = 0
                            continue
                # Last resort: break if stuck too long
                if stall_count >= 10:
                    a = best_a
                    stall_count = 0
                    step_scale = max(0.5, step_scale * 0.8)
                    if step_scale < 0.1:
                        break
            # Small step toward LP solution even without improvement
            t_small = 0.02 * step_scale
            a_small = (1 - t_small) * a + t_small * g_lp
            a_small = np.maximum(a_small, 0)
            s_small = float(np.sum(a_small))
            if s_small > 1e-10:
                a = a_small / s_small
            a = best_a  # Reset to best

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def dual_guided_direction(a, dual):
    """
    Use LP dual variables to construct a direction that emphasizes
    positions where the convolution is close to the maximum.
    """
    if dual is None:
        return a.copy()
    n = len(a)
    # dual[i] >= 0 indicates how "tight" constraint i is
    # Positions with large dual values are where conv is at max
    # We want to reduce those positions
    # Direction: move mass away from high-dual positions
    conv = compute_conv(a / np.sum(a))
    # Weighted average position of max convolution
    w = dual / max(np.max(dual), 1e-15)
    # This gives a signal of where to "push"
    # For now, just return a slightly perturbed version
    return a.copy()
