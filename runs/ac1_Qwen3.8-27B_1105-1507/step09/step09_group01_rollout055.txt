import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def nesterov_subgradient(a, time_budget=200.0, verbose=False):
    """Nesterov-accelerated subgradient method on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    v = np.zeros(n)
    prev_a = a.copy()
    it = 0
    best_it = 0
    lr_init = 5.0 / np.sqrt(n)
    lr = lr_init

    while time.monotonic() - start < time_budget:
        if it % 10000 == 0 and verbose:
            print(f"    NS it{it}: obj={best_obj:.8f}, lr={lr:.4f}, t={time.monotonic()-start:.0f}s")
        
        y = a + beta * v
        y = project_simplex(y)
        
        # Subgradient of max_k conv(a,a)[k] w.r.t. a:
        # If k* is argmax, subgrad[i] = 2 * sum_{k*: k*=i+j} a[j] = 2 * a_conv[i]
        # where a_conv[i] = sum_j a[j]*a[i+j] ... no.
        # conv(a,a)[k] = sum_j a[j]*a[k-j]
        # d(conv[k])/d(a[i]) = a[k-i] + a[k-i] (if k-i != i) or 2*a[i] (if k=i... wait)
        # Actually: d/d(a[i]) sum_j a[j]*a[k-j] = a[k-i] (from j=i term) + a[k-i] (from j=k-i term, if k-i != i)
        # = 2*a[k-i] if k != 2i, and 2*a[i] if k=2i
        # So subgrad[i] for peak k* is: 2*a[k*-i] (if k* != 2i) or 2*a[i] (if k*=2i)
        # More simply: subgrad[i] = 2 * a_conv_at_peak[i] where a_conv_at_peak is the "cross-correlation"
        
        conv = compute_conv(a)
        peak_idx = np.argmax(conv)
        
        # Subgradient: for peak at k*, subgrad[i] = 2*a[k*-i] if 0 <= k*-i < n, else 0
        # (with the special case k*=2i giving 2*a[i], which is the same formula)
        k_star = int(peak_idx)
        subgrad = np.zeros(n)
        j_start = max(0, k_star - (n - 1))
        j_end = min(n - 1, k_star)
        for j in range(j_start, j_end + 1):
            subgrad[j] = 2.0 * a[k_star - j]
        
        v = beta * v - lr * subgrad
        a_new = y + v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        prev_a = a.copy()
        a = a_new
        it += 1
        lr = max(1e-6, lr_init / np.sqrt(1.0 + it * 0.01))
        
        if it - best_it > 50000:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diversified_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        # Beta-like distribution
        alpha = rng.uniform(0.5, 3.0)
        beta_p = rng.uniform(0.5, 3.0)
        x = np.linspace(0.01, 0.99, n)
        a = x**alpha * (1-x)**beta_p
        a = a / a.sum()
    elif mode == 3:
        # Semicircle
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x**2, 0))
        a = a / a.sum()
    elif mode == 4:
        # Gaussian centered
        x = np.linspace(0, 1, n)
        mu = rng.uniform(0.3, 0.7)
        sigma = rng.uniform(0.1, 0.3)
        a = np.exp(-((x - mu)**2) / (2*sigma**2))
        a = a / a.sum()
    elif mode == 5:
        # Two peaks
        x = np.linspace(0, 1, n)
        mu1, mu2 = 0.3, 0.7
        sigma = rng.uniform(0.05, 0.15)
        a = np.exp(-((x-mu1)**2)/(2*sigma**2)) + np.exp(-((x-mu2)**2)/(2*sigma**2))
        a = a / a.sum()
    elif mode == 6:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x**rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 7:
        # High-pass filtered uniform
        a = np.ones(n)
        x = np.linspace(0, 1, n)
        for f in rng.integers(2, 8, size=3):
            a += 0.2 * np.sin(2*np.pi*f*x + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses exact coordinate descent (solving 1D subproblems via ternary search),
LP-based direction finding with golden-section line search, Nesterov-accelerated
subgradient method, multi-length exploration, and perturbation restarts.
Key innovation: coordinate descent with exact 1D optimization of each coordinate,
exploiting the linearity of convolution in each coordinate.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coordinate_descent(a, time_budget=200.0, verbose=False):
    """
    Coordinate descent with exact 1D subproblem solving.
    For each coordinate i, finds the delta that minimizes the objective
    when a[i] is changed by delta (with renormalization to sum=1).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    sweep = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        sweep_improved = False

        for i in range(n):
            # When we change a[i] to a[i] + delta (with sum=1 constraint):
            # New normalized: a_new = (a + delta*e_i) / (1 + delta)
            # conv(a_new, a_new)[k] = [conv(a,a)[k] + 2*delta*a[k-i]*1[0<=k-i<n] + delta^2*1[k=2i]] / (1+delta)^2
            # Objective = 2n * max_k(conv_new[k])
            
            # Affected k: k in [i, i+n-1] (where k-i in [0, n-1])
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            # Non-affected max
            if k_start > 0 or k_end < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_start:k_end + 1] = False
                non_affected_max = float(np.max(conv[mask]))
            else:
                non_affected_max = 0.0

            # Affected part
            affected_k = np.arange(k_start, k_end + 1)
            affected_conv = conv[affected_k].copy()
            k_offset = affected_k - i  # = k - i, in [0, n-1]
            lin_coeffs = 2.0 * a[k_offset]
            quad_mask = (affected_k == 2 * i)

            # f(delta) = max(non_affected_max, max(affected_conv + lin_coeffs*delta + quad_mask*delta^2)) / (1+delta)^2
            # subject to delta > -1 (for normalization to be valid) and a[i]+delta >= 0
            
            delta_min = -a[i] + 1e-12
            # Upper bound: delta should not make a[i]+delta too large
            # If a[i]+delta = D, then conv_new[2i] >= D^2/(1+delta)^2
            # We want this <= current max, so D/(1+delta) <= sqrt(max_conv)
            # (a[i]+delta)/(1+delta) <= sqrt(max_conv)
            # a[i]+delta <= sqrt(max_conv)*(1+delta)
            # a[i] - sqrt(max_conv) <= delta*(sqrt(max_conv) - 1)
            # This gives a bound but let's just use a practical one
            delta_max = 0.5
            
            if delta_max <= delta_min:
                continue

            # Ternary search for optimal delta
            def f_delta(delta):
                if delta <= -a[i] or delta <= -1.0:
                    return 1e10
                denom = (1.0 + delta) ** 2
                new_vals = affected_conv + lin_coeffs * delta
                if quad_mask.any():
                    new_vals = new_vals.copy()
                    new_vals[quad_mask] += delta * delta
                numerator = max(non_affected_max, float(np.max(new_vals)))
                return numerator / denom

            lo = delta_min
            hi = delta_max
            # Quick check: is delta=0 already optimal?
            f0 = f_delta(0.0)
            
            # Sample to find a good bracket
            d_test1 = lo + 0.3 * (hi - lo)
            d_test2 = lo + 0.7 * (hi - lo)
            f1 = f_delta(d_test1)
            f2 = f_delta(d_test2)
            
            if f1 < f0 and f1 < f2:
                hi = d_test2
                lo = d_test1 if f1 < f2 else lo
            elif f2 < f0 and f2 < f1:
                lo = d_test1
                hi = d_test2
            elif f1 < f2:
                hi = d_test2
            else:
                lo = d_test1

            # Ensure 0 is in the bracket if it's good
            if f0 <= f1 and f0 <= f2:
                lo = min(lo, 0.0)
                hi = max(hi, 0.0)
            lo = max(lo, delta_min)
            hi = min(hi, delta_max)
            
            if hi - lo < 1e-8:
                delta_opt = (lo + hi) / 2.0
            else:
                # Ternary search
                for _ in range(35):
                    if hi - lo < 1e-10:
                        break
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    fm1 = f_delta(m1)
                    fm2 = f_delta(m2)
                    if fm1 < fm2:
                        hi = m2
                    else:
                        lo = m1
                delta_opt = (lo + hi) / 2.0
                delta_opt = max(delta_opt, delta_min)
                delta_opt = min(delta_opt, delta_max)

            if abs(delta_opt) > 1e-10:
                a[i] += delta_opt
                a[i] = max(a[i], 0.0)
                sweep_improved = True

        if sweep_improved:
            a = a / a.sum()
            obj_new = compute_obj(a)
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        sweep += 1
        if verbose and sweep % 5 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")

        if no_improve >= 4:
            break

    return best_a


def lp_polish(a, time_budget=200.0, verbose=False, seed=42):
    """
    LP-based polishing: solve LP to find direction minimizing max(conv(a,g)),
    then golden-section line search along the blend direction.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0

    # Build constraint matrix A where (A@g)[k] = conv(a,g)[k]
    # A[k,m] = a[k-m] if max(0,k-n+1) <= m <= min(k,n-1)
    A = np.zeros((2 * n - 1, n))
    for m in range(n):
        k_lo = m
        k_hi = min(m + n, 2 * n - 1)
        A[k_lo:k_hi, m] = a[:k_hi - k_lo]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    while time.monotonic() - start < time_budget:
        # Solve LP
        g_val = None
        M_val = None
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver, warm_start=True)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    M_val = float(M.value)
                    break
            except Exception:
                continue

        if g_val is None:
            break

        s_g = float(np.sum(g_val))
        if s_g < 1e-10:
            break
        g_val = g_val / s_g

        # Golden section line search
        def blend_obj(t):
            a_try = (1.0 - t) * best_a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return 1e10
            a_try = a_try / s_try
            return compute_obj(a_try)

        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.01, 1.0
        c_val = hi - gr * (hi - lo)
        d_val = lo + gr * (hi - lo)
        fc = blend_obj(c_val)
        fd = blend_obj(d_val)
        for _ in range(28):
            if hi - lo < 0.015:
                break
            if fc < fd:
                hi = d_val
                d_val = c_val
                fd = fc
                c_val = hi - gr * (hi - lo)
                fc = blend_obj(c_val)
            else:
                lo = c_val
                c_val = d_val
                fc = fd
                d_val = lo + gr * (hi - lo)
                fd = blend_obj(d_val)

        t_opt = (lo + hi) / 2.0
        a_try = (1.0 - t_opt) * best_a + t_opt * g_val
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            break
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        it += 1

        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            no_improve = 0
            if verbose and it % 3 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.3f}, M={M_val:.6f}, el={time.monotonic()-start:.0f}s")
        else:
            no_improve += 1
            if no_improve == 2:
                # Try smaller step
                t_small = 0.05
                a_try2 = (1.0 - t_small) * best_a + t_small * g_val
                a_try2 = np.maximum(a_try2, 0)
                a_try2 = a_try2 / a_try2.sum()
                obj_try2 = compute_obj(a_try2)
                if obj_try2 < best_obj - 1e-14:
                    best_obj = obj_try2
                    best_a = a_try2.copy()
                    no_improve = 0
            elif no_improve == 4:
                # Perturb
                scale = 0.015
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                best_a = a_pert
                best_obj = compute_obj(best_a)
                no_improve = 0
            elif no_improve >= 6:
                if it > 40:
                    break
                no_improve = 0

    return best_a


def nesterov_subgradient(a, time_budget=200.0, verbose=False):
    """Nesterov-accelerated subgradient method on the probability simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    v = np.zeros(n)
    it = 0
    best_it = 0
    lr_init = 3.0 / np.sqrt(n)
    lr = lr_init

    while time.monotonic() - start < time_budget:
        if it % 5000 == 0 and it > 0 and verbose:
            print(f"    NS it{it}: obj={best_obj:.8f}, lr={lr:.5f}, t={time.monotonic()-start:.0f}s")

        y = a + beta * v
        y = project_simplex(y)

        # Compute subgradient at y
        conv_y = compute_conv(y)
        peak_idx = int(np.argmax(conv_y))
        k_star = peak_idx

        # Subgradient: subgrad[i] = 2*y[k_star - i] if 0 <= k_star - i < n
        subgrad = np.zeros(n)
        j_lo = max(0, k_star - (n - 1))
        j_hi = min(n - 1, k_star)
        subgrad[j_lo:j_hi + 1] = 2.0 * y[k_star - j_hi:k_star - j_lo + 1][::-1]

        v = beta * v - lr * subgrad
        a_new = y + v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-7, lr_init / (1.0 + 0.005 * it))

        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diversified_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.05, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        alpha = rng.uniform(0.3, 4.0)
        beta_p = rng.uniform(0.3, 4.0)
        x = np.linspace(0.005, 0.995, n)
        a = x ** alpha * (1 - x) ** beta_p
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x ** 2, 0.01))
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        mu = rng.uniform(0.25, 0.75)
        sigma = rng.uniform(0.08, 0.3)
        a = np.exp(-((x - mu) ** 2) / (2 * sigma ** 2))
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        sigma = rng.uniform(0.05, 0.12)
        a = np.exp(-((x - 0.3) ** 2) / (2 * sigma ** 2)) + np.exp(-((x - 0.7) ** 2) / (2 * sigma ** 2))
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

        alloc1 = min(remaining * 0.25, 50.0)
        if alloc1 > 10:
            a_try = coordinate_descent(a_try, time_budget=alloc1)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.3, 100.0)
        if alloc2 > 15:
            a_try = lp_polish(a_try, time_budget=alloc2, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_curr = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Perturbation + Re-optimize ===
    print(f"\n=== Phase 6: Perturbation Restarts ===")
    remaining = deadline - time.monotonic()

    for r in range(12):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        # Various perturbation strategies
        if r < 4:
            # Gaussian perturbation
            scale = 0.005 + 0.005 * r
            a_pert = best_seq * np.exp(rng.normal(0, scale, n_curr))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
        elif r < 8:
            # Blend with diverse init
            a_div = diversified_init(n_curr, rng, mode=(r - 4) % 7)
            blend = 0.2 + 0.05 * (r - 4)
            a_pert = (1 - blend) * best_seq + blend * a_div
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
        else:
            # Random coordinate zeroing + perturbation
            a_pert = best_seq.copy()
            n_zero = rng.integers(5, max(6, n_curr // 20))
            zero_idx = rng.choice(n_curr, size=n_zero, replace=False)
            a_pert[zero_idx] = 0
            a_pert = a_pert * np.exp(rng.normal(0, 0.02, n_curr))
            a_pert = np.maximum(a_pert, 0)
            s_p = a_pert.sum()
            if s_p < 1e-6:
                continue
            a_pert = a_pert / s_p

        # Optimize the perturbed sequence
        alloc1 = min(remaining * 0.3, 60.0)
        if alloc1 > 10:
            a_pert = lp_polish(a_pert, time_budget=alloc1, seed=300 + r)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.2, 40.0)
        if alloc2 > 10:
            a_pert = coordinate_descent(a_pert, time_budget=alloc2)

        obj_pert = compute_obj(a_pert)
        if obj_pert < best_score:
            best_score = obj_pert
            best_seq = a_pert.copy()
            n_curr = len(best_seq)
            print(f"  ** Perturb {r}: obj={best_score:.8f}")
        else:
            print(f"  Perturb {r}: obj={obj_pert:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Alternating CD + LP for remaining time
    cycles = 0
    while remaining > 40 and cycles < 6:
        # LP
        alloc = min(remaining * 0.4, 120.0)
        if alloc > 20:
            best_seq = lp_polish(best_seq.copy(), time_budget=alloc, seed=600 + cycles)
            best_score = compute_obj(best_seq)
            print(f"  Final LP {cycles}: obj={best_score:.8f}")
            cycles += 1

        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # CD
        alloc = min(remaining * 0.4, 80.0)
        if alloc > 15:
            best_seq = coordinate_descent(best_seq.copy(), time_budget=alloc)
            best_score = compute_obj(best_seq)
            print(f"  Final CD {cycles}: obj={best_score:.8f}")
            cycles += 1

        remaining = deadline - time.monotonic()

    # Final Nesterov pass
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.5, 60.0)
        a_fin = nesterov_subgradient(best_seq.copy(), time_budget=alloc)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Final NS: obj={best_score:.8f} **")

    # Validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
