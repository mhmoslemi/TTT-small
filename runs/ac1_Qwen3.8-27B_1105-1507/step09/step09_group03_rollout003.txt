"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based smoothed gradient with Nesterov acceleration, Frank-Wolfe hybrid,
and multi-scale exploration. Key: correct cross-correlation gradient via FFT,
adaptive temperature, and KKT-based refinement.
"""

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def compute_gradient(a, p, N):
    """
    Compute gradient of smoothed objective.
    g_m = 2 * sum_k p_k * a_{k-m} (cross-correlation of p and a)
    p has length 2n-1, a has length n.
    """
    n = len(a)
    P = fft(p, N)
    A = fft(a, N)
    corr = ifft(P * np.conj(A)).real
    return 2.0 * corr[:n]

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_smoothing(conv, tau):
    """Compute softmax weights p_k from convolution values."""
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    s = float(np.sum(exp_vals))
    if s < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_vals / s

def nesterov_smoothed_descent(a, time_budget=100.0, tau_init=20.0, tau_end=2000.0, verbose=False):
    """
    Nesterov accelerated gradient descent on log-sum-exp smoothed objective.
    Operates on simplex (sum a = 1, a >= 0).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    # FFT size for gradient computation
    N = 1
    while N < 3 * n:
        N <<= 1
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.7  # momentum
    v = np.zeros(n)
    lr = 2.0 / np.sqrt(n)
    it = 0
    best_it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        # Exponential tau schedule
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        # Compute gradient
        conv = compute_conv(y)
        p = softmax_smoothing(conv, tau)
        g = compute_gradient(y, p, N)
        
        # Gradient step with momentum
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
        
        a = a_new
        it += 1
        
        # Adaptive learning rate
        if stall > 500:
            lr *= 0.5
            stall = 0
            if it - best_it > 5000:
                # Reset
                a = best_a.copy()
                v = np.zeros(n)
                lr = 2.0 / np.sqrt(n)
                beta = 0.7
                best_it = it
        
        if verbose and it % 2000 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")
    
    return best_a

def mirror_descent(a, time_budget=60.0, tau=50.0, verbose=False):
    """
    Mirror descent with entropy regularization (multiplicative weight updates).
    a_i_new ∝ a_i * exp(-eta * g_i)
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    N = 1
    while N < 3 * n:
        N <<= 1
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = 1.0 / np.sqrt(n)
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        p = softmax_smoothing(conv, tau)

def subgradient_descent(a, time_budget=60.0, verbose=False):
    """
    Direct subgradient method on the non-smooth objective.
    Uses the active peak to compute exact subgradient.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    t0 = 1.0
    it = 0
    stall = 0
    rng = np.random.default_rng(123)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Use top-k peaks for a richer subgradient
        k = min(5, len(conv))
        top_k = np.argsort(conv)[-k:]
        weights = np.zeros(len(conv))
        for idx in top_k:
            weights[idx] = 1.0
        weights /= k
        p = weights
        
        N = 1
        while N < 3 * n:
            N <<= 1
        g = compute_gradient(a, p, N)
        
        # Center gradient
        g_centered = g - np.mean(g)
        
        # Decreasing step size
        t = t0 / (np.sqrt(it + 1))
        
        # Gradient step
        a_new = a - t * g_centered
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 100:
                # Perturb
                a = best_a * np.exp(rng.normal(0, 0.01, n))
                a = np.maximum(a, 0)
                s_a = float(np.sum(a))
                if s_a > 1e-10:
                    a = a / s_a
                else:
                    a = best_a.copy()
                stall = 0
                t0 *= 0.5
        
        it += 1
        if verbose and it % 100 == 0:
            print(f"    SGD it{it}: obj={best_obj:.8f}, t={t:.6f}")
    
    return best_a

def frank_wolfe_step(a, p, N):
    """
    Frank-Wolfe (conditional gradient) step.
    Find vertex e_i that minimizes g_i, move towards it.
    """
    n = len(a)
    g = compute_gradient(a, p, N)
    i_star = int(np.argmin(g))
    # Optimal step size for FW: minimize f(a + alpha*(e_i - a))
    # Use line search
    e = np.zeros(n)
    e[i_star] = 1.0
    
    # Simple step size
    alpha_candidates = [1.0 / (it + 2) for it in [0, 1, 2]]
    # Actually, let's just use a fixed fraction
    best_alpha = 0.1
    best_obj = compute_obj(a)
    
    for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
        a_try = (1 - alpha) * a + alpha * e
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_alpha = alpha
    
    return (1 - best_alpha) * a + best_alpha * e, best_obj

def fw_hybrid(a, time_budget=60.0, verbose=False):
    """

    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)

    if s < 1e-10:

    beta = 0.7
    v = np.zeros(n)
    lr = 1.5 / np.sqrt(n)
    stall = 0

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

            conv = compute_conv(a)
            p = softmax_smoothing(conv, tau)

        else:
            # Gradient step
            y = a + beta * v
            y = np.maximum(y, 0)
            s_y = float(np.sum(y))
            if s_y < 1e-10:
                y = a.copy()
            else:
                y = y / s_y
            
            conv = compute_conv(y)
            p = softmax_smoothing(conv, tau)
            g = compute_gradient(y, p, N)
            
            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussians
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        # Fourier-based
        freqs = rng.integers(1, 8, size=6)
        phases = rng.uniform(0, 2 * np.pi, size=6)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 1)
        a = a / a.sum()
    elif mode == 6:
        # Bimodal
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

def kkt_refine(a, time_budget=30.0, verbose=False):
    """
    KKT-based refinement: at optimum, gradient should be constant on support.
    Iteratively adjust to equalize gradient on support.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    N = 1
    while N < 3 * n:
        N <<= 1
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    tau = 200.0
    lr = 0.1 / np.sqrt(n)
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        p = softmax_smoothing(conv, tau)
        g = compute_gradient(a, p, N)
        
        # Center gradient
        g_c = g - np.mean(g)
        
        # Only update on support (where a > threshold)
        support = a > 1e-6
        if np.sum(support) < 2:
            break
        
        # On support, try to make gradient constant
        # a_new = a - lr * (g_c - mean(g_c[support])) * support
        g_target = np.mean(g_c[support])
        g_diff = np.zeros(n)
        g_diff[support] = g_c[support] - g_target
        
        a_new = a - lr * g_diff
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        
        it += 1
        lr *= 0.999
        
        if verbose and it % 500 == 0:
            print(f"    KKT it{it}: obj={best_obj:.8f}, lr={lr:.8f}")
    
    return best_a
