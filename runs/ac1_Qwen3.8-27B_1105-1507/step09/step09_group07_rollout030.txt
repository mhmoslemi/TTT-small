Optimized step function search using enhanced LP alternating with golden-section
line search, dual-guided local refinement, adaptive n-sweep, and multi-scale
perturbation restarts.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, warm_start=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    m = 2 * n - 1
    
    # Try solvers in order
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            if warm_start is not None:
                try:
                    prob.solve(solver=solver, warm_start=True)
                except:
                    prob.solve(solver=solver)
            else:
                prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    # Fallback to scipy linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def golden_section_max(obj_func, a, lo, hi, tol=1e-6, max_iter=50):
    """Find minimum of obj_func in [lo, hi] using golden section search."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ~0.618
    
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj_func(c)
    fd = obj_func(d)
    
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj_func(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_func(d)
    
    t_opt = (lo + hi) / 2
    return t_opt, obj_func(t_opt)

def smooth_optimize_v2(a, time_budget, verbose=False):
    """
    Smooth optimization using log-sum-exp surrogate with mirror descent.
    Uses FFT-accelerated gradient computation.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 0.1 / np.sqrt(n)
    beta = 10.0
    it = 0
    stall = 0
    patience = 50
    
    while time.monotonic() - start < time_budget:
        # Compute convolution
        conv = compute_conv(a)
        
        # Softmax weights
        shifted = beta * (conv - conv.max())
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            break
        w = w / w_sum
        
        # Gradient: grad[i] = 2 * sum_j w[i+j] * a[j]
        # Vectorized using index matrix
        idx = np.arange(n)[:, None] + np.arange(n)[None, :]
        # clip to valid range
        idx = np.clip(idx, 0, 2*n-2)
        W = w[idx]  # shape (n, n)
        grad = 2.0 * (W @ a)
        
        # Mirror descent update
        log_a = np.log(a)
        log_a_new = log_a - lr * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30:
            lr *= 0.5
            it += 1
            continue
        a_new = a_new / a_new_sum
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = min(0.5, lr * 1.02)
            beta = beta * 1.05
            stall = 0
        else:
            lr = max(1e-6, lr * 0.9)
            stall += 1
            if stall > patience:
                # Increase beta more aggressively
                beta = beta * 2
                lr = 0.1 / np.sqrt(n)
                stall = 0
        
        it += 1
        if verbose and it % 100 == 0:
            print(f"    Smooth-v2 it{it}: obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.6f}")
    
    return best_a


def simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def projected_subgradient(a, time_budget, verbose=False):
    """
    Projected subgradient method for minimizing max_k conv(a,a)_k
    subject to a >= 0, sum(a) = 1.
    Uses exact subgradient (the a[k-i] for active lags k).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 1.0 / np.sqrt(n)
    it = 0
    stall = 0
    patience = 100
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = conv.max()
        
        # Subgradient: for the max lag k*, grad[i] = 2*a[k*-i] (or a[k*-i] if k*=i)
        # Use all lags within epsilon of max for a "fat" subgradient
        eps = 1e-6 * max_val
        active = np.where(conv >= max_val - eps)[0]
        
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j]
        
        # Normalize subgradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad = grad / grad_norm
        
        # Update
        a_new = a - lr * grad
        a_new = simplex_project(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = min(0.5, lr * 1.01)
            stall = 0
        else:
            lr = max(1e-8, lr * 0.95)
            stall += 1
            if stall > patience:
                # Reset with perturbation
                rng_local = np.random.default_rng(it)
                a = best_a * np.exp(rng_local.normal(0, 0.05, n))
                a = a / a.sum()
                lr = 0.5 / np.sqrt(n)
                stall = 0
        
        it += 1
        if verbose and it % 200 == 0:
            print(f"    PGD it{it}: obj={best_obj:.8f}, lr={lr:.6f}")
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    """

        max_val = conv.max()
        
        # Find top contributing lags
        top_lags = np.argsort(conv)[::-1][:5]
        
        # Find "quiet" positions (small a[i] that don't contribute to peaks)
        peak_contrib = np.zeros(n)
        for k in top_lags:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    peak_contrib[i] += a[j]
        
        # Sort by contribution (least contributing first)
        quiet_order = np.argsort(peak_contrib)
        quiet_positions = quiet_order[:10]
        
        improved = False
        for k_star in top_lags[:3]:
            if time.monotonic() - start > time_budget:
                break
            k_star = int(k_star)
            
            # Find largest contributors to this lag
            contribs = []
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    contribs.append((a[i] * a[j], i, j))
            contribs.sort(reverse=True)

                if c_val < 1e-12:
                    continue
                src = i if a[i] >= a[j] else j
                for target in quiet_positions[:5]:
                    if target == src:
                        continue
                    if a[target] > a[src] * 0.8:  # target should be smaller
                        continue

                        delta = frac * a[src]
                        if delta <= 0 or delta >= a[src] * 0.5:
                            continue
                        a[src] -= delta
                        a[target] += delta
                        new_conv = compute_conv(a)
                        new_obj = 2.0 * n * float(np.max(new_conv))
                        if new_obj < best_obj - 1e-14:
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[src] += delta
                            a[target] -= delta
                            break
                if improved:
                    break

                a = best_a.copy()  # sync
                conv = compute_conv(a)
        
        if not improved and round_num > 20:

                a = best_a * np.exp(rng.normal(0, scale, n))

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    m = 2 * n - 1

    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def golden_section_min(obj_func, lo, hi, tol=0.02, max_iter=40):
    """Golden section search for minimum of obj_func in [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj_func(c)
    fd = obj_func(d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj_func(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_func(d)
    t_opt = (lo + hi) / 2.0
    return t_opt, obj_func(t_opt)


def lp_alternate_v2(a_in, time_budget, verbose=False):
    """Enhanced LP alternating with golden-section line search and adaptive perturbation."""
    n = len(a_in)
    a = np.maximum(a_in, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    patience = 20
    last_t = 0.3
    iteration = 0
    rng = np.random.default_rng(54321 + n)

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        sim = 1.0 - np.linalg.norm(a - g_lp) / (np.linalg.norm(a) + np.linalg.norm(g_lp) + 1e-15)
        if sim > 0.9999:
            pert = rng.normal(0, 0.04 + 0.01 * min(stall_count // 5, 10), n)
            a = best_a * np.exp(pert)
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall_count = 0
            continue

        # Golden section search for step size
        def obj_t(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                return np.inf
            a_try = a_try / s_try
            return compute_obj(a_try)

        # Narrow search range based on last good t
        lo = max(0.01, last_t - 0.15)
        hi = min(0.99, last_t + 0.15)
        if hi - lo < 0.05:
            lo, hi = 0.01, 0.99
        t_opt, obj_opt = golden_section_min(obj_t, lo, hi, tol=0.015, max_iter=35)

        iteration += 1
        if obj_opt < best_obj - 1e-13:
            a_try = (1.0 - t_opt) * a + t_opt * g_lp
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            best_obj = obj_opt
            best_a = a_try.copy()
            a = a_try.copy()
            last_t = t_opt
            stall_count = 0
        else:
            # Try with last known good t
            a_try = (1.0 - last_t) * a + last_t * g_lp
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > patience:
                    scale = 0.03 + 0.015 * min(stall_count // patience, 8)
                    pert = rng.normal(0, scale, n)
                    a = best_a * np.exp(pert)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    stall_count = 0
                    last_t = 0.4

        if verbose and iteration % 15 == 0:
            el = time.monotonic() - start
            print(f"    LP2 it{iteration}: obj={best_obj:.8f} t={last_t:.3f} stall={stall_count} t+={el:.0f}s")

    return best_a


def smooth_mirror(a_in, time_budget, verbose=False):
    """Smooth optimization via log-sum-exp surrogate + mirror descent (multiplicative updates)."""
    n = len(a_in)
    a = np.maximum(a_in, 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.3 / np.sqrt(n)
    beta = 5.0
    it = 0
    stall = 0
    patience = 40

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        shifted = beta * (conv - conv.max())
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            beta *= 3
            it += 1
            continue
        w = w / ws

        # grad[i] = 2 * sum_j w[i+j] * a[j], vectorized
        idx = np.arange(n)[:, None] + np.arange(n)[None, :]
        idx = np.minimum(idx, 2 * n - 2)
        Wmat = w[idx]
        grad = 2.0 * (Wmat @ a)

        # Mirror descent (multiplicative)
        log_a = np.log(a)
        log_a_new = log_a - lr * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        sn = a_new.sum()
        if sn < 1e-30:
            lr *= 0.5
            it += 1
            continue
        a_new = a_new / sn

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = min(1.0, lr * 1.03)
            beta *= 1.04
            stall = 0
        else:
            lr = max(1e-7, lr * 0.92)
            stall += 1
            if stall > patience:
                beta *= 2.0
                lr = 0.3 / np.sqrt(n)
                stall = 0

        it += 1
        if verbose and it % 80 == 0:
            print(f"    Smooth it{it}: obj={best_obj:.8f} beta={beta:.1f} lr={lr:.5f}")

    return best_a


def multi_direction_lp(a_in, time_budget, verbose=False):
    """Solve LP from multiple perturbed viewpoints, pick best direction."""
    n = len(a_in)
    a = np.maximum(a_in, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(31415 + n)
    it = 0

    while time.monotonic() - start < time_budget:
        directions = []
        for ps in [0.0, 0.03, 0.06, 0.10, 0.15]:
            if ps == 0.0:
                a_p = a.copy()
            else:
                a_p = a * np.exp(rng.normal(0, ps, n))
                a_p = np.maximum(a_p, 0)
                sp = a_p.sum()
                if sp < 1e-10:
                    continue
                a_p = a_p / sp
            g, _ = solve_lp(a_p)
            if g is not None:
                directions.append(g)

        if not directions:
            break

        for g_dir in directions:
            def obj_t2(t):
                a_try = (1.0 - t) * a + t * g_dir
                a_try = np.maximum(a_try, 0)
                st = a_try.sum()
                if st < 1e-10:
                    return np.inf
                a_try = a_try / st
                return compute_obj(a_try)

            t_opt, obj_opt = golden_section_min(obj_t2, 0.01, 0.99, tol=0.04, max_iter=25)
            if obj_opt < best_obj - 1e-13:
                a_try = (1.0 - t_opt) * a + t_opt * g_dir
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                best_obj = obj_opt
                best_a = a_try.copy()
                a = a_try.copy()

        # Escape check
        g_main, _ = solve_lp(a)
        if g_main is not None:
            sim = 1.0 - np.linalg.norm(a - g_main) / (np.linalg.norm(a) + np.linalg.norm(g_main) + 1e-15)
            if sim > 0.9998:
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / a.sum()

        it += 1
        if verbose and it % 5 == 0:
            print(f"    MultiLP it{it}: obj={best_obj:.8f}")

    return best_a


def targeted_refine(a_in, time_budget, verbose=False):
    """Targeted mass transfer: identify peak lags, move mass from contributors to quiet spots."""
    n = len(a_in)
    a = np.maximum(a_in, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(27182 + n)

    for rnd in range(300):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_val = conv.max()
        top_lags = np.argsort(conv)[::-1][:4]

        # Compute peak contribution per position
        peak_contrib = np.zeros(n)
        for k in top_lags:
            ki = int(k)
            for i in range(n):
                j = ki - i
                if 0 <= j < n:
                    peak_contrib[i] += a[j]

        quiet_order = np.argsort(peak_contrib)
        quiet_pos = quiet_order[:12]

        improved = False
        for k_star in top_lags[:3]:
            if time.monotonic() - start > time_budget:
                break
            ki = int(k_star)
            contribs = []
            for i in range(n):
                j = ki - i
                if 0 <= j < n:
                    contribs.append((a[i] * a[j], i, j))
            contribs.sort(key=lambda x: -x[0])

            for cval, i, j in contribs[:6]:
                if cval < 1e-14:
                    continue
                src = i if a[i] >= a[j] else j
                for tgt in quiet_pos[:6]:
                    if tgt == src:
                        continue
                    if a[tgt] > a[src] * 0.7:
                        continue
                    for frac in [0.003, 0.007, 0.015, 0.03]:
                        delta = frac * a[src]
                        if delta <= 0 or delta >= a[src] * 0.4:
                            continue
                        a[src] -= delta
                        a[tgt] += delta
                        nc = compute_conv(a)
                        no = 2.0 * n * float(np.max(nc))
                        if no < best_obj - 1e-14:
                            best_obj = no
                            best_a = a.copy()
                            improved = True
                            a = best_a.copy()
                            break
                        else:
                            a[src] += delta
                            a[tgt] -= delta
                            break
                if improved:
                    conv = compute_conv(a)
                    break
            if improved:
                break

        if not improved and rnd > 30:
            if rnd % 15 == 0:
                scale = 0.01 + 0.005 * (rnd // 15)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()

        if verbose and rnd % 30 == 0:
            print(f"    Targeted r{rnd}: obj={best_obj:.8f}")

    return best_a


def resample_seq(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    m = mode % 10
    if m == 0:
        a = np.ones(n) / n
    elif m == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif m == 2:
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif m == 3:
        a = 1.0 + 0.3 * np.sin(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif m == 4:
        a = np.exp(-3 * (x - 0.3) ** 2) + np.exp(-3 * (x - 0.7) ** 2)
        a = a / a.sum()
    elif m == 5:
        a = x**2 + (1 - x)**2 + 0.05
        a = a / a.sum()
    elif m == 6:
        f1, f2 = rng.integers(2, 9, 2)
        p1, p2 = rng.uniform(0, 2 * np.pi, 2)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * f1 * x + p1) + 0.2 * np.sin(2 * np.pi * f2 * x + p2)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif m == 7:
        a = rng.dirichlet(np.ones(n) * 0.5)
    elif m == 8:
        a = np.abs(np.sin(np.pi * x)) + 0.1
        a = a / a.sum()
    else:
        a = (1.0 + 0.4 * np.cos(6 * np.pi * x) + 0.2 * np.cos(10 * np.pi * x))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    return a
