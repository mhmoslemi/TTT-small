import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.maximum(v, 0)
    s = float(np.sum(v))
    if s < 1e-300:
        return np.ones(n) / n
    u = np.sort(v / s)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v / s - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # (conv(a,g))[k] = sum_j a[j] * g[k-j]
        # For fixed j, g[k-j] appears at row k where k = j, j+1, ..., j+n-1
        # So A[k, j] = a[j] for k in [j, j+n-1]
        row_start = j
        row_end = min(m, j + n)
        A[row_start:row_end, j] = a[j]
    return A

    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient_fft(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of Q_k(a) using FFT cross-correlation.
    """
    n = len(a)
    conv = compute_conv(a)

    max_conv = float(np.max(conv))
    if tau < 1e-10:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1

    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]

    return g, conv


def adaptive_smoothed_descent(a, time_budget=100.0, tau_init=10.0, tau_end=500.0, verbose=False):
    """
    Nesterov accelerated gradient descent with adaptive tau and learning rate.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 3.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    it = 0
    best_it = 0
    recent_gnorms = []

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed < 1e-10:
            elapsed = 1e-10

        # Adaptive tau: cosine schedule
        frac = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * 0.5 * (1 - np.cos(np.pi * frac))

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute gradient
        g, _ = smoothed_gradient_fft(y, tau)
        g_norm = float(np.linalg.norm(g))
        recent_gnorms.append(g_norm)
        if len(recent_gnorms) > 200:
            recent_gnorms = recent_gnorms[-200:]

        # Adaptive learning rate
        if recent_gnorms and np.mean(recent_gnorms) > 1e-15:
            adaptive_lr = 1.0 / (float(np.mean(recent_gnorms)) + 1e-10)
            lr = 0.6 * lr + 0.4 * min(adaptive_lr, 3.0 * base_lr)

        # Update
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        # Slow decay
        lr = max(1e-10, lr * 0.9998)

        # Reset if stuck
        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it

    return best_a


def fourier_perturbation(a, rng, strength=0.03):
    """Add Fourier modes as structured perturbation in log domain."""
    n = len(a)
    a_safe = np.maximum(a, 1e-15)
    pert = np.zeros(n)
    n_modes = 5
    for k in range(1, n_modes + 1):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a_safe * np.exp(pert)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def gauss_perturbation(a, rng, strength=0.05):
    """Add random Gaussian bumps."""
    n = len(a)
    pert = np.zeros(n)
    n_bumps = rng.integers(2, 6)
    for _ in range(n_bumps):
        center = rng.uniform(0, n)
        width = rng.uniform(n/20, n/4)
        amp = strength * rng.uniform(0.5, 1.5)
        x = np.arange(n)
        pert += amp * np.exp(-((x - center) ** 2) / (2 * width ** 2))
    a_safe = np.maximum(a, 1e-15)
    a_new = a_safe * np.exp(pert)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lp_with_sa(a, time_budget=200.0, verbose=False, seed=42, t_init=0.8, t_floor=0.05):
    """
    LP-based optimization with simulated annealing acceptance.
    At each step, solve LP and accept/reject based on temperature.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    current_a = a.copy()
    current_obj = best_obj
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    stall_count = 0
    last_improve_it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Temperature cooling
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        temp = t_init * (t_floor / t_init) ** frac

        # Solve LP
        sol_start = time.monotonic()
        g_lp, M_lp = solve_lp_minpeak(current_a)
        lp_time = time.monotonic() - sol_start

        if g_lp is None:
            break

        # Try multiple blending factors
        candidates_t = [0.3, 0.5, 0.7, 0.9, 1.0]
        best_local_a = current_a.copy()
        best_local_obj = current_obj

        for t in candidates_t:
            a_try = (1.0 - t) * current_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()

        it += 1

        # Acceptance criterion (SA)
        delta = best_local_obj - current_obj
        if delta < -1e-14:
            # Improvement: always accept
            accept = True
        else:
            # Worse: accept with probability exp(-delta/temp)
            if temp > 1e-10:
                accept_prob = np.exp(-delta / temp)
                accept = rng.random() < accept_prob
            else:
                accept = False

        if accept:
            current_a = best_local_a.copy()
            current_obj = best_local_obj
            stall_count = 0
            if current_obj < best_obj - 1e-14:
                best_obj = current_obj
                best_a = current_a.copy()
                last_improve_it = it
                if verbose and it % 10 == 0:
                    print(f"    SA-LP it{it}: obj={best_obj:.8f}, t={temp:.4f}")
        else:
            stall_count += 1
            # Diversify when stuck
            if stall_count > 3:
                current_a = gauss_perturbation(best_a, rng, strength=0.03 + 0.01 * min(stall_count, 5))
                current_obj = compute_obj(current_a)
                stall_count = 0
            if stall_count > 10:
                current_a = fourier_perturbation(best_a, rng, strength=0.05)
                current_obj = compute_obj(current_a)
                stall_count = 0

        # Global restart if severely stuck
        if it - last_improve_it > 50:
            current_a = best_a.copy()
            current_obj = best_obj
            current_a = gauss_perturbation(current_a, rng, strength=0.1)
            stall_count = 0
            last_improve_it = it

    return best_a


def multi_peak_init(n, n_peaks, rng, peak_width_frac=0.05):
    """Initialize with evenly spaced Gaussian peaks."""
    a = np.zeros(n)
    if n_peaks <= 1:
        a[:] = 1.0 / n
        return a
    centers = np.linspace(0, n - 1, n_peaks)
    width = max(1.0, n * peak_width_frac / (2 * n_peaks))
    for c in centers:
        a += np.exp(-((np.arange(n) - c) ** 2) / (2 * width ** 2))
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / max(a.sum(), 1e-15)
    elif mode == 2:
        # Multi-peak
        n_peaks = rng.integers(3, max(4, n // 20))
        a = multi_peak_init(n, n_peaks, rng, peak_width_frac=rng.uniform(0.02, 0.15))
    elif mode == 3:
        # Oscillatory
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 5:
        # Two clusters
        c1 = rng.integers(0, n // 3)
        c2 = rng.integers(2 * n // 3, n)
        w = max(1, n // 16)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 6:
        # Block structure
        n_blocks = rng.integers(2, 8)
        block_size = n // n_blocks
        for b in range(n_blocks):
            start = b * block_size
            end = min(n, start + block_size)
            a_block = rng.uniform(0.1, 1.0, end - start)
            a[start:end] = a_block
        a = a / a.sum()
    elif mode == 7:
        # Triangular
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 4 * np.abs(x - 0.5), 0.01)
        a = a / a.sum()
    elif mode == 8:
        # Exponential decay from center
        a = np.exp(-np.abs(np.arange(n) - n / 2) / (n / 5))
        a = a / a.sum()
    elif mode == 9:
        # Random multi-peak with varying widths
        n_peaks = rng.integers(2, max(3, n // 15))
        a = np.zeros(n)
        for _ in range(n_peaks):
            c = rng.uniform(0, n)
            w = rng.uniform(n/30, n/5)
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def spectral_init(n, rng):
    """Initialize in frequency domain."""
    size = 1
    while size < 2 * n:
        size <<= 1
    freqs = np.fft.rfftfreq(size)
    mag = np.exp(-freqs * n * 0.3)
    phases = rng.uniform(0, 2 * np.pi, size=size // 2 + 1)
    F = mag * np.exp(1j * phases)
    F[0] = abs(F[0])
    a = np.fft.irfft(F, size)[:n]
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP-based SA, smoothed GD,
    multi-length exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        best_seq = np.ones(n_current) / n_current
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")

    # === Phase 1: Quick LP-based SA on parent ===
    print(f"\n=== Phase 1: LP-SA on parent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.2, 250.0)
    if p1_budget > 30:
        a_lp = lp_with_sa(best_seq, time_budget=p1_budget, verbose=True, seed=100)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1 LP-SA: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 LP-SA: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: Smoothed GD refinement ===
    print(f"\n=== Phase 2: Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.1, 120.0)
    if p2_budget > 20:
        a_gd = adaptive_smoothed_descent(best_seq, time_budget=p2_budget,
                                         tau_init=20.0, tau_end=800.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase2 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 GD: obj={obj_gd:.8f}")

    # === Phase 3: Intensive LP-SA (main workhorse) ===
    print(f"\n=== Phase 3: Intensive LP-SA ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.3, 400.0)
    if p3_budget > 30:
        # Multiple rounds with different seeds and perturbation
        for rnd in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(p3_budget / (5 - rnd), remaining * 0.5)
            if alloc < 15:
                break

            # Perturb before each round (except first)
            if rnd > 0:
                strength = 0.02 + 0.01 * rnd
                working = gauss_perturbation(best_seq, rng, strength=strength)
                working = working / working.sum()
            else:
                working = best_seq.copy()

            a_lp = lp_with_sa(working, time_budget=alloc, verbose=(rnd == 0),
                              seed=500 + rnd * 7)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP-SA round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP-SA round {rnd}: obj={obj_lp:.8f}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(300, n_best - 200), max(300, n_best - 100),
        n_best + 100, n_best + 200, n_best + 400, n_best + 600,
        500, 700, 900, 1100, 1300, 1500
    ]))
    # Filter to reasonable range and diversity
    n_options = [n for n in n_options if 300 <= n <= 2000]
    # Keep at most 6, spread across the range
    if len(n_options) > 6:
        indices = np.linspace(0, len(n_options) - 1, 6, dtype=int)
        n_options = [n_options[i] for i in indices]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        if n_try == n_best:
            continue

        # Resample best sequence to new length
        a_try = resample_sequence(best_seq, n_try)

        # Quick optimization: GD then LP
        alloc1 = min(remaining * 0.2, 50.0)
        if alloc1 > 15:
            a_try = adaptive_smoothed_descent(a_try, time_budget=alloc1,
                                              tau_init=20.0, tau_end=400.0)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 60.0)
        if alloc2 > 15:
            a_try = lp_with_sa(a_try, time_budget=alloc2, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        # Different initialization strategies
        if r == 0:
            a_r = spectral_init(n_cur, rng)
        elif r == 1:
            a_r = multi_peak_init(n_cur, rng.integers(5, max(6, n_cur // 15)), rng,
                                  peak_width_frac=rng.uniform(0.03, 0.1))
        elif r == 2:
            a_r = multi_peak_init(n_cur, rng.integers(3, max(4, n_cur // 25)), rng,
                                  peak_width_frac=rng.uniform(0.05, 0.2))
        elif r == 3:
            a_r = diverse_init(n_cur, rng, mode=rng.integers(0, 10))
        else:
            mode = r % 10
            a_r = diverse_init(n_cur, rng, mode=mode)
            # Blend with best for warm start
            blend = 0.2 + 0.1 * (r - 3)
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

        # Optimize
        alloc_gd = min(remaining * 0.2, 50.0)
        alloc_lp = min(remaining * 0.15, 50.0)

        if alloc_gd > 15:
            a_r = adaptive_smoothed_descent(a_r, time_budget=alloc_gd,
                                            tau_init=30.0, tau_end=500.0)

        remaining = deadline - time.monotonic()
        if alloc_lp > 15 and remaining > alloc_lp + 10:
            a_r = lp_with_sa(a_r, time_budget=alloc_lp, seed=2000 + r * 13)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={r}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={r}): obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Multiple LP passes with decreasing perturbation
    for lp_rnd in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * 0.2, 60.0)
        if alloc < 10:
            break

        if lp_rnd > 0:
            small_pert = gauss_perturbation(best_seq, rng, strength=0.01 * lp_rnd)
            small_pert = small_pert / small_pert.sum()
        else:
            small_pert = best_seq.copy()

        a_lp = lp_with_sa(small_pert, time_budget=alloc, verbose=False,
                          seed=5000 + lp_rnd * 37)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP {lp_rnd}: obj={best_score:.8f} **")
        else:
            if lp_rnd < 5:
                print(f"  Final LP {lp_rnd}: obj={obj_lp:.8f}")
                break

    # Final GD
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 60.0)
        a_final = adaptive_smoothed_descent(best_seq, time_budget=alloc,
                                            tau_init=100.0, tau_end=1500.0)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # One more LP if time
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 50.0)
        a_final = lp_with_sa(best_seq, time_budget=alloc, verbose=False, seed=9999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Ultra-final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
