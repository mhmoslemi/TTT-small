```python
def optimize_ift(a, time_budget, verbose=False):
    """
    IFT-based optimization with peak elimination.
    Alternates between:
    1. Time domain: project to simplex (a >= 0, sum = 1)
    2. Frequency domain: push DFT magnitude toward flat
    3. Peak elimination: explicitly reduce the dominant lag
    """
    n = len(a)
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # IFT parameters
    target_mag = None  # will be set
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Step 1: Compute DFT
        N = 1
        while N < 2*n:
            N <<= 1
        A = np.fft.rfft(a, N)
        
        # Step 2: Push magnitude toward flat
        if target_mag is None:
            target_mag = np.median(np.abs(A[A != 0]))
        
        # Blend toward target
        alpha = 0.3  # blending factor
        A_new_mag = (1-alpha) * np.abs(A) + alpha * target_mag
        A_new = A_new_mag * np.exp(1j * np.angle(A))
        
        # Step 3: Go back to time domain
        a_new = np.fft.irfft(A_new, N)[:n].real
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
        
        # Step 4: Peak elimination
        conv = compute_conv(a_new)
        max_k = np.argmax(conv)
        
        # Compute gradient to reduce the peak
        # For peak at k*, reduce a_i proportional to a_{k*-i}
        # This is a simple "anti-peak" step
        decay = 0.01
        a_anti = a_new.copy()
        i_start = max(0, max_k - (n-1))
        i_end = min(n-1, max_k)
        for i in range(i_start, i_end+1):
            j = max_k - i
            a_anti[i] -= decay * a_new[i] * a_new[j]
        a_anti = np.maximum(a_anti, 0)
        s = a_anti.sum()
        if s > 1e-10:
            a_anti = a_anti / s
        else:
            a_anti = a_new.copy()
        
        # Choose between a_new and a_anti
        obj_new = compute_obj(a_new)
        obj_anti = compute_obj(a_anti)
        
        if obj_anti < obj_new:
            a_new = a_anti
            obj_new = obj_anti
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        it += 1
    
    return best_a
```

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of:
- IFT (Iterative Fourier Transform) for spectral shaping initialization
- Population-based search with diverse candidates
- Smoothed gradient descent with FFT-based gradient computation
- Lag-specific targeted modifications
- LP polishing for final refinement
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_gradient_fft(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a).
    Uses FFT for O(n log n) computation.
    Returns (gradient, convolution values, max index).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    # grad_i = 2 * sum_j p_{i+j} * a_j
    # = 2 * correlate(p, a, 'full')[i]
    # Compute via FFT
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    
    max_idx = int(np.argmax(conv))
    return g, conv, max_idx


def ift_initialize(n, iterations=200, alpha=0.5, seed=0):
    """
    Initialize a sequence using IFT (Gerchberg-Saxton) to get flat spectrum.
    """
    rng = np.random.default_rng(seed)
    N = 1
    while N < 2 * n:
        N <<= 1
    
    # Start with random non-negative
    a = rng.uniform(0.1, 1.0, n)
    a = a / a.sum()
    
    for it in range(iterations):
        # Forward transform
        A = fft(a, N)
        
        # Shape the magnitude
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: flat magnitude (but keep DC component)
        # Use a blend between current and flat
        target = np.median(mag[1:])  # exclude DC
        new_mag = (1 - alpha) * mag + alpha * target
        new_mag[0] = mag[0]  # preserve DC (sum)
        
        # Inverse transform with new magnitude
        A_new = new_mag * np.exp(1j * phase)
        a = ifft(A_new, N).real[:n]
        
        # Project onto non-negative
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    
    return a


def ift_with_phase_diversity(n, iterations=100, alpha=0.3, seed=0, n_modes=3):
    """
    Generate diverse IFT initializations with different phase patterns.
    """
    candidates = []
    for mode in range(n_modes):

        elif mode == 1:

        else:
            x = np.linspace(0, 1, n)
            a = np.exp(-((x - 0.5)**2) / (2 * 0.1**2))
        
        a = a / a.sum()
        
        for it in range(iterations):
            A = fft(a, N)
            mag = np.abs(A)
            phase = np.angle(A)
            target = np.median(mag[1:])
            new_mag = (1 - alpha) * mag + alpha * target
            new_mag[0] = mag[0]
            A_new = new_mag * np.exp(1j * phase)
            a = ifft(A_new, N).real[:n]
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-10:

            else:

def lag_surgery(a, conv, max_k, strength=0.05):
    """
    Apply targeted modification to reduce the peak at lag max_k.
    Reduces a_i proportionally to its contribution to conv[max_k].
    """
    n = len(a)
    a_new = a.copy()
    
    i_start = max(0, max_k - (n - 1))
    i_end = min(n - 1, max_k)
    
    for i in range(i_start, i_end + 1):
        j = max_k - i
        if 0 <= j < n:
            a_new[i] -= strength * a[i] * a[j]
    
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    
    return a_new


def multi_lag_surgery(a, conv, top_k=5, strength=0.02):
    """
    Apply surgery to reduce the top k lags simultaneously.
    """
    n = len(a)
    a_new = a.copy()
    
    top_indices = np.argsort(conv)[::-1][:top_k]
    
    for k in top_indices:
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if 0 <= j < n:
                a_new[i] -= strength * a_new[i] * a[j]
    
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    
    return a_new


def build_conv_matrix(a):
    """Build matrix A_mat (2n-1 x n) such that A_mat @ g = convolve(a, g)."""
    n = len(a)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A_mat[j:j + r, j] = a[:r]
    return A_mat


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A_mat = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def gd_with_spectral(a, time_budget=60.0, tau_init=10.0, tau_end=300.0,
                     spectral_freq=50, spectral_alpha=0.2, verbose=False, seed=0):
    """
    Gradient descent with periodic spectral shaping steps.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    N = 1
    while N < 2 * n:
        N <<= 1

    lr = 3.0 / np.sqrt(n)
    beta = 0.8
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init + (tau_end - tau_init) * progress

        g, conv, max_k = smoothed_gradient_fft(a, tau)

        # Nesterov step
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y > 1e-10:
            y = y / s_y
        else:
            y = a.copy()

        g_y, _, _ = smoothed_gradient_fft(y, tau)
        v = beta * v - lr * g_y
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        # Periodic spectral shaping
        if it % spectral_freq == spectral_freq - 1:
            A = fft(a_new, N)
            mag = np.abs(A)
            phase = np.angle(A)
            target = np.median(mag[1:])
            new_mag = (1 - spectral_alpha) * mag + spectral_alpha * target
            new_mag[0] = mag[0]
            A_new = new_mag * np.exp(1j * phase)
            a_spec = ifft(A_new, N).real[:n]
            a_spec = np.maximum(a_spec, 0)
            s_spec = a_spec.sum()
            if s_spec > 1e-10:
                a_spec = a_spec / s_spec
                obj_spec = compute_obj(a_spec)
                if obj_spec < compute_obj(a_new):
                    a_new = a_spec

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        lr = max(1e-8, lr * 0.99995)
        it += 1

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n)
            best_it = it

    return best_a


def lp_polish(a, time_budget=100.0, verbose=False, seed=0):
    """LP polishing with blending."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    consecutive_fails = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        
        best_t_obj = best_obj
        best_t_val = 0.3
        best_a_local = best_a.copy()

        for t in candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        iter_count += 1

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a.copy()
            consecutive_fails = 0
        else:
            consecutive_fails += 1
            # Perturb to escape
            conv = compute_conv(a)
            a = multi_lag_surgery(a, conv, top_k=5, strength=0.03)
            a = 0.7 * a + 0.3 * best_a
            a = np.maximum(a, 0)
            a = a / a.sum()
            if consecutive_fails > 30:
                break

    return best_a


def population_search(n, time_budget, n_pop=5, verbose=False, seed=0):
    """
    Population-based search: maintain multiple candidates, evolve them.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    
    # Initialize population
    population = []
    for i in range(n_pop):
        if i == 0:
            a = np.ones(n) / n
        elif i == 1:
            a = ift_initialize(n, iterations=100, alpha=0.3, seed=seed + i)
        else:
            a = rng.uniform(0.01, 1.0, n)
            a = a / a.sum()
        population.append(a)
    
    # Evaluate initial population
    objs = [compute_obj(a) for a in population]
    
    # Evolution loop
    gen = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 30:
            break
        
        gen_time = min(remaining / max(1, n_pop - 1), 30.0)
        
        # Remove worst, add new
        worst_idx = int(np.argmax(objs))
        best_idx = int(np.argmin(objs))
        
        # Create new candidate: blend of best with random perturbation
        a_base = population[best_idx].copy()
        a_rand = rng.uniform(0.01, 1.0, n)
        a_rand = a_rand / a_rand.sum()
        blend = 0.3 + 0.3 * rng.random()
        a_new = (1 - blend) * a_base + blend * a_rand
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        # Optimize the new candidate
        a_opt = gd_with_spectral(a_new, time_budget=gen_time,
                                  tau_init=10.0, tau_end=200.0,
                                  spectral_freq=30, spectral_alpha=0.15,
                                  verbose=False, seed=seed + gen * 10 + worst_idx)
        obj_opt = compute_obj(a_opt)
        
        population[worst_idx] = a_opt
        objs[worst_idx] = obj_opt
        
        if verbose and gen % 5 == 0:
            print(f"    Pop gen {gen}: best obj = {min(objs):.8f}")
        
        gen += 1
    
    best_idx = int(np.argmin(objs))
    return population[best_idx], objs[best_idx]


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
