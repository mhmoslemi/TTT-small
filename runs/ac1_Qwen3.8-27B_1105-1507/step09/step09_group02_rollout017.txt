"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Intensive LP-based alternating minimization with adaptive line search,
peak-flattening perturbations, multi-length exploration, and final smoothing.
Key improvements: better line search, peak-flatness objective, longer LP runs

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_top_peaks(a, K=10):
    """Return the indices and values of the top K peaks in conv(a,a)."""
    conv = compute_conv(a)
    idx = np.argsort(conv)[::-1][:K]
    return idx, conv[idx]


def build_conv_matrix(a):
    """Build matrix (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A


def solve_lp(a, time_limit=5.0):
    """
    Solve LP: min M s.t. conv(a/g_norm, g)[k] <= M, sum(g)=1, g>=0.
    Returns g_optimal.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def solve_lp_weighted(a, weights, time_limit=5.0):

    slacks = cp.Variable(m, nonneg=True)

def smoothed_mg(a, time_budget, tau=100.0, verbose=False):
    """
    Smoothed minimization using log-sum-exp approximation of max.
    Uses FFT-based gradient computation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 1.0 / np.sqrt(n)
    beta = 0.9
    v = np.zeros(n)
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break
        
        # Compute smoothed gradient
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient of sum_k p[k] * conv[k] w.r.t. a
        # d(conv[k])/da[i] = a[k-i] + a[i-k] (symmetric)
        # So grad[i] = 2 * sum_k p[k] * a[k-i] (for valid k)
        # This is 2 * conv(p, a reversed) ... let me compute via FFT
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]
        
        # Nesterov step
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y > 1e-10:
            y = y / s_y
        
        # Recompute gradient at y
        conv_y = compute_conv(y)
        max_y = float(np.max(conv_y))
        exp_y = np.exp(tau * (conv_y - max_y))
        p_y = exp_y / float(np.sum(exp_y))
        P_y = fft(p_y, N)
        A_y = fft(y, N)
        g_y = 2.0 * ifft(P_y * np.conj(A_y)).real[:n]
        
        v = beta * v - lr * g_y
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        it += 1
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 1000:
                lr *= 0.5
                v *= 0.5
                stall = 0
                if lr < 1e-10:
                    break
        
        lr = max(1e-8, lr * 0.9999)
    
    return best_a


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def build_conv_matrix(a):
    """Build matrix (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)

    for j in range(n):
        A[j:j + n, j] = a
    return A

                mode = stall % 3
                if mode == 0:
                    a = best_a * np.exp(rng.normal(0, 0.08, n))
                elif mode == 1:
                    # Bipolar perturbation
                    half = n // 2
                    pert = rng.normal(0, 0.05, n)
                    pert[:half] *= 1.5
                    pert[half:] *= 0.5
                    a = best_a * np.exp(pert)
                else:
                    # Smooth low-freq perturbation
                    freqs = rng.integers(1, 5, size=2)
                    phases = rng.uniform(0, 2 * np.pi, size=2)
                    x = np.linspace(0, 1, n)
                    pert = np.zeros(n)
                    for f, p in zip(freqs, phases):
                        pert += 0.05 * np.sin(2 * np.pi * f * x + p)
                    a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0

            if stall > 80:
                break

    return best_a


def subgradient_polish(a, time_budget, K=15, verbose=False):
    """
    Subgradient method targeting top-K peaks with adaptive weighting and momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.05
    momentum = np.zeros(n)
    beta = 0.8
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        top_vals = conv[top_idx]
        max_val = top_vals[0]

        # Exponential weighting: heavier on higher peaks
        temp = max(0.01, (max_val - top_vals[-1]) * 3.0)
        weights = np.exp((top_vals - max_val) / temp)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        momentum = beta * momentum + g / g_norm
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        it += 1

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                lr *= 0.7
                momentum *= 0.5
                stall = 0
                if lr < 1e-6:
                    break

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initial sequences."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 4) * x + rng.uniform(0, 2 * np.pi))
    elif mode == 2:
        a = 1.0 + 0.2 * np.cos(2 * np.pi * rng.uniform(2, 6) * x)
        a = a + 0.1 * np.sin(2 * np.pi * rng.uniform(5, 12) * x)
    elif mode == 3:
        a = np.exp(-((x - 0.3) ** 2) / 0.05) + np.exp(-((x - 0.7) ** 2) / 0.05)
    elif mode == 4:
        # Arcsine-like distribution
        a = 1.0 / np.sqrt(np.maximum(x * (1 - x), 0.01))
    else:
        a = rng.uniform(0.5, 1.5, n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function.
    Phase 1: Intensive LP alternating (45%)
    Phase 2: Subgradient polish (10%)
    Phase 3: Multi-length exploration (20%)
    Phase 4: Final LP + subgradient polish (25%)
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_parent = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    n = len(best_seq)
    print(f"Starting: n={n}, obj={best_score:.8f}")

    # === Phase 1: Intensive LP Alternating (45%) ===
    print(f"\n=== Phase 1: LP Alternating ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.45, 600.0)

    if p1_budget > 30:
        # 3 rounds with perturbations between
        round_budgets = [p1_budget * 0.4, p1_budget * 0.35, p1_budget * 0.25]
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            alloc = min(round_budgets[rnd], remaining * 0.5)
            if alloc < 15:
                break

            # Round 0: from best
            # Round 1+: from perturbed
            if rnd == 0:
                a_input = best_seq
            else:
                scale = 0.015 * (rnd + 1)
                a_input = best_seq * np.exp(rng.normal(0, scale, n))
                a_input = np.maximum(a_input, 0)
                a_input = a_input / a_input.sum()

            a_out = lp_core_loop(a_input, time_budget=alloc, verbose=True, rng=rng)
            obj_out = compute_obj(a_out)
            print(f"  LP round {rnd+1}: obj={obj_out:.8f}")
            if obj_out < best_score:
                best_score = obj_out
                best_seq = a_out.copy()
                n = len(best_seq)
                print(f"  ** Improved! n={n}, obj={best_score:.8f} **")

    # === Phase 2: Subgradient Polish (10%) ===
    print(f"\n=== Phase 2: Subgradient ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.1, 100.0)
    if p2_budget > 10:
        a_sg = subgradient_polish(best_seq, time_budget=p2_budget, K=15, verbose=True)
        obj_sg = compute_obj(a_sg)
        print(f"  SubGrad: obj={obj_sg:.8f}")
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  ** Improved! obj={best_score:.8f} **")

    # === Phase 3: Multi-length Exploration (20%) ===
    print(f"\n=== Phase 3: Multi-length ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    n_options = sorted(set([
        600, 700, 850, 900, 950, 1000, 1100, 1200
    ]))
    n_options = [x for x in n_options if x != n_current and 400 <= x <= 1500]

    per_n = min(remaining * 0.2, 60.0)
    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        # Resample and do quick LP
        a_try = resample_seq(best_seq, n_try)
        alloc = min(per_n, remaining * 0.25)
        if alloc < 10:
            break

        a_try = lp_core_loop(a_try, time_budget=alloc, rng=rng)
        obj_try = compute_obj(a_try)
        print(f"  n={n_try}: obj={obj_try:.8f}")
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            n = n_try
            print(f"  ** New best: n={n_try}, obj={best_score:.8f} **")

    # === Phase 4: Final Polish (25%) ===
    print(f"\n=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Final LP polish
    if remaining > 30:
        alloc = min(remaining * 0.4, 180.0)
        a_final = lp_core_loop(best_seq, time_budget=alloc, verbose=True, rng=rng)
        obj_final = compute_obj(a_final)
        print(f"  Final LP: obj={obj_final:.8f}")
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Improved! obj={best_score:.8f} **")

    # Final subgradient
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 80.0)
        a_final = subgradient_polish(best_seq, time_budget=alloc, K=20)
        obj_final = compute_obj(a_final)
        print(f"  Final SG: obj={obj_final:.8f}")
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()

    # One last short LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 50.0)
        a_final = lp_core_loop(best_seq, time_budget=alloc, rng=rng)
        obj_final = compute_obj(a_final)
        print(f"  Final LP2: obj={obj_final:.8f}")
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()

    # Validate
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
