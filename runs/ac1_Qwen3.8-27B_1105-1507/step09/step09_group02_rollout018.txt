2. Adaptive line search with golden-section refinement

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.sparse import coo_matrix, csr_matrix
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_sparse(a):
    """Build sparse matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    rows = np.zeros(n * min(n, 2 * n - 1), dtype=np.int32)
    cols = np.zeros(n * min(n, 2 * n - 1), dtype=np.int32)
    vals = np.zeros(n * min(n, 2 * n - 1), dtype=np.float64)
    
    idx = 0
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        length = k_end - k_start
        if length > 0:
            rows[idx:idx+length] = range(k_start, k_end)
            cols[idx:idx+length] = j
            vals[idx:idx+length] = a[:length]
            idx += length
    
    A = coo_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2*n-1, n))
    return A.tocsr()


def solve_lp(a, prefer_solver=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal, active_lags) or (None, None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_sparse(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solver_order = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    if prefer_solver:
        solver_order = [prefer_solver] + [s for s in solver_order if s != prefer_solver]
    
    for solver in solver_order:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Get active constraints (dual values)
                    active_lags = []
                    try:
                        for i, c in enumerate(constraints[:-1]):
                            dual_val = c.dual_value
                            if dual_val is not None and dual_val > 1e-6:
                                active_lags.append(i)
                    except:
                        pass
                    return g_val, float(M.value), active_lags
        except Exception:
            continue
    return None, None, None

def fourier_init(n, n_harmonics=5, rng=None):
    """Initialize using a parametric Fourier family."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.arange(n) / n
    a = np.ones(n)
    for j in range(1, n_harmonics + 1):
        amp = rng.uniform(0.1, 0.8) / j
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * j * x + phase)
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a

    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 2.0 / np.sqrt(n)
    momentum = 0.8
    v = np.zeros(n)
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        
        # Softmax weights over top peaks
        temp = max_pv * 0.1 + 1e-10
        weights = np.exp((peak_vals - max_pv) / temp)
        weights /= weights.sum()
        
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]
        
        v = momentum * v + (1 - momentum) * g
        # Riemannian gradient: project onto tangent space (sum = 0)
        v = v - np.mean(v)
        
        a_new = a - lr * v
        # Project onto simplex
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                lr *= 0.8
            if stall > 300:
                lr = 2.0 / np.sqrt(n)
                v = np.zeros(n)
                stall = 0
            if lr < 1e-10:
                break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP polishing with adaptive line search + subgradient + multi-n.
    Time allocation: LP main (50%), subgradient (10%), multi-n (20%), final polish (20%).
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Try to load parent
    n_base = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_base = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = np.ones(n_base) / n_base
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_base}): obj={best_score:.8f}")
    
    # === Phase 1: Quick LP convergence (150s) ===
    print("\n=== Phase 1: Quick LP ===")
    remaining = deadline - time.monotonic()

    if p1_budget > 30:
        a1 = lp_polish(best_seq, time_budget=p1_budget, verbose=True, seed=seed)

        else:
            print(f"  Phase1 LP: obj={obj1:.8f}")
    
    # === Phase 2: Heavy LP polishing (500s) ===
    print("\n=== Phase 2: Heavy LP ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.45, 500.0)
    if p2_budget > 60:
        # Multiple rounds with perturbation between
        num_rounds = 3
        per_round = p2_budget / num_rounds
        for rnd in range(num_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(per_round, remaining * 0.6)
            if alloc < 20:
                break
            s_offset = seed + 100 * (rnd + 1)
            a2 = lp_polish(best_seq, time_budget=alloc, verbose=(rnd == 0), seed=s_offset)
            obj2 = compute_obj(a2)
            if obj2 < best_score:
                best_score = obj2
                best_seq = a2.copy()
                print(f"  LP round {rnd+1}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd+1}: obj={obj2:.8f}")
                if rnd < num_rounds - 1:
                    # Perturb for next round using Fourier noise
                    n = len(best_seq)
                    x = np.arange(n) / n
                    noise = np.zeros(n)
                    for j in range(1, 4):
                        noise += 0.02 * rng.uniform(0.5, 2.0) * np.cos(2 * np.pi * j * x + rng.uniform(0, 2*np.pi))
                    best_seq = best_seq * np.exp(noise)
                    best_seq = np.maximum(best_seq, 0)
                    best_seq = best_seq / best_seq.sum()
    
    # === Phase 3: Subgradient refinement (100s) ===
    print("\n=== Phase 3: Subgradient ===")
    remaining = deadline - time.monotonic()

            print(f"  Phase3 SG: obj={obj3:.8f}")
    
    # === Phase 4: Multi-n exploration (250s) ===
    print("\n=== Phase 4: Multi-n ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    
    n_options = []
    for n_try in [600, 700, 750, 800, 850, 900, 950, 1000, 1100, 1200, 1300, 1500]:
        if n_try != n_current and 400 <= n_try <= 2000:
            n_options.append(n_try)
    
    # Sort by proximity to current best n first
    n_options.sort(key=lambda x: abs(x - n_current))
    
    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        
        a_try = resample_sequence(best_seq, n_try)
        
        # Quick LP polish (limited time)
        alloc = min(remaining * 0.15, 60.0)
        if alloc > 15:

        obj_try = compute_obj(a_try)
        if obj_try < best_score:

            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")

    if remaining > 20:
        alloc = min(remaining * 0.3, 60.0)
        a6 = subgradient_refine(best_seq, time_budget=alloc, K=20)
        obj6 = compute_obj(a6)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Final SG: obj={best_score:.8f} **")
    
    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 80.0)
        a7 = lp_polish(best_seq, time_budget=alloc, seed=seed + 7777)
        obj7 = compute_obj(a7)
        if obj7 < best_score:
            best_score = obj7
            best_seq = a7.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    if best_seq is None:

    return result
