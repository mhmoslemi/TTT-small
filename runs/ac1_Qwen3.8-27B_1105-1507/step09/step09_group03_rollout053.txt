Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements: fixed LP solver bug, peak-directed local search,
efficient pair moves, and multi-n exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fixed LP (bug fix from previous), LSE-smoothed Nesterov descent,

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_minmax(a, time_limit=30.0):
    """LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, time_limit=time_limit)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_active_sum(a, active_idx, time_limit=20.0):
    """LP: min sum(conv(a,g)[k] for k in active) s.t. sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    A_act = A[active_idx, :]
    
    g = cp.Variable(n, nonneg=True)
    obj = cp.sum(A_act @ g)
    constraints = [cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(obj), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, time_limit=time_limit)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv
        except Exception:
            continue
    return None


def lse_grad_conv(a, tau):

    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    p = np.exp(tau * (conv - max_c))
    p_sum = float(np.sum(p))
    if p_sum < 1e-300:
        p = np.zeros_like(conv)
        p[np.argmax(conv)] = 1.0
    else:
        p = p / p_sum
    
    # cross_corr[i] = sum_k p[k] * a[k-i] = ifft(fft(p) * conj(fft(a)))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross_corr = ifft(P * np.conj(A)).real
    
    grad = (2.0 / tau) * cross_corr[:n]
    return grad, conv


def nesterov_lse_descent(a, time_budget=200.0, tau_start=20.0, tau_end=500.0):
    """Nesterov accelerated gradient descent on LSE-smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.85
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / (time_budget * 0.95))
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy < 1e-10:
            y = best_a.copy()
        else:
            y = y / sy
        
        grad, conv = lse_grad_conv(y, tau)
        
        # Gradient step
        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / sn
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)
        
        # Reset if stuck
        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.7
            best_it = it
    
    return best_a


def peak_pair_search(a, time_budget=150.0, num_peaks=8):
    """
    Peak-directed pair move search.
    For top peaks, find (source, target) pairs that reduce the max peak.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        
        # Top peaks
        top_k = np.argsort(conv)[::-1][:num_peaks]
        peak_vals = conv[top_k]
        
        # Compute profiles: profile[m, i] = a[k_m - i] + a[i - k_m] (with bounds)
        # where k_m = top_k[m]
        profiles = np.zeros((num_peaks, n))
        for m in range(num_peaks):
            k = int(top_k[m])
            for i in range(n):
                val = 0.0
                if 0 <= k - i < n:
                    val += a[k - i]
                if 0 <= i - k < n:
                    val += a[i - k]
                profiles[m, i] = val
        
        # For moving δ from j to i: Δpeak_m ≈ δ * (profiles[m, i] - profiles[m, j])
        # We want to reduce the max peak, so we want profiles[m, i] < profiles[m, j] for large m
        
        # Strategy: find j with high "weighted profile" and i with low "weighted profile"
        # Weight by how much each peak is above the second-highest
        
        # Simple approach: use the worst peak's profile
        worst_profile = profiles[0]
        
        # Sample candidate sources (high profile) and targets (low profile)
        source_idx = np.argsort(worst_profile)[::-1]
        target_idx = np.argsort(worst_profile)
        
        improved = False
        tried_this_iter = 0
        
        for j in source_idx[:30]:
            if a[int(j)] < 1e-12:
                continue
            for i in target_idx[:30]:
                if i == j:
                    continue
                i, j = int(i), int(j)
                
                # Compute effect on all peaks
                delta_profile = profiles[:, i] - profiles[:, j]
                
                # Try a few delta values
                max_d = 0.3 * a[j]
                for d_frac in [0.1, 0.3, 0.5, 0.7]:
                    d = d_frac * max_d
                    if d <= 0:
                        continue
                    
                    new_peaks = peak_vals + d * delta_profile
                    new_max = float(np.max(new_peaks))
                    # Also check other peaks not in top_k (approximate)
                    # For now, just check the top ones
                    
                    if new_max < max_val * (1 - 1e-12):
                        # Do the move
                        a[i] += d
                        a[j] -= d
                        # Recompute true objective
                        obj_new = compute_obj(a)
                        if obj_new < best_obj - 1e-14:
                            best_obj = obj_new
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            # Undo
                            a[j] += d
                            a[i] -= d
                    
                    tried_this_iter += 1
                if improved:
                    break
            if improved:
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            if stall % 5 == 0:
                # Random perturbation to escape
                scale = 0.005 + 0.003 * min(stall // 5, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_a = float(np.sum(a))
                if s_a > 1e-10:
                    a = a / s_a
                    obj_a = compute_obj(a)
                    if obj_a < best_obj:
                        best_obj = obj_a
                        best_a = a.copy()
                        stall = 0
            
            if stall > 100:
                break
    
    return best_a


def lp_polish(a, time_budget=200.0, max_iters=50):
    """
    LP-based polishing: alternate minmax LP and active-set LP,
    with line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget and it < max_iters:
        # Stage 1: minmax LP
        g1, M1 = solve_lp_minmax(a, time_limit=min(20.0, (time_budget - (time.monotonic() - start)) * 0.3))
        if g1 is None:
            break
        
        # Find active set
        conv_ag = build_conv_matrix(a / np.sum(a)) @ g1
        max_ag = float(np.max(conv_ag))
        tol = 0.05 * max_ag
        active = np.where(conv_ag >= max_ag - tol)[0]
        if len(active) < 3:
            active = np.argsort(conv_ag)[::-1][:10]
        
        # Stage 2: min active sum LP
        g2 = solve_lp_active_sum(a, active, time_limit=min(15.0, (time_budget - (time.monotonic() - start)) * 0.2))
        
        # Choose direction
        if g2 is not None:
            # Blend
            g_dir = 0.5 * g1 + 0.5 * g2
        else:
            g_dir = g1
        
        # Line search
        best_t = 0.0
        best_f = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_dir
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            f_try = compute_obj(a_try)
            if f_try < best_f - 1e-14:
                best_f = f_try
                best_t = t
        
        if best_t > 0:
            a_new = (1 - best_t) * a + best_t * g_dir
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            best_obj = best_f
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            # Perturb
            if stall > 3:
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                stall = 0
            
            if stall > 20:
                break
        
        it += 1
    
    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
