"""
Fast LP-based Frank-Wolfe optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses scipy HiGHS solver for the LP subproblem, with proper line search and
KKT-based convergence detection. Multi-length exploration for global search.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        A[k, j_lo:j_hi+1] = a[k - j_lo::-1][:j_hi - j_lo + 1]
    return A

def solve_lp_fast(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    a = a / float(np.sum(a))
    A_conv = build_conv_matrix(a)
    
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    # Inequality: A_conv @ g - M <= 0
    A_ub = np.hstack([A_conv, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]  # M >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status == 0 and res.x is not None:
        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s, float(res.x[-1])

_global_best_score = np.inf

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a.astype(np.float64), size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_conv_xy(x, y):
    """Full linear convolution x*y via FFT."""
    n = max(len(x), len(y))
    size = 1
    while size < len(x) + len(y) - 1:
        size <<= 1
    fx = rfft(x.astype(np.float64), size)
    fy = rfft(y.astype(np.float64), size)
    return irfft(fx * fy, size)[:len(x)+len(y)-1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(n):
        A[k, :k+1] = a[k::-1]
    for k in range(n, 2*n-1):
        A[k, k-n+1:] = a[n-1 : k-n : -1]
    return A

def solve_lp(a):

    n = len(a)

    A_conv = build_conv_matrix(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A_conv, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})

        s = float(np.sum(g))

    fx = rfft(x.astype(np.float64), size)
    fy = rfft(y.astype(np.float64), size)
    return irfft(fx * fy, size)[:nx + ny - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(n):
        A[k, :k + 1] = a[k::-1]
    for k in range(n, 2 * n - 1):
        A[k, k - n + 1:] = a[n - 1:k - n:-1]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0, M>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a = a.astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a = a / s
    A_conv = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([A_conv, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x[:n], 0)
            s_g = float(np.sum(g))
            if s_g > 1e-10:
                return g / s_g, float(max(0, res.x[-1]))
    except Exception:
        pass
    return None, None


def line_search_convex(C_aa, C_ag, C_gg):
    """
    Golden section search to minimize f(t) = max_k q_k(t) over t in [0,1],
    where q_k(t) = (1-t)^2*C_aa[k] + 2t(1-t)*C_ag[k] + t^2*C_gg[k].
    f is convex (max of convex quadratics), so golden section works.
    """
    def f(t):
        return float(np.max((1.0 - t) ** 2 * C_aa + 2.0 * t * (1.0 - t) * C_ag + t ** 2 * C_gg))

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc, fd = f(c), f(d)

    for _ in range(45):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)

    t_opt = (lo + hi) * 0.5
    return t_opt, f(t_opt)


def lp_frank_wolfe(a, time_budget, verbose=False, seed=42):
    """
    LP-based Frank-Wolfe with golden-section line search and adaptive perturbation.
    Returns (best_a, best_obj).
    """
    rng = np.random.default_rng(seed)
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n, dtype=np.float64) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    perturb_level = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g, M_lp = solve_lp(a)
        if g is None:
            break

        it += 1

        C_aa = compute_conv(a)
        C_ag = compute_conv_xy(a, g)
        C_gg = compute_conv(g)

        t_opt, _ = line_search_convex(C_aa, C_ag, C_gg)

        if t_opt > 1e-10:
            a_new = (1.0 - t_opt) * a + t_opt * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = a.copy()
                s_new = 1.0
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)

            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                a = a_new
                stall += 1
        else:
            stall += 1

        if verbose and it % 25 == 0:
            print(f"        LW it{it}: obj={best_obj:.8f}, t={t_opt:.5f}, M_lp={M_lp:.8f}, stall={stall}")

        # KKT check: if M_lp is very close to max(conv(a,a)), we're at a fixed point
        if it % 10 == 0:
            max_conv = float(np.max(compute_conv(a)))
            if abs(M_lp - max_conv) < 1e-9:
                if verbose:
                    print(f"        LW: KKT near-optimal at it{it}, obj={best_obj:.8f}")
                # Try one more perturbation before declaring done
                stall = max(stall, 31)

        # Adaptive perturbation when stalled
        if stall >= 25:
            if perturb_level == 0:
                scale = 0.005
            elif perturb_level == 1:
                scale = 0.01
            elif perturb_level == 2:
                scale = 0.02
            elif perturb_level == 3:
                scale = 0.05
            elif perturb_level == 4:
                scale = 0.1
            else:
                # Full restart with random
                a = rng.uniform(0.1, 1.0, n)
                a = a / a.sum()
                perturb_level = 0
                stall = 0
                continue

            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            else:
                a = best_a.copy()
            stall = 0
            perturb_level += 1

            # Also try targeted perturbation: reduce peak contributors
            if perturb_level <= 2:
                conv = compute_conv(best_a)
                top_k = np.argmax(conv)
                n_half = n // 2
                # Reduce elements that contribute to the peak
                peak_contrib = compute_conv_xy(best_a, best_a)
                # Reduce a at positions that are "hot" for the peak
                i_lo = max(0, int(top_k) - n + 1)
                i_hi = min(n - 1, int(top_k))
                for i in range(i_lo, i_hi + 1):
                    j = int(top_k) - i
                    if 0 <= j < n:
                        best_a[i] *= 0.95
                        best_a[j] *= 0.95
                s = float(np.sum(best_a))
                if s > 1e-10:
                    best_a = best_a / s

            if verbose:
                print(f"        LW: perturbed (level={perturb_level}, scale={scale})")

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0], dtype=np.float64)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0).astype(np.float64)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n, dtype=np.float64)
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        # Arcsine-like (more mass at edges)
        x = np.linspace(0.01, 0.99, n)
        a = 1.0 / np.sqrt(x * (1.0 - x) + 1e-8)
    elif mode == 3:
        # Beta(0.5, 0.5) arcsine
        x = np.linspace(0.005, 0.995, n)
        a = 1.0 / np.sqrt(x * (1.0 - x) + 1e-10)
    elif mode == 4:
        # Beta(2,2) - more mass at center
        x = np.linspace(0.005, 0.995, n)
        a = x * (1.0 - x)
    elif mode == 5:
        # Beta(0.5, 2) - asymmetric
        x = np.linspace(0.005, 0.995, n)
        a = x * (1.0 - x) ** 1.5
    elif mode == 6:
        # Two bumps
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * (n // 10) ** 2))
    elif mode == 7:
        # Cosine taper
        x = np.linspace(0, np.pi, n)
        a = np.cos(x * 0.5) ** 2
    else:
        a = np.ones(n, dtype=np.float64)

    a = np.maximum(a, 1e-6).astype(np.float64)
    a = a / a.sum()
    return a


def try_symmetrize(a):
    """Try symmetrizing a and return the better of a and (a+rev(a))/2."""
    a_sym = (a + a[::-1]) * 0.5
    a_sym = np.maximum(a_sym, 0)
    s = float(np.sum(a_sym))
    if s < 1e-10:
        return a, compute_obj(a)
    a_sym = a_sym / s
    obj_orig = compute_obj(a)
    obj_sym = compute_obj(a_sym)
    if obj_sym < obj_orig:
        return a_sym, obj_sym
    return a, obj_orig


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses LP-based Frank-Wolfe with:
    - Fast scipy HiGHS solver (no cvxpy overhead)
    - Golden section line search (exact 1D minimization)
    - Adaptive perturbation for escaping local minima
    - Multi-length exploration
    - Symmetry exploitation
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    rng = np.random.default_rng(seed)

    best_a = None
    best_obj = np.inf

    # Phase 0: Load parent and set up initial candidates
    n_parent = 800
    parent_loaded = False
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_a = parent_arr / s
                best_obj = compute_obj(best_a)
                n_parent = parent_arr.size
                parent_loaded = True
                print(f"Parent loaded: n={n_parent}, obj={best_obj:.8f}")
    except Exception:
        pass

    if not parent_loaded:
        best_a = np.ones(n_parent, dtype=np.float64) / n_parent
        best_obj = compute_obj(best_a)
        print(f"Default uniform init: n={n_parent}, obj={best_obj:.8f}")

    def remaining_time():
        return max(0.0, deadline - time.monotonic())

    # Phase 1: LP Frank-Wolfe on parent (50% of time)
    print(f"\n=== Phase 1: LP Frank-Wolfe (n={len(best_a)}) ===")
    p1_budget = min(remaining_time() * 0.45, 600.0)
    if p1_budget > 30:
        a_out, obj_out = lp_frank_wolfe(best_a, time_budget=p1_budget, verbose=True, seed=seed)
        if obj_out < best_obj:
            best_obj = obj_out
            best_a = a_out
            print(f"  Phase 1 result: obj={best_obj:.8f}")
        else:
            print(f"  Phase 1 result: obj={obj_out:.8f} (no improvement)")

    # Try symmetrizing
    a_sym, obj_sym = try_symmetrize(best_a)
    if obj_sym < best_obj:
        best_obj = obj_sym
        best_a = a_sym
        print(f"  Symmetrized: obj={best_obj:.8f}")

    # Phase 2: Continue LP with perturbation (20% of time)
    print(f"\n=== Phase 2: Continuation LP ===")
    p2_budget = min(remaining_time() * 0.2, 250.0)
    if p2_budget > 20:
        # Perturb slightly then continue
        scale = 0.008
        a_pert = best_a * np.exp(rng.normal(0, scale, len(best_a)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        a_out, obj_out = lp_frank_wolfe(a_pert, time_budget=p2_budget, verbose=False, seed=seed + 1)
        if obj_out < best_obj:
            best_obj = obj_out
            best_a = a_out
            print(f"  Phase 2 result: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 2 result: obj={obj_out:.8f}")

    # Phase 3: Multi-length exploration (20% of time)
    print(f"\n=== Phase 3: Multi-length Exploration ===")
    p3_budget = remaining_time() * 0.22
    n_current = len(best_a)

    n_options = []
    for dn in [-200, -100, 100, 200, 300, 400]:
        n_try = n_current + dn
        if 300 <= n_try <= 2000 and n_try != n_current:
            n_options.append(n_try)
    # Add some fixed options
    for n_fixed in [500, 600, 700, 800, 900, 1000, 1200]:
        if 300 <= n_fixed <= 2000 and n_fixed not in n_options:
            n_options.append(n_fixed)

    n_options = sorted(set(n_options))
    # Sort by proximity to current (try nearby lengths first)
    n_options.sort(key=lambda x: abs(x - n_current))
    n_options = n_options[:6]  # Try at most 6 lengths

    for n_try in n_options:
        if remaining_time() < 40:
            break
        if n_try == n_current:
            continue

        alloc = min(p3_budget / 3.0, remaining_time() * 0.4, 100.0)
        if alloc < 20:
            alloc = min(60.0, remaining_time() * 0.3)
        if alloc < 15:
            break

        print(f"  Trying n={n_try} (budget={alloc:.0f}s)...")
        a_try = resample_sequence(best_a, n_try)
        a_out, obj_out = lp_frank_wolfe(a_try, time_budget=alloc, verbose=False, seed=seed + n_try)

        # Try symmetrizing
        a_sy, obj_sy = try_symmetrize(a_out)
        if obj_sy < obj_out:
            obj_out = obj_sy
            a_out = a_sy

        if obj_out < best_obj:
            best_obj = obj_out
            best_a = a_out
            print(f"  ** n={n_try}: obj={best_obj:.8f} IMPROVED!")
        else:
            print(f"  n={n_try}: obj={obj_out:.8f}")

    # Phase 4: Final intensive LP polish (remaining time)
    print(f"\n=== Phase 4: Final Polish ===")
    remaining = remaining_time()
    if remaining > 20:
        alloc = min(remaining * 0.6, 200.0)
        a_out, obj_out = lp_frank_wolfe(best_a, time_budget=alloc, verbose=True, seed=seed + 999)
        if obj_out < best_obj:
            best_obj = obj_out
            best_a = a_out
            print(f"  Final LP: obj={best_obj:.8f} **")
        else:
            print(f"  Final LP: obj={obj_out:.8f}")

    # Try symmetrize one more time
    a_sym, obj_sym = try_symmetrize(best_a)
    if obj_sym < best_obj:
        best_obj = obj_sym
        best_a = a_sym
        print(f"  Final sym: obj={best_obj:.8f}")

    # Phase 5: Very small perturbation + quick LP (last few seconds)
    remaining = remaining_time()
    if remaining > 15:
        alloc = min(remaining * 0.5, 30.0)
        for pert_scale in [0.002, 0.005]:
            if remaining_time() < 10:
                break
            a_p = best_a * np.exp(rng.normal(0, pert_scale, len(best_a)))
            a_p = np.maximum(a_p, 0)
            a_p = a_p / a_p.sum()
            a_out, obj_out = lp_frank_wolfe(a_p, time_budget=alloc, verbose=False, seed=seed + 1000)
            if obj_out < best_obj:
                best_obj = obj_out
                best_a = a_out
                print(f"  Micro-perturb ({pert_scale}): obj={best_obj:.8f} **")

    # Final validation
    if best_a is None:
        best_a = np.ones(200, dtype=np.float64)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_obj:
        best_obj = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")
    return result
