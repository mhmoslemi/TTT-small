import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    if n == 0: return np.array([])
    size = 2**np.ceil(np.log2(2*n-1)).astype(int)
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

def compute_obj(a):
    """Objective: 2*n*max(conv)/(sum)^2"""
    n = len(a)
    s = np.sum(a)
    if s < 0.01: return np.inf
    a_n = a/s
    return 2.0*n*float(np.max(compute_conv(a_n)))

def solve_lp(a, tight=True):
    """LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0"""
    n = len(a)
    a_n = a/np.sum(a)
    # Build constraint matrix
    # (a*g)[k] = sum_j a[j] * g[k-j] for valid j
    # Use FFT approach: we can express this as a circulant-like matrix
    # But for LP we need the explicit matrix
    # Actually for n=800, the matrix is 1599 x 800 which is fine
    A = np.zeros((2*n-1, n))
    for j in range(n):
        start = j
        end = min(2*n-1, j+n)
        A[start:end, j] = a_n[:end-start]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    # Try solvers in order of preference
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCIP, cp.GUROBI, cp.PDLP, cp.SCS]:
        try:
            kwargs = {}
            if solver == cp.MOSEK:
                kwargs = {'MSDK_MSQKTOL': 1e-10, 'MSDK_MSFEASTOL': 1e-10,
                         'MSDK_MSDINFESTOL': 1e-10, 'MSDK_MSPDINFEASTOL': 1e-10,
                         'MSDK_MSDEGINTOL': 0, 'MSDK_MSTHREADS': 1}
            if solver == cp.CLARABEL:
                kwargs = {'tol_gap_rel': 1e-10, 'tol_feas': 1e-12,
                         'tol_infeas': 1e-12, 'max_iter': 5000}
            prob.solve(solver=solver, **kwargs)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = np.sum(g_val)
                if s > 1e-10:
                    return g_val/s, float(M.value), prob.status
        except:
            continue
    return None, None, None

def exact_line_search(a, g, n_coarse=200, n_fine=400):

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous: exact line search for LP steps, tighter solver
tolerances, dual-guided perturbation, systematic multi-n exploration, and
convolution profile analysis to focus LP effort on active constraints.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    target = 2 * n - 1
    while size < target:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:target]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_conv_profile(a):
    """Return convolution and its max info."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_val = float(np.max(conv))
    max_idx = int(np.argmax(conv))
    return conv, max_idx, max_val


def solve_lp(a, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (optimal_g, optimal_M, dual_vars) or (None, None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix A (2n-1 x n) where (A@g)[k] = (a*g)[k]
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a_norm
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.GUROBI, cp.PDLP, cp.SCIP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            kwargs = {}
            if solver == cp.MOSEK:
                kwargs = {'MSDK_MSQKTOL': 1e-10, 'MSDK_MSFEASTOL': 1e-10,
                          'MSDK_MSDINFESTOL': 1e-10, 'MSDK_MSPDINFEASTOL': 1e-10,
                          'MSDK_MSTHREADS': 1}
            elif solver == cp.CLARABEL:
                kwargs = {'tol_gap_rel': 1e-10, 'tol_feas': 1e-12,
                          'tol_infeas': 1e-12, 'max_iter': 10000}
            t0 = time.monotonic()
            prob.solve(solver=solver, **kwargs)
            elapsed = time.monotonic() - t0
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    # Try to get dual variables for the inequality constraints
                    dual_vars = None
                    try:
                        dual_vars = constraints[0].dual_value
                    except Exception:
                        pass
                    if verbose:
                        print(f"      LP solved by {solver} in {elapsed:.3f}s, status={prob.status}")
                    return g_val / s, float(M.value), dual_vars
        except Exception:
            continue
    
    return None, None, None


def exact_line_search(a, g, n_coarse=300, n_fine=500):
    """Find optimal step size t for (1-t)*a + t*g that minimizes the objective."""
    best_t = 0.0
    best_val = compute_obj(a)
    
    # Coarse search
    for t in np.linspace(0.005, 1.0, n_coarse):
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0)
        s = float(np.sum(a_t))
        if s < 1e-10:
            continue
        a_t = a_t / s
        val = compute_obj(a_t)
        if val < best_val:
            best_val = val
            best_t = t
    
    # Fine search around best
    if best_t > 0:
        lo = max(0.001, best_t - 0.005)
        hi = min(1.0, best_t + 0.005)
        for t in np.linspace(lo, hi, n_fine):
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0)
            s = float(np.sum(a_t))
            if s < 1e-10:
                continue
            a_t = a_t / s
            val = compute_obj(a_t)
            if val < best_val:
                best_val = val
                best_t = t
    
    return best_t, best_val

def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of convolution using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    N = 1
    while N < 3 * n:
        N <<= 1
    
    from scipy.fft import fft, ifft
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def gradient_descent(a, time_budget=100.0, tau_init=20.0, tau_end=500.0,
                     lr_init=None, verbose=False):
    """
    Adaptive smoothed gradient descent with Nesterov momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if lr_init is None:
        lr = 2.0 / np.sqrt(n)
    else:
        lr = lr_init
    beta = 0.9
    v = np.zeros(n)
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        # Cosine annealing for tau
        tau = tau_init + (tau_end - tau_init) * (1 - 0.5 * (1 + np.cos(np.pi * progress)))
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        g, conv = smoothed_gradient_fft(y, tau)
        
        # Update momentum
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        
        # Decay learning rate slowly
        lr = max(1e-10, lr * 0.99995)
        
        # Reset if stuck
        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)
            best_it = it
        
        if verbose and it % 2000 == 0:
            print(f"      GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.arange(n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian peaks
        c1, c2 = n // 4, 3 * n // 4
        w = n // 8
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        xn = x / (n - 1)
        a = 1.0 / (np.sqrt(np.maximum(1 - (2*xn - 1)**2, 0.01)) + 1e-3)
        a = a / a.sum()
    elif mode == 4:
        # Cosine window
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Exponential decay from center
        a = np.exp(-np.abs(x - n / 2) / (n / 6))
        a = a / a.sum()
    elif mode == 6:
        # Four peaks
        centers = [n//8, 3*n//8, 5*n//8, 7*n//8]
        w = n // 16
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 7:
        # Triangular
        a = np.maximum(1 - 4 * np.abs(x / (n - 1) - 0.5), 0.01)
        a = a / a.sum()
    elif mode == 8:
        # S_MOD
        freqs = rng.integers(2, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x / n + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

        a_lp, obj_lp = lp_polish(best_seq, time_budget=alloc, verbose=False,
                                  seed=12345)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Ultra-final LP: obj={best_score:.8f} **")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
