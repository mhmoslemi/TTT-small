    """Initialize with 1/sqrt(i+1) shape."""

def multi_peak_gradient_descent(a, time_budget, top_k=10, verbose=False, symmetrize_every=50):
    """
    Gradient descent minimizing max(conv(a,a)) on the simplex.
    Uses weighted multi-peak gradient with backtracking line search.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    
    lr = 0.05
    j_arr = np.arange(n)
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(conv.max())
        mean_val = float(conv.mean())
        
        # Find top peaks weighted by excess
        k = min(top_k, len(conv))
        top_idx = np.argsort(conv)[::-1][:k]
        weights = np.maximum(conv[top_idx] - mean_val, 0)
        w_sum = weights.sum()
        if w_sum < 1e-15:
            # All peaks are at or below mean - use uniform
            weights = np.ones(k) / k
        else:
            weights = weights / w_sum

        grad = np.zeros(n)
        for ki, wi in zip(top_idx, weights):
            offset = ki - j_arr
            valid = (offset >= 0) & (offset < n)
            if np.any(valid):
                grad[valid] += 2.0 * wi * a[offset[valid]]

        grad = grad - grad.mean()

        improved = False

            a_new = project_simplex(a - alpha * grad)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            alpha *= 0.5

            stall = 0
        else:
            stall += 1

        if verbose and it % 200 == 0:
            print(f"    MPGD it{it}: obj={best_obj:.8f}, max_conv={max_val:.6f}, lr={lr:.4f}")
    
    return best_a
