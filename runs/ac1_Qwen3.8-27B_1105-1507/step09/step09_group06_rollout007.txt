Uses exact active-set subgradient method with adaptive step sizes,

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu * rho > (cssv - 1.0)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_val = idx[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

    """
    n = len(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    # Gradient: sum_k p[k] * grad(c_k)
    grad = np.zeros(n)
    for k in range(len(conv)):
        if p[k] < 1e-10:
            continue
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_end >= i_start:
            indices = np.arange(i_start, i_end + 1)
            corresponding = k - indices
            grad[indices] += p[k] * 2.0 * a[corresponding]
    
    return grad


def build_conv_matrix(a):
    """Build matrix A_mat (2n-1 x n) such that A_mat @ g = convolve(a, g)."""
    n = len(a)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A_mat[j:j + r, j] = a[:r]
    return A_mat


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A_mat = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def active_set_gd(a, time_budget=100.0, verbose=False, seed=0):
    """
    Gradient descent using exact active-set subgradient.
    With backtracking line search and Nesterov acceleration.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Parameters
    momentum = 0.75
    v = np.zeros(n)
    lr = 2.0 / np.sqrt(n)
    it = 0
    best_it = 0
    last_improve = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Determine mixing: sometimes use exact subgradient, sometimes smoothed
        if it % 50 < 30:
            # Use exact subgradient
            grad, n_active = exact_subgradient(a, conv, n_active=min(10, max(3, n//100)), rel_tol=5e-4)
        else:
            # Use smoothed subgradient for smoother landscape
            tau = 20.0 + 30.0 * min(1.0, it / 500.0)
            grad = smoothed_subgradient(a, conv, tau=tau)
        
        # Nesterov momentum
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        # Backtracking line search
        obj_new = compute_obj(a_new)
        if obj_new > best_obj:
            # Try smaller step
            for backoff in [0.5, 0.25, 0.1]:
                a_try = a + backoff * v
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj or obj_try < obj_new:
                        a_new = a_try
                        obj_new = obj_try
                        break
            else:
                a_new = a  # No improvement, stay
                obj_new = compute_obj(a)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        
        # Decay learning rate
        lr = max(1e-6, lr * 0.9995)
        
        # Reset if stuck
        if it - best_it > 2000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)
            best_it = it
            # Add small perturbation
            noise = rng.normal(0, 0.01, n)
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
        
        if verbose and it % 500 == 0:
            print(f"      ASG it{it}: obj={best_obj:.8f}, lr={lr:.6f}")
    
    return best_a


def lp_with_line_search(a, time_budget=100.0, verbose=False, seed=0):
    """
    LP-based direction finding with line search along the LP direction.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    it = 0
    consecutive_fails = 0
    
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break
        
        # Direction: g_lp - a, projected to tangent of simplex
        direction = g_lp - a
        
        # Line search along direction
        best_t = 0.0
        best_t_obj = best_obj
        
        # Try a range of t values (log-spaced)
        t_candidates = np.concatenate([
            np.logspace(-2.5, 0, 15),  # 0.003 to 1.0
            np.array([0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9])
        ])
        t_candidates = np.unique(t_candidates)
        
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        it += 1
        
        if best_t > 0 and best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = ((1.0 - best_t) * a + best_t * g_lp)
            best_a = np.maximum(best_a, 0)
            best_a = best_a / best_a.sum()
            a = best_a.copy()
            consecutive_fails = 0
            if verbose and it % 3 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t:.4f}")
        else:
            consecutive_fails += 1
            # Perturb to escape
            scale = 0.02 * min(consecutive_fails, 5) + 0.01
            noise = rng.normal(0, 1, n) * scale
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            # Blend with best
            a = 0.5 * a + 0.5 * best_a
            a = a / a.sum()
            
            if consecutive_fails > 20:
                break
    
    return best_a


def freq_domain_smooth(a, alpha=0.5, seed=0):
    """
    Apply frequency-domain smoothing: reduce high-frequency components.
    This can help by reducing oscillatory patterns that create peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    # DFT
    A = fft(a, n)
    
    # Low-pass filter: reduce high frequencies
    freqs = np.fft.fftfreq(n)
    # Smoothly attenuate high frequencies
    # Use a raised cosine filter
    cutoff = 0.3  # fraction of Nyquist
    attenuation = 1.0 / (1.0 + alpha * (np.abs(freqs) / cutoff) ** 2)
    A_filtered = A * attenuation
    
    a_new = ifft(A_filtered).real
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new > 1e-10:
        a_new = a_new / s_new
    return a_new


def peak_redundancy_check(a):
    """
    Check if the convolution has redundant peaks (many peaks at the same level).
    Returns the number of near-max peaks.
    """
    conv = compute_conv(a)
    max_v = float(np.max(conv))
    near_max = np.sum(conv > max_v * (1 - 1e-3))
    return near_max


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Raised cosine (Hann window)
        x = np.arange(n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x / n))
        a = a / a.sum()
    elif mode == 3:
        # Quarter-sine
        x = np.arange(n)
        a = np.sin(np.pi * x / (2 * n)) ** 2
        a = a / a.sum()
    elif mode == 4:
        # Exponential decay from center
        center = n / 2
        x = np.arange(n)
        sigma = n / 6
        a = np.exp(-(x - center) ** 2 / (2 * sigma ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Flat top with tapered edges
        tap = max(1, n // 10)
        a = np.ones(n)
        for i in range(tap):
            a[i] *= (i + 1) / (tap + 1)
            a[n-1-i] *= (i + 1) / (tap + 1)
        a = a / a.sum()
    elif mode == 6:
        # Modulated: 1 + eps*cos
        x = np.arange(n)
        eps = 0.3
        a = 1 + eps * np.cos(2 * np.pi * x / n)
        a = a / a.sum()
    elif mode == 7:
        # Fejér kernel approximation
        x = np.arange(n)
        a = np.sinc(x / (n/4)) ** 2
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 8:
        # Two-hump
        c1, c2 = n//3, 2*n//3
        x = np.arange(n)
        w = n // 10
        a = np.exp(-(x-c1)**2/(2*w**2)) + np.exp(-(x-c2)**2/(2*w**2))
        a = a / a.sum()
    else:
        x = np.linspace(0, 1, n)
        a = 1 - abs(2*x - 1)  # Triangle
        a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    # Use scipy-like interpolation via numpy
    from scipy.interpolate import interp1d
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    f = interp1d(x_old, a, kind='cubic', fill_value=0, bounds_error=False)
    a_new = f(x_new)
    a_new = np.maximum(a_new, 0)
    return a_new


def full_optimize(a, time_budget=400.0, verbose=False, seed=0):
    """
    Full optimization: alternate between ASG and LP, with periodic
    frequency smoothing and restarts.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    phase = 0
    gｄ_count = 0
    lp_count = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break
        
        # Alternate between GD and LP
        # Use a round-robin: 2 GD steps, then 1 LP step
        if phase % 3 < 2:
            # GD step
            alloc = min(30.0, remaining * 0.15)
            if alloc > 5:
                a_gd = active_set_gd(a, time_budget=alloc, verbose=False, seed=seed + phase)
                obj_gd = compute_obj(a_gd)
                if obj_gd < best_obj:
                    best_obj = obj_gd
                    best_a = a_gd.copy()
                    a = a_gd.copy()
                    if verbose:
                        print(f"    FullOpt GD: obj={best_obj:.8f}")
        else:
            # LP step
            alloc = min(40.0, remaining * 0.2)
            if alloc > 5:
                a_lp = lp_with_line_search(a, time_budget=alloc, verbose=verbose, seed=seed + phase)
                obj_lp = compute_obj(a_lp)
                if obj_lp < best_obj:
                    best_obj = obj_lp
                    best_a = a_lp.copy()
                    a = a_lp.copy()
                    if verbose:
                        print(f"    FullOpt LP: obj={best_obj:.8f}")
                else:
                    # Try frequency smoothing
                    a_sm = freq_domain_smooth(a, alpha=0.3 + 0.1 * (phase // 20))
                    a = 0.7 * a + 0.3 * a_sm
                    a = np.maximum(a, 0)
                    a = a / a.sum()
        
        phase += 1
    
    return best_a
