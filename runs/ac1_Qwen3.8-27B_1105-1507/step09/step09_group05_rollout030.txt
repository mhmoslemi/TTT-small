def build_conv_matrix(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            # A[k, j] = a[k-j] for j in [j_start, j_end]

            A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
