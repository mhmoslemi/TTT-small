def frank_wolfe_step(a, grad, step_size=None):
    """
    Frank-Wolfe (conditional gradient) step on the simplex.
    Find v = e_j where j = argmin(grad), then move towards it.
    """
    n = len(a)
    j = int(np.argmin(grad))
    v = np.zeros(n)
    v[j] = 1.0
    
    if step_size is None:
        # Standard FW step size: 2/(t+2) or use backtracking
        step_size = 2.0 / (len(a) + 2)  # crude
    
    a_new = (1.0 - step_size) * a + step_size * v
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new, j

def mirror_descent_step(a, grad, eta):
    """
    Mirror descent step with negative entropy (exponential) update.
    a_new[i] = a[i] * exp(-eta * grad[i])
    """
    # Use log domain for stability
    log_a = np.log(np.maximum(a, 1e-30))
    log_a_new = log_a - eta * grad
    # Normalize
    log_a_new -= np.max(log_a_new)
    a_new = np.exp(log_a_new)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new

def lp_solve_active(a, active_indices, margin=0.0):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M for k in active_indices, sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    num_constraints = len(active_indices)
    
    # Build constraint matrix for only active indices
    A_rows = []
    for k in active_indices:
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[i] = a_norm[j]
            # Also the symmetric term
            if 0 <= k - i < n and i != k - i:
                # Actually we already counted both in the sum
                pass
        # Wait, conv(a,g)[k] = sum_i a_norm[i] * g[k-i] (with bounds)
        # So the row for constraint k is: row[i] = a_norm[k-i] if 0<=k-i<n
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[i] = a_norm[j]
        A_rows.append(row)
    
    A_mat = np.array(A_rows)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def identify_active_set(conv, top_k=50, threshold_frac=0.95):
    """Identify which convolution positions are 'active' (close to max)."""
    max_conv = float(np.max(conv))
    threshold = max_conv * threshold_frac
    active = np.where(conv >= threshold)[0]
    # Also include the top_k
    top = np.argsort(conv)[::-1][:top_k]
    active = np.union1d(active, top)
    return sorted(active.tolist())

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 3:
        # Multiple peaks
        n_peaks = rng.integers(2, 6)
        centers = rng.integers(0, n, n_peaks)
        widths = rng.uniform(n // 20, n // 5, n_peaks)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 4:
        # Sinusoidal
        freqs = rng.uniform(0.5, 5, 3)
        phases = rng.uniform(0, 2 * np.pi, 3)
        x = np.arange(n) / n
        a = np.zeros(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * (1 + np.sin(2 * np.pi * f * x + p))
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 5:
        # Triangular
        a = np.abs(1 - 2 * np.arange(n) / n)
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 6:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1, 3)
        a = a / a.sum()
    elif mode == 7:
        # Random smooth
        a = rng.uniform(0, 1, n)
        k = max(2, n // 100)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 8:
        # Alternating
        a = np.ones(n)
        a[::2] = rng.uniform(0.3, 1.0)
        a = a / a.sum()
    elif mode == 9:
        # Lorentzian
        c = n // 2
        w = max(1, n // 10)
        x = np.arange(n)
        a = 1.0 / (1 + ((x - c) / w) ** 2)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

def optimize_single(a_init, time_budget, seed=0):
    """
    Optimize a single sequence using a combination of:
    1. Smoothed gradient descent (with adaptive tau)
    2. Mirror descent (exponential updates)
    3. Active-set LP polishing
    4. Random perturbations to escape local minima
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Parameters
    tau = 20.0
    tau_multiplier = 1.02
    eta = 1.0 / np.sqrt(n)
    fw_lr = 0.1
    lp_interval = 50  # Every 50 iterations, try LP
    lp_top_k = 30
    perturb_interval = 100
    perturb_scale = 0.03
    patience = 500
    last_improve = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        
        # Adaptive tau: increase over time for sharper max approximation
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = 10.0 + 490.0 * progress ** 0.5  # sqrt schedule: starts at 10, ends at 500
        
        # Compute smoothed gradient
        smoothed_val, grad, conv = smoothed_obj_and_grad(a, tau)
        
        # Choose update method based on iteration
        if it % 3 == 0:
            # Gradient descent with projected step
            lr = eta / (1.0 + 0.1 * it)
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
            else:
                a_new = a.copy()
            
        elif it % 3 == 1:
            # Mirror descent (exponential)
            a_new = mirror_descent_step(a, grad, eta * 0.5)
            
        else:
            # Frank-Wolfe step (move towards best vertex) combined with GD
            a_gd = a - lr * grad
            a_gd = np.maximum(a_gd, 0)
            s_gd = float(np.sum(a_gd))
            if s_gd > 1e-10:
                a_gd = a_gd / s_gd
            j = int(np.argmin(grad))
            v = np.zeros(n)
            v[j] = 1.0
            t_fw = 0.1 / (1.0 + 0.01 * it)
            a_new = (1.0 - t_fw) * a_gd + t_fw * v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
            else:
                a_new = a.copy()
        
        # Evaluate
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            last_improve = it
        
        a = a_new
        it += 1
        
        # Periodic LP polish
        if it % lp_interval == 0 and remaining > 30:
            conv_cur = compute_conv(a)
            active = identify_active_set(conv_cur, top_k=lp_top_k, threshold_frac=0.9)
            g_lp, M_lp = lp_solve_active(a, active)
            if g_lp is not None:
                # Try blending
                for t in [0.1, 0.3, 0.5, 0.7, 1.0]:
                    a_try = (1 - t) * a + t * g_lp
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try > 1e-10:
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try.copy()
                            last_improve = it
                            break
        
        # Periodic perturbation to escape local minima
        if it % perturb_interval == 0:
            conv_cur = compute_conv(a)
            # Identify peak positions
            peak_idx = int(np.argmax(conv_cur))
            # Perturb positions that contribute to the peak
            i_start = max(0, peak_idx - (n - 1))
            i_end = min(n - 1, peak_idx)
            noise = rng.normal(0, perturb_scale, n)
            # Larger perturbation near peak
            for i in range(i_start, i_end + 1):
                j = peak_idx - i
                if 0 <= j < n:
                    noise[i] += perturb_scale * a[i] * a[j] / max(a.max(), 1e-10)
            a_pert = a * np.exp(noise)
            a_pert = np.maximum(a_pert, 0)
            s_pert = float(np.sum(a_pert))
            if s_pert > 1e-10:
                a_pert = a_pert / s_pert
                # Blend with best
                a = 0.7 * a_pert + 0.3 * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()
        
        # Reset if stuck
        if it - last_improve > patience:
            a = best_a.copy()
            # Add random perturbation
            noise = rng.normal(0, 0.05, n)
            a = a * np.exp(noise)
            a = np.maximum(a, 0)
            a = a / a.sum()
            last_improve = it
            patience = max(200, patience // 2)
            perturb_scale = min(0.1, perturb_scale * 1.2)
    
    return best_a, best_obj

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-6)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.

Approach:
- Adaptive smoothed gradient with mixed updates (GD, mirror descent, Frank-Wolfe)
- Active-set LP polishing (faster than full LP)
- Peak-aware perturbations targeting convolution peaks
- Sqrt tau schedule for progressive max sharpening
- Multi-length exploration and diversified restarts
- Backtracking line search for robustness

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_obj_and_grad(a, tau):
    """
    Compute smoothed objective (log-sum-exp approximation of max) and gradient.
    a is assumed to be normalized (sum=1).
    Returns (smoothed_val, gradient, conv).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv = compute_conv(a_norm)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    log_sum = np.log(float(np.sum(exp_vals)))
    smoothed_val = max_conv + log_sum / tau

    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n - 2:
        N <<= 1
    P = fft(p, N)
    A = fft(a_norm, N)
    grad = 2.0 * ifft(P * np.conj(A)).real[:n]

    return smoothed_val, grad, conv


def mirror_descent_step(a, grad, eta):
    """Mirror descent with exponential (negative entropy) update."""
    log_a = np.log(np.maximum(a, 1e-30))
    log_a_new = log_a - eta * grad
    log_a_new -= np.max(log_a_new)
    a_new = np.exp(log_a_new)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu * rho > (cssv - 1.0)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_val = int(idx[-1])
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_for_a(a_norm, active_indices):
    """Build constraint matrix rows for active indices only.
    Row k: row[i] = a_norm[k-i] if 0 <= k-i < n else 0.
    """
    n = len(a_norm)
    rows = []
    for k in active_indices:
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[i] = a_norm[j]
        rows.append(row)
    return np.array(rows)


def lp_solve_active(a, active_indices):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M for k in active,
    sum(g)=1, g>=0. Returns (g_opt, M_opt) or (None, None).
    Uses sparse matrix for efficiency.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    a_norm = a_norm.copy()

    # Build sparse constraint matrix
    rows_list = []
    cols_list = []
    vals_list = []
    row_starts = [0]

    for k in active_indices:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                rows_list.append(len(row_starts) - 1)
                cols_list.append(i)
                vals_list.append(a_norm[j])
        row_starts.append(len(rows_list))

    if len(rows_list) == 0:
        return np.ones(n) / n, float(np.max(compute_conv(a_norm)))

    A_mat = np.array([rows_list, cols_list, vals_list])
    # Need to build dense for cvxpy (sparse might work too)
    num_constraints = len(active_indices)
    A_dense = np.zeros((num_constraints, n))
    for idx in range(len(rows_list)):
        A_dense[rows_list[idx], cols_list[idx]] += vals_list[idx]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_dense @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def identify_active_set(conv, top_k=50, threshold_frac=0.92):
    """Identify which convolution positions are 'active' (close to max)."""
    max_conv = float(np.max(conv))
    if max_conv <= 0:
        return list(range(len(conv)))
    threshold = max_conv * threshold_frac
    active = np.where(conv >= threshold)[0]
    top = np.argsort(conv)[::-1][:top_k]
    active = np.union1d(active, top)
    return sorted(int(x) for x in active)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
        a = np.maximum(a, 1e-8)
        a = a / a.sum()
    elif mode == 3:
        n_peaks = int(rng.integers(2, 6))
        centers = rng.integers(0, n, n_peaks)
        widths = rng.uniform(n // 20, n // 5, n_peaks)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 1e-8)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.uniform(0.5, 5, 3)
        phases = rng.uniform(0, 2 * np.pi, 3)
        x = np.arange(n) / n
        a = np.zeros(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * (1 + np.sin(2 * np.pi * f * x + p))
        a = np.maximum(a, 1e-8)
        a = a / a.sum()
    elif mode == 5:
        a = np.abs(1 - 2 * np.arange(n) / n)
        a = np.maximum(a, 1e-8)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0.01, 1, n)
        a = x ** float(rng.uniform(-1, 3))
        a = a / a.sum()
    elif mode == 7:
        a = rng.uniform(0, 1, n)
        k = max(2, n // 80)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-8)
        a = a / a.sum()
    elif mode == 8:
        a = np.ones(n)
        a[::2] = rng.uniform(0.3, 1.0)
        a = a / a.sum()
    elif mode == 9:
        c = n // 2
        w = max(1, n // 8)
        x = np.arange(n)
        a = 1.0 / (1 + ((x - c) / w) ** 2)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def optimize_single(a_init, time_budget, seed=0, verbose=False):

    """

    s = float(np.sum(a))
    if s < 1e-10:

    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    eta_base = 1.0 / np.sqrt(n)
    lp_interval = 40
    lp_top_k = 40
    perturb_interval = 80
    patience = 400
    last_improve = 0
    it = 0
    perturb_scale = 0.03

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 5:
            break

        # Adaptive tau: sqrt schedule
        progress = min(1.0, elapsed / max(time_budget * 0.85, 1.0))
        tau = 15.0 + 585.0 * (progress ** 0.4)

        # Compute smoothed gradient
        smoothed_val, grad, conv = smoothed_obj_and_grad(a, tau)

        # Choose update method
        method = it % 4
        if method == 0:
            # Standard gradient descent with backtracking
            lr = eta_base / (1.0 + 0.05 * it)
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
            else:
                a_new = a.copy()
        elif method == 1:
            # Mirror descent (exponential update)
            a_new = mirror_descent_step(a, grad, eta_base * 0.4)
        elif method == 2:
            # Frank-Wolfe style: blend GD step with move towards best vertex
            lr = eta_base / (1.0 + 0.05 * it)
            a_gd = a - lr * grad
            a_gd = np.maximum(a_gd, 0)
            s_gd = float(np.sum(a_gd))
            if s_gd > 1e-10:
                a_gd = a_gd / s_gd
            else:
                a_gd = a.copy()
            j = int(np.argmin(grad))
            v = np.zeros(n)
            v[j] = 1.0
            t_fw = 0.15 / (1.0 + 0.02 * it)
            a_new = (1.0 - t_fw) * a_gd + t_fw * v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
            else:
                a_new = a.copy()
        else:
            # Nesterov-style: use momentum
            if not hasattr(optimize_single, '_momentum'):
                optimize_single._momentum = {}
            mom_key = id(a)
            if mom_key not in optimize_single._momentum:
                optimize_single._momentum[mom_key] = np.zeros(n)
            beta = 0.7
            v_mom = optimize_single._momentum[mom_key]
            v_mom = beta * v_mom - eta_base * grad
            a_new = a + v_mom
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                optimize_single._momentum[mom_key] = v_mom
            else:
                a_new = a.copy()
                optimize_single._momentum[mom_key] = np.zeros(n)

        # Evaluate
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            last_improve = it

        a = a_new
        it += 1

        # Periodic LP polish
        if it % lp_interval == 0 and remaining > 25:
            conv_cur = compute_conv(a)
            active = identify_active_set(conv_cur, top_k=lp_top_k, threshold_frac=0.88)
            g_lp, M_lp = lp_solve_active(a, active)
            if g_lp is not None:
                best_blend = None
                best_blend_obj = best_obj
                for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    a_try = (1 - t) * a + t * g_lp
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try > 1e-10:
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_blend_obj:
                            best_blend_obj = obj_try
                            best_blend = a_try.copy()
                if best_blend is not None:
                    best_obj = best_blend_obj
                    best_a = best_blend.copy()
                    a = best_blend.copy()
                    last_improve = it

        # Periodic peak-aware perturbation
        if it % perturb_interval == 0:
            conv_cur = compute_conv(a)
            # Find top peaks and create targeted perturbation
            top_peaks = np.argsort(conv_cur)[::-1][:5]
            noise = rng.normal(0, perturb_scale, n)
            for pk in top_peaks:
                pk = int(pk)
                i_lo = max(0, pk - (n - 1))
                i_hi = min(n - 1, pk)
                for i in range(i_lo, i_hi + 1):
                    j = pk - i
                    if 0 <= j < n:
                        weight = a[i] * a[j]
                        noise[i] += perturb_scale * 2.0 * weight / max(float(np.max(a)), 1e-10)
            a_pert = a * np.exp(noise)
            a_pert = np.maximum(a_pert, 0)
            s_pert = float(np.sum(a_pert))
            if s_pert > 1e-10:
                a_pert = a_pert / s_pert
                # Blend with best to not lose progress
                a = 0.6 * a_pert + 0.4 * best_a
                a = np.maximum(a, 0)
                a = a / float(np.sum(a))

        # Reset if stuck
        if it - last_improve > patience:
            a = best_a.copy()
            noise = rng.normal(0, 0.04, n)
            a = a * np.exp(noise)
            a = np.maximum(a, 0)
            a = a / float(np.sum(a))
            last_improve = it
            patience = max(150, patience * 3 // 4)

            a = np.maximum(a, 0)
            a = a / float(np.sum(a))

        if verbose and it % 100 == 0:
            print(f"    it={it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-8)

        start_p2 = time.monotonic()
        it_p2 = 0
        stagnant = 0
        while time.monotonic() - start_p2 < p2_budget:
            conv = compute_conv(a_cur)
            for top_k in [25, 50, 80]:
                active = identify_active_set(conv, top_k=top_k, threshold_frac=0.85)
                g_lp, M_lp = lp_solve_active(a_cur, active)
                if g_lp is not None:
                    for t in [0.05, 0.1, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0]:
                        a_try = (1 - t) * a_cur + t * g_lp
                        a_try = np.maximum(a_try, 0)
                        s_try = float(np.sum(a_try))
                        if s_try > 1e-10:
                            a_try = a_try / s_try
                            obj_try = compute_obj(a_try)
                            if obj_try < best_score:
                                best_score = obj_try
                                best_seq = a_try.copy()
                                a_cur = a_try.copy()
                                stagnant = 0
                                conv = compute_conv(a_cur)
                                break
            it_p2 += 1
            stagnant += 1
            if stagnant > 30:
                # Perturb and continue
                noise = rng.normal(0, 0.04, n_cur)
                a_cur = a_cur * np.exp(noise)
                a_cur = np.maximum(a_cur, 0)
                a_cur = a_cur / a_cur.sum()
                stagnant = 0
            if time.monotonic() - start_p2 < p2_budget and it_p2 % 20 == 0:
                print(f"    LP it{it_p2}: obj={best_score:.8f}")
        print(f"  Phase 2 done: obj={best_score:.8f}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq) if best_seq is not None else 800

    n_options = []
    for delta in [-200, -100, 100, 200, 400, 600]:
        n_try = n_current + delta
        if 400 <= n_try <= 2500:
            n_options.append(n_try)
    for n_abs in [600, 900, 1200, 1500, 2000]:
        if abs(n_abs - n_current) > 80 and 400 <= n_abs <= 2500:
            n_options.append(n_abs)
    n_options = sorted(set(n_options))
    n_options = n_options[:4]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break
        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.12, 120.0)
        if alloc > 25:
            a_opt, obj_opt = optimize_single(a_try, time_budget=alloc,
                                              seed=seed + n_try)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                n_current = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 4: Diversified restarts ===
    print(f"\n=== Phase 4: Diversified Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq) if best_seq is not None else 800

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 70:
            break
        mode = (r * 3 + 1) % 10
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.1 + 0.05 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / float(np.sum(a_r))

        alloc = min(remaining * 0.18, 100.0)
        if alloc > 25:
            a_opt, obj_opt = optimize_single(a_r, time_budget=alloc,
                                              seed=seed + 2000 + r)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_opt:.8f}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 40 and best_seq is not None:
        alloc = min(remaining * 0.4, 200.0)
        a_opt, obj_opt = optimize_single(best_seq, time_budget=alloc,
                                          seed=seed + 5000, verbose=True)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final GD: obj={best_score:.8f} **")
        else:
            print(f"  Final GD: obj={obj_opt:.8f}")

    # Final LP pass
    remaining = deadline - time.monotonic()
    if remaining > 15 and best_seq is not None:
        a_fin = best_seq.copy()
        a_fin = a_fin / float(np.sum(a_fin))
        conv = compute_conv(a_fin)
        for top_k in [30, 60, 100]:
            active = identify_active_set(conv, top_k=top_k, threshold_frac=0.8)
            g_lp, M_lp = lp_solve_active(a_fin, active)
            if g_lp is not None:
                for t in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    a_try = (1 - t) * a_fin + t * g_lp
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try > 1e-10:
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_score:
                            best_score = obj_try
                            best_seq = a_try.copy()
                            a_fin = a_try.copy()
                            conv = compute_conv(a_fin)
                            print(f"  Final LP (k={top_k}, t={t}): obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
