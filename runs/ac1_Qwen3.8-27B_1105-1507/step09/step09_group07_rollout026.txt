"""
Optimized LP alternating with backtracking line search, smoothing,
peak shaving, and two-scale approach for minimizing peak autocorrelation
of non-negative step functions.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(np.asarray(a, dtype=np.float64), size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = np.asarray(a, dtype=np.float64) / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A such that A @ g = conv(a, g) (truncated to 2n-1)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. conv(a, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a_norm))
    if s < 1e-15:
        return None, None
    a_norm = a_norm / s
    
    A = build_conv_matrix(a_norm)
    
    # Formulate: min M
    # s.t. A @ g - M <= 0 (2n-1 constraints)
    #      sum(g) = 1
    #      g >= 0
    #      M free
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s_g = float(np.sum(g_val))
            if s_g > 1e-10:
                return g_val / s_g, float(res.x[n])
    except Exception:
        pass
    
    return None, None


def solve_lp_with_dual(a):

    n = len(a)
    a_norm = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a_norm))
    if s < 1e-15:
        return None, None, None
    a_norm = a_norm / s
    
    A = build_conv_matrix(a_norm)
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:

def peak_shave(a, time_budget, verbose=False):
    """
    Direct peak shaving: identify top peaks of conv(a,a), find contributors,
    and move mass to reduce them.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = float(np.sum(a))
    if s < 1e-15:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for rnd in range(500):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        peak_val = float(np.max(conv))
        peak_lag = int(np.argmax(conv))
        
        # Find top contributors to the peak
        contributors = []
        for i in range(n):
            j = peak_lag - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)
        
        # Find "safe" target positions (low contribution to all top peaks)
        top_peaks = np.argsort(conv)[::-1][:5]
        risk = np.zeros(n)
        for i in range(n):
            for k in top_peaks:
                k = int(k)
                j = k - i
                if 0 <= j < n:
                    risk[i] += a[j]
        
        safe_positions = np.argsort(risk)[:10]
        
        improved = False
        for contrib, i, j in contributors[:8]:
            if contrib < 1e-12:
                continue
            src = i if a[i] >= a[j] else j
            for target in safe_positions:
                if target == src:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02]:
                    delta = frac * a[src]
                    if delta <= 0 or delta >= 0.5 * a[src]:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    obj = compute_obj(a)
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break
            if improved:
                break
        
        if not improved:
            break
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(np.asarray(a, dtype=np.float64), size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

    s = float(np.sum(a))

        return np.inf

    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A such that A @ g = conv(a, g) (length 2n-1)."""

    return A
