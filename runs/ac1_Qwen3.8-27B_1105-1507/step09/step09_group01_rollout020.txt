"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses KL mirror descent, active-set LP polishing, frequency-domain equalization,
and multi-scale exploration to beat previous bounds.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def kl_project(v, n):
    """Project v onto simplex using KL divergence (iterative Bregman projection)."""
    a = np.maximum(v, 1e-30)
    for _ in range(50):
        s = np.sum(a)
        if abs(s - 1.0) < 1e-12:
            break
        a = a / s
    return np.maximum(a, 0)


def gradient_conv_at_peak(a, k):
    """Compute d(conv[k])/da, which is 2*a[k-m] for m=0..n-1 (zero-padded)."""
    n = len(a)
    g = np.zeros(n)
    for m in range(n):
        idx = k - m
        if 0 <= idx < n:
            g[m] = 2.0 * a[idx]
    return g


def smoothed_gradient(a, tau, num_peaks=10):
    """Compute smoothed gradient using top peaks with softmax weighting."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    # Use softmax over all conv values
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    # Compute gradient via FFT
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def kl_mirror_descent(a, time_budget=100.0, tau_init=10.0, tau_end=500.0, verbose=False):
    """KL mirror descent on the smoothed objective."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 1.0 / np.sqrt(n)
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** progress
        
        g, conv = smoothed_gradient(a, tau)
        
        # Quadratic (HKD) mirror descent update
        # a_new = a * exp(-eta * g) / sum(a * exp(-eta * g))
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - lr * g
        log_a_new -= np.max(log_a_new)  # numerical stability
        a_new = np.exp(log_a_new)
        s = np.sum(a_new)
        if s > 1e-15:
            a_new = a_new / s
        else:
            a_new = a.copy()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stale = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.7
                stall = 0
                if lr < 1e-10:
                    break
            else:
                lr *= 0.999
        
        a = a_new
        it += 1
        
        if verbose and it % 1000 == 0:
            print(f"    KMD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")
    
    return best_a


def euclidean_projected_descent(a, time_budget=100.0, tau_init=10.0, tau_end=500.0, verbose=False):

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, M_target=None, warm_start=None):
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    if warm_start is not None:
        g = cp.Variable(n, nonneg=True, value=warm_start)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue
    
    return None, None


def lp_polish(a, time_budget=200.0, verbose=False, seed=42):
    """LP polishing with golden-section blend search and warm starts."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    warm_g = None
    iter_count = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        
        g_lp, M_lp = solve_lp_minpeak(a, warm_start=warm_g)
        if g_lp is None:
            break
        warm_g = g_lp.copy()
        
        # Golden section search for blend factor
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.01, 1.0
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)
        
        def blend_eval(t):
            at = (1.0 - t) * best_a + t * g_lp
            at = np.maximum(at, 0)
            st = float(np.sum(at))
            if st < 1e-10:
                return 1e10
            at = at / st
            return compute_obj(at)
        
        fc, fd = blend_eval(c), blend_eval(d)
        for _ in range(30):
            if abs(hi - lo) < 0.01:
                break
            if fc < fd:
                hi, d, fd = d, c, fc
                c = hi - gr * (hi - lo)
                fc = blend_eval(c)
            else:
                lo, c, fc = c, d, fd
                d = lo + gr * (hi - lo)
                fd = blend_eval(d)
        
        t_opt = (lo + hi) / 2.0
        a_try = (1.0 - t_opt) * best_a + t_opt * g_lp
        a_try = np.maximum(a_try, 0)
        a_try = a_try / a_try.sum()
        obj_try = compute_obj(a_try)
        
        iter_count += 1
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a.copy()
            no_improve = 0
            if verbose and iter_count % 3 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={t_opt:.3f}, M={M_lp:.6f}, el={elapsed:.0f}s")
        else:
            no_improve += 1
            if no_improve >= 3:
                scale = 0.005 * min(no_improve, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                if iter_count > 40:
                    break
    
    return best_a


def freq_equalize(a, time_budget=60.0, verbose=False):
    """Try to equalize the frequency spectrum to reduce time-domain peaks."""
    n = len(a)

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-8))
        return a / a.sum()
    elif mode == 3:
        freqs = rng.integers(1, 6, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 4:
        # Split into alternating high/low
        a = np.ones(n)
        a[::2] = 2.0
        a[1::2] = 0.5
        return a / a.sum()
    elif mode == 5:
        # Gaussian bumps
        centers = rng.integers(0, n, size=3)
        widths = rng.integers(n // 20, n // 5, size=3)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif mode == 6:
        # Sine-based
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.5 * np.abs(np.sin(x * rng.uniform(1, 5)))
        return a / a.sum()
    else:
        return np.ones(n) / n


def peak_shaving(a, time_budget=60.0, verbose=False):
    """Iteratively shave the peak convolution by reducing contributing elements."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = 0.02
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_k = int(np.argmax(conv))
        max_val = conv[max_k]
        
        # Find elements that contribute most to the peak
        contrib = np.zeros(n)
        for m in range(n):
            idx = max_k - m
            if 0 <= idx < n:
                contrib[m] = a[m] * a[idx]
        
        # Reduce top contributors, redistribute to bottom
        total_mass = float(np.sum(a))
        top_frac = max(1, n // 10)
        bottom_frac = max(1, n // 10)
        
        top_idx = np.argsort(contrib)[::-1][:top_frac]
        bottom_idx = np.argsort(contrib)[:bottom_frac]
        
        # Transfer mass from top to bottom
        transfer = eta * a[top_idx].mean()
        a[top_idx] -= transfer
        a[bottom_idx] += transfer * len(top_idx) / len(bottom_idx)
        
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
        else:
            a = best_a.copy()
            eta *= 0.8
            if eta < 1e-6:
                break
    
    if verbose:
        print(f"    PeakShave: obj={best_obj:.8f}")
    return best_a

"""

Uses KL mirror descent, dual-guided refinement, LP polishing with warm starts,

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient(a, tau):
    """Compute smoothed gradient of log-sum-exp of convolution."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def kl_mirror_descent(a, time_budget=100.0, tau_init=10.0, tau_end=500.0, verbose=False):
    """KL mirror descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 1.5 / np.sqrt(n)
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.8)
        
        g, conv = smoothed_gradient(a, tau)
        
        # KL (quadratic entropy) mirror descent step
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - lr * g
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        s = float(np.sum(a_new))
        if s > 1e-15:
            a_new = a_new / s
        else:
            a_new = a.copy()

            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0

            stall += 1

                stall = 0

def build_conv_matrix(a):
    """Build matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, M_target=None, warm_start=None):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M, dual_y)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    if warm_start is not None:
        g = cp.Variable(n, nonneg=True, value=warm_start)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value) if M.value is not None else None
                    # Try to extract dual variables
                    dual_y = None
                    try:
                        # The dual variables for constraints A@g <= M
                        dual_val = None
                        for constraint in constraints:
                            if hasattr(constraint, 'dual_value') and constraint.dual_value is not None:
                                dv = np.asarray(constraint.dual_value)
                                if dv.shape == (2 * n - 1,):
                                    dual_y = np.maximum(dv, 0)
                                    dual_y = dual_y / dual_y.sum() if dual_y.sum() > 0 else None
                                break
                    except Exception:
                        pass
                    return g_val, M_val, dual_y
            elif prob.status == 'infeasible':
                return None, None, None
        except Exception:
            continue
    
    return None, None, None

def peak_shaving(a, time_budget=60.0, verbose=False):
    """Iteratively reduce the peak convolution by targeted mass transfer."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.03
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_k = int(np.argmax(conv))
        max_val = conv[max_k]
        
        # Find contributing pairs
        contrib = np.zeros(n)
        for m in range(n):
            idx = max_k - m
            if 0 <= idx < n:
                contrib[m] = a[m] * a[idx]
        
        top_frac = max(1, n // 8)
        bot_frac = max(1, n // 8)
        top_idx = np.argsort(contrib)[::-1][:top_frac]
        bot_idx = np.argsort(contrib)[:bot_frac]
        
        transfer = eta * float(np.mean(a[top_idx]))
        a[top_idx] -= transfer
        a[bot_idx] += transfer * len(top_idx) / len(bot_idx)
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            eta = min(eta * 1.05, 0.1)
        else:
            a = best_a.copy()
            eta *= 0.85
            if eta < 5e-4:
                break
    
    if verbose:
        print(f"    PeakShave: obj={best_obj:.8f}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-8)
        return a / a.sum()
    elif mode == 3:
        freqs = rng.integers(1, 6, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        return a / a.sum()
    elif mode == 4:
        a = np.ones(n)
        a[::2] = 2.0
        a[1::2] = 0.4
        return a / a.sum()
    elif mode == 5:
        centers = rng.integers(0, n, size=3)
        widths = rng.integers(max(2, n // 20), max(3, n // 5), size=3)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = x ** rng.uniform(-1, 3) + (1 - x) ** rng.uniform(-1, 3)
        return a / a.sum()
    else:
        return np.ones(n) / n
