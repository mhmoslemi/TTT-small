import numpy as np

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_fast(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g) using stride tricks."""
    n = len(a)
    # Use a sliding window approach
    A_rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            row[i_start:i_end + 1] = a[k - i_end:k - i_start + 1][::-1]
        A_rows.append(row)
    return np.array(A_rows)


def solve_lp_fast(a, warm_g=None, solver=None):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt, active_lags) or (None, None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_fast(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers_to_try = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    if solver:
        solvers_to_try = [solver] + [s for s in solvers_to_try if s != solver]

    for solver in solvers_to_try:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    # Find active constraints
                    conv_check = A @ g_val
                    active = np.where(conv_check >= M_val - 1e-6)[0]
                    return g_val, M_val, active
        except Exception:
            continue
    return None, None, None


def solve_lp_symmetric(a):
    """
    Solve LP with symmetry constraint: g[i] = g[n-1-i].
    This halves the variables and may find better solutions.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    half_n = (n + 1) // 2

    # Symmetric g: g[i] = g[n-1-i]
    # Variables: g_sym[0], ..., g_sym[half_n-1]
    g_sym = cp.Variable(half_n, nonneg=True)
    M = cp.Variable()

    # Build constraints using symmetric g
    # conv(a, g_sym_full)[k] where g_sym_full[i] = g_sym[min(i, n-1-i)]
    constraints = []
    for k in range(2 * n - 1):
        expr = 0
        for i in range(n):
            gi = g_sym[min(i, n - 1 - i)]
            ai = a_norm[min(k - i, n - 1)] if 0 <= k - i < n else 0
            if ai != 0:
                expr += ai * gi
        constraints.append(expr <= M)

    # Sum constraint: sum of symmetric g = 1
    if n % 2 == 0:
        sum_constraint = 2 * cp.sum(g_sym) == 1
    else:
        sum_constraint = 2 * cp.sum(g_sym) - g_sym[half_n - 1] == 1

    constraints.append(sum_constraint)
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g_sym.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_sym_val = np.maximum(g_sym.value, 0)
                # Expand to full length
                g_full = np.zeros(n)
                for i in range(n):
                    g_full[i] = g_sym_val[min(i, n - 1 - i)]
                s = float(np.sum(g_full))
                if s > 1e-10:
                    g_full = g_full / s
                    return g_full, float(M.value)
        except Exception:
            continue
    return None, None


def spectral_flat_init(n, rng):
    """
    Create initialization with flat-ish autocorrelation using iterative spectral method.
    """
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    # Start with flat spectrum
    A = np.ones(N) / N
    # Add some random phase
    phases = rng.uniform(0, 2 * np.pi, N)
    A = np.abs(A) * np.exp(1j * phases)
    # Ensure conjugate symmetry for real signal
    A[N // 2] = np.abs(A[N // 2])
    for k in range(1, N // 2):
        A[N - k] = np.conj(A[k])

    # Iterative projection: alternate between flat magnitude and non-negative time domain
    a = np.zeros(n)
    for it in range(50):
        a = np.real(ifft(A, N)[:n])
        # Enforce non-negativity
        a = np.maximum(a, 0)
        a = a / (np.sum(a) + 1e-15)
        # Update spectrum
        A = fft(a, N)
        # Flatten magnitude (keep only first n components meaningful)
        mag = np.abs(A)
        # Target: flat magnitude
        target_mag = np.ones(N) / np.sqrt(N)
        # Blend: partially flatten
        new_mag = 0.7 * target_mag + 0.3 * mag / (np.max(mag) + 1e-15)
        A = new_mag * np.exp(1j * np.angle(A))
        for k in range(1, N // 2):
            A[N - k] = np.conj(A[k])
        A[N // 2] = np.abs(A[N // 2])

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        # Uniform
        a = np.ones(n) / n
    elif mode == 1:
        # Cosine window
        x = np.arange(n)
        a = np.sin(np.pi * (x + 0.5) / n) ** 2
        a = a / a.sum()
    elif mode == 2:
        # Raised cosine
        x = np.linspace(0, np.pi, n)
        a = (1 + np.cos(x)) / 2
        a = a / a.sum()
    elif mode == 3:
        # Gaussian centered
        x = np.arange(n)
        a = np.exp(-((x - n / 2) ** 2) / (2 * (n / 4) ** 2))
        a = a / a.sum()
    elif mode == 4:
        # Spectral flat
        a = spectral_flat_init(n, rng)
    elif mode == 5:
        # Two-peak (anti-symmetric tendency)
        x = np.arange(n)
        sigma = n / 6
        a = np.exp(-((x - n * 0.25) ** 2) / (2 * sigma ** 2)) + \
            np.exp(-((x - n * 0.75) ** 2) / (2 * sigma ** 2))
        a = a / a.sum()
    elif mode == 6:
        # Random
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 7:
        # Symmetric cosine
        x = np.arange(n)
        center = (n - 1) / 2
        a = np.cos(np.pi * (x - center) / (n + 1)) ** 2
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 8:
        # Bernoulli-like (sparse)
        density = 0.7
        mask = rng.random(n) < density
        a = np.where(mask, 1.0, 0.0)
        if a.sum() == 0:
            a = np.ones(n)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

def smoothed_max_grad(a, tau, top_k=50):
    """
    Compute gradient of smoothed max of conv(a,a) using top-K selection.
    More efficient than full softmax for large n.
    """
    n = len(a)
    conv = compute_conv(a)

    # Use top-k selection for efficiency
    if 2 * n - 1 > top_k:
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        conv_top = conv[top_idx]
        max_top = conv_top.max()
        exp_vals = np.exp(tau * (conv_top - max_top))
        weights = exp_vals / exp_vals.sum()
    else:
        top_idx = np.arange(len(conv))
        max_conv = conv.max()
        exp_vals = np.exp(tau * (conv - max_conv))
        weights = exp_vals / exp_vals.sum()

    # Gradient: grad_i = 2 * sum_k w_k * a[k-i]
    # Use FFT for efficiency
    N = 1
    while N < 3 * n:
        N <<= 1

    p = np.zeros(N)
    p[top_idx] = weights
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    grad = 2.0 * g_full[:n]

    return grad, conv


def nesterov_smoothed_descent(a, time_budget, tau_init=20.0, tau_end=500.0, top_k=50, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.95))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)

        if verbose and it % 2000 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_max_grad(y, tau, top_k)

        # Barzilai-Borwein step size adaptation
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.4 * lr + 0.6 * min(max(bb_lr, 1e-8), 20.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)

        # Reset if no improvement for too long
        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 5.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def peak_shave_advanced(a, time_budget, K=30, verbose=False):

    n = len(a)

    start = time.monotonic()
    lr = 0.5
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Get top K peaks
        K_eff = min(K, 2 * n - 1)
        top_idx = np.argpartition(conv, -K_eff)[-K_eff:]

        # Weighted subgradient
        peak_vals = conv[top_idx]
        max_pv = peak_vals.max()
        weights = np.exp(5.0 * (peak_vals - max_pv) / (max_pv + 1e-15))
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):

            i_start = max(0, k - (n - 1))

                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        a_new = a - lr * g / g_norm
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.02, 2.0)
        else:
            stall += 1
            lr *= 0.7
            if stall > 50:
                lr = 0.5 / np.sqrt(n)
                a = best_a.copy()
                stall = 0
            if lr < 1e-10:
                break

        it += 1
        if verbose and it % 500 == 0:
            print(f"    PS it{it}: obj={best_obj:.8f}, lr={lr:.4f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation with reflection."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    # Extend with reflection for better boundary handling
    a_ext = np.concatenate([a, a[::-1]])
    total = len(a_ext)
    x_old = np.linspace(0, 1, total)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a_ext)
    return np.maximum(a_new, 0)


def quadratic_polish(a, time_budget, n_inner=5, verbose=False):
    """
    Quadratic polish: minimize a quadratic approximation of the objective.
    Uses the Hessian of the max-conv term at the current point.
    """

        if len(top_indices) == 0:
            top_indices = np.array([np.argmax(conv)])

        # Build quadratic form for the top peaks
        # For each peak k, conv(a,a)[k] = a^T B_k a where B_k is the "circulant" matrix
        # Approximate with top-3 peaks

        # Compute gradient and approximate Hessian
        g = np.zeros(n)
        H = np.zeros((n, n))
        for k in top3:

            # Actually for conv(a,a)[k] = sum_i a[i]*a[k-i]:
            # d/d a[m] = sum_i (delta_{m,i} a[k-i] + a[i] delta_{m,k-i})
            #           = a[k-m] + a[m] (when valid)
            # Hessian: H[m,l] = delta_{m+k-l} + delta_{l+k-m} (roughly)
            # For simplicity, use diagonal approximation
            for m in range(n):
                km = k - m
                if 0 <= km < n:
                    g[m] += a[km]

        # Scale gradient
        g /= len(top3) + 1e-15
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Adaptive step with backtracking
        step = 1.0 / (g_norm * np.sqrt(n))
        for _ in range(n_inner):
            a_try = a - step * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                break
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                break
            step *= 0.5

        a = best_a
        if obj_try >= best_obj - 1e-13:
            # Perturb
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function.
    Strategy:
    - Phase 1 (20%): Diversified initialization + quick LP across multiple lengths
    - Phase 2 (45%): Intensive LP polishing on top candidates
    - Phase 3 (15%): Smoothed gradient fine-tuning
    - Phase 4 (20%): Final polish (peak shave + gradient + LP)
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0  # 30s buffer for safety

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Diversified Initialization + Quick LP (20%) ===
    print(f"\n=== Phase 1: Diversified Init + Quick LP ===")
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.20

    candidates = [(best_seq.copy(), best_score, len(best_seq))]

    # Try different lengths and initializations
    n_options = sorted(set([
        500, 600, 700, 800, 900, 1000, 1200, 1400
    ]))
    n_options = [n for n in n_options if n <= 1500]

    modes = [0, 1, 2, 3, 4, 5, 7]
    init_per_n = max(1, int(p1_budget * 0.7 / (len(n_options) * len(modes) * 2.0)))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        for mode in modes[:max(1, len(modes) // len(n_options))]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            a_init = diverse_init(n_try, rng, mode)
            obj_init = compute_obj(a_init)
            candidates.append((a_init, obj_init, n_try))

            # Quick LP (2-3 iterations)
            if remaining > 10:
                a_lp = lp_polish_v2(a_init, time_budget=min(5.0, remaining * 0.3),
                                     verbose=False)
                obj_lp = compute_obj(a_lp)
                candidates.append((a_lp, obj_lp, n_try))

    # Sort by score and keep top 5
    candidates.sort(key=lambda x: x[1])
    print(f"  Top 5 candidates after Phase 1:")
    for i, (a_c, s_c, n_c) in enumerate(candidates[:5]):
        print(f"    #{i+1}: n={n_c}, obj={s_c:.8f}")

    if len(candidates) > 0 and candidates[0][1] < best_score:
        best_seq = candidates[0][0].copy()
        best_score = candidates[0][1]
        n_target = candidates[0][2]

    # === Phase 2: Intensive LP Polishing (45%) ===
    print(f"\n=== Phase 2: Intensive LP ===")
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.45

    # Work on top 3 candidates
    top_candidates = candidates[:3]
    if not top_candidates:
        top_candidates = [(best_seq.copy(), best_score, n_target)]

    per_candidate_time = p2_budget / max(1, len(top_candidates))

    for idx, (a_c, s_c, n_c) in enumerate(top_candidates):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        alloc = min(per_candidate_time, remaining * 0.6)
        if alloc < 10:
            break

        print(f"  Polishing candidate #{idx+1} (n={n_c}, obj={s_c:.8f}), budget={alloc:.0f}s")

        # Multiple rounds with perturbation
        num_rounds = max(2, int(alloc / 60))
        per_round = alloc / num_rounds

        for rnd in range(num_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc_r = min(per_round, remaining * 0.5)
            if alloc_r < 5:
                break

            use_sym = (rnd % 3 == 2) and n_c <= 1000  # Try symmetric every 3rd round
            a_pol = lp_polish_v2(a_c, time_budget=alloc_r, verbose=(rnd == 0),
                                 use_symmetric=use_sym)
            obj_pol = compute_obj(a_pol)

            if obj_pol < s_c - 1e-12:
                s_c = obj_pol
                a_c = a_pol
                if obj_pol < best_score:
                    best_score = obj_pol
                    best_seq = a_pol.copy()
                    n_target = n_c
                    print(f"    ** New best: obj={best_score:.8f}, n={n_c}")
            else:
                # Perturb for next round
                if rnd < num_rounds - 1:
                    scale = 0.01 * (rnd + 1)
                    a_c = a_c * np.exp(rng.normal(0, scale, n_c))
                    a_c = np.maximum(a_c, 0)
                    a_c = a_c / a_c.sum()

    # === Phase 3: Smoothed Gradient (15%) ===
    print(f"\n=== Phase 3: Smoothed Gradient ===")
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.15

    if p3_budget > 15:
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=p3_budget,
                                          tau_init=30.0, tau_end=500.0,
                                          top_k=80, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase3 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 GD: obj={obj_sgd:.8f}")

    # === Phase 4: Final Polish (20%) ===
    print(f"\n=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.20

    # 4a: Peak shaving
    if p4_budget > 15:
        alloc = p4_budget * 0.3
        a_ps = peak_shave_advanced(best_seq, time_budget=alloc, K=50, verbose=True)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  PeakShave: obj={best_score:.8f} **")
        else:
            print(f"  PeakShave: obj={obj_ps:.8f}")

    # 4b: Quadratic polish
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 60.0)
        a_qp = quadratic_polish(best_seq, time_budget=alloc, verbose=True)
        obj_qp = compute_obj(a_qp)
        if obj_qp < best_score:
            best_score = obj_qp
            best_seq = a_qp.copy()
            print(f"  QuadPolish: obj={best_score:.8f} **")
        else:
            print(f"  QuadPolish: obj={obj_qp:.8f}")

    # 4c: Final LP
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.4, 80.0)
        a_lp = lp_polish_v2(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_lp:.8f}")

    # 4d: High-tau gradient
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.4, 50.0)
        a_sgd2 = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                           tau_init=100.0, tau_end=1000.0,
                                           top_k=100)
        obj_sgd2 = compute_obj(a_sgd2)
        if obj_sgd2 < best_score:
            best_score = obj_sgd2
            best_seq = a_sgd2.copy()
            print(f"  Final GD: obj={best_score:.8f} **")
        else:
            print(f"  Final GD: obj={obj_sgd2:.8f}")

    # 4e: One more LP
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.5, 40.0)
        a_lp2 = lp_polish_v2(best_seq, time_budget=alloc)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
