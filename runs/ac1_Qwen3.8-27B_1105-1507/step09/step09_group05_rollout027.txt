Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequence a.
Uses LP-direction iteration with golden-section line search for step size,
combined with smoothed gradient descent for initialization and multi-n exploration.
Key improvement over previous: proper line search (exploiting convexity of
the objective in the blending parameter) and more aggressive LP iteration.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_conv_cross(a, b):
    """Compute linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def smoothed_gradient(a, tau):
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    # Clip to avoid overflow
    shifted = np.minimum(tau * (conv - max_conv), 50.0)
    exp_vals = np.exp(shifted)
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g


def nesterov_smoothed_descent(a, time_budget=100.0, tau_init=5.0, tau_end=300.0, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget * 0.9, 1e-6))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)

        if it % 10000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)

        if prev_g is not None and it % 3 == 0 and it > 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1

        # Adaptive learning rate decay
        lr = max(1e-8, lr * 0.9995)

        # Reset if stuck
        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a

def line_search_convex(C, D, E, t_lo=0.0, t_hi=1.0):
    """
    Minimize F(t) = max_k [(1-t)^2 C_k + 2t(1-t) D_k + t^2 E_k]
    over t in [t_lo, t_hi], where F is convex.
    Uses golden section search.
    """
    def F(t):
        vals = (1-t)**2 * C + 2*t*(1-t) * D + t**2 * E
        return float(np.max(vals))
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    a_t = t_lo
    b_t = t_hi
    c = b_t - gr * (b_t - a_t)
    d = a_t + gr * (b_t - a_t)
    fc = F(c)
    fd = F(d)
    
    for _ in range(50):  # 50 iterations gives precision ~ (2/phi)^50 * (b-a) ≈ 1e-10
        if fc < fd:
            b_t = d
            d = c
            fd = fc
            c = b_t - gr * (b_t - a_t)
            fc = F(c)
        else:
            a_t = c
            c = d
            fc = fd
            d = a_t + gr * (b_t - a_t)
            fd = F(d)
    
    t_opt = (a_t + b_t) / 2
    return t_opt, F(t_opt)

def lp_iteration(a, time_budget, verbose=False):
    """
    LP-direction iteration with line search.
    """
    n = len(a)
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    no_improve = 0
    
    # Precompute A matrix
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a  # This changes with a! Need to rebuild each iteration.
    
    while time.monotonic() - start < time_budget:
        # Rebuild A with current a
        A = np.zeros((2*n-1, n))
        for j in range(n):
            A[j:j+n, j] = a
        
        # Solve LP
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        solved = False
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        g_val = g_val / s_g
                        solved = True
                        break
            except:
                continue
        
        if not solved:
            break
        
        # Precompute convolutions for line search
        C = compute_conv(a)  # conv(a, a)
        D = compute_conv_cross(a, g_val)  # conv(a, g)
        E = compute_conv(g_val)  # conv(g, g)
        
        # Line search
        t_opt, _ = line_search_convex(C, D, E)
        
        # Compute new a
        a_new = (1 - t_opt) * a + t_opt * g_val
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-direction iteration
with golden-section line search (exploiting convexity of the blended
objective in the step parameter), combined with smoothed gradient
descent warm-start and multi-length exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def golden_search(C, D, E, t_lo=0.0, t_hi=1.0, max_iter=40):
    """
    Minimize F(t) = max_k [(1-t)^2 C_k + 2t(1-t) D_k + t^2 E_k]
    over t in [t_lo, t_hi]. F is convex in t.
    """
    def F(t):
        one_minus_t = 1.0 - t
        return float(np.max(one_minus_t**2 * C + 2.0 * t * one_minus_t * D + t**2 * E))
    
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    a_g = t_lo
    b_g = t_hi
    c = b_g - inv_phi * (b_g - a_g)
    d = a_g + inv_phi * (b_g - a_g)
    fc = F(c)
    fd = F(d)
    
    for _ in range(max_iter):
        if fc < fd:
            b_g = d
            d = c
            fd = fc
            c = b_g - inv_phi * (b_g - a_g)
            fc = F(c)
        else:
            a_g = c
            c = d
            fc = fd
            d = a_g + inv_phi * (b_g - a_g)
            fd = F(d)
    
    t_opt = (a_g + b_g) / 2.0
    return t_opt, F(t_opt)


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. conv(a, g)_k <= M for all k, sum(g) = 1, g >= 0.
    Returns g_optimal or None.
    """
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def lp_iterate(a, time_budget, verbose=False):
    """
    LP-direction iteration with golden-section line search.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf, 0
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        g = solve_lp_direction(a)
        if g is None:
            break
        
        C = compute_conv(a)
        D = compute_conv_cross(a, g)
        E = compute_conv(g)
        
        t_opt, _ = golden_search(C, D, E)
        
        if t_opt < 1e-6:
            # No improvement possible in this direction
            no_improve += 1
            if no_improve >= 3:
                break
            # Try a tiny step to perturb
            t_opt = 1e-4
        else:
            no_improve = 0
        
        a_new = (1.0 - t_opt) * a + t_opt * g
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        it += 1
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}")
        else:

            # (the LP direction for a_new might be different)
            a = a_new
            no_improve += 1
            if no_improve >= 5:
                a = best_a.copy()
                break
    
    return best_a, best_obj, it


def smoothed_gradient(a, tau):
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = np.minimum(tau * (conv - max_conv), 50.0)
    exp_vals = np.exp(shifted)
    p = exp_vals / float(np.sum(exp_vals))
    from scipy.fft import fft, ifft
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def smoothed_descent(a, time_budget=60.0, tau_init=5.0, tau_end=200.0):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget * 0.8, 1e-6))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)
        
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        g = smoothed_gradient(y, tau)
        
        if prev_g is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 8.0 / np.sqrt(n))
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.999)
        
        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it
    
    return best_a


def perturb(a, rng, scale=0.03):
    n = len(a)
    method = rng.integers(0, 5)
    a = a.copy()
    if method == 0:
        a *= np.exp(rng.normal(0, scale, n))
    elif method == 1:
        idx = rng.choice(n, size=n // 5, replace=False)
        a[idx] *= rng.uniform(0.5, 1.5, len(idx))
    elif method == 2:
        shift = rng.integers(1, max(2, n // 20))
        a = np.roll(a, int(shift))
    elif method == 3:
        a = (1 - scale) * a + scale / n
    else:
        mask = rng.random(n) > 0.05
        a *= mask
    
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return np.maximum(a, 0) / float(np.sum(np.maximum(a, 0)))
