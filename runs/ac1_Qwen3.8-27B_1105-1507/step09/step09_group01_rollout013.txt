def lbnb_smoothed(a_init, tau_schedule, time_budget):
    """L-BFGS-B optimization on simplex with smoothed max objective."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for tau in tau_schedule:
        if time.monotonic() - start > time_budget * 0.9:
            break
        
        # Define smoothed objective on the simplex (parameterized by log transform or just use B bounds)
        def obj_func(x):
            x = np.maximum(x, 0)
            s = x.sum()
            if s < 1e-10:
                return 1e10
            x = x / s
            conv = compute_conv(x)
            max_c = conv.max()
            shifted = tau * (conv - max_c)
            lse = max_c + np.log(np.sum(np.exp(shifted))) / tau
            return 2.0 * n * lse
        
        def grad_func(x):
            x = np.maximum(x, 0)
            s = x.sum()
            if s < 1e-10:
                return np.zeros_like(x)
            x = x / s
            conv = compute_conv(x)
            max_c = conv.max()
            shifted = tau * (conv - max_c)
            p = np.exp(shifted)
            p /= p.sum()
            
            # Gradient of 2n * LSE(tau*conv) / tau with respect to x
            # = 2n * sum_k p_k * d(conv_k)/dx
            # d(conv_k)/dx_i = 2 * (cross-corr of x with delta_k)[i] approximately
            # Using FFT:

            P = fft(p, N)
            X = fft(x, N)
            g = ifft(P * np.conj(X)).real[:n]
            return 2.0 * n * 2.0 * g  # factor of 2 from quadratic form
        
        # Use L-BFGS-B with bounds [0, inf)
        result = minimize(obj_func, a, jac=grad_func, method='L-BFGS-B',

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS-B on log-sum-exp smoothed objective for fast local optimization,
combined with LP-based direction finding, spectral shaping initialization,

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft, next_fast_len
from scipy.optimize import minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = next_fast_len(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_set = np.where(cond)[0]
    if len(rho_set) == 0:
        return np.ones(n) / n
    rho = rho_set[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_grad_fft(a, p):
    """Compute gradient of 2n * LSE(tau*conv)/tau given softmax weights p."""
    n = len(a)
    N = next_fast_len(3 * n)
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def lbnb_smoothed_optimize(a_init, time_budget=60.0, tau_values=None, verbose=False):
    """
    L-BFGS-B optimization with increasing tau (log-sum-exp smoothing).
    Works on unnormalized a (objective is scale-invariant).
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    if tau_values is None:
        tau_values = [5, 15, 40, 100, 250, 500, 1000]

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for tau in tau_values:
        if time.monotonic() - start > time_budget * 0.92:
            break

        # Time allocation per tau
        tau_time = min((time_budget * 0.9 - (time.monotonic() - start)), time_budget / len(tau_values) * 2)
        if tau_time < 2:
            break

        def obj_func(x):
            x = np.maximum(x, 0)
            sx = float(np.sum(x))
            if sx < 1e-10:
                return 1e10
            x = x / sx
            conv = compute_conv(x)
            max_c = float(np.max(conv))
            shifted = tau * (conv - max_c)
            # Clip for numerical stability
            shifted = np.clip(shifted, -500, 500)
            lse = max_c + float(np.log(np.sum(np.exp(shifted)))) / tau
            return 2.0 * n * lse

        def grad_func(x):
            x = np.maximum(x, 0)
            sx = float(np.sum(x))
            if sx < 1e-10:
                return np.zeros_like(x)
            x = x / sx
            conv = compute_conv(x)
            max_c = float(np.max(conv))
            shifted = tau * (conv - max_c)
            shifted = np.clip(shifted, -500, 500)
            p = np.exp(shifted)
            p_sum = float(np.sum(p))
            if p_sum < 1e-300:
                p = np.zeros_like(conv)
                p[np.argmax(conv)] = 1.0
            else:
                p = p / p_sum
            # Gradient of 2n * LSE(tau*conv)/tau = 2n * sum_k p_k * d(conv_k)/dx
            # d(conv_k)/dx_i = 2 * a_{k-i} (approximately)
            g = smoothed_grad_fft(x, p)
            return 2.0 * n * g

        max_iter = max(50, int(n / 2))
        try:
            result = minimize(
                obj_func, a, jac=grad_func, method='L-BFGS-B',
                bounds=[(0, None)] * n,
                options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                         'maxls': 50, 'maxfun': max_iter * 10}
            )
            if result.x is not None:

                        best_a = a_new.copy()
                    a = a_new
                    if verbose:
                        print(f"    L-BFGS tau={tau}: obj={best_obj:.8f}")
        except Exception as e:
            if verbose:
                print(f"    L-BFGS tau={tau} failed: {e}")

def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j, j] = a_norm[n - 1 - j] if n - 1 - j >= 0 else 0
    # Actually, let's build it properly.
    # conv(a, g)_k = sum_j a_j * g_{k-j}
    # For a fixed a, this is linear in g: (A @ g)_k = sum_j M_{k,j} * g_j
    # M_{k,j} = a_{k-j} if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a_norm[idx]
    return A


def build_conv_matrix_fast(a_norm):
    """Build convolution matrix efficiently using structure."""
    n = len(a_norm)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # A[k, j] = a_norm[k-j] when 0 <= k-j < n
        # k ranges from j to j+n-1
        k_start = j
        k_end = min(j + n, m)
        indices = np.arange(k_start, k_end)
        A[indices, j] = a_norm[indices - j]
    return A


def solve_lp_minpeak(a, M_target=None, verbose=False):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_fast(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue

    return None, None

def subgradient_refine(a, time_budget=60.0, K=8, verbose=False):
    """Subgradient method with top-K peaks and adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.5 / np.sqrt(n)
    momentum = 0.6
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break

        # Soft weighting of peaks
        temp = max_pv * 0.1
        weights = np.exp((peak_vals - max_pv) / temp)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            # d(conv_k)/da_i = 2 * a_{k-i} for valid i
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * 2.0 * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.5
                v *= 0.2
                stall = 0
                if lr < 1e-12:
                    break
            else:
                lr *= 0.997

        it += 1

    return best_a


def spectral_init(n, rng, mode=0):
    """Generate sequences with spectral shaping for good initial autocorrelation properties."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        # Uniform
        a = np.ones(n) / n
    elif mode == 1:
        # Raised cosine (good spectral rolloff)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif mode == 2:
        # Exponential decay (one-sided)
        a = np.exp(-3.0 * x)
    elif mode == 3:
        # Double exponential (symmetric)
        a = np.exp(-3.0 * np.abs(x - 0.5) * 2)
    elif mode == 4:
        # Prolate-like (concentrated in time)
        a = 1.0 / (1.0 + 10.0 * (x - 0.5) ** 2)
    elif mode == 5:
        # Sine-modulated
        f = rng.integers(1, 4)
        a = 0.5 + 0.4 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
    elif mode == 6:
        # Spectrally white (random with flat spectrum)
        f = np.random.default_rng(rng.integers(0, 2**31))
        A = f.standard_normal(n) + 1j * f.standard_normal(n)
        A[0] = np.abs(A[0])  # Make real
        A[-1] = np.abs(A[-1]) if n > 1 else A[0]
        if n > 1:
            for i in range(1, n // 2):
                A[n - i] = np.conj(A[i])
        a = np.abs(ifft(A)) + 0.1
    elif mode == 7:
        # Gaussian bump
        c = 0.3 + 0.4 * rng.random()
        w = 0.05 + 0.15 * rng.random()
        a = np.exp(-(x - c) ** 2 / (2 * w ** 2))
    elif mode == 8:
        # Multi-bump
        n_bumps = rng.integers(2, 5)
        a = np.zeros(n)
        for _ in range(n_bumps):
            c = rng.uniform(0.1, 0.9)
            w = rng.uniform(0.03, 0.15)
            a += np.exp(-(x - c) ** 2 / (2 * w ** 2))
    elif mode == 9:
        # Inverse raised cosine (concentrated in middle)
        a = 1.0 / (0.3 + 0.7 * np.cos(np.pi * (2 * x - 1)) ** 2)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    # Use numpy interp (linear) for speed, or spline for quality
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def golden_section_min(f, a_lo, a_hi, max_iter=30, tol=0.01):
    """Golden section search for minimum."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if abs(a_hi - a_lo) < tol:
            break
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = f(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = f(d)
    return (a_lo + a_hi) / 2.0


def spectral_shaping_step(a, time_budget=30.0):
    """
    One step of spectral shaping: reduce spectral components that contribute
    most to the autocorrelation peak.
    """
    n = len(a)
    a = a / np.sum(a)

    conv = compute_conv(a)
    peak_idx = int(np.argmax(conv))

    # Compute gradient direction in spectral domain
    # The peak is at lag peak_idx, which corresponds to frequency 0 in the
    # cross-correlation sense. We want to reduce the components at peak_idx.
    
    # Compute FFT of a
    N = next_fast_len(2 * n)
    A = fft(a, N)

    # The autocorrelation at lag k is ifft(|A|^2)[k] (circular, with padding)
    # To reduce the peak at peak_idx, we reduce |A|^2 at the frequencies
    # that contribute most to that lag.
    
    # Simple approach: reduce the overall spectral peak
    # Find the frequency with maximum |A|^2 relative to the average
    mag = np.abs(A)
    mag_avg = mag.mean()
    
    # Suppress the top few spectral peaks
    top_k = max(3, n // 20)
    top_freqs = np.argsort(mag)[::-1][:top_k]
    
    # Reduce those frequencies
    A_new = A.copy()
    for f_idx in top_freqs:
        A_new[f_idx] *= 0.85
        # Maintain Hermitian symmetry for real signal
        if f_idx < N - f_idx and f_idx != 0:
            A_new[N - f_idx] = np.conj(A_new[f_idx])
    
    a_new = np.real(ifft(A_new, N)[:n])
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    
    return a_new


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining L-BFGS-B smoothed optimization,
    LP polishing, spectral shaping, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_base = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_base = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_base) / n_base
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_base}): obj={best_score:.8f}")

    # === Phase 1: L-BFGS-B Smoothed Optimization ===
    print(f"\n=== Phase 1: L-BFGS-B Smoothed Optimization ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.2, 180.0)
    if p1_budget > 20:
        a_opt = lbnb_smoothed_optimize(best_seq, time_budget=p1_budget,
                                        tau_values=[5, 15, 40, 100, 250, 500],
                                        verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase1 L-BFGS: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 L-BFGS: obj={obj_opt:.8f}")

    # === Phase 2: LP Polishing ===
    print(f"\n=== Phase 2: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.3, 300.0)
    if p2_budget > 30:
        a_lp = lp_polish_improved(best_seq, time_budget=p2_budget, verbose=True, seed=42)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f}")

    # === Phase 3: L-BFGS-B again (after LP) ===
    print(f"\n=== Phase 3: L-BFGS-B Refine ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 120.0)
    if p3_budget > 15:
        a_opt2 = lbnb_smoothed_optimize(best_seq, time_budget=p3_budget,
                                         tau_values=[50, 150, 400, 1000, 2000],
                                         verbose=True)
        obj_opt2 = compute_obj(a_opt2)
        if obj_opt2 < best_score:
            best_score = obj_opt2
            best_seq = a_opt2.copy()
            print(f"  Phase3 L-BFGS: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 L-BFGS: obj={obj_opt2:.8f}")

    # === Phase 4: Spectral Shaping + Re-optimize ===
    print(f"\n=== Phase 4: Spectral Shaping ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_spec = a.copy() if False else best_seq.copy()
        for step in range(3):
            a_spec = spectral_shaping_step(a_spec, time_budget=5.0)
            # Quick L-BFGS after each shaping step
            alloc = min(20.0, remaining * 0.1)
            if alloc > 5:
                a_spec = lbnb_smoothed_optimize(a_spec, time_budget=alloc,
                                                 tau_values=[50, 200, 500])
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_score:
                best_score = obj_spec
                best_seq = a_spec.copy()
                print(f"  Phase4 spec step{step}: obj={best_score:.8f} **")
            else:
                print(f"  Phase4 spec step{step}: obj={obj_spec:.8f}")
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

    # === Phase 5: Multi-length Exploration ===
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Generate diverse n options
    n_options = set()
    for delta in [-300, -200, -100, 100, 200, 300, 500, 800]:
        n_try = n_current + delta
        if 400 <= n_try <= 2500:
            n_options.add(n_try)
    # Also add some specific values
    for n_spec in [500, 700, 900, 1000, 1100, 1300, 1500, 1800, 2000]:
        if abs(n_spec - n_current) > 50:
            n_options.add(n_spec)

    n_options = sorted(n_options)
    n_tried = 0

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80 or n_tried >= 8:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        n_tried += 1

        # Resample and try different inits
        best_for_n = None
        best_obj_n = np.inf

        for init_mode in [0, 1, 3, 8]:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            if init_mode == 0:
                a_try = resample_sequence(best_seq, n_try)
            else:
                a_try = spectral_init(n_try, rng, mode=init_mode)

            # Quick L-BFGS
            alloc1 = min(remaining * 0.2, 30.0)
            if alloc1 > 5:
                a_try = lbnb_smoothed_optimize(a_try, time_budget=alloc1,
                                                tau_values=[20, 80, 300])

            # Quick LP
            remaining = deadline - time.monotonic()
            alloc2 = min(remaining * 0.2, 50.0)
            if alloc2 > 15:
                a_try = lp_polish_improved(a_try, time_budget=alloc2, seed=n_try + init_mode)

            obj_try = compute_obj(a_try)
            if obj_try < best_obj_n:
                best_obj_n = obj_try
                best_for_n = a_try.copy()

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_current = n_try
                print(f"  ** n={n_try} mode={init_mode}: obj={best_score:.8f} **")

        if best_for_n is not None and best_obj_n < best_score + 0.01:
            # Do a final polish on the best for this n
            remaining = deadline - time.monotonic()
            if remaining > 30:
                alloc = min(remaining * 0.3, 80.0)
                a_final = lbnb_smoothed_optimize(best_for_n, time_budget=alloc,
                                                  tau_values=[100, 400, 1000])
                obj_final = compute_obj(a_final)
                if obj_final < best_score:
                    best_score = obj_final
                    best_seq = a_final.copy()
                    print(f"  ** n={n_try} final: obj={best_score:.8f} **")

    # === Phase 6: Perturbation + LP restarts ===
    print(f"\n=== Phase 6: Perturbation + Re-LP ===")
    remaining = deadline - time.monotonic()

    for r in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        scale = 0.008 + 0.004 * r
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        # Quick LP
        alloc = min(remaining * 0.2, 80.0)
        if alloc > 15:
            a_pert = lp_polish_improved(a_pert, time_budget=alloc, seed=200 + r)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Perturb {r} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                if r % 3 == 0:
                    print(f"  Perturb {r} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        # L-BFGS with high tau
        alloc = min(remaining * 0.4, 150.0)
        a_final = lbnb_smoothed_optimize(best_seq, time_budget=alloc,
                                          tau_values=[200, 500, 1000, 2000, 4000],
                                          verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final L-BFGS: obj={best_score:.8f} **")
        else:
            print(f"  Final L-BFGS: obj={obj_final:.8f}")

        # Final LP
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.4, 100.0)
            a_final2 = lp_polish_improved(best_seq, time_budget=alloc, seed=999)
            obj_final2 = compute_obj(a_final2)
            if obj_final2 < best_score:
                best_score = obj_final2
                best_seq = a_final2.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        # Final subgradient
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.3, 50.0)
            a_sg = subgradient_refine(best_seq, time_budget=alloc, K=10)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  Final SG: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    
    # Validate with evaluate_sequence
    try:
        val = evaluate_sequence(result)
        if val < np.inf:
            if val < best_score:
                best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
