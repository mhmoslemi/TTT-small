"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS-B on log-sum-exp smoothed objective, alternating LP with line search,
active set balancing, and systematic multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_obj_and_grad(a, tau):
    """
    Compute the log-sum-exp smoothed objective and its gradient.
    Objective: (1/tau) * log(sum_k exp(tau * conv_k(a))) where conv_k = conv(a,a)_k
    We work with normalized a (sum=1) so the full objective is 2*n * smoothed_max.
    
    Returns (value, gradient) where value is the smoothed max conv value
    and gradient is the gradient w.r.t. a (before the 2n factor).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    # Log-sum-exp for numerical stability
    shifted = tau * (conv - max_conv)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
        smooth_val = max_conv
    else:
        p = exp_vals / sum_exp
        smooth_val = max_conv + np.log(sum_exp) / tau
    
    # Gradient: d/d(a_j) of sum_k p_k * conv_k = 2 * sum_k p_k * a_{k-j}
    # Where a_{k-j} is a[k-j] if 0<=k-j<n else 0
    # This is 2 * (p correlated with a), i.e., 2 * cross-correlation
    
    # Using FFT: the gradient is 2 * ifft(fft(p) * conj(fft(a)))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    grad = 2.0 * g_full[:n]
    
    # The full scaled objective is 2*n * smooth_val
    # Its gradient is 2*n * grad
    return 2.0 * n * smooth_val, 2.0 * n * grad


def lbgfss_smoothed(a, tau_schedule=None, time_budget=100.0, verbose=False):

    Run L-BFGS-B on the smoothed objective with a temperature schedule.
    The sequence is normalized to sum=1 before optimization.
    """

    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if tau_schedule is None:
        tau_schedule = [(10.0, 0.2), (40.0, 0.2), (100.0, 0.2), (300.0, 0.2), (800.0, 0.2)]
    
    for tau, frac_time in tau_schedule:
        t_budget = time_budget * frac_time
        t_start = time.monotonic()
        
        def obj_fun(x):
            x = np.maximum(x, 0)
            s = float(np.sum(x))
            if s < 1e-10:
                return 1e10
            x = x / s
            val, _ = smoothed_obj_and_grad(x, tau)
            return val
        
        def obj_grad_fun(x):
            x = np.maximum(x, 0)
            s = float(np.sum(x))
            if s < 1e-10:
                return 1e10, np.ones(n)
            x = x / s
            val, grad = smoothed_obj_and_grad(x, tau)
            return val, grad
        
        result = minimize(obj_grad_fun, a, jac=obj_grad_fun, method='L-BFGS-B',
                         options={'maxiter': 5000, 'gtol': 1e-10, 'maxfun': 10000,
                                  'iprint': -1})
        
        if result.x is not None:
            a_new = np.maximum(result.x, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                a = a_new
        
        if verbose:
            elapsed = time.monotonic() - t_start
            print(f"      L-BFGS-B tau={tau}: obj={best_obj:.8f} (time={elapsed:.1f}s)")
        
        if time.monotonic() - start >= time_budget:

_g_tau = 10.0

def fft_conv(a, b=None):
    """Full linear convolution a*b via FFT."""
    if b is None:
        b = a
    n1, n2 = len(a), len(b)
    size = 1
    while size < n1 + n2 - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n1 + n2 - 1]


def compute_conv_auto(a):
    """Full auto-convolution a*a via FFT. Length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_auto(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax_weights(conv, tau):
    """Compute softmax weights p_k = exp(tau*conv_k) / sum exp(tau*conv_j)."""
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    return p, max_conv

    """Compute (1/tau)*log(sum exp(tau*conv_k))."""
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    sum_exp = float(np.sum(np.exp(shifted)))
    if sum_exp < 1e-300:
        return max_conv
    return max_conv + np.log(sum_exp) / tau

    smooth_val = max_conv + np.log(float(np.sum(np.exp(tau * (conv - max_conv))))) / tau
    
    # Gradient: 2 * cross_corr(p, a)

    N = 1

    A = fft(a, N)

    # In matrix form: A[k, j] = a[k-j] if 0<=k-j<n, else 0.

    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP, cp.CBC, cp.GLOP]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(M.value)
        except Exception:
            continue
    
    return None, None
