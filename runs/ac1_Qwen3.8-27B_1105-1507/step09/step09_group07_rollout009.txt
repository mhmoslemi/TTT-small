   """
   Enhanced LP-alternating optimization for minimizing peak autocorrelation
   of non-negative step functions. Uses multiple restarts, n-sweep, and
   adaptive step sizes to break through local optima.
   """
   
   import numpy as np
   import time
   import cvxpy as cp
   from scipy.fft import rfft, irfft
   
   def compute_conv(a):
       """Full linear convolution a*a via FFT."""
       n = len(a)
       if n == 0:
           return np.array([])
       size = 1
       while size < 2 * n - 1:
           size <<= 1
       fa = rfft(a, size)
       return np.maximum(irfft(fa * fa, size)[:2 * n - 1], 0)
   
   def compute_conv_cross(a, b):
       """Cross convolution a*b via FFT."""
       n = len(a)
       m = len(b)
       size = 1
       while size < n + m - 1:
           size <<= 1
       fa = rfft(a, size)
       fb = rfft(b, size)
       return np.maximum(irfft(fa * fb, size)[:n + m - 1], 0)
   
   def compute_obj(a):
       """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
       n = len(a)
       if n == 0:
           return np.inf
       s = float(np.sum(a))
       if s < 0.01:
           return np.inf
       a_norm = a / s
       conv = compute_conv(a_norm)
       return 2.0 * n * float(np.max(conv))
   
   def build_conv_matrix(a):
       """Build matrix A such that A @ g = conv(a, g). Shape: (2n-1, n)."""
       n = len(a)
       A = np.zeros((2 * n - 1, n))
       for k in range(2 * n - 1):
           # A[k, j] = a[k-j] if 0 <= k-j < n
           j_start = max(0, k - (n - 1))
           j_end = min(n - 1, k)
           if j_start <= j_end:
               A[k, j_start:j_end + 1] = a[k - np.arange(j_start, j_end + 1)]
       return A
   
   def solve_lp(a, solver_preference=None):
       """
       Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.
       Returns (g, M) or (None, None).
       """
       n = len(a)
       a_norm = a / np.sum(a)
       A = build_conv_matrix(a_norm)
       
       solvers = solver_preference or [cp.CLARABEL, cp.MOSEK, cp.SCS]
       
       for solver in solvers:
           try:
               g = cp.Variable(n, nonneg=True)
               M = cp.Variable()
               constraints = [A @ g <= M, cp.sum(g) == 1]
               prob = cp.Problem(cp.Minimize(M), constraints)
               prob.solve(solver=solver)
               if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                   g_val = np.maximum(g.value, 0)
                   s = float(np.sum(g_val))
                   if s > 1e-10:
                       return g_val / s, float(M.value)
           except Exception:
               continue
       
       # Fallback: scipy linprog
       try:
           from scipy.optimize import linprog
           c = np.zeros(n + 1)
           c[-1] = 1.0
           A_ub = np.zeros((2 * n - 1, n + 1))
           A_ub[:, :n] = A
           A_ub[:, n] = -1.0
           b_ub = np.zeros(2 * n - 1)
           A_eq = np.zeros((1, n + 1))
           A_eq[0, :n] = 1.0
           b_eq = np.array([1.0])
           bounds = [(0, None)] * n + [(None, None)]
           res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
           if res.status == 0:
               g_val = np.maximum(res.x[:n], 0)
               s = float(np.sum(g_val))
               if s > 1e-10:
                   return g_val / s, float(res.x[n])
       except Exception:
           pass
       
       return None, None

                       a = a_try.copy()

                       break

import numpy as np

import cvxpy as cp
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    result = irfft(fa * fa, size)[:2 * n - 1]
    return np.maximum(result, 0)


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        A[k_start:k_end + 1, j] = a[0:k_end - k_start + 1]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. conv(a,g) <= M for all lags, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_alternate(a, time_budget, verbose=False, seed=42, patience_base=25):
    """
    LP alternating with adaptive step sizes and smart restarts.
    Returns the best sequence found.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()

    start = time.monotonic()

    stall = 0
    patience = patience_base
    iteration = 0
    rng = np.random.default_rng(seed)
    obj_history = []

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        # Check convergence via similarity
        diff_norm = np.linalg.norm(a - g_lp)
        ref_norm = np.linalg.norm(a) + np.linalg.norm(g_lp) + 1e-15
        similarity = 1.0 - diff_norm / ref_norm

        if similarity > 0.9999:
            stall += 5
        else:
            # Try full step first
            a_try = g_lp.copy()
            obj_try = compute_obj(a_try)

            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                stall = 0
                patience = max(15, patience_base)
                iteration += 1
                obj_history.append(best_obj)
                if len(obj_history) > 100:
                    obj_history.pop(0)
                if verbose and iteration % 30 == 0:
                    elapsed = time.monotonic() - start
                    print(f"          LP-alt it{iteration}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")
                continue

            found_improve = False
            for t in [0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05, 0.02]:
                a_try = (1.0 - t) * a + t * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    stall = 0

                    iteration += 1
                    obj_history.append(best_obj)
                    if len(obj_history) > 100:
                        obj_history.pop(0)
                    found_improve = True
                    break

            if not found_improve:
                stall += 1

        if stall >= patience:
            # Smart restart: use multiple perturbation strategies
            strategy = stall % 4
            if strategy == 0:
                # Multiplicative Gaussian noise
                scale = min(0.3, 0.05 + 0.02 * (stall // patience))
                noise = rng.normal(0, scale, n)
                a = best_a * np.exp(noise)
            elif strategy == 1:
                # Blend with uniform
                t_blend = 0.1 + 0.1 * (stall // patience)
                a = (1.0 - t_blend) * best_a + t_blend * np.ones(n) / n
            elif strategy == 2:
                # Blend with LP solution (even if slightly worse)
                t_blend = 0.5
                a = (1.0 - t_blend) * best_a + t_blend * g_lp
                # Add small noise

                target = sorted_idx[:5]

                    for ti in target:
                        delta = 0.01 * a[si]
                        a[si] -= delta
                        a[ti] += delta

            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            else:
                a = best_a.copy()
            stall = 0
            patience = max(15, patience + 5)

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def cosine_init(n, freq, amp, phase=0.0):

    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a

    return a

def w_rev_func(weights):
