def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu * rho > (cssv - 1.0)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_val = idx[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_descent(a, time_budget=60.0, seed=0, verbose=False, use_improved=True):

            g_lp, M_lp = solve_lp_improved(a)

        t_candidates = sorted(set([max(0.01, min(1.0, t)) for t in t_candidates]))
        
        best_t_obj = best_obj
        best_t_val = last_t
        best_a_local = best_a.copy()
        
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)

def coordinate_descent_fast(a, time_budget=60.0, seed=0):
    """Fast coordinate descent using gradient information to select which coordinates to update."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Compute gradient to determine which coordinates to focus on
    g, conv = smoothed_gradient_fft(a, 50.0)
    # Coordinates with large gradient magnitude are more promising
    grad_mags = np.abs(g)
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Select coordinates proportional to gradient magnitude
        probs = grad_mags + 1e-10
        probs = probs / probs.sum()
        
        # Update a random subset
        n_update = max(10, n // 10)
        indices = rng.choice(n, size=n_update, replace=False, p=probs)
        
        for i in indices:
            if time.monotonic() - start >= time_budget:
                break
            
            old_val = a[i]
            # Try a few steps
            best_local = best_obj
            best_local_val = old_val
            
            for step in [1.0/n*0.1, -1.0/n*0.1, 1.0/n*0.3, -1.0/n*0.3, 1.0/n*0.05, -1.0/n*0.05]:
                a_try = a.copy()
                a_try[i] = max(0, old_val + step)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_local:
                    best_local = obj_try
                    best_local_val = max(0, old_val + step)
            
            if best_local_val != old_val:
                a[i] = best_local_val
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                if best_local < best_obj:
                    best_obj = best_local
                    best_a = a.copy()
        
        it += 1
        if it % 5 == 0:
            # Recompute gradient
            g, conv = smoothed_gradient_fft(a, 50.0)
            grad_mags = np.abs(g)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
        a = a / a.sum()
    elif mode == 4:
        # Chirp-like
        x = np.linspace(0, 1, n)
        freq = rng.uniform(5, 20)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x)
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 5:
        # Multi-scale
        base = np.ones(n) / n
        detail = rng.normal(0, 0.1, n)
        a = base + detail
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 6:
        # Tukey window
        x = np.linspace(0, 1, n)
        r = 0.5
        a = np.ones(n)
        m = int(r * n // 2)
        for i in range(m):
            t = (i + 1) / (m + 1)
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * t - 1) / r))
            a[n-1-i] = a[i]
        a = a / a.sum()
    elif mode == 7:
        # Raised cosine with different center
        c = rng.uniform(0.3, 0.7) * n
        w = rng.uniform(0.2, 0.5) * n
        x = np.arange(n)
        a = np.exp(-((x - c) / w) ** 2)
        a = a / a.sum()
    elif mode == 8:
        # Periodic with random frequency
        f = rng.uniform(3, 15)
        x = np.arange(n) / n
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f * x) + 0.2 * np.cos(2 * np.pi * 2*f * x)
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of:
- FFT-based smoothed gradient descent with Nesterov momentum and adaptive tau
- Improved LP-based direction finding (second-order aware)
- Spectral domain optimization (power spectrum flattening via zero-padded DFT)
- Gradient-guided coordinate descent
- Multi-length exploration and diverse restarts

Key insight: The objective uses LINEAR CONVOLUTION (not autocorrelation).
The peak is typically near the middle of the convolution. Spectral shaping
on the zero-padded sequence directly targets flattening the convolution.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu * rho > (cssv - 1.0)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_val = idx[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient_fft(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT.
    Returns (gradient array of length n, convolution values).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def spectral_shaping(a, n_iters=200, beta_init=0.5, beta_end=10.0, seed=0):
    """
    Iterative amplitude adjustment in frequency domain (zero-padded) to flatten
    the linear convolution. Uses the zero-padded DFT so that circular convolution
    of the padded sequence equals linear convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    # Zero-pad to length 2n for correct linear conv via circular conv
    N = 2 * n
    rng = np.random.default_rng(seed)

    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        beta = beta_init + (beta_end - beta_init) * progress

        # Compute DFT of zero-padded sequence
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        mag = np.abs(A)
        phase = np.angle(A)

        # Compute current convolution (linear)
        conv = compute_conv(a)
        peak_idx = np.argmax(conv)
        peak_val = conv[peak_idx]

        # Strategy: flatten power spectrum, with extra emphasis on reducing
        # the peak lag contribution
        power = mag ** 2
        mean_power = np.mean(power)

        # Blend power spectrum toward flat
        alpha = 0.05 + 0.4 * progress
        new_power = (1.0 - alpha) * power + alpha * mean_power

        # Additional: reduce power at frequencies that contribute to peak
        # The contribution of freq f to conv[k] is |A(f)|^2 * exp(2pi i f k / N) / N
        # For the peak lag, we can identify which frequencies contribute most
        if peak_val > 0 and it > 10:
            # Compute the phase alignment at peak lag
            k_peak = int(peak_idx)
            freqs = np.arange(N)
            alignment = np.exp(2j * np.pi * freqs * k_peak / N)
            # Frequencies where alignment is positive (constructive) contribute most
            constructive = np.real(alignment) > 0.5
            # Slightly reduce power at those frequencies
            new_power[constructive] *= (1.0 - 0.1 * beta / max(beta_init, 1.0))
            new_power = np.maximum(new_power, 1e-15)

        new_mag = np.sqrt(new_power)
        new_A = new_mag * np.exp(1j * phase)
        new_a_pad = ifft(new_A).real
        new_a = new_a_pad[:n]
        new_a = np.maximum(new_a, 0)
        s_new = float(np.sum(new_a))
        if s_new > 1e-10:
            new_a = new_a / s_new
            obj_new = compute_obj(new_a)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = new_a.copy()
                a = new_a.copy()
            else:
                # Try gentler step
                a_try = 0.9 * a + 0.1 * new_a
                a_try = np.maximum(a_try, 0)
                if a_try.sum() > 1e-10:
                    a_try = a_try / a_try.sum()
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try.copy()
        else:
            a = best_a.copy()

    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix (2n-1 x n) such that A_mat @ g = convolve(a, g)."""
    n = len(a)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A_mat[j:j + r, j] = a[:r]
    return A_mat


def solve_lp_improved(a):
    """
    Improved LP: min M s.t. 2*conv(a,g)_k - conv(a,a)_k <= M for all k, sum(g)=1, g>=0.
    This accounts for the current autocorrelation profile for better descent direction.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv_aa = compute_conv(a_norm)

    A_mat = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    # 2 * A_mat @ g - conv_aa <= M
    constraints = [2.0 * A_mat @ g - conv_aa <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_standard(a):
    """Standard LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A_mat = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_gd(a, time_budget=100.0, tau_init=20.0, tau_end=600.0, seed=0, verbose=False):
    """Nesterov smoothed gradient descent with cosine tau annealing and resets."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.8
    lr = 4.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_g = None
    prev_y = a.copy()
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init + (tau_end - tau_init) * (1.0 - np.cos(np.pi * progress)) / 2.0

        # Periodic tau reset
        if it > 0 and it % 8000 == 0:
            tau = tau_init * 1.5

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)

        # Barzilai-Borwein step size update
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_y
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.6 * lr + 0.4 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_y = y.copy()
        prev_g = g.copy()
        a = a_new
        it += 1

        lr = max(1e-9, lr * 0.99997)

        # Reset if stuck
        if it - best_it > 25000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 4.0 / np.sqrt(n)
            best_it = it

        if verbose and it % 5000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")

    return best_a


def lp_descent(a, time_budget=60.0, seed=0, verbose=False, use_improved=True):
    """LP-based descent with multiple blending factors and adaptive perturbation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    last_t = 0.3
    fails = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        if use_improved:
            g_lp, M_lp = solve_lp_improved(a)
            if g_lp is None:
                g_lp, M_lp = solve_lp_standard(a)
        else:
            g_lp, M_lp = solve_lp_standard(a)

        if g_lp is None:
            break

        # Try multiple blending factors
        t_candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        t_candidates.append(max(0.02, last_t * 0.6))
        t_candidates.append(min(1.0, last_t * 1.5))
        t_candidates.append(max(0.02, last_t * 0.8))
        t_candidates = sorted(set([max(0.01, min(1.0, t)) for t in t_candidates]))

        best_t_obj = best_obj
        best_t_val = last_t
        best_a_local = best_a.copy()

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        it += 1

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a.copy()
            last_t = best_t_val
            fails = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t_val:.2f}")
        else:
            fails += 1
            if fails > 15:
                break
            # Adaptive perturbation to escape
            conv = compute_conv(a)
            peak_idx = np.argmax(conv)
            # Perturb more near the peak-contributing positions
            scale = 0.01 + 0.005 * min(fails, 6)
            noise = rng.normal(0, 1, n) * scale
            # Extra perturbation near center (where convolution peak typically is)
            center = n // 2
            center_weight = np.exp(-((np.arange(n) - center) ** 2) / (2 * (n // 8) ** 2))
            noise += 0.5 * scale * rng.normal(0, 1, n) * center_weight
            a = a * np.exp(noise)
            a = np.maximum(a, 0)
            if a.sum() > 1e-10:
                a = a / a.sum()
            a = 0.85 * a + 0.15 * best_a
            a = a / a.sum()

    return best_a


def coordinate_descent_fast(a, time_budget=60.0, seed=0):
    """Gradient-guided coordinate descent: focuses on high-gradient coordinates."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Compute gradient to determine which coordinates matter most
    g, conv = smoothed_gradient_fft(a, 50.0)
    grad_mags = np.abs(g) + 1e-10

    it = 0
    while time.monotonic() - start < time_budget:
        # Focus on top-gradient coordinates
        top_frac = max(10, n // 8)
        top_idx = np.argsort(grad_mags)[::-1][:top_frac]
        # Add some random ones for exploration
        rand_idx = rng.choice(n, size=top_frac // 4, replace=False)
        indices = np.unique(np.concatenate([top_idx, rand_idx]))

        for i in indices:
            if time.monotonic() - start >= time_budget:
                break

            old_val = a[i]
            best_local = best_obj
            best_local_val = old_val

            # Adaptive step sizes based on current value
            base_step = max(1e-4, 0.1 / n)
            for step in [base_step, -base_step, base_step * 3, -base_step * 3,
                         base_step * 0.3, -base_step * 0.3]:
                a_try = a.copy()
                a_try[i] = max(0.0, old_val + step)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_local:
                    best_local = obj_try
                    best_local_val = max(0.0, old_val + step)

            if best_local_val != old_val:
                a[i] = best_local_val
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                if best_local < best_obj:
                    best_obj = best_local
                    best_a = a.copy()

        it += 1
        if it % 3 == 0:
            g, conv = smoothed_gradient_fft(a, 50.0)
            grad_mags = np.abs(g) + 1e-10

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations with different structures."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Sinusoidal modulation
        f = rng.uniform(3, 12)
        phi = rng.uniform(0, 2 * np.pi)
        x = np.arange(n) / n
        a = 1.0 + 0.7 * np.sin(2 * np.pi * f * x + phi)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 3:
        # Multi-frequency
        freqs = rng.uniform(2, 15, size=3)
        phis = rng.uniform(0, 2 * np.pi, size=3)
        x = np.arange(n) / n
        a = 1.0
        for f, p in zip(freqs, phis):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Asymmetric bump (favors one side to reduce middle convolution)
        c = rng.uniform(0.2, 0.8) * n
        w = rng.uniform(0.15, 0.4) * n
        x = np.arange(n)
        a = np.exp(-((x - c) / w) ** 2)
        a = a / a.sum()
    elif mode == 5:
        # Two bumps (asymmetric)
        c1 = rng.uniform(0.1, 0.3) * n
        c2 = rng.uniform(0.6, 0.9) * n
        w1 = rng.uniform(0.1, 0.25) * n
        w2 = rng.uniform(0.1, 0.25) * n
        x = np.arange(n)
        a = 0.7 * np.exp(-((x - c1) / w1) ** 2) + 0.3 * np.exp(-((x - c2) / w2) ** 2)
        a = a / a.sum()
    elif mode == 6:
        # Cosine taper with random duty
        x = np.linspace(0, np.pi, n)
        a = (1 + np.cos(x)) / 2
        a = a / a.sum()
    elif mode == 7:
        # Random smooth (low-pass filtered noise)
        a = rng.uniform(0, 1, n)
        k = max(3, n // 30)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 8:
        # Chirp-like (linear frequency sweep)
        f0, f1 = rng.uniform(3, 8), rng.uniform(10, 25)
        x = np.arange(n)
        phase = 2 * np.pi * (f0 * x / n + (f1 - f0) * x ** 2 / (2 * n))
        a = 0.5 + 0.5 * np.cos(phase)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 9:
        # Exponential decay from center
        c = n // 2
        decay = rng.uniform(0.5, 2.0)
        x = np.arange(n)
        a = np.exp(-decay * np.abs(x - c) / n)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    if remaining > 40:
        alloc_lp = min(remaining * 0.15, 100.0)
        a_lp3 = lp_descent(best_seq, time_budget=alloc_lp, seed=seed + 400,
                           verbose=False, use_improved=True)
        obj_lp3 = compute_obj(a_lp3)
        if obj_lp3 < best_score:
            best_score = obj_lp3
            best_seq = a_lp3.copy()
            print(f"  LP3: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_cd = coordinate_descent_fast(best_seq, time_budget=min(60, remaining * 0.3), seed=seed + 500)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  CD: obj={best_score:.8f} **")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = [n_current + 100, n_current + 200, n_current + 400,
                 max(500, n_current - 100), 1000, 1200]
    n_options = [n for n in n_options if 500 <= n <= 2000 and abs(n - n_current) >= 50]
    n_options = sorted(set(n_options))[:4]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        # Quick optimization on new length
        alloc = min(remaining * 0.18, 100.0)
        if alloc > 30:
            a_try = smoothed_gd(a_try, time_budget=alloc * 0.5,
                                tau_init=20, tau_end=400, seed=seed + n_try)
            remaining = deadline - time.monotonic()
            if remaining > 25:
                a_try = lp_descent(a_try, time_budget=min(40, remaining * 0.3),
                                   seed=seed + n_try + 1, use_improved=True)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            n = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse Restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = (r * 3 + 1) % 10
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.05 + 0.03 * r
        a_r = (1.0 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.18, 70.0)
        if alloc > 20:
            a_r = smoothed_gd(a_r, time_budget=alloc * 0.5,
                              tau_init=15, tau_end=300, seed=seed + 1000 + r * 10)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_r = lp_descent(a_r, time_budget=min(30, remaining * 0.3),
                                 seed=seed + 1000 + r * 10 + 1, use_improved=True)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Final Intensive Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau spectral
    if remaining > 25:
        a_sp = spectral_shaping(best_seq, n_iters=80, beta_init=3.0, beta_end=20.0, seed=seed + 2000)
        obj_sp = compute_obj(a_sp)
        if obj_sp < best_score:
            best_score = obj_sp
            best_seq = a_sp.copy()
            print(f"  Final Spectral: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_gd = smoothed_gd(best_seq, time_budget=min(50, remaining * 0.4),
                           tau_init=100.0, tau_end=2000.0, seed=seed + 2100)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_lp = lp_descent(best_seq, time_budget=min(40, remaining * 0.5),
                          seed=seed + 2200, use_improved=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_cd = coordinate_descent_fast(best_seq, time_budget=min(30, remaining * 0.5), seed=seed + 2300)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
