import numpy as np

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        return a

    best_it = 0

        y = np.maximum(y, 0)

        a = a_new

def lp_direction(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Uses the structure of the problem - avoids building full matrix for small n.
    Returns g (the optimal direction).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build the convolution matrix A (2n-1 x n)
    # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    # This is a Toeplitz-like matrix
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i  # k = i + 0, i + 1, ..., i + n-1 (but max is 2n-2)
        k_end = min(2 * n - 1, i + n)
        A[k_start:k_end, i] = a_norm[:k_end - k_start]
    
    # Solve LP using scipy
    from scipy.optimize import linprog
    
    # Variables: g[0..n-1], M
    # Total: n+1 variables
    # Minimize: M (index n)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Constraints: A @ g - M <= 0 (2n-1 constraints)
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g[i] >= 0, M free (but M >= 0 practically)
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = res.x[:n]
            g = np.maximum(g, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    
    return None, None

def lp_polish(a, time_budget, verbose=False):
    """
    Iterative LP-based polishing: solve LP direction, line search, repeat.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0

    while time.monotonic() - start < time_budget:
        g, M_val = lp_direction(a)
        if g is None:
            break

        # Line search: blend a with g
        best_t = 0
        best_obj_line = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_line:
                best_obj_line = obj_try
                best_t = t

        if best_t > 0:
            a_new = (1 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            best_a = a_new.copy()
            best_obj = best_obj_line
            a = a_new
            stall = 0
            if verbose:
                print(f"    LP t={best_t:.2f}: obj={best_obj:.8f}")
        else:
            stall += 1
            # Perturb
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            if stall > 20:
                break

    return best_a

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.arange(n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        a = np.exp(-((x - n//4)**2) / (2*(n//10)**2)) + np.exp(-((x - 3*n//4)**2) / (2*(n//10)**2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        xx = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - xx**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        # Multi-frequency
        phases = rng.uniform(0, 2*np.pi, size=3)
        freqs = rng.integers(1, 5, size=3)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2*np.pi*f*x/n + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        a = (x + 1) ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        # Cosine bump
        a = 0.5 + 0.5 * np.cos(2*np.pi*x/n)
        a = a / a.sum()
    elif mode == 7:
        # Concentrated center
        a = np.exp(-((x - n//2)**2) / (2*(n//5)**2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def pairwise_exchange(a, time_budget, verbose=False):
    """
    Try to reduce the objective by swapping mass between pairs of positions.
    Focus on reducing the dominant peak.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_k = int(np.argmax(conv))
        
        # Find contributing indices to the peak
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        
        improved = False
        # Try to reduce the peak by moving mass
        # For each pair (i, j) contributing to the peak, try moving mass from i to j' or vice versa
        for _ in range(50):
            i = rng.integers(i_start, i_end + 1)
            j = max_k - i
            if j < 0 or j >= n or i == j:
                continue
            
            # Try moving a small amount from max(i,j) to a random position
            src = max(i, j)
            # Find a good target: somewhere that doesn't contribute to the peak
            tgt = rng.integers(0, n)
            if tgt == src:
                continue
            
            delta = 0.01 * a[src]
            if delta > a[src]:
                continue
            
            a_try = a.copy()
            a_try[src] -= delta
            a_try[tgt] += delta
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break

    return best_a

"""
Optimizes step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent with log-sum-exp smoothing (FFT-based O(n log n) gradients),
LP polishing via HiGHS, multi-n exploration, diverse restarts, and subgradient methods.
Key innovations: entropic mirror map for simplex, adaptive tau scheduling,
Nesterov acceleration, and efficient cross-correlation gradients.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient(a, tau):
    """
    Compute gradient of f_tau(a) = 2n * (1/tau) * log(sum_k exp(tau*C_k))
    using FFT cross-correlation. Returns (grad, conv, p).
    grad[m] = 4n * sum_k p[k] * a[k-m] * I(0<=k-m<n)
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    # Cross-correlation via FFT: xc[m] = sum_k p[k] * a[k-m]
    N = 1
    while N < 3 * n:
        N <<= 1

    P = fft(p, N)
    A = fft(a, N)
    xc = ifft(np.conj(P) * A).real

    grad = 4.0 * n * xc[:n]
    return grad, conv, p


def subgradient_at_peak(a):
    """Compute the exact subgradient at the argmax of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    # subgrad[m] = 4n * a[k_star - m] if 0 <= k_star - m < n else 0
    grad = np.zeros(n)
    for m in range(n):
        idx = k_star - m
        if 0 <= idx < n:
            grad[m] = 4.0 * n * a[idx]
    return grad, k_star


def mirror_descent_smoothed(a, time_budget, tau_init, tau_end, lr_init, seed=42, use_nesterov=True, verbose=False):
    """
    Mirror descent on smoothed objective over simplex using entropic mirror map.
    Update: a_new = a * exp(-eta * grad) / Z (multiplicative weights)
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    eta = lr_init
    v = np.zeros(n)
    beta = 0.8 if use_nesterov else 0.0
    it = 0
    best_it = 0
    stagnation = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.88))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.55)

        # Nesterov extrapolation
        if use_nesterov:
            y = a + beta * v
            y = np.maximum(y, 0)
            s_y = float(np.sum(y))
            if s_y < 1e-10:
                y = best_a.copy()
            else:
                y = y / s_y
        else:
            y = a

        # Compute smoothed gradient
        grad, conv, p = smoothed_gradient(y, tau)

        # Mirror descent update (multiplicative weights)
        # a_new[m] = y[m] * exp(-eta * grad[m]) / Z
        scaled = -eta * grad
        scaled -= np.max(scaled)  # numerical stability
        a_new = y * np.exp(scaled)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        # Also try Euclidean projected step
        a_euc = project_simplex(y - eta * grad)

        obj_new = compute_obj(a_new)
        obj_euc = compute_obj(a_euc)

        if obj_euc < obj_new:
            a_new = a_euc
            obj_new = obj_euc

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stagnation = 0
        else:
            stagnation += 1

        if use_nesterov:
            v = a_new - a
        a = a_new
        it += 1

        # Adaptive learning rate (decay)
        eta = lr_init / (1.0 + 0.05 * it * (0.1 + progress))
        eta = max(eta, 1e-8)

        # Adaptive momentum
        if use_nesterov:
            beta = min(0.95, 0.8 + 0.15 * progress)

        # Reset if stuck for too long
        if stagnation > 3000:
            a = best_a.copy()
            v = np.zeros(n)
            beta = 0.8
            eta = lr_init * 0.5
            stagnation = 0
            # Small random perturbation
            if it > 500:
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / a.sum()

        if verbose and it % 2000 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, eta={eta:.5f}")

    return best_a


def subgradient_descent(a, time_budget, lr_init, seed=42, verbose=False):
    """
    Subgradient descent with averaging on the non-smooth objective.
    Step in direction of the subgradient at the current peak.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    eta = lr_init
    running_avg = a.copy()
    it = 0
    best_it = 0
    stagnation = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))

        # Compute subgradient at current peak
        grad, k_star = subgradient_at_peak(a)

        # Normalize gradient for step size
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm > 1e-10:
            grad = grad / grad_norm

        # Step
        step = eta / np.sqrt(1 + it * 0.01)
        a_new = project_simplex(a - step * grad * grad_norm)

        # Also try multiplicative update
        scaled = -eta / (1 + it * 0.05) * grad
        scaled -= np.max(scaled)
        a_mult = a * np.exp(scaled)
        s_m = float(np.sum(a_mult))
        if s_m > 1e-10:
            a_mult = a_mult / s_m
            obj_mult = compute_obj(a_mult)
            obj_euc = compute_obj(a_new)
            if obj_mult < obj_euc:
                a_new = a_mult

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stagnation = 0
        else:
            stagnation += 1

        # Running average (for convergence guarantee)
        running_avg = (it * running_avg + a_new) / (it + 1)
        obj_avg = compute_obj(running_avg)
        if obj_avg < best_obj:
            best_obj = obj_avg
            best_a = running_avg.copy()
            best_it = it

        a = a_new
        it += 1

        if stagnation > 2000:
            a = best_a.copy()
            eta = lr_init * 0.5
            stagnation = 0
            if it > 1000:
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / a.sum()

        if verbose and it % 5000 == 0:
            print(f"    SD it{it}: obj={best_obj:.8f}, eta={eta:.4f}")

    return best_a


def lp_direction(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M_val) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))

    # Build convolution matrix A (2n-1 x n) where A[k,i] = a_norm[k-i] if valid
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A[k_start:k_end, i] = a_norm[:k_end - k_start]

    from scipy.optimize import linprog

    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[n] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s_g = float(np.sum(g))
            if s_g > 1e-10:
                g = g / s_g
                return g, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_polish(a, time_budget, verbose=False):
    """Iterative LP-based polishing with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(54321)
    stall = 0

    while time.monotonic() - start < time_budget:
        g, M_val = lp_direction(a)
        if g is None:
            break

        # Line search
        best_t = 0
        best_obj_line = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_line:
                best_obj_line = obj_try
                best_t = t

        if best_t > 0:
            a_new = (1 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / float(np.sum(a_new))
            best_a = a_new.copy()
            best_obj = best_obj_line
            a = a_new
            stall = 0
            if verbose:
                print(f"    LP t={best_t:.2f}: obj={best_obj:.8f}")
        else:
            stall += 1
            # Perturb and retry
            scale = 0.01 + 0.005 * min(stall, 10)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / float(np.sum(a))
            if stall > 30:
                break

    return best_a


def peak_reducing_cd(a, time_budget, verbose=False):
    """

    Tries to move mass away from positions contributing to the top peaks.

    while time.monotonic() - start < time_budget:

        max_k = int(np.argmax(conv))

        # Find top contributing pairs to the peak
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)

        improved = False
        # Try multiple random attempts to reduce the peak
        for _ in range(100):
            # Pick a random source from the contributing set
            if i_end - i_start + 1 > 0:
                src = int(rng.integers(i_start, i_end + 1))
            else:
                src = int(rng.integers(0, n))

            if a[src] < 1e-8:

            tgt = int(rng.integers(0, n))

            # Fractional move

            a_try[src] -= delta

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    xm = x - (n - 1) / 2.0  # centered
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two bumps
        w = max(1, n // 12)
        a = np.exp(-(x - n * 0.25) ** 2 / (2 * w ** 2)) + np.exp(-(x - n * 0.75) ** 2 / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine
        xx = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - xx ** 2 + 1e-8))
        a = a / a.sum()
    elif mode == 4:
        # Multi-sine
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x / n + p)
        a = np.maximum(a, 0.005)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        a = (x + 1) ** float(rng.uniform(-3, 3))
        a = a / a.sum()
    elif mode == 6:
        # Cosine window
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
        a = a / a.sum()
    elif mode == 7:
        # Center concentrated
        w = max(1, n // 8)
        a = np.exp(-xm ** 2 / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 8:
        # Edges concentrated
        a = np.abs(xm) ** float(rng.uniform(1, 3)) + 1.0
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
