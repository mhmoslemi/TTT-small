**The key algorithm: Mirror Descent with LogSumExp**

We want to minimize `f(a) = max_k conv(a,a)[k]` subject to `sum(a) = 1, a >= 0`.

Using the log-sum-exp smooth approximation:
`f_beta(a) = (1/beta) * log(sum_k exp(beta * conv(a,a)[k]))`

The gradient is:
`grad_i = (1/beta) * sum_k w_k * (a[i+k] + a[i-k]) * beta = sum_k w_k * (a[i+k] + a[i-k])`

where `w_k = exp(beta * conv(a,a)[k]) / sum_j exp(beta * conv(a,a)[j])`.

(The beta cancels out in the gradient!)

Mirror descent update on the simplex:
`a_new[i] = a[i] * exp(-lr_i * grad_i) / Z`

where `Z = sum_j a[j] * exp(-lr_j * grad_j)`.

This is the natural gradient / multiplicative weights update.

For the learning rate, we can use:
- `lr = 1 / (beta * sqrt(t))` (standard subgradient)
- Or adaptive: `lr = alpha / (1 + cumulative_grad_norm)`
- Or simply: try a few fixed values and use the best.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))

def compute_lse_grad(a, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*conv(a,a)[k]))."""
    n = len(a)
    conv = compute_conv(a)
    shifted = beta * (conv - conv.max())
    w = np.exp(shifted)
    w /= w.sum()
    # grad[i] = sum_k w[k] * (a[i+k] + a[i-k])
    size = 1
    while size < 2 * n:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    # g2[i] = conv(w, a)[i] = sum_k w[k]*a[i-k] (for valid i-k)
    conv_wa = irfft(W * A, size)
    g2 = conv_wa[:n]
    # g1[i] = conv(w_rev, a)[n-1+i] = sum_k w[k]*a[i+k]
    W_rev = rfft(w[::-1], size)
    conv_wra = irfft(W_rev * A, size)
    g1 = conv_wra[n-1:n-1+n]
    return g1 + g2, w

def mirror_descent(a, time_budget, beta_start=10, beta_end=500, verbose=False):
    """Mirror descent with LSE smoothing on the simplex."""
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Polyak average
    avg_a = a.copy()
    count = 0
    
    lr = 1.0 / np.sqrt(n)
    it = 0
    
    while time.monotonic() - start < time_budget:
        # Adaptive beta
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        grad, w = compute_lse_grad(a, beta)
        
        # Mirror descent update
        log_a = np.log(a)
        log_a_new = log_a - lr * grad
        log_a_new -= log_a_new.max()  # for numerical stability
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        count += 1
        
        # Polyak average (weighted by 1/t)
        avg_a = (avg_a * (count - 1) + a_new) / count
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            lr = max(1e-8, lr * 1.001)  # slightly increase lr
        else:
            # Check averaged solution
            obj_avg = compute_obj(avg_a)
            if obj_avg < best_obj - 1e-10:

            lr = max(1e-8, lr * 0.999)

            print(f"          MD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.6f}, t={elapsed:.1f}s")

    if obj_avg_final < best_obj:
        best_a = avg_a.copy()
        best_obj = obj_avg_final

def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / np.sum(a)
    A = build_conv_matrix(a_n)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, float(res.x[n])
    except Exception:
        pass
    return None, None

Hybrid optimization: Mirror Descent (LSE) + LP Alternating with interleaving,
n-sweep, and diverse restarts. Target: score < 1.5027.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def compute_lse_grad(a, beta):
    n = len(a)
    conv = compute_conv(a)
    shifted = beta * (conv - conv.max())
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= w_sum
    size = 1
    while size < 2 * n:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    conv_wa = irfft(W * A, size)
    g2 = conv_wa[:n]
    W_rev = rfft(w[::-1], size)
    conv_wra = irfft(W_rev * A, size)
    g1 = conv_wra[n - 1:n - 1 + n]
    return g1 + g2, w

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-15:
        return None, None
    a_n = a / s
    A = build_conv_matrix(a_n)
    c = np.zeros(n + 1)
    c[-1] = 1.0

    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

            g_sum = float(np.sum(g))
            if g_sum > 1e-10:
                return g / g_sum, float(res.x[n])
    except Exception:
        pass
    return None, None

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
