import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}"""
    n = len(v)
    v = v.copy()
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.where(u * np.arange(1, n+1) > (cssv - target_sum))[0] + 1) if np.any(u * np.arange(1, n+1) > (cssv - target_sum)) else n
    theta = (cssv[rho-1] - target_sum) / rho
    return np.maximum(v - theta, 0)

Optimized search for minimizing peak autocorrelation of non-negative step functions.
Uses fast LP alternating with precomputed line search, subgradient refinement,
diverse restarts, and dense n-sweep. Target: score < 1.5027.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_pair(a, b):
    """Full linear convolution a*b via FFT. Returns array of length 2n-1 (assuming same length)."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:2 * n - 1]

def compute_obj_from_conv(a, conv):
    """Compute objective given precomputed convolution."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(conv)) / (s * s)

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def solve_lp_fast(a, n):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses scipy linprog with HiGHS.
    Returns (g_normalized, M_opt) or (None, None).
    """
    a_norm = a / float(np.sum(a))
    
    # Build sparse-like convolution matrix
    # conv(a, g)[k] = sum_j a[j] * g[k-j] (with bounds)
    # = sum_j g[j] * a[k-j]
    # So we need A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    # A is (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n
    
    # For efficiency, build it as a list of (row, col, val) triples
    # But for n=800, a dense matrix is 1599*800 = 1.28M entries, which is fine
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a[k-j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
        A[j:j+n, j] = a_norm
    
    # LP: min M
    # s.t. A@g - M <= 0  (i.e., A@g <= M)
    #      sum(g) = 1
    #      g >= 0
    
    # Variables: [g(0), ..., g(n-1), M]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    # A_ub * x <= b_ub
    # [A, -I] * [g; M] <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # A_eq * x = b_eq
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n  # g >= 0
    bounds.append((None, None))  # M is free
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None

def line_search_precomputed(a, g, conv_a, conv_ag, conv_g):
    """
    Line search on (1-t)*a + t*g using precomputed convolutions.
    Returns (best_t, best_obj).
    """
    n = len(a)
    s_a = float(np.sum(a))
    s_g = float(np.sum(g))
    if s_a < 1e-10 or s_g < 1e-10:
        return 0.0, np.inf
    
    # Normalize to sum=1
    a1 = a / s_a
    g1 = g / s_g
    
    # conv((1-t)*a1 + t*g1, (1-t)*a1 + t*g1)[k]
    # = (1-t)^2 * conv_a1[k] + 2t(1-t) * conv_ag1[k] + t^2 * conv_g1[k]
    # where conv_a1 = conv(a1, a1), etc.
    
    # But we have conv_a = conv(a, a), conv_ag = conv(a, g), conv_g = conv(g, g)
    # conv_a1[k] = conv_a[k] / (s_a^2)
    # conv_ag1[k] = conv_ag[k] / (s_a * s_g)
    # conv_g1[k] = conv_g[k] / (s_g^2)
    
    ca = conv_a / (s_a * s_a)
    cb = conv_ag / (s_a * s_g)
    cg = conv_g / (s_g * s_g)
    
    best_t = 0.0
    best_obj = np.inf
    
    # Try a grid of t values
    t_values = np.array([0.0, 0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 
                         0.6, 0.7, 0.8, 0.9, 0.95, 1.0])
    
    for t in t_values:
        t1 = 1.0 - t
        # obj = 2*n * max_k [(1-t)^2 * ca[k] + 2t(1-t) * cb[k] + t^2 * cg[k]]
        conv_t = t1*t1 * ca + 2*t*t1 * cb + t*t * cg
        obj = 2.0 * n * float(np.max(conv_t))
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    # Refine around best_t
    if 0 < best_t < 1:
        t_refine = np.linspace(max(0, best_t - 0.1), min(1, best_t + 0.1), 11)
        for t in t_refine:
            t1 = 1.0 - t
            conv_t = t1*t1 * ca + 2*t*t1 * cb + t*t * cg
            obj = 2.0 * n * float(np.max(conv_t))
            if obj < best_obj:
                best_obj = obj
                best_t = t
    
    return best_t, best_obj

def lp_alternate_fast(a, time_budget, max_iter=None, verbose=False):

def subgradient_optimize(a, time_budget, verbose=False):
    """
    Subgradient method for minimizing max convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 1.0 / np.sqrt(n)
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Find active set: lags where conv is close to M
        tol = 0.05 * M
        active = np.where(conv >= M - tol)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(conv)])
        
        # Compute subgradient
        sg = np.zeros(n)
        for k in active:
            k = int(k)
            # sg[j] += a[j-k] + a[j+k] (with bounds)
            # a[j-k] is non-zero when 0 <= j-k < n, i.e., k <= j <= k+n-1
            # a[j+k] is non-zero when 0 <= j+k < n, i.e., -k <= j <= n-1-k
            j_start1 = max(0, k)
            j_end1 = min(n, k + n)
            if j_start1 < j_end1:
                sg[j_start1:j_end1] += a[:j_end1 - j_start1]
            
            j_start2 = max(0, -k)
            j_end2 = min(n, n - k)
            if j_start2 < j_end2:
                sg[j_start2:j_end2] += a[j_start2 + k:j_end2 + k]
        
        sg /= len(active)
        
        # Update
        a_new = a - lr * sg
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = max(1e-8, lr * 0.999)
        else:
            lr = max(1e-8, lr * 0.95)
        
        it += 1
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"          Subgrad it{it}: obj={best_obj:.8f}, lr={lr:.6f}, |S|={len(active)}, elapsed={elapsed:.1f}s")
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    mode = mode % 10
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + np.exp(-3 * (x - 0.7) ** 2) + 0.05
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = x**2 + (1 - x)**2 + 0.1
        a = a / a.sum()
    elif mode == 6:
        freqs = rng.integers(2, 10, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 7:
        a = rng.dirichlet(np.ones(n) * 0.3)
    elif mode == 8:
        # Triangular
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2*x - 1)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        # Two-peak
        x = np.linspace(0, 1, n)
        a = np.exp(-5*(x-0.25)**2) + np.exp(-5*(x-0.75)**2) + 0.02
        a = a / a.sum()
    return a
