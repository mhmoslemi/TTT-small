"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv via FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1

    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, time_limit=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (optimal_g, optimal_M, active_set) or (None, None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            if time_limit:
                prob.solve(solver=solver, time_limit_sec=int(time_limit))
            else:
                prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Find active set
                    conv_ags = A @ g_val
                    M_val = float(M.value)
                    tol = 1e-5 * max(1.0, M_val)
                    active = np.where(conv_ags >= M_val - tol)[0]
                    return g_val, M_val, active
        except Exception:
            continue

    return None, None, None


def solve_weighted_lp(a, weights, time_limit=None):
    """
    Solve weighted LP: min M s.t. weights[k]*conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    This focuses more on constraints with higher weights (typically the active ones).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    A_w = weights[:, None] * A

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_w @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            if time_limit:
                prob.solve(solver=solver, time_limit_sec=int(time_limit))
            else:
                prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def solve_socp_direction(a, radius, time_limit=None):
    """
    Solve SOCP: find direction d minimizing max_k [(a*a)[k] + 2*(a*d)[k]]
    s.t. ||d||_2 <= radius, a+d >= 0, sum(d) = 0.
    Returns the direction d or None.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv_aa = compute_conv(a_norm)
    A = build_conv_matrix(a_norm)

    d = cp.Variable(n)
    M = cp.Variable()
    constraints = [
        cp.quad_form(d) <= radius ** 2,
        a_norm + d >= 0,
        cp.sum(d) == 0,
        A @ d + conv_aa <= M
    ]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS]
    for solver in solvers:
        try:
            if time_limit:
                prob.solve(solver=solver, time_limit_sec=int(time_limit))
            else:
                prob.solve(solver=solver)
            if d.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return d.value
        except Exception:
            continue
    return None


def golden_section_search(f, a, b, tol=1e-6, max_iter=50):
    """Golden section search for minimum of unimodal f on [a, b]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    x = (a + b) / 2.0
    return x, f(x)

def adaptive_smoothed_descent(a, time_budget=100.0, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent with cosine annealing."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 3.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Cosine anneal tau
        cos_val = 0.5 * (1 + np.cos(np.pi * progress))
        tau = tau_init + (tau_end - tau_init) * (1 - cos_val)

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute gradient
        g, conv = smoothed_gradient_fft(y, tau)

        # Barzilai-Borwein step adjustment
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                bb_lr = min(max(bb_lr, 1e-8), 4.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb_lr

        # Update
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1

        # Decay
        lr = max(1e-8, lr * 0.99995)

        # Reset if stuck
        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 3:
        freqs = rng.integers(1, 6, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Bell-shaped
        c = n / 2
        w = n / 4
        a = np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Two peaks
        c1 = n // 4
        c2 = 3 * n // 4
        w = max(1, n // 10)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 6:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        # Linear ramp
        a = np.linspace(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 8:
        # Exponential decay from center
        a = np.exp(-np.abs(np.arange(n) - n / 2) / (n / 5))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        n_current = 800
        best_seq = np.ones(n_current) / n_current
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Quick GD warmup ===
    print(f"\n=== Phase 1: GD Warmup ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.1, 90.0)
    if p1_budget > 20:
        a_sgd = adaptive_smoothed_descent(best_seq, time_budget=p1_budget,
                                          tau_init=5.0, tau_end=300.0)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase1 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 GD: obj={obj_sgd:.8f} (no improvement)")

    # === Phase 2: Extended LP polishing with line search ===
    print(f"\n=== Phase 2: LP Polishing with Line Search ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.4, 500.0)
    if p2_budget > 30:
        # Multiple rounds with increasing perturbation
        for rnd in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(p2_budget / (8 - rnd), remaining * 0.6)
            if alloc < 15:
                break
            a_lp = lp_polish_with_linesearch(best_seq, time_budget=alloc,
                                             verbose=True, seed=2000 + rnd * 17)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f} (stall)")
                # Inject diversity
                if rnd < 7:
                    strength = 0.008 * (rnd + 1)
                    pert = rng.normal(0, strength, size=len(best_seq))
                    new_seq = best_seq * np.exp(pert)
                    new_seq = np.maximum(new_seq, 0)
                    s = float(np.sum(new_seq))
                    if s > 1e-10:
                        best_seq = new_seq / s
                    # Also try a coarser perturbation
                    if rnd >= 3:
                        n_pert = max(20, len(best_seq) // 10)
                        idx = rng.choice(len(best_seq), size=n_pert, replace=False)
                        new_seq2 = best_seq.copy()
                        new_seq2[idx] *= rng.uniform(0.3, 3.0, size=n_pert)
                        new_seq2 = np.maximum(new_seq2, 0)
                        s2 = float(np.sum(new_seq2))
                        if s2 > 1e-10:
                            new_seq2 = new_seq2 / s2
                            obj2 = compute_obj(new_seq2)
                            if obj2 < best_score + 0.01:  # Allow slight worsening
                                best_seq = new_seq2

    # === Phase 3: Multi-n exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Focus on longer sequences and nearby lengths
    n_options = sorted(set([
        n_best, n_best + 100, n_best + 200, n_best + 400,
        n_best + 600, n_best + 800,
        1000, 1200, 1400, 1600, 2000
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2500]
    # Remove duplicates and sort
    n_options = sorted(set(n_options))
    # Prioritize longer sequences
    n_options = [n for n in n_options if n >= n_best - 50]
    n_options = n_options[:8]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        if n_try == n_best:
            continue
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break

        a_try = resample_sequence(best_seq, n_try)
        print(f"  Resampled to n={n_try}, initial obj={compute_obj(a_try):.8f}")

        # GD phase
        alloc1 = min(remaining * 0.15, 50.0)
        if alloc1 > 15:
            a_try = adaptive_smoothed_descent(a_try, time_budget=alloc1,
                                              tau_init=10.0, tau_end=300.0)

        # LP phase
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.2, 60.0)
        if alloc2 > 15:
            a_try = lp_polish_with_linesearch(a_try, time_budget=alloc2,
                                              verbose=False, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Intensive LP polish on best ===
    print(f"\n=== Phase 4: Intensive LP Polish ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.3, 300.0)
    if p4_budget > 20:
        for rnd in range(10):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(p4_budget / (10 - rnd), remaining * 0.7)
            if alloc < 10:
                break
            a_lp = lp_polish_with_linesearch(best_seq, time_budget=alloc,
                                             verbose=(rnd < 3), seed=5000 + rnd * 31)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Intensive LP {rnd}: obj={best_score:.8f} **")
            else:
                # Small perturbation
                pert = rng.normal(0, 0.005, size=len(best_seq))
                new_s = best_seq * np.exp(pert)
                new_s = np.maximum(new_s, 0)
                s = float(np.sum(new_s))
                if s > 1e-10:
                    best_seq = new_s / s

    # === Phase 5: Final GD with high tau ===
    print(f"\n=== Phase 5: Final High-Tau GD ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.3, 120.0)
        a_final = adaptive_smoothed_descent(best_seq, time_budget=alloc,
                                            tau_init=50.0, tau_end=1500.0, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final GD: obj={best_score:.8f} **")
        else:
            print(f"  Final GD: obj={obj_final:.8f}")

    # === Phase 6: One last LP squeeze ===
    print(f"\n=== Phase 6: Final LP Squeeze ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 80.0)
        a_squeeze = lp_polish_with_linesearch(best_seq, time_budget=alloc,
                                              verbose=True, seed=99999)
        obj_squeeze = compute_obj(a_squeeze)
        if obj_squeeze < best_score:
            best_score = obj_squeeze
            best_seq = a_squeeze.copy()
            print(f"  Squeeze: obj={best_score:.8f} **")
        else:
            print(f"  Squeeze: obj={obj_squeeze:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
