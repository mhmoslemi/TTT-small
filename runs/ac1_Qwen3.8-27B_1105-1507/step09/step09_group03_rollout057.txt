"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
- Fixed LP solver (previous had bug: undefined variable)
- Barzilai-Borwein adaptive step size for faster convergence
- Peak-pair local search for targeted peak reduction
- Spectral initialization for diverse starting points
- Hierarchical n-sweep (coarse-to-fine)
- Dual-objective LP (min-max then min-sum-of-active peaks)
- Adaptive temperature schedule with restarts
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = int(rho[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[k, j] = a[k-j] if 0 <= k-j < n
        lo = j
        hi = min(2 * n - 1, j + n)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_lp_minmax(a, timeout=20):
    """
    Solve LP: min M s.t. conv(a_norm, g) <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g_var = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g_var <= M, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver, timeout=timeout)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_minactive(a, active_indices, timeout=20):
    """
    Solve LP: min sum(conv(a_norm,g)[k] for k in active) s.t. sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    A_active = A[active_indices, :]

    g_var = cp.Variable(n, nonneg=True)
    obj = cp.sum(A_active @ g_var)
    constraints = [cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(obj), constraints)

    solvers = [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver, timeout=timeout)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def two_stage_lp_polish(a, time_budget=100.0, verbose=False):

    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        # Stage 1: min max
        g1, M1 = solve_lp_minmax(a, timeout=15)
        if g1 is None:
            stall += 1
            if stall > 3:
                break
            continue

        # Find active constraints
        conv_ag1 = compute_conv(a)  # This is conv(a,a), not conv(a,g1)
        # We need conv(a_norm, g1)

        A = build_conv_matrix(a_norm)
        conv_a_g1 = A @ g1
        max_val = float(np.max(conv_a_g1))
        tol = 0.02 * max(max_val, 1e-10)
        active = np.where(conv_a_g1 >= max_val - tol)[0]

        if len(active) == 0:
            active = np.array([int(np.argmax(conv_a_g1))])
        # Limit active set size for tractability

            peak_g2 = float(np.max(conv_a_g2))
            if peak_g2 < peak_g1 * 0.99:
                g_dir = g2
            else:
                g_dir = (g1 + g2) / 2.0
        else:
            g_dir = g1

        # Line search: try several t values
        best_t = 0.0

        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:

            obj_try = compute_obj(a_try)

                best_t_obj = obj_try
                best_t = t

            stall = 0
            it += 1
            if verbose and it % 5 == 0:

def nesterov_bb_descent(a, time_budget=300.0, tau_init=5.0, tau_end=2000.0,
                         verbose=False):
    """
    Nesterov accelerated gradient descent with Barzilai-Borwein step size
    and adaptive temperature on log-sum-exp smoothed objective over simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    restart_count = 0

    # BB parameters
    lr_min = 1e-10
    lr_max = 50.0 / np.sqrt(n)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.95))
        # Geometric temperature schedule
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        g, conv = smoothed_gradient(y, tau)

        # Barzilai-Borwein step size update
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15:
                if abs(sy) > 1e-12:
                    bb_lr = ss / abs(sy)
                    # Blend BB with current
                    lr = 0.6 * lr + 0.4 * np.clip(bb_lr, lr_min, lr_max)
                else:
                    lr = lr * 0.95

        # Gradient step
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1

        # Adaptive momentum
        if progress > 0.7:
            beta = 0.7
        elif progress > 0.4:
            beta = 0.8
        else:
            beta = 0.85

        # Periodic restart to escape local minima
        if it - best_it > 5000:
            restart_count += 1
            a = best_a.copy()
            v = np.zeros(n)
            lr = 5.0 / np.sqrt(n)
            # Small perturbation
            rng = np.random.default_rng(it + restart_count)
            a = a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_it = it
            # Reduce step size after restart
            lr *= 0.5

        if verbose and it % 2000 == 0:
            print(f"      NBD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.4f}")

    return best_a

    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(4242)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_k = int(np.argmax(conv))
        max_val = float(conv[max_k])

        # Find contributing pairs for this peak
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)

        if i_start > i_end:
            break

        # Compute contributions
        contribs = []
        for i in range(i_start, i_end + 1):
            j = max_k - i
            if 0 <= j < n:
                contribs.append((a[i] * a[j], i, j))

        if not contribs:
            break

        contribs.sort(reverse=True)

        improved = False
        # Try reducing top contributors
        for cval, src, dst in contribs[:5]:
            if src == dst:
                continue
            if a[src] < 1e-12:
                continue

            # Try moving mass from src to various targets
            deltas = [0.01, 0.005, 0.002]
            for df in deltas:
                delta = df * a[src]
                if delta > a[src] * 0.5:
                    continue

                # Target selection: indices far from the peak
                candidates = [
                    src + max(1, n // 10),
                    src - max(1, n // 10),
                    (n - 1 - src + n) % n,
                    int(rng.integers(0, n)),
                    int(rng.integers(0, n)),
                ]

                for tgt in candidates:
                    tgt = int(tgt) % n
                    if tgt == src or tgt == dst:
                        continue
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[tgt] += delta
                    if a_try[src] < 0:
                        continue
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
            # Perturb
            if stall % 10 == 0:
                a = best_a * np.exp(rng.normal(0, 0.01, n))
                a = np.maximum(a, 0)
                a = a / a.sum()

        if verbose and (time.monotonic() - start) % 10 < 0.1:
            print(f"      PPS: obj={best_obj:.8f}, stall={stall}")

    return best_a


def spectral_init(n, rng, mode=0):
    """Generate spectral-based initializations with low peak autocorrelation."""
    if mode == 0:
        # Random phase -> flat spectrum
        phases = rng.uniform(0, 2 * np.pi, n)
        a = np.cos(phases) + 1.0
        a = np.maximum(a, 0)
    elif mode == 1:
        # Zadoff-Chu inspired
        c = rng.integers(1, n // 2)
        x = np.arange(n)
        phase = 2 * np.pi * c * x * (x + 1) / (2 * n)
        a = np.cos(phase) + 1.0
        a = np.maximum(a, 0)
    elif mode == 2:
        # Hann window
        x = np.arange(n) / (n - 1)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = a + 0.01 * rng.uniform(0, 1, n)
    elif mode == 3:
        # Blackman window
        x = np.arange(n) / (n - 1)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0)
    elif mode == 4:
        # Raised cosine with random width
        w = rng.uniform(0.3, 0.8)
        x = np.arange(n) / (n - 1)
        a = np.cos(np.pi * (x - 0.5) / w) ** 2
        a = np.where(np.abs(x - 0.5) < w, a, 0)
    else:
        # Multi-frequency
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)

    a = a / a.sum()
    return a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        a = rng.power(0.5, n)
        a = a / a.sum()
    else:
        x = np.linspace(0, 1, n)
        a = x ** 0.5 * (1 - x) ** 0.5
        a = a / (a.sum() + 1e-15)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def golden_section_search(a, direction, n_evals=8):
    """Find optimal blend t in [0,1] for a + t*(direction - a)."""
    phi = (np.sqrt(5) - 1) / 2
    lo = 0.0
    hi = 1.0

    best_t = 0.0
    best_val = compute_obj(a)

    for _ in range(n_evals):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)

        a1 = (1 - t1) * a + t1 * direction
        a1 = np.maximum(a1, 0)
        s1 = float(np.sum(a1))
        f1 = compute_obj(a1) if s1 > 1e-10 else np.inf

        a2 = (1 - t2) * a + t2 * direction
        a2 = np.maximum(a2, 0)
        s2 = float(np.sum(a2))
        f2 = compute_obj(a2) if s2 > 1e-10 else np.inf

        if f1 < f2:
            hi = t2
        else:
            lo = t1

    t_best = (lo + hi) / 2
    a_best = (1 - t_best) * a + t_best * direction
    a_best = np.maximum(a_best, 0)
    s_best = float(np.sum(a_best))
    if s_best > 1e-10:
        a_best = a_best / s_best
    return a_best


def propose_candidate(seed=42, budget_s=1500):

    best_score = np.inf

    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent available: {e}")

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Quick LP directions to improve starting point ===
    print("\n=== Phase 1: LP Direction Finding ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.15, 180.0)
    if p1_budget > 20:
        a_lp = two_stage_lp_polish(best_seq, time_budget=p1_budget, verbose=False)

        if obj_lp < best_score:

            print(f"  NBD: obj={obj_nbd:.8f}")

    # === Phase 3: Peak-pair local search ===
    print("\n=== Phase 3: Peak-Pair Search ===")

        a_pp = peak_pair_search(best_seq, time_budget=p3_budget)

        else:
            print(f"  PPS: obj={obj_pp:.8f}")

    # === Phase 4: Second Nesterov pass with higher tau ===
