import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build sparse-friendly matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

        progress = min(1.0, elapsed / time_budget)
        tau_cur = tau * (1 + 5 * progress)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau_cur * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        if prev_g is not None and it % 5 == 0:
            s_vec = a - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.4 * lr + 0.6 * min(max(bb_lr, 1e-8), 20.0 / np.sqrt(n))

        v = momentum * v - lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9995)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def peak_targeted_perturbation(a, rng, scale=0.05):
    """
    Perturb the sequence to specifically reduce the peak convolution values.
    Identifies the top convolution positions and reduces contributions from them.
    """
    n = len(a)
    a = a.copy()
    a_norm = a / float(np.sum(a))
    conv = compute_conv(a_norm)

    # Find top convolution positions
    top_k = np.argsort(conv)[::-1][:8]
    n_top = min(8, len(top_k))

    # For each top position, compute the gradient contribution to a
    delta = np.zeros(n)
    for k in top_k[:n_top]:
        k = int(k)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                delta[j] += a_norm[idx]

    max_d = np.max(np.abs(delta))
    if max_d > 1e-10:
        # Reduce elements that contribute to peaks
        a = a - scale * delta / max_d
        # Also add small random noise to avoid getting stuck
        a = a + scale * 0.3 * rng.normal(0, 1, n)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_perturbation(a, rng, scale=0.03):
    """Random multiplicative perturbation."""
    n = len(a)
    a = a * np.exp(rng.normal(0, scale, n))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def structural_perturbation(a, rng, scale=0.04):
    """Perturb using low-frequency modes (preserves smooth structure)."""
    n = len(a)
    a = a.copy()
    # Add random low-frequency cosine modes
    num_modes = 3
    for _ in range(num_modes):
        freq = rng.integers(1, max(2, n // 20))
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n)
        a = a * (1.0 + scale * np.cos(2 * np.pi * freq * x / n + phase))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        freqs = rng.integers(1, 4, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = x ** rng.uniform(-1, 1) * (1 - x) ** rng.uniform(-1, 1)
        a = a / a.sum()
    elif mode == 5:
        centers = rng.uniform(0.2, 0.8, size=2)
        widths = rng.uniform(0.05, 0.2, size=2)
        a = np.zeros(n)
        x = np.linspace(0, 1, n)
        for c, w in zip(centers, widths):
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Full LP alternation with line search + hybrid smoothed-gradient
    + convolution-aware perturbations + broad multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Quick LP alternation on parent ===
    print(f"\n=== Phase 1: Quick LP Alternation ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 80.0)
    if p1_budget > 15:
        a_sgd = smoothed_gradient_descent(best_seq, time_budget=min(p1_budget * 0.3, 30.0), tau=50.0)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase1 SGD: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        lp_budget = min(remaining * 0.1, 60.0)
        if lp_budget > 10:
            a_lp, obj_lp, iters = lp_alternation(best_seq, max_iter=200, time_limit=lp_budget, verbose=True)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Phase1 LP: obj={best_score:.8f} ({iters} iters) **")
            else:
                print(f"  Phase1 LP: obj={obj_lp:.8f} ({iters} iters)")

    # === Phase 2: Multi-n exploration ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Generate n candidates around the current best and some spread out
    n_candidates = set()
    for delta in [-200, -100, -50, 50, 100, 200, 400, 600, 800, 1000, 1500, 2000]:
        n_try = n_current + delta
        if 300 <= n_try <= 5000:
            n_candidates.add(n_try)
    # Add some absolute values
    for n_abs in [400, 500, 600, 700, 800, 900, 1000, 1200, 1500, 2000, 2500, 3000]:
        if 300 <= n_abs <= 5000:
            n_candidates.add(n_abs)

    n_candidates = [n for n in n_candidates if n != n_current]
    n_candidates.sort(key=lambda x: abs(x - n_current))

    explored = 0
    for n_try in n_candidates[:12]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        if explored >= 8:
            break

        a_try = resample_sequence(best_seq, n_try)
        obj_init = compute_obj(a_try)

        # Quick optimization: a few LP iterations
        alloc = min(remaining * 0.08, 45.0)
        if alloc > 10:
            a_try, obj_try, _ = lp_alternation(a_try, max_iter=50, time_limit=alloc)
        else:
            obj_try = obj_init

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")
        explored += 1

    # === Phase 3: Deep hybrid optimization ===
    print(f"\n=== Phase 3: Deep Hybrid Optimization ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    a_current = best_seq.copy()

    cycle = 0
    while remaining > 30:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Alternate between LP and smoothed gradient
        if cycle % 3 == 0:
            # LP step
            alloc = min(remaining * 0.15, 90.0)
            a_lp, obj_lp, iters = lp_alternation(a_current, max_iter=100, time_limit=alloc)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a_current = a_lp.copy()
                print(f"  P3 cycle{cycle} LP: obj={best_score:.8f} ({iters} iters) **")
            else:
                # Perturb to escape
                if obj_lp < a_current_obj_threshold(a_current, 1e-6):
                    a_current = peak_targeted_perturbation(a_current, rng, scale=0.04)
                else:
                    a_current = random_perturbation(a_current, rng, scale=0.03)
                print(f"  P3 cycle{cycle} LP: obj={obj_lp:.8f} ({iters} iters) [perturb]")
        elif cycle % 3 == 1:
            # Smoothed gradient step
            alloc = min(remaining * 0.1, 60.0)
            tau = 50.0 + 20.0 * (cycle // 3)
            a_sgd = smoothed_gradient_descent(a_current, time_budget=alloc, tau=tau)
            obj_sgd = compute_obj(a_sgd)
            if obj_sgd < best_score:
                best_score = obj_sgd
                best_seq = a_sgd.copy()
                a_current = a_sgd.copy()
                print(f"  P3 cycle{cycle} SGD: obj={best_score:.8f} **")
            else:
                print(f"  P3 cycle{cycle} SGD: obj={obj_sgd:.8f}")
        else:
            # Structural perturbation + quick LP
            a_pert = structural_perturbation(a_current, rng, scale=0.03)
            alloc = min(remaining * 0.08, 40.0)
            if alloc > 10:
                a_pert, obj_pert, _ = lp_alternation(a_pert, max_iter=40, time_limit=alloc)
            else:
                obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                a_current = a_pert.copy()
                print(f"  P3 cycle{cycle} Struct: obj={best_score:.8f} **")
            else:
                print(f"  P3 cycle{cycle} Struct: obj={obj_pert:.8f}")

        cycle += 1

    # === Phase 4: Aggressive escape attempts ===
    print(f"\n=== Phase 4: Escape Attempts ===")
    remaining = deadline - time.monotonic()

    for attempt in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if attempt % 2 == 0:
            a_try = peak_targeted_perturbation(best_seq, rng, scale=0.05)
        else:
            a_try = structural_perturbation(best_seq, rng, scale=0.05)

        # Quick optimization
        alloc = min(remaining * 0.12, 60.0)
        if alloc > 10:
            a_sgd = smoothed_gradient_descent(a_try, time_budget=alloc * 0.4, tau=30.0)
            remaining = deadline - time.monotonic()
            lp_alloc = min(alloc * 0.6, remaining * 0.3)
            if lp_alloc > 10:
                a_try, obj_try, _ = lp_alternation(a_sgd, max_iter=50, time_limit=lp_alloc)
            else:
                a_try = a_sgd
                obj_try = compute_obj(a_try)
        else:
            obj_try = compute_obj(a_try)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  Escape {attempt}: obj={best_score:.8f} **")
        else:
            print(f"  Escape {attempt}: obj={obj_try:.8f}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Multiple rounds of LP with decreasing perturbation
    for polish_rnd in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * 0.25, 80.0)
        if alloc < 10:
            break

        a_cur = best_seq.copy()
        if polish_rnd > 0 and polish_rnd < 4:
            a_cur = random_perturbation(a_cur, rng, scale=0.01 + 0.005 * polish_rnd)

        a_lp, obj_lp, iters = lp_alternation(a_cur, max_iter=200, time_limit=alloc, verbose=(polish_rnd == 0))
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP {polish_rnd}: obj={best_score:.8f} ({iters} iters) **")
        else:
            print(f"  Final LP {polish_rnd}: obj={obj_lp:.8f} ({iters} iters)")

    # Final: one more smoothed gradient with high tau
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 50.0)
        a_sgd = smoothed_gradient_descent(best_seq, time_budget=alloc, tau=200.0)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Final SGD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result


def a_current_obj_threshold(a, eps):
    """Helper: return current objective + epsilon for comparison."""
    return compute_obj(a) + eps
