def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

"""

Hessian-exploiting Newton corrections, momentum subgradient for escaping,
and multi-n exploration. Exploits the structured Hessian of convolution constraints.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix C (2n-1 x n) such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        # Row k, col j: C[k,j] = a[k-j] if 0<=k-j<n
        # For fixed j, k-j ranges from 0 to n-1, so k ranges from j to j+n-1
        # But k must be in [0, 2n-2]
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        C[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return C


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    C = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [C @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def hessian_apply_active(a, active_idx, v):
    """
    Apply the Hessian of the active constraint k to vector v.
    H_k * v has components: (H_k * v)[m] = 2 * v[k-m] (if 0<=k-m<n)
    For multiple active constraints with equal weight, sum them up.
    """
    n = len(a)
    result = np.zeros(n)
    for k in active_idx:
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                result[m] += 2.0 * v[j]
    return result


def find_active_set(conv, max_conv, tol=1e-6):

    At each step: solve LP to get direction, then backtrack to find optimal t.

        return a
    a = a / s

        g_lp, M_lp = solve_lp_direction(a)
        lp_count += 1

        # Check if LP solution is close to current (convergence)

            break

        # Backtracking line search
        best_t = 0.0

def smoothed_subgradient(a, tau):
    """Compute smoothed subgradient of max(conv(a,a)) using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    # Gradient of sum_k p_k * (a*a)[k] with respect to a
    # d/da[m] (a*a)[k] = 2*a[k-m] (if valid)
    # So grad[m] = 2 * sum_k p[k] * a[k-m] (if 0<=k-m<n)
    # This is 2 * (p * a) circular convolution flipped... let's use FFT
    N = 1
    while N < 3 * n:
        N <<= 1
    P = np.fft.fft(p, N)
    A = np.fft.fft(a, N)
    # We need sum_k p[k] * a[k-m] for each m
    # This is the cross-correlation of p and a
    # corr[m] = sum_k p[k] * a[k-m] = ifft(fft(p) * conj(fft(a)))[m] ... 
    # Actually: sum_k p[k] a[k-m] = (p flipped) * a at position... 
    # Let's use: sum_k p[k] a[k-m] = sum_j p[j+m] a[j] = cross_correlation(p, a)[m]
    # In FFT: cross_correlation = ifft(conj(FFT(p)) * FFT(a))
    # Wait, let me think more carefully.
    # We want g[m] = sum_k p[k] * a[k-m] (summed over valid k where 0<=k-m<n)
    # This is the "offset" inner product. 
    # Define b = p reversed: b[i] = p[len(p)-1-i]
    # Then g[m] = sum_j b[j] * a[m+j] (for valid indices) = (b * a)[m] (linear conv)
    # Actually it's simpler to just do:
    # g[m] = sum_{k: 0<=k-m<n} p[k] * a[k-m] = sum_{j: 0<=j<n} p[j+m] * a[j]
    # This is (p shifted by m) dotted with a
    # = sum_j a[j] * p[j+m]
    # This is the cross-correlation of a and p at lag m
    # FFT: ifft(conj(fft(a)) * fft(p)) gives cross-correlation
    g_full = np.fft.ifft(np.conj(np.fft.fft(a, N)) * P).real
    # g_full[m] = sum_j a[j] * p[j+m] (with circular wrapping, but N is large enough)
    # We need the linear version, which is g_full[:n] if N >= 2n
    grad = 2.0 * g_full[:n]
    return grad, conv


def momentum_descent(a, time_budget, tau_init=10.0, tau_end=500.0, beta=0.9, verbose=True):
    """Nesterov momentum smoothed subgradient descent."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    it = 0
    lr = 2.0 / np.sqrt(n)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy < 1e-10:
            y = best_a.copy()
        else:
            y = y / sy

        grad, conv = smoothed_subgradient(y, tau)
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break

        # Adaptive step size
        lr = max(1e-10, min(lr, 1.5 / (g_norm + 1e-10)))

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 500 == 0:
            print(f"    Mom it{it}: obj={best_obj:.10f}, tau={tau:.1f}, lr={lr:.6f}")

        if it > 50000:
            break

    return best_a

    # Perturb phases of mid frequencies
    mid = slice(max(1, n // 20), min(size // 2, 3 * n // 20 + 1))
    if stop_idx := mid.stop > mid.start:
        phase[mid] += strength * rng.uniform(-1, 1, size=mid.stop - mid.start)
        mag[mid] *= (1 + 0.02 * rng.normal(size=mid.stop - mid.start))
        mag[mid] = np.maximum(mag[mid], 0)
    A_new = mag * np.exp(1j * phase)
    a_new = irfft(A_new, size)[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining LP backtracking, active-set Newton, momentum descent,
    and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    # Initialize from parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                a = parent_arr / s
            else:
                a = np.ones(800) / 800
        else:
            a = np.ones(800) / 800
    except Exception:
        a = np.ones(800) / 800

    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    print(f"Initial: n={n}, obj={best_obj:.10f}")

    rng = np.random.default_rng(seed)

    # === Phase 1: Aggressive LP with backtracking (main workhorse) ===
    print(f"\n=== Phase 1: LP Backtracking ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 500.0)
    if p1_budget > 30:
        a1 = lp_polish_backtrack(best_a, time_budget=p1_budget, verbose=True, seed=seed)
        obj1 = compute_obj(a1)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()
            print(f"  Phase1 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase1 result: obj={obj1:.10f} (no improvement)")

    # === Phase 2: Active-set Newton ===
    print(f"\n=== Phase 2: Active-Set Newton ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 250.0)
    if p2_budget > 30:
        a2 = active_set_newton(best_a, time_budget=p2_budget, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            print(f"  Phase2 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase2 result: obj={obj2:.10f} (no improvement)")

    # === Phase 3: LP again (with perturbation if needed) ===
    print(f"\n=== Phase 3: LP Round 2 ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 250.0)
    if p3_budget > 20:
        # Slight perturbation before second LP round
        a_pert = fourier_perturb(best_a, rng, strength=0.01)
        a3 = lp_polish_backtrack(a_pert, time_budget=p3_budget, verbose=True, seed=seed + 100)
        obj3 = compute_obj(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            print(f"  Phase3 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase3 result: obj={obj3:.10f} (no improvement)")
            # Also try without perturbation
            a3b = lp_polish_backtrack(best_a, time_budget=min(p3_budget * 0.5, 60.0),
                                      verbose=False, seed=seed + 200)
            obj3b = compute_obj(a3b)
            if obj3b < best_obj:
                best_obj = obj3b
                best_a = a3b.copy()
                print(f"  Phase3b result: obj={best_obj:.10f} **")

    # === Phase 4: Momentum descent for escaping ===
    print(f"\n=== Phase 4: Momentum Descent ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.1, 100.0)
    if p4_budget > 20:
        a4 = momentum_descent(best_a, time_budget=p4_budget,
                              tau_init=20.0, tau_end=400.0, verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()
            print(f"  Phase4 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase4 result: obj={obj4:.10f} (no improvement)")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    n_options = sorted(set([
        n_best - 50, n_best + 50, n_best + 100,
        n_best + 200, n_best + 400,
        600, 700, 900, 1000, 1100, 1200, 1400
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_best) > 20]
    n_options = n_options[:5]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_a, n_try)
        alloc = min(remaining * 0.15, 80.0)
        if alloc > 20:
            a_try = lp_polish_backtrack(a_try, time_budget=alloc, verbose=False, seed=n_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_obj:.10f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Multiple short LP passes
    for round_idx in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining * 0.25, 80.0)
        if round_idx > 0:
            best_a = fourier_perturb(best_a, rng, strength=0.005 * round_idx)
        a_final = lp_polish_backtrack(best_a, time_budget=alloc, verbose=False,
                                      seed=seed + 1000 + round_idx * 100)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
            print(f"  Final LP {round_idx}: obj={best_obj:.10f} **")
        else:
            print(f"  Final LP {round_idx}: obj={obj_final:.10f}")
            if round_idx >= 2:
                break

    # One more momentum pass
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 60.0)
        a_mom = momentum_descent(best_a, time_budget=alloc,
                                 tau_init=50.0, tau_end=800.0, verbose=False)
        obj_mom = compute_obj(a_mom)
        if obj_mom < best_obj:
            best_obj = obj_mom
            best_a = a_mom.copy()
            print(f"  Final momentum: obj={best_obj:.10f} **")

    # Final validation
    if best_a is None:
        best_a = np.ones(200)
        best_obj = compute_obj(best_a)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")

    return result
