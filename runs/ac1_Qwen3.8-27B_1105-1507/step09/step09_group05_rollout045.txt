```python
"""
Optimizes step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast warm-started LP iterations, Frank-Wolfe fallback,
active-set-aware perturbations, and multi-length exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a, g_warm=None, M_warm=None):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def lp_iteration(a, t=1.0):
    """One LP iteration: solve for g, blend with a."""
    g, M = solve_lp(a)
    if g is None:
        return a, M, 0
    a_new = (1 - t) * a + t * g
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a, M, -1
    return a_new / s, M, 1

def fp_iteration(a, max_iter=200, time_limit=100.0, verbose=False):
    """Fixed-point iteration using LP."""
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    no_improve = 0
    t = 0.5
    
    for it in range(max_iter):
        if time.monotonic() - start > time_limit:
            break
        a_new, M, status = lp_iteration(a, t)
        if status < 0:
            break
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Adapt t
            t = min(1.0, t * 1.1)
        else:
            no_improve += 1
            if no_improve >= 3:
                break
            t = max(0.1, t * 0.7)
    
    return best_a, best_obj

```

```
        Solve for coefficients alpha_0, ..., alpha_{m-1} minimizing:
```

```python
"""
Optimizes step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Anderson-accelerated LP fixed-point iteration, lag-targeted perturbations,
multi-length exploration, and adaptive blending. Key diferencia: Anderson
acceleration for superlinear convergence of the fixed-point map, and
structure-aware perturbations targeting the active constraint lags.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, max_time=2.0):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    start = time.monotonic()
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        if time.monotonic() - start > max_time:
            break
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M_var.value)
        except Exception:
            continue
    return None, None


def anderson_lp_iteration(a, max_iter=500, time_limit=200.0, verbose=False, m_hist=4):
    """
    Anderson-accelerated LP fixed-point iteration.
    f(a) = LP solution g. We want a fixed point a = f(a).
    Anderson acceleration uses history of iterates and residuals.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf

    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Anderson acceleration state
    history_a = [a.copy()]
    history_r = []
    it = 0
    no_improve = 0
    last_t = 0.5

    while it < max_iter and (time.monotonic() - start) < time_limit:
        # Solve LP
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        # Residual: r = a - f(a) = a - g_lp (we want this to be 0)
        r = a - g_lp

        if len(history_a) < m_hist:
            # Simple blending step
            a_new = (1.0 - last_t) * a + last_t * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            history_a.append(a_new.copy())
            history_r.append(r.copy())
        else:
            # Anderson acceleration: find alpha minimizing ||sum alpha_i * r_i||
            # subject to sum(alpha) = 1
            R = np.array(history_r[-m_hist:])  # (m_hist, n)
            # Solve: min ||R^T @ alpha||^2 s.t. 1^T alpha = 1
            # Normal equations: (R R^T) alpha = R 1
            Gram = R @ R.T  # (m_hist, m_hist)
            rhs = R @ np.ones(n)
            # Add constraint via Lagrange: [Gram, 1; 1^T, 0] [alpha; lambda] = [rhs; 1]
            sys_mat = np.zeros((m_hist + 1, m_hist + 1))
            sys_mat[:m_hist, :m_hist] = Gram
            sys_mat[:m_hist, m_hist] = 1.0
            sys_mat[m_hist, :m_hist] = 1.0
            sys_rhs = np.concatenate([rhs, [1.0]])
            try:
                sol = np.linalg.solve(sys_mat + 1e-12 * np.eye(m_hist + 1), sys_rhs)
                alpha = sol[:m_hist]
                alpha = np.maximum(alpha, 0)  # Keep convex combination
                alpha_sum = float(np.sum(alpha))
                if alpha_sum < 1e-10:
                    alpha = np.ones(m_hist) / m_hist
                else:
                    alpha = alpha / alpha_sum
            except np.linalg.LinAlgError:
                alpha = np.ones(m_hist) / m_hist

            # New iterate: convex combination of recent history
            a_new = np.zeros(n)
            for i in range(m_hist):
                a_new += alpha[i] * history_a[-m_hist + i]
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = a.copy()
                s_new = float(np.sum(a_new))
            a_new = a_new / s_new

            history_a.append(a_new.copy())
            history_r.append(r.copy())

            # Trim history
            if len(history_a) > m_hist + 5:
                history_a = history_a[-(m_hist + 1):]
                history_r = history_r[-(m_hist + 1):]

        it += 1

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Adapt step size
            last_t = min(1.0, last_t * 1.05)
        else:
            no_improve += 1
            if no_improve >= 10:
                # Reset Anderson state, try simple blending
                history_a = [a.copy()]
                history_r = []
                no_improve = 0
                last_t = 0.3
            last_t = max(0.1, last_t * 0.8)
            # If very stuck, just use the LP solution directly
            if no_improve >= 5:
                a = g_lp.copy()
                history_a = [a.copy()]
                history_r = []
                no_improve = 0

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    Anderson it{it}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a, best_obj


def simple_lp_iteration(a, max_iter=200, time_limit=100.0, verbose=False):
    """Simple LP fixed-point iteration with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    no_improve = 0
    t = 0.5

    for it in range(max_iter):
        if time.monotonic() - start > time_limit:
            break
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        a_new = (1.0 - t) * a + t * g_lp
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            t = min(1.0, t * 1.08)
        else:
            no_improve += 1
            t = max(0.1, t * 0.75)
            if no_improve >= 5:
                break

        if verbose and it % 30 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t:.2f}")

    return best_a, best_obj


def lag_targeted_perturbation(a, rng, scale=0.05):
    """
    Perturb a by targeting the peak lags.
    For the peak lag k*, reduce a[j]*a[k*-j] for large contributors.
    """
    n = len(a)
    a = a.copy()
    conv = compute_conv(a)
    peak_val = float(np.max(conv))
    peak_lags = np.where(conv > 0.95 * peak_val)[0]

    # For each near-peak lag, compute which pairs contribute most
    deltas = np.zeros(n)
    for k in peak_lags[:10]:
        k = int(k)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                # Contribution of a[j]*a[idx] to conv[k]
                contrib = a[j] * a[idx]
                # Move mass from the larger element
                if contrib > 0:
                    if a[j] > a[idx]:
                        deltas[j] += contrib
                    else:
                        deltas[idx] += contrib

    max_d = np.max(np.abs(deltas))
    if max_d > 1e-15:
        a = a - scale * deltas / max_d
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s

    # Add small random noise to break symmetry
    noise = rng.normal(0, scale * 0.3, n)
    a = a * np.exp(noise)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    return a


def symmetric_perturbation(a, rng, scale=0.03):
    """Perturb with symmetry-aware noise (a[j] and a[n-1-j] perturbed together)."""
    n = len(a)
    a = a.copy()
    half = n // 2
    for j in range(half):
        partner = n - 1 - j
        factor = np.exp(rng.normal(0, scale))
        a[j] *= factor
        a[partner] *= factor
    if n % 2 == 1:
        a[half] *= np.exp(rng.normal(0, scale))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def edge_concentrated_init(n, rng):
    """Initialize with mass concentrated at edges."""
    a = np.zeros(n)
    edge_width = max(1, n // 10)
    a[:edge_width] = 1.0
    a[-edge_width:] = 1.0
    a += rng.uniform(0, 0.3, n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s


def gaussian_init(n, rng, sigma_frac=0.25):
    """Gaussian-like initialization centered."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a += 0.01 * rng.random(n)
    a = np.maximum(a, 0)
    return a / a.sum()


def cosinit(n, rng):
    """Cosine initialization (high in middle)."""
    x = np.linspace(0, np.pi, n)
    a = 0.5 + 0.5 * np.cos(x)
    a += 0.05 * rng.random(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def ring_init(n, rng):
    """Ring initialization: high at 1/4 and 3/4 positions."""
    x = np.arange(n)
    a = np.exp(-((x - n*0.25)**2) / (2*(n*0.08)**2)) + np.exp(-((x - n*0.75)**2) / (2*(n*0.08)**2))
    a += 0.01 * rng.random(n)
    a = np.maximum(a, 0)
    return a / a.sum()


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
