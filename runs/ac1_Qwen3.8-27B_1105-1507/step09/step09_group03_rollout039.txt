def eval_obj_vectorized(conv_full, a, j, t, n):
    delta = t - a[j]
    # Indices of contributing lags
    ks = np.arange(max(0, j), min(2*n-1, j+n))
    k_minus_j = ks - j
    # Linear contribution: 2*delta*a[k_minus_j]
    contrib_linear = 2 * delta * a[k_minus_j]
    # Quadratic correction at k=2j
    b_candidates = conv_full[ks].copy()
    b_candidates += contrib_linear
    # Fix the quadratic term (where k_minus_j == j, i.e., k == 2j)
    quad_idx = np.where(k_minus_j == j)[0]
    if len(quad_idx) > 0:

    # Non-contributing lags
    mask = np.ones(2*n-1, dtype=bool)
    mask[ks] = False
    max_noncontrib = np.max(conv_full[mask]) if mask.any() else 0.0
    return max(np.max(b_candidates), max_noncontrib)

def golden_1d(f, lo, hi, iters=30):
    phi = (np.sqrt(5) - 1) / 2
    a_ = lo + (1-phi)*(hi-lo)
    b_ = lo + phi*(hi-lo)
    fa = f(a_)
    fb = f(b_)
    for _ in range(iters):
        if fa < fb:
            hi = b_
            b_ = a_
            fb = fa
            a_ = lo + (1-phi)*(hi-lo)
            fa = f(a_)
        else:
            lo = a_
            a_ = b_
            fa = fb
            b_ = lo + phi*(hi-lo)
            fb = f(b_)
    return (lo + hi) / 2

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses exact coordinate descent with golden-section 1D subproblem solving,
peak-priority sweep ordering, subgradient-based perturbation, symmetric
variant, and multi-n exploration. Key innovation: exact solution of each
1D coordinate subproblem enables thousands of fast sweeps, extracting
fine-grained improvements over heuristic approaches.
"""

import numpy as np
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    from scipy.fft import rfft, irfft
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def golden_section_min(f, lo, hi, n_iters=28):
    """Golden section search for minimun of f on [lo, hi]. Returns optimal x."""
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = lo + (1.0 - phi) * (hi - lo)
    b_pt = lo + phi * (hi - lo)
    fa = f(a_pt)
    fb = f(b_pt)
    for _ in range(n_iters):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = lo + (1.0 - phi) * (hi - lo)
            fa = f(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + phi * (hi - lo)
            fb = f(b_pt)
    return (lo + hi) / 2.0

def compute_obj_from_conv(n, conv, total):
    """Compute 2n*max(conv)/total^2 from precomputed convolution."""

    return 2.0 * n * float(np.max(conv)) / (total * total)

def peak_priority_order(a, n_golden_check=5):
    """
    Generate sweep order: coordinates contributing most to the peak first.
    """
    n = len(a)
    conv = compute_conv(a)
    peak_k = int(np.argmax(conv))

    # Coordinates that contribute to the peak lag
    i_start = max(0, peak_k - (n - 1))
    i_end = min(n - 1, peak_k)

    # Score each coordinate by its contribution to the peak
    scores = np.zeros(n)
    for i in range(i_start, i_end + 1):
        j = peak_k - i
        if 0 <= j < n:
            scores[i] = a[i] * a[j]
            if i != j:
                scores[j] += a[i] * a[j]

    # Sort: high score first, then fill in the rest
    order = np.argsort(-scores)
    return order


def subgradient_perturb(a, scale, rng):
    """
    Perturb a in the negative subgradient direction of max_k b[k].
    The subgradient at the peak uses the gradient of that lag.
    """
    n = len(a)
    conv = compute_conv(a)
    max_k = int(np.argmax(conv))

    # Gradient of b[max_k] with respect to a[j]:
    # grad[j] = 2 * a[max_k - j] if 0 <= max_k-j < n, else 0
    grad = np.zeros(n)
    j_start = max(0, max_k - (n - 1))
    j_end = min(n - 1, max_k)
    for j in range(j_start, j_end + 1):
        grad[j] = 2.0 * a[max_k - j]

    # Normalize gradient
    g_norm = np.linalg.norm(grad)
    if g_norm > 1e-15:
        grad = grad / g_norm

    # Negative gradient step + random perturbation
    a_new = a - scale * grad
    a_new += scale * 0.1 * rng.standard_normal(n)
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a
    return a_new / s

        #   2*t*a[k-j] for valid k-j, k-j != j
        #   t^2 for k = 2j
        # contributions from position partner:
        #   2*t*a[k-partner] for valid k-partner, k-partner != partner
        #   t^2 for k = 2*partner
        # If j == partner (center of odd-length): only one set

        if j == partner:
            # Center element, only one position
            k_start = max(0, j)
            k_end = min(2 * n - 2, j + n - 1)
            if k_start <= k_end:
                ks = np.arange(k_start, k_end + 1)
                a_kj = a_sym[ks - j].copy()
                base = conv[ks].copy()
                base -= 2.0 * old_val * a_kj
                qm = (ks - j == j)
                if np.any(qm):
                    base[qm] += old_val * old_val

                max_nc = 0.0
                if k_start > 0:
                    max_nc = max(max_nc, float(np.max(conv[:k_start])))
                if k_end < 2 * n - 2:
                    max_nc = max(max_nc, float(np.max(conv[k_end + 1:])))

                def ft(t):
                    b_c = base + 2.0 * t * a_kj
                    if np.any(qm):
                        b_c[qm] = base[qm] + t * t
                    max_b = max(float(np.max(b_c)), max_nc)
                    denom = (S + t) * (S + t)
                    return max_b / denom if denom > 1e-30 else 1e30

                t_hi = max(4.0 * old_val / n, 2.0 / n)
                t_opt = golden_section_min(ft, 0.0, t_hi, n_golden)
                t_opt = max(0.0, t_opt)
                a_sym[j] = t_opt
                # Update conv
                conv[ks] = base + 2.0 * t_opt * a_kj
                if np.any(qm):
                    conv[ks[qm]] = base[qm] + t_opt * t_opt
                total = S + t_opt
            continue

        # Two positions: j and partner
        # Collect contributions from both
        all_ks = set()
        # From j
        ks_j = np.arange(max(0, j), min(2 * n - 1, j + n))
        # From partner
        ks_p = np.arange(max(0, partner), min(2 * n - 1, partner + n))
        all_ks = np.unique(np.concatenate([ks_j, ks_p]))

        base = conv[all_ks].copy()
        # Remove old contributions
        for k in range(len(all_ks)):
            kk = all_ks[k]
            # From j
            if j <= kk < j + n:
                kmj = kk - j
                if kmj == j:
                    base[k] += old_val * old_val  # undo t^2, add back
                    base[k] -= 2.0 * old_val * a_sym[kmj]  # already subtracted
                else:
                    base[k] -= 2.0 * old_val * a_sym[kmj]
            # From partner
            if partner <= kk < partner + n:
                kmp = kk - partner
                if kmp == partner:
                    base[k] += old_partner * old_partner
                    base[k] -= 2.0 * old_partner * a_sym[kmp]
                else:
                    base[k] -= 2.0 * old_partner * a_sym[kmp]

        # Now base[k] is conv with a[j]=0, a[partner]=0
        # f(t): set a[j]=a[partner]=t
        # contribution at k:
        #   from j: 2*t*a[k-j] (if valid, k-j!=j) or t^2 (if k=2j)
        #   from partner: 2*t*a[k-partner] (if valid, k-partner!=partner) or t^2 (if k=2*partner)

        # Precompute
        contrib_coef = np.zeros(len(all_ks))
        quad_count = np.zeros(len(all_ks), dtype=int)
        for idx_k, kk in enumerate(all_ks):
            if j <= kk < j + n:
                kmj = int(kk) - j
                if kmj == j:
                    quad_count[idx_k] += 1
                else:
                    contrib_coef[idx_k] += 2.0 * a_sym[kmj]
            if partner <= kk < partner + n:
                kmp = int(kk) - int(partner)
                if kmp == partner:
                    quad_count[idx_k] += 1
                else:
                    contrib_coef[idx_k] += 2.0 * a_sym[kmp]

        max_nc = 0.0
        all_conv_ks = np.arange(2 * n - 1)
        nc_mask = np.isin(all_conv_ks, all_ks) == False
        if np.any(nc_mask):
            max_nc = float(np.max(conv[all_conv_ks[nc_mask]]))

        def ft2(t):
            b_c = base + contrib_coef * t
            for idx_k in range(len(all_ks)):
                if quad_count[idx_k] > 0:
                    b_c[idx_k] += quad_count[idx_k] * t * t
            max_b = max(float(np.max(b_c)), max_nc)
            denom = (S + 2.0 * t) * (S + 2.0 * t)
            return max_b / denom if denom > 1e-30 else 1e30

        t_hi = max(4.0 * old_val / n, 2.0 / n)
        t_opt = golden_section_min(ft2, 0.0, t_hi, n_golden)
        t_opt = max(0.0, t_opt)
        a_sym[j] = t_opt
        a_sym[partner] = t_opt
        total = S + 2.0 * t_opt

        # Update conv
        conv[all_ks] = base + contrib_coef * t_opt
        for idx_k in range(len(all_ks)):
            if quad_count[idx_k] > 0:
                conv[all_ks[idx_k]] += quad_count[idx_k] * t_opt * t_opt

    a_sym = np.maximum(a_sym, 0.0)
    a_sym = a_sym / float(np.sum(a_sym))
    return a_sym


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        # Frequency-domain flat spectrum approach
        n_fft = 2 * n
        phase = rng.uniform(0, 2 * np.pi, n_fft)
        mag = np.ones(n_fft)
        # Shape magnitude to be roughly flat
        spectrum = mag * np.exp(1j * phase)
        a = np.real(np.fft.ifft(spectrum))[:n]
        a = np.maximum(a, 0)
        if a.sum() < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / a.sum()
    elif mode == 5:
        # Raised cosine window
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = a / a.sum()
    elif mode == 6:
        # Two-level sequence
        levels = [1.0, 0.3]
        a = rng.choice(levels, n).astype(float)
        # Ensure enough of each level
        n_high = max(n // 4, 1)
        idx = rng.permutation(n)
        a[idx[:n_high]] = levels[0]
        a[idx[n_high:]] = levels[1]
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def fast_cd_loop(a, time_budget, order_type="mixed", n_golden=25, verbose=False):
    """
    Run multiple coordinate descent sweeps until time budget is exhausted.
    Returns best a found.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    sweep_count = 0
    stall = 0
    order_type_cycle = ["forward", "reverse", "random", "peak"]
    order_idx = 0

    while time.monotonic() - start < time_budget:
        # Choose sweep order
        ot = order_type_cycle[order_idx % len(order_type_cycle)]
        order_idx += 1

        if ot == "forward":
            order = np.arange(n)
        elif ot == "reverse":
            order = np.arange(n - 1, -1, -1)
        elif ot == "random":
            order = rng.permutation(n)
        else:  # peak
            try:
                order = peak_priority_order(a)
            except:
                order = rng.permutation(n)

        # Run sweep
        a_new, conv_new, _ = cd_sweep(a, order, n_golden=n_golden)
        obj_new = compute_obj(a_new)

        sweep_count += 1

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and sweep_count % 50 == 0:
                elapsed = time.monotonic() - start
                print(f"    CD sweep {sweep_count}: obj={best_obj:.8f}, t={elapsed:.1f}s")
        else:
            stall += 1
            a = best_a.copy()

            if stall > 15:
                # Perturb using subgradient
                scale = min(0.005 + 0.001 * stall, 0.05)
                a = subgradient_perturb(best_a, scale, rng)
                stall = 0

                # Try a few more sweeps after perturbation
                for s2 in range(5):
                    a_new2, _, _ = cd_sweep(a, rng.permutation(n), n_golden=15)
                    obj_s2 = compute_obj(a_new2)
                    if obj_s2 < best_obj:
                        best_obj = obj_s2
                        best_a = a_new2.copy()
                        a = a_new2
                        break

                if stall > 40:
                    # Bigger perturbation
                    scale = 0.05
                    a = subgradient_perturb(best_a, scale, rng)
                    # Or random restart blend
                    blend = 0.2
                    a_rand = diverse_init(n, rng, mode=rng.integers(0, 7))
                    a = (1 - blend) * best_a + blend * a_rand
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    stall = 0

                if stall > 80:
                    break

    return best_a


def symmetric_cd_loop(a, time_budget, n_golden=20, verbose=False):
    """Run symmetric coordinate descent sweeps."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    sweep_count = 0
    stall = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        a_new = symmetric_cd_sweep(a, n_golden)
        obj_new = compute_obj(a_new)
        sweep_count += 1

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and sweep_count % 30 == 0:
                elapsed = time.monotonic() - start
                print(f"    SymCD sweep {sweep_count}: obj={best_obj:.8f}, t={elapsed:.1f}s")
        else:
            stall += 1
            a = best_a.copy()
            if stall > 10:
                a = subgradient_perturb(best_a, 0.01, rng)
                a = (a + a[::-1]) / 2.0  # Re-symmetrize
                a = a / a.sum()
                stall = 0
            if stall > 40:
                break

    return best_a


def lp_direction(a, n_active=10):
    """
    Solve min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0 using a simple
    projected subgradient approach (avoids cvxpy for speed).
    Returns a direction g on the simplex.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))

    # Build the linear operator: conv(a, g) = A @ g
    # A[k, j] = a_norm[k-j] if 0 <= k-j < n, else 0
    # This is a (2n-1) x n matrix (Toeplitz-like)
    # But for n up to 1500, this is 3000 x 1500 = 4.5M entries. Manageable.

    # Use iterative method: Frank-Wolfe (conditional gradient)
    # Minimize max_k (A@G)[k] over G in simplex
    # Equivalent to: minimize M s.t. A@G <= M, sum(G)=1, G>=0

    g = np.ones(n) / n  # Start at center of simplex
    t = 0.3
    best_g = g.copy()
    best_M = np.inf

    n_iter = 200
    for it in range(n_iter):
        # Compute current conv values
        conv_ag = np.convolve(a_norm, g)
        M_cur = float(np.max(conv_ag))
        if M_cur < best_M:
            best_M = M_cur
            best_g = g.copy()

        # Find the vertex of the simplex that minimizes the linearized objective
        # Linearize: minimize grad^T (g - g0) where grad is subgradient of max
        peak_k = int(np.argmax(conv_ag))
        # Subgradient: A[peak_k, :] = a_norm[peak_k - j] for valid j
        grad_k = np.zeros(n)
        j_start = max(0, peak_k - (n - 1))
        j_end = min(n - 1, peak_k)
        for j in range(j_start, j_end + 1):
            grad_k[j] = a_norm[peak_k - j]

        # Also include nearby peaks for robustness
        top_kinds = np.argsort(-conv_ag)[:n_active]
        for kk in top_kinds:
            j_s = max(0, int(kk) - (n - 1))
            j_e = min(n - 1, int(kk))
            for j in range(j_s, j_e + 1):
                grad_k[j] += a_norm[int(kk) - j]

        # Frank-Wolfe: pick vertex e_j that minimizes grad_k[j]
        j_min = int(np.argmin(grad_k))
        vertex = np.zeros(n)
        vertex[j_min] = 1.0

        # Update
        g = (1 - t) * g + t * vertex
        t = max(0.1, 2.0 / (it + 2.0))

    return best_g


def lp_polish(a, time_budget=60.0, verbose=False):
    """
    LP-based polishing using Frank-Wolfe to find direction, then blend.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    it = 0
    while time.monotonic() - start < time_budget:
        g_dir = lp_direction(a, n_active=8)
        if g_dir is None:
            break

        # Line search: minimize obj((1-t)*a + t*g_dir)
        best_t = 0.0
        best_blend_obj = best_obj
        for t_candidate in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t_candidate) * a + t_candidate * g_dir
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_t = t_candidate

        if best_t > 0:
            a_new = (1 - best_t) * a + best_t * g_dir
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            if best_blend_obj < best_obj - 1e-14:
                best_obj = best_blend_obj
                best_a = a_new.copy()
                a = a_new
        else:
            a = subgradient_perturb(best_a, 0.02, rng)

        it += 1
        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}")

    return best_a
