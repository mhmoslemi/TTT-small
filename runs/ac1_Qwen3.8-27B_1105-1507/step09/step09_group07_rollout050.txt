"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    m = 2 * n - 1

    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def fast_conv_pair(a, g):
    """Compute conv(a, g) via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g, size)
    return irfft(fa * fg, size)[:2 * n - 1]


def peak_directed_descent(a, time_budget, verbose=False):
    """
    Direct gradient descent on the log-sum-exp smoothed objective.
    Uses momentum and adaptive learning rate.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.5 / np.sqrt(n)
    momentum = np.zeros(n)
    beta_m = 0.9
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Log-sum-exp with adaptive beta
        it_frac = min(1.0, it / 100)
        beta = 10.0 + 200.0 * it_frac

        m = conv.max()
        shifted = beta * (conv - m)
        # Clip to avoid overflow
        shifted = np.clip(shifted, -500, 0)
        weights = np.exp(shifted)
        weights /= weights.sum()

        # Gradient of log-sum-exp(conv(a,a)) w.r.t. a
        # d/d a[j] log(sum_k exp(beta*(conv[k]-m))) = beta * sum_k weights[k] * d conv[k]/d a[j]
        # d conv[k] / d a[j] = a[k-j] + a[j-k] (for valid indices)
        # This is equivalent to: grad[j] = beta * (weights_cross_corr)[j]
        # where weights_cross_corr[j] = sum_k weights[k] * (a[k-j] + a[j-k])

        # Compute via FFT: grad = beta * 2 * ifft(fft(weights) * conj(fft(a)))
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        fw = rfft(weights, size)
        fa = rfft(a, size)
        # cross-correlation
        cc = irfft(fw * np.conj(fa), size)[:n]
        # Also need the "flipped" part
        # Actually for conv(a,a), d/d a[j] = sum_k (da/da[j] * conv[k]) where
        # d conv[k]/d a[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
        # = cross-correlation of a with a at shift j
        # The full gradient is: grad[j] = beta * sum_k weights[k] * (a[(k-j)%valid] + a[(j-k)%valid])
        # = beta * 2 * (weights * a)_cross[j] (by symmetry of conv)
        # Actually let me just compute it directly using the relationship:
        # conv(a,a) = ifft(fft(a)^2), so d conv/d a = ifft(2*fft(a)*grad_a_in_freq)
        # equivalently: grad_in_time = 2 * ifft(fft(a) * fft(weights_shifted))

        # Simpler: grad = beta * (a * weights_correlated) where we take the circular cross-correlation
        # Let me just use: grad[j] = beta * 2 * sum_k weights[k] * a[(k-j) mod (2n-1)]
        # This is the cross-correlation of weights with a (padded)

        # Most efficient: use FFT
        # pad weights to length 2n-1
        w_padded = np.zeros(2 * n - 1)
        w_padded[:len(weights)] = weights
        # cross-correlation: cc[j] = sum_k w_padded[k] * a_padded[k-j]
        # = sum_k w_padded[k] * a[(k-j)] where a is padded
        a_padded = np.zeros(2 * n - 1)
        a_padded[:n] = a

        # Use FFT for cross-correlation
        size2 = 1
        while size2 < 4 * n - 3:
            size2 <<= 1
        fwp = rfft(w_padded, size2)
        fa_padded = rfft(a_padded, size2)
        # cross-correlation: ifft(fw * conj(fa))
        cc_full = irfft(fwp * np.conj(fa_padded), size2)
        # The cross-correlation at lag j is cc_full[j] (for j >= 0) and cc_full[size2-j] (for j < 0)
        # We want grad[j] = beta * (cc_full[j] + cc_full[size2-j]) for j = 0..n-1
        # which is beta * 2 * cc_full[j] by symmetry (since both are real)
        grad = beta * 2.0 * cc_full[:n]

        # Apply momentum
        momentum = beta_m * momentum + (1 - beta_m) * grad
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = min(lr * 1.01, 5.0)
        else:
            lr *= 0.9

        it += 1
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"          PDD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.4f}, time={elapsed:.1f}s")

    return best_a

def targeted_peak_reduction(a, time_budget, verbose=False):
    """
    Targeted reduction of the maximum convolution peak.
    Identifies the peak lag and finds optimal 2-swap moves to reduce it.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0

    while time.monotonic() - start < time_budget:
        round_num += 1
        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]

        # Find which pairs (i, j) contribute most to this peak
        # conv[peak_idx] = sum_i a[i] * a[peak_idx - i] for valid i
        contributors = []
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)

        # Find "safe" positions: positions that don't contribute much to the peak
        # and where adding mass won't create a new peak
        peak_lags = np.argsort(conv)[::-1][:5]
        danger = np.zeros(n)
        for k in peak_lags:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    danger[i] += a[j]
                    danger[j] += a[i]
        safe_positions = np.argsort(danger)[:10]

        improved = False
        for contrib, i, j in contributors[:5]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-10:
                continue
            src = i if a[i] >= a[j] else j
            for target in safe_positions:
                if target == src:
                    continue
                # Try moving a small fraction from src to target
                for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                    delta = frac * a[src]
                    if delta <= 0 or delta >= 0.9 * a[src]:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_peak = float(np.max(new_conv))
                    new_obj = 2.0 * n * new_peak
                    if new_obj < best_obj - 1e-10:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break
            if improved:
                break

        if not improved:
            if round_num > 5:
                break
            # Try larger moves
            for i in range(n):
                if time.monotonic() - start > time_budget:
                    break
                for j in range(n):
                    if i == j:
                        continue
                    for frac in [0.1, 0.2, 0.3]:
                        delta = frac * a[i]
                        if delta > 0.9 * a[i]:
                            continue
                        a[i] -= delta
                        a[j] += delta
                        a = np.maximum(a, 0)
                        s = float(np.sum(a))
                        if s < 1e-10:
                            a[i] += delta
                            a[j] -= delta
                            continue
                        a = a / s
                        obj = compute_obj(a)
                        if obj < best_obj - 1e-10:
                            best_obj = obj
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[i] += delta
                            a[j] -= delta
                            a = np.maximum(a, 0)
                            a = a / a.sum()
                    if improved:
                        break
                if improved:
                    break

        if verbose and round_num % 20 == 0:
            print(f"          Targeted r{round_num}: obj={best_obj:.8f}, peak_idx={peak_idx}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode % 10 == 0:
        a = np.ones(n) / n
    elif mode % 10 == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode % 10 == 2:
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 3:
        # Inverted U (peaked in middle, small at edges)
        a = np.exp(-4 * (x - 0.5) ** 2) + 0.05
        a = a / a.sum()
    elif mode % 10 == 4:
        # U-shaped (peaked at edges)
        a = np.exp(-4 * (x - 0.5) ** 2) * 0.1 + 1.0
        a = a / a.sum()
    elif mode % 10 == 5:
        # Two bumps
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2) + 0.05
        a = a / a.sum()
    elif mode % 10 == 6:
        # Frequency-based
        freqs = rng.integers(2, 10, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 10 == 7:
        # Power law
        a = x**2 + (1-x)**2 + 0.1
        a = a / a.sum()
    elif mode % 10 == 8:
        # Dirichlet
        a = rng.dirichlet(np.ones(n) * 0.3)
    else:
        # Flat with noise
        a = np.ones(n) + rng.normal(0, 0.3, n)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase strategy:
    Phase 1: Quick LP polish on parent (10%)
    Phase 2: n-sweep with quick optimization (15%)
    Phase 3: Intensive optimization on best n (40%)
    Phase 4: Peak-directed + targeted refinement (20%)
    Phase 5: Diverse restarts + symmetric search (10%)
    Phase 6: Final polish (5%)
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"\nInitial: n={n_current}, obj={best_score:.8f}, target=1.5027")

    # === Phase 1: Quick LP polish (10%) ===
    print(f"\n=== Phase 1: Quick LP Polish ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.10, 80.0)
    if p1_budget > 15:
        a_p1 = lp_alternate_v2(best_seq, time_budget=p1_budget, verbose=True)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_p1:.8f}")

    # === Phase 2: n-sweep (15%) ===
    print(f"\n=== Phase 2: n-Sweep ===")
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.15

    n_options = sorted(set([
        600, 640, 700, 720, 750, 768, 780, 800, 820, 840,
        850, 860, 880, 900, 920, 960, 1000, 1024, 1050, 1100, 1200,
        n_current - 50, n_current + 50,
        n_current - 100, n_current + 100,
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1500]

    per_n_budget = p2_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_n_budget, remaining * 0.1)
        if alloc < 8:
            alloc = 8

        a_try = resample_sequence(best_seq, n_try)
        a_opt = lp_alternate_v2(a_try, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)
        marker = " **" if obj_opt < best_score else ""
        if obj_opt < best_score or n_try == n_options[0]:
            print(f"  n={n_try}: obj={obj_opt:.8f}{marker}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try

    # === Phase 3: Intensive optimization (40%) ===
    print(f"\n=== Phase 3: Intensive Optimization (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.40

    # 3a: LP alternating (50% of phase 3)
    alloc_3a = p3_budget * 0.50
    if alloc_3a > 20:
        a_3a = lp_alternate_v2(best_seq, time_budget=alloc_3a, verbose=True)
        obj_3a = compute_obj(a_3a)
        if obj_3a < best_score:
            best_score = obj_3a
            best_seq = a_3a.copy()
            print(f"  LP-v2: obj={best_score:.8f} **")

    # 3b: Peak-directed descent (25% of phase 3)
    remaining = deadline - time.monotonic()
    alloc_3b = min(p3_budget * 0.25, remaining * 0.3)
    if alloc_3b > 20:
        a_3b = peak_directed_descent(best_seq, time_budget=alloc_3b, verbose=True)
        obj_3b = compute_obj(a_3b)
        if obj_3b < best_score:
            best_score = obj_3b
            best_seq = a_3b.copy()
            print(f"  PDD: obj={best_score:.8f} **")

    # 3c: Symmetric LP alternating (25% of phase 3)
    remaining = deadline - time.monotonic()
    alloc_3c = min(p3_budget * 0.25, remaining * 0.3)
    if alloc_3c > 20:
        a_3c = symmetric_lp_alternate(best_seq, time_budget=alloc_3c, verbose=True)
        obj_3c = compute_obj(a_3c)
        if obj_3c < best_score:
            best_score = obj_3c
            best_seq = a_3c.copy()
            print(f"  SymLP: obj={best_score:.8f} **")

    # 3d: Another LP round with perturbation
    remaining = deadline - time.monotonic()
    alloc_3d = min(remaining * 0.25, 60.0)
    if alloc_3d > 20:
        rng_3d = np.random.default_rng(12345)
        a_pert = best_seq * np.exp(rng_3d.normal(0, 0.03, n_current))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        a_3d = lp_alternate_v2(a_pert, time_budget=alloc_3d, verbose=False)
        obj_3d = compute_obj(a_3d)
        if obj_3d < best_score:
            best_score = obj_3d
            best_seq = a_3d.copy()
            print(f"  LP-pert: obj={best_score:.8f} **")
        else:
            print(f"  LP-pert: obj={obj_3d:.8f}")

    # === Phase 4: Targeted refinement (20%) ===
    print(f"\n=== Phase 4: Targeted Refinement ===")
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.20

    # 4a: Targeted peak reduction
    alloc_4a = p4_budget * 0.50
    if alloc_4a > 15:
        a_4a = targeted_peak_reduction(best_seq, time_budget=alloc_4a, verbose=True)
        obj_4a = compute_obj(a_4a)
        if obj_4a < best_score:
            best_score = obj_4a
            best_seq = a_4a.copy()
            print(f"  Targeted: obj={best_score:.8f} **")
        else:
            print(f"  Targeted: obj={obj_4a:.8f}")

    # 4b: More LP after targeted
    remaining = deadline - time.monotonic()
    alloc_4b = min(p4_budget * 0.50, remaining * 0.3)
    if alloc_4b > 15:
        a_4b = lp_alternate_v2(best_seq, time_budget=alloc_4b, verbose=True)
        obj_4b = compute_obj(a_4b)
        if obj_4b < best_score:
            best_score = obj_4b
            best_seq = a_4b.copy()
            print(f"  LP-post: obj={best_score:.8f} **")

    # === Phase 5: Diverse restarts + symmetric (10%) ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p5_budget = remaining * 0.10
    n_use = len(best_seq)
    n_restarts = 3
    per_restart = p5_budget / max(1, n_restarts)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_restart, remaining * 0.2)
        if alloc < 15:
            break

        mode = (r * 3 + 3) % 10
        a_r = diverse_init(n_use, rng, mode=mode)
        # Blend with best
        blend = 0.4 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        # Quick optimization
        half = alloc * 0.6
        if half > 10:
            a_r = lp_alternate_v2(a_r, time_budget=half, verbose=False)
        remaining = deadline - time.monotonic()
        lp_alloc = min(remaining * 0.1, alloc * 0.4)
        if lp_alloc > 10:
            a_r = peak_directed_descent(a_r, time_budget=lp_alloc, verbose=False)

        obj_r = compute_obj(a_r)
        marker = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}{marker}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()

    # === Phase 6: Final polish (5%) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # One more LP
    alloc = min(remaining * 0.5, 40.0)
    if alloc > 15:
        a_final = lp_alternate_v2(best_seq, time_budget=alloc, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Targeted refinement
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.3, 20.0)
    if alloc > 10:
        a_targ = targeted_peak_reduction(best_seq, time_budget=alloc, verbose=False)
        obj_targ = compute_obj(a_targ)
        if obj_targ < best_score:
            best_score = obj_targ
            best_seq = a_targ.copy()
            print(f"  Final Targeted: obj={best_score:.8f} **")

    # One last quick LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = lp_alternate_v2(best_seq, time_budget=min(remaining * 0.5, 20.0), verbose=False)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    # Round to avoid floating point issues
    best_seq = np.maximum(best_seq, 0.0)
    result = [float(x) for x in best_seq]

    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"=== Target: 1.5027, Achieved: {best_score:.8f}, Gap: {best_score - 1.5027:.8f} ===")

    return result
