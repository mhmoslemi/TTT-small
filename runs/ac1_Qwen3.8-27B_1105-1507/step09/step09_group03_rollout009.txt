def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_max_conv(a, top_k=5, tol=0.0):
    """
    Compute subgradient of max_k (a*a)[k] with respect to a.
    Uses the top_k active constraints.
    Returns gradient vector of size n.
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    
    # Find active indices
    if tol == 0.0:
        active = np.where(conv >= max_val - 1e-10)[0]
    else:
        active = np.where(conv >= max_val - tol)[0]
    
    if len(active) > top_k:
        # Keep only top_k
        sorted_idx = np.argsort(conv)[::-1][:top_k]
        active = sorted_idx
    
    # Compute subgradient
    # d/d(a[j]) (a*a)[k] = a[(k-j) mod ...] handling bounds
    # For linear convolution: (a*a)[k] = sum_i a[i]*a[k-i] where 0<=i<n and 0<=k-i<n
    # d/d(a[j]) = a[k-j] (if 0<=k-j<n) + a[k-j] (same term counted twice if k-2j=0... no)
    # Actually: d/d(a[j]) sum_i a[i]*a[k-i] 
    #   = (if 0<=k-j<n) a[k-j] + (if 0<=j<n, which is always true, and 0<=k-j<n) a[j]
    # Wait, let me be more careful.
    # (a*a)[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i] * a[k-i]
    # d/d(a[j]) = sum_{i: 0<=i<n, 0<=k-i<n, i=j} a[k-i] + sum_{i: 0<=i<n, 0<=k-i<n, k-i=j} a[i]
    # = (if 0<=k-j<n) a[k-j] + (if 0<=k-j<n) a[k-j]  [since i=k-j must satisfy 0<=k-j<n]
    # Wait, the second sum: k-i=j means i=k-j. The condition is 0<=k-j<n and 0<=k-(k-j)=j<n (always true).
    # So both terms require 0<=k-j<n, and both give a[k-j] (first) and a[j] (second).
    # So d/d(a[j]) (a*a)[k] = (if 0<=k-j<n) [a[k-j] + a[j]]
    # But wait, if k-j == j (i.e., k=2j), then both terms are the same, so it's 2*a[j].
    # Actually no: first term is a[k-j] = a[j], second term is a[j]. So it's a[j] + a[j] = 2*a[j]. Correct.
    
    grad = np.zeros(n)
    for k in active:
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += a[idx] + a[j]
    
    # Average over active set (for a "smoothed" subgradient)
    grad /= len(active)
    return grad, conv, active

def smoothed_subgradient_fft(a, tau, n_evals=None):
    """
    Compute smoothed gradient of log-sum-exp(tau * conv) using FFT.
    Returns (gradient, conv, weights).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    # Gradient of sum_k p[k] * (a*a)[k] with respect to a
    # = sum_k p[k] * d(a*a)[k]/da
    # = sum_k p[k] * (shift of a)
    # Using FFT: the gradient is related to the cross-correlation of p with a
    
    # d/d(a[j]) sum_k p[k] * (a*a)[k]
    # = sum_k p[k] * (a[k-j] + a[j]) for valid k-j
    # = sum_k p[k] * a[k-j] + sum_k p[k] * a[j] (for valid k-j)
    # First term: (p * a)[j] convolved appropriately
    # Second term: a[j] * sum_{k: 0<=k-j<n} p[k]
    
    # Let's use the FFT approach directly
    # The "unconstrained" gradient (ignoring bounds) of sum_k p[k] * (a*a)[k]
    # with respect to a is: 2 * ifft(fft(p_padded) * conj(fft(a_padded)))[:n]
    # where p_padded is p zero-padded to appropriate size
    
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    grad = 2.0 * g_full[:n]
    
    return grad, conv, p

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Anti-symmetric bias
        x = np.linspace(-1, 1, n)
        a = 1.0 + 0.3 * x  # Linear ramp (anti-symmetric component)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 3:
        # Quadratic anti-symmetric
        x = np.linspace(-1, 1, n)
        a = 1.0 + 0.2 * x**2 * np.sign(x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Cosine modulation
        x = np.linspace(0, 2*np.pi, n)
        a = 1.0 + 0.15 * np.sin(x) + 0.05 * np.sin(2*x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Exponential ramp
        a = np.exp(np.linspace(-1, 1, n))
        a = a / a.sum()
    elif mode == 6:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * (n//10) ** 2))
        a = a / a.sum()
    else:
        a = 1.0 / (1.0 + 0.1 * np.sin(2 * np.pi * np.arange(n) / n))
        a = a / a.sum()
    return a

def multi_scale_refine(a, time_budget=100.0, verbose=False):
    """
    Coarse-to-fine refinement:
    1. Downsample the sequence
    2. Optimize at coarse level
    3. Upsample and optimize at fine level
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Coarse level
    coarse_n = max(50, n // 8)
    a_coarse = resample_sequence(a, coarse_n)
    
    # Optimize at coarse level
    remaining = deadline_safe()  # We'll pass time budget
    
    # Actually, let's just do a few iterations of subgradient at coarse level
    for _ in range(min(200, int(time_budget / 5))):
        if time.monotonic() - start > time_budget * 0.4:
            break
        tau = 30.0
        grad, conv, p = smoothed_subgradient_fft(a_coarse, tau)
        lr = 3.0 / np.sqrt(coarse_n)
        a_coarse = a_coarse - lr * grad
        a_coarse = np.maximum(a_coarse, 0)
        s_c = float(np.sum(a_coarse))
        if s_c > 1e-10:
            a_coarse = a_coarse / s_c
    
    # Upsample back
    a_new = resample_sequence(a_coarse, n)
    a_new = a_new / a_new.sum()
    
    # Blend with original
    for alpha in [1.0, 0.8, 0.6, 0.4]:
        a_try = (1 - alpha) * best_a + alpha * a_new
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try > 1e-10:
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                break
    
    # Fine-level optimization
    if time.monotonic() - start < time_budget * 0.7:
        remaining_fine = time_budget - (time.monotonic() - start)
        a_fine = long_lookahead_subgradient(best_a, time_budget=remaining_fine * 0.8)
        obj_fine = compute_obj(a_fine)
        if obj_fine < best_obj:
            best_obj = obj_fine
            best_a = a_fine.copy()
    
    return best_a
