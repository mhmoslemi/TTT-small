import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy import sparse
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_conv_cross(a, b):
    """Compute conv(a, b) for possibly different lengths (but we'll use same length)."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix_sparse(a):
    """Build sparse matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # Column j: A[k, j] = a[k-j] for k where 0 <= k-j < n
        # k ranges from j to j+n-1, but clipped to [0, 2n-2]
        k_start = j
        k_end = min(j + n, 2*n - 1)
        for k in range(k_start, k_end):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def solve_lp(a, solver_list=None):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_conv_matrix_sparse(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:

            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(g_val.sum())
                if s > 1e-10:

        return 2.0 * n * float(np.max(val))
    
    # Golden section search on [0, 1]

    a_gs, b_gs = 0.0, 1.0
    c_gs = b_gs - gr * (b_gs - a_gs)

    fc = f(c_gs)

    best_a = a.copy()
    
    for t in np.linspace(0, 1, 50):
        val = (1-t)**2 * c_aa + 2*t*(1-t) * c_ag + t**2 * c_gg
        obj = 2.0 * n * float(np.max(val))
        if obj < best_val:
            best_val = obj
            best_t = t
            best_a = (1 - t) * a + t * g
            best_a = np.maximum(best_a, 0)
            s = float(best_a.sum())
            if s > 1e-10:
                best_a = best_a / s
    
    return best_a, best_t, best_val
```

This is simple and effective. 50 evaluations * O(n) each = O(50n) which is fine.

Now, let me think about the overall structure more carefully.

The key question is: **how many LP iterations can we do in the time budget, and will they make enough progress?**

Each LP iteration:
- Build sparse matrix: O(n^2) = O(640K) for n=800
- Solve LP: ~1-3 seconds (depends on solver)
- Line search: ~0.01 seconds
- Total: ~1-3 seconds

In 1000 seconds (leaving 500s for other phases), we can do ~300-1000 iterations.

Will each iteration make progress? Not necessarily. The LP finds a direction that minimizes the cross-convolution peak, but the self-convolution peak might not decrease.

To ensure progress, we need:
1. Good initial point (parent is good)
2. Multiple LP solves with different references
3. Perturbation to escape local minima
4. Accepting only improving moves (or occasional non-improving moves for exploration)

Let me also add a **direct objective-based step**: after the LP iteration, if we're at a point where the conv peak is at a specific set of positions, we can try to directly reduce those by adjusting the sequence.

- Wait, for j=0: rows 0 to n-1 (n nonzeros)
- For j=n-1: rows n-1 to 2n-2 (n nonzeros)

```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses intensive LP-based iteration with line search, coordinate descent polishing,
and multi-length exploration. Key innovation: dense-grid line search on the
blended objective for better step size selection, and adaptive perturbation
to escape local minima.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy import sparse
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        for k in range(k_start, k_end):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp(a, solvers=None):
    n = len(a)
    A = build_conv_matrix_sparse(a)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    if solvers is None:
        solvers = [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(g_val.sum())
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g, c_aa, c_ag, c_gg, n, num_points=50):
    best_t = 0.0
    best_val = np.inf
    best_a = a.copy()
    ts = np.linspace(0, 1, num_points)
    for t in ts:
        val = (1 - t) ** 2 * c_aa + 2 * t * (1 - t) * c_ag + t ** 2 * c_gg
        obj = 2.0 * n * float(np.max(val))
        if obj < best_val:
            best_val = obj
            best_t = t
            best_a = (1 - t) * a + t * g
            best_a = np.maximum(best_a, 0)
            s = float(best_a.sum())
            if s > 1e-10:
                best_a = best_a / s
    return best_a, best_t, best_val


def perturb_sequence(a, rng, scale=0.05):
    a_pert = a * np.exp(rng.normal(0, scale, len(a)))
    a_pert = np.maximum(a_pert, 0)
    s = float(a_pert.sum())
    if s < 1e-10:
        return a.copy()
    return a_pert / s


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coord_descent_sweep(a, time_budget=60.0, start_time=None, rng=None):
    if start_time is None:
        start_time = time.monotonic()
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        return a
    a = a / S
    conv = compute_conv(a)
    best_obj = float(np.max(conv))
    best_a = a.copy()
    gr = (np.sqrt(5) - 1) / 2.0
    stagnant = 0

    while time.monotonic() - start_time < time_budget:
        obj_before = float(np.max(conv))
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for k in order:
            d = np.zeros(2 * n - 1)
            d[k:k + n] = a

            lo_delta = -a[k]
            max_conv_cur = float(np.max(conv))
            upper_est = np.sqrt(max(0.0, max_conv_cur - conv[2 * k] + a[k] ** 2 + 0.05 * max_conv_cur)) - a[k]
            hi_delta = max(upper_est, 2.0 / n)
            hi_delta = max(hi_delta, lo_delta + 1.0 / n)

            a_gs = lo_delta
            b_gs = hi_delta
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)

            # Evaluate at current points
            val_c = conv + 2.0 * c_gs * d
            val_c[2 * k] += c_gs * c_gs
            fc = float(np.max(val_c))

            val_d = conv + 2.0 * d_gs * d
            val_d[2 * k] += d_gs * d_gs
            fd = float(np.max(val_d))

            for _ in range(40):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    val_c = conv + 2.0 * c_gs * d
                    val_c[2 * k] += c_gs * c_gs
                    fc = float(np.max(val_c))
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    val_d = conv + 2.0 * d_gs * d
                    val_d[2 * k] += d_gs * d_gs
                    fd = float(np.max(val_d))

            delta_opt = (a_gs + b_gs) / 2.0
            if abs(delta_opt) < 1e-10:
                continue

            new_conv = conv + 2.0 * delta_opt * d
            new_conv[2 * k] += delta_opt * delta_opt
            new_obj = float(np.max(new_conv))

            if new_obj < obj_before - 1e-14:
                a[k] = max(0.0, a[k] + delta_opt)
                conv = new_conv
                S_new = float(np.sum(a))
                if S_new < 1e-10:
                    a[:] = 1.0 / n
                    conv = compute_conv(a)
                else:
                    a = a / S_new
                    conv = conv / (S_new * S_new)

        obj_after = float(np.max(conv))
        if obj_after < best_obj - 1e-12:
            best_obj = obj_after
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 8:
                a = best_a * np.exp(rng.normal(0, 0.02 * (1 + stagnant * 0.3), n))
                a = np.maximum(a, 0)
