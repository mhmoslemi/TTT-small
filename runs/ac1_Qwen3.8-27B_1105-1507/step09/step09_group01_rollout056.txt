def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max using FFT cross-correlation."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, M_target=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:

    return None, None


def golden_section_search(f, a_lo, a_hi, tol=0.01, max_iter=40):
    """Golden section search for minimum of f on [a_lo, a_hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if abs(a_hi - a_lo) < tol:
            break
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = f(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = f(d)
    return (a_lo + a_hi) / 2.0


def lp_polish_adaptive(a, time_budget=200.0, verbose=False, seed=42):
    """LP-based polishing with golden-section line search and constraint tightening."""
    n = len(a)

def subgradient_refine(a, time_budget=100.0, K=5, verbose=False):
    """Subgradient method with top-K peaks, momentum, and adaptive step."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]

        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        weights = np.exp((peak_vals - max_pv) * 5.0 / max_pv)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 300:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-10:
                    break
            else:
                lr *= 0.995
        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        c1 = rng.integers(0, n // 3)
        c2 = rng.integers(2 * n // 3, n)
        w = max(1, n // 10)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 8:
        # Bi-modal with varying widths
        c1 = n // 3
        c2 = 2 * n // 3
        w1 = n // rng.integers(6, 15)
        w2 = n // rng.integers(6, 15)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w1 ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w2 ** 2))
        a = a / a.sum()
    elif mode == 9:
        # Raised cosine window
        x = np.arange(n) / n
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = a / a.sum()
    elif mode == 10:
        # Triangular window
        x = np.arange(n) / (n - 1)
        a = 1 - np.abs(2 * x - 1)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def block_coordinate_descent(a, time_budget=120.0, block_size=16, verbose=False):
    """
    Block coordinate descent: optimize one block of coordinates at a time
    while keeping others fixed. Uses the fact that the objective changes
    in a structured way when only a subset of coordinates change.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    
    num_blocks = (n + block_size - 1) // block_size
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        # Shuffle block order
        block_order = rng.permutation(num_blocks)
        
        improved = False
        for bi in block_order:
            if time.monotonic() - start >= time_budget:
                break
            
            start_idx = bi * block_size
            end_idx = min(n, start_idx + block_size)
            bs = end_idx - start_idx
            
            # Current block values
            a_block = a[start_idx:end_idx].copy()
            
            # Try several perturbations of the block
            best_block = a_block.copy()
            best_block_obj = best_obj
            
            for attempt in range(20):
                # Perturb the block
                scale = 0.1 / (1 + attempt * 0.5)
                a_block_try = a_block * np.exp(rng.normal(0, scale, bs))
                a_block_try = np.maximum(a_block_try, 0)
                
                # Renormalize to maintain sum = 1
                rest_sum = s - a_block.sum()  # This is wrong, s should be 1
                # Actually a is already normalized to sum=1
                rest_sum = 1.0 - a_block.sum()
                block_sum = a_block_try.sum()
                
                if block_sum > rest_sum:
                    a_block_try = a_block_try * rest_sum / block_sum
                # Scale to maintain total sum = 1
                # Actually, just normalize the whole thing
                a_full_try = a.copy()
                a_full_try[start_idx:end_idx] = a_block_try
                a_full_try = np.maximum(a_full_try, 0)
                s_try = a_full_try.sum()
                if s_try < 1e-10:
                    continue
                a_full_try = a_full_try / s_try
                
                obj_try = compute_obj(a_full_try)
                if obj_try < best_block_obj - 1e-15:
                    best_block_obj = obj_try
                    best_block = a_block_try.copy()
                    improved = True
            
            if improved:
                a[start_idx:end_idx] = best_block
                a = a / a.sum()
        
        obj_check = compute_obj(a)
        if obj_check < best_obj - 1e-15:
            best_obj = obj_check
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
        
        it += 1
        if verbose and it % 10 == 0:
            print(f"    BCD it{it}: obj={best_obj:.8f}")
    
    return best_a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a hybrid approach combining frequency-domain spectral shaping (novel),
LP-based polishing, FFT-smoothed gradient descent, block coordinate descent,
and aggressive multi-length exploration with diverse perturbative restarts.
Key new idea: spectral error-reduction in frequency domain to rapidly flatten
autocorrelation peaks before LP polishing, plus block coordinate descent for
local refinement.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_shaping(a, time_budget=60.0, num_peaks=3, verbose=False):
    """
    Frequency-domain peak reduction: iteratively reduce peak autocorrelation
    by shaping the power spectrum. Reduces |A[f]|^2 at frequencies aligned
    with the peak lag, creating a flatter autocorrelation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    freqs = np.arange(N)
    lr = 0.25
    decay = 0.9995
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)

        sorted_idx = np.argsort(conv)[::-1]
        top_k = sorted_idx[:num_peaks]

        A = fft(a, N)
        P = np.abs(A) ** 2
        phase = np.angle(A)
        P0 = P[0]

        sensitivity = np.zeros(N)
        for k in top_k:
            k = int(k)
            sensitivity += np.cos(2.0 * np.pi * freqs * k / N)

        sensitivity[0] = 0.0

        P_updated = P * np.exp(-lr * sensitivity)
        P_updated[0] = P0
        P_updated = np.maximum(P_updated, 0.0)

        A_new = np.sqrt(P_updated) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new
            stall += 1
            if stall > 80:
                a = best_a.copy()
                lr *= 0.75
                if lr < 0.005:
                    break
                stall = 0

        it += 1
        lr = max(0.005, lr * decay)

        if verbose and it % 40 == 0:
            elapsed = time.monotonic() - start
            print(f"    Spectral it{it}: obj={best_obj:.8f}, lr={lr:.4f}, t={elapsed:.0f}s")

    return best_a


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max using FFT cross-correlation."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def nesterov_smoothed_descent(a, time_budget=300.0, tau_init=10.0, tau_end=400.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        if it % 5000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)

        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)

        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, M_target=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue

    return None, None

def subgradient_refine(a, time_budget=100.0, K=5, verbose=False):
    """Subgradient method with top-K peaks, momentum, and adaptive step."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]

        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        weights = np.exp((peak_vals - max_pv) * 5.0 / max_pv)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 300:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-10:
                    break
            else:
                lr *= 0.995
        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def block_coordinate_descent(a, time_budget=120.0, block_size=16, verbose=False):
    """Block coordinate descent: optimize blocks of coordinates with random perturbations."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)

    num_blocks = (n + block_size - 1) // block_size
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        block_order = rng.permutation(num_blocks)
        improved = False

        for bi in block_order:
            if time.monotonic() - start >= time_budget:
                break

            start_idx = bi * block_size
            end_idx = min(n, start_idx + block_size)
            bs = end_idx - start_idx
            a_block = a[start_idx:end_idx].copy()
            best_block_obj = best_obj

            for attempt in range(15):
                scale = 0.1 / (1 + attempt * 0.5)
                a_block_try = a_block * np.exp(rng.normal(0, scale, bs))
                a_block_try = np.maximum(a_block_try, 0)

                a_full_try = a.copy()
                a_full_try[start_idx:end_idx] = a_block_try
                a_full_try = np.maximum(a_full_try, 0)
                s_try = a_full_try.sum()
                if s_try < 1e-10:
                    continue
                a_full_try = a_full_try / s_try

                obj_try = compute_obj(a_full_try)
                if obj_try < best_block_obj - 1e-15:
                    best_block_obj = obj_try
                    best_block = a_block_try.copy()
                    improved = True

            if improved:
                a[start_idx:end_idx] = best_block
                a = a / a.sum()

        obj_check = compute_obj(a)
        if obj_check < best_obj - 1e-15:
            best_obj = obj_check
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 40:
                break

        it += 1
        if verbose and it % 10 == 0:
            print(f"    BCD it{it}: obj={best_obj:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines spectral shaping, LP polishing, smoothed GD,
    block coordinate descent, multi-n exploration, and perturbation restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Spectral Shaping ===
    print(f"\n=== Phase 1: Spectral Shaping ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 80.0)
    if p1_budget > 10:
        a_spec = spectral_shaping(best_seq, time_budget=p1_budget, num_peaks=3, verbose=True)
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Phase1 Spectral: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 Spectral: obj={obj_spec:.8f}")

    # === Phase 2: Smoothed GD ===
    print(f"\n=== Phase 2: Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.08, 80.0)
    if p2_budget > 10:
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=p2_budget,
                                          tau_init=5.0, tau_end=300.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase2 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 GD: obj={obj_sgd:.8f}")

    # === Phase 3: LP Polishing (main workhorse) ===
    print(f"\n=== Phase 3: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.32, 400.0)
    if p3_budget > 30:
        a_lp = lp_polish_adaptive(best_seq, time_budget=p3_budget, verbose=True, seed=42)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase3 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 LP: obj={obj_lp:.8f}")

    # === Phase 4: Block Coordinate Descent ===
    print(f"\n=== Phase 4: Block Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.06, 60.0)
    if p4_budget > 10:
        a_bcd = block_coordinate_descent(best_seq, time_budget=p4_budget,
                                         block_size=16, verbose=True)
        obj_bcd = compute_obj(a_bcd)
        if obj_bcd < best_score:
            best_score = obj_bcd
            best_seq = a_bcd.copy()
            print(f"  Phase4 BCD: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 BCD: obj={obj_bcd:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(300, n_current - 200), max(300, n_current - 100),
        n_current + 100, n_current + 200, n_current + 400,
        700, 900, 1100, 1300, 1500, 1800
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2200 and abs(n - n_current) > 60]

    for n_try in n_options[:7]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)

        alloc1 = min(remaining * 0.12, 25.0)
        if alloc1 > 8:
            a_try = spectral_shaping(a_try, time_budget=alloc1, num_peaks=2)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.10, 20.0)
        if alloc2 > 6:
            a_try = nesterov_smoothed_descent(a_try, time_budget=alloc2,
                                              tau_init=10.0, tau_end=200.0)

        remaining = deadline - time.monotonic()
        alloc3 = min(remaining * 0.22, 70.0)
        if alloc3 > 12:
            a_try = lp_polish_adaptive(a_try, time_budget=alloc3, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Perturbation + re-LP ===
    print(f"\n=== Phase 6: Perturbation + Re-LP ===")
    remaining = deadline - time.monotonic()

    for r in range(12):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        scale = 0.004 + 0.003 * r
        a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        alloc_spec = min(remaining * 0.08, 15.0)
        if alloc_spec > 5:
            a_pert = spectral_shaping(a_pert, time_budget=alloc_spec, num_peaks=2)

        alloc_lp = min(remaining * 0.28, 70.0)
        if alloc_lp > 12:
            a_pert = lp_polish_adaptive(a_pert, time_budget=alloc_lp, seed=300 + r)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Perturb {r} (scale={scale:.4f}): obj={best_score:.8f} **")
            else:
                print(f"  Perturb {r} (scale={scale:.4f}): obj={obj_pert:.8f}")

    # === Phase 7: Final intensive LP ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.65, 150.0)
        a_final = lp_polish_adaptive(best_seq, time_budget=alloc, verbose=True, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # Final subgradient
    remaining = deadline - time.monotonic()
    if remaining > 12:
        alloc = min(remaining * 0.3, 25.0)
        a_sg = subgradient_refine(best_seq, time_budget=alloc, K=8)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
