    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0"""
    n = len(a)

    # Build constraint matrix A_mat where A_mat @ g gives conv(a_norm, g)
    # A_mat has 2n-1 rows and n columns
    # A_mat[k, i] = a_norm[k-i] if 0 <= k-i < n, else 0
    A_mat = np.zeros((2*n-1, n))
    for i in range(n):
        # For column i, A_mat[k, i] = a_norm[k-i] if valid
        # k-i must be in [0, n-1], so k in [i, i+n-1]
        # But k also must be in [0, 2n-2]
        k_start = max(0, i)
        k_end = min(2*n-2, i+n-1)
        for k in range(k_start, k_end+1):
            A_mat[k, i] = a_norm[k-i]
    
    # LP: min M
    # Variables: [g_0, ..., g_{n-1}, M]

    A_eq = np.zeros((1, n+1))

def solve_lp_fast(a, n_top=None, time_limit=10):

    n = len(a)
    a_s = a / np.sum(a)

    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-10:
                g /= s
            return g, float(res.x[n])
    except Exception:
        pass
    return None, None

def smoothed_grad(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) via FFT."""
    n = len(a)
    conv = compute_conv(a)
    mx = float(np.max(conv))
    expv = np.exp(tau * (conv - mx))
    p = expv / float(np.sum(expv))
    
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    grad = 2.0 * g_full[:n]
    return grad, conv

        v = beta * v - lr * g

            a_r = np.sin(x) ** 2
        elif mode == 2:
            a_r = rng.uniform(0.1, 1, n_best)
            k = max(2, n_best // 30)
            kernel = np.ones(k) / k
            a_r = np.convolve(a_r, kernel, mode='same')
        else:
            x = np.linspace(-1, 1, n_best)
            a_r = 1.0 / (np.sqrt(1 - x ** 2 + 0.01))
        a_r = np.maximum(a_r, 0)
        a_r /= a_r.sum()
        blend = 0.25
        a_r = (1 - blend) * best_seq + blend * a_r
        a_r = np.maximum(a_r, 0)
        a_r /= a_r.sum()
        alloc = min(remaining * 0.2, 80.0)
        if alloc > 15:
            a_r = gd_lp_alternate(a_r, time_budget=alloc, verbose=False, seed=seed + 1000 + r)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # Phase 6: Final intensive polish
    print("\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 100.0)
        a_fp = lp_loop(best_seq, time_budget=alloc, n_top=max(15, len(best_seq) // 5),
                       verbose=True, seed=seed + 5000)
        obj_fp = compute_obj(a_fp)
        if obj_fp < best_score:
            best_score = obj_fp
            best_seq = a_fp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_fp:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 60.0)
        a_fs = gd_loop(best_seq, time_budget=alloc, tau_init=50.0, tau_end=800.0, seed=seed + 6000)
        obj_fs = compute_obj(a_fs)
        if obj_fs < best_score:
            best_score = obj_fs
            best_seq = a_fs.copy()
            print(f"  Final GD: obj={best_score:.8f} **")
        else:
            print(f"  Final GD: obj={obj_fs:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
