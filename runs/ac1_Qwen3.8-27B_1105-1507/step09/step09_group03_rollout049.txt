def mirror_descent_subgrad(a, time_budget, eta_init=0.1, eta_end=0.01, verbose=False):
    """
    Mirror descent (multiplicative weights) with subgradients of the max convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        eta = eta_init * (eta_end / eta_init) ** progress
        
        # Find active peaks (top few)
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        # Use log-sum-exp for smooth subgradient
        tau = 50.0 * (1 + progress * 20)  # increase smoothing
        exp_vals = np.exp(tau * (conv - max_val))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Weighted subgradient
        grad = np.zeros(n)
        for k in range(len(p)):
            if p[k] > 1e-10:
                grad += p[k] * conv_subgradient(a, k)
        
        # Mirror descent update
        log_update = -eta * grad
        log_update = log_update - np.max(log_update)  # numerical stability
        weights = np.exp(log_update)
        a_new = a * weights
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        
        if it % 100 == 0 and verbose:
            print(f"    MD it{it}: obj={best_obj:.8f}, eta={eta:.4f}")
        
        if it - best_it > 5000:
            a = best_a.copy()
            it = best_it
    
    return best_a

    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def conv_subgradient(a, peak_idx):
    """Compute subgradient of (a*a)[peak_idx] w.r.t. a.
    grad[m] = 2*a[peak_idx - m] if 0 <= peak_idx - m < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, peak_idx - (n - 1))
    hi = min(n - 1, peak_idx)
    for m in range(lo, hi + 1):
        grad[m] = 2.0 * a[peak_idx - m]
    return grad


def smooth_subgradient(a, tau=50.0):
    """Compute smoothed subgradient using log-sum-exp over all convolution entries."""
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_val))
    p = exp_vals / float(np.sum(exp_vals))
    # Weighted sum of subgradients
    grad = np.zeros(n)
    for k in range(len(conv)):
        if p[k] > 1e-12:
            grad += p[k] * conv_subgradient(a, k)
    return grad, max_val


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minmax(a, active_indices=None, max_iter=50):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    If active_indices given, only constrain those.
    Returns (g, M) or (None, None).
    """
    if not HAS_CVXPY:
        return None, None
    
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()

    if active_indices is not None:
        constraints = [A[active_indices] @ g <= M, cp.sum(g) == 1]
    else:
        constraints = [A @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver, max_iter=max_iter)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value) if M.value is not None else float(np.max(A @ g_val))
                    return g_val, M_val
        except Exception:
            continue
    return None, None


def solve_lp_minsum_active(a, active_indices):
    """Solve LP: min sum(conv(a,g)[k] for k in active) s.t. sum(g)=1, g>=0."""
    if not HAS_CVXPY:
        return None
    
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    obj = cp.sum(A[active_indices] @ g)
    constraints = [cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(obj), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def mirror_descent_subgrad(a, time_budget, eta_init=0.05, eta_end=0.005, tau_init=20.0, tau_end=200.0, verbose=False):
    """
    Mirror descent (multiplicative weights) with smoothed subgradients.
    Uses entropy-regularized updates that naturally stay in the positive orthant.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    best_it = 0
    momentum = 0.7
    v = np.zeros(n)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        eta = eta_init * (eta_end / eta_init) ** progress
        tau = tau_init * (tau_end / tau_init) ** progress

        # Smoothed subgradient
        grad, max_val = smooth_subgradient(a, tau)

        # Heavy-ball momentum on the gradient
        v = momentum * v + grad
        g_eff = v

        # Multiplicative weights update
        # a_new_j ∝ a_j * exp(-eta * g_eff_j)
        log_update = -eta * g_eff
        log_update = log_update - np.max(log_update)
        weights = np.exp(log_update)
        a_new = a * weights
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = best_a.copy()
            v = np.zeros(n)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        # Adapt momentum
        momentum = 0.5 + 0.4 * (1 - progress)

        # Reset if stuck
        if it - best_it > 3000:
            a = best_a.copy()
            v = np.zeros(n)
            # Perturb
            rng = np.random.default_rng(it % 10000)
            a = a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            best_it = it

        if it % 200 == 0 and verbose:
            print(f"    MD it{it}: obj={best_obj:.8f}, eta={eta:.5f}, tau={tau:.1f}")

    return best_a


def two_stage_lp(a, time_budget=100.0, verbose=False):
    """Two-stage LP: min-max then min-sum-on-active, with blending."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    it = 0

    while time.monotonic() - start < time_budget:
        # Stage 1: min max
        g1, M1 = solve_lp_minmax(a)
        if g1 is None:
            break

        # Find active set
        conv_ag = build_conv_matrix(a) @ g1
        max_cv = float(np.max(conv_ag))
        tol = 0.005 * max(max_cv, 1e-10)
        active = np.where(conv_ag >= max_cv - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv_ag)])

        # Stage 2: min sum on active
        g2 = solve_lp_minsum_active(a, active)

        # Blend
        if g2 is not None:
            g_blend = 0.5 * g1 + 0.5 * g2
        else:
            g_blend = g1

        # Line search
        best_t = 0.0
        best_f = compute_obj(a)
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1 - t) * a + t * g_blend
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                f_try = compute_obj(a_try)
                if f_try < best_f:
                    best_f = f_try
                    best_t = t

        if best_t > 0:
            a = (1 - best_t) * a + best_t * g_blend
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            obj = compute_obj(a)
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()

        it += 1
        if it % 5 == 0 and verbose:
            print(f"    2SLP it{it}: obj={best_obj:.8f}, M1={M1:.6f}, |S|={len(active)}")

        # Perturb if stuck
        if it > 20 and (time.monotonic() - start) > time_budget * 0.5:
            a = best_a * np.exp(rng.normal(0, 0.03, n))
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a


def peak_shaving(a, time_budget=60.0, verbose=False):
    """
    Iterative peak shaving: for the dominant peak, move mass from top contributors
    to positions that minimize new peak creation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0

    while time.monotonic() - start < time_budget and stall < 80:
        conv = compute_conv(a)
        peak_k = int(np.argmax(conv))
        peak_val = float(conv[peak_k])

        # Find contributing pairs
        i_lo = max(0, peak_k - (n - 1))
        i_hi = min(n - 1, peak_k)
        
        # Compute contributions
        contribs = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
            j = peak_k - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]

        # Top contributors
        top_frac = max(1, n // 10)
        top_idx = np.argsort(contribs)[::-1][:top_frac]

        # Find good targets: positions with low contribution to the peak
        # and also low contribution to other high peaks
        top_peaks = np.argsort(conv)[::-1][:5]
        target_score = np.zeros(n)
        for k in top_peaks:
            k_lo = max(0, k - (n - 1))
            k_hi = min(n - 1, k)
            for i in range(k_lo, k_hi + 1):
                j = k - i
                if 0 <= j < n:
                    target_score[i] += a[i] * a[j]

        # Good targets are those with low target_score that are not top contributors
        candidates = np.where(np.array([i not in set(top_idx) for i in range(n)]))[0]
        if len(candidates) == 0:
            stall += 1
            break

        cand_scores = target_score[candidates]
        good_targets = candidates[np.argsort(cand_scores)[:max(1, len(candidates) // 4)]]

        improved = False
        for src in top_idx[:3]:
            if a[src] < 1e-10:
                continue
            for tgt in good_targets[:5]:
                if tgt == src:
                    continue
                for frac in [0.02, 0.01, 0.005, 0.001]:
                    delta = frac * a[src]
                    if delta > a[src]:
                        continue
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[tgt] += delta
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            stall = 0
        else:
            stall += 1
            # Try random perturbation
            if stall % 10 == 0:
                src = int(rng.integers(0, n))
                tgt = int(rng.integers(0, n))
                if src != tgt:
                    delta = 0.01 * a[src]
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[tgt] += delta
                    a_try = np.maximum(a_try, 0)
                    a_try = a_try / a_try.sum()
                    if compute_obj(a_try) < best_obj:
                        a = a_try
                        best_obj = compute_obj(a)
                        best_a = a_try.copy()
                        improved = True
                        stall = 0

        if verbose and (time.monotonic() - start) % 5 < 0.1:
            print(f"    PS: obj={best_obj:.8f}, stall={stall}")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Comb pattern: alternating high/low
        a = np.zeros(n)
        for i in range(n):
            if i % 3 == 0:
                a[i] = 2.0
            else:
                a[i] = 0.5
        a = a / a.sum()
    elif mode == 3:
        # Golomb-ruler inspired: place mass at prime positions
        primes = [p for p in range(2, n) if all(p % q != 0 for q in range(2, int(p**0.5)+1))]
        a = np.zeros(n)
        for p in primes[:n//3]:
            a[p] = 1.0
        if a.sum() == 0:
            a = np.ones(n) / n
        else:
            a = a / a.sum()
    elif mode == 4:
        # Sinc-like: high in middle, decaying
        x = np.linspace(-1, 1, n)
        a = np.sinc(x * 3) ** 2
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Two-cluster
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            width = n // 8
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * width ** 2))
        a = a / a.sum()
    elif mode == 6:
        # Inverse comb
        a = np.zeros(n)
        for i in range(n):
            if i % 4 < 2:
                a[i] = 1.0
        a = a / a.sum()
    else:
        x = np.linspace(0, 1, n)
        freqs = rng.integers(1, 7, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def accelerated_subgrad(a, time_budget=100.0, verbose=False):
    """
    Accelerated subgradient method with Nesterov momentum and Euclidean projection.
    Different from mirror descent - uses projected updates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    best_it = 0
    
    y = a.copy()
    t_k = 1.0
    lr0 = 4.0 / (np.sqrt(n) * float(np.max(compute_conv(a))) + 1)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        lr = lr0 / (1 + 0.5 * t_k)
        
        # Smoothed subgradient at y
        tau = 30.0 * (1 + progress * 10)
        grad, _ = smooth_subgradient(y, tau)

        # Nesterov update
        t_next = (1 + np.sqrt(1 + 4 * t_k ** 2)) / 2
        beta = (t_k - 1) / t_next
        y_new = project_simplex(a + beta * (y - a) - lr * grad)
        
        obj_new = compute_obj(y_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = y_new.copy()
            best_it = it

        a = y_new
        y = a + beta * (a - a)  # simplify: y = a for next iteration's momentum
        y = project_simplex(a - lr * grad + beta * (a - (a + beta * (y - a) - lr * grad)))
        # Simplify: just set y = a for stability
        y = a.copy()
        t_k = t_next
        it += 1

        if it - best_it > 5000:
            a = best_a.copy()
            y = best_a.copy()
            t_k = 1.0
            best_it = it

        if it % 500 == 0 and verbose:
            print(f"    ASG it{it}: obj={best_obj:.8f}, lr={lr:.6f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization combining mirror descent, LP, peak shaving,
    and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # === Phase 0: Load parent or initialize ===
    print("Phase 0: Initialization")
    n_target = 800
    parent_loaded = False
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                parent_loaded = True
                print(f"  Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"  No parent: {e}")

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"  Default (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Try diverse initializations with quick polish ===
    print("\nPhase 1: Diverse Initialization Search")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        for mode in range(7):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            a_try = diverse_init(n_target, rng, mode=mode)
            # Blend with parent
            blend = 0.4 if parent_loaded else 0.0
            a_try = (1 - blend) * a_try + blend * best_seq
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            
            # Quick mirror descent
            alloc = min(remaining * 0.1, 20.0)
            if alloc > 3:
                a_try = mirror_descent_subgrad(a_try, time_budget=alloc,
                                              eta_init=0.1, eta_end=0.02,
                                              tau_init=10, tau_end=50)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Mode {mode}: obj={best_score:.8f} **")
            else:
                print(f"  Mode {mode}: obj={obj_try:.8f}")

    # === Phase 2: Mirror Descent (main workhorse) ===
    print("\nPhase 2: Mirror Descent")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.3, 300.0)
    if p2_budget > 30:
        print(f"  Budget: {p2_budget:.0f}s")
        a_md = mirror_descent_subgrad(best_seq, time_budget=p2_budget,
                                       eta_init=0.05, eta_end=0.003,
                                       tau_init=20.0, tau_end=300.0,
                                       verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Mirror Descent: obj={best_score:.8f} **")
        else:
            print(f"  Mirror Descent: obj={obj_md:.8f}")

    # === Phase 3: Two-stage LP ===
    print("\nPhase 3: Two-Stage LP")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 200.0)
    if p3_budget > 30:
        print(f"  Budget: {p3_budget:.0f}s")
        a_lp = two_stage_lp(best_seq, time_budget=p3_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  2SLP: obj={best_score:.8f} **")
        else:
            print(f"  2SLP: obj={obj_lp:.8f}")

    # === Phase 4: Peak Shaving ===
    print("\nPhase 4: Peak Shaving")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.1, 100.0)
    if p4_budget > 20:
        a_ps = peak_shaving(best_seq, time_budget=p4_budget, verbose=False)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  Peak Shaving: obj={best_score:.8f} **")
        else:
            print(f"  Peak Shaving: obj={obj_ps:.8f}")

    # === Phase 5: Accelerated Subgradient ===
    print("\nPhase 5: Accelerated Subgradient")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.15, 150.0)
    if p5_budget > 30:
        a_asg = accelerated_subgrad(best_seq, time_budget=p5_budget, verbose=True)
        obj_asg = compute_obj(a_asg)
        if obj_asg < best_score:
            best_score = obj_asg
            best_seq = a_asg.copy()
            print(f"  ASG: obj={best_score:.8f} **")
        else:
            print(f"  ASG: obj={obj_asg:.8f}")

    # === Phase 6: Multi-n Exploration ===
    print("\nPhase 6: Multi-n Exploration")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    n_options = sorted(set([
        max(200, n_current - 200), max(300, n_current - 100),
        n_current + 100, n_current + 200,
        400, 600, 900, 1100, 1400, 1800
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000 and abs(n - n_current) > 80]

    for n_try in sorted(n_options, key=lambda x: abs(x - n_current))[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        
        # Quick optimization
        alloc = min(remaining * 0.2, 50.0)
        if alloc > 10:
            half = alloc / 2
            a_try = mirror_descent_subgrad(a_try, time_budget=half,
                                           eta_init=0.08, eta_end=0.02,
                                           tau_init=15, tau_end=80)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_try = peak_shaving(a_try, time_budget=min(remaining * 0.3, 20.0))

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print("\nPhase 7: Final Polish")
    remaining = deadline - time.monotonic()

    # Round 1: Mirror descent
    if remaining > 30:
        alloc = min(remaining * 0.3, 120.0)
        a_f = mirror_descent_subgrad(best_seq, time_budget=alloc,
                                      eta_init=0.03, eta_end=0.002,
                                      tau_init=50, tau_end=500)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # Round 2: LP
    remaining = deadline - time.monotonic()
    if remaining > 25 and HAS_CVXPY:
        alloc = min(remaining * 0.25, 80.0)
        a_f = two_stage_lp(best_seq, time_budget=alloc)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Round 3: Peak shaving
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 50.0)
        a_f = peak_shaving(best_seq, time_budget=alloc)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final PS: obj={best_score:.8f} **")

    # Round 4: One more MD with very fine parameters
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 60.0)
        a_f = mirror_descent_subgrad(best_seq, time_budget=alloc,
                                      eta_init=0.01, eta_end=0.001,
                                      tau_init=100, tau_end=1000)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final Fine MD: obj={best_score:.8f} **")

    # Round 5: PEAK TRIMMING - zero out smallest elements
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_trim = best_seq.copy()
        for thresh_frac in [0.001, 0.002, 0.005]:
            median_val = np.median(a_trim[a_trim > 0])
            thresh = thresh_frac * median_val
            a_try = a_trim.copy()
            a_try[a_try < thresh] = 0
            s_try = float(np.sum(a_try))
            if s_try > 0.01:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_score:
                    best_score = obj_try
                    best_seq = a_try.copy()
                    print(f"  Trim ({thresh_frac}): obj={best_score:.8f} **")
                    a_trim = a_try.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    
    # Validate
    try:
        val = evaluate_sequence(result)
        if val < np.inf:
            best_score = val
        else:
            result = [1.0] * 100
            best_score = evaluate_sequence(result)
    except Exception:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
