5. Better time budget allocation

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A_mat (2n-1 x n) such that A_mat @ g = convolve(a, g)."""
    n = len(a)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A_mat[j:j + r, j] = a[:r]
    return A_mat


def solve_lp_standard(a):
    """Standard LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A_mat = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

    Coarse-to-fine multigrid optimization.
    Downsample, optimize at coarse level, upsample, refine.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Downsample by factor 2
    n_coarse = n // 2
    if n_coarse < 50:
        return a

    a_coarse = np.zeros(n_coarse)
    for i in range(n_coarse):
        a_coarse[i] = a[2*i] + a[2*i+1] if 2*i+1 < n else a[2*i]
    a_coarse = a_coarse / a_coarse.sum()

    # Optimize at coarse level (brief)
    coarse_budget = min(20.0, time_budget * 0.3)
    a_coarse_opt = lp_optimize(a_coarse, time_budget=coarse_budget, seed=seed)
    obj_coarse = compute_obj(a_coarse_opt)
    if obj_coarse < best_obj:
        print(f"    Multigrid: coarse n={n_coarse} improved to {obj_coarse:.8f}")

    # Upsample
    a_up = np.zeros(n)
    for i in range(n_coarse):
        a_up[2*i] = a_coarse_opt[i]
        if 2*i+1 < n:
            a_up[2*i+1] = a_coarse_opt[i]
    a_up = a_up / a_up.sum()

    # Blend with current
    a_new = 0.5 * a + 0.5 * a_up
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    obj_new = compute_obj(a_new)

    if obj_new < best_obj:
        # Refine at fine level
        fine_budget = time_budget - (time.monotonic() - start) - 5
        if fine_budget > 10:
            a_new = lp_optimize(a_new, time_budget=fine_budget, seed=seed+1)
            obj_new = compute_obj(a_new)

    if obj_new < best_obj:
        return a_new
    return a


def peak_shave(a, time_budget=30.0, seed=0):
    """
    Targeted coordinate descent on the peak.
    Identify peak lag, then adjust coordinates to reduce it.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        conv_a = compute_conv(a)
        peak_lag = int(np.argmax(conv_a))
        peak_val = conv_a[peak_lag]

        # Find the coordinates contributing most to the peak
        i_start = max(0, peak_lag - (n - 1))
        i_end = min(n - 1, peak_lag)

        # Try reducing high-contributing coordinates
        best_improvement = 0
        best_i = -1
        best_new_val = None

        for i in range(i_start, i_end + 1):
            j = peak_lag - i
            if j < 0 or j >= n:
                continue
            contribution = a[i] * a[j]
            if contribution > 0:
                # Try reducing a[i] by a small amount
                for frac in [0.7, 0.5, 0.3]:
                    new_val = a[i] * frac
                    delta = a[i] - new_val
                    # Redistribute delta to low-contribution coordinates
                    a_try = a.copy()
                    a_try[i] = new_val
                    # Add delta to a random low-contribution position
                    low_cand = []
                    for k in range(n):
                        if k == i or k == j:
                            continue
                        jk = peak_lag - k
                        if 0 <= jk < n:
                            low_cand.append((a[k] * a[jk], k))
                    if low_cand:
                        low_cand.sort()
                        # Spread delta among the lowest contribution positions
                        n_redist = min(5, len(low_cand))
                        for idx in range(n_redist):
                            a_try[low_cand[idx][1]] += delta / n_redist
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    improvement = best_obj - obj_try
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_i = i
                        best_new_val = a_try.copy()

        if best_improvement > 1e-12 and best_new_val is not None:
            a = best_new_val
            best_obj = compute_obj(a)
        else:
            # No improvement - small random perturbation
            noise = rng.normal(0, 0.005, n)
            a = a * np.exp(noise)
            a = np.maximum(a, 0)
            a = a / a.sum()

    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses aggressive LP-based optimization with
    multiple variants, peak-targeted refinement, multigrid, and multi-n search.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0  # Safety buffer

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_parent = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_parent) / n_parent
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_parent}): obj={best_score:.8f}")

    # === Phase 1: Main LP Optimization (50% of time) ===
    print(f"\n=== Phase 1: Aggressive LP Optimization ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.50, 700.0)
    if p1_budget > 30:
        a_opt = lp_optimize(best_seq, time_budget=p1_budget, seed=seed, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase1 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 LP: obj={obj_opt:.8f}")

    # === Phase 2: Peak Shaving + Multigrid (15% of time) ===
    print(f"\n=== Phase 2: Peak Shaving & Multigrid ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 150.0)
    if p2_budget > 20:
        # Peak shaving
        ps_budget = p2_budget * 0.5
        a_ps = peak_shave(best_seq, time_budget=ps_budget, seed=seed+100)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  Peak shave: obj={best_score:.8f} **")

        # Multigrid
        remaining = deadline - time.monotonic()
        mg_budget = min(remaining * 0.1, 60.0)
        if mg_budget > 10:
            a_mg = multigrid_refine(best_seq, time_budget=mg_budget, seed=seed+200)
            obj_mg = compute_obj(a_mg)
            if obj_mg < best_score:
                best_score = obj_mg
                best_seq = a_mg.copy()
                print(f"  Multigrid: obj={best_score:.8f} **")

    # === Phase 3: Second LP Pass with perturbation (20% of time) ===
    print(f"\n=== Phase 3: LP Pass 2 (Perturbed) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.20, 250.0)
    if p3_budget > 30:
        # Perturb the current best
        conv = compute_conv(best_seq)
        n = len(best_seq)
        peak_lag = int(np.argmax(conv))
        # Structured perturbation: anti-symmetric around peak
        pert = np.zeros(n)
        i_start = max(0, peak_lag - (n - 1))
        i_end = min(n - 1, peak_lag)
        for i in range(i_start, i_end + 1):
            j = peak_lag - i
            if 0 <= j < n:
                if i < j:
                    pert[i] = 0.03
                    pert[j] = -0.03
                elif i > j:
                    pert[i] = -0.03
                    pert[j] = 0.03
        a_pert = best_seq + pert
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        a_opt2 = lp_optimize(a_pert, time_budget=p3_budget, seed=seed+300, verbose=True)
        obj_opt2 = compute_obj(a_opt2)
        if obj_opt2 < best_score:
            best_score = obj_opt2
            best_seq = a_opt2.copy()
            print(f"  Phase3 LP2: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 LP2: obj={obj_opt2:.8f}")

    # === Phase 4: Multi-n Exploration (10% of time) ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = []
    for dn in [-100, -50, 50, 100, 200, 300]:
        n_try = n_current + dn
        if 400 <= n_try <= 2500:
            n_options.append(n_try)
    # Also try some fixed values
    for n_fixed in [600, 700, 900, 1000, 1200]:
        if abs(n_fixed - n_current) > 80:
            n_options.append(n_fixed)

    n_options = sorted(set(n_options))[:5]  # Limit to 5

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        # Brief LP optimization
        alloc = min(remaining * 0.15, 80.0)
        if alloc > 15:
            a_try = lp_optimize(a_try, time_budget=alloc, seed=seed + n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Final Intensive Polish (5% of time) ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        final_budget = min(remaining * 0.9, 100.0)
        a_final = lp_optimize(best_seq, time_budget=final_budget, seed=seed+500, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final polish: obj={best_score:.8f} **")
        else:
            print(f"  Final polish: obj={obj_final:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
