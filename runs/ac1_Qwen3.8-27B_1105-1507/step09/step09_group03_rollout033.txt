def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * np.max(conv) / (s*s)

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0)

"""
Optimizes step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based direction finding, smoothed gradient descent with Nesterov acceleration,
and local refinement. Explores multiple sequence lengths and diverse initializations.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.signal import fftconvolve
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_direction(a, active_indices=None, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    If active_indices given, only constrain those.
    Returns (g, M) where g is the direction.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    if active_indices is not None:
        # Only constrain active indices
        A = np.zeros((len(active_indices), n))
        for i, k in enumerate(active_indices):
            # Row k of the full conv matrix
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            A[i, j_start:j_end + 1] = a_norm[k - j_end:k - j_start + 1][::-1]
    else:
        A = build_conv_matrix(a_norm)
    
    num_rows = A.shape[0]
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, direction, max_iter=12):
    """
    Golden-section line search for t in [0,1] minimizing compute_obj((1-t)*a + t*direction).
    """
    a = np.maximum(a, 0)
    s_a = float(np.sum(a))
    if s_a < 1e-10:
        return a, compute_obj(a)
    a = a / s_a
    
    def f(t):
        a_t = (1 - t) * a + t * direction
        a_t = np.maximum(a_t, 0)
        s_t = float(np.sum(a_t))
        if s_t < 1e-10:
            return np.inf
        return compute_obj(a_t / s_t)
    
    # Try full step first
    f1 = f(1.0)
    f0 = f(0.0)
    
    if f1 <= f0:
        # Full step is good, try to extend beyond 1? No, keep in [0,1]
        # Actually just return t=1
        return a + (direction - a), f1  # (1-1)*a + 1*direction = direction
    
    # Golden section in [0, 1]
    phi = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    f_lo = f0
    f_hi = f1
    
    for _ in range(max_iter - 2):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)
        f_t1 = f(t1)
        f_t2 = f(t2)
        
        if f_t1 < f_t2:
            hi = t2
            f_hi = f_t2
        else:
            lo = t1
            f_lo = f_t1
    
    t_best = (lo + hi) / 2
    f_best = f(t_best)
    a_best = (1 - t_best) * a + t_best * direction
    a_best = np.maximum(a_best, 0)
    s_best = float(np.sum(a_best))
    if s_best > 1e-10:
        a_best = a_best / s_best
    return a_best, f_best

def smoothed_gradient(a, p):
    """
    Compute gradient of smoothed objective.
    grad_i = 2 * sum_{j=0}^{n-1} a[j] * p[i+j]

    n = len(a)
    p_rev = p[::-1]

def nesterov_smoothed_descent(a, time_budget=200.0, tau_init=20.0, tau_end=500.0, verbose=False):
    """
    Nesterov accelerated gradient descent on smoothed objective over simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9  # momentum
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.95))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        # Compute gradient
        conv = compute_conv(y)
        max_c = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_c))
        p = exp_vals / float(np.sum(exp_vals))
        
        grad = smoothed_gradient(y, p)
        
        # Update
        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
