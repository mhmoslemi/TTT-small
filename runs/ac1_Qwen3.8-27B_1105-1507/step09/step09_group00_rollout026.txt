"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-driven optimization with proper line search as the main workhorse,
combined with FFT-accelerated smoothed gradient descent for polishing.
Key innovations: aggressive LP iteration with adaptive line search, 
frequency-domain perturbation for escaping local minima, and multi-n exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


# More efficient vectorized version
def build_conv_matrix_fast(a):
    """Build convolution matrix using vectorized operations."""
    n = len(a)
    # A[k, j] = a[k-j] if 0 <= k-j < n
    # Using broadcasting
    k_idx = np.arange(2 * n - 1)[:, None]  # (2n-1, 1)
    j_idx = np.arange(n)[None, :]  # (1, n)
    diff = k_idx - j_idx  # (2n-1, n)
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_fast(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_direct(a):
    """
    Alternative LP: minimize the max of conv(g,g) directly.
    This is actually a different (harsher) problem. Let's try:
    min M s.t. conv(g,g)[k] <= M for all k, sum(g)=1, g>=0

    min M s.t. conv(a,g)[k] + (g*g - a*a)[linearized] <= M
    Actually, let's just stick with the standard LP.
    """
    return solve_lp_minpeak(a)

    # Coarse grid
    ts = np.linspace(0.01, 0.99, n_evals)
    objs = []
    for t in ts:
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            objs.append(np.inf)
        else:
            objs.append(compute_obj(a_try))
    objs = np.array(objs)
    
    # Find best on grid
    best_idx = np.argmin(objs)
    best_t = ts[best_idx]
    best_obj = objs[best_idx]
    
    # Parabolic refinement around best
    if 0 < best_idx < len(ts) - 1:
        t0, t1, t2 = ts[best_idx-1], ts[best_idx], ts[best_idx+1]
        f0, f1, f2 = objs[best_idx-1], objs[best_idx], objs[best_idx+1]
        denom = (f0 - 2*f1 + f2)
        if abs(denom) > 1e-15:
            t_par = t1 - 0.5 * (t2 - t0) * (f0 - f1) / (f0 - 2*f1 + f2)
            t_par = max(ts[0], min(ts[-1], t_par))
            a_par = (1 - t_par) * a + t_par * g
            a_par = np.maximum(a_par, 0)
            s_par = float(np.sum(a_par))
            if s_par > 1e-10:
                f_par = compute_obj(a_par)
                if f_par < best_obj:
                    best_t = t_par
                    best_obj = f_par
    
    return best_t, best_obj


def lp_driven_optimize(a, time_budget=600.0, verbose=True):
    """
    Main optimization: repeatedly solve LP, do line search, update.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)

def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def gd_polish(a, time_budget=120.0, tau_init=50.0, tau_end=800.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    base_lr = 3.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress**1.5
        
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        g = smoothed_gradient(y, tau)
        
        # Adaptive LR
        g_norm = float(np.linalg.norm(g))
        if g_norm > 1e-15:
            adapt = 1.0 / g_norm
            adapt = min(max(adapt, 0.3 * base_lr), 2.0 * base_lr)
            lr = 0.8 * lr + 0.2 * adapt
        
        # BB step
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-8), 4.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9998)
        
        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it
    
    if verbose:
        print(f"    GD polish: obj={best_obj:.8f}, iters={it}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.03):
    """Add low-frequency Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 4):
        freq = k * 2 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses aggressive LP-driven iteration with proper line search as the main workhorse,
combined with FFT-accelerated smoothed gradient descent for polishing.
Key improvements over previous: more LP iterations, proper line search,
dual-based convergence detection, broader n exploration, and better perturbation.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_fast(a):
    """Build convolution matrix A (2n-1 x n) such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[diff[valid]]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_fast(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, g, n_evals=30):
    """
    Line search: find best t in [0,1] to minimize obj((1-t)*a + t*g).
    Uses coarse grid + golden-section refinement.
    """
    n = len(a)
    ts = np.linspace(0.005, 0.995, n_evals)
    objs = np.full(n_evals, np.inf)
    for i, t in enumerate(ts):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            objs[i] = compute_obj(a_try)

    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]

    # Parabolic refinement
    if 0 < best_idx < n_evals - 1:
        t0, t1, t2 = ts[best_idx - 1], ts[best_idx], ts[best_idx + 1]
        f0, f1, f2 = objs[best_idx - 1], objs[best_idx], objs[best_idx + 1]
        denom = f0 - 2.0 * f1 + f2
        if abs(denom) > 1e-15:
            t_par = t1 - 0.5 * (t2 - t0) * (f0 - f1) / denom
            t_par = float(np.clip(t_par, 0.005, 0.995))
            a_par = (1 - t_par) * a + t_par * g
            a_par = np.maximum(a_par, 0)
            s_par = float(np.sum(a_par))
            if s_par > 1e-10:
                f_par = compute_obj(a_par)
                if f_par < best_obj:
                    best_t = t_par
                    best_obj = f_par

    return best_t, best_obj


def lp_driven_optimize(a, time_budget=300.0, verbose=True):
    """
    Main LP-driven optimization loop:
    1. Solve LP to get direction g
    2. Line search for optimal step size
    3. Update and repeat
    Uses dual gap to detect convergence.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    stall_count = 0
    iter_count = 0
    consecutive_no_improve = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1.5:
            break

        g, M = solve_lp_minpeak(a)
        if g is None:
            break

        # Current max autocorrelation peak
        conv_curr = compute_conv(a)
        max_curr = float(np.max(conv_curr))

        # Line search
        best_t, obj_at_t = line_search(a, g, n_evals=25)

        iter_count += 1

        if obj_at_t < best_obj - 1e-12:
            a_try = (1 - best_t) * a + best_t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                best_obj = obj_at_t
                best_a = a_try.copy()
                a = a_try
                stall_count = 0
                consecutive_no_improve = 0
                if verbose and (iter_count % 10 == 0 or iter_count == 1):
                    elapsed = time.monotonic() - start
                    print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t:.3f}, "
                          f"LP_M={M:.6f}, max_conv={max_curr:.6f}, time={elapsed:.0f}s")
        else:
            consecutive_no_improve += 1
            stall_count += 1

            if verbose and iter_count % 20 == 0:
                print(f"    LP it{iter_count}: stalled obj={best_obj:.8f}, "
                      f"LP_M={M:.6f}, max_conv={max_curr:.6f}")

            # Try smaller step
            if consecutive_no_improve <= 2:
                for t_small in [0.05, 0.02, 0.01]:
                    a_small = (1 - t_small) * a + t_small * g
                    a_small = np.maximum(a_small, 0)
                    s_small = float(np.sum(a_small))
                    if s_small > 1e-10:
                        a_small = a_small / s_small
                        obj_small = compute_obj(a_small)
                        if obj_small < best_obj - 1e-12:
                            best_obj = obj_small
                            best_a = a_small.copy()
                            a = a_small
                            stall_count = 0
                            consecutive_no_improve = 0
                            if verbose:
                                print(f"    LP it{iter_count}: small step t={t_small}, obj={best_obj:.8f}")
                            break

            if stall_count >= 5:
                # Perturb using subgradient direction
                tau = 200.0
                conv_p = compute_conv(a)
                max_p = float(np.max(conv_p))
                exp_p = np.exp(tau * (conv_p - max_p))
                p = exp_p / exp_p.sum()
                N_fft = 1
                while N_fft < 3 * n:
                    N_fft <<= 1
                P = fft(p, N_fft)
                Af = fft(a, N_fft)
                grad = 2.0 * ifft(P * np.conj(Af)).real[:n]

                # Move opposite to subgradient (descent direction)
                lr_pert = 0.02 / (float(np.linalg.norm(grad)) + 1e-10)
                a = a - lr_pert * grad
                a = np.maximum(a, 0)
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new
                else:
                    a = best_a.copy()

                # Also add small random component
                a = a + 0.005 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()

                stall_count = 0
                consecutive_no_improve = 0

                if iter_count > 100:
                    if verbose:
                        print(f"    LP it{iter_count}: giving up after {iter_count} iterations")
                    break

    return best_a


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def gd_polish(a, time_budget=100.0, tau_init=50.0, tau_end=1000.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed max autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 3.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.5

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g = smoothed_gradient(y, tau)
        g_norm = float(np.linalg.norm(g))

        # Adaptive LR
        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm + 1e-10)
            adapt = min(max(adapt, 0.2 * base_lr), 3.0 * base_lr)
            lr = 0.75 * lr + 0.25 * adapt

        # Barzilai-Borwein
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-8), 5.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9998)

        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.4
            best_it = it

    if verbose:
        print(f"    GD: obj={best_obj:.8f}, iters={it}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturb(a, rng, strength=0.03):
    """Add low-frequency Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 5):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining aggressive LP-driven optimization with line search,
    smoothed GD polishing, multi-n exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Initialize from parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Aggressive LP-Driven Optimization ===
    print(f"\n=== Phase 1: LP-Driven Optimization (main) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.4, 550.0)
    if p1_budget > 30:
        a_lp = lp_driven_optimize(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: GD Polish ===
    print(f"\n=== Phase 2: GD Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.12, 120.0)
    if p2_budget > 15:
        a_gd = gd_polish(best_seq, time_budget=p2_budget, tau_init=50.0, tau_end=800.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase2: obj={best_score:.8f} **")
        else:
            print(f"  Phase2: obj={obj_gd:.8f} (no improvement)")

    # === Phase 3: Second LP Round ===
    print(f"\n=== Phase 3: Second LP Round ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 300.0)
    if p3_budget > 20:
        a_lp2 = lp_driven_optimize(best_seq, time_budget=p3_budget, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj_lp2:.8f} (no improvement)")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        n_best - 150, n_best - 100, n_best - 50,
        n_best + 50, n_best + 100, n_best + 200, n_best + 400,
        500, 600, 700, 900, 1000, 1100, 1200, 1400
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1800 and abs(n - n_best) > 30]
    n_options = n_options[:6]
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)

        # LP optimization
        alloc_lp = min(remaining * 0.3, 60.0)
        if alloc_lp > 12:
            a_try = lp_driven_optimize(a_try, time_budget=alloc_lp, verbose=False)

        # GD polish
        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.1, 30.0)
        if alloc_gd > 8:
            a_try = gd_polish(a_try, time_budget=alloc_gd, tau_init=100.0, tau_end=500.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse Restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    restart_strategies = [
        ("mirror", lambda: ((best_seq + best_seq[::-1]) / 2.0)),
        ("fourier", lambda: fourier_perturb(best_seq, rng, strength=0.04)),
        ("smooth_blend", None),
        ("random_blend", None),
    ]

    for r, (name, func) in enumerate(restart_strategies):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if name == "mirror":
            a_r = func()
            a_r = a_r / a_r.sum()
        elif name == "fourier":
            a_r = func()
        elif name == "smooth_blend":
            ks = max(5, n_cur // 15)
            if ks % 2 == 0:
                ks += 1
            a_smooth = uniform_filter1d(best_seq, size=ks)
            a_r = 0.8 * best_seq + 0.2 * a_smooth
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
        else:
            a_r = 0.75 * best_seq + 0.25 * (rng.uniform(0, 1, n_cur) / n_cur)
            a_r = a_r / a_r.sum()

        # Optimize
        alloc_lp = min(remaining * 0.3, 50.0)
        if alloc_lp > 10:
            a_r = lp_driven_optimize(a_r, time_budget=alloc_lp, verbose=False)

        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.08, 25.0)
        if alloc_gd > 8:
            a_r = gd_polish(a_r, time_budget=alloc_gd, tau_init=100.0, tau_end=400.0)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart ({name}): obj={best_score:.8f} **")
        else:
            print(f"  Restart ({name}): obj={obj_r:.8f}")

    # === Phase 6: Final Intensive Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Final LP
    if remaining > 25:
        alloc = min(remaining * 0.45, 200.0)
        a_final = lp_driven_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final high-tau GD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 80.0)
        a_final2 = gd_polish(best_seq, time_budget=alloc, tau_init=200.0, tau_end=2000.0, verbose=True)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
