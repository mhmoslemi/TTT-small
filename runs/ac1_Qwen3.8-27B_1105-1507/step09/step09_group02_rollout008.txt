def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a):
    """Solve: min M s.t. conv(a_norm, g) <= M, sum(g)=1, g>=0"""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def subgradient_refine(a, time_budget, K=10, verbose=False):
    """Subgradient method with weighted top-K peaks."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 1.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        
        # Exponential weighting
        if max_pv > 1e-15:
            weights = np.exp((peak_vals - max_pv) * 5.0 / max_pv)
        else:
            weights = np.ones(K)
        weights /= weights.sum()
        
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]
        
        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 100:
                lr *= 0.8
                v *= 0.5
                stall = 0
                if lr < 1e-10:
                    break
        
        it += 1
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        # Cosine-modulated
        x = np.arange(n)
        a = 1 + 0.3 * np.cos(2 * np.pi * x / n)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Beta distribution
        a = rng.beta(2, 2, n)
        a = a / a.sum()
    elif mode == 6:
        # U-shape (high at edges)
        x = np.linspace(0, 1, n)
        a = x**2 + (1-x)**2 + 0.1
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

    """Solve: min M + (1/(2η))||g-a||^2 s.t. A@g <= M, sum(g)=1, g>=0"""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n)
    M = cp.Variable()
    objective = cp.Minimize(M + (1/(2*eta)) * cp.sum_squares(g - a_norm))
    constraints = [A @ g <= M, cp.sum(g) == 1, g >= 0]
    
    prob = cp.Problem(objective, constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:

                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None
