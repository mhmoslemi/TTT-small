"""
Multi-method optimization for minimum peak autocorrelation of non-negative sequences.
Combines spectral initialization, penalty-based gradient descent with Nesterov acceleration,

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    b = a / s
    conv = compute_conv(b)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(b):
    """Build matrix A (2n-1 x n) such that A @ g = conv(b, g)."""
    n = len(b)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = b[:r]
    return A

def solve_lp_minmax(b, active_only=None):
    """Solve LP for direction: min M s.t. conv(b,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(b)
    b_norm = b / np.sum(b)
    A = build_conv_matrix(b_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    if active_only is not None:
        constraints = [A[active_only] @ g <= M, cp.sum(g) == 1]
    else:
        constraints = [A @ g <= M, cp.sum(g) == 1]
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def penalty_gradient(b, tau):
    """Compute gradient of smoothed penalty: sum_k softplus(tau*(conv_k - max_conv))."""
    n = len(b)
    conv = compute_conv(b)
    max_c = float(np.max(conv))
    # Use log-sum-exp smoothed max
    shifted = tau * (conv - max_c)
    # Safe log-sum-exp
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    p = exp_vals / float(np.sum(exp_vals))
    # Gradient: p_k * d(conv_k)/db = p_k * (b convolved with delta at appropriate position)
    # d(conv_k)/db_j = b_{k-j} (for valid indices)
    # So gradient_j = sum_k p_k * b_{k-j}
    # This is a weighted sum of shifts of b, which is a convolution-like operation
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    B = fft(b, N)
    g_full = ifft(P * np.conj(B)).real
    g = 2.0 * g_full[:n]
    return g, conv

def nesterov_penalty_descent(b, time_budget=200.0, tau_init=20.0, tau_end=200.0, verbose=False):

    n = len(b)
    b = np.maximum(b, 0).astype(np.float64)

        return b

    v = np.zeros(n)
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)
        
        if it % 5000 == 0 and verbose:
            print(f"    NPD it{it}: obj={best_obj:.8f}, tau={tau:.1f}")
        
        # Nesterov lookahead
        y = b + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_b.copy()
        else:
            y = y / s_y
        
        g, conv = penalty_gradient(y, tau)
        
        # Update
        v = beta * v - lr * g
        b_new = y + v
        b_new = np.maximum(b_new, 0)
        s_new = float(np.sum(b_new))
        if s_new < 1e-10:
            b_new = best_b.copy()
            v = np.zeros(n)
            b = b_new
            it += 1
            continue
        b_new = b_new / s_new
        
        obj_new = compute_obj(b_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_b = b_new.copy()
            best_it = it
        
        b = b_new
        it += 1
        lr = max(1e-9, lr * 0.99995)
        
        # Reset if stuck

            b = best_b.copy()
            v = np.zeros(n)

def spectral_init(n, rng):
    """Initialize using spectral method: find non-negative seq with flat-ish power spectrum."""
    # Start with a random phase, take magnitude as initial sequence
    # Then iteratively enforce non-negativity and flatness
    b = np.ones(n) / np.sqrt(n)
    for _ in range(50):
        # Go to frequency domain
        B = fft(b, 2 * n)
        # Make magnitude flat
        mag = np.abs(B)
        target_mag = np.mean(mag[mag > 0])
        B_new = B / (mag + 1e-10) * target_mag
        # Come back
        b = ifft(B_new, 2 * n).real[:n]
        # Enforce non-negativity
        b = np.maximum(b, 0)
        s = float(np.sum(b))
        if s > 1e-10:
            b = b / s
        else:
            b = np.ones(n) / n
    
    return b

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

"""
Multi-phase optimizer for minimum peak autocorrelation of non-negative sequences.
Combines spectral initialization, Nesterov-accelerated penalty descent,
peak-shaving LP with active-set refinement, subgradient fast iterations,
multi-n exploration, and coordinate polish to beat the 1.5027 target.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    b = a / s
    conv = compute_conv(b)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(b):
    """Build matrix A (2n-1 x n) such that A @ g = conv(b, g)."""
    n = len(b)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = b[:r]
    return A


def solve_lp_minmax(b, active_set=None, time_limit=10.0):
    """Solve LP: min M s.t. conv(b,g)[k] <= M (all or active), sum(g)=1, g>=0."""
    n = len(b)
    b_norm = b / float(np.sum(b))
    A = build_conv_matrix(b_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()

    if active_set is not None and len(active_set) < len(A):
        constraints = [A[active_set] @ g <= M, cp.sum(g) == 1]
    else:
        constraints = [A @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(M), constraints)
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_peak_shave(b, peak_idx, active_set=None):
    """
    Solve LP: min conv(b,g)[peak_idx] s.t. conv(b,g)[k] <= conv(b,b)[k] for k in active_set,
    sum(g)=1, g>=0.
    """
    n = len(b)
    b_norm = b / float(np.sum(b))
    A = build_conv_matrix(b_norm)
    conv_bb = A @ b_norm
    g = cp.Variable(n, nonneg=True)

    constraints = [cp.sum(g) == 1]
    if active_set is not None:
        for k in active_set:
            if k == peak_idx:
                continue
            constraints.append(A[k] @ g <= conv_bb[k] + 1e-12)

    prob = cp.Problem(cp.Minimize(A[peak_idx] @ g), constraints)
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def penalty_gradient(b, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(b,b)."""
    n = len(b)
    conv = compute_conv(b)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    B = fft(b, N)
    g_full = ifft(P * np.conj(B)).real
    g = 2.0 * g_full[:n]
    return g, conv


def subgradient_step(b):
    """
    Fast subgradient step: move mass away from the peak-contributing positions.
    Returns a direction vector (not normalized).
    """
    n = len(b)
    conv = compute_conv(b)
    peak_k = int(np.argmax(conv))

    # Subgradient of (b*b)[peak_k] w.r.t. b is 2*b shifted
    # (b*b)[k] = sum_i b_i * b_{k-i}, d/db_j = sum_i [delta_{i,j}*b_{k-i} + b_i*delta_{k-i,j}]
    # = b_{k-j} + b_{j-k} (with appropriate bounds)
    sg = np.zeros(n)
    for j in range(n):
        val = 0.0
        i1 = peak_k - j
        if 0 <= i1 < n:
            val += b[i1]
        i2 = j - peak_k
        if 0 <= i2 < n:
            val += b[i2]
        sg[j] = 2.0 * val
    return sg, peak_k

    best_obj = compute_obj(b)

        if it % 5000 == 0 and verbose:
            print(f"    NPD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.2e}")

        # Nesterov lookahead
        y = b + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_b.copy()
        else:
            y = y / s_y

        g, conv = penalty_gradient(y, tau)

        # Update
        v = beta * v - lr * g
        b_new = y + v
        b_new = np.maximum(b_new, 0)
        s_new = float(np.sum(b_new))
        if s_new < 1e-10:
            b_new = best_b.copy()
            v = np.zeros(n)
            b = b_new
            it += 1
            continue
        b_new = b_new / s_new

        obj_new = compute_obj(b_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_b = b_new.copy()
            best_it = it

        b = b_new
        it += 1
        lr = max(min_lr, lr * 0.99997)

        # Reset if stuck
        if it - best_it > 25000:
            b = best_b.copy()
            v = np.zeros(n)
            lr = max(8.0 / np.sqrt(n) * 0.3, min_lr)
            beta = 0.8
            best_it = it

    return best_b


def fast_subgradient_phase(b, time_budget=60.0, verbose=False):
    """
    Very fast phase using subgradient steps with Nesterov acceleration.
    No LP required - just FFT for convolution.
    """
    n = len(b)
    b = np.maximum(b, 0).astype(np.float64)
    s = float(np.sum(b))
    if s < 1e-10:
        return b
    b = b / s

    best_b = b.copy()
    best_obj = compute_obj(b)
    start = time.monotonic()

    beta = 0.92
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        if it % 10000 == 0 and verbose:
            print(f"    FSG it{it}: obj={best_obj:.8f}, lr={lr:.2e}")

        y = b + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_b.copy()
        else:
            y = y / s_y

        sg, peak_k = subgradient_step(y)

        # Normalize subgradient for stable steps
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm > 1e-15:
            sg = sg / sg_norm

        v = beta * v - lr * sg
        b_new = y + v
        b_new = np.maximum(b_new, 0)
        s_new = float(np.sum(b_new))
        if s_new < 1e-10:
            b_new = best_b.copy()
            v = np.zeros(n)
            b = b_new
            it += 1
            continue
        b_new = b_new / s_new

        obj_new = compute_obj(b_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_b = b_new.copy()
            best_it = it

        b = b_new
        it += 1
        lr = max(1e-10, lr * 0.99999)

        if it - best_it > 50000:
            b = best_b.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_b


def lp_polish(b, time_budget=100.0, verbose=False):
    """LP-based polishing with adaptive active set and peak-shaving."""
    n = len(b)
    b = np.maximum(b, 0).astype(np.float64)
    s = float(np.sum(b))
    if s < 1e-10:
        return b
    b = b / s

    best_b = b.copy()
    best_obj = compute_obj(b)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        # Determine strategy: alternate between min-max and peak-shave
        conv = compute_conv(b)
        peak_k = int(np.argmax(conv))

        if it % 3 == 2:
            # Peak-shaving LP
            A = build_conv_matrix(b)
            conv_bg = A @ b
            max_val = float(np.max(conv_bg))
            tol = 0.1 * max_val
            active = np.where(conv_bg >= max_val - tol)[0]
            # Exclude the peak itself from constraints
            active = [k for k in active if k != peak_k]
            if len(active) < 3:
                active = None

            g_dir = solve_lp_peak_shave(b, peak_k, active_set=active)
        else:
            # Standard min-max LP
            g_dir, M_lp = solve_lp_minmax(b)

        if g_dir is None:
            stall += 5
            if stall > 40:
                break
            # Perturb and retry
            scale = 0.01 + 0.003 * min(stall, 20)
            b = best_b * np.exp(rng.normal(0, scale, n))
            b = np.maximum(b, 0)
            b = b / float(np.sum(b))
            it += 1
            continue

        # Line search for best step size
        found = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.85, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1]:
            b_new = (1.0 - t) * b + t * g_dir
            b_new = np.maximum(b_new, 0)
            s_new = float(np.sum(b_new))
            if s_new < 1e-10:
                continue
            b_new = b_new / s_new
            obj_new = compute_obj(b_new)
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_b = b_new.copy()
                b = b_new
                found = True
                break

        it += 1
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 20:
                scale = 0.015 + 0.005 * min(stall, 15)
                b = best_b * np.exp(rng.normal(0, scale, n))
                b = np.maximum(b, 0)
                s_b = float(np.sum(b))
                if s_b > 1e-10:
                    b = b / s_b
                stall = 0
                if it > 80:
                    break

        if verbose and it % 3 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}")

    return best_b


def coordinate_polish(b, time_budget=60.0, verbose=False):
    """Targeted coordinate descent on peak-contributing pairs."""
    n = len(b)
    b = np.maximum(b, 0).astype(np.float64)
    s = float(np.sum(b))
    if s < 1e-10:
        return b
    b = b / s

    best_b = b.copy()
    best_obj = compute_obj(b)
    start = time.monotonic()
    rng = np.random.default_rng(999)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(b)
        top_peaks = np.argsort(conv)[::-1][:3]

        improved = False
        for pk in top_peaks:
            i_start = max(0, int(pk) - (n - 1))
            i_end = min(n - 1, int(pk))

            if i_start > i_end:
                continue

            # Find top contributing pairs
            contribs = np.array([b[i] * b[int(pk) - i] for i in range(i_start, i_end + 1)])
            top_ci = np.argsort(contribs)[::-1][:8]

            for ci in top_ci:
                i = i_start + int(ci)
                j = int(pk) - i

                if i == j:
                    targets = [rng.integers(0, n), (i + n // 3) % n, (i + 2 * n // 3) % n]
                else:
                    targets = [rng.integers(0, n), rng.integers(0, n), (i + n // 4) % n]

                for tgt in targets:
                    tgt = int(tgt) % n
                    if tgt == i and not (i != j and tgt == j):
                        continue

                    for frac in [0.03, 0.015, 0.008, 0.004]:
                        src = i if (i != j and b[i] >= b[j]) else (i if i == j else min(i, j))
                        delta = frac * b[src]
                        if delta > b[src] * 0.9:
                            continue

                        b_try = b.copy()
                        b_try[src] -= delta
                        b_try[tgt] += delta
                        b_try = np.maximum(b_try, 0)
                        s_try = float(np.sum(b_try))
                        if s_try < 1e-10:
                            continue
                        b_try = b_try / s_try
                        obj_try = compute_obj(b_try)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_b = b_try.copy()
                            b = b_try
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        it += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break

        if verbose and it % 30 == 0:
            print(f"    CD it{it}: obj={best_obj:.8f}")

    return best_b


def spectral_init(n, rng):
    """Initialize using iterative spectral flatness enforcement."""
    b = np.ones(n) / np.sqrt(n)
    for _ in range(80):
        B_full = fft(b, 2 * n)
        mag = np.abs(B_full)
        nonzero_mask = mag > 1e-10
        if nonzero_mask.sum() == 0:
            break
        target_mag = np.mean(mag[nonzero_mask])
        B_new = B_full / (mag + 1e-10) * target_mag
        b = ifft(B_new, 2 * n).real[:n]
        b = np.maximum(b, 0)
        s = float(np.sum(b))
        if s > 1e-10:
            b = b / s
        else:
            b = np.ones(n) / n
            break
    return b


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        b = rng.uniform(0, 1, n)
        return b / b.sum()
    elif mode == 2:
        period = int(rng.integers(3, 12))
        b = np.zeros(n)
        b[::period] = 1.0
        b[1::period] = rng.uniform(0.1, 0.4, size=(n - 1) // period + 1)
        b = b[:n]
        return b / b.sum() if b.sum() > 0 else np.ones(n) / n
    elif mode == 3:
        x = np.linspace(0, 1, n)
        b = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** rng.uniform(0.5, 3.0)
        return b / b.sum()
    elif mode == 4:
        n_comp = int(rng.integers(2, 7))
        b = np.zeros(n)
        for _ in range(n_comp):
            c = rng.uniform(0, n)
            w = rng.uniform(n * 0.03, n * 0.25)
            amp = rng.uniform(0.5, 2.0)
            b += amp * np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        return b / b.sum() if b.sum() > 0 else np.ones(n) / n
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        b = x ** rng.uniform(-3, 3)
        return b / b.sum()
    elif mode == 6:
        f = rng.uniform(1, 15)
        f2 = rng.uniform(0.5, 5)
        x = np.linspace(0, 1, n)
        b = 1 + 0.4 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        b += 0.2 * np.sin(2 * np.pi * f2 * x + rng.uniform(0, 2 * np.pi))
        b = np.maximum(b, 0.01)
        return b / b.sum()
    elif mode == 7:
        return spectral_init(n, rng)
    elif mode == 8:
        # Two-block with different intensities
        b = np.zeros(n)
        split = int(n * rng.uniform(0.3, 0.7))
        b[:split] = rng.uniform(0.8, 1.5)
        b[split:] = rng.uniform(0.2, 0.6)
        return b / b.sum()
    else:
        b = rng.exponential(1.0, n)
        return b / b.sum()


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization:
    1. Load/initialize sequence
    2. Fast subgradient phase (many iterations, no LP)
    3. Nesterov penalty descent (smoothed)
    4. LP polishing with peak-shaving (precise)
    5. Multi-n exploration
    6. Diverse restarts
    7. Final intensive polish (LP + NPD + CD)
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        print("No parent available, using fresh start.")

    if best_seq is None:
        best_seq = spectral_init(n_target, rng)
        best_score = compute_obj(best_seq)
        print(f"Spectral init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Fast subgradient (quick wins, no LP) ===
    print(f"\n=== Phase 1: Fast Subgradient ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.1, 90.0)
    if p1_budget > 10:
        a1 = fast_subgradient_phase(best_seq, time_budget=p1_budget, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  FSG: obj={best_score:.8f} **")
        else:
            print(f"  FSG: obj={obj1:.8f}")

    # === Phase 2: Nesterov penalty descent ===
    print(f"\n=== Phase 2: Nesterov Penalty Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 200.0)
    if p2_budget > 20:
        a2 = nesterov_penalty_descent(best_seq, time_budget=p2_budget,
                                       tau_init=15.0, tau_end=400.0, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  NPD: obj={best_score:.8f} **")
        else:
            print(f"  NPD: obj={obj2:.8f}")

    # === Phase 3: LP polishing with peak-shaving ===
    print(f"\n=== Phase 3: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.25, 300.0)
    if p3_budget > 30:
        a3 = lp_polish(best_seq, time_budget=p3_budget, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj3:.8f}")

    # === Phase 4: High-tau Nesterov for fine-tuning ===
    print(f"\n=== Phase 4: High-tau Nesterov ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 150.0)
    if p4_budget > 20:
        a4 = nesterov_penalty_descent(best_seq, time_budget=p4_budget,
                                       tau_init=100.0, tau_end=2000.0, verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  NPD-hi: obj={best_score:.8f} **")
        else:
            print(f"  NPD-hi: obj={obj4:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(300, n_current - 200),
        max(300, n_current - 100),
        n_current + 100,
        n_current + 200,
        n_current + 400,
        400, 600, 1000, 1200, 1600, 2000
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2500 and abs(n - n_current) > 150]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick NPD
        alloc = min(remaining * 0.3, 80.0)
        if alloc > 15:
            a_try = nesterov_penalty_descent(a_try, time_budget=alloc,
                                              tau_init=20.0, tau_end=600.0)

        # Quick LP
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.2, 50.0)
        if alloc > 15:
            a_try = lp_polish(a_try, time_budget=alloc)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 70:
            break

        mode = (r * 3 + 2) % 9
        a_r = diverse_init(n_best, rng, mode=mode)
        a_r = 0.4 * a_r + 0.6 * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / float(np.sum(a_r))

        # NPD
        alloc = min(remaining * 0.3, 70.0)
        if alloc > 15:
            a_r = nesterov_penalty_descent(a_r, time_budget=alloc,
                                            tau_init=20.0, tau_end=500.0)

        # LP
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.2, 50.0)
        if alloc > 15:
            a_r = lp_polish(a_r, time_budget=alloc)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 40:
        alloc = min(remaining * 0.3, 100.0)
        a7 = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj7 = compute_obj(a7)
        if obj7 < best_score:
            best_score = obj7
            best_seq = a7.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 70.0)
        a7 = nesterov_penalty_descent(best_seq, time_budget=alloc,
                                       tau_init=300.0, tau_end=5000.0, verbose=True)
        obj7 = compute_obj(a7)
        if obj7 < best_score:
            best_score = obj7
            best_seq = a7.copy()
            print(f"  Final NPD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 50.0)
        a7 = coordinate_polish(best_seq, time_budget=alloc, verbose=True)
        obj7 = compute_obj(a7)
        if obj7 < best_score:
            best_score = obj7
            best_seq = a7.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 30.0)
        a7 = lp_polish(best_seq, time_budget=alloc)
        obj7 = compute_obj(a7)
        if obj7 < best_score:
            best_score = obj7
            best_seq = a7.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = compute_obj(np.array(result))

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
