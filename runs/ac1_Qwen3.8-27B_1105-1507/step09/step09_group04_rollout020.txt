import numpy as np

from scipy.fft import rfft, irfft
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def convolve_pair(a, b):
    """Compute linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

        end = min(j + n, 2 * n - 1)

    # = sum_j g[j] * a[k-j] for valid j, k-j
    # So row k of A has entries A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    return A


def build_conv_matrix_sparse(a):
    """Build the convolution matrix efficiently using the Toeplitz structure."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def find_best_blend(a, g, conv_a, conv_ag, conv_gg, n_t_samples=200):
    """
    Find t in [0,1] that minimizes max_k [(1-t)^2*conv_a[k] + 2t(1-t)*conv_ag[k] + t^2*conv_gg[k]].
    Uses fine grid search followed by local refinement.
    """
    best_obj = np.inf
    best_t = 0.0
    
    # Grid search
    ts = np.linspace(0, 1, n_t_samples)
    for t in ts:
        obj = (1 - t) ** 2 * np.max(conv_a)  # quick upper bound check
        # Full computation
        blended_conv = (1 - t) ** 2 * conv_a + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        peak = float(np.max(blended_conv))
        if peak < best_obj:
            best_obj = peak
            best_t = t
    
    # Local refinement around best_t
    if 0 < best_t < 1:
        local_ts = np.linspace(max(0, best_t - 0.02), min(1, best_t + 0.02), 100)
        for t in local_ts:
            blended_conv = (1 - t) ** 2 * conv_a + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
            peak = float(np.max(blended_conv))
            if peak < best_obj:
                best_obj = peak
                best_t = t
    
    return best_t, best_obj


def lp_iteration(a, conv_a, mode='active', active_frac=0.3, slack_mult=2.0):
    """
    One LP iteration: solve LP for g, find best blend t, return new a.
    """
    n = len(a)
    
    if mode == 'active':
        g, M_lp = solve_lp_active_set(a, conv_a, active_frac, slack_mult)
    else:
        g, M_lp = solve_lp_full(a, conv_a)
    
    if g is None:
        return a, conv_a, 0.0
    
    # Compute cross-convolutions
    conv_ag = convolve_pair(a, g)
    conv_gg = compute_conv(g)
    
    # Find best blend
    best_t, best_peak = find_best_blend(a, g, conv_a, conv_ag, conv_gg)
    
    if best_t <= 0:
        return a, conv_a, best_peak
    
    # Apply blend
    new_a = (1 - best_t) * a + best_t * g
    new_a = np.maximum(new_a, 0)
    s = float(np.sum(new_a))
    if s < 1e-10:
        return a, conv_a, best_peak
    new_a = new_a / s
    new_conv = compute_conv(new_a)
    
    return new_a, new_conv, float(np.max(new_conv))

                val[2 * k] += delta * delta
                return float(np.max(val))
            
            fc = peak_at_delta(c_gs)
            fd = peak_at_delta(d_gs)
            
            for _ in range(40):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = peak_at_delta(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = peak_at_delta(d_gs)
            
            delta_opt = (a_gs + b_gs) / 2.0
            new_peak = peak_at_delta(delta_opt)
            
            if new_peak < best_obj - 1e-14:
                # Apply update
                a[k] = max(0.0, a[k] + delta_opt)
                s_new = float(np.sum(a))
                if s_new < 1e-10:
                    a = np.ones(n) / n
                    conv = compute_conv(a)
                    best_obj = float(np.max(conv))
                    best_a = a.copy()
                    break
                a = a / s_new
                conv = compute_conv(a)
                best_obj = float(np.max(conv))
                best_a = a.copy()
                improved = True
        
        if not improved:
            stationary_count += 1
            if stationary_count > 3:
                break
        else:
            stationary_count = 0
    
    if verbose:
        print(f"    CD polish: obj={2*n*best_obj:.8f}")
    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.001)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = x ** rng.uniform(0.5, 3.0)
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** rng.uniform(0.5, 3.0)
    elif mode == 4:
        # Two bumps
        c1 = n * rng.uniform(0.1, 0.3)
        c2 = n * rng.uniform(0.7, 0.9)
        w = n * rng.uniform(0.05, 0.15)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
    elif mode == 5:
        # Sine wave
        x = np.linspace(0, 1, n)
        f = rng.uniform(1, 4)
        p = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
    elif mode == 6:
        # Triangular
        x = np.linspace(0, 1, n)
        a = 1.0 - 2.0 * np.abs(x - 0.5)
        a = np.maximum(a, 0.01)
    elif mode == 7:
        # Inverse triangular (U-shape)
        x = np.linspace(0, 1, n)
        a = 0.5 + 2.0 * np.abs(x - 0.5)
    elif mode == 8:
        # Random smooth
        a = np.ones(n) * 0.5
        for _ in range(3):
            c = rng.integers(n // 4, 3 * n // 4)
            w = rng.integers(n // 10, n // 3)
            amp = rng.uniform(-0.3, 0.3)
            a += amp * np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 0.01)
    else:
        a = rng.uniform(0.1, 1.0, n)
    a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def polished_lp_loop(a, time_budget, verbose=False):
    """
    Main LP iteration loop with adaptive parameters.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s
    conv_a = compute_conv(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv_a))
    start = time.monotonic()
    
    active_fracs = [0.2, 0.3, 0.4, 0.5, 1.0]
    iter_count = 0
    stall_count = 0
    last_improvement = best_obj
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 2:
            break
        
        # Choose LP mode: start with active set, switch to full if stalling
        if stall_count < 3:
            frac = active_fracs[min(stall_count, len(active_fracs) - 1)]
            mode = 'active'
        else:
            mode = 'full'
        
        new_a, new_conv, new_peak = lp_iteration(a, conv_a, mode=mode, active_frac=frac if frac < 1.0 else 0.3)
        
        new_obj = 2.0 * n * float(np.max(new_conv))
        iter_count += 1
        
        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = new_a.copy()
            a = new_a.copy()
            conv_a = new_conv.copy()
            stall_count = 0
            last_improvement = best_obj
            if verbose and iter_count % 10 == 0:
                print(f"      LP it{iter_count}: obj={best_obj:.8f}, t={elapsed:.1f}s")
        else:
            stall_count += 1
            if stall_count > 8:
                # Perturb and reset
                rng = np.random.default_rng(hash(iter_count) % 100000)
                scale = 0.01 + 0.005 * min(stall_count, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    conv_a = compute_conv(a)
                stall_count = 0
        
        # Update with new_conv even if no improvement (for next LP)
        if new_a is not a:
            conv_a = new_conv
    
    if verbose:
        print(f"    LP loop done: {iter_count} iters, obj={best_obj:.8f}, time={time.monotonic()-start:.1f}s")
    
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP iterations + coordinate descent + multi-length + diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent sequence
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")
    
    # === Phase 1: Intensive LP loop on parent length ===
    print(f"\n=== Phase 1: LP Loop (n={n_target}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 500.0)
    if p1_budget > 20:
        a_lp = polished_lp_loop(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 LP: obj={obj_lp:.8f}")
    
    # === Phase 2: Coordinate Descent Polish ===
    print(f"\n=== Phase 2: CD Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 150.0)
    if p2_budget > 15:
        a_cd = coord_descent_polish(best_seq, time_budget=p2_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase2 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 CD: obj={obj_cd:.8f}")
    
    # === Phase 3: Another LP round ===
    print(f"\n=== Phase 3: LP Round 2 ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.20, 250.0)
    if p3_budget > 20:
        a_lp2 = polished_lp_loop(best_seq, time_budget=p3_budget, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase3 LP: obj={best_score:.8f} **")
        # CD polish after LP
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_cd2 = coord_descent_polish(best_seq, time_budget=min(remaining * 0.3, 60.0))
            obj_cd2 = compute_obj(a_cd2)
            if obj_cd2 < best_score:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                print(f"  Phase3 CD: obj={best_score:.8f} **")
    
    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    
    n_options = []
    for delta in [-200, -100, -50, 50, 100, 200, 300, 400]:
        n_try = n_current + delta
        if 300 <= n_try <= 2000:
            n_options.append(n_try)
    # Also try some fixed lengths
    for n_fixed in [400, 500, 600, 800, 1000, 1200, 1500]:
        if abs(n_fixed - n_current) > 30 and 300 <= n_fixed <= 2000:
            n_options.append(n_fixed)
    
    n_options = sorted(set(n_options))
    n_options = n_options[:6]  # Limit to 6 attempts
    
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        
        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)
        
        # LP iterations
        alloc_lp = min(remaining * 0.4, 60.0)
        if alloc_lp > 10:
            a_try = polished_lp_loop(a_try, time_budget=alloc_lp)
        
        # CD polish
        remaining = deadline - time.monotonic()
        alloc_cd = min(remaining * 0.2, 30.0)
        if alloc_cd > 5:
            a_try = coord_descent_polish(a_try, time_budget=alloc_cd)
        
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")
    
    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    restart_modes = [1, 2, 3, 4, 5, 6, 7, 8, 9, 0]
    for r in range(min(6, len(restart_modes))):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        
        mode = restart_modes[r]
        a_r = diverse_init(n_best, rng, mode=mode)
        
        # Blend with best to keep some structure
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()
        
        # LP
        alloc_lp = min(remaining * 0.4, 50.0)
        if alloc_lp > 10:
            a_r = polished_lp_loop(a_r, time_budget=alloc_lp)
        
        # CD
        remaining = deadline - time.monotonic()
        alloc_cd = min(remaining * 0.15, 20.0)
        if alloc_cd > 5:
            a_r = coord_descent_polish(a_r, time_budget=alloc_cd)
        
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")
    
    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    
    if remaining > 30:
        alloc = min(remaining * 0.4, 100.0)
        a_f = polished_lp_loop(best_seq, time_budget=alloc, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
    
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_f2 = coord_descent_polish(best_seq, time_budget=min(remaining * 0.5, 60.0), verbose=True)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
