import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """

    n = len(a)

                    # Two-peak
                    x = np.linspace(0, 1, n)
                    a = np.exp(-30 * (x - 0.25) ** 2) + np.exp(-30 * (x - 0.75) ** 2) + 0.05
                    a = a / a.sum()
                elif mode == 3:
                    # Random smooth
                    a = np.exp(rng.normal(0, 0.5, n))
                    a = a / a.sum()
                elif mode == 4:
                    # Blend with best
                    a_new = rng.dirichlet(np.ones(n) * 0.3)
                    a = 0.5 * a_new + 0.5 * best_a
                    a = a / a.sum()
                else:
                    # Pure random
                    a = rng.dirichlet(np.ones(n) * 0.1)

                best_obj = compute_obj(a)
                if best_obj > compute_obj(best_a):
                    # Keep best_a as reference but continue from new a
                    pass
                stall = 0
                last_good_t = 0.5

        if verbose and iteration % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"        LP it{iteration}: obj={best_obj:.8f}, t={last_good_t:.3f}, "
                  f"stall={stall}, elapsed={elapsed:.1f}s")

        # If we've been stuck a long time, try a medium perturbation
        if stall >= 10 and stall < max_stall:
            if stall % 5 == 0:
                rng = np.random.default_rng(iteration * 31 + n)
                scale = 0.1 + 0.05 * (stall // 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()

    return best_a


def penalty_optimize(a, time_budget, verbose=False):
    """
    Smooth optimization using log-sum-exp surrogate with multiplicative weights.
    More aggressive than the previous version.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    beta = 50.0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        m = conv.max()
        shifted = beta * (conv - m)
        weights = np.exp(shifted)
        weights = weights / weights.sum()

        # Gradient: g[i] = 2 * sum_k weights[k] * a[k-i]
        # Use FFT for efficiency
        size = 1
        while size < 2 * n:
            size <<= 1
        Aw = rfft(weights, size)
        Aa = rfft(a, size)
        # cross-correlation: sum_k w[k] * a[k-i]
        # This is conv(w, reversed_a) or equivalently the valid part of conv
        g_full = irfft(Aw * Aa, size)
        # g_full[k] = sum_i w[i] * a[k-i], we want g[i] = sum_k w[k] * a[k-i]
        # g[i] = g_full[2*n-1-i] ... let me think more carefully
        # Actually, g[i] = sum_k w[k] * a[k-i] for valid k-i
        # This is the cross-correlation of w and a at lag i
        # cross_corr[i] = sum_k w[k+i] * a[k] = sum_j w[j] * a[j-i]
        # So g[i] = cross_correlation(w, a)[i] = sum_j w[j] * a[j-i]
        # In terms of FFT: cross_corr = irfft(fft(w) * conj(fft(a)))
        # But since these are real, conj(fft(a)) = fft(a[::-1]) (with padding)
        # Simpler: just compute it directly using the convolution formula
        # g[i] = sum_{valid j} w[j] * a[j-i]
        # For each i, j ranges over max(0,i) to min(2n-2, i+n-1) and j-i ranges over 0 to n-1
        # Let's just use the direct formula with numpy
        g = np.zeros(n)
        for i in range(n):
            # g[i] = sum_j w[j] * a[j-i] where 0<=j<2n-1 and 0<=j-i<n
            j_start = max(0, i)
            j_end = min(2 * n - 1, i + n - 1)
            if j_start <= j_end:
                g[i] = np.dot(weights[j_start:j_end + 1], a[j_start - i:j_end - i + 1])

        # Multiplicative weights update
        log_a = np.log(a)
        log_a_new = log_a - lr * g
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = min(lr * 1.01, 5.0)
            beta = min(beta * 1.05, 5000.0)
        else:
            lr = max(lr * 0.9, 0.01)

        it += 1
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"        Penalty it{it}: obj={best_obj:.8f}, beta={beta:.0f}, "
                  f"lr={lr:.3f}, elapsed={elapsed:.1f}s")

    return best_a


def shape_init(n, mode, rng=None):
    """Generate diverse shape initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    x = np.linspace(0, 1, n)

    if mode % 12 == 0:
        a = np.ones(n)
    elif mode % 12 == 1:
        # Anti-symmetric (large left, small right)
        a = x + 0.1
    elif mode % 12 == 2:
        # Anti-symmetric (small left, large right)
        a = (1 - x) + 0.1
    elif mode % 12 == 3:
        # Inverted V (large middle)
        a = 1.0 - np.abs(2 * x - 1)
        a = np.maximum(a, 0.05)
    elif mode % 12 == 4:
        # U-shape (large ends)
        a = (2 * x - 1) ** 2 + 0.3
    elif mode % 12 == 5:
        # Sine
        a = 0.5 + 0.4 * np.sin(2 * np.pi * x)
        a = np.maximum(a, 0.05)
    elif mode % 12 == 6:
        # Two peaks
        a = np.exp(-40 * (x - 0.3) ** 2) + np.exp(-40 * (x - 0.7) ** 2) + 0.05
    elif mode % 12 == 7:
        # Exponential ramp
        a = np.exp(2 * x)
    elif mode % 12 == 8:
        # Random smooth (interpolated noise)
        coarse = rng.uniform(0.5, 1.5, max(2, n // 10))
        a = np.interp(x, np.linspace(0, 1, len(coarse)), coarse)
    elif mode % 12 == 9:
        # Cosine bump
        a = np.cos(np.pi * x) ** 2 + 0.1
    elif mode % 12 == 10:
        # Sawtooth
        a = (x * 7) % 1.0
        a = np.maximum(a, 0.05)
    else:
        # Dirichlet random
        a = rng.dirichlet(np.ones(n) * 0.5)

    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def local_search(a, time_budget, verbose=False):
    """
    Local search: try small random perturbations and keep improvements.
    Uses both additive and multiplicative perturbations.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    rng = np.random.default_rng(12345 + n)

    while time.monotonic() - start < time_budget:
        # Random direction on the simplex
        direction = rng.normal(0, 1, n)
        # Project to sum-zero subspace
        direction -= direction.mean()
        norm = np.linalg.norm(direction)
        if norm < 1e-10:
            continue
        direction /= norm

        # Try different step sizes
        improved = False
        for scale in [0.01, 0.005, 0.02, 0.001, 0.05]:
            a_try = best_a + scale * direction / n
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 0.01:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if not improved and rng.random() < 0.1:
            # Occasionally try a larger jump
            a_try = best_a * np.exp(rng.normal(0, 0.3, n))
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining fast LP alternating, penalty optimization,
    shape-based exploration, local search, and n-sweep.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"\nInitial: n={n_current}, obj={best_score:.8f}, target=1.5027")

    # === Phase 1: Quick LP polish (8%) ===
    print(f"\n=== Phase 1: Quick LP Polish ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 80.0)
    if p1_budget > 20:
        a_p1 = lp_alternate_fast(best_seq, time_budget=p1_budget, verbose=True)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_p1:.8f}")

    # === Phase 2: Wide n-sweep (20%) ===
    print(f"\n=== Phase 2: n-Sweep ===")
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.20

    # Try a wide range of n values
    n_options = list(range(500, 1800, 50))
    # Add the current n and nearby values
    for dn in [-20, -10, 10, 20]:
        n_options.append(n_current + dn)
    n_options = sorted(set(n for n in n_options if 400 <= n <= 2000))

    per_n_budget = p2_budget / max(1, len(n_options))
    n_results = []
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = max(10, min(per_n_budget, remaining * 0.1))

        a_try = resample_sequence(best_seq, n_try)
        # Quick LP alternating (limited)
        a_opt = lp_alternate_fast(a_try, time_budget=alloc, verbose=False, max_stall=10)
        obj_opt = compute_obj(a_opt)
        marker = " **" if obj_opt < best_score else ""
        if len(n_results) < 20 or obj_opt < best_score:
            print(f"  n={n_try}: obj={opt_opt:.8f}{marker}" if False else
                  f"  n={n_try}: obj={obj_opt:.8f}{marker}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try
        n_results.append((n_try, obj_opt, a_opt.copy()))

    # Sort by objective and keep top
    n_results.sort(key=lambda x: x[1])
    print(f"  Best n from sweep: {n_results[0][0]}, obj={n_results[0][1]:.8f}")

    # === Phase 3: Intensive LP with diverse restarts (35%) ===
    print(f"\n=== Phase 3: Intensive LP (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.35

    n_rounds = 5
    per_round = p3_budget / n_rounds
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        alloc = min(per_round, remaining * 0.25)
        if alloc < 30:
            break

        if rnd == 0:
            a_start = best_seq.copy()
        elif rnd == 1:
            # Anti-symmetric shape blended with best
            x = np.linspace(0, 1, n_current)
            a_shape = x + 0.1
            a_shape = a_shape / a_shape.sum()
            a_start = 0.4 * a_shape + 0.6 * best_seq
            a_start = a_start / a_start.sum()
        elif rnd == 2:
            # From the best n-sweep result (if different)
            if len(n_results) > 1:
                a_start = n_results[1][2].copy()
                if len(a_start) != n_current:
                    a_start = resample_sequence(a_start, n_current)
            else:
                a_start = best_seq * np.exp(rng.normal(0, 0.1, n_current))
                a_start = a_start / a_start.sum()
        elif rnd == 3:
            # Sine-based anti-symmetric
            x = np.linspace(0, 1, n_current)
            a_shape = 0.5 + 0.45 * np.sin(2 * np.pi * x + np.pi * 0.3)
            a_shape = np.maximum(a_shape, 0.05)
            a_shape = a_shape / a_shape.sum()
            a_start = 0.3 * a_shape + 0.7 * best_seq
            a_start = a_start / a_start.sum()
        else:
            # Larger perturbation of best
            a_start = best_seq * np.exp(rng.normal(0, 0.3, n_current))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()

        a_lp = lp_alternate_fast(a_start, time_budget=alloc, verbose=True, max_stall=20)
        obj_lp = compute_obj(a_lp)
        marker = " **" if obj_lp < best_score else ""
        print(f"  LP round {rnd}: obj={obj_lp:.8f}{marker}")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()

    # === Phase 4: Shape exploration + penalty optimization (15%) ===
    print(f"\n=== Phase 4: Shape Exploration ===")
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.15

    shapes_to_try = [1, 2, 5, 6, 8]  # anti-symmetric ramps, sine, two-peak, exp ramp
    per_shape = p4_budget / max(1, len(shapes_to_try))

    for shape_mode in shapes_to_try:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(per_shape, remaining * 0.2)
        if alloc < 15:
            alloc = 15

        a_shape = shape_init(n_current, shape_mode, rng)
        # Blend with best
        a_shape = 0.3 * a_shape + 0.7 * best_seq
        a_shape = a_shape / a_shape.sum()

        # Short penalty optimization
        half_alloc = alloc * 0.4
        if half_alloc > 10:
            a_pen = penalty_optimize(a_shape, time_budget=half_alloc, verbose=False)
            obj_pen = compute_obj(a_pen)
        else:
            a_pen = a_shape
            obj_pen = obj_pen  # won't reach here

        # Short LP
        remaining = deadline - time.monotonic()
        lp_alloc = min(remaining * 0.1, alloc * 0.5)
        if lp_alloc > 10:
            a_lp = lp_alternate_fast(a_pen, time_budget=lp_alloc, verbose=False, max_stall=10)
            obj_lp = compute_obj(a_lp)
        else:
            a_lp = a_pen
            obj_lp = obj_pen

        marker = " **" if obj_lp < best_score else ""
        print(f"  Shape {shape_mode}: obj={obj_lp:.8f}{marker}")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()

    # === Phase 5: Local search + final polish (15%) ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Local search
    alloc = min(remaining * 0.3, 60.0)
    if alloc > 15:
        a_ls = local_search(best_seq, time_budget=alloc, verbose=False)
        obj_ls = compute_obj(a_ls)
        if obj_ls < best_score:
            best_score = obj_ls
            best_seq = a_ls.copy()
            print(f"  Local search: obj={best_score:.8f} **")
        else:
            print(f"  Local search: obj={obj_ls:.8f}")

    # Final LP polish
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.4, 80.0)
    if alloc > 20:
        a_final = lp_alternate_fast(best_seq, time_budget=alloc, verbose=True, max_stall=30)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # Try a few more n values near the best
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for n_try in [n_current - 25, n_current + 25, n_current - 50, n_current + 50]:
            remaining = deadline - time.monotonic()
            if remaining < 20 or n_try < 300:
                break
            a_try = resample_sequence(best_seq, n_try)
            a_opt = lp_alternate_fast(a_try, time_budget=min(30, remaining * 0.3),
                                      verbose=False, max_stall=10)
            obj_opt = compute_obj(a_opt)
            marker = " **" if obj_opt < best_score else ""
            print(f"  Final n={n_try}: obj={obj_opt:.8f}{marker}")
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"=== Target: 1.5027, Achieved: {best_score:.8f}, Gap: {best_score - 1.5027:.8f} ===")

    return result
