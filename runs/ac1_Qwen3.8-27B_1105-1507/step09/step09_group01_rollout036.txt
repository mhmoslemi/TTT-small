from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Build matrix A (2n-1 x n) such that A @ g = conv(p, g)."""
    n = len(p)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = p[:r]
    return A


def solve_lp_direction(p, M_target=None):
    """
    Solve LP: min M s.t. conv(p,g)_k <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(p)
    p_norm = p / np.sum(p)
    A = build_conv_matrix(p_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    # Try solvers in order of speed
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            if solver == cp.PDLP:
                prob.solve(solver=cp.PDLP, verbose=False, maxiter=500)
            else:
                prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None

    # We want to decrease p[i] where sensitivity is high, increase p[j] where sensitivity is low

    # conv[peak_k] = sum_m p[m] * p[peak_k - m]
    # d conv[peak_k] / d p[m] = p[peak_k - m] + p[m - peak_k]

    for m in range(n):

        s2 = m - peak_k
        if 0 <= s1 < n:
            sens[m] += p[s1]
        if 0 <= s2 < n:
            sens[m] += p[s2]

    source_score = p * sens
    # Target: minimize sens[m] (positions that least "drive" the peak)

    target_score = sens + 0.1 * p  # want low sensitivity and low current value
    j = int(np.argmin(target_score))

        A[j:j + r, j] = p[:r]
    return A


def solve_lp(p, M_target=None, solvers=None):

def cd_polish(p, time_budget=100.0, verbose=False, seed=42):
    """
    Pairwise coordinate descent polishing.
    Directly targets peaks of the self-convolution.
    """
    n = len(p)
    p = project_simplex(p.copy())
    best_p = p.copy()
    best_obj = compute_obj(p)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    conv = compute_conv(p)
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        if it % 1000 == 0:
            # Recompute conv from scratch to avoid drift
            conv = compute_conv(p)
        
        # Find top peaks
        top_k = min(5, 2 * n - 1)
        peak_vals, peak_idx = np.partition(conv, range(-top_k, 0)), np.argsort(conv)[::-1][:top_k]
        peak_idx = np.argsort(conv)[::-1][:top_k]
        
        # For the top peak, find best (i, j) pair
        k_star = int(peak_idx[0])
        
        # Sensitivity: d conv[k_star] / d p[m] = p[k_star - m] (if valid) + p[m - k_star] (if valid)
        sens = np.zeros(n)
        for m in range(n):
            if 0 <= k_star - m < n:
                sens[m] += p[k_star - m]
            if 0 <= m - k_star < n:
                sens[m] += p[m - k_star]
        
        # Source: high p[m] * sens[m]
        source_score = p * sens
        i = int(np.argmax(source_score))
        
        # Target: low sens[m] + low p[m]
        target_score = sens + 0.01 * p
        j = int(np.argmin(target_score))
        
        if i == j:
            # Try next peak
            if len(peak_idx) > 1:
                k_star = int(peak_idx[1])
                sens = np.zeros(n)
                for m in range(n):
                    if 0 <= k_star - m < n:
                        sens[m] += p[k_star - m]
                    if 0 <= m - k_star < n:
                        sens[m] += p[m - k_star]
                source_score = p * sens
                i = int(np.argmax(source_score))
                target_score = sens + 0.01 * p
                j = int(np.argmin(target_score))
                if i == j:
                    # Random pair
                    i = int(rng.integers(0, n))
                    j = int(rng.integers(0, n))
                    if i == j:
                        continue
            else:
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
        
        # Compute optimal delta
        # b[k] = p[k-j] (if valid) - p[k-i] (if valid)
        b = np.zeros(2 * n - 1)
        # Vectorized:
        ks = np.arange(2 * n - 1)
        valid_j = (ks - j >= 0) & (ks - j < n)
        valid_i = (ks - i >= 0) & (ks - i < n)
        b[valid_j] += p[ks[valid_j] - j]
        b[valid_i] -= p[ks[valid_i] - i]
        
        # Bisection for optimal delta
        delta_lo = -p[j]
        delta_hi = p[i]
        
        if abs(delta_hi - delta_lo) < 1e-15:
            continue
        
        lo = delta_lo
        hi = delta_hi
        for _ in range(50):
            mid = (lo + hi) / 2
            vals = conv + 2.0 * mid * b
            f_val = float(np.max(vals))
            # Subgradient (slope)
            active = np.where(np.isclose(vals, f_val, atol=1e-14))[0]
            if len(active) == 0:
                active = np.array([int(np.argmax(vals))])
            slope = 2.0 * float(np.sum(b[active]))
            
            if abs(slope) < 1e-12:
                break
            elif slope > 0:
                hi = mid
            else:
                lo = mid
        
        delta = (lo + hi) / 2.0
        delta = max(delta_lo + 1e-15, min(delta_hi - 1e-15, delta))
        
        # Apply update
        p[i] -= delta
        p[j] += delta
        
        # Update conv (approximate, skip quadratic terms for speed)
        conv += 2.0 * delta * b
        
        # Fix the quadratic correction at affected positions
        for k_check in [2 * i, 2 * j, i + j]:
            if 0 <= k_check < 2 * n - 1:
                Q = 0.0
                if k_check == 2 * i:
                    Q += 1.0
                if k_check == 2 * j:
                    Q += 1.0
                if k_check == i + j:
                    Q -= 2.0
                # We already added 2*delta*b[k_check], but the true change is
                # 2*delta*b[k_check] + delta^2 * Q
                # So we need to add delta^2 * Q
                conv[k_check] += delta * delta * Q
        
        # Check if improvement
        new_max = float(np.max(conv))
        new_obj = 2.0 * n * new_max
        if new_obj < best_obj - 1e-13:
            best_obj = new_obj
            best_p = p.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                # Revert and try a different approach
                p = best_p.copy()
                conv = compute_conv(p)
                stall = 0
                # Small random perturbation
                p *= np.exp(rng.normal(0, 0.005, n))
                p = project_simplex(p)
                conv = compute_conv(p)
        
        it += 1
    
    # Final verification
    final_obj = compute_obj(best_p)
    if final_obj < best_obj:
        best_obj = final_obj
    
    return best_p

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(p):
    """Build matrix A (2n-1 x n) such that A @ g = conv(p, g)."""
    n = len(p)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = p[:r]
    return A


def solve_lp(p, M_target=None):

    Returns (g_opt, M_opt) or (None, None).
    """

    s = float(np.sum(p))

    constraints = [A @ g <= M, cp.sum(g) == 1]

    # Try PDLP first (fastest for this size), then CLARABEL, then SCS

    if final_obj < best_obj:
        best_obj = final_obj
        best_p = project_simplex(best_p.copy())

    return best_p


def smoothed_gd(p, time_budget=60.0, tau_init=5.0, tau_end=300.0):
    """Nesterov accelerated gradient descent on smoothed max(conv(p,p))."""
    n = len(p)
    p = project_simplex(p.copy())
    best_p = p.copy()
    best_obj = compute_obj(p)
    start = time.monotonic()

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_p = p.copy()
    prev_g = None
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        y = p + beta * v
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy < 1e-10:
            y = best_p.copy()
        else:
            y = y / sy

        # Compute smoothed gradient
        conv = compute_conv(y)
        max_c = float(np.max(conv))
        exp_vals = np.exp(np.clip(tau * (conv - max_c), -700, 700))
        w = exp_vals / float(np.sum(exp_vals))

        # Gradient: g_j = 2 * sum_k w_k * y_{k-j} (with bounds)
        # Use FFT: cross-correlation of w and y
        N = 1
        while N < 3 * n:
            N <<= 1
        W = fft(w, N)
        Y = fft(y, N)
        # cross_corr[k] = sum_j y_j * w_{j+k} = sum_j y_j * w[j+k]
        # We want: for each j, sum_k w_k * y_{k-j} (with 0<=k-j<n)
        # This is the cross-correlation: (w * y) in appropriate sense
        # Actually: sum_k w[k] * y[k-j] = (w cross y)[j] shifted
        # Let's use: ifft(conj(fft(y)) * fft(w)) gives cross-correlation
        # cross[j] = sum_k y[k] * conj(w[k-j]) ... let me just be explicit
        # g_j = 2 * sum_{k: 0<=k-j<n} w[k] * y[k-j]
        # Let l = k-j, then k = l+j, so g_j = 2 * sum_{l: 0<=l<n} w[l+j] * y[l]
        # This is the cross-correlation of w and y at lag j
        # cross[j] = sum_l y[l] * conj(w[l+j]) = sum_l y[l] * w[l+j] (real)
        # = ifft(conj(fft(y, N)) * fft(w, N))[j] ... need to be careful

        # Simpler: direct computation via FFT
        # sum_l y[l] * w[l+j] = cross_correlation(y, w)[j]
        # cross_corr(a, b)[j] = sum_k a[k] * b[k+j]
        # = ifft(conj(fft(a)) * fft(b)) with appropriate windowing

        g_full = ifft(np.conj(fft(y, N)) * fft(w, N)).real
        g = 2.0 * g_full[:n]

        # Update
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_p
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy2 = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and sy2 > 1e-12:
                bb_lr = ss / sy2
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-10), 5.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        sa = float(np.sum(a_new))
        if sa < 1e-10:
            a_new = best_p.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / sa

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_p = a_new.copy()

        prev_p = p.copy()
        prev_g = g.copy()
        p = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)

        if it % 5000 == 0:
            p = best_p.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n)

    return best_p


def propose_candidate():
    """
    Main search function. Combines LP polishing, pairwise CD peak-shaving,
    smoothed gradient descent, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + 1470.0  # 30s safety margin
    rng = np.random.default_rng(12345)

    # --- Get parent ---
    best_seq = None
    best_score = np.inf

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={len(best_seq)}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.8f}")

    # === Phase 1: LP Polishing (30% budget) ===
    print(f"\n=== Phase 1: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 400.0)
    if p1_budget > 30:
        a_lp = lp_polish(best_seq, time_budget=p1_budget, verbose=True, seed=42)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1 LP done: obj={best_score:.8f}")
        else:
            print(f"  Phase1 LP: no improvement (obj={obj_lp:.8f})")

    # === Phase 2: Pairwise CD Peak Shaving (25% budget) ===
    print(f"\n=== Phase 2: CD Peak Shaving ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 350.0)
    if p2_budget > 20:
        a_cd = cd_peak_shave(best_seq, time_budget=p2_budget, verbose=True, seed=77)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase2 CD done: obj={best_score:.8f}")
        else:
            print(f"  Phase2 CD: no improvement (obj={obj_cd:.8f})")

    # === Phase 3: Smoothed GD (10% budget) ===
    print(f"\n=== Phase 3: Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.10, 120.0)
    if p3_budget > 15:
        a_gd = smoothed_gd(best_seq, time_budget=p3_budget, tau_init=5.0, tau_end=200.0)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase3 GD done: obj={best_score:.8f}")
        else:
            print(f"  Phase3 GD: no improvement (obj={obj_gd:.8f})")

    # === Phase 4: Multi-n Exploration (25% budget) ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = [n for n in [600, 700, 850, 900, 1000, 1100, 1200, 1400]
                 if abs(n - n_current) > 50 and 400 <= n <= 2000]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_try = resample_sequence(best_seq, n_try)
        a_try = project_simplex(a_try)

        # Quick LP polish
        alloc_lp = min(remaining * 0.2, 80.0)
        if alloc_lp > 15:
            a_try = lp_polish(a_try, time_budget=alloc_lp, seed=n_try)

        # Quick CD
        remaining = deadline - time.monotonic()
        alloc_cd = min(remaining * 0.15, 60.0)
        if alloc_cd > 10:
            a_try = cd_peak_shave(a_try, time_budget=alloc_cd, seed=n_try + 1)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Intensive combined polish (remaining budget) ===
    print(f"\n=== Phase 5: Final Combined Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        # Alternate LP and CD
        round_budget = min(remaining * 0.4, 200.0)
        if round_budget > 20:
            a1 = lp_polish(best_seq, time_budget=round_budget, verbose=True, seed=999)
            o1 = compute_obj(a1)
            if o1 < best_score:
                best_score = o1
                best_seq = a1.copy()
                print(f"  Final LP: obj={best_score:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            round_budget = min(remaining * 0.4, 150.0)
            a2 = cd_peak_shave(best_seq, time_budget=round_budget, seed=1234)
            o2 = compute_obj(a2)
            if o2 < best_score:
                best_score = o2
                best_seq = a2.copy()
                print(f"  Final CD: obj={best_score:.8f}")

        # Perturbation restarts
        remaining = deadline - time.monotonic()
        for r in range(5):
            if remaining < 30:
                break
            scale = 0.005 + 0.003 * r
            a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
            a_pert = project_simplex(a_pert)

            alloc = min(remaining * 0.3, 80.0)
            if alloc > 15:
                a_pert = lp_polish(a_pert, time_budget=alloc, seed=500 + r)
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  Perturb {r} (scale={scale:.4f}): obj={best_score:.8f} **")
                else:
                    print(f"  Perturb {r}: obj={obj_pert:.8f}")

            remaining = deadline - time.monotonic()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0)
    s_final = float(np.sum(best_seq))
    if s_final < 0.01:
        best_seq = np.ones(100)
        best_score = 2.0

    result = [float(min(1000.0, max(0.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
