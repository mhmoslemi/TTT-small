```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses multiplicative weights (HW algorithm) for fast LP approximation,
accelerated gradient descent on smoothed objective, and adaptive perturbation.
Key innovation: Game-theoretic LP approximation replaces expensive exact LP solves,
enabling 10x more polishing iterations within the time budget.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a, b=None):
    if b is None:
        b = a
    na = len(a)
    nb = len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:na + nb - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def at_transpose_y(a, y):
    """Compute A^T @ y where A[k,j] = a[j] * 1[j<=k<=j+n-1].
    Result[j] = a[j] * sum_{k=j}^{min(j+n-1, len(y)-1)} y[k]
    """
    n = len(a)
    m = len(y)
    ps = np.zeros(m + 1)
    ps[1:] = np.cumsum(y)
    result = np.zeros(n)
    for j in range(n):
        hi = min(j + n, m)
        result[j] = a[j] * (ps[hi] - ps[j])
    return result


def fast_lp_mw(a, T, eta=2.0):
    """
    Approximate LP min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0
    using multiplicative weights (Hedge algorithm) on the associated zero-sum game.
    Returns (g_opt, M_value).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    g_hist = np.zeros(n)
    y_hist = np.zeros(2 * n - 1)
    g = np.ones(n) / n
    y = np.ones(2 * n - 1) / (2 * n - 1)
    
 for t in range(T):
        # Column player update: g ∝ exp(-eta_scaled * (A^T y))
        Ay = at_transpose_y(a_norm, y)
        range_Ay = np.max(Ay) - np.min(Ay)
        if range_Ay < 1e-15:
            scale = eta
        else:
            scale = eta / range_Ay
        exp_vals = np.exp(-scale * (Ay - np.min(Ay)))
        g = exp_vals / float(np.sum(exp_vals))
        g_hist += g
        
        # Row player update: y ∝ exp(eta_scaled * (A g))
        Ag = compute_conv(a_norm, g)
        range_Ag = np.max(Ag) - np.min(Ag)
        if range_Ag < 1e-15:
            scale = eta
        else:
            scale = eta / range_Ag
        exp_vals = np.exp(scale * (Ag - np.min(Ag)))
        y = exp_vals / float(np.sum(exp_vals))
        y_hist += y
    
    g_avg = g_hist / T
    y_avg = y_hist / T
    M_val = float(np.dot(y_avg, compute_conv(a_norm, g_avg)))
    return g_avg, M_val


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    # grad[j] = 2 * sum_k p[k] * a[k-j] (approximately, ignoring the 2j=k correction)
    # More precisely: grad[j] = sum_k p[k] * (2 - 1[2j==k]) * a[k-j]
    # For large n, the correction is negligible, so we use 2 * cross_correlation
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    # cross_correlation: sum_k p[k] * a[k-j]
    # This is ifft(fft(p) * conj(fft(a)))... wait, need to be careful.
    # sum_k p[k] * a[k-j] is the cross-correlation of p and a at lag j.
    # In FFT: ifft(P * conj(A)) gives sum_k p[k] * a[k-j]? Let me verify.
    # Actually, ifft(P * A) gives circular convolution (p * a)[j] = sum_k p[k] * a[j-k]
    # We want sum_k p[k] * a[k-j] = sum_k p[k] * a[-(j-k)] (with a extended periodically)
    # = ifft(P * conj(A))[j] (for the linear part)
    xcorr = ifft(P * np.conj(A)).real
    grad = 2.0 * xcorr[:n]
    return grad, conv, float(max_conv)


def nesterov_descent(a, time_budget=60.0, tau_init=10.0, tau_end=500.0):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.85
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)
        
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        g, conv, max_c = smoothed_gradient(y, tau)
        
        # Project gradient onto tangent space of simplex
        g = g - np.mean(g)
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        lr = max(1e-8, lr * 0.9999)
    
    return best_a


def line_search_blend(a, g, best_obj, max_evals=20):
    """Find best t in (0,1) for (1-t)*a + t*g."""
    a = a.copy()
    g = g.copy()
    best_t = 0.0
    best_val = best_obj
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = compute_obj(np.maximum((1-c)*a + c*g, 0))
    fd = compute_obj(np.maximum((1-d)*a + d*g, 0))
    
    for _ in range(max_evals):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = compute_obj(np.maximum((1-c)*a + c*g, 0))
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = compute_obj(np.maximum((1-d)*a + d*g, 0))
    
    best_t = (lo + hi) / 2
    best_val = compute_obj(np.maximum((1-best_t)*a + best_t*g, 0))
    return best_t, best_val


def perturb(a, rng, method=0, scale=0.03):
    """Apply perturbation to escape local minima."""
    n = len(a)
    a = a.copy()
    if method == 0:
        a *= np.exp(rng.normal(0, scale, n))
    elif method == 1:
        a *= rng.uniform(1 - scale, 1 + scale, n)
    elif method == 2:
        mask = rng.random(n) > scale
        a *= mask
        if a.sum() < 1e-10:
            a = np.ones(n) / n
    elif method == 3:
        shift = int(rng.integers(1, max(2, n // 5)))
        a = np.roll(a, shift)
    elif method == 4:
        a = (1 - scale) * a + scale / n
    elif method == 5:
        # Perturb based on gradient of current conv
        conv = compute_conv(a / a.sum())
        max_c = conv.max()
        # Increase weights where conv is low (to flatten)
        target = np.full(len(conv), max_c)
        # Gradient direction: reduce high conv, increase low conv
        delta = np.zeros(n)
        for k in np.argsort(conv)[:n//4]:  # lowest conv values
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    delta[j] += a[idx] / a.sum()
        a = a + scale * 0.3 * delta / max(np.max(np.abs(delta)), 1e-10)
        a = np.maximum(a, 0)
    elif method == 6:
        # Random restart with blend
        a = (1 - scale * 2) * a + scale * 2 / n
    elif method == 7:
        # Frequency domain perturbation
        A = fft(a, 2 * n)
        A[1:-1] *= np.exp(1j * rng.normal(0, scale, len(A[1:-1])))
        a = ifft(A).real[:n]
        a = np.maximum(a, 0)
    else:
        a *= np.exp(rng.normal(0, scale, n))
    
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
```

```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses multiplicative weights (Hedge algorithm) on the associated zero-sum game for fast LP approximation,
Nesterov accelerated gradient descent on smoothed objective, and adaptive perturbation.
Key innovation: Game-theoretic LP approximation via multiplicative weights replaces expensive exact LP solves,
enabling far more polishing iterations within the time budget. Precomputed FFTs and vectorized operations
keep each iteration at O(n log n).
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a, b=None):
    """Full linear convolution a*b via FFT. If b is None, compute a*a."""
    if b is None:
        b = a
    na = len(a)
    nb = len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:na + nb - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def at_transpose_y(a, y):
    """Compute A^T @ y where A[k,j] = a[j] * 1[j<=k<=j+n-1].
    Result[j] = a[j] * sum_{k=j}^{min(j+n-1, len(y)-1)} y[k]. Vectorized O(n)."""
    n = len(a)
    m = len(y)
    ps = np.concatenate([[0.0], np.cumsum(y)])
    lo = np.arange(n)
    hi = np.minimum(lo + n, m)
    return a * (ps[hi] - ps[lo])


def fast_lp_mw(a, T, eta=2.0, verbose=False):
    """
    Approximate LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0
    via multiplicative weights (Hedge) on the zero-sum game.
    Row player (y over k) maximizes, column player (g over j) minimizes.
    Returns (g_optimal, M_value).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    m = 2 * n - 1

    # Precompute FFT of a for fast conv(a, g)
    size = 1
    while size < m:
        size <<= 1
    A_fft = rfft(a_norm, size)

    def conv_with_a(g):
        G_fft = rfft(g, size)
        return irfft(A_fft * G_fft, size)[:m]

    g_hist = np.zeros(n)
    y_hist = np.zeros(m)
    g = np.ones(n) / n
    y = np.ones(m) / m

    for t in range(T):
        # Column player (minimizer): g ∝ exp(-scale * (A^T y))
        Ay = at_transpose_y(a_norm, y)
        range_Ay = float(Ay.max() - Ay.min())
        scale = eta / max(range_Ay, 1e-10)
        exp_vals = np.exp(-scale * (Ay - Ay.min()))
        g_sum = float(exp_vals.sum())
        if g_sum > 1e-30:
            g = exp_vals / g_sum
        else:
            g = np.ones(n) / n
        g_hist += g

        # Row player (maximizer): y ∝ exp(scale * (A g))
        Ag = conv_with_a(g)
        range_Ag = float(Ag.max() - Ag.min())
        scale = eta / max(range_Ag, 1e-10)
        exp_vals = np.exp(scale * (Ag - Ag.min()))
        y_sum = float(exp_vals.sum())
        if y_sum > 1e-30:
            y = exp_vals / y_sum
        else:
            y = np.ones(m) / m
        y_hist += y

    g_avg = g_hist / T
    y_avg = y_hist / T
    M_val = float(np.dot(y_avg, conv_with_a(g_avg)))
    if verbose:
        print(f"    MW: T={T}, M={M_val:.8f}")
    return g_avg, M_val


def smoothed_gradient(a, tau):
    """Compute gradient of smoothed log-sum-exp max of conv(a,a) via FFT cross-correlation."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    p = exp_vals / float(np.sum(exp_vals))
    # grad[j] ≈ 2 * sum_k p[k] * a[k-j] (cross-correlation)
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    xcorr = ifft(P * np.conj(A)).real
    grad = 2.0 * xcorr[:n]
    return grad, conv


def nesterov_descent(a, time_budget=60.0, tau_init=10.0, tau_end=500.0):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    stagnant = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-6))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient(y, tau)
        # Project onto tangent space (sum of gradient = 0)
        g = g - np.mean(g)

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stagnant = 0
        else:
            stagnant += 1

        a = a_new
        lr = max(1e-8, lr * 0.9998)

        if stagnant > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 1.5 / np.sqrt(n)
            stagnant = 0

    return best_a


def line_search_blend(a, g, current_best, max_evals=18):
    """Golden section search for best t in [0,1] minimizing obj((1-t)*a + t*g)."""
    n = len(a)
    best_t = 0.0
    best_val = current_best

    # Check t=0 and t=1 first
    val_1 = compute_obj(np.maximum(g, 0))
    if val_1 < best_val:
        best_val = val_1
        best_t = 1.0

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)

    def eval_at(t):
        blend = np.maximum((1.0 - t) * a + t * g, 0)
        s = float(np.sum(blend))
        if s < 1e-10:
            return np.inf
        return compute_obj(blend / s)

    fc = eval_at(c)
    fd = eval_at(d)

    for _ in range(max_evals - 2):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = eval_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = eval_at(d)

    mid = (lo + hi) / 2.0
    fm = eval_at(mid)
    if fm < best_val:
        best_val = fm
        best_t = mid

    # Also check some discrete points for robustness
    for t_check in [0.1, 0.3, 0.5, 0.7, 0.9]:
        vc = eval_at(t_check)
        if vc < best_val:
            best_val = vc
            best_t = t_check

    return best_t, best_val


def perturb(a, rng, method=0, scale=0.03):
    """Apply structured perturbation to escape local minima."""
    n = len(a)
    a = a.copy()
    if method == 0:
        a *= np.exp(rng.normal(0, scale, n))
    elif method == 1:
        a *= rng.uniform(1 - scale, 1 + scale, n)
    elif method == 2:
        mask = rng.random(n) > scale * 1.5
        a *= mask
        if a.sum() < 1e-10:
            a = np.ones(n) / n
    elif method == 3:
        shift = int(rng.integers(1, max(2, n // 8)))
        a = np.roll(a, shift)
    elif method == 4:
        a = (1 - scale) * a + scale / n
    elif method == 5:
        # Gradient-based perturbation: push mass away from high-conv regions
        a_norm = a / a.sum()
        conv = compute_conv(a_norm)
        max_c = conv.max()
        # Find low-conv positions and add mass there
        low_mask = conv < max_c * 0.7
        delta = np.zeros(n)
        for k in np.where(low_mask)[0][:n // 3]:
            for j in range(n):
                idx = int(k) - j
                if 0 <= idx < n:
                    delta[j] += a_norm[idx]
        max_d = np.max(np.abs(delta))
        if max_d > 1e-12:
            a = a + scale * 0.5 * delta / max_d
        a = np.maximum(a, 0)
    elif method == 6:
        # Frequency perturbation
        A = fft(a, 2 * n)
        A[1:-1] *= np.exp(1j * rng.normal(0, scale + 0.02, len(A[1:-1])))
        a = ifft(A).real[:n]
        a = np.maximum(a, 0)
    elif method == 7:
        # Partial random restart
        frac = scale * 2
        a = (1 - frac) * a + frac / n
    elif method == 8:
        # Mirror-then-noise
        a = (a + a[::-1]) / 2
        a *= np.exp(rng.normal(0, scale * 0.5, n))
    else:
        a *= np.exp(rng.normal(0, scale, n))

    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation on [0,1]."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a /= a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a /= a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        freq = rng.integers(2, 6)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        a /= a.sum()
    elif mode == 4:
        # Triangle
        mid = n // 2
        a = 1.0 - np.abs(np.arange(n) - mid) / mid
        a = np.maximum(a, 0.01)
        a /= a.sum()
    elif mode == 5:
        # Sinc-like
        x = np.linspace(-3, 3, n)
        a = np.sinc(x / 3.0)
        a = np.maximum(a, 0.01)
        a /= a.sum()
    elif mode == 6:
        # Gaussian bumps
        centers = np.array([n // 4, 3 * n // 4])
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * (n // 6) ** 2))
        a = np.maximum(a, 0.01)
        a /= a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)


def polish_loop(a, time_budget, rng, mw_T, verbose=True):
    """Main polishing loop using MW LP approximation + line search + perturbation."""
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    no_improve = 0
    iteration = 0

    while (time.monotonic() - start) < time_budget:
        iteration += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        # Adaptive T: start larger, shrink as we converge
        T_cur = mw_T
        if iteration > 15:
            T_cur = max(200, mw_T * 200 // (200 + iteration * 30))

        g, M_val = fast_lp_mw(best_a, T=T_cur, eta=2.0)

        best_t, best_blend_obj = line_search_blend(best_a, g, best_obj, max_evals=15)

        if best_blend_obj < best_obj - 1e-12:
            best_obj = best_blend_obj
            best_a = np.maximum((1.0 - best_t) * best_a + best_t * g, 0)
            s = float(np.sum(best_a))
            if s > 1e-10:
                best_a = best_a / s
            no_improve = 0
            if verbose and (iteration % 3 == 0 or best_t > 0.5):
                elapsed = time.monotonic() - start
                print(f"    Polish it{iteration}: obj={best_obj:.8f}, t={best_t:.3f}, "
                      f"M={M_val:.6f}, T={T_cur}, elapsed={elapsed:.0f}s")
        else:
            no_improve += 1
            if no_improve >= 4:
                method = int(rng.integers(0, 9))
                scale = 0.015 + 0.01 * min(no_improve, 6)
                best_a = perturb(best_a, rng, method=method, scale=scale)
                no_improve = 0
                if verbose:
                    print(f"    Polish it{iteration}: no improve, perturb m={method}, "
                          f"elapsed={time.monotonic()-start:.0f}s")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multiplicative weights LP approximation + Nesterov descent
    + multi-n exploration + adaptive perturbation.
    Returns list of non-negative floats minimizing the evaluation objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 12.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")

    # === Phase 1: Nesterov warm-start ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 1: Nesterov (remaining={remaining:.0f}s) ===")
    if remaining > 40:
        alloc = min(remaining * 0.12, 80.0)
        a = nesterov_descent(best_seq, time_budget=alloc, tau_init=5.0, tau_end=400.0)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov: obj={obj:.8f}")

    # === Phase 2: Main MW polishing ===
    n = len(best_seq)
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 2: MW Polish (remaining={remaining:.0f}s, n={n}) ===")
    mw_T = max(400, min(2500, int(40 * n)))
    p2_budget = min(remaining * 0.45, 500.0)
    if p2_budget > 20:
        best_seq, best_score = polish_loop(best_seq, p2_budget, rng, mw_T, verbose=True)

    # === Phase 3: Nesterov refinement ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 3: Nesterov Refine (remaining={remaining:.0f}s) ===")
    if remaining > 30:
        alloc = min(remaining * 0.15, 80.0)
        a = nesterov_descent(best_seq, time_budget=alloc, tau_init=50.0, tau_end=1500.0)
        obj = compute_obj(a)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  Refine: obj={best_score:.8f} **")
        else:
            print(f"  Refine: obj={obj:.8f}")

    # === Phase 4: Multi-n exploration ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 4: Multi-n (remaining={remaining:.0f}s) ===")
    n_best = len(best_seq)

    n_candidates = [n_best - 200, n_best - 100, n_best + 100, n_best + 200,
                    n_best + 400, n_best + 600, 500, 600, 700, 800, 1000, 1200, 1500]
    n_candidates = sorted(set([n for n in n_candidates if 300 <= n <= 2000 and n != n_best]))
    n_candidates.sort(key=lambda x: abs(x - n_best))

    for n_try in n_candidates[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        a_try = resample(best_seq, n_try)

        # Nesterov
        alloc1 = min(remaining * 0.06, 30.0)
        if alloc1 > 5:
            a_try = nesterov_descent(a_try, time_budget=alloc1, tau_init=10.0, tau_end=300.0)

        # MW polish (shorter)
        remaining = deadline - time.monotonic()
        if remaining > 20:
            T_n = max(300, min(1200, int(25 * n_try)))
            p_budget = min(remaining * 0.12, 60.0)
            if p_budget > 10:
                a_try, obj_attempt = polish_loop(a_try, p_budget, rng, T_n, verbose=False)
                if obj_attempt < best_score:
                    best_score = obj_attempt
                    best_seq = a_try.copy()
                    n_best = n_try
                    print(f"  ** n={n_try}: obj={best_score:.8f} **")
                else:
                    print(f"  n={n_try}: obj={obj_attempt:.8f}")
            else:
                obj_attempt = compute_obj(a_try)
                if obj_attempt < best_score:
                    best_score = obj_attempt
                    best_seq = a_try.copy()
                    n_best = n_try
                    print(f"  ** n={n_try}: obj={best_score:.8f} (quick) **")
                else:
                    print(f"  n={n_try}: obj={obj_attempt:.8f} (quick)")

    # === Phase 5: Final intensive polish ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 5: Final Polish (remaining={remaining:.0f}s) ===")
    n = len(best_seq)
    mw_T_final = max(500, min(3000, int(45 * n)))

    # MW polish
    if remaining > 20:
        p5_budget = min(remaining * 0.4, 200.0)
        best_seq, best_score = polish_loop(best_seq, p5_budget, rng, mw_T_final, verbose=True)

    # Final Nesterov with high tau
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 50.0)
        if alloc > 5:
            a = nesterov_descent(best_seq, time_budget=alloc, tau_init=200.0, tau_end=3000.0)
            obj = compute_obj(a)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  Final Nesterov: obj={best_score:.8f} **")

    # One more MW round if time permits
    remaining = deadline - time.monotonic()
    if remaining > 15:
        g, _ = fast_lp_mw(best_seq, T=max(500, int(30 * n)), eta=2.0)
        best_t, obj_new = line_search_blend(best_seq, g, best_score, max_evals=12)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = np.maximum((1 - best_t) * best_seq + best_t * g, 0)
            best_seq = best_seq / best_seq.sum()
            print(f"  Final MW: obj={best_score:.8f} **")

    # Validate and return
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
