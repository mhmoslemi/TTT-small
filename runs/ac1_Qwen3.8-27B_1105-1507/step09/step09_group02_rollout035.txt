"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: LP-based polishing with optimal step size computation,
active set tracking, and multi-length exploration.
Key improvement: analytical step size from quadratic upper bound on new peak.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Cross correlation conv(a, b) via FFT."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, solver=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    if solver:
        solvers = [solver] + [s for s in solvers if s != solver]
    
    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M.value)
        except Exception:
            continue
    return None, None


def optimal_step_size(M_f, M_lp, M_g):
    """
    Compute optimal t that minimizes the upper bound on new peak:
    q(t) = (1-t)^2 * M_f + 2t(1-t) * M_lp + t^2 * M_g
    """
    # q(t) = M_f(1-t)^2 + 2*M_lp*t*(1-t) + M_g*t^2
    #      = M_f(1 - 2t + t^2) + 2*M_lp*(t - t^2) + M_g*t^2
    #      = M_f - 2*M_f*t + M_f*t^2 + 2*M_lp*t - 2*M_lp*t^2 + M_g*t^2
    #      = M_f + (2*M_lp - 2*M_f)*t + (M_f - 2*M_lp + M_g)*t^2
    # q'(t) = (2*M_lp - 2*M_f) + 2*(M_f - 2*M_lp + M_g)*t = 0
    # t* = (M_f - M_lp) / (M_f - 2*M_lp + M_g)
    
    denom = M_f - 2.0 * M_lp + M_g
    if abs(denom) < 1e-15:
        # Linear case
        if M_lp < M_f:
            return 1.0
        return 0.1
    t_star = (M_f - M_lp) / denom
    t_star = max(0.0, min(1.0, t_star))
    return t_star

def smoothed_max_grad(a, conv, tau):
    """Compute gradient of smoothed max of conv via FFT."""
    n = len(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    N = 1
    while N < 2 * n:
        N <<= 1
    from scipy.fft import fft, ifft
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n]


def gd_polish(a, time_budget, tau=100.0, verbose=False):

        return a

    start = time.monotonic()

        g = smoothed_max_grad(a, conv, tau)

        s_new = float(np.sum(a_new))

        else:
            stall += 1
            a = best_a

                lr *= 0.5
                stall = 0
                if lr < 1e-12:

                tau *= 2
                lr = 1.0 / (tau * np.sqrt(n))
        
        it += 1
        if verbose and it % 500 == 0:

Strategy: LP polishing with analytically optimal step size (quadratic bound),
dedicated peak subgradient phase for final equalization, and multi-n exploration.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:

                g_val = np.maximum(g.value, 0)

def compute_optimal_t(M_f, M_lp, M_g):

    Compute optimal t minimizing the quadratic upper bound:
    q(t) = (1-t)^2*M_f + 2t(1-t)*M_lp + t^2*M_g
    Returns t in [0, 1].
    """
    if M_g < M_f - 1e-15:
        return 1.0
    denom = M_f - 2.0 * M_lp + M_g
    if abs(denom) < 1e-15:
        t = 0.5
    else:
        t = (M_f - M_lp) / denom
    return max(0.01, min(1.0, t))

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        return a

    best_obj = compute_obj(a)
    start = time.monotonic()
    tau = 100.0
    lr = 1.0 / (tau * np.sqrt(n))
    it = 0
    stall = 0

    while time.monotonic() - start < budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 2 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

            print(f"  Final PeakEq: obj={best_score:.10f} **")
        else:
            print(f"  Final PeakEq: obj={obj_fe:.10f}")

    # One more quick LP if time
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = remaining * 0.6
        a_fl = lp_polish(best_seq, time_budget=alloc, verbose=False, max_iter=200)
        obj_fl = compute_obj(a_fl)
        if obj_fl < best_score:
            best_score = obj_fl
            best_seq = a_fl.copy()
            print(f"  Final LP2: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
