    n = len(a)
    a_norm = a / np.sum(a)
    m = len(active_indices)
    
    # Build constraint matrix: m x n
    A = np.zeros((m, n))
    for idx, k in enumerate(active_indices):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A[idx, j] = a_norm[k-j]

    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)

    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])

            # k-hi = k - min(n-1, k) = max(0, k-(n-1)) = lo
            # k-lo = k - max(0, k-n+1) = min(k, n-1) = hi
            # So A[idx, j] = a[k-j] for j in [lo, hi]
            # a[k-j] for j=lo gives a[k-lo] = a[hi]
            # a[k-j] for j=hi gives a[k-hi] = a[lo]
            # So A[idx, lo:hi+1] = a[hi:lo-1:-1] = a[hi::-1][:hi-lo+1]
            A[idx, lo:hi+1] = a[lo:hi+1][::-1]
    return A


def solve_lp_constraints(a, constraint_indices, time_limit=5.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for k in constraint_indices,
    sum(g) = 1, g >= 0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    m = len(constraint_indices)
    
    if m == 0:
        return np.ones(n) / n, 0.0
    
    A = build_constraint_matrix(a_norm, constraint_indices)
    
    # LP variables: [g_0, ..., g_{n-1}, M]
    # min M
    # s.t. A@g - M <= 0  (m constraints)
    #      sum(g) = 1
    #      g >= 0, M free
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    try:
        start = time.monotonic()
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.success:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[-1])
    except Exception:
        pass
    
    return None, None

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses active-set LP with constraint generation for fast inner iterations,
combined with smoothed gradient descent, multi-length exploration, and
adaptive blending. Key innovation: constraint-generation LP that maintains
a small active set for speed, with feasibility checks via FFT.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_ag(a, g):
    """Compute convolve(a, g) using FFT. Returns array of length n1+n2-1."""
    n1, n2 = len(a), len(g)
    size = 1
    while size < n1 + n2 - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g, size)
    return irfft(fa * fg, size)[:n1 + n2 - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_abmat(a, indices):
    """Build matrix (m x n) where row i gives conv(a,g) at lag indices[i]."""
    n = len(a)
    m = len(indices)
    A = np.zeros((m, n))
    for idx, k in enumerate(indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[idx, lo:hi + 1] = a[lo:hi + 1][::-1]
    return A


def solve_lp(a, constraint_indices, time_limit=4.0):
    """
    Solve LP: min M s.t. conv(a_norm,g)[k] <= M for k in constraint_indices,
    sum(g) = 1, g >= 0. Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    m = len(constraint_indices)
    if m == 0:
        return np.ones(n) / n, 0.0

    A = build_abmat(a_norm, constraint_indices)

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(-np.inf, np.inf)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[-1])
    except Exception:
        pass
    return None, None


def smoothed_gradient_fft(a, tau):
    """Compute gradient of smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = np.fft.fft(p, N)
    A = np.fft.fft(a, N)
    g_full = np.fft.ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv

def smoothed_gd(a, time_budget, tau_init=10.0, tau_end=500.0, verbose=False, seed=0):
    """Nesterov smoothed gradient descent on simplex with cosine tau annealing."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.8
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init + (tau_end - tau_init) * (1.0 - np.cos(np.pi * progress)) / 2.0

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, _ = smoothed_gradient_fft(y, tau)
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9998)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n)
            best_it = it

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        c1 = n // 4
        c2 = 3 * n // 4
        w = max(1, n // 8)
        a = np.zeros(n)
        for c in [c1, c2]:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine
        x = np.linspace(-0.9, 0.9, n)
        a = 1.0 / np.sqrt(1 - x**2)
        a = a / a.sum()
    elif mode == 4:
        # Random smooth
        a = rng.uniform(0, 1, n)
        k = max(2, n // 40)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 5:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.3 + 0.7 * np.cos(x / 2) ** 2
        a = a / a.sum()
    elif mode == 6:
        # Sparse
        a = np.zeros(n)
        k = n // 5
        indices = rng.choice(n, k, replace=False)
        a[indices] = 1.0
        a = a / a.sum()
    elif mode == 7:
        # Periodic
        x = np.linspace(0, 1, n)
        f = rng.integers(2, 8)
        a = 0.5 + 0.4 * np.sin(2 * np.pi * f * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def coordinate_descent(a, time_budget, seed=0):
    """Cyclic coordinate descent with small steps."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        order = np.arange(n)
        rng.shuffle(order)
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            old_val = a[i]
            step = 0.005 / np.sqrt(n)
            best_local = best_obj
            best_local_val = old_val
            for delta in [step, -step, 2 * step, -2 * step]:
                a_try = a.copy()
                a_try[i] = max(0, a_try[i] + delta)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_local:
                    best_local = obj_try
                    best_local_val = max(0, old_val + delta)
            if best_local_val != old_val:
                a[i] = best_local_val
                a = a / a.sum()
                if best_local < best_obj:
                    best_obj = best_local
                    best_a = a.copy()

    return best_a

                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    # === Phase 1: Active-set LP (main workhorse, 45% time) ===
    print(f"\n=== Phase 1: Active-Set LP (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.45, 600.0)
    if p1_budget > 30:
        a_lp = active_set_lp_optimize(best_seq, time_budget=p1_budget,
                                       max_active=250, verbose=True, seed=seed)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 LP: obj={obj_lp:.8f}")

    # === Phase 2: Smoothed GD polish (15% time) ===
    print(f"\n=== Phase 2: Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 200.0)
    if p2_budget > 20:
        a_gd = smoothed_gd(best_seq, time_budget=p2_budget,
                           tau_init=20.0, tau_end=800.0, verbose=False, seed=seed + 100)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase2 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 GD: obj={obj_gd:.8f}")

    # === Phase 3: Multi-n exploration (25% time) ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(300, n_current - 200), max(300, n_current - 100),
        n_current + 100, n_current + 200, n_current + 300,
        500, 600, 700, 900, 1000, 1100, 1200, 1400
    ]))
    # Filter: only lengths that differ from current and are reasonable
    n_options = [n for n in n_options if 300 <= n <= 1600 and abs(n - n_current) > 50]
    n_options = n_options[:5]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        alloc = min(remaining * 0.2, 120.0)
        if alloc > 20:
            a_try = active_set_lp_optimize(a_try, time_budget=alloc,
                                            max_active=200, verbose=False, seed=seed + n_try)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_try = smoothed_gd(a_try, time_budget=min(remaining * 0.1, 30.0),
                                     tau_init=30.0, tau_end=500.0, seed=seed + n_try + 1)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Intensive final polish (10% time) ===
    print(f"\n=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()

    # More LP iterations
    if remaining > 30:
        alloc = min(remaining * 0.4, 150.0)
        a_final = active_set_lp_optimize(best_seq, time_budget=alloc,
                                          max_active=300, verbose=True, seed=seed + 500)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 60.0)
        a_cd = coordinate_descent(best_seq, time_budget=alloc, seed=seed + 600)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_cd:.8f}")

    # One more GD at high tau
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 40.0)
        a_gd2 = smoothed_gd(best_seq, time_budget=alloc,
                            tau_init=100.0, tau_end=2000.0, seed=seed + 700)
        obj_gd2 = compute_obj(a_gd2)
        if obj_gd2 < best_score:
            best_score = obj_gd2
            best_seq = a_gd2.copy()
            print(f"  Final GD (high tau): obj={best_score:.8f} **")
        else:
            print(f"  Final GD: obj={obj_gd2:.8f}")

    # === Phase 5: Diverse restarts (remaining time) ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = (r * 3 + 1) % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.15 + 0.05 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.25, 80.0)
        if alloc > 15:
            a_r = active_set_lp_optimize(a_r, time_budget=alloc,
                                          max_active=200, verbose=False, seed=seed + 2000 + r)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
