```python
import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time
import scipy.sparse as sp

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * np.max(conv)

def build_sparse_conv_matrix(a):
    """Build sparse convolution matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):
            if k < 2*n - 1:
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def solve_lp(a, solver=None, warm_start=None):
    """Solve: min M s.t. A@g <= M, sum(g)=1, g>=0"""
    n = len(a)
    A = build_sparse_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    constraints = [
        A @ g <= M,
        cp.sum(g) == 1
    ]
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers_to_try = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.GLPK, cp.CBC]
    if solver:
        solvers_to_try = [solver] + solvers_to_try
    
    for sol in solvers_to_try:
        try:
            prob.solve(solver=sol)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = np.sum(g_val)
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def lp_iteration(a, max_iters=100, tol=1e-8):
    """Iterative LP: alternate between fixing a and solving for g."""
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(max_iters):
        g, M = solve_lp(best_a)
        if g is None:
            break
        
        # Full step
        new_a = g
        new_obj = compute_obj(new_a)
        
        if new_obj < best_obj:
            best_obj = new_obj
            best_a = new_a
        
        # Check convergence
        diff = np.max(np.abs(new_a - best_a))
        if diff < tol:
            break
        
        best_a = new_a  # Always update, even if obj didn't improve
    
    return best_a, best_obj
```

So `max_i h_i = ν` and `h_i = ν` exactly on the support of `a`.

This is a "water-filling" condition! The support of `a` is exactly the set of indices where `h` achieves its maximum.

1. Given `a`, determine the active set `S` (indices where `(a*a)_k ≈ M`)

3. Determine the new support: `new_support = {i : h_i >= ν - ε}`
4. Update `a` to be concentrated on the new support

```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast sparse LP-based iterative method with solver selection, subgradient polishing,
multi-n exploration, diverse initializations, and KKT-informed refinement.
Key ideas: exploit LP dual structure, warm-start solvers, use subgradient for final polish,
explore multiple sequence lengths, and leverage diverse starting points.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import scipy.sparse as sp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_sparse_conv_matrix(a):
    """Build sparse convolution matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        for k in range(k_start, k_end):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_fast(a, solver_pref=None):
    """
    Solve LP: min M s.t. A(a) @ g <= M, sum(g)=1, g>=0.
    Returns (g, M, dual_y) or (None, None, None).
    """
    n = len(a)
    A = build_sparse_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()

    conv_constraint = A @ g <= M
    constraints = [conv_constraint, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = []
    if solver_pref:
        solvers.append(solver_pref)
    for s in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK, cp.CBC]:
        if s not in solvers:
            solvers.append(s)

    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    g_val = g_val / s_val
                    m_val = float(M.value)
                    # Try to extract dual
                    dual_y = None
                    try:
                        dual_y = conv_constraint.dual_value
                        if dual_y is not None:
                            dual_y = np.maximum(dual_y, 0)
                            dy_sum = float(np.sum(dual_y))
                            if dy_sum > 1e-10:
                                dual_y = dual_y / dy_sum
                    except Exception:
                        dual_y = None
                    return g_val, m_val, dual_y
        except Exception:
            continue
    return None, None, None


def find_best_solver(n_test=200, budget=10.0):
    """Find the fastest LP solver for the given problem size."""
    a = np.ones(n_test) / n_test
    A = build_sparse_conv_matrix(a)

    g = cp.Variable(n_test, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    results = {}
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK, cp.CBC]
    start = time.monotonic()

    for solver in solvers:
        try:
            prob.solve(solver=solver)
            t = time.monotonic() - start
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                results[solver] = t
        except Exception:
            pass
        if time.monotonic() - start > budget:
            break

    if results:
        best = min(results, key=results.get)
        print(f"  Solver selection: {best} ({results[best]:.3f}s)")
        return best
    return cp.CLARABEL


def lp_iterate(a, max_iters=200, tol=1e-10, solver=None, verbose=False,
               alpha_schedule=None, time_budget=None, start_time=None):
    """
    Iterative LP: at each step, solve LP with fixed a, update a <- g.
    Uses blended updates with adaptive alpha.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64) / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)

    alpha = 1.0
    stalled = 0
    it = 0

    while it < max_iters:
        if time_budget is not None:
            if time.monotonic() - start_time > time_budget:
                break

        g, M_lp, dual_y = solve_lp_fast(a, solver=solver)
        if g is None:
            break

        # Adaptive alpha: start with full step, reduce if oscillating
        if alpha_schedule is None:
            if stalled < 3:
                alpha = 1.0
            elif stalled < 6:
                alpha = 0.5
            else:
                alpha = 0.2

        a_new = (1.0 - alpha) * a + alpha * g
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stalled = 0
        else:
            stalled += 1
            if stalled > 10:
                # Try full step anyway
                a_new = g.copy()
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-14:
                    best_a = a_new.copy()
                    best_obj = obj_new
                    stalled = 0
                else:
                    break

        # Convergence check
        diff = np.max(np.abs(a_new - a))
        if diff < tol:
            break

        a = a_new
        it += 1

        if verbose and it % 20 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, alpha={alpha:.2f}, diff={diff:.2e}")

    return best_a, best_obj


def subgradient_polish(a, time_budget=60.0, solver=None, verbose=False):
    """
    Subgradient method with exact max for final polishing.
    Uses Polyak step size and simplex projection.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64) / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Estimate optimal gap (use theoretical lower bound)
    lower_bound = 2.0 * n / (2 * n - 1)
    gap = best_obj - lower_bound

    lr = 0.1
    v = np.zeros(n)
    beta = 0.5
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        max_idx = np.where(conv == max_val)[0]

        # Subgradient at the max positions (average for stability)
        grad = np.zeros(n)
        for k in max_idx:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j]
        grad /= len(max_idx)

        # Momentum
        v = beta * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            gap = max(gap * 0.9, 1e-10)
        else:
            lr *= 0.95
            v *= 0.5

        a = a_new
        it += 1

        if verbose and it % 100 == 0:
            print(f"    Subgrad it{it}: obj={best_obj:.8f}, lr={lr:.6f}")

    return best_a, best_obj


def smoothed_gradient_descent(a, time_budget=120.0, tau_start=5.0, tau_end=500.0, verbose=False):
    """
    Nesterov accelerated smoothed gradient descent with log-sum-exp smoothing.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64) / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress ** 1.5

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient via cross-correlation (FFT)
        N = 1
        while N < 3 * n:
            N <<= 1
        P = np.fft.fft(p, N)
        A_f = np.fft.fft(y, N)
        g_full = np.fft.ifft(P * np.conj(A_f)).real
        g = 2.0 * g_full[:n]

        # Update
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        # Adaptive lr
        if it % 100 == 0:
            lr *= 0.99
            if it % 1000 == 0 and verbose:
                print(f"    SGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")

        # Reset if stuck for too long
        if it > 0 and it % 5000 == 0:
            obj_check = compute_obj(a)
            if obj_check > best_obj * 1.01:
                a = best_a.copy()
                v = np.zeros(n)
                lr = 3.0 / np.sqrt(n)

    return best_a, best_obj


def diverse_init(n, rng, mode=0):
    """Generate diverse initial sequences."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        # U-shaped (high at endpoints)
        x = np.linspace(0, 1, n)
        a = 1.0 / (0.1 + 4.0 * x * (1 - x))
        return a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(0.001, 0.999, n)
        a = 1.0 / np.sqrt(x * (1 - x))
        return a / a.sum()
    elif mode == 4:
        # Cosine bell
        x = np.linspace(-1, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * x))
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif mode == 5:
        # Two peaks
        c1, c2 = n // 3, 2 * n // 3
        w = n // 10
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 6:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** 3
        return a / a.sum()
    elif mode == 7:
        # Symmetric U
        x = np.linspace(0, 1, n)
        a = 1.0 + 2.0 * (2 * x - 1) ** 2
        return a / a.sum()
    elif mode == 8:
        # Optimal-like: heavy endpoints, cosine taper
        x = np.arange(n) / (n - 1)
        a = 0.3 + 0.7 * (1 - np.sin(np.pi * x)) ** 2
        return a / a.sum()
    elif mode == 9:
        # Random sparse
        a = np.zeros(n)
        n_sparse = max(2, n // 5)
        idx = rng.choice(n, n_sparse, replace=False)
        a[idx] = rng.uniform(0.5, 1.5, n_sparse)
        return a / a.sum()
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def kkt_check(a):
    """
    Check KKT conditions and return diagnostics.
    Returns (max_violation, active_set_fraction).
    """
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    tol = 1e-6

    # Active set: positions where conv is close to M
    active = np.where(conv > M - tol)[0]

    # Compute h_i = 2 * sum_k y_k * a[k-i] using uniform y over active set
    y = np.zeros(2 * n - 1)
    y[active] = 1.0 / len(active)

    # h = 2 * (y * a)
    N = 1
    while N < 3 * n:
        N <<= 1
    Y = np.fft.fft(y, N)
    A_f = np.fft.fft(a, N)
    h = 2.0 * np.fft.ifft(Y * A_f).real[:n]

    # On support of a, h should be constant
    support = np.where(a > 1e-10)[0]
    if len(support) == 0:
        return np.inf, 0.0

    h_support = h[support]
    h_mean = np.mean(h_support)
    max_violation = float(np.max(np.abs(h_support - h_mean)))

    # Fraction of active set
    active_frac = len(active) / (2 * n - 1)

    return max_violation, active_frac

    # Sort by objective

    print(f"  Best from scan: n={n_results[0][1]}, obj={n_results[0][0]:.8f}")

    if n_results[0][0] < best_score:
        best_score = n_results[0][0]
        best_seq = n_results[0][2].copy()

    # Pick top 3 n values for detailed optimization
    top_ns = []
    for obj_r, n_r, a_r in n_results:
        if n_r not in top_ns:
            top_ns.append(n_r)
        if len(top_ns) >= 3:
            break

    # === Phase 2: Detailed LP optimization (700s) ===
    print("\n=== Phase 2: Detailed LP Optimization ===")
    p2_deadline = start_time + 900.0

    for n_detail in top_ns:
        remaining = p2_deadline - time.monotonic()
        if remaining < 30:
            break

        # Start from the scan result for this n
        a_start = None
        for obj_r, n_r, a_r in n_results:
            if n_r == n_detail:
                a_start = a_r.copy()
                break
        if a_start is None:
            a_start = resample_sequence(best_seq, n_detail)

        # Multiple LP rounds with restarts
        n_alloc = remaining * 0.9
        for rnd in range(5):
            remaining = p2_deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(n_alloc / (5 - rnd) * 0.5, remaining * 0.4)
            if alloc < 10:
                break

            a_res, obj_res = lp_iterate(a_start, max_iters=200, tol=1e-10,
                                         solver=solver, time_budget=alloc,
                                         start_time=time.monotonic(),
                                         verbose=(rnd == 0))
            if obj_res < best_score:
                best_score = obj_res
                best_seq = a_res.copy()
                print(f"  n={n_detail}, round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_detail}, round {rnd}: obj={obj_res:.8f}")

            # For next round, use the result but add small perturbation
            if rnd < 4:
                pert = rng.normal(0, 0.01, n_detail)
                a_start = a_res + pert
                a_start = np.maximum(a_start, 0)
                a_start = a_start / a_start.sum()

    # === Phase 3: Subgradient polish (200s) ===
    print("\n=== Phase 3: Subgradient Polish ===")
    p3_deadline = start_time + 1100.0

    remaining = p3_deadline - time.monotonic()
    if remaining > 30:
        a_sub, obj_sub = subgradient_polish(best_seq, time_budget=min(remaining * 0.6, 120.0),
                                             solver=solver, verbose=True)
        if obj_sub < best_score:
            best_score = obj_sub
            best_seq = a_sub.copy()
            print(f"  Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj_sub:.8f}")

    # === Phase 4: SGD with different tau (150s) ===
    print("\n=== Phase 4: Smoothed GD ===")
    p4_deadline = start_time + 1250.0

    remaining = p4_deadline - time.monotonic()
    if remaining > 30:
        a_sgd, obj_sgd = smoothed_gradient_descent(best_seq, time_budget=min(remaining * 0.7, 100.0),
                                                    tau_start=10.0, tau_end=800.0, verbose=True)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  SGD: obj={best_score:.8f} **")
        else:
            print(f"  SGD: obj={obj_sgd:.8f}")

    # === Phase 5: Diverse restarts (150s) ===
    print("\n=== Phase 5: Diverse Restarts ===")
    p5_deadline = start_time + 1400.0
    n_best = len(best_seq)

    for r in range(8):
        remaining = p5_deadline - time.monotonic()
        if remaining < 20:
            break

        mode = r % 10
        a_div = diverse_init(n_best, rng, mode=mode)
        # Blend with current best
        blend = 0.3
        a_div = (1 - blend) * a_div + blend * best_seq
        a_div = np.maximum(a_div, 0)
        a_div = a_div / a_div.sum()

        alloc = min(remaining * 0.3, 25.0)
        if alloc < 5:
            break

        a_res, obj_res = lp_iterate(a_div, max_iters=50, tol=1e-9,
                                     solver=solver, time_budget=alloc,
                                     start_time=time.monotonic())
        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode {mode}): obj={obj_res:.8f}")

    # === Phase 6: Final intensive LP (100s) ===
    print("\n=== Phase 6: Final LP ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final, obj_final = lp_iterate(best_seq, max_iters=500, tol=1e-12,
                                         solver=solver, time_budget=remaining * 0.7,
                                         start_time=time.monotonic(), verbose=True)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

        # One more subgradient
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_sub2, obj_sub2 = subgradient_polish(best_seq, time_budget=remaining * 0.6,
                                                   solver=solver)
            if obj_sub2 < best_score:
                best_score = obj_sub2
                best_seq = a_sub2.copy()
                print(f"  Final subgrad: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = 2.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
