- sum(a) stays the same (if normalized)
- max_conv stays roughly the same (the peak of the convolution of a smooth function with itself is approximately proportional to the integral of f^2, which is shape-dependent)
- So the objective scales as ~n

Wait no, that can't be right because we know the objective can be < 2 for all n. Let me reconsider.

If a is a "bump" of height h and width w in a sequence of length n (with sum = 1, so h*w ≈ 1), then:
- max_conv ≈ h^2 * w = h^2 * (1/h) = h = 1/w (approximately)
- objective ≈ 2n / w

So for a bump of fixed width w, larger n gives larger objective. But we can also make the bump wider...

Actually the key tradeoff is: wider bumps have lower peaks but the convolution is spread out more. The optimal shape depends on n.

For the uniform sequence (wider than any bump), objective = 2. For very concentrated sequences, the peak is high. The optimum is somewhere in between.

I think the sweet spot for n is around 500-1500 based on the previous results (n=800 gave 1.506).

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of:
1. Nesterov accelerated smoothed gradient descent (mirror descent on simplex)
2. LP-based alternating optimization with golden-section line search
3. Multiplicative weight descent targeting top peaks
4. Peak-chasing via direct element modification
5. Multi-length exploration with structured initializations
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, M_target=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue
    
    return None, None


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def nesterov_mirror_descent(a, time_budget=300.0, tau_init=5.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated mirror descent (multiplicative) on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.0 / np.sqrt(n)
    beta = 0.85
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.95))
        # Exponential ramp for tau
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        if it % 10000 == 0 and verbose:
            print(f"    NMD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.4f}, t={elapsed:.0f}s")

        # Nesterov extrapolation
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        g, conv = smoothed_gradient_fft(y, tau)

        # Gradient step with multiplicative (mirror) update
        a_new = y * np.exp(-lr * g)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        # Update momentum
        v = a_new - a
        a = a_new
        it += 1

        # Adaptive learning rate
        if it % 100 == 0:
            if it - best_it > 10000:
                lr *= 0.5
                v = np.zeros(n)
                a = best_a.copy()
                best_it = it
            elif it - best_it > 5000:
                lr *= 0.8

    return best_a


def multiplicative_weight_descent(a, time_budget=200.0, K=10, tau_schedule=None, verbose=False, seed=42):
    """
    Multiplicative weight descent targeting top-K peaks.
    Directly reduces elements that contribute most to the highest peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    lr = 2.0 / np.sqrt(n)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = elapsed / time_budget

        # Compute convolution
        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Get top K peaks with softmax weights
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        # Temperature for peak selection
        peak_tau = 5.0 + 50.0 * progress
        weights = np.exp(peak_tau * (peak_vals - max_conv) / max_conv)
        weights /= weights.sum()

        # Compute gradient (pressure) at each index
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            # conv[k] = sum_i a[i]*a[k-i] for valid i
            # d conv[k] / d a[j] = a[k-j] if 0 <= k-j < n
            j_range = np.arange(max(0, k - n + 1), min(n, k + 1))
            a_vals = a[k - j_range]
            g[j_range] += w * a_vals

        # Multiplicative update
        scale = lr * (1.0 - 0.5 * progress)  # decrease step size over time
        a_new = a * np.exp(-scale * g)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall % 100 == 0:
                lr *= 0.8
                if lr < 1e-6:
                    # Perturb and restart
                    a = best_a * np.exp(rng.normal(0, 0.01, n))
                    a = np.maximum(a, 1e-15)
                    a = a / a.sum()
                    lr = 2.0 / np.sqrt(n)
                    stall = 0

        a = a_new
        it += 1

        if verbose and it % 5000 == 0:
            print(f"    MWD it{it}: obj={best_obj:.8f}, lr={lr:.4f}, t={elapsed:.0f}s")

    return best_a


def peak_chase(a, time_budget=100.0, verbose=False, seed=42):
    """
    Direct peak chasing: identify the peak, then modify elements to reduce it.
    Uses a "deflation" approach: reduce elements that are heavily involved in the peak.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    alpha = 0.1  # deflation factor

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_k = int(np.argmax(conv))
        max_val = conv[max_k]

        # Compute contribution of each element to the peak
        # conv[max_k] = sum_i a[i]*a[max_k-i] for valid i
        # The "pressure" on element j is a[max_k - j] (if valid)
        j_range = np.arange(max(0, max_k - n + 1), min(n, max_k + 1))
        pressure = a[max_k - j_range]

        # Deflate: reduce elements with high pressure
        # But redistribute to low-pressure elements
        a[j_range] -= alpha * pressure
        a[j_range] = np.maximum(a[j_range], 0)

        # Normalize
        s_new = float(np.sum(a))
        if s_new < 1e-10:
            a = best_a.copy()
            continue
        a = a / s_new

        obj_new = compute_obj(a)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a.copy()
            # Slightly increase deflation
            alpha = min(0.3, alpha * 1.02)
        else:
            # Revert and try less aggressive
            a = best_a.copy()
            alpha *= 0.7
            if alpha < 0.001:
                # Perturb
                a = best_a * np.exp(rng.normal(0, 0.005, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                alpha = 0.05

        it += 1
        if verbose and it % 100 == 0:
            print(f"    PC it{it}: obj={best_obj:.8f}, alpha={alpha:.4f}, t={time.monotonic()-start:.0f}s")

    return best_a


def lp_polish(a, time_budget=200.0, verbose=False, seed=42):

    a = a / s

    start = time.monotonic()

    it = 0

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:

        best_t = 0
        best_blend_obj = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)

            a_try = a_try / s_try
            obj_try = compute_obj(a_try)

        if best_t > 0:

            a_try = a_try / a_try.sum()
            best_obj = best_blend_obj
            best_a = a_try.copy()
            a = best_a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                # Perturb
                scale = 0.01 * min(no_improve, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                if it > 50:

        it += 1
        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}, M={M_lp:.6f}, el={time.monotonic()-start:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-0.999, 0.999, n)
        a = 1.0 / np.sqrt(1 - x**2)
        a = a / a.sum()
    elif mode == 4:
        # Raised cosine
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1.5, 0.5)
        a = a / a.sum()
    elif mode == 6:
        # Chirp-like (Barker inspired)
        phases = rng.uniform(0, 2*np.pi, n)
        freqs = rng.uniform(0.5, 3.0, n)
        a = 0.5 + 0.5 * np.cos(np.cumsum(freqs))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) Nesterov mirror descent with adaptive temperature, (2) LP-based
alternating with exact convex 1D line search exploiting quadratic structure,
(3) multiplicative weight descent targeting top peaks, (4) peak-chasing via
deflation, (5) systematic multi-n exploration with structured initializations,
and (6) perturbative restarts with re-optimization.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, M_target=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M <= M_target)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
            elif prob.status == 'infeasible':
                return None, None
        except Exception:
            continue

    return None, None


def conv_peak_function(a, g, t_val):
    """Compute max of ((1-t)*a + t*g) * ((1-t)*a + t*g) efficiently."""
    a_try = (1.0 - t_val) * a + t_val * g
    a_try = np.maximum(a_try, 0)
    s_try = float(np.sum(a_try))
    if s_try < 1e-10:
        return 1e10
    a_try = a_try / s_try
    conv = compute_conv(a_try)
    return float(np.max(conv))


def convex_1d_search(a, g, max_iter=60):
    """
    Find optimal t in [0,1] minimizing max(((1-t)*a + t*g)^2 / sum^2).
    Exploits convexity: the function is convex in t.
    Uses ternary search.
    """
    lo = 0.0
    hi = 1.0
    for _ in range(max_iter):
        if hi - lo < 1e-10:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = conv_peak_function(a, g, m1)
        f2 = conv_peak_function(a, g, m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def nesterov_mirror_descent(a, time_budget=300.0, tau_init=5.0, tau_end=500.0, verbose=False):

    n = len(a)

    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.5 / np.sqrt(n)
    beta = 0.85
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.95))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)

        if it % 10000 == 0 and verbose:
            print(f"    NMD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.4f}, t={elapsed:.0f}s")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:

        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)

def multiplicative_weight_descent(a, time_budget=200.0, K=8, verbose=False, seed=42):
    """
    Multiplicative weight descent targeting top-K peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    lr = 2.5 / np.sqrt(n)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        peak_tau = 3.0 + 40.0 * progress
        weights = np.exp(peak_tau * (peak_vals - max_conv) / max_conv)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            g[j_start:j_end] += w * a[k - j_start:k - j_end + 1][::-1]

        scale = lr * (1.0 - 0.6 * progress)
        a_new = a * np.exp(-scale * g)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall % 200 == 0:
                lr *= 0.7
                if lr < 1e-7:
                    a = best_a * np.exp(rng.normal(0, 0.008, n))
                    a = np.maximum(a, 1e-15)
                    a = a / a.sum()
                    lr = 2.5 / np.sqrt(n)
                    stall = 0

        a = a_new
        it += 1

        if verbose and it % 10000 == 0:
            print(f"    MWD it{it}: obj={best_obj:.8f}, lr={lr:.4f}, t={elapsed:.0f}s")

    return best_a
