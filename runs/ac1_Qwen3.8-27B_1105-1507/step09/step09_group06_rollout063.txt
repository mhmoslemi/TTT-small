In the LP, the peak M is the same for all lags. But some lags are "closer" to being active than others. We can add a small penalty to lags that are close to the peak, making the LP more focused on reducing the top peaks.

Modified LP:
min M + epsilon * sum_k max(0, conv(a,g)[k] - (1-delta)*M)^2

This is no longer an LP (it's a QP), but it might give better results. Alternatively, we can add a small linear penalty:
min M + epsilon * sum_k max(0, conv(a,g)[k] - (1-delta)*M_prev)

This is still an LP! The idea is to penalize lags that are close to the previous peak level.

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses aggressive LP-based optimization with t=1.0 updates, diverse spectral
perturbations, variable-length exploration, and subgradient refinement.
Key innovations: always-take LP solution, frequency-domain perturbation,
constrained active-set LP variant, adaptive n search.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu * rho > (cssv - 1.0)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_val = idx[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A_mat (2n-1 x n) such that A_mat @ g = convolve(a, g)."""
    n = len(a)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A_mat[j:j + r, j] = a[:r]
    return A_mat


def build_conv_matrix_sparse(a):
    """Build convolution matrix more efficiently using toeplitz-like structure."""
    n = len(a)
    # conv(a, g)[k] = sum_j a[j] * g[k-j] for valid j, k-j
    # This is a Toeplitz matrix
    from scipy.linalg import toeplitz
    # The first column is a[0], a[1], ..., a[n-1], 0, 0, ..., 0 (length 2n-1)
    # The first row is a[0], 0, 0, ..., 0
    col = np.zeros(2 * n - 1)
    col[:n] = a
    row = np.zeros(2 * n - 1)
    row[0] = a[0]
    # Actually, this isn't quite right for full convolution
    # Let me just use the direct construction
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_mat[j:j+n, j] = a
    return A_mat


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A_mat = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_minpeak_fast(a, solver=cp.CLARABEL):

    a_norm = a / np.sum(a)
    A_mat = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)

def lp_aggressive(a, time_budget=100.0, verbose=False, seed=0):
    """
    Aggressive LP optimization: always take t=1.0, iterate until convergence.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    no_improve_count = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Always take full LP solution (t=1.0)
        a_new = g_lp
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve_count = 0
            it += 1
            if verbose and it % 5 == 0:
                print(f"      LP aggressive it{it}: obj={best_obj:.8f}, M={M_lp:.6f}")
        else:
            no_improve_count += 1
            if no_improve_count > 5:
                break
            # Small blend to escape
            a = 0.9 * a + 0.1 * g_lp
            a = np.maximum(a, 0)
            a = a / a.sum()
            it += 1

    return best_a


def spectral_perturbation(a, scale=0.1, seed=0):
    """
    Perturb in the frequency domain: add random phase noise to the spectrum,
    then transform back and project to simplex.
    This creates smooth, structured perturbations.
    """
    n = len(a)
    rng = np.random.default_rng(seed)

    # FFT
    A = fft(a, n)

    # Add random phase perturbation (preserve amplitude roughly)
    freqs = np.arange(n)
    # More perturbation at low frequencies (affects global shape)
    low_freq_mask = freqs < n // 4
    phase_noise = rng.uniform(-np.pi, np.pi, n)
    amp_noise = 1 + scale * rng.normal(0, 1, n) * low_freq_mask
    amp_noise = np.maximum(0.5, amp_noise)

    A_new = A * amp_noise * np.exp(1j * phase_noise * scale)

    # Transform back
    a_new = ifft(A_new).real
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()

    return a_new


def peak_targeted_perturbation(a, scale=0.05, seed=0):
    """
    Perturb specifically at positions that contribute to the peak.
    Reduce values at peak-contributing positions, increase elsewhere.
    """
    n = len(a)
    rng = np.random.default_rng(seed)
    conv = compute_conv(a)

    # Find the top peak lag
    k_peak = int(np.argmax(conv))

    # Find positions that contribute most to this peak
    contrib = np.zeros(n)
    i_start = max(0, k_peak - (n - 1))
    i_end = min(n - 1, k_peak)
    for i in range(i_start, i_end + 1):
        j = k_peak - i
        if 0 <= j < n:
            contrib[i] = a[i] * a[j]

    # Reduce at high-contribution positions, increase elsewhere
    max_c = contrib.max() if contrib.max() > 0 else 1.0
    normalized_contrib = contrib / max_c

    # Create perturbation: decrease where contribution is high
    noise = rng.normal(0, 1, n) * scale
    adjustment = -normalized_contrib * scale * 2 + noise * 0.5
    a_new = a * np.exp(adjustment)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussians
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-0.999, 0.999, n)
        a = 1.0 / np.sqrt(1 - x**2)
        a = a / a.sum()
    elif mode == 4:
        # Random smooth
        a = rng.uniform(0, 1, n)
        k = max(2, n // 30)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = a / a.sum()
    elif mode == 5:
        # Cosine taper
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
        a = a / a.sum()
    elif mode == 6:
        # Sinc-like
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 7:
        # Fourier series
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def subgradient_descent(a, time_budget=100.0, top_k=5, verbose=False, seed=0):
    """
    Subgradient method using the active set (top k peaks).
    Different from smoothed gradient: uses only the top peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 2.0 / np.sqrt(n)
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Find top k peaks
        peak_indices = np.argsort(conv)[::-1][:top_k]

        # Compute subgradient: average of gradients at top peaks
        g = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            for i in range(n):
                j1 = k - i
                j2 = i - k
                term = 0.0
                if 0 <= j1 < n:
                    term += a[j1]
                if 0 <= j2 < n and j2 != j1:
                    term += a[j2]
                g[i] += term

        g /= top_k

        # Gradient step with projection
        a_new = a - lr * g
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        it += 1

        # Decay learning rate
        lr *= 0.999
        if lr < 1e-6:
            lr = 2.0 / np.sqrt(n)
            no_improve = 0

        if no_improve > 5000:
            break

        if verbose and it % 500 == 0:
            print(f"      Subgrad it{it}: obj={best_obj:.8f}, lr={lr:.6f}")

    return best_a


def multi_lp_restart(a, n_restarts=5, time_per_restart=30.0, verbose=False, seed=0):
    """
    For each restart, apply a different perturbation then run LP optimization.
    """
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    for r in range(n_restarts):
        elapsed = time.monotonic() - start
        if elapsed + time_per_restart > 120:
            break

        # Different perturbation strategy for each restart
        if r == 0:
            a_try = spectral_perturbation(a, scale=0.1, seed=seed + r * 100)
        elif r == 1:
            a_try = peak_targeted_perturbation(a, scale=0.08, seed=seed + r * 100)
        elif r == 2:
            a_try = (0.7 * a + 0.3 * diverse_init(n, rng, mode=(r % 8)))
            a_try = a_try / a_try.sum()
        elif r == 3:
            # Random smooth perturbation
            noise = rng.normal(0, 0.05, n)
            k = max(2, n // 20)
            kernel = np.ones(k) / k
            noise = np.convolve(noise, kernel, mode='same')
            a_try = a * np.exp(noise)
            a_try = np.maximum(a_try, 0) / a_try.sum()
        else:
            # Frequency perturbation with different scale
            a_try = spectral_perturbation(a, scale=0.15 + 0.05 * r, seed=seed + r * 100)

        # LP optimize
        a_opt = lp_aggressive(a_try, time_budget=time_per_restart,
                              verbose=False, seed=seed + r)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"      Multi-LP restart {r}: obj={best_obj:.8f} **")
        elif verbose:
            print(f"      Multi-LP restart {r}: obj={obj_opt:.8f}")

    return best_a

The gradient of (a*a)[k] with respect to a[i] is:
∂(a*a)[k]/∂a[i] = a[k-i] (if 0 <= k-i < n) + a[i-k] (if 0 <= i-k < n and i-k != k-i)

Actually, let me be more careful. The convolution is:
(a*a)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]

The derivative with respect to a[i]:

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses aggressive LP optimization with t=1.0 updates, frequency-domain perturbations,
peak-targeted reduction, subgradient refinement with active sets, variable-n search,
and diverse restart strategies. Key innovations: always-take LP solution (no blending),
spectral shaping perturbations, fast subgradient with corrected gradient formula,
and systematic multi-scale exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu * rho > (cssv - 1.0)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_val = idx[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A_mat (2n-1 x n) such that A_mat @ g = convolve(a, g)."""
    n = len(a)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A_mat[j:j + r, j] = a[:r]
    return A_mat


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A_mat = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_topk(a, top_k=3):
    """
    Solve LP targeting only the top-k peaks:
    min M s.t. conv(a,g)[k] <= M for k in top-k, sum(g)=1, g>=0.
    This is a relaxed LP (fewer constraints) that may give a more targeted update.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv_current = compute_conv(a_norm)
    peak_indices = np.argsort(conv_current)[::-1][:top_k]

    A_mat = build_conv_matrix(a_norm)
    # Use only constraints for peak indices (plus a few neighbors for safety)
    constraint_indices = set()
    for k in peak_indices:
        for dk in range(-2, 3):
            kk = int(k) + dk
            if 0 <= kk < 2 * n - 1:
                constraint_indices.add(kk)
    constraint_indices = sorted(constraint_indices)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat[k] @ g <= M for k in constraint_indices]
    constraints.append(cp.sum(g) == 1)
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Check if this is actually better on the full objective
                    return g_val, None
        except Exception:
            continue
    return None, None


def lp_aggressive(a, time_budget=100.0, verbose=False, seed=0):
    """
    Aggressive LP optimization: always take t=1.0 (full LP solution).
    Iterates until no improvement or time budget exhausted.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    no_improve_count = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Always take full LP solution
        a_new = g_lp
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve_count = 0
            it += 1
            if verbose and (it % 5 == 0 or it <= 3):
                print(f"      LP it{it}: obj={best_obj:.8f}, M={M_lp:.6f}, t={time.monotonic()-start:.1f}s")
        else:
            no_improve_count += 1
            if no_improve_count > 5:
                break
            # Small blend to try to escape
            a = 0.85 * a + 0.15 * g_lp
            a = np.maximum(a, 0)
            a = a / a.sum()
            it += 1

    return best_a

def spectral_perturbation(a, scale=0.1, seed=0, mode=0):
    """
    Perturb in the frequency domain to create smooth structured perturbations.
    mode=0: random phase + amplitude
    mode=1: phase only
    mode=2: amplitude only (low freq boost)
    """
    n = len(a)
    rng = np.random.default_rng(seed)

    A = fft(a, n)
    freqs = np.arange(n)
    low_mask = freqs < n // 4

    if mode == 0:
        phase_noise = rng.uniform(-np.pi, np.pi, n) * scale
        amp_noise = 1.0 + scale * rng.normal(0, 1, n) * low_mask
        amp_noise = np.maximum(0.3, amp_noise)
        A_new = A * amp_noise * np.exp(1j * phase_noise)
    elif mode == 1:
        phase_noise = rng.uniform(-np.pi, np.pi, n) * scale
        A_new = A * np.exp(1j * phase_noise)
    elif mode == 2:
        amp_noise = 1.0 + scale * rng.normal(0, 1, n)
        amp_noise[n//2 - n//8:n//2 + n//8] += scale * 0.5
        amp_noise = np.maximum(0.3, amp_noise)
        A_new = A * amp_noise
    else:
        phase_noise = rng.uniform(-np.pi, np.pi, n) * scale * 0.5
        amp_noise = 1.0 + 0.5 * scale * rng.normal(0, 1, n) * low_mask
        amp_noise = np.maximum(0.5, amp_noise)
        A_new = A * amp_noise * np.exp(1j * phase_noise)

    a_new = ifft(A_new).real
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()

    return a_new


def peak_targeted_perturbation(a, scale=0.08, seed=0):
    """
    Reduce values at positions contributing to the peak, increase elsewhere.
    """
    n = len(a)
    rng = np.random.default_rng(seed)
    conv = compute_conv(a)

    # Find top 3 peak lags
    top_peaks = np.argsort(conv)[::-1][:3]

    # Compute contribution to these peaks
    contrib = np.zeros(n)
    for k in top_peaks:
        k = int(k)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                contrib[i] += a[j]

    max_c = contrib.max() if contrib.max() > 0 else 1.0
    norm_contrib = contrib / max_c

    # Decrease at high-contribution, slight increase elsewhere
    noise = rng.normal(0, 1, n) * scale * 0.3
    adjustment = -norm_contrib * scale + noise
    a_new = a * np.exp(adjustment)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def anti_concentration_perturbation(a, scale=0.1, seed=0):
    """
    Push the sequence toward more uniform distribution (anti-concentration).
    Useful when the sequence has developed sharp peaks.
    """
    n = len(a)
    rng = np.random.default_rng(seed)

    # Add uniform component
    uniform = np.ones(n) / n
    blend = scale
    a_new = (1 - blend) * a + blend * uniform
    # Add small smooth noise
    noise = rng.normal(0, 1, n)
    k = max(2, n // 20)
    kernel = np.ones(k) / k
    noise = np.convolve(noise, kernel, mode='same')
    a_new = a_new + 0.02 * noise
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    return a_new


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2.0 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-0.999, 0.999, n)
        a = 1.0 / np.sqrt(1.0 - x**2)
        a = a / a.sum()
    elif mode == 4:
        a = rng.uniform(0, 1, n)
        k = max(2, n // 30)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 7:
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 8:
        # Multiple Gaussian bumps
        n_bumps = rng.integers(2, 5)
        a = np.zeros(n)
        for _ in range(n_bumps):
            c = rng.integers(0, n)
            w = rng.integers(max(1, n // 20), n // 5)
            a += np.exp(-((np.arange(n) - c) ** 2) / (2.0 * w ** 2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def subgradient_descent(a, time_budget=100.0, top_k=5, verbose=False, seed=0):
    """
    Subgradient method using active set (top k peaks).
    Gradient formula: d(a*a)[k]/d(a[i]) = 2*a[k-i] if 0<=k-i<n, else 0.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 1.0 / np.sqrt(n)
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_indices = np.argsort(conv)[::-1][:top_k]

        # Subgradient: g[i] = (1/top_k) * sum_{k in peaks} 2*a[k-i]
        g = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            # For each i, j = k-i must be in [0, n)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            for i in range(i_start, i_end):
                j = k - i
                if 0 <= j < n:
                    g[i] += 2.0 * a[j]
        g /= float(top_k)

        # Gradient step with projection
        a_new = a - lr * g
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9995)

        if no_improve > 2000:
            a = best_a.copy()
            lr = 1.0 / np.sqrt(n)
            no_improve = 0

        if verbose and it % 1000 == 0:
            print(f"      Subgrad it{it}: obj={best_obj:.8f}, lr={lr:.6f}, t={time.monotonic()-start:.1f}s")

    return best_a


def smoothed_gradient_step(a, tau, lr):
    """
    Single step of smoothed gradient descent on the log-sum-exp smoothed max.
    """
    n = len(a)
    a_norm = a / a.sum()
    conv = compute_conv(a_norm)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a_norm, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]

    a_new = a_norm - lr * g
    a_new = project_simplex(a_new)
    return a_new


def multi_lp_restart(a, n_restarts=5, time_per_restart=30.0, verbose=False, seed=0):
    """
    For each restart, apply a different perturbation then run LP optimization.
    Uses diverse strategies to explore different basins of attraction.
    """
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    strategies = [
        ("spectral0", lambda: spectral_perturbation(a, scale=0.12, seed=seed, mode=0)),
        ("spectral1", lambda: spectral_perturbation(a, scale=0.08, seed=seed+1, mode=1)),
        ("spectral2", lambda: spectral_perturbation(a, scale=0.15, seed=seed+2, mode=2)),
        ("peak_target", lambda: peak_targeted_perturbation(a, scale=0.1, seed=seed+3)),
        ("anti_conc", lambda: anti_concentration_perturbation(a, scale=0.15, seed=seed+4)),
        ("blend_div", lambda: (0.6 * a + 0.4 * diverse_init(n, rng, mode=(3 + n_restarts) % 9))),
        ("smooth_noise", None),  # handled separately
        ("spectral4", lambda: spectral_perturbation(a, scale=0.06, seed=seed+7, mode=3)),
    ]

    for r in range(min(n_restarts, len(strategies))):
        elapsed = time.monotonic() - start
        if elapsed + time_per_restart > 150:
            break

        name, func = strategies[r]
        if func is None:
            # Smooth noise
            noise = rng.normal(0, 0.06, n)
            k = max(2, n // 15)
            kernel = np.ones(k) / k
            noise = np.convolve(noise, kernel, mode='same')
            a_try = a * np.exp(noise)
            a_try = np.maximum(a_try, 0) / a_try.sum()
        else:
            a_try = func()
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try

        # LP optimize
        a_opt = lp_aggressive(a_try, time_budget=time_per_restart,
                              verbose=False, seed=seed + r * 100)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"      Restart {r} ({name}): obj={best_obj:.8f} **")
        elif verbose:
            print(f"      Restart {r} ({name}): obj={obj_opt:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining aggressive LP optimization, spectral
    perturbations, peak-targeted strategies, subgradient refinement,
    and variable-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Phase 0: Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={n_target}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Aggressive LP (always t=1.0) ===
    print(f"\n=== Phase 1: Aggressive LP ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.18, 200.0)
    if p1_budget > 20:
        a_lp = lp_aggressive(best_seq, time_budget=p1_budget,
                              verbose=True, seed=seed)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj_lp:.8f}")

    # === Phase 2: Top-k focused LP ===
    print(f"\n=== Phase 2: Top-k LP ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 150.0)
    if p2_budget > 20:
        a_topk = lp_with_topk_focus(best_seq, time_budget=p2_budget,
                                     verbose=True, seed=seed + 50)
        obj_topk = compute_obj(a_topk)
        if obj_topk < best_score:
            best_score = obj_topk
            best_seq = a_topk.copy()
            print(f"  Phase2: obj={best_score:.8f} **")
        else:
            print(f"  Phase2: obj={obj_topk:.8f}")

    # === Phase 3: Multi-perturbation LP restart ===
    print(f"\n=== Phase 3: Multi-Perturbation LP ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.25, 350.0)
    if p3_budget > 50:
        n_restarts = max(4, int(p3_budget / 45))
        time_per = min(50.0, p3_budget / max(1, n_restarts))
        a_multi = multi_lp_restart(best_seq, n_restarts=n_restarts,
                                    time_per_restart=time_per,
                                    verbose=True, seed=seed + 100)
        obj_multi = compute_obj(a_multi)
        if obj_multi < best_score:
            best_score = obj_multi
            best_seq = a_multi.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj_multi:.8f}")

    # === Phase 4: Subgradient refinement ===
    print(f"\n=== Phase 4: Subgradient Refinement ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.1, 100.0)
    if p4_budget > 20:
        a_sg = subgradient_descent(best_seq, time_budget=p4_budget,
                                    top_k=5, verbose=True, seed=seed + 200)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Phase4: obj={best_score:.8f} **")
        else:
            print(f"  Phase4: obj={obj_sg:.8f}")

    # === Phase 5: Variable-n search ===
    print(f"\n=== Phase 5: Variable-n Search ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(500, n_current - 100), max(600, n_current - 50),
        n_current + 50, n_current + 100, n_current + 200,
        n_current + 500,
        700, 800, 900, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_current) >= 50]
    n_options = n_options[:5]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        alloc = min(remaining * 0.12, 60.0)
        if alloc > 15:
            a_try = lp_aggressive(a_try, time_budget=alloc, verbose=False, seed=seed + n_try)

        remaining = deadline - time.monotonic()
        if remaining > 25:
            a_try = multi_lp_restart(a_try, n_restarts=2,
                                      time_per_restart=min(20.0, remaining * 0.1),
                                      verbose=False, seed=seed + n_try + 50)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Second round of multi-perturbation on best ===
    print(f"\n=== Phase 6: Second Multi-Perturbation ===")
    remaining = deadline - time.monotonic()
    p6_budget = min(remaining * 0.2, 200.0)
    if p6_budget > 40:
        # Use different seed range to explore different directions
        a_multi2 = multi_lp_restart(best_seq, n_restarts=4,
                                     time_per_restart=min(40.0, p6_budget / 4),
                                     verbose=True, seed=seed + 5000)
        obj_multi2 = compute_obj(a_multi2)
        if obj_multi2 < best_score:
            best_score = obj_multi2
            best_seq = a_multi2.copy()
            print(f"  Phase6: obj={best_score:.8f} **")
        else:
            print(f"  Phase6: obj={obj_multi2:.8f}")

    # === Phase 7: Final intensive LP polish ===
    print(f"\n=== Phase 7: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    p7_budget = min(remaining * 0.3, 150.0)
    if p7_budget > 15:
        a_final = lp_aggressive(best_seq, time_budget=p7_budget,
                                 verbose=True, seed=seed + 9000)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

        # One more subgradient pass
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_sg2 = subgradient_descent(best_seq, time_budget=min(remaining * 0.4, 40.0),
                                         top_k=3, verbose=False, seed=seed + 9500)
            obj_sg2 = compute_obj(a_sg2)
            if obj_sg2 < best_score:
                best_score = obj_sg2
                best_seq = a_sg2.copy()
                print(f"  Final Subgrad: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
