def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.

        if np.isnan(x) or np.isinf(x):

    sequence = [float(x) for x in sequence]

    sequence = [max(0, x) for x in sequence]

        return np.inf

def spectral_shaping(a, time_budget, verbose=False):
    """
    Try to flatten the power spectrum to reduce autocorrelation peaks.
    Iteratively: compute spectrum, flatten it, invert, project to simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        if time.monotonic() - start > time_budget:
            break
        for it in range(20):
            if time.monotonic() - start > time_budget:
                break
            # Compute power spectrum
            size = 1
            while size < 2 * n:
                size <<= 1
            A = rfft(a, size)
            P = np.abs(A) ** 2

            # Flatten: P_new = (1-alpha)*P + alpha*mean(P)
            P_flat = (1 - alpha) * P + alpha * np.mean(P)
            P_flat = np.maximum(P_flat, 0)

            # Invert to get candidate sequence
            A_new = np.sqrt(P_flat) * (A / (np.abs(A) + 1e-30))
            a_new = irfft(A_new, size)[:n]
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
                if verbose:
                    print(f"          Spectral alpha={alpha}, it{it}: obj={best_obj:.8f}")
            else:
                break

    return best_a

                break

            # Find contributors to this lag

            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    contributors.append((a[i] * a[j], i, j))

                if time.monotonic() - start > time_budget:
                    break
                if contrib < 1e-10:
                    continue
                src = i if a[i] >= a[j] else j
                for target in quiet_positions:
                    if target == src:
                        continue

def smooth_optimize(a, time_budget, beta_start=50.0, beta_end=3000.0, verbose=False):
    """
    Smooth optimization using log-sum-exp surrogate with mirror descent.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.0 / np.sqrt(n)
    it = 0
    total_iters = max(1, int(time_budget * 3))

    while time.monotonic() - start < time_budget:
        frac = min(1.0, it / total_iters)
        beta = beta_start * (beta_end / beta_start) ** frac

        conv = compute_conv(a)
        m = conv.max()
        shifted = beta * (conv - m)
        weights = np.exp(shifted)
        weights /= weights.sum()

        # Gradient: g[i] = 2 * sum_k weights[k] * a[k-i] (cross-correlation)
        g = np.zeros(n)
        for i in range(n):
            k_start = 0
            k_end = min(2 * n - 1 - 1, i + n - 1)
            a_start = max(0, k_start - i)
            a_end = min(n - 1, k_end - i)
            if a_start <= a_end:
                ks = i + a_start
                ke = i + a_end
                g[i] = 2.0 * np.dot(a[a_start:a_end + 1], weights[ks:ke + 1])

        # Mirror descent
        log_a = np.log(a)
        log_a_new = log_a - lr * g
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = max(1e-6, lr * 0.995)
        else:
            lr = max(1e-6, lr * 0.9)

        it += 1
        if verbose and it % 100 == 0:
            print(f"          Smooth it{it}: obj={best_obj:.8f}, beta={beta:.0f}")

    return best_a


def diverse_init(n, rng, mode=0):

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
