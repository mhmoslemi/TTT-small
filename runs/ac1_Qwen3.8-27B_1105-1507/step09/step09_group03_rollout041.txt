def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses multiplicative weights (mirror descent) with adaptive temperature,
periodic LP polishing, block refinement, multi-n exploration, and diverse restarts.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_grad_fft(a, tau):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*C[k])) where C=conv(a,a).
    Returns (grad, C, p) where p=softmax(tau*C).
    """
    n = len(a)
    C = compute_conv(a)
    maxC = float(np.max(C))
    p = np.exp(tau * (C - maxC))
    p /= float(np.sum(p))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    xc = ifft(P * np.conj(A)).real
    grad = 2.0 * xc[:n]
    return grad, C, p


def multiplicative_weights(a, time_budget, tau_init=10.0, tau_end=300.0,
                           eta_init=0.5, verbose=False):
    """
    Mirror descent with multiplicative weights on the simplex.
    a_new ∝ a * exp(-eta * grad), then normalize.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = eta_init
    it = 0
    best_it = 0
    patience = 5000

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.95))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)

        grad, C, p = smoothed_grad_fft(a, tau)

        # Multiplicative update
        # Scale grad to have appropriate magnitude
        grad_scale = float(np.max(np.abs(grad))) + 1e-10
        normalized_grad = grad / grad_scale

        a_new = a * np.exp(-eta * normalized_grad)
        s_new = float(np.sum(a_new))
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            a = a_new
        else:

            a = a_new  # Still update to continue the trajectory

        it += 1
        eta *= 0.9999  # Slowly decrease step size

        if verbose and it % 10000 == 0:
            print(f"    MW it{it}: obj={best_obj:.8f}, tau={tau:.1f}, eta={eta:.4f}")

        if it - best_it > patience:
            # Reset
            a = best_a.copy()
            eta = eta_init
            best_it = it
            patience = max(1000, patience // 2)

    return best_a

    while time.monotonic() - start < time_budget:

            if float(np.dot(diff, grad_diff)) <= 0.5 * L * float(np.dot(diff, diff)) + 1e-14:

            L *= 2.0
            if L > 1e6:

        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y > 1e-15:
            y = y / s_y
        else:
            y = z_new.copy()

        z = z_new
        t_k = t_next

        obj = compute_obj(z)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = z.copy()
            best_it = it

            y = best_a.copy()
            t_k = 1.0

def solve_lp_minmax(a, solver_list=None):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    
    # Build constraint matrix
    # A[k, j] = a_n[k-j] if 0 <= k-j < n
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[mask] = a_n[diff[mask]]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget, verbose=False):
    """Polish using LP iterations with golden section line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        if it % 5 == 0:
            g_lp, M_lp = solve_lp_minmax(a)
            if g_lp is None:
                break

            # Golden section line search between a and g_lp
            best_t = 0.0
            best_t_obj = best_obj
            lo, hi = 0.0, 1.0
            for _ in range(6):
                t1 = hi - 0.618 * (hi - lo)
                t2 = lo + 0.618 * (hi - lo)
                a1 = (1 - t1) * a + t1 * g_lp
                a1 = np.maximum(a1, 0)
                s1 = float(np.sum(a1))
                f1 = compute_obj(a1) if s1 > 1e-10 else np.inf
                a2 = (1 - t2) * a + t2 * g_lp
                a2 = np.maximum(a2, 0)
                s2 = float(np.sum(a2))
                f2 = compute_obj(a2) if s2 > 1e-10 else np.inf
                if f1 < f2:
                    hi = t2
                else:
                    lo = t1
                if f1 < best_t_obj:
                    best_t_obj = f1
                    best_t = t1
                if f2 < best_t_obj:
                    best_t_obj = f2
                    best_t = t2

            if best_t_obj < best_obj - 1e-13:
                a_new = (1 - best_t) * a + best_t * g_lp
                a_new = np.maximum(a_new, 0)
                a_new = a_new / float(np.sum(a_new))
                best_obj = best_t_obj
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                # Perturb
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / float(np.sum(a))
                if stall > 30:
                    break
        else:
            # Use gradient step between LP calls
            tau = 50.0
            grad, _, _ = smoothed_grad_fft(a, tau)
            eta = 0.1
            a_new = a * np.exp(-eta * grad / (float(np.max(np.abs(grad))) + 1e-10))
            a_new = np.maximum(a_new, 0)
            s_n = float(np.sum(a_new))
            if s_n > 1e-10:
                a_new = a_new / s_n
                if compute_obj(a_new) < best_obj:
                    best_obj = compute_obj(a_new)
                    best_a = a_new.copy()
                    a = a_new

        it += 1
        if verbose and it % 10 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}")

    return best_a


def block_lp_refine(a, time_budget, block_size=50, verbose=False):
    """
    Block coordinate refinement: split into blocks, for each block solve
    a small LP that minimizes the max convolution contribution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    n_blocks = n // block_size
    if n_blocks < 2:
        return best_a

    for round_pass in range(20):
        if time.monotonic() - start > time_budget:
            break

        for b in range(n_blocks):
            if time.monotonic() - start > time_budget:
                break

            b_start = b * block_size
            b_end = min((b + 1) * block_size, n)
            b_len = b_end - b_start

            # Fix all except block b, optimize block b
            # The contribution of block b to conv(a,a) involves:
            # - conv within block b
            # - cross terms with other blocks
            # For simplicity, we solve: min max_k (C_fixed[k] + C_block[k](x))
            # where x is the block variable, C_fixed is the convolution of the
            # fixed part, and C_block is the contribution involving x.

            # Build the "fixed" convolution (zero out block b)
            a_fixed = a.copy()
            a_fixed[b_start:b_end] = 0

            # Convolution of fixed part
            C_fixed = compute_conv(a_fixed)

            # For the block, we need: C_block[k](x) = conv(a_fixed, x shifted) + conv(x shifted, a_fixed) + conv(x, x)
            # This is complex. Let's use a simpler approach:
            # Just do gradient-based optimization on the block.

            # Compute gradient restricted to the block
            tau = 30.0
            grad_full, C, _ = smoothed_grad_fft(a, tau)
            grad_block = grad_full[b_start:b_end]

            # Multiplicative update on the block
            eta = 0.3
            a_try = a.copy()
            block_vals = a[b_start:b_end].copy()
            block_vals = block_vals * np.exp(-eta * grad_block / (float(np.max(np.abs(grad_block))) + 1e-10))
            block_vals = np.maximum(block_vals, 0)

            # Keep sum of block approximately constant
            s_block_old = float(np.sum(a[b_start:b_end]))
            s_block_new = float(np.sum(block_vals))
            if s_block_new > 1e-15 and s_block_old > 1e-15:
                block_vals = block_vals * (s_block_old / s_block_new)
            a_try[b_start:b_end] = block_vals
            a_try = a_try / float(np.sum(a_try))

            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-10))
        a = a / a.sum()
    elif mode == 4:
        # Random Fourier
        freqs = rng.integers(1, 6, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Cosine
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    else:
        # Kronecker-like: concentrated at edges
        a = np.zeros(n)
        a[0] = 1
        a[-1] = 1
        a[n//2] = 0.5
        a = a / a.sum()
    return a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses active-set LP polishing, multiplicative weights (mirror descent) with FFT-based
smoothed gradients, FISTA, spectral diversification, multi-n exploration,

import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_grad_fft(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*C[k])) where C=conv(a,a).
    Returns (grad, C, p) where p = softmax(tau*C).
    grad[j] = 2 * sum_m p[j+m] * a[m]
    """
    n = len(a)
    C = compute_conv(a)
    maxC = float(np.max(C))
    p = np.exp(tau * (C - maxC))
    p /= float(np.sum(p))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    xc = ifft(P * np.conj(A)).real
    grad = 2.0 * xc[:n]
    return grad, C, p


def peak_entropy(p):
    """Shannon entropy of the softmax over peaks. Low = concentrated, High = flat."""
    p = p[p > 1e-15]
    if len(p) == 0:
        return 0.0
    return float(-np.sum(p * np.log(p)))


def multiplicative_weights(a, time_budget, tau_init=10.0, tau_end=400.0,
                           eta_init=0.8, verbose=False):
    """Mirror descent with multiplicative weights on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    eta = eta_init
    it = 0
    best_it = 0
    patience = 8000

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.95))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)

        grad, C, p = smoothed_grad_fft(a, tau)

        # Adaptive eta based on peak concentration
        h = peak_entropy(p)
        max_h = np.log(2 * n - 1)
        eta_scale = 0.5 + 0.5 * (h / max_h)
        eta_eff = eta * eta_scale

        gscale = float(np.max(np.abs(grad))) + 1e-10
        a_new = a * np.exp(-eta_eff * grad / gscale)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        a = a_new

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        it += 1
        eta *= 0.99985

        if verbose and it % 20000 == 0:
            print(f"    MW it{it}: obj={best_obj:.8f}, tau={tau:.1f}, eta={eta:.4f}, H={h:.2f}")

        if it - best_it > patience:
            a = best_a.copy()
            eta = eta_init * 0.5
            best_it = it
            patience = max(2000, patience // 2)
            if patience <= 2000:
                break

    return best_a


def fista_robust(a, time_budget, tau_init=20.0, tau_end=800.0, verbose=False):
    """FISTA on smoothed objective over simplex with simple step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    z = a.copy()
    y = a.copy()
    t_k = 1.0
    L = 1.0
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.95))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)

        grad_y, _, _ = smoothed_grad_fft(y, tau)
        z_new = project_simplex(y - grad_y / L)

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = z_new + (t_k - 1.0) / t_next * (z_new - z)
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y > 1e-15:
            y = y / s_y
        else:
            y = z_new.copy()

        z = z_new
        t_k = t_next
        L *= 0.999  # Slowly decrease L (increase step size)

        obj = compute_obj(z)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = z.copy()
            best_it = it

        it += 1
        if verbose and it % 2000 == 0:
            print(f"    FISTA it{it}: obj={best_obj:.8f}, tau={tau:.1f}, L={L:.3f}")

        if it - best_it > 30000:
            z = best_a.copy()
            y = best_a.copy()
            t_k = 1.0
            L = 0.5
            best_it = it

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[mask] = a[diff[mask]]
    return A


def solve_lp_minmax(a, active_indices=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M (for all k or active only), sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)

    if active_indices is not None:
        A_sub = A[active_indices, :]
    else:
        A_sub = A

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_sub @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solver_list = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_min_sum_active(a, active_indices):
    """Solve LP: min sum(conv(a,g)[k] for k in active) s.t. sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    A_sub = A[active_indices, :]

    g = cp.Variable(n, nonneg=True)
    obj = cp.sum(A_sub @ g)
    constraints = [cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(obj), constraints)

    solver_list = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def golden_section_search(f, lo, hi, n_evals=7):
    """Golden section search for minimum of f on [lo, hi]."""
    phi = (np.sqrt(5) - 1) / 2
    for _ in range(n_evals):
        t1 = hi - phi * (hi - lo)
        t2 = lo + phi * (hi - lo)
        f1 = f(t1)
        f2 = f(t2)
        if f1 < f2:
            hi = t2
        else:
            lo = t1
    return (lo + hi) / 2


def active_set_lp_polish(a, time_budget, verbose=False):
    """
    Two-stage active-set LP polishing:
    1. Solve min max LP to get direction g1
    2. Identify truly active constraints
    3. Solve min-sum-active LP to get direction g2
    4. Golden section blend
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        # Stage 1: min max LP
        g1, M1 = solve_lp_minmax(a)
        if g1 is None:
            break

        # Find active constraints
        A_mat = build_conv_matrix(a / float(np.sum(a)))
        conv_ag = A_mat @ g1
        max_conv = float(np.max(conv_ag))
        tol = 0.005 * max_conv
        active = np.where(conv_ag >= max_conv - tol)[0]

        if len(active) == 0:
            active = np.array([int(np.argmax(conv_ag))])

        # Stage 2: min sum at active indices
        g2 = solve_lp_min_sum_active(a, active)

        # Choose direction
        if g2 is not None:
            # Compare: which direction gives lower max when blended?
            g_blend = 0.5 * g1 + 0.5 * g2
        else:
            g_blend = g1

        # Golden section line search
        def obj_at_t(t):
            a_t = (1 - t) * a + t * g_blend
            a_t = np.maximum(a_t, 0)
            s_t = float(np.sum(a_t))
            if s_t < 1e-10:
                return np.inf
            return compute_obj(a_t / s_t)

        t_best = golden_section_search(obj_at_t, 0.0, 1.0, n_evals=7)
        a_new = (1 - t_best) * a + t_best * g_blend
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        it += 1

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 3 == 0:
                print(f"    ASLP it{it}: obj={best_obj:.8f}, M={M1:.6f}, |S|={len(active)}")
        else:
            stall += 1
            # Perturb to escape
            scale = 0.01 + 0.005 * min(stall, 20)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s_a = float(np.sum(a))
            if s_a > 1e-10:
                a = a / s_a
            else:
                a = best_a.copy()
            if stall > 40:
                break

    return best_a


def spectral_perturb(a, rng, n_modes=3):
    """Perturb sequence with random low-frequency modulation."""
    n = len(a)
    x = np.arange(n) / n
    for _ in range(n_modes):
        freq = int(rng.integers(1, 6))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4)
        a = a * (1.0 + amp * np.sin(2 * np.pi * freq * x + phase))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    else:
        a = a / max(a)
    return a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-10))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 6, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** float(rng.uniform(-3, 3))
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    else:
        # Edge-concentrated
        a = np.zeros(n)
        a[0] = 1.0
        a[-1] = 1.0
        if n > 2:
            a[n // 2] = 0.3
        a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def quick_optimize(a, time_budget):
    """Quick multi-method optimization for a given sequence."""
    n = len(a)
    if time_budget < 5:
        return a

    # MW
    alloc1 = time_budget * 0.4
    a = multiplicative_weights(a, time_budget=alloc1,
                               tau_init=15.0, tau_end=300.0, eta_init=0.7)
    remaining = time_budget - alloc1

    # FISTA
    if remaining > 3:
        alloc2 = remaining * 0.5
        a = fista_robust(a, time_budget=alloc2,
                         tau_init=30.0, tau_end=500.0)
        remaining = time_budget - alloc1 - alloc2

    # LP
    if remaining > 3:
        a = active_set_lp_polish(a, time_budget=remaining * 0.8)

    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization:
    1. Active-set LP polishing (main workhorse)
    2. Multiplicative weights fine-tuning
    3. FISTA with high tau
    4. Multi-n exploration
    5. Spectral diversification + diverse restarts
    6. Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"No parent: {e}")

    if best_seq is None:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    # === Phase 1: Active-set LP polishing ===
    print(f"\n=== Phase 1: Active-Set LP ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.3, 350.0)
    if p1_budget > 30:
        a_lp = active_set_lp_polish(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ASLP: obj={best_score:.8f} **")
        else:
            print(f"  ASLP: obj={obj_lp:.8f}")

    # === Phase 2: Multiplicative Weights ===
    print(f"\n=== Phase 2: Multiplicative Weights ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 200.0)
    if p2_budget > 20:
        a_mw = multiplicative_weights(best_seq, time_budget=p2_budget,
                                      tau_init=20.0, tau_end=500.0,
                                      eta_init=0.6, verbose=True)
        obj_mw = compute_obj(a_mw)
        if obj_mw < best_score:
            best_score = obj_mw
            best_seq = a_mw.copy()
            print(f"  MW: obj={best_score:.8f} **")
        else:
            print(f"  MW: obj={obj_mw:.8f}")

    # === Phase 3: FISTA ===
    print(f"\n=== Phase 3: FISTA ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    if p3_budget > 20:
        a_f = fista_robust(best_seq, time_budget=p3_budget,
                           tau_init=30.0, tau_end=1000.0, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  FISTA: obj={best_score:.8f} **")
        else:
            print(f"  FISTA: obj={obj_f:.8f}")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(200, n_current - 200),
        max(200, n_current - 100),
        n_current + 100,
        n_current + 200,
        min(2000, n_current + 400),
        600, 700, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000 and abs(n - n_current) > 30]

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.1, 40.0)
        if alloc > 10:
            a_try = quick_optimize(a_try, alloc)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Spectral Diversification + Restarts ===
    print(f"\n=== Phase 5: Spectral Diversification ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if r < 2:
            # Spectral perturbation of current best
            a_r = spectral_perturb(best_seq, rng, n_modes=r + 2)
        else:
            # Diverse init blended with best
            mode = (r - 1) % 7
            a_r = diverse_init(n_best, rng, mode=mode)
            blend = 0.3 + 0.1 * r
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.12, 50.0)
        if alloc > 10:
            a_r = quick_optimize(a_r, alloc)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r}: obj={best_score:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 6: Final Intensive Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    if remaining > 30:
        alloc = min(remaining * 0.3, 80.0)
        a_f1 = active_set_lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # MW with very high tau
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.25, 50.0)
        a_f2 = multiplicative_weights(best_seq, time_budget=alloc,
                                      tau_init=100.0, tau_end=2000.0, eta_init=0.2)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final MW: obj={best_score:.8f} **")

    # FISTA
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.25, 40.0)
        a_f3 = fista_robust(best_seq, time_budget=alloc,
                            tau_init=100.0, tau_end=2000.0)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Final FISTA: obj={best_score:.8f} **")

    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 30.0)
        a_f4 = active_set_lp_polish(best_seq, time_budget=alloc)
        obj_f4 = compute_obj(a_f4)
        if obj_f4 < best_score:
            best_score = obj_f4
            best_seq = a_f4.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
