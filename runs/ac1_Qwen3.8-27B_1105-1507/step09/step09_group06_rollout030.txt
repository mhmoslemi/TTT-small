def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)

def build_conv_matrix(a):
    """Build matrix A_mat (2n-1 x n) such that A_mat @ g = convolve(a, g)."""
    n = len(a)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A_mat[j:j + r, j] = a[:r]
    return A_mat

def solve_lp_minpeak_fixed(a_fixed, time_limit=30.0):
    """
    Solve LP: min M s.t. conv(a_fixed, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a_fixed)
    a_norm = a_fixed / np.sum(a_fixed)
    A_mat = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def spectral_flattening(a, alpha=0.3):
    """
    Project a towards a flatter power spectrum while maintaining non-negativity.
    Uses water-filling in the frequency domain.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Compute power spectrum
    A = fft(a_norm, 2*n)
    P = np.abs(A)**2
    
    # Target: flat spectrum (all equal to mean)
    P_mean = np.mean(P)
    
    # Blend towards flat
    P_new = (1 - alpha) * P + alpha * P_mean
    
    # Take square root to get amplitude
    A_new = np.sqrt(P_new) * A / (np.abs(A) + 1e-12)
    
    # Inverse FFT
    a_new = ifft(A_new).real[:n]
    
    # Project to non-negative
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a_norm.copy()
    return a_new / s

def nesterov_smoothed_descent(a, time_budget=300.0, tau_init=10.0, tau_end=500.0, seed=0):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    lr = 4.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init + (tau_end - tau_init) * (1.0 - np.cos(np.pi * progress)) / 2.0
        
        if it > 0 and it % 5000 == 0:
            tau = tau_init * 2.0

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))

        p = exp_vals / float(np.sum(exp_vals))

        N = 1
        while N < 3 * n:

        A = fft(y, N)
        g_full = ifft(P * np.conj(A)).real

        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Alternating LP with adaptive line search, spectral domain perturbation,
high-tau Nesterov smoothed descent, and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A_mat (2n-1 x n) such that A_mat @ g = convolve(a, g)."""
    n = len(a)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A_mat[j:j + r, j] = a[:r]
    return A_mat


def solve_lp(a_fixed):
    """
    Solve LP: min M s.t. conv(a_fixed, g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a_fixed)
    a_norm = a_fixed / np.sum(a_fixed)
    A_mat = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_mat @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP, cp.MOSEK]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def spectral_flatten(a, alpha=0.2):
    """
    Blend a towards a sequence with flatter power spectrum.
    This reduces autocorrelation peaks by spreading energy in frequency domain.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return a.copy()
    a_norm = a / s

    # Compute FFT (zero-padded to avoid circular convolution artifacts)
    N = 1
    while N < 2 * n:
        N <<= 1
    A = fft(a_norm, N)
    P = np.abs(A) ** 2

    # Target: geometric mean level (preserves total energy)
    P_target = np.full(N, np.mean(P))

    # Blend in log domain for better behavior
    log_P = np.log(P + 1e-15)
    log_P_target = np.log(P_target + 1e-15)
    log_P_new = (1 - alpha) * log_P + alpha * log_P_target
    P_new = np.exp(log_P_new)

    # Reconstruct with original phase
    phase = A / (np.abs(A) + 1e-15)
    A_new = np.sqrt(P_new) * phase

    a_new = ifft(A_new).real[:n]
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        return a_norm.copy()
    return a_new / s_new


def nesterov_gd(a, time_budget=100.0, tau_init=50.0, tau_end=1000.0, seed=0):
    """Nesterov accelerated smoothed gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.8
    lr = 3.0 / np.sqrt(n)
    momentum = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-6))
        tau = tau_init + (tau_end - tau_init) * (1.0 - np.cos(np.pi * progress)) / 2.0

        # Nesterov step
        y = a + beta * momentum
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        conv = compute_conv(y)
        max_c = float(np.max(conv))
        weights = np.exp(tau * (conv - max_c))
        weights = weights / float(np.sum(weights))

        # Gradient: 2 * convolve(a, weights_padded)
        N = 1
        while N < 3 * n:
            N <<= 1
        W = fft(weights, N)
        Y = fft(y, N)
        grad_full = ifft(W * np.conj(Y)).real
        grad = 2.0 * grad_full[:n]

        # Update
        momentum = beta * momentum - lr * grad
        a_new = y + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9998)

        if it - best_it > 20000:
            a = best_a.copy()
            momentum = np.zeros(n)
            lr = 3.0 / np.sqrt(n)
            best_it = it

    return best_a


def lp_optimize(a, time_budget=200.0, seed=0):
    """
    Main LP-based optimizer:
    - Solve LP direction
    - Line search for optimal blend
    - Alternate direction when stuck
    - Spectral perturbation to escape
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    fail_count = 0
    last_t = 0.3

    while time.monotonic() - start < time_budget:
        # Solve LP
        g_dir, M_val = solve_lp(a)
        if g_dir is None:
            break

        # Line search: golden section over t in [0.01, 1.0]
        best_t = last_t
        best_t_obj = best_obj
        best_a_try = best_a.copy()

        # Try a set of t values, focusing around last_t
        t_vals = np.unique(np.concatenate([
            np.linspace(0.01, 1.0, 15),
            [max(0.01, last_t * 0.5), min(1.0, last_t * 1.5), last_t]
        ]))

        for t in t_vals:
            a_try = (1.0 - t) * a + t * g_dir
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_try = a_try.copy()

        it += 1

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_try.copy()
            a = best_a.copy()
            last_t = best_t
            fail_count = 0
        else:
            fail_count += 1

            # Try alternate: use g_dir as fixed, solve LP for new direction
            if fail_count % 3 == 0:
                h_dir, _ = solve_lp(g_dir)
                if h_dir is not None:
                    for t in [0.1, 0.2, 0.3, 0.5]:
                        a_try = (1.0 - t) * a + t * h_dir
                        a_try = np.maximum(a_try, 0)
                        s_try = float(np.sum(a_try))
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try.copy()
                            fail_count = 0
                            break

            # Spectral perturbation
            if fail_count % 5 == 0:
                alpha = 0.05 + 0.02 * min(fail_count // 5, 8)
                a_spec = spectral_flatten(best_a, alpha=alpha)
                # Blend with current best
                for mix in [0.3, 0.5, 0.7]:
                    a_try = mix * a_spec + (1 - mix) * best_a
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        fail_count = 0
                        break
                if fail_count > 0:
                    a = 0.5 * a_spec + 0.5 * best_a
                    a = np.maximum(a, 0)
                    a = a / a.sum()

            # Random perturbation
            if fail_count % 10 == 0:
                noise = rng.normal(0, 0.03, n)
                a_pert = best_a * np.exp(noise)
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                a = 0.3 * a_pert + 0.7 * best_a
                a = a / a.sum()

            if fail_count > 50:
                break

    return best_a


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n <= 1:
        return np.full(new_n, a[0] if n == 1 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initial sequences."""
    x = np.arange(n) / max(n - 1, 1)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        a = 1.0 / (np.sqrt(1 - (2 * x - 1) ** 2 + 1e-6))
    elif mode == 3:
        a = np.sin(np.pi * x) ** 2
    elif mode == 4:
        a = 1 - np.abs(2 * x - 1)
    elif mode == 5:
        a = np.exp(-((x - 0.5) ** 2) / 0.05)
    elif mode == 6:
        a = np.exp(-((x - 0.3) ** 2) / 0.02) + np.exp(-((x - 0.7) ** 2) / 0.02)
    elif mode == 7:
        a = 1 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.2 * np.sin(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.01)
    elif mode == 8:
        a = x ** rng.uniform(-1, 3)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
