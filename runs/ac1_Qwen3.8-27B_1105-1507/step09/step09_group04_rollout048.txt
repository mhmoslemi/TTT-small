"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_standard(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_trust_region(a, radius=0.5):
    """Solve LP with trust region: min M s.t. conv(a,g)<=M, sum(g)=1, g>=0, ||g-a||_1 <= radius."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [
        A @ g <= M,
        cp.sum(g) == 1,
        cp.norm1(g - a_norm) <= radius
    ]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def optimal_delta_analytical(conv, d, k, a_k, verbose=False):
    """
    Find delta minimizing max_m (conv[m] + 2*delta*d[m] + delta^2 * [m==2k])
    subject to delta >= -a_k.
    Uses bisection on subgradient for the convex piecewise-linear (+quadratic) function.
    """
    n = len(conv)
    
    def f_at(delta):
        val = conv + 2.0 * delta * d
        val[2 * k] += delta * delta
        return float(np.max(val))
    
    def subgrad_at(delta):
        val = conv + 2.0 * delta * d
        val[2 * k] += delta * delta
        idx = np.argmax(val)
        sg = 2.0 * d[idx]
        if idx == 2 * k:
            sg += 2.0 * delta
        return sg
    
    # Bracket the minimum
    lo = -a_k
    # Upper bound: find where the function is clearly increasing
    hi = 0.0
    f_mid = f_at(0.0)
    # Expand hi
    step = max(1.0 / len(conv), 1e-4)
    hi = step
    for _ in range(50):
        if f_at(hi) > f_mid + 1e-12:
            break
        hi *= 2.0
        if hi > 1.0:
            break
    
    if hi <= lo:
        hi = lo + 1e-6
    
    # Bisection on subgradient
    for _ in range(60):
        mid = (lo + hi) / 2.0
        sg = subgrad_at(mid)
        if sg > 0:
            hi = mid
        else:
            lo = mid
        if hi - lo < 1e-14:
            break
    
    delta_opt = (lo + hi) / 2.0
    delta_opt = max(delta_opt, -a_k)
    
    # Verify and refine with a few evaluations
    best_d = delta_opt
    best_f = f_at(delta_opt)
    for frac in [0.5, 0.7, 1.3, 1.5, 2.0]:
        trial = max(-a_k, delta_opt * frac)
        ft = f_at(trial)
        if ft < best_f:
            best_f = ft
            best_d = trial
    
    return best_d

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 3:
        a = np.sin(np.pi * x)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 4:
        a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 2
        a = np.maximum(a, 0.0001)
        a = a / a.sum()
    elif mode == 5:
        centers = [n * 0.25, n * 0.75]
        widths = [n * 0.15, n * 0.15]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 6:
        x2 = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x2**2 + 0.01)
        a = a / a.sum()
    elif mode == 7:
        freqs = rng.integers(1, 4, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 8:
        a = np.exp(-((x - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = a / a.sum()
    elif mode == 9:
        # Fejer-like kernel
        a = 1.0 - x
        a = a * (1.0 - a)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def sensitivity_guided_cd(a, time_budget=100.0, start_time=0, rng=None, verbose=False):

        impacts = np.zeros(n)

            # d[m] = a[m-k] for m in [k, k+n-1]
            # At active position m, d[m] = a[m-k] if k <= m < k+n

                    max_impact = max(max_impact, a[m - k])
            impacts[k] = max_impact

        # Sort by impact (descending) - most impactful first
        order = np.argsort(-impacts)

        obj_before = float(np.max(conv))

        for k in order:
            d = np.zeros(2 * n - 1)
            d[k:k + n] = a

            a_lp, obj_lp = lp_polish_advanced(best_seq, time_budget=alloc, verbose=True, rng=rng)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                n_best = len(best_seq)
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
                if rnd < 2:
                    scale = 0.02 * (rnd + 1)
                    best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                    best_seq = np.maximum(best_seq, 0)
                    best_seq = best_seq / best_seq.sum()

    # === Phase 3: Sensitivity-guided CD ===
    print(f"\n=== Phase 3: Sensitivity CD ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    if p3_budget > 30:
        a_scd, obj_scd = sensitivity_guided_cd(best_seq, time_budget=p3_budget,
                                               start_time=time.monotonic(), rng=rng, verbose=True)
        if obj_scd < best_score:
            best_score = obj_scd
            best_seq = a_scd.copy()
            n_best = len(best_seq)
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj_scd:.8f}")

    # === Phase 4: Smoothed Gradient Descent ===
    print(f"\n=== Phase 4: Smoothed Gradient ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 180.0)
    if p4_budget > 30:
        a_sgd, obj_sgd = smoothed_nesterov(best_seq, time_budget=p4_budget,
                                           tau_init=20.0, tau_end=800.0, verbose=True)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            n_best = len(best_seq)
            print(f"  Phase4: obj={best_score:.8f} **")
        else:
            print(f"  Phase4: obj={obj_sgd:.8f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Systematic length exploration
    n_options = []
    base = n_best
    for delta_n in [-200, -100, -50, 50, 100, 200, 300, 400]:
        n_try = base + delta_n
        if 300 <= n_try <= 2000 and n_try != base:
            n_options.append(n_try)
    # Add some absolute values
    for n_abs in [400, 500, 600, 700, 900, 1000, 1100, 1200, 1500]:
        if abs(n_abs - base) > 50:
            n_options.append(n_abs)

    n_options = sorted(set(n_options))
    # Limit to 6 best candidates (closest to current + spread)
    n_options.sort(key=lambda x: abs(x - base))
    n_options = n_options[:6]
    # Ensure diversity
    selected = [n_options[0]]
    for n in n_options[1:]:
        if all(abs(n - s) > 100 for s in selected):
            selected.append(n)
        if len(selected) >= 5:
            break
    n_options = selected

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        print(f"  Trying n={n_try}...")
        a_try = resample_sequence(best_seq, n_try)

        # Quick CD
        alloc1 = min(remaining * 0.4, 60.0)
        if alloc1 > 20:
            a_try, obj_try = coord_descent_improved(a_try, time_budget=alloc1,
                                                    start_time=time.monotonic(), rng=rng)

        # Quick LP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.3, 40.0)
        if alloc2 > 15:
            a_try, obj_try2 = lp_polish_advanced(a_try, time_budget=alloc2, rng=rng)
            if obj_try2 < obj_try:
                obj_try = obj_try2

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        mode = (r + 1) % 10
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.2 + 0.12 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc1 = min(remaining * 0.35, 70.0)
        if alloc1 > 20:
            a_r, obj_r = coord_descent_improved(a_r, time_budget=alloc1,
                                                start_time=time.monotonic(), rng=rng)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.2, 35.0)
        if alloc2 > 15:
            a_r, obj_r2 = lp_polish_advanced(a_r, time_budget=alloc2, rng=rng)
            if obj_r2 < obj_r:
                obj_r = obj_r2

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            n_best = len(best_seq)
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 60:
        alloc = min(remaining * 0.35, 120.0)
        a_cd_f, obj_cd_f = coord_descent_improved(best_seq, time_budget=alloc,
                                                  start_time=time.monotonic(), rng=rng, verbose=True)
        if obj_cd_f < best_score:
            best_score = obj_cd_f
            best_seq = a_cd_f.copy()
            n_best = len(best_seq)
            print(f"  Final CD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.35, 100.0)
        a_lp_f, obj_lp_f = lp_polish_advanced(best_seq, time_budget=alloc, verbose=True, rng=rng)
        if obj_lp_f < best_score:
            best_score = obj_lp_f
            best_seq = a_lp_f.copy()
            n_best = len(best_seq)
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 80.0)
        a_scd_f, obj_scd_f = sensitivity_guided_cd(best_seq, time_budget=alloc,
                                                   start_time=time.monotonic(), rng=rng)
        if obj_scd_f < best_score:
            best_score = obj_scd_f
            best_seq = a_scd_f.copy()
            n_best = len(best_seq)
            print(f"  Final SCD: obj={best_score:.8f} **")

    # Final CD fine pass
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 60.0)
        a_cd_f2, obj_cd_f2 = coord_descent_improved(best_seq, time_budget=alloc,
                                                    start_time=time.monotonic(), rng=rng)
        if obj_cd_f2 < best_score:
            best_score = obj_cd_f2
            best_seq = a_cd_f2.copy()
            n_best = len(best_seq)
            print(f"  Final CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
