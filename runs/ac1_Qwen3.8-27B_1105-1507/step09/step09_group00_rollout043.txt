```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses weighted QP iterations with Hankel matrix structure, alternating forward/backward LP,
and smoothed gradient descent. New approach focuses on the algebraic structure of the
convolution constraint via the Hankel matrix representation.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def hankel_matrix(w):
    """Build Hankel matrix W[i,j] = w[i+j] where w has length 2n-1."""
    n = (len(w) + 1) // 2
    W = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            W[i, j] = w[i + j]
    return W

def hankel_matrix_fast(w):
    """Build Hankel matrix W[i,j] = w[i+j] using numpy vectorization."""
    n = (len(w) + 1) // 2
    idx = np.arange(n)
    W = w[idx[:, None] + idx[None, :]]
    return W

def solve_qp_hankel(w, eps=1e-8):
    """
    Solve: min a^T W a s.t. sum(a)=1, where W is Hankel with entries W[i,j]=w[i+j].
    Returns the solution a (projected to simplex if needed).
    """
    n = (len(w) + 1) // 2
    W = hankel_matrix_fast(w)
    W += eps * np.eye(n)  # Regularization for numerical stability
    
    try:
        ones = np.ones(n)
        x = np.linalg.solve(W, ones)
    except np.linalg.LinAlgError:
        # Fallback: use pseudo-inverse
        x = np.linalg.lstsq(W, np.ones(n), rcond=None)[0]
    
    # Project to simplex
    s = float(np.sum(x))
    if s < 1e-10:
        return np.ones(n) / n
    a = x / s
    a = np.maximum(a, 0)
    s2 = float(np.sum(a))
    if s2 < 1e-10:
        return np.ones(n) / n
    return a / s2

def softmax_shifted(c, sigma):
    """Compute softmax(c/sigma) with numerical stability."""
    c = np.asarray(c, dtype=np.float64)
    c_shifted = c / sigma - np.max(c) / sigma
    w = np.exp(c_shifted)
    s = float(np.sum(w))
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= s
    return w

def weighted_qp_iteration(a, max_iters=100, sigma_start=None, sigma_end=1.0, 
                          blend_rate=0.5, eps_reg=1e-8, verbose=False, time_budget=None, start_time=None):
    """
    Iterative weighted QP: alternately compute softmax weights from conv(a,a)
    and solve the QP min a'^T W a' s.t. sum(a')=1.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s
    
    if sigma_start is None:
        sigma_start = float(n)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    if time_budget is None:
        time_budget = 100.0
    if start_time is None:
        start_time = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start_time >= time_budget:
            break
        
        # Anneal sigma
        frac = it / max(max_iters - 1, 1)
        sigma = sigma_start * (sigma_end / sigma_start) ** frac
        
        c = compute_conv(a)
        M = float(np.max(c))
        w = softmax_shifted(c, sigma)
        
        # Solve QP
        a_qp = solve_qp_hankel(w, eps=eps_reg)
        
        # Blend
        a_new = (1 - blend_rate) * a + blend_rate * a_qp
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 10 == 0:
            print(f"    WQP it{it}: obj={best_obj:.8f}, sigma={sigma:.1f}, M={M:.6f}")
    
    return best_a
```

```python
    A = np.zeros((2 * n - 1, n))

        k_min = i

def solve_lp_forward(fixed_a, time_budget=30.0):
    """
    Solve: min M s.t. conv(fixed_a, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M).
    """
    import cvxpy as cp
    n = len(fixed_a)
    a_norm = fixed_a / np.sum(fixed_a)
    A = build_convolve_matrix_fixed(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

        except Exception:
            continue
    return None, None
```

```python
def alternating_lp(a, time_budget=200.0, blend_rate=0.5, verbose=True, seed=42):
    """
    Alternating LP: fix one variable, solve LP for the other.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    iter_count = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        # Forward: fix a, solve for g
        g, M_f = solve_lp_forward(a)
        if g is None:
            break
        
        # Blend a towards g
        a_fwd = (1 - blend_rate) * a + blend_rate * g
        a_fwd = np.maximum(a_fwd, 0)
        a_fwd = a_fwd / np.sum(a_fwd)
        
        # Backward: fix g (the LP solution), solve for a
        a_bwd, M_b = solve_lp_backward(g)
        if a_bwd is not None:
            a_new = (1 - blend_rate) * a_fwd + blend_rate * a_bwd
        else:
            a_new = a_fwd
        a_new = np.maximum(a_new, 0)
        a_new = a_new / np.sum(a_new)
        
        obj_new = compute_obj(a_new)
        a = a_new
        iter_count += 1
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a.copy()
            if verbose and iter_count % 3 == 0:
                print(f"    ALP it{iter_count}: obj={best_obj:.8f}, M_f={M_f:.6f}, M_b={M_b:.6f}")
    
    return best_a
```

```python
def smoothed_gd(a, time_budget=100.0, tau_init=10.0, tau_end=500.0, lr=None, verbose=False):
    """Smoothed gradient descent with cosine annealing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s
    
    if lr is None:
        lr = 2.0 / np.sqrt(n)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)  # momentum
    beta = 0.9
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(time_budget, 1e-10))
        # Cosine anneal tau
        tau = tau_init + (tau_end - tau_init) * 0.5 * (1 - np.cos(np.pi * frac))
        
        # Nesterov step
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        # Compute gradient
        g, conv = smoothed_gradient(y, tau)
        
        # Update
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        lr = max(1e-10, lr * 0.99995)
    
    return best_a
```

```python
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / np.sum(a)
    
    if lr is None:
        lr = 1.0 / np.sqrt(n)

    best_obj = compute_obj(a)

        tau = tau_init + (tau_end - tau_init) * 0.5 * (1 - np.cos(np.pi * frac))
        
        g, conv = smoothed_gradient(a, tau)
        avg_g = float(np.dot(a, g))
        
        # Multiplicative update
        a_new = a * np.exp(-lr * (g - avg_g))
        a_new = np.maximum(a_new, 1e-30)  # Prevent underflow
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s
        
        obj_new = compute_obj(a_new)

            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        lr = max(1e-6, lr * 0.9999)
    
    return best_a
```

```python
def build_convolve_matrix_fast(fixed):
    """Build matrix A (2n-1 x n) such that A @ x = conv(fixed, x) using vectorization."""
    n = len(fixed)
    # A[k, i] = fixed[k-i] if 0 <= k-i < n
    k_idx = np.arange(2*n-1)[:, None]  # (2n-1, 1)
    i_idx = np.arange(n)[None, :]  # (1, n)
    diff = k_idx - i_idx  # (2n-1, n)
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2*n-1, n))
    A[mask] = fixed[diff[mask]]
```

```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations: (1) Weighted QP iterations exploiting Hankel matrix structure,
(2) Alternating forward/backward LP for joint bilinear optimization,
(3) Multiplicative weight gradient descent in log-space,
(4) Adaptive sigma annealing, targeted lag reduction, and multi-scale n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def hankel_matrix_fast(w):
    """Build Hankel matrix W[i,j] = w[i+j]. w has length 2n-1."""
    n = (len(w) + 1) // 2
    idx = np.arange(n)
    return w[idx[:, None] + idx[None, :]]


def solve_qp_hankel(w, eps=1e-8):
    """
    Solve: min a^T W a s.t. sum(a)=1, where W[i,j]=w[i+j].
    Returns solution projected to simplex.
    """
    n = (len(w) + 1) // 2
    W = hankel_matrix_fast(w)
    W = W + eps * np.eye(n)
    try:
        ones = np.ones(n)
        x = np.linalg.solve(W, ones)
    except np.linalg.LinAlgError:
        try:
            x = np.linalg.lstsq(W, np.ones(n), rcond=None)[0]
        except Exception:
            return np.ones(n) / n
    s = float(np.sum(x))
    if abs(s) < 1e-10:
        return np.ones(n) / n
    a = x / s
    a = np.maximum(a, 0.0)
    s2 = float(np.sum(a))
    if s2 < 1e-10:
        return np.ones(n) / n
    return a / s2


def softmax_shifted(c, sigma):
    """Compute numerically stable softmax(c/sigma)."""
    c = np.asarray(c, dtype=np.float64)
    shifted = c / sigma - np.max(c) / sigma
    w = np.exp(shifted)
    s = float(np.sum(w))
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= s
    return w


def weighted_qp_iteration(a, max_iters=80, sigma_start=None, sigma_end=1.0,
                          blend_rate=0.6, eps_reg=1e-8, verbose=False,
                          time_budget=120.0, start_time=None):
    """
    Iterative weighted QP: compute softmax weights from conv(a,a),
    solve QP min a'^T W a' s.t. sum(a')=1, blend with current.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    if sigma_start is None:
        sigma_start = float(n)
    if start_time is None:
        start_time = time.monotonic()

    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(max_iters):
        if time.monotonic() - start_time >= time_budget:
            break

        frac = it / max(max_iters - 1, 1)
        sigma = sigma_start * (sigma_end / sigma_start) ** frac

        c = compute_conv(a)
        w = softmax_shifted(c, sigma)
        a_qp = solve_qp_hankel(w, eps=eps_reg)

        a_new = (1.0 - blend_rate) * a + blend_rate * a_qp
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

        if verbose and (it % 15 == 0 or it == max_iters - 1):
            print(f"    WQP it{it}: obj={best_obj:.8f}, sigma={sigma:.1f}")

    return best_a


def build_convolve_matrix_fast(fixed):
    """Build matrix A (2n-1 x n) such that A @ x = conv(fixed, x)."""
    n = len(fixed)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[mask] = fixed[diff[mask]]
    return A


def solve_lp_minpeak(fixed, verbose=False):
    """
    Solve LP: min M s.t. conv(fixed, x)[k] <= M for all k, sum(x)=1, x>=0.
    Returns (x, M) or (None, None).
    """
    import cvxpy as cp
    n = len(fixed)
    fixed_norm = fixed / float(np.sum(fixed))
    A = build_convolve_matrix_fast(fixed_norm)

    x_var = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ x_var <= M, cp.sum(x_var) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if x_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                x_val = np.maximum(x_var.value, 0)
                s = float(np.sum(x_val))
                if s > 1e-10:
                    return x_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def alternating_lp(a, time_budget=200.0, blend_rate=0.5, verbose=True, seed=42):
    """
    Alternating LP: fix one variable, solve LP for the other, blend.
    Forward: fix a, find g. Backward: fix g, find a.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    iter_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break

        g, M_f = solve_lp_minpeak(a)
        if g is None:
            break

        a_fwd = (1.0 - blend_rate) * a + blend_rate * g
        a_fwd = np.maximum(a_fwd, 0)
        s_fwd = float(np.sum(a_fwd))
        if s_fwd < 1e-10:
            a_fwd = best_a.copy()
        else:
            a_fwd = a_fwd / s_fwd

        a_bwd, M_b = solve_lp_minpeak(g)
        if a_bwd is not None:
            a_new = (1.0 - blend_rate * 0.5) * a_fwd + blend_rate * 0.5 * a_bwd
        else:
            a_new = a_fwd

        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a_fwd
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        a = a_new
        iter_count += 1

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a.copy()
            if verbose and iter_count % 3 == 0:
                print(f"    ALP it{iter_count}: obj={best_obj:.8f}, M_f={M_f:.6f}, M_b={M_b if M_b else 0:.6f}")

    return best_a


def smoothed_gradient(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    corr = ifft(np.conj(fft(a, N)) * fft(p, N)).real[:n]
    grad = 2.0 * corr
    return grad, conv


def smoothed_gd(a, time_budget=100.0, tau_init=10.0, tau_end=500.0,
                lr=None, verbose=False, momentum=0.9):
    """Smoothed gradient descent with Nesterov momentum and cosine annealing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    if lr is None:
        lr = 2.0 / np.sqrt(n)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    no_improve = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * 0.5 * (1.0 - np.cos(np.pi * frac))

        y = a + momentum * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, _ = smoothed_gradient(y, tau)
        v = momentum * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        lr = max(1e-10, lr * 0.9999)

        if no_improve > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            no_improve = 0

        if verbose and no_improve % 2000 == 0 and no_improve > 0:
            print(f"    GD: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6f}")

    return best_a


def multiplicative_gd(a, time_budget=100.0, tau_init=10.0, tau_end=400.0,
                      lr=None, verbose=False):
    """Multiplicative weight (exponential) gradient descent."""
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / np.sum(a)

    if lr is None:
        lr = 1.0 / np.sqrt(n)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * 0.5 * (1.0 - np.cos(np.pi * frac))

        g, _ = smoothed_gradient(a, tau)
        avg_g = float(np.dot(a, g))

        log_update = lr * (avg_g - g)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-30)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr = max(1e-6, lr * 0.9999)

        if verbose and best_obj < 1.51:
            elapsed_now = time.monotonic() - start
            if elapsed_now % 10 < 1:
                print(f"    MGD: obj={best_obj:.8f}, tau={tau:.0f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 6, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        x = np.linspace(0, 1, n)
        a = np.abs(np.sin(np.pi * x * rng.uniform(2, 5)))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 8:
        # Two-pass Gaussian
        c = n // 2
        w = n // 6
        a = np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

            if s > 0.01:

                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        n_current = 800
        best_seq = np.ones(n_current) / n_current
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Weighted QP iterations ===
    print(f"\n=== Phase 1: Weighted QP (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 30:
        a_qp = weighted_qp_iteration(best_seq, max_iters=100, sigma_start=float(n_current),
                                     sigma_end=1.0, blend_rate=0.6, eps_reg=1e-8,
                                     verbose=True, time_budget=p1_budget)
        obj_qp = compute_obj(a_qp)
        if obj_qp < best_score:
            best_score = obj_qp
            best_seq = a_qp.copy()
            print(f"  Phase1 WQP: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 WQP: obj={obj_qp:.8f} (no improvement)")

    # === Phase 2: Alternating LP ===
    print(f"\n=== Phase 2: Alternating LP ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.30, 350.0)
    if p2_budget > 30:
        # Multiple rounds with perturbation between
        for rnd in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(p2_budget / (4 - rnd), remaining * 0.6)
            if alloc < 20:
                break
            a_lp = alternating_lp(best_seq, time_budget=alloc, blend_rate=0.5,
                                  verbose=True, seed=1000 + rnd * 7)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
                # Perturb to escape
                if rnd < 3:
                    noise = rng.normal(0, 0.01, len(best_seq))
                    best_seq = best_seq * np.exp(noise)
                    best_seq = np.maximum(best_seq, 0)
                    best_seq = best_seq / best_seq.sum()

    # === Phase 3: Multiplicative GD (new) ===
    print(f"\n=== Phase 3: Multiplicative GD ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.20, 250.0)
    if p3_budget > 30:
        a_mgd = multiplicative_gd(best_seq, time_budget=p3_budget,
                                  tau_init=15.0, tau_end=600.0, verbose=True)
        obj_mgd = compute_obj(a_mgd)
        if obj_mgd < best_score:
            best_score = obj_mgd
            best_seq = a_mgd.copy()
            print(f"  Phase3 MGD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 MGD: obj={obj_mgd:.8f}")

    # === Phase 4: Targeted lag reduction ===
    print(f"\n=== Phase 4: Targeted Lag Reduction ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.10, 100.0)
    if p4_budget > 20:
        a_tar = targeted_lag_reduction(best_seq, time_budget=p4_budget, verbose=True)
        obj_tar = compute_obj(a_tar)
        if obj_tar < best_score:
            best_score = obj_tar
            best_seq = a_tar.copy()
            print(f"  Phase4 Target: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 Target: obj={obj_tar:.8f}")

    # === Phase 5: Smoothed GD with high tau ===
    print(f"\n=== Phase 5: Smoothed GD (high tau) ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.15, 180.0)
    if p5_budget > 20:
        a_gd = smoothed_gd(best_seq, time_budget=p5_budget,
                           tau_init=50.0, tau_end=1500.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase5 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase5 GD: obj={obj_gd:.8f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(400, n_best - 200), max(400, n_best - 100),
        n_best + 100, n_best + 200, n_best + 400,
        600, 700, 900, 1000, 1200, 1400
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500 and abs(n - n_best) > 50]
    n_options = n_options[:5]
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick WQP
        alloc1 = min(remaining * 0.3, 45.0)
        if alloc1 > 15:
            a_try = weighted_qp_iteration(a_try, max_iters=30,
                                          sigma_start=float(n_try), sigma_end=2.0,
                                          blend_rate=0.5, time_budget=alloc1)

        # Quick GP
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.2, 40.0)
        if alloc2 > 15:
            a_try = smoothed_gd(a_try, time_budget=alloc2, tau_init=20.0, tau_end=500.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Another WQP round with lower sigma
    if remaining > 40:
        alloc = min(remaining * 0.25, 100.0)
        a_qp2 = weighted_qp_iteration(best_seq, max_iters=60,
                                      sigma_start=float(n_current) * 0.3, sigma_end=0.5,
                                      blend_rate=0.4, eps_reg=1e-9,
                                      time_budget=alloc)
        obj_qp2 = compute_obj(a_qp2)
        if obj_qp2 < best_score:
            best_score = obj_qp2
            best_seq = a_qp2.copy()
            print(f"  Final WQP: obj={best_score:.8f} **")

    # Another LP round
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 80.0)
        a_lp2 = alternating_lp(best_seq, time_budget=alloc, blend_rate=0.4,
                               verbose=False, seed=9999)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final MGD
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.3, 60.0)
        a_mgd2 = multiplicative_gd(best_seq, time_budget=alloc,
                                   tau_init=100.0, tau_end=2000.0)
        obj_mgd2 = compute_obj(a_mgd2)
        if obj_mgd2 < best_score:
            best_score = obj_mgd2
            best_seq = a_mgd2.copy()
            print(f"  Final MGD: obj={best_score:.8f} **")

    # Ultimate GD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 50.0)
        a_gd2 = smoothed_gd(best_seq, time_budget=alloc,
                            tau_init=200.0, tau_end=3000.0, lr=0.5 / np.sqrt(n_current))
        obj_gd2 = compute_obj(a_gd2)
        if obj_gd2 < best_score:
            best_score = obj_gd2
            best_seq = a_gd2.copy()
            print(f"  Ultimate GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
