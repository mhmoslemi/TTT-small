import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def multiplicative_update(a, tau, eta):
    """One step of multiplicative weight update."""
    n = len(a)
    conv = compute_conv(a)
    # Softmax weights
    max_conv = float(np.max(conv))
    p = np.exp(tau * (conv - max_conv))
    p /= p.sum()
    # Compute pressure = cross_correlation(p, a)
    # pressure[i] = sum_k p[k] * a[k-i] (with validity)
    # This is the convolution of p (reversed) with a, or equivalently
    # the correlation of p and a
    # Using FFT: pressure = ifft(fft(p_padded) * conj(fft(a_padded)))
    N = 1
    while N < 2 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    pressure = np.real(ifft(P * np.conj(A)))[:n]
    # Update
    a_new = a * np.exp(-eta * pressure)
    # Floor
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    return a_new

def multiplicative_descent(a, time_budget, tau_init=10.0, tau_end=200.0, eta_init=0.5, verbose=False):
    """Iteratively apply multiplicative updates with cooling."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** progress
        # Cool eta
        eta = eta_init * (1.0 - 0.5 * progress)
        
        # Try a few eta values (line search)
        best_eta_obj = best_obj
        best_eta_a = best_a
        for eta_try in [eta * 0.5, eta, eta * 1.5, eta * 2.0]:
            a_try = multiplicative_update(best_a, tau, eta_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_eta_obj:
                best_eta_obj = obj_try
                best_eta_a = a_try
        
        if best_eta_obj < best_obj - 1e-14:
            best_obj = best_eta_obj
            best_a = best_eta_a
        else:
            # Stuck, try smaller eta
            ...
        
        it += 1
        if verbose and it % 500 == 0:
            print(f"    MU it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")
    
    return best_a

                if not np.any(active_mask):
                    # All slopes must be <= 0 for decreasing, all >= 0 for increasing
                    # If all base_j + x*slope_j < base_flat, then f(x) = base_flat (constant)
                    # The minimum is at the boundary where some becomes active
                    # This happens at x = (base_flat - base_j) / slope_j for slope_j > 0
                    # Or x = (base_flat - base_j) / slope_j for slope_j < 0
                    # Since we want the smallest x where something becomes active:
                    # For slopes > 0: x = (base_flat - base_j) / slope_j (increasing, becomes active as x grows)
                    # For slopes < 0: x = (base_flat - base_j) / slope_j (decreasing, becomes active as x shrinks)
                    # At x_mid, none are active, so f(x) = base_flat for all x in some range
                    # The minimum is at x = 0 if base_flat is decreasing... this is getting complex
                    break
                active_slopes = slopes[active_mask]
                min_s = float(np.min(active_slopes))
                max_s = float(np.max(active_slopes))
                if min_s <= 0 <= max_s:
                    break  # Optimal
                elif min_s > 0:
                    # f is increasing, go left
                    x_high = x_mid
                else:
                    # f is decreasing, go right
                    x_low = x_mid
            
            x_opt = 0.5 * (x_low + x_high)
            
            # Update a_i
            a_new_i = x_opt
            a[i] = a_new_i
            a = a / a.sum()  # Renormalize
            
            # Check for improvement
            if epoch % 5 == 0:
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
        
    return best_a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Fast multiplicative weight descent (FFT-based) as main workhorse,
coordinate descent for fine polishing, adaptive multi-n exploration,
and periodic LP corrections. Key innovation: O(n log n) per-step multiplicative
updates that avoid expensive LP solves while still making consistent progress.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def multiplicative_step(a, tau, eta):
    """One multiplicative weight update step using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    p = np.exp(tau * (conv - max_conv))
    p_sum = float(p.sum())
    if p_sum < 1e-30:
        return a.copy()
    p /= p_sum
    # Compute pressure: pressure[i] = 2 * sum_k p[k] * a[k-i] (valid indices)
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    pressure = np.real(ifft(P * np.conj(A)))[:n]
    pressure *= 2.0
    # Center the pressure for more stable updates
    p_mean = float(np.mean(pressure))
    a_new = a * np.exp(-eta * (pressure - p_mean))
    a_new = np.maximum(a_new, 1e-18)
    a_new = a_new / a_new.sum()
    return a_new, conv


def multiplicative_descent(a, time_budget, tau_init=5.0, tau_end=300.0,
                           eta_init=0.8, verbose=False):
    """Iterative multiplicative weight descent with adaptive parameters."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(123)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.8)
        eta = eta_init * (1.0 - 0.6 * progress)

        improved_this_iter = False
        # Try multiple eta scalings
        for eta_scale in [1.0, 0.5, 2.0, 0.25, 3.0]:
            eta_try = eta * eta_scale
            if eta_try < 1e-6:
                continue
            a_try, _ = multiplicative_step(best_a, tau, eta_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                improved_this_iter = True
                break

        # Also try direct gradient step (negative pressure direction)
        if not improved_this_iter:
            conv = compute_conv(best_a)
            max_c = float(np.max(conv))
            # Subgradient of max
            active = np.where(conv >= max_c - 1e-10)[0]
            if len(active) > 0 and len(active) < len(conv):
                g = np.zeros(n)
                for k in active[:min(20, len(active))]:
                    k = int(k)
                    i_start = max(0, k - (n - 1))
                    i_end = min(n - 1, k)
                    if i_start <= i_end:
                        j_start = k - i_end
                        j_end = k - i_start
                        g[i_start:i_end + 1] += 2.0 * best_a[j_start:j_end + 1][::-1]
                g /= len(active[:min(20, len(active))])
                # Projected gradient step
                for lr in [0.02, 0.01, 0.005, 0.05]:
                    a_try = best_a - lr * g
                    a_try = np.maximum(a_try, 0)
                    s_try = float(a_try.sum())
                    if s_try < 1e-10:
                        continue
                    a_try /= s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved_this_iter = True
                        break

        if improved_this_iter:
            stall = 0
        else:
            stall += 1
            if stall > 50:
                # Perturb to escape local minimum
                scale = 0.02 + 0.01 * min(stall - 50, 20)
                best_a = best_a * np.exp(rng.normal(0, scale, n))
                best_a = np.maximum(best_a, 0)
                best_a = best_a / best_a.sum()
                stall = 0
            if stall > 200:
                break

        it += 1
        if verbose and it % 1000 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, eta={eta:.3f}, t={elapsed:.0f}s")

    return best_a


def coordinate_descent(a, time_budget, verbose=False):
    """Coordinate descent: optimize each coefficient in turn."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    epoch = 0
    rng = np.random.default_rng(77)

    while time.monotonic() - start < time_budget:
        epoch += 1
        # Random permutation for shuffling
        order = rng.permutation(n) if epoch > 1 else np.arange(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break
            a_i = a[i]
            if a_i < 1e-15:
                continue

            # Compute current convolution
            conv = compute_conv(a)

            # The change in conv when a_i changes to x:
            # For k where 0 <= k-i < n: conv_k changes by 2*(x - a_i)*a_{k-i}
            # For k = 2*i: additional (x-a_i)^2 term (the a_i*a_i term)
            # Actually: the a_i*a_{k-i} term and a_{k-i}*a_i term
            # When k = 2i: the term is a_i^2, changes to x^2, delta = x^2 - a_i^2
            # When k != 2i: delta = 2*(x-a_i)*a_{k-i}

            # Find the optimal x >= 0 minimizing max_k conv_k(new)
            # conv_k(new) = conv_k + delta_k(x)
            # where delta_k(x) = 2*(x-a_i)*a_{k-i} for k in [i, i+n-1], k != 2i
            #                  (x-a_i)^2 + 2*(x-a_i)*a_{k-i} - 2*(x-a_i)*a_{k-i} ... 
            # Let me just compute: conv_k(new) = conv_k - 2*a_i*a_{k-i} + 2*x*a_{k-i} (for k!=2i)
            #                                   = conv_k - a_i^2 + x^2 (for k=2i, if applicable)

            k_min = i
            k_max = min(2 * n - 2, i + n - 1)
            if k_min > k_max:
                continue

            indices = np.arange(k_min, k_max + 1) - i
            # base_k = conv_k - contribution of current a_i
            base = conv[k_min:k_max + 1].copy()
            base -= 2.0 * a_i * a[indices]
            # Handle k = 2i case
            if 2 * i >= k_min and 2 * i <= k_max:
                pos = 2 * i - k_min
                base[pos] = conv[2 * i] - a_i * a_i
            # For k outside [k_min, k_max], conv is unchanged
            base_flat = float(np.max(np.concatenate([conv[:k_min], conv[k_max + 1:]]))) if k_min > 0 or k_max < 2*n-2 else -1e15

            # Slopes for linear terms
            slopes = 2.0 * a[indices].copy()
            # For k = 2i, the function is quadratic: base[2i-k_min] + x^2
            quad_pos = None
            if 2 * i >= k_min and 2 * i <= k_max:
                quad_pos = 2 * i - k_min

            # Minimize max(max_j (base_j + x*slope_j [+ x^2 if quad]), base_flat) for x >= 0
            # Use bisection on the subgradient
            x_low = 0.0
            # Upper bound: where x^2 dominates
            x_high = max(a_i * 2.0, np.sqrt(base_flat + 1) if base_flat > 0 else 1.0)
            x_high = min(x_high, 2.0)  # Cap it

            for _ in range(40):
                x_mid = 0.5 * (x_low + x_high)
                vals = base + x_mid * slopes
                if quad_pos is not None:
                    vals[quad_pos] = base[quad_pos] + x_mid * x_mid
                f_val = max(float(np.max(vals)), base_flat)
                # Subgradient: min and max slope among active constraints
                active_mask = vals >= base_flat - 1e-12
                active_mask = vals >= f_val - 1e-10
                if not np.any(active_mask):
                    active_mask = vals >= f_val - 1e-8
                if np.any(active_mask):
                    active_slopes = slopes[active_mask].copy()
                    if quad_pos is not None and active_mask[quad_pos]:
                        active_slopes = np.append(active_slopes, 2.0 * x_mid)
                    min_s = float(np.min(active_slopes))
                    max_s = float(np.max(active_slopes))
                    if min_s <= 1e-10 and max_s >= -1e-10:
                        break
                    elif min_s > 1e-10:
                        x_high = x_mid
                    else:
                        x_low = x_mid
                else:
                    # f = base_flat (constant region), any x works
                    break

            x_opt = max(0.0, 0.5 * (x_low + x_high))
            a[i] = x_opt

            # Renormalize periodically
            if i % 100 == 0:
                s = float(a.sum())
                if s < 1e-10:
                    a = a / a.sum() if a.sum() > 0 else np.ones(n) / n
                else:
                    a = a / s

        # End of epoch: check improvement
        s = float(a.sum())
        if s < 1e-10:
            a = best_a.copy()
            continue
        a = a / s
        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        if verbose and epoch % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD epoch{epoch}: obj={best_obj:.8f}, t={elapsed:.0f}s")

        # Check time
        if time.monotonic() - start > time_budget - 5:
            break

    return best_a


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)

    # Build constraint matrix A where (A@g)[k] = (conv(a_norm, g))[k]
    # A is (2n-1) x n, A[k, j] = a_norm[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        for k in range(k_start, k_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a_norm[k - j])

    A_matrix = np.zeros((2 * n - 1, n))
    for k, j in zip(rows, cols):
        A_matrix[k, j] = vals[rows.index(k) if k in rows else 0]  # This is slow

    # Better: build directly
    A_matrix = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A_matrix[k_start:k_end + 1, j] = a_norm[0:k_end - k_start + 1]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_matrix @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget, verbose=False):
    """LP-based polishing with adaptive blending."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    last_t = 0.5
    rng = np.random.default_rng(99)

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 10:
            break
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0,
                      max(0.05, last_t * 0.5), min(1.0, last_t * 1.5)]
        best_t_obj = best_obj
        best_t_val = last_t
        best_local = best_a.copy()

        for t in sorted(set([round(c, 2) for c in candidates])):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(a_try.sum())
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_local = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_local.copy()
            a = best_a
            last_t = best_t_val
            if verbose and it % 3 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t_val:.2f}")
        else:
            # Perturb
            scale = 0.02
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            if it % 20 == 0 and verbose:
                print(f"    LP it{it}: stuck, perturbing. best={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-8))
        a = a / a.sum()
    elif mode == 3:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Tent-like: high in middle, low at edges
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2 * x - 1) ** rng.uniform(1, 3)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses multiplicative descent as primary optimizer,
    coordinate descent for polishing, LP for corrections, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent sequence
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Quick multiplicative descent from multiple inits (200s) ===
    print(f"\n=== Phase 1: Quick MD from diverse inits ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.15, 200.0)
    if p1_budget > 30:
        # First, polish the parent a bit
        alloc = p1_budget * 0.4
        a_md = multiplicative_descent(best_seq, time_budget=alloc,
                                      tau_init=5.0, tau_end=150.0, eta_init=0.8, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  MD on parent: obj={best_score:.8f} **")
        else:
            print(f"  MD on parent: obj={obj_md:.8f}")

        # Try a few diverse inits
        remaining = deadline - time.monotonic()
        if remaining > 60:
            per_init = (remaining - 10) / 4
            for mode in [2, 3, 4]:
                if time.monotonic() > deadline - 30:
                    break
                a_init = diverse_init(n_target, rng, mode)
                a_try = multiplicative_descent(a_init, time_budget=per_init,
                                               tau_init=5.0, tau_end=100.0, eta_init=0.6)
                obj_try = compute_obj(a_try)
                if obj_try < best_score:
                    best_score = obj_try

                else:
                    print(f"  Init mode{mode}: obj={obj_try:.8f}")

    # === Phase 2: Extended multiplicative descent (300s) ===
    print(f"\n=== Phase 2: Extended MD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 300.0)
    if p2_budget > 50:
        # Multiple rounds with perturbation
        num_rounds = 3
        per_round = p2_budget / num_rounds
        for rnd in range(num_rounds):
            if time.monotonic() > deadline - 60:

                                          tau_init=3.0 + 5*rnd, tau_end=200.0 + 100*rnd,
