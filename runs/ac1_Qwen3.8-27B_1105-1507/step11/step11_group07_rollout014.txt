"""
Multi-phase optimizer for step function coefficient minimization.
Uses Adam-based log-sum-exp gradient descent, peak-shaving SOCP,
Frank-Wolfe subgradient method, and LP polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective with sum(a)=1 normalization."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def softmax_from_x(x):
    """Convert unconstrained x to simplex via softmax."""
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    return exp_x / np.sum(exp_x)

def adam_logsumexp_descent(a, time_budget=200.0, beta_start=2.0, beta_end=150.0,
                           lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                           verbose=True, seed=42):

    Adam-based log-sum-exp gradient descent on the simplex.
    Uses softmax parameterization to maintain feasibility.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Initialize in log-space
    a_clipped = np.clip(a, 1e-10, None)
    x = np.log(a_clipped)  # log-space parameter

    v = np.zeros(n)  # second moment
    t = 0
    stall = 0
    best_t = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * (progress ** 0.7)
        lr_t = lr / (1.0 + 0.5 * progress)

        a_current = project_simplex(a_current)  # ensure exact feasibility

        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = project_simplex(a_new)
        
        # Evaluate

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_t = t
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Reduce learning rate and reset momentum
                lr_t *= 0.5
                m = np.zeros(n)
                v = np.zeros(n)
                stall = 0
                # Small perturbation

                x = np.log(np.clip(best_a + pert, 1e-10, None))

        t_iter = t

def frank_wolfe_step(a, time_budget=50.0, max_iters=10000, verbose=False):
    """
    Frank-Wolfe (conditional gradient) method using exact subgradients.
    At each step, the linear minimization over the simplex picks the
    coordinate with the most negative gradient.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_val = np.max(conv)
        # Active set: positions where conv is within 1% of max
        active = conv >= max_val * 0.99
        if not np.any(active):
            active = conv >= max_val - 1e-10
        
        # Subgradient: average of gradients at active positions
        # grad_i = (1/|S|) * sum_{k in S} 2*a[k-i] (for valid k-i)
        # Using cross-correlation with the indicator of active set
        p = np.zeros(len(conv))
        p[active] = 1.0 / np.sum(active)
        
        a_flipped = a[::-1]
        size = 1
        while size < len(p) + n - 1:
            size <<= 1
        P = rfft(p, size)
        A = rfft(a_flipped, size)
        cross = irfft(P * A, size)
        grad = 2.0 * cross[n-1:n-1+n]
        
        # Frank-Wolfe: minimize grad^T v over simplex => v = e_argmin(grad)
        v_idx = np.argmin(grad)
        v = np.zeros(n)
        v[v_idx] = 1.0
        
        # Step size: gamma_t = 2/(t+2)
        gamma = 2.0 / (it + 2)
        a_new = (1 - gamma) * a + gamma * v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 500 == 0:
            print(f"    FW it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def peak_shaving(a, time_budget=60.0, max_iters=20, verbose=False):
    """
    Peak shaving: iteratively solve SOCP to reduce the top peaks.
    Formulation: minimize t s.t. conv(a+delta, a+delta)_k <= t + peak_k for top peaks,
    sum(delta)=0, a+delta>=0, ||delta||_2 <= eps
    Linearized: conv(a+delta)_k ≈ conv(a)_k + 2*conv(a, delta)_k
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break
        
        conv = compute_conv(a)
        max_val = np.max(conv)
        
        # Identify top peaks
        n_peaks = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]
        
        # Build linearized constraint matrix for these peaks
        # conv(a, delta)_k = sum_i A[k,i] * delta[i] where A is conv matrix of a
        A_mat = build_conv_matrix(a)
        A_peaks = A_mat[peak_indices, :]  # (n_peaks x n)
        
        # SOCP: min t
        # s.t. conv[k] + 2*(A_peaks @ delta)[i] <= t + conv[peak_indices[i]] for i in range(n_peaks)
        #      i.e., 2*(A_peaks @ delta)[i] <= t for i in range(n_peaks)
        #      sum(delta) = 0
        #      a + delta >= 0  =>  delta >= -a
        #      ||delta||_2 <= eps
        eps_delta = 0.05  # step size in L2
        
        delta = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            2 * (A_peaks @ delta) <= t_var,
            cp.sum(delta) == 0,
            delta >= -a + 1e-10,
            cp.norm(delta, 2) <= eps_delta
        ]
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        
        solved = False
        for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    delta_val = delta.value
                    a_new = project_simplex(np.maximum(a + delta_val, 0))
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj - 1e-10:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new
                    solved = True
                    break
            except Exception:
                continue
        
        if not solved:
            break
        
        if verbose and it % 5 == 0:
            print(f"    PS it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
    
    return best_a, best_obj


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    return project_simplex(a)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = (np.exp(-((idx - n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) +
             np.exp(-((idx - 3 * n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) + 0.01)
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        if n > 7:
            kernel = np.ones(7) / 7
            a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 6:
        a = 1.0 / np.cosh(3.0 * (2 * x - 1))
    elif mode == 7:
        a = np.zeros(n)
        a[0] = 1.0
        a[-1] = 1.0
        if n > 4:
            a[1:-1] += 0.05
    elif mode == 8:
        # Gaussian bump in middle
        a = np.exp(-((x - 0.5) ** 2) / (2 * 0.15 ** 2)) + 0.01
    elif mode == 9:
        # Two bumps
        a = (np.exp(-((x - 0.2) ** 2) / (2 * 0.1 ** 2)) +
             np.exp(-((x - 0.8) ** 2) / (2 * 0.1 ** 2)) + 0.01)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def lp_polish(a, time_budget=120.0, max_iters=60, verbose=False):
    """LP-based polishing with golden section line search."""
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break
        
        # Line search: try multiple t values
        best_t = 0
        best_obj_t = best_obj
        for t_try in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1 - t_try) * a + t_try * g_lp
            a_try = project_simplex(np.maximum(a_try, 0))
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t_try
        
        # Refine around best_t with golden section
        if best_t > 0:
            lo = max(0, best_t - 0.1)
            hi = min(1.5, best_t + 0.1)
            for _ in range(10):
                mid1 = lo + 0.382 * (hi - lo)
                mid2 = lo + 0.618 * (hi - lo)
                a1 = project_simplex(np.maximum((1 - mid1) * a + mid1 * g_lp, 0))
                a2 = project_simplex(np.maximum((1 - mid2) * a + mid2 * g_lp, 0))
                o1 = compute_obj(a1)
                o2 = compute_obj(a2)
                if o1 < o2:
                    hi = mid2
                else:
                    lo = mid1
                if o1 < best_obj_t:
                    best_obj_t = o1
                    best_t = mid1
                if o2 < best_obj_t:
                    best_obj_t = o2
                    best_t = mid2
        
        if best_t > 0 and best_obj_t < best_obj - 1e-12:
            a_new = project_simplex(np.maximum((1 - best_t) * a + best_t * g_lp, 0))
            best_obj = best_obj_t
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
        
        it += 1
        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.4f}, "
                  f"elapsed={time.monotonic()-start:.1f}s")
        
        if stall >= 5:
            if stall <= 8:
                noise = rng.normal(0, 0.003 * (stall - 4), n)
                a = project_simplex(best_a + noise)
            else:
                x = np.linspace(0, 1, n)
                a_div = 1.0 / (0.05 + 4.0 * (2 * x - 1) ** 2)
                a = project_simplex(0.5 * a_div + 0.5 * best_a)
            stall = 0
    
    return best_a, best_obj
