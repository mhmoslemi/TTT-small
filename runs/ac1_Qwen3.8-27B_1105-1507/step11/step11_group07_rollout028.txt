```python
import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # a[j] contributes to positions j, j+1, ..., min(j+n-1, m-1)
        # Wait, conv(a, g)[k] = sum_i a[i] * g[k-i]
        # So for fixed j (index in g), g[j] contributes a[i] to position k = i + j
        # for i = 0, ..., n-1. So g[j] has coefficients a[0], a[1], ..., a[n-1] at positions j, j+1, ..., j+n-1.
        lo = j
        hi = min(j + n, m)
        A[lo:hi, j] = a[:hi - lo]
    return A

def solve_lp_cycling(a):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def lp_cycling(a, time_budget, max_iters=5000, verbose=True, seed=42, perturbation=0.0, annealing_T=0.0):
    """LP cycling with optional perturbation and simulated annealing."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.1:
            break
        
        g, M = solve_lp_cycling(a)
        if g is None:
            break
        
        it += 1
        obj_g = compute_obj(g)
        
        if annealing_T > 0 and obj_g > best_obj:
            # Simulated annealing: accept with probability exp(-(obj_g - best_obj) / T)
            T = annealing_T * max(0, 1.0 - it / max_iters)
            if T > 0 and rng.random() < np.exp(-(obj_g - best_obj) / T):
                a = g.copy()
            else:
                stall += 1
                if stall > 20:
                    break
                continue
        else:
            a = g.copy()
        
        # Perturbation
        if perturbation > 0:
            noise = rng.normal(0, perturbation, n)
            a = project_simplex(a + noise)
        
        obj_a = compute_obj(a)
        if obj_a < best_obj:
            best_obj = obj_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if verbose and it % 100 == 0:
            print(f"    LP-cycle it{it}: obj={best_obj:.8f}, T={annealing_T * max(0, 1.0 - it / max_iters):.4f}, t={time.monotonic()-start:.1f}s")
    
    return best_a, best_obj
```

The theoretical minimum of (a*a)[n-1] = sum_i a[i] * a[n-1-i] subject to a >= 0, sum(a) = 1 is... 

If we set a[i] = a[n-1-i] (palindromic), then (a*a)[n-1] = sum_i a[i]^2. By Cauchy-Schwarz, sum a[i]^2 >= 1/n, with equality when a is uniform.

But we want to minimize the MAX of all convolution coefficients, not just the central one. The peak might not be at the center.

For a U-shaped sequence, the peak is likely at k=0 or k=2n-2 (the edges), where (a*a)[0] = a[0]^2. If a[0] is large, this could be the peak.

So the trade-off is:
- Making a[0] small reduces the edge peak but increases the central peak
- Making a[0] large increases the edge peak but the central peak might still be the max

The optimal balance is somewhere in between, which is what the optimization is finding.

```python
"""
LP-cycling with simulated annealing and diverse restarts for step function optimization.
Uses alternating linear programming to minimize peak autocorrelation of non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        hi = min(j + n, m)
        A[j:hi, j] = a[:hi - j]
    return A


def solve_lp_cycling(a, solver_cache=None):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.CBC]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_cycling_annealed(a, time_budget, max_iters=10000, verbose=True, seed=42):
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    stall = 0
    
    # Annealing schedule
    T_init = 0.1
    T_end = 0.001
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.05:
            break
        
        progress = min(1.0, it / max_iters)
        T = T_init + (T_end - T_init) * progress
        
        g, M = solve_lp_cycling(a)
        if g is None:
            break
        
        it += 1
        
        # With probability related to T, accept g even if worse
        obj_g = compute_obj(g)
        if obj_g <= best_obj or (T > 0 and rng.random() < np.exp(-(obj_g - best_obj) / T)):
            a = g.copy()
            obj_a = obj_g
        else:
            # Reject: add small perturbation to current best
            noise_scale = T * 0.1
            noise = rng.normal(0, noise_scale, n)
            a = project_simplex(best_a + noise)
            obj_a = compute_obj(a)
        
        if obj_a < best_obj - 1e-10:
            best_obj = obj_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Occasionally do a more aggressive restart
        if stall > 100 and it > 100:
            # Perturb more aggressively
            noise = rng.normal(0, 0.01, n)
            a = project_simplex(best_a + noise)
            stall = 0
            if it % 500 == 0:
                print(f"    LP it{it}: restart, obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
        
        if verbose and it % 200 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, T={T:.5f}, t={time.monotonic()-start:.1f}s")
        
        # Early stop if converged
        if it > 5000 and best_it and it - best_it > 2000:
            break
    
    return best_a, best_obj


def soft_max_gd(a, time_budget, beta_start=10.0, beta_end=50.0, lr_start=0.01, momentum=0.9, seed=42, verbose=False):
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    rng = np.random.default_rng(seed)
    it = 0
    stall = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress
        lr = lr_start * (1.0 - 0.3 * progress)
        
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient: grad_i = 2 * sum_j a_j * p_{i+j}
        # Use FFT cross-correlation
        a_rev = a[::-1]
        size = 1
        while size < n + len(p) - 1:
            size <<= 1
        A_f = rfft(a_rev, size)
        P_f = rfft(p, size)
        full = irfft(A_f * P_f, size)
        q = full[n - 1:n - 1 + n]
        grad = 2.0 * q
        
        v = momentum * v + grad
        a_new = project_simplex(a - lr * v)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall > 200:
                v = np.zeros(n)
                a = project_simplex(best_a + rng.normal(0, 0.002, n))
                stall = 0
        
        a = a_new
        it += 1
        
        if verbose and it % 500 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_a, best_obj

def interp_length(a, new_n):
    old_n = len(a)
```

```python
"""
LP-cycling with simulated annealing, soft-max gradient descent, and diverse restarts
for step function optimization. Uses alternating linear programming to minimize peak
autocorrelation of non-negative sequences, combined with FFT-accelerated gradient methods.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Cross-correlation a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Assumes sum(a)=1 for speed."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        hi = min(j + n, m)
        A[j:hi, j] = a[:hi - j]
    return A


def solve_lp_cycling(a):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_cycling_annealed(a, time_budget, max_iters=20000, verbose=True, seed=42,
                        T_init=0.05, T_end=0.0005):
    """LP cycling with simulated annealing for escaping local minima."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    stall = 0
    best_it = 0

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.05:
            break

        progress = min(1.0, it / max_iters)
        T = T_init * (T_end / T_init) ** progress

        g, M = solve_lp_cycling(a)
        if g is None:
            break

        it += 1
        obj_g = compute_obj(g)

        # Acceptance criterion: always accept if better, otherwise annealing
        if obj_g <= best_obj or (T > 1e-6 and rng.random() < np.exp(-(obj_g - best_obj) / T)):
            a = g.copy()
            obj_a = obj_g
        else:
            # Reject: perturb current best
            noise_scale = max(T * 0.2, 0.001)
            noise = rng.normal(0, noise_scale, n)
            a = project_simplex(best_a + noise)
            obj_a = compute_obj(a)

        if obj_a < best_obj - 1e-10:
            best_obj = obj_a
            best_a = a.copy()
            best_it = it
            stall = 0
        else:
            stall += 1

        # Aggressive restarts when stalled
        if stall > 150 and it > 200:
            if stall <= 300:
                noise = rng.normal(0, 0.005 * (stall // 50 + 1), n)
            else:
                # Mix with a fresh random init
                fresh = rng.dirichlet(np.ones(n) * 0.1)
                a = project_simplex(0.3 * a + 0.7 * fresh)
            stall = 0

        if verbose and it % 200 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, T={T:.6f}, stall={stall}, "
                  f"t={time.monotonic()-start:.1f}s")

        # Early stop if fully converged
        if it > 3000 and it - best_it > 3000:
            break

    return best_a, best_obj


def soft_max_gd(a, time_budget, beta_start=10.0, beta_end=50.0, lr_start=0.01,
                momentum=0.9, seed=42, verbose=False):
    """Soft-max gradient descent with Nesterov momentum on the simplex."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    rng = np.random.default_rng(seed)
    it = 0
    stall = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress
        lr = lr_start * (1.0 - 0.3 * progress)

        conv = compute_conv(a)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p = exp_vals / float(np.sum(exp_vals))

        # Gradient via FFT cross-correlation
        a_rev = a[::-1]
        size = 1
        while size < n + len(p) - 1:
            size <<= 1
        A_f = rfft(a_rev, size)
        P_f = rfft(p, size)
        full = irfft(A_f * P_f, size)
        q = full[n - 1:n - 1 + n]
        grad = 2.0 * q

        v = momentum * v + grad
        a_new = project_simplex(a - lr * v)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall > 300:
                v = np.zeros(n)
                a = project_simplex(best_a + rng.normal(0, 0.003, n))
                stall = 0

        a = a_new
        it += 1

        if verbose and it % 500 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, t={elapsed:.1f}s")

        if it > 2000 and it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)

    return best_a, best_obj


def multiplicative_weights(a, time_budget, eta_start=0.5, eta_end=0.01,
                           seed=42, verbose=False):
    """
    Multiplicative weights / exponentiated gradient method.
    Update: g[i] = a[i] * exp(-eta * grad_i) / sum(...)
    This naturally stays in the simplex and can escape L2-projected minima.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    stall = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        eta = eta_start * (eta_end / eta_start) ** progress

        conv = compute_conv(a)
        c_max = float(np.max(conv))
        # Soft target
        beta = 20.0 + 40.0 * progress
        exp_vals = np.exp(beta * (conv - c_max))
        p = exp_vals / float(np.sum(exp_vals))

        # Gradient
        a_rev = a[::-1]
        size = 1
        while size < n + len(p) - 1:
            size <<= 1
        A_f = rfft(a_rev, size)
        P_f = rfft(p, size)
        full = irfft(A_f * P_f, size)
        q = full[n - 1:n - 1 + n]
        grad = 2.0 * q

        # Multiplicative update with damping
        damping = 0.5 + 0.5 * progress
        update = a * np.exp(-eta * damping * grad)
        update = project_simplex(update)

        obj_new = compute_obj(update)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = update.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall > 200:
                a = project_simplex(best_a + rng.normal(0, 0.005, n))
                stall = 0

        a = update
        it += 1

        if verbose and it % 500 == 0:
            print(f"    MW it{it}: obj={best_obj:.8f}, eta={eta:.4f}, t={elapsed:.1f}s")

    return best_a, best_obj


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    return project_simplex(a)


def arcsine_init(n):
    """Arcsine distribution initialization."""
    x = np.linspace(0.01, 0.99, n)
    a = 1.0 / np.sqrt(1 - (2 * x - 1) ** 2)
    return project_simplex(a)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = (np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) +
             np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01)
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / np.cosh(4.0 * (2 * x - 1))
    elif mode == 6:
        a = np.zeros(n)
        a[0] = 1.0
        a[-1] = 1.0
        a[1:min(5, n)] = 0.4
        a[max(0, n - 5):n - 1] = 0.4
        if n > 6:
            a[1:-1] += 0.03
    elif mode == 7:
        # Comb pattern
        a = np.zeros(n)
        a[::2] = 1.0
        a = project_simplex(a)
    elif mode == 8:
        # Power-U
        a = np.abs(x - 0.5) ** 0.3 + 0.01
    elif mode == 9:
        # Double peak
        a = (np.exp(-((idx - n // 5) ** 2) / (2 * (n // 10) ** 2)) +
             np.exp(-((idx - 4 * n // 5) ** 2) / (2 * (n // 10) ** 2)) + 0.01)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interp_length(a, new_n):
    """Interpolate sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_simplex(np.maximum(a_new, 0))


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search: GD coarse -> LP cycling (annealed) -> MW updates ->
    diverse restarts -> length exploration -> final polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = u_shaped_init(n_current, sharpness=3.0)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")

    # Try multiple initializations at current length
    for init_mode in [1, 5, 8]:
        a_init = diverse_init(n_current, rng, mode=init_mode)
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"Init mode {init_mode}: obj={best_score:.8f} **")

    # === Phase 1: Soft-max GD (coarse shaping) ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 1: SoftMax GD (coarse), time={remaining:.0f}s ===")
        alloc = min(remaining * 0.15, 150.0)
        a1, obj1 = soft_max_gd(best_seq, time_budget=alloc,
                                beta_start=5.0, beta_end=25.0,
                                lr_start=0.015, momentum=0.88, seed=seed, verbose=True)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1: obj={best_score:.8f} **")

    # === Phase 2: LP Cycling (main refinement) ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 2: LP Cycling (annealed), time={remaining:.0f}s ===")
        alloc = min(remaining * 0.30, 350.0)
        a2, obj2 = lp_cycling_annealed(best_seq, time_budget=alloc,
                                        max_iters=30000, verbose=True, seed=seed,
                                        T_init=0.03, T_end=0.0003)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2: obj={best_score:.8f} **")

        # Second LP round with different annealing
        remaining = deadline - time.monotonic()
        if remaining > 60:
            alloc2 = min(remaining * 0.20, 200.0)
            a2b, obj2b = lp_cycling_annealed(best_seq, time_budget=alloc2,
                                              max_iters=20000, verbose=False, seed=seed + 500,
                                              T_init=0.01, T_end=0.0001)
            if obj2b < best_score:
                best_score = obj2b
                best_seq = a2b.copy()
                print(f"  Phase2b: obj={best_score:.8f} **")

    # === Phase 3: Multiplicative Weights (different geometry) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 3: MultiWeigh

ts, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.12, 100.0)
        a3, obj3 = multiplicative_weights(best_seq, time_budget=alloc,
                                           eta_start=0.3, eta_end=0.02,
                                           seed=seed + 300, verbose=True)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3: obj={best_score:.8f} **")

        # Follow up with LP if MW helped
        remaining = deadline - time.monotonic()
        if remaining > 40:
            alloc_lp = min(remaining * 0.2, 80.0)
            a3b, obj3b = lp_cycling_annealed(a3, time_budget=alloc_lp,
                                              max_iters=10000, verbose=False,
                                              seed=seed + 400, T_init=0.02, T_end=0.0002)
            if obj3b < best_score:
                best_score = obj3b
                best_seq = a3b.copy()
                print(f"  Phase3 LP: obj={best_score:.8f} **")

    # === Phase 4: Diverse Restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 4: Diverse Restarts, time={remaining:.0f}s ===")
        n_cur = len(best_seq)

        for r in range(10):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            mode = r % 10
            a_r = diverse_init(n_cur, rng, mode=mode)
            # Gradient blend with current best
            blend = 0.3 + 0.05 * r
            a_r = project_simplex((1 - blend) * a_r + blend * best_seq)

            # Quick GD
            alloc_sgd = min(remaining * 0.15, 30.0)
            if alloc_sgd > 8:
                a_r, obj_r = soft_max_gd(a_r, time_budget=alloc_sgd,
                                           beta_start=15.0, beta_end=50.0,
                                           lr_start=0.01, momentum=0.92,
                                           verbose=False, seed=seed + 100 + r)

            # LP polish
            remaining = deadline - time.monotonic()
            if remaining > 20:
                alloc_lp = min(remaining * 0.25, 50.0)
                a_r, obj_r = lp_cycling_annealed(a_r, time_budget=alloc_lp,
                                                  max_iters=5000, verbose=False,
                                                  seed=seed + 200 + r,
                                                  T_init=0.02, T_end=0.001)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 5: Length Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 5: Length Exploration, time={remaining:.0f}s ===")
        n_best = len(best_seq)

        n_options = sorted(set([
            max(200, n_best - 300), max(200, n_best - 150),
            n_best + 150, n_best + 300, n_best + 500,
            400, 600, 900, 1100, 1400, 1800
        ]))
        n_options = [n for n in n_options if 200 <= n <= 2500]
        n_options = [n for n in n_options if abs(n - n_best) >= 100]
        n_options = n_options[:5]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            a_try = interp_length(best_seq, n_try)

            alloc_sgd = min(remaining * 0.25, 35.0)
            if alloc_sgd > 8:
                a_try, obj_try = soft_max_gd(a_try, time_budget=alloc_sgd,
                                               beta_start=8.0, beta_end=35.0,
                                               lr_start=0.012, momentum=0.9,
                                               verbose=False, seed=seed + 700)

            remaining = deadline - time.monotonic()
            if remaining > 20:
                alloc_lp = min(remaining * 0.35, 50.0)
                a_try, obj_try = lp_cycling_annealed(a_try, time_budget=alloc_lp,
                                                      max_iters=8000, verbose=False,
                                                      seed=seed + 800,
                                                      T_init=0.02, T_end=0.001)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final Intensive Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 6: Final Polish, time={remaining:.0f}s ===")

        # Sustained LP cycling (near-greedy)
        alloc = min(remaining * 0.5, 150.0)
        if alloc > 15:
            a_f, obj_f = lp_cycling_annealed(best_seq, time_budget=alloc,
                                              max_iters=50000, verbose=True,
                                              seed=seed + 2000,
                                              T_init=0.005, T_end=0.0001)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        # Last MW pass
        remaining = deadline - time.monotonic()
        if remaining > 25:
            alloc2 = min(remaining * 0.35, 60.0)
            a_f2, obj_f2 = multiplicative_weights(best_seq, time_budget=alloc2,
                                                   eta_start=0.1, eta_end=0.005,
                                                   seed=seed + 3000, verbose=False)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final MW: obj={best_score:.8f} **")

        # Tiny perturbation + LP
        remaining = deadline - time.monotonic()
        if remaining > 15:
            for attempt in range(3):
                remaining = deadline - time.monotonic()
                if remaining < 5:
                    break
                noise = rng.normal(0, 0.002, len(best_seq))
                a_p = project_simplex(best_seq + noise)
                alloc = min(remaining * 0.5, 15.0)
                if alloc > 3:
                    a_p, obj_p = lp_cycling_annealed(a_p, time_budget=alloc,
                                                      max_iters=5000, verbose=False,
                                                      seed=seed + 4000 + attempt,
                                                      T_init=0.003, T_end=0.0001)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    print(f"  Perturb {attempt}: obj={best_score:.8f} **")

    # Final validation with actual evaluator
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
