"""
Hybrid optimizer combining frequency-domain shaping (Gerchberg-Saxton style),
peak-directed subgradient descent with line search, coordinate descent, and
multi-n exploration to minimize 2*n*max(autocorr)/sum^2 for non-negative sequences.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv)/sum^2."""
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def freq_domain_shaping(a, time_budget=60.0, alpha_start=0.5, alpha_end=0.9, verbose=False, seed=42):
    """Gerchberg-Saxton style frequency domain shaping."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    patience = 0
    max_patience = 50

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        # FFT of current sequence
        size = 1
        while size < 2 * n:
            size <<= 1
        A = fft(a, size)
        P = np.abs(A) ** 2

        # Shape the power spectrum to be flatter
        # P_new = P^alpha (alpha < 1 flattens)
        P_flat = np.power(P + 1e-30, alpha)
        
        # Also try adding a small constant to flatten
        P_flat = P_flat + 1e-6 * np.mean(P_flat)

        # Reconstruct with same phase
        A_new = np.sqrt(P_flat) * np.exp(1j * np.angle(A))
        a_new = ifft(A_new).real

        # Enforce non-negativity and normalization
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            patience += 1
            alpha = max(0.1, alpha - 0.05)
            continue
        a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        # Blend with best
        t = 0.3 + 0.4 * (1 - progress)
        a = (1 - t) * best_a + t * a_new
        a = np.maximum(a, 0)
        s_a = float(np.sum(a))

            a = a / s_a

            patience = 0

        it += 1
        if verbose and it % 200 == 0:
            print(f"    FDS it{it}: obj={best_obj:.8f}, alpha={alpha:.3f}, t={elapsed:.1f}s")

def peak_directed_descent(a, time_budget=120.0, verbose=False, seed=42):
    """
    Peak-directed optimization: identify the lags where autocorrelation peaks,
    then optimize in the direction that most reduces those peaks.
    Uses exact line search.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    eta = 2.0 / n
    it = 0
    patience = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 2:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        
        # Find active lags (within 1% of max)
        threshold = max_val * 0.999
        active_lags = np.where(conv >= threshold)[0]

        if len(active_lags) == 0:
            active_lags = np.array([np.argmax(conv)])

        # Compute gradient direction: we want to decrease the peak lags
        # grad[i] = sum_{k in active} (a[k-i] + a[i-k]) / |active|
        direction = np.zeros(n)
        for k in active_lags:
            for j in range(n):
                idx1 = k - j  # a[k-j] if 0 <= k-j < n
                idx2 = j - k  # a[j-k] if 0 <= j-k < n
                if 0 <= idx1 < n:
                    direction[j] += a[idx1]
                if 0 <= idx2 < n:
                    direction[j] += a[idx2]
        direction /= (2.0 * len(active_lags))

        # We want to move in -direction (to decrease peaks)
        # But we need to stay in the feasible set: a >= 0, sum(a) = 1
        # Use a projected subgradient step with line search
        
        # Normalize direction
        dir_norm = float(np.linalg.norm(direction))
        if dir_norm < 1e-15:
            break
        direction = direction / dir_norm

        # Try multiple step sizes
        candidates_t = [eta, eta*0.5, eta*0.25, eta*2, eta*4]
        best_t_obj = best_obj
        best_t_a = best_a.copy()

        for t in candidates_t:
            a_try = a - t * direction
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_a = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = best_t_a
            patience = 0
            eta = min(eta * 1.2, 1.0)
        else:
            patience += 1
            if patience % 10 == 0:
                eta *= 0.8
                if eta < 1e-6:
                    # Random restart from best
                    pert = rng.normal(0, 0.03, n)
                    a = best_a * np.exp(pert)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    eta = 2.0 / n
                    patience = 0

        it += 1
        if verbose and it % 100 == 0:
            print(f"    PDD it{it}: obj={best_obj:.8f}, eta={eta:.6f}, t={elapsed:.1f}s")

    return best_a


def coordinate_descent(a, time_budget=300.0, verbose=False, seed=42):

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a

    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    stall_count = 0

    while time.monotonic() - start < time_budget:
        sweep += 1

        # Periodically recompute conv
        if sweep % 3 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    a = a / S_check

        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask].copy()

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            def f_1d(x):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Quick candidate evaluation

            if x_hi - x_lo < 1e-14:

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)

            fd = f_1d(d_pt)

            for _ in range(35):

                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-15:
                continue

            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta

            S_new = float(np.sum(a))
            if S_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if sweep % 10 == 0:
            obj_now = 2.0 * n * float(np.max(conv)) / (float(np.sum(a)) ** 2)
            if best_obj < prev_obj - 1e-13:
                prev_obj = best_obj
                stall_count = 0
            else:
                stall_count += 1
            if verbose and sweep % 30 == 0:
                elapsed = time.monotonic() - start
                print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s, stall={stall_count}")

def simulated_annealing(a, time_budget=120.0, verbose=False, seed=42):
    """Simulated annealing with structured moves."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    T = 0.1
    T_min = 1e-6
    cooling = 0.995
    it = 0

    while time.monotonic() - start < time_budget and T > T_min:
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break

        # Propose a move
        move_type = rng.integers(0, 4)
        a_proposed = a.copy()

        if move_type == 0:
            # Single coordinate change
            i = rng.integers(0, n)
            delta = rng.normal(0, 0.05 / n)
            a_proposed[i] = max(0, a_proposed[i] + delta)
        elif move_type == 1:
            # Pair swap
            i, j = rng.integers(0, n, 2)
            if i != j:
                delta = rng.normal(0, 0.05 / n)
                a_proposed[i] = max(0, a_proposed[i] + delta)
                a_proposed[j] = max(0, a_proposed[j] - delta)
        elif move_type == 2:
            # Segment perturbation
            seg_len = rng.integers(3, max(4, n // 10))
            start_idx = rng.integers(0, n - seg_len)
            noise = rng.normal(0, 0.02 / n, seg_len)
            a_proposed[start_idx:start_idx + seg_len] = np.maximum(
                a_proposed[start_idx:start_idx + seg_len] + noise, 0)
        else:
            # Global scale perturbation
            a_proposed = a * np.exp(rng.normal(0, 0.01))

        s_prop = float(np.sum(a_proposed))
        if s_prop < 1e-10:
            T *= cooling
            continue
        a_proposed = a_proposed / s_prop

        obj_prop = 2.0 * n * float(np.max(compute_conv(a_proposed)))
        obj_curr = 2.0 * n * float(np.max(compute_conv(a)))

        delta_obj = obj_prop - obj_curr
        if delta_obj < 0 or rng.random() < math.exp(-delta_obj / T):
            a = a_proposed
            if obj_prop < best_obj:
                best_obj = obj_prop
                best_a = a_proposed.copy()

        T *= cooling
        it += 1

        if verbose and it % 500 == 0:
            print(f"    SA it{it}: best={best_obj:.8f}, T={T:.6f}, t={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:
                    g_val = g_val / g_sum
                    return g_val
        except Exception:
            continue
    return None


def lp_blend(a, time_budget=40.0, verbose=False, seed=42):
    """LP-based blending with multiple t values."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

        a = best_a

        # Add small perturbation to avoid cycling
        pert = rng.normal(0, 0.01, n)
        a = a * np.exp(pert)
        a = np.maximum(a, 0)
        a = a / a.sum()

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        x = np.arange(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-0.999, 0.999, n)
        a = 1.0 / np.sqrt(1 - x * x)
        a = a / a.sum()
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = a + 1e-10
        a = a / a.sum()
    elif mode == 5:
        # Random smooth
        freqs = rng.integers(1, 8, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Power law
        x = np.arange(1, n + 1)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 7:
        # Triangular
        x = np.arange(n)
        mid = n // 2
        a = 1.0 - np.abs(x - mid) / (mid + 1)
        a = a / a.sum()
    elif mode == 8:
        # Periodic cosine
        x = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x / rng.integers(3, 10))
        a = np.maximum(a, 0.1)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Hybrid optimizer combining frequency-domain shaping (Gerchberg-Saxton style),
peak-directed subgradient descent, exact coordinate descent, LP blending,
simulated annealing, and multi-n exploration to minimize 2*n*max(autocorr)/sum^2.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv)/sum^2."""
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def dist_object(x, v, S_other, n_val, c_k_lin, a_ki_lin, c_2i):
    """Evaluate 1D subproblem: objective when coordinate i is set to x."""
    if len(a_ki_lin) > 0:
        lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
        max_lin = float(np.max(lin_vals))
    else:
        max_lin = 0.0
    quad_val = c_2i + x * x - v * v
    max_val = max(max_lin, quad_val)
    denom = (S_other + x) * (S_other + x)
    if denom < 1e-30:
        return 1e30
    return 2.0 * n_val * max_val / denom


def freq_domain_shaping(a, time_budget=60.0, alpha_start=0.5, alpha_end=0.9, verbose=False, seed=42):
    """Gerchberg-Saxton style frequency domain shaping to flatten autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    patience = 0
    max_patience = 80

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        size = 1
        while size < 2 * n:
            size <<= 1
        A = fft(a, size)
        P = np.abs(A) ** 2

        # Flatten power spectrum
        P_flat = np.power(P + 1e-30, alpha)
        P_flat = P_flat + 1e-6 * np.mean(P_flat)

        A_new = np.sqrt(P_flat) * np.exp(1j * np.angle(A))
        a_new = ifft(A_new).real

        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            patience += 1
            continue
        a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        # Blend
        t_blend = 0.2 + 0.5 * (1.0 - progress)
        a = (1.0 - t_blend) * best_a + t_blend * a_new
        a = np.maximum(a, 0)
        s_a = float(np.sum(a))
        if s_a > 1e-10:
            a = a / s_a

        if patience > max_patience:
            pert = rng.normal(0, 0.05, n)
            a = best_a * np.exp(pert)
            a = np.maximum(a, 0)
            a = a / a.sum()
            patience = 0

        it += 1
        if verbose and it % 200 == 0:
            print(f"    FDS it{it}: obj={best_obj:.8f}, alpha={alpha:.3f}, t={elapsed:.1f}s")

    return best_a


def peak_directed_descent(a, time_budget=120.0, verbose=False, seed=42):
    """Peak-directed optimization targeting the maximum autocorrelation lags."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    eta = 2.0 / n
    it = 0
    patience = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 2:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        threshold = max_val * 0.999
        active_lags = np.where(conv >= threshold)[0]

        if len(active_lags) == 0:
            active_lags = np.array([np.argmax(conv)])

        # Compute gradient of max-peak w.r.t. each coordinate
        direction = np.zeros(n)
        for k in active_lags:
            for j in range(n):
                idx1 = k - j
                idx2 = j - k
                if 0 <= idx1 < n:
                    direction[j] += a[idx1]
                if 0 <= idx2 < n:
                    direction[j] += a[idx2]
        direction /= (2.0 * len(active_lags))

        dir_norm = float(np.linalg.norm(direction))
        if dir_norm < 1e-15:
            break
        direction = direction / dir_norm

        # Line search over multiple step sizes
        candidates_t = [eta, eta * 0.5, eta * 0.25, eta * 2.0, eta * 4.0, eta * 0.1]
        best_t_obj = best_obj
        best_t_a = best_a.copy()

        for t in candidates_t:
            a_try = a - t * direction
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_a = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = best_t_a
            patience = 0
            eta = min(eta * 1.15, 2.0)
        else:
            patience += 1
            if patience % 15 == 0:
                eta *= 0.75
                if eta < 1e-5:
                    pert = rng.normal(0, 0.03, n)
                    a = best_a * np.exp(pert)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    eta = 2.0 / n
                    patience = 0

        it += 1
        if verbose and it % 200 == 0:
            print(f"    PDD it{it}: obj={best_obj:.8f}, eta={eta:.6f}, t={elapsed:.1f}s")

    return best_a


def coordinate_descent(a, time_budget=300.0, verbose=False, seed=42):
    """Exact coordinate descent with golden section search on 1D subproblems."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    prev_obj = best_obj
    stall_count = 0

    while time.monotonic() - start < time_budget:
        sweep += 1

        if sweep % 3 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    a = a / S_check

        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask].copy()

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            # Candidate evaluation
            scale = max(v, S / n, 1e-6)
            candidates = [0.0, v * 0.25, v * 0.5, v * 0.75, v,
                          v * 1.25, v * 1.5, v * 2.0,
                          S / (2 * n), S / n, 2.0 * S / n]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = dist_object(v, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
            for x in candidates:
                fx = dist_object(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section
            x_lo = max(0.0, best_x - scale * 0.3)
            x_hi = best_x + scale * 0.5
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = dist_object(c_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
            fd = dist_object(d_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)

            for _ in range(35):
                if x_hi - x_lo < 1e-13 * max(1.0, abs(x_hi)):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = dist_object(c_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = dist_object(d_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-15:
                continue

            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta

            S_new = float(np.sum(a))
            if S_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if sweep % 10 == 0:
            S_now = float(np.sum(a))
            if S_now > 1e-10:
                obj_now = 2.0 * n * float(np.max(conv)) / (S_now * S_now)
            else:
                obj_now = best_obj
            if best_obj < prev_obj - 1e-13:
                prev_obj = best_obj
                stall_count = 0
            else:
                stall_count += 1
            if verbose and sweep % 30 == 0:
                elapsed = time.monotonic() - start
                print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s, stall={stall_count}")

        if stall_count > 40:
            break

    return best_a


def simulated_annealing(a, time_budget=120.0, verbose=False, seed=42):
    """Simulated annealing with structured moves."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    T = 0.05
    T_min = 1e-7
    cooling = 0.997
    it = 0

    while time.monotonic() - start < time_budget and T > T_min:
        if time.monotonic() - start > time_budget - 1:
            break

        move_type = rng.integers(0, 4)
        a_proposed = a.copy()

        if move_type == 0:
            i = rng.integers(0, n)
            delta = rng.normal(0, 0.03 / n)
            a_proposed[i] = max(0.0, a_proposed[i] + delta)
        elif move_type == 1:
            i, j = rng.integers(0, n, 2)
            if i != j:
                delta = rng.normal(0, 0.03 / n)
                a_proposed[i] = max(0.0, a_proposed[i] + delta)
                a_proposed[j] = max(0.0, a_proposed[j] - delta)
        elif move_type == 2:
            seg_len = int(rng.integers(3, max(4, n // 8)))
            start_idx = int(rng.integers(0, max(1, n - seg_len)))
            noise = rng.normal(0, 0.01 / n, seg_len)
            a_proposed[start_idx:start_idx + seg_len] = np.maximum(
                a_proposed[start_idx:start_idx + seg_len] + noise, 0.0)
        else:
            a_proposed = a * np.exp(rng.normal(0, 0.005))

        s_prop = float(np.sum(a_proposed))
        if s_prop < 1e-10:
            T *= cooling
            it += 1
            continue
        a_proposed = a_proposed / s_prop

        obj_prop = 2.0 * n * float(np.max(compute_conv(a_proposed)))
        obj_curr = 2.0 * n * float(np.max(compute_conv(a)))

        delta_obj = obj_prop - obj_curr
        if delta_obj < 0 or (T > T_min and rng.random() < math.exp(-delta_obj / T)):
            a = a_proposed
            if obj_prop < best_obj:
                best_obj = obj_prop
                best_a = a_proposed.copy()

        T *= cooling
        it += 1

        if verbose and it % 1000 == 0:
            print(f"    SA it{it}: best={best_obj:.8f}, T={T:.6f}, t={time.monotonic()-start:.1f}s")

    return best_a


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:
                    g_val = g_val / g_sum
                    return g_val
        except Exception:
            continue
    return None


def lp_blend(a, time_budget=40.0, verbose=False, seed=42):
    """LP-based blending with perturbation to avoid cycling."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        improved = False
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7]:
            a_try = (1.0 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                improved = True

        if not improved:
            # Perturb to escape
            pert = rng.normal(0, 0.02, n)
            a = best_a * np.exp(pert)
            a = np.maximum(a, 0)
            a = a / a.sum()
        else:
            a = best_a

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        x = np.arange(n)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((x - c) ** 2) / (2 * w * w))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-0.999, 0.999, n)
        a = 1.0 / np.sqrt(1.0 - x * x)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
        a = a + 1e-10
        a = a / a.sum()
    elif mode == 5:
        freqs = rng.integers(1, 8, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2.0 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        x = np.arange(1, n + 1)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 7:
        x = np.arange(n)
        mid = n // 2
        a = 1.0 - np.abs(x - mid) / (mid + 1)
        a = a / a.sum()
    elif mode == 8:
        x = np.arange(n)
        period = int(rng.integers(3, 10))
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * x / period)
        a = np.maximum(a, 0.1)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase hybrid search combining:
    1. Quick CD polish
    2. Frequency-domain shaping (Gerchberg-Saxton)
    3. Peak-directed subgradient descent
    4. Simulated annealing
    5. LP blending
    6. Multi-n exploration
    7. Diverse restarts
    8. Final polish (CD + FDS)
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_default = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = 2.0 * parent.size * float(np.max(compute_conv(parent)))
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(n_default) / n_default
        best_score = 2.0

    print(f"  Initial: n={len(best_seq)}, obj={best_score:.8f}")

    # === Phase 1: Quick CD Polish ===
    print(f"\n=== Phase 1: CD Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        p1_budget = min(remaining * 0.18, 180.0)
        a1 = coordinate_descent(best_seq, time_budget=p1_budget, verbose=True, seed=seed)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj1:.8f}")

    # === Phase 2: Frequency Domain Shaping ===
    print(f"\n=== Phase 2: Freq Domain Shaping ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        for rd in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            p2_alloc = min(remaining * 0.1, 80.0)
            a2 = freq_domain_shaping(best_seq, time_budget=p2_alloc,
                                     alpha_start=0.3 + 0.05 * rd,
                                     alpha_end=0.7 + 0.05 * rd,
                                     verbose=False, seed=100 + rd * 7)
            obj2 = compute_obj(a2)
            if obj2 < best_score:
                best_score = obj2
                best_seq = a2.copy()
                print(f"  FDS round {rd}: obj={best_score:.8f} **")
            else:
                print(f"  FDS round {rd}: obj={obj2:.8f}")

    # === Phase 3: CD after FDS ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = min(remaining * 0.1, 80.0)
        a3 = coordinate_descent(best_seq, time_budget=p3_budget, verbose=False, seed=seed + 10)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  CD after FDS: obj={best_score:.8f} **")

    # === Phase 4: Peak-Directed Descent ===
    print(f"\n=== Phase 4: Peak-Directed Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p4_budget = min(remaining * 0.15, 120.0)
        a4 = peak_directed_descent(best_seq, time_budget=p4_budget,
                                   verbose=True, seed=seed + 20)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  PDD: obj={best_score:.8f} **")
        else:
            print(f"  PDD: obj={obj4:.8f}")

    # === Phase 5: CD + FDS after PDD ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a5 = coordinate_descent(best_seq, time_budget=min(50, remaining * 0.15),
                                verbose=False, seed=seed + 30)
        obj5 = compute_obj(a5)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  CD after PDD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 25:
        a5b = freq_domain_shaping(best_seq, time_budget=min(40, remaining * 0.15),
                                  alpha_start=0.4, alpha_end=0.8,
                                  verbose=False, seed=seed + 31)
        obj5b = compute_obj(a5b)
        if obj5b < best_score:
            best_score = obj5b
            best_seq = a5b.copy()
            print(f"  FDS after PDD: obj={best_score:.8f} **")

    # === Phase 6: Simulated Annealing ===
    print(f"\n=== Phase 6: Simulated Annealing ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p6_budget = min(remaining * 0.1, 80.0)
        a6 = simulated_annealing(best_seq, time_budget=p6_budget,
                                 verbose=False, seed=seed + 40)
        obj6 = compute_obj(a6)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  SA: obj={best_score:.8f} **")
        else:
            print(f"  SA: obj={obj6:.8f}")
        # Quick CD after SA
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a6b = coordinate_descent(best_seq, time_budget=min(30, remaining * 0.2),
                                     verbose=False, seed=seed + 41)
            obj6b = compute_obj(a6b)
            if obj6b < best_score:
                best_score = obj6b
                best_seq = a6b.copy()
                print(f"  CD after SA: obj={best_score:.8f} **")

    # === Phase 7: LP Blending ===
    print(f"\n=== Phase 7: LP Blending ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p7_budget = min(remaining * 0.08, 50.0)
        a7 = lp_blend(best_seq, time_budget=p7_budget, verbose=False, seed=seed + 50)
        obj7 = compute_obj(a7)
        if obj7 < best_score:
            best_score = obj7
            best_seq = a7.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj7:.8f}")

    # === Phase 8: Multi-n Exploration ===
    print(f"\n=== Phase 8: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_candidates = sorted(set([
        max(300, n_current - 150),
        n_current + 100, n_current + 250, n_current + 500,
        500, 700, 1000, 1300, 1700, 2200
    ]))
    n_filtered = []
    for nc in n_candidates:
        if 300 <= nc <= 2500:
            if all(abs(nc - nf) > 100 for nf in n_filtered):
                n_filtered.append(nc)
    n_filtered = n_filtered[:5]
    print(f"  n candidates: {n_filtered}")

    for n_try in n_filtered:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.1, 50.0)
        if alloc > 15:
            a_try = coordinate_descent(a_try, time_budget=alloc, verbose=False, seed=n_try)

        # Also try FDS
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            a_try2 = freq_domain_shaping(a_try, time_budget=min(25, remaining2 * 0.3),
                                         alpha_start=0.4, alpha_end=0.8,
                                         verbose=False, seed=n_try + 1)
            obj_try2 = compute_obj(a_try2)
            obj_try = compute_obj(a_try)
            if obj_try2 < obj_try:
                a_try = a_try2
                obj_try = obj_try2

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 9: Diverse Restarts ===
    print(f"\n=== Phase 9: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = (r * 3 + 1) % 9
        a_r = diverse_init(n_cur, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1.0 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.12, 50.0)
        if alloc > 15:
            a_r = coordinate_descent(a_r, time_budget=alloc, verbose=False,
                                     seed=5000 + r * 100)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode {mode}): obj={obj_r:.8f}")

    # === Phase 10: Final Polish ===
    print(f"\n=== Phase 10: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        for fr in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            # CD
            cd_alloc = min(remaining * 0.3, 70.0)
            if cd_alloc > 10:
                a_f = coordinate_descent(best_seq, time_budget=cd_alloc,
                                         verbose=False, seed=9000 + fr)
                obj_f = compute_obj(a_f)
                if obj_f < best_score:
                    best_score = obj_f
                    best_seq = a_f.copy()
                    print(f"  Final CD {fr}: obj={best_score:.8f} **")

            # FDS
            remaining = deadline - time.monotonic()
            if remaining > 15:
                fds_alloc = min(remaining * 0.25, 40.0)
                if fds_alloc > 8:
                    a_f2 = freq_domain_shaping(best_seq, time_budget=fds_alloc,
                                               alpha_start=0.4, alpha_end=0.85,
                                               verbose=False, seed=9500 + fr)
                    obj_f2 = compute_obj(a_f2)
                    if obj_f2 < best_score:
                        best_score = obj_f2
                        best_seq = a_f2.copy()
                        print(f"  Final FDS {fr}: obj={best_score:.8f} **")

            # PDD
            remaining = deadline - time.monotonic()
            if remaining > 15:
                pdd_alloc = min(remaining * 0.2, 30.0)
                if pdd_alloc > 8:
                    a_f3 = peak_directed_descent(best_seq, time_budget=pdd_alloc,
                                                 verbose=False, seed=9700 + fr)
                    obj_f3 = compute_obj(a_f3)
                    if obj_f3 < best_score:
                        best_score = obj_f3
                        best_seq = a_f3.copy()
                        print(f"  Final PDD {fr}: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = 2.0

    best_seq = np.maximum(best_seq, 0.0)
    result = [float(min(1000.0, x)) for x in best_seq]

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
