"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1

    n = len(a)

            elapsed = time.monotonic() - start
            print(f"      LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                  f"lp_time={lp_time:.2f}s, stall={stall}, elapsed={elapsed:.1f}s")
        if stall >= 3:
            if stall <= 5:
                freq = rng.integers(1, max(2, n // 50))
                phase = rng.uniform(0, 2 * np.pi)
                idx = np.arange(n, dtype=np.float64)
                pert = 0.02 * stall * np.cos(2 * np.pi * freq * idx / n + phase)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 0.01:
                    a = a / s
                else:
                    a = best_a.copy()
            else:
                mode = stall % 5
                a_div = diverse_init(n, rng, mode)
                a = 0.4 * a_div + 0.6 * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()
            stall = 0
    return best_a, best_obj


def diverse_init(n, rng, mode):
    x = np.linspace(0, 1, n)
    idx = np.arange(n, dtype=np.float64)
    if mode == 0:
        a = 1.0 / (0.05 + 3.0 * (2 * x - 1) ** 2)
    elif mode == 1:
        a = np.exp(-((idx - n * 0.2) ** 2) / (2 * (n * 0.08) ** 2)) + \
            np.exp(-((idx - n * 0.8) ** 2) / (2 * (n * 0.08) ** 2)) + 0.01
    elif mode == 2:
        a = 4.0 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(4.0 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.03 + 8.0 * (2 * x - 1) ** 4)
    elif mode == 6:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * 3 * idx / n)
        a = np.maximum(a, 0.01)
    elif mode == 7:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 5 * idx / n)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a_src, n_dst):
    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s


def symmetric_init(n, rng, mode):
    m = (n + 1) // 2
    x = np.linspace(0, 1, m)
    if mode == 0:
        half = 1.0 / (0.1 + 2.0 * (2 * x - 0) ** 2)
    elif mode == 1:
        half = 0.5 + 0.5 * np.cos(np.pi * x)
    elif mode == 2:
        half = np.exp(-2.0 * (x - 0.5) ** 2) + 0.1
    else:
        half = 1.0 / (0.05 + 4.0 * x ** 2)
    half = np.maximum(half, 0.01)
    if n % 2 == 1:
        a = np.concatenate([half, half[-2::-1]])
    else:
        a = np.concatenate([half, half[:-1][::-1]])
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 0.01:
        return np.ones(n) / n
    return a / s


def symmetry_project(a):
    n = len(a)
    rev = a[::-1]
    a_sym = (a + rev) / 2.0
    a_sym = np.maximum(a_sym, 0)
    s = float(np.sum(a_sym))
    if s < 1e-10:
        return a
    return a_sym / s


def propose_candidate(seed=42, budget_s=1500):
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        s = float(np.sum(parent_arr))
        if s > 0.01:
            best_seq = parent_arr / s
            best_score = compute_obj(best_seq)
            n_best = len(best_seq)
            print(f"Parent loaded: n={n_best}, obj={best_score:.8f}")
        else:
            raise ValueError("Sum too small")
    except Exception:
        best_seq = diverse_init(800, rng, 0)
        best_score = compute_obj(best_seq)
        n_best = len(best_seq)
        print(f"Default init: n={n_best}, obj={best_score:.8f}")

    # Try alternative inits at same n
    for mode in range(8):
        a_alt = diverse_init(n_best, rng, mode)
        obj_alt = compute_obj(a_alt)
        if obj_alt < best_score:
            best_score = obj_alt
            best_seq = a_alt.copy()
            print(f"  Alt init mode={mode}: obj={best_score:.8f} **")

    # Try symmetric inits
    for mode in range(4):
        a_sym = symmetric_init(n_best, rng, mode)
        obj_sym = compute_obj(a_sym)
        if obj_sym < best_score:
            best_score = obj_sym
            best_seq = a_sym.copy()
            print(f"  Sym init mode={mode}: obj={best_score:.8f} **")

    # === Phase 1: Softmax gradient descent ===
    remaining = deadline - time.monotonic()
    if remaining > 150:
        print(f"\n=== Phase 1: Softmax Descent (n={n_best}, budget={min(remaining, 250):.0f}s) ===")
        alloc = min(remaining * 0.2, 250.0)
        a_sg, obj_sg = softmax_descent(best_seq, time_budget=alloc,
                                       beta_start=0.5, beta_end=1000.0,
                                       lr0=0.4, momentum_decay=0.6, verbose=True)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Phase 1 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 done: obj={obj_sg:.8f} (no improvement from {best_score:.8f})")

    # === Phase 2: LP Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 2: LP Polish (budget={min(remaining, 350):.0f}s) ===")
        alloc = min(remaining * 0.3, 350.0)
        a_lp, obj_lp = lp_polish(best_seq, time_budget=alloc, max_iters=200, verbose=True)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase 2 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 done: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Refuge with softmax at high beta ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 3: High-Beta Softmax ===")
        alloc = min(remaining * 0.1, 80.0)
        a_sg2, obj_sg2 = softmax_descent(best_seq, time_budget=alloc,
                                         beta_start=50.0, beta_end=5000.0,
                                         lr0=0.15, momentum_decay=0.5, verbose=True)
        if obj_sg2 < best_score:
            best_score = obj_sg2
            best_seq = a_sg2.copy()
            print(f"  Phase 3 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3 done: obj={obj_sg2:.8f} (no improvement)")

    # === Phase 4: LP Polish Round 2 ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: LP Polish R2 ===")
        alloc = min(remaining * 0.15, 150.0)
        a_lp2, obj_lp2 = lp_polish(best_seq, time_budget=alloc, max_iters=80, verbose=True)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase 4 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4 done: obj={obj_lp2:.8f} (no improvement)")

    # === Phase 5: Symmetry projection + re-optimize ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 5: Symmetry Exploration ===")
        a_sym = symmetry_project(best_seq)
        obj_sym = compute_obj(a_sym)
        if obj_sym < best_score:
            best_score = obj_sym
            best_seq = a_sym.copy()
            print(f"  Symmetric projection: obj={best_score:.8f} **")
        else:
            print(f"  Symmetric projection: obj={obj_sym:.8f} (worse, keeping asymmetric)")
            # Try partial symmetry
            for blend in [0.3, 0.5, 0.7]:
                a_blend = (1 - blend) * best_seq + blend * symmetry_project(best_seq)
                obj_blend = compute_obj(a_blend)
                if obj_blend < best_score:
                    best_score = obj_blend
                    best_seq = a_blend.copy()
                    print(f"  Partial sym blend={blend}: obj={best_score:.8f} **")

    # === Phase 6: N Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 200:
        print(f"\n=== Phase 6: N Exploration (budget~{remaining:.0f}s) ===")
        n_cur = len(best_seq)
        n_options = sorted(set([
            n_cur - 100, n_cur - 50, n_cur - 200,
            n_cur + 50, n_cur + 100, n_cur + 200, n_cur + 400,
            n_cur + 600, n_cur + 800, n_cur + 1000, n_cur + 1500,
            500, 600, 700, 900, 1000, 1100, 1200, 1500, 1800, 2000, 2500, 3000
        ]))
        n_options = [n for n in n_options if 300 <= n <= 3500 and abs(n - n_cur) >= 50]
        n_options.sort(key=lambda x: abs(x - n_cur))
        n_options = n_options[:8]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 80:
                break
            a_try = interpolate_seq(best_seq, n_try)
            if a_try is None:
                continue
            # Also try a symmetric version
            a_sym_try = symmetry_project(a_try)
            obj_sym_try = compute_obj(a_sym_try)
            obj_try = compute_obj(a_try)
            if obj_sym_try < obj_try:
                a_try = a_sym_try
                obj_try = obj_sym_try
            # Quick LP polish
            alloc = min(remaining * 0.3, 100.0)
            a_try2, obj_try2 = lp_polish(a_try, time_budget=alloc, max_iters=50, verbose=False)
            # Quick softmax
            if remaining > 120:
                alloc2 = min(remaining * 0.1, 40.0)
                a_try3, obj_try3 = softmax_descent(a_try2, time_budget=alloc2,
                                                   beta_start=10.0, beta_end=2000.0,
                                                   lr0=0.2, momentum_decay=0.5, verbose=False)
                if obj_try3 < obj_try2:
                    a_try2 = a_try3
                    obj_try2 = obj_try3
            if obj_try2 < best_score:
                best_score = obj_try2
                best_seq = a_try2.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try2:.8f}")

    # === Phase 7: Final intensive LP at best n ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 7: Final LP Polish ===")
        alloc = min(remaining - 10, 200.0)
        if alloc > 20:
            a_final, obj_final = lp_polish(best_seq, time_budget=alloc, max_iters=100, verbose=True)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
            print(f"  Final LP: obj={obj_final:.8f} (best={best_score:.8f})")

    # === Phase 8: Final squeeze ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 8: Final Squeeze ===")
        # Short softmax
        alloc = min(remaining * 0.4, 40.0)
        a_sq, obj_sq = softmax_descent(best_seq, time_budget=alloc,
                                       beta_start=100.0, beta_end=10000.0,
                                       lr0=0.1, momentum_decay=0.4, verbose=False)
        if obj_sq < best_score:
            best_score = obj_sq
            best_seq = a_sq.copy()
            print(f"  SG squeeze: obj={best_score:.8f} **")
        # Short LP
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            a_sq2, obj_sq2 = lp_polish(best_seq, time_budget=min(remaining2 - 5, 50.0),
                                       max_iters=30, verbose=False)
            if obj_sq2 < best_score:
                best_score = obj_sq2
                best_seq = a_sq2.copy()
                print(f"  LP squeeze: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(200)
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
