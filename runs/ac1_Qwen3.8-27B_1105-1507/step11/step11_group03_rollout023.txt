def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def autocorr_gradient(a, lag):
    """Compute gradient of R(lag) = sum_i a[i]*a[i+lag] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 2.0 * a
    else:
        # dR/d a[j] = a[j+lag] (if j+lag < n) + a[j-lag] (if j-lag >= 0)
        forward = n - lag
        grad[:forward] += a[lag:]
        grad[lag:] += a[:forward]
    return grad


def smoothed_grad(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    shifted = tau * (conv - exact_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max

def nesterov_gd(a, time_budget, tau_levels=None, verbose=False, beta=0.9, lr_scale=1.5):
    """Nesterov accelerated gradient descent with staircase tau schedule."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [3, 8, 20, 50, 120, 300, 700, 1500]

    for tau in tau_levels:
        lr = lr_scale / np.sqrt(n)
        v = np.zeros(n)
        tau_iters = 0
        max_iters_per_level = 3000

        while time.monotonic() - start < time_budget:
            if tau_iters >= max_iters_per_level:
                break

            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new
            tau_iters += 1
            lr = max(1e-7, lr * 0.999995)

        if verbose:
            print(f"    tau={tau:5d}: obj={best_obj:.10f}, iters={tau_iters}")

    return best_a


def lp_iteration_main(a, time_budget, verbose=False, seed=42, max_iters=3000):

    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))

        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall_count = 0
    delta = 0.003
    lp_interval = 1  # do LP every this many iterations
    gd_interval = 5  # do GD every this many iterations

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        improved = False

        # LP step
        if it % lp_interval == 0 and remaining > 10:
            d = lp_steepest_descent_v2(a, K=80, delta=delta)
            if d is not None:
                # Line search
                best_local_obj = best_obj
                best_local_a = best_a.copy()
                for alpha in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
                    a_try = a + alpha * d
                    a_try = np.maximum(a_try, 0)
                    st = float(np.sum(a_try))
                    if st < 1e-10:
                        continue
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_local_obj - 1e-15:
                        best_local_obj = obj_try
                        best_local_a = a_try.copy()

                if best_local_obj < best_obj - 1e-15:
                    best_obj = best_local_obj
                    best_a = best_local_a.copy()
                    a = best_a
                    improved = True
                    stall_count = 0
                    # Shrink delta slowly
                    delta = max(0.0005, delta * 0.999)
                    if verbose and it % 50 == 0:
                        print(f"    LP it{it:4d}: obj={best_obj:.10f}, delta={delta:.6f}")
                else:
                    # LP didn't help, try smaller delta
                    delta = max(0.0003, delta * 0.5)

        # Smoothed GD step
        if it % gd_interval == 0 and remaining > 15 and not improved:
            tau = 20 + 10 * (it // 100)
            lr = 1.0 / np.sqrt(n)
            g, _ = smoothed_grad(a, tau)
            gn = float(np.linalg.norm(g))
            if gn > 1e-15:
                a_try = a - lr * g / gn * 0.01
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st > 1e-10:
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-15:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        stall_count = 0

        if not improved:
            stall_count += 1
            if stall_count > 20:
                # Perturbation to escape
                pert_strength = min(0.05, 0.01 * (stall_count // 10))
                a = a + pert_strength * rng.standard_normal(n)
                a = np.maximum(a, 0)
                sa = float(np.sum(a))
                if sa > 1e-10:
                    a = a / sa
                else:
                    a = best_a.copy()
                stall_count = 0
                delta = 0.003

    return best_a

                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None


def cross_corr_lp_fixed_point(a, time_budget, verbose=False, seed=42, max_iters=500):
    """Original cross-correlation LP fixed point (for comparison/fallback)."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall_count = 0
    t_current = 0.15

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = cross_corr_lp(a)
        if g_lp is None:
            stall_count += 1
            if stall_count > 10:
                break
            continue
        stall_count = 0

        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t_current

        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a
            t_current = max(0.03, min(1.0, best_local_t * 0.8))
            if verbose and it % 20 == 0:
                print(f"    XLP it{it:4d}: obj={best_obj:.10f}, t={best_local_t:.3f}")
        else:
            stall_count += 1
            if stall_count > 15:
                a = best_a + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_count = 0
                t_current = 0.1

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_init(n, mode, rng):
    """Create structured initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.maximum(np.sinc(x), 0.01)
        return a / a.sum()
    elif mode == 2:
        a = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
        return a / a.sum()
    elif mode == 3:
        x = np.arange(n) / (n - 1)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        return a / a.sum()
    elif mode == 4:
        x = np.linspace(-3, 3, n)
        a = np.exp(-x ** 2)
        return a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
        return a / a.sum()
    elif mode == 6:
        a = rng.uniform(0.05, 1.0, n)
        return a / a.sum()
    elif mode == 7:
        # Random Fourier
        x = np.arange(n) / n
        a = 1.0
        for _ in range(3):
            f = rng.integers(1, 5)
            p = rng.uniform(0, 2 * np.pi)
            a += 0.2 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def combined_optimize(a, time_budget, verbose=False, seed=42):
    """Combined optimization: LP + GD alternation with adaptive scheduling."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall_count = 0
    phase = 0  # 0=LP, 1=GD, 2=perturb

    for it in range(10000):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        improved = False

        if phase == 0:
            # Direct autocorrelation LP
            K = 60 if remaining > 100 else 40
            delta = 0.002 if stall_count > 10 else 0.004
            d = lp_steepest_descent_v2(a, K=K, delta=delta)
            if d is not None:
                for alpha in [1.0, 0.6, 0.4, 0.2, 0.1]:
                    a_try = a + alpha * d
                    a_try = np.maximum(a_try, 0)
                    st = float(np.sum(a_try))
                    if st < 1e-10:
                        continue
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-15:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        stall_count = 0
                        break
                if improved:
                    if verbose and it % 30 == 0:
                        print(f"    it{it:4d} [LP]: obj={best_obj:.10f}")
            phase = 1 if not improved else 0

        elif phase == 1:
            # Smoothed GD
            tau = 30
            lr = 0.5 / np.sqrt(n)
            g, _ = smoothed_grad(a, tau)
            gn = float(np.linalg.norm(g))
            if gn > 1e-15:
                a_try = a - lr * g
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st > 1e-10:
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-15:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        stall_count = 0
            if not improved:
                stall_count += 1
                phase = 0 if stall_count < 15 else 2

        else:
            # Perturbation
            strength = 0.01 + 0.005 * (stall_count // 20)
            strength = min(strength, 0.04)
            a = a + strength * rng.standard_normal(n)
            a = np.maximum(a, 0)
            sa = float(np.sum(a))
            if sa > 1e-10:
                a = a / sa
            else:
                a = best_a.copy()
            stall_count = 0
            phase = 0

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining direct autocorrelation LP, cross-correlation LP,
    Nesterov GD, multi-n exploration, and diverse initialization.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                parent_n = len(parent_arr)
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent_n}): obj={best_score:.10f}")
    except Exception:
        pass

    n_current = len(best_seq) if best_seq is not None else 800

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.10f}")
        n_current = 800

    # === Phase 1: Quick multi-n screening ===
    print(f"\n=== Phase 1: Multi-n Screening ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        n_options = [700, 800, 900, 1000, 1100, 1200, 1400, 1500]
        n_options = [n for n in n_options if n != n_current]
        n_options.sort(key=lambda x: abs(x - n_current))
        n_options = n_options[:4]
        
        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            a_try = resample(best_seq, n_try)
            # Quick GD
            alloc = min(remaining * 0.1, 25.0)
            if alloc > 8:
                a_try = nesterov_gd(a_try, time_budget=alloc,
                                     tau_levels=[10, 50, 200], verbose=False)
            obj_try = compute_obj(a_try)
            if obj_try < best_score - 1e-10:
                best_score = obj_try
                best_seq = a_try.copy()
                n_current = n_try
                print(f"  ** n={n_try}: obj={best_score:.10f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 2: Direct autocorrelation LP (main) ===
    print(f"\n=== Phase 2: Direct AutoCorr LP ===")
    remaining = deadline - time.monotonic()
    if remaining > 200:
        alloc = min(remaining * 0.35, 500.0)
        a2 = lp_iteration_main(best_seq, time_budget=alloc, verbose=True, seed=42)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 AutoLP: obj={best_score:.10f} **")
        else:
            print(f"  Phase2 AutoLP: obj={obj2:.10f}")

    # === Phase 3: Cross-correlation LP (alternative) ===
    print(f"\n=== Phase 3: Cross-Corr LP ===")
    remaining = deadline - time.monotonic()
    if remaining > 150:
        alloc = min(remaining * 0.20, 300.0)
        a3 = cross_corr_lp_fixed_point(best_seq, time_budget=alloc, verbose=True, seed=77)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3 XLP: obj={best_score:.10f} **")
        else:
            print(f"  Phase3 XLP: obj={obj3:.10f}")

    # === Phase 4: Combined optimization ===
    print(f"\n=== Phase 4: Combined LP+GD ===")
    remaining = deadline - time.monotonic()
    if remaining > 150:
        alloc = min(remaining * 0.25, 350.0)
        a4 = combined_optimize(best_seq, time_budget=alloc, verbose=True, seed=123)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4 Combined: obj={best_score:.10f} **")
        else:
            print(f"  Phase4 Combined: obj={obj4:.10f}")

    # === Phase 5: Nesterov GD polish ===
    print(f"\n=== Phase 5: Nesterov GD Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.15, 200.0)
        a5 = nesterov_gd(best_seq, time_budget=alloc,
                         tau_levels=[20, 60, 150, 400, 800], verbose=True)
        obj5 = compute_obj(a5)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  Phase5 GD: obj={best_score:.10f} **")
        else:
            print(f"  Phase5 GD: obj={obj5:.10f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        for mode in [1, 4, 7]:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            a_init = resample(best_seq, n_current)
            # Perturb strongly
            a_init = a_init + 0.05 * rng.standard_normal(n_current)
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()
            
            alloc = min(remaining * 0.15, 60.0)
            if alloc > 15:
                a_restart = combined_optimize(a_init, time_budget=alloc, verbose=False,
                                              seed=1000 + mode)
                obj_restart = compute_obj(a_restart)
                if obj_restart < best_score:
                    best_score = obj_restart
                    best_seq = a_restart.copy()
                    print(f"  Restart mode {mode}: obj={best_score:.10f} **")
                else:
                    print(f"  Restart mode {mode}: obj={obj_restart:.10f}")

    # === Phase 7: Second direct LP pass with refined params ===
    print(f"\n=== Phase 7: Refined LP Pass ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        alloc = min(remaining * 0.30, 250.0)
        a7 = lp_iteration_main(best_seq, time_budget=alloc, verbose=True, seed=9999)
        obj7 = compute_obj(a7)
        if obj7 < best_score:
            best_score = obj7
            best_seq = a7.copy()
            print(f"  Phase7 LP2: obj={best_score:.10f} **")
        else:
            print(f"  Phase7 LP2: obj={obj7:.10f}")

    # === Phase 8: Ultra-fine GD ===
    print(f"\n=== Phase 8: Ultra-fine GD ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 80.0)
        a8 = nesterov_gd(best_seq, time_budget=alloc,
                         tau_levels=[100, 300, 800, 2000], verbose=False)
        obj8 = compute_obj(a8)
        if obj8 < best_score:
            best_score = obj8
            best_seq = a8.copy()
            print(f"  Phase8 UltraGD: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    return result
