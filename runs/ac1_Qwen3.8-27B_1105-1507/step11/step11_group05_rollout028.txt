import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a, b=None):
    """Compute conv(a, b) or conv(a, a) via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def solve_full_lp(a, solver=None):
    """Solve: min M s.t. conv(a, g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / float(np.sum(a))
    
    # Build A matrix: A[k, j] = a[k-j] if 0 <= k-j < n
    A = np.zeros((2*n - 1, n))
    for j in range(n):
        # a[k-j] is valid when 0 <= k-j < n, i.e., j <= k < j+n
        k_start = j
        k_end = min(2*n - 1, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver is None:
        solver = cp.CLARABEL
    
    try:
        prob.solve(solver=solver)
        if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
            g_val = np.maximum(np.asarray(g.value).flatten(), 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception as e:
        print(f"  LP solve failed: {e}")
    
    return None

def find_best_blend(a, g, n_trials=20):
    """Find best blend factor between a and g."""
    n = len(a)
    best_t = 0.0
    best_obj = compute_obj(a)
    
    # Coarse search
    for t in np.linspace(0.05, 1.0, n_trials):
        a_try = (1 - t) * a + t * g
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
    
    # Refine around best_t
    if best_t > 0:
        for dt in np.linspace(-0.2, 0.2, 10):
            t_try = best_t + dt
            if 0 < t_try < 1.2:
                a_try = (1 - t_try) * a + t_try * g
                s = float(np.sum(a_try))
                if s < 0.01:
                    continue
                a_try = a_try / s
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_t = t_try
    
    return best_t, best_obj

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    result = irfft(fa * fb, size)
    return result[:n + m - 1]

def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_A(a):
    """Build (2n-1) x n matrix for LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A

def solve_full_lp(a, solver=None):
    """Solve min M s.t. A@b <= M, sum(b)=1, b>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_A(a_norm)
    
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ b <= M, cp.sum(b) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    if solver is not None:
        solvers = [solver] + solvers
    
    for s in solvers:
        try:
            prob.solve(solver=s)
            if b.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b.value).flatten(), 0)
                sv = float(np.sum(b_val))

    return None

def coordinate_descent(a, time_budget):
    """Coordinate descent with quantile-based candidates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_obj = compute_obj(a)
    start = time.monotonic()
    
    improved_any = True
    while improved_any and (time.monotonic() - start) < time_budget:
        improved_any = False
        for i in range(n):
            if (time.monotonic() - start) >= time_budget:
                break
            old = a[i]
            if old < 1e-12:
                continue
            
            # Try several multipliers
            best_val = old
            best_local = best_obj
            for mult in [0.0, 0.3, 0.6, 1.0, 1.4, 2.0, 3.0]:
                a[i] = old * mult
                s_new = float(np.sum(a))
                if s_new < 0.01:
                    continue
                obj_try = compute_obj(a)
                if obj_try < best_local:
                    best_local = obj_try
                    best_val = old * mult
            
            if best_val != old:
                a[i] = best_val
                improved_any = True
                if best_local < best_obj:
                    best_obj = best_local
                    best_a = a.copy()
            else:
                a[i] = old

    return best_a

"""
Full-LP iterative optimization for step function coefficients.
Key improvements over previous approach:
1. Complete LP (all 2n-1 convolution constraints) for precise peak control
2. Progressive constraint refinement (start truncated, expand to full)
3. Adaptive blending with bisection refinement
4. Periodic multi-perturbation restarts to escape local optima
5. Efficient coordinate descent with reduced candidate set
6. Strategic n exploration with interpolation transfer
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a, b=None):
    """Full linear convolution via FFT. Returns array of length n+m-1."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    result = irfft(fa * fb, size)
    return result[:n + m - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_A(a, subset=None):
    """Build matrix A where A[k,j] = a[k-j] if 0<=k-j<n. Optionally use subset of rows."""
    n = len(a)
    A_full = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_full[j:j + n, j] = a
    if subset is not None:
        return A_full[subset, :]
    return A_full


def solve_lp(a, top_m=None, solver=None):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0. If top_m, use only top constraint rows."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    if top_m is not None and top_m < 2 * n - 1:
        conv_a = compute_conv(a_norm)
        idx = np.argsort(conv_a)[::-1][:top_m]
        A = build_A(a_norm, subset=idx)
    else:
        A = build_A(a_norm)
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solver_list = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    if solver is not None:
        solver_list = [solver] + [s for s in solver_list if s != solver]
    
    for s in solver_list:
        try:
            prob.solve(solver=s)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g_var.value).flatten(), 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv
        except Exception:
            continue
    return None


def find_best_blend(a, g, n_coarse=20, n_fine=15):
    """Find optimal blend factor between a and g via coarse+fine search."""
    best_t = 0.0
    best_obj = compute_obj(a)
    
    # Coarse grid
    for t in np.linspace(0.05, 1.0, n_coarse):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
    
    # Fine refinement around best
    if best_t > 0.01:
        lo = max(0.005, best_t - 0.08)
        hi = min(1.1, best_t + 0.08)
        for t in np.linspace(lo, hi, n_fine):
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_t = t
    
    return best_t, best_obj


def lp_iteration(a, time_budget, max_iters=500, top_m_start=None, top_m_end=None,
                 verbose=False, perturb_period=8):
    """
    Main LP iteration loop with progressive constraint refinement.
    Starts with few constraints, expands to full over time.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    if top_m_start is None:
        top_m_start = min(30, 2 * n - 1)
    if top_m_end is None:
        top_m_end = 2 * n - 1
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    it = 0
    
    while (time.monotonic() - start) < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # Progressive top_m: start small, grow to full
        frac = min(1.0, it / max(1, max_iters * 0.5))
        top_m = int(round(top_m_start + frac * (top_m_end - top_m_start)))
        top_m = max(10, min(top_m, 2 * n - 1))
        
        # Solve LP
        g = solve_lp(a, top_m=top_m)
        if g is None:
            if verbose:
                print(f"    LP failed at iter {it}")
            stall += 2
            if stall > 10:
                break
            it += 1
            continue
        
        # Find best blend
        t_opt, obj_opt = find_best_blend(a, g)
        
        if t_opt > 0.005 and obj_opt < best_obj - 1e-13:
            a_new = (1 - t_opt) * a + t_opt * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new.copy()
                    stall = 0
                    if verbose and (it % 10 == 0 or obj_new < 1.505):
                        print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, top_m={top_m}, "

                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1
        
        it += 1
        
        # Periodic perturbation to escape stalls
        if stall >= perturb_period:
            rng_local = np.random.default_rng(hash((it, n)) % (2**32))

            noise = rng_local.normal(0, noise_scale, n)
            a_pert = best_a + noise
            a_pert = np.maximum(a_pert, 0)
            s_p = float(np.sum(a_pert))
            if s_p > 0.01:
                a = a_pert / s_p
                best_a = a.copy()
                best_obj = compute_obj(a)
            stall = 0
    
    return best_a


def coordinate_descent(a, time_budget):
    """Fast coordinate descent with adaptive candidates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while (time.monotonic() - start) < time_budget:
        sweep_improved = False
        
        for i in range(n):
            if (time.monotonic() - start) >= time_budget:
                break
            old = a[i]
            if old < 1e-12:
                continue
            
            best_val = old
            best_local = best_obj
            # Try fewer candidates for speed
            for mult in [0.0, 0.4, 0.7, 1.3, 1.8, 2.5]:
                a[i] = old * mult
                s_new = float(np.sum(a))
                if s_new < 0.01:
                    continue
                obj_try = compute_obj(a)
                if obj_try < best_local:
                    best_local = obj_try
                    best_val = old * mult
            
            if best_val != old:
                a[i] = best_val
                sweep_improved = True
                if best_local < best_obj:
                    best_obj = best_local
                    best_a = a.copy()
            else:
                a[i] = old
        
        # Normalize at end of sweep
        s = float(np.sum(a))
        if s > 0.01:
            a = a / s
            obj_norm = compute_obj(a)
            if obj_norm < best_obj:
                best_obj = obj_norm
                best_a = a.copy()
        
        if not sweep_improved:
            break
    
    return best_a


def diverse_init(n, mode):
    """Generate diverse non-negative initializations (normalized to sum=1)."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = 1.0 / (0.03 + 4.0 * (2 * x - 1) ** 2)
    elif mode == 1:
        a = 1.0 / np.sqrt(np.maximum(1e-8, 1 - (2 * x - 1) ** 2))
    elif mode == 2:
        a = (2 * x - 1) ** 6 + 0.03
    elif mode == 3:
        a = np.exp(6.0 * (2 * x - 1) ** 2)
    elif mode == 4:
        a = 3.0 * np.abs(x - 0.5) ** 0.7 + 0.03
    elif mode == 5:
        d1 = np.abs(x - 0.08)
        d2 = np.abs(x - 0.92)
        a = np.exp(-80 * d1**2) + np.exp(-80 * d2**2) + 0.005
    elif mode == 6:
        a = 0.3 + 0.7 * np.abs(np.cos(np.pi * x)) ** 1.5
    elif mode == 7:
        a = (1 + np.cos(2 * np.pi * x)) ** 0.6 + 0.02
    elif mode == 8:
        # Step-like: high at edges, low in middle
        thresh = 0.2
        a = np.where((x < thresh) | (x > 1 - thresh), 1.0, 0.05)
    elif mode == 9:
        # Power law with sharp edges
        d = np.maximum(np.abs(x - 0.5), 1.0 / (2 * n))
        a = d ** (-1.8) + 0.01
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 0:
        a = a / s
    return a


def lse_gradient_descent(a, time_budget, T_start=5.0, T_end=50.0):

        conv = compute_conv(a)
        cmax = np.max(conv)
        exp_vals = np.exp(T * (conv - cmax))
        w = exp_vals / exp_vals.sum()
        
        # Gradient of sum_k w[k] * conv_k(a,a)
        # Using FFT: grad_i = sum_k w[k] * (a[k-i] + a[i-k]) (valid terms)
        grad = np.zeros(n)
        w_padded = np.zeros(2 * n - 1)
        w_padded[:len(w)] = w
        # grad[i] = sum_k w[k]*a[k-i] + sum_k w[k]*a[i-k] (valid indices)
        # First term: cross-correlation of w and a
        for i in range(n):
            # sum_k w[k] * a[k-i] where 0<=k-i<n, 0<=k<2n-1
            lo1 = max(0, i)
            hi1 = min(2 * n - 2, i + n - 1)
            if lo1 <= hi1:
                grad[i] += np.dot(w_padded[lo1:hi1+1], a[lo1-i:hi1-i+1])
            # sum_k w[k] * a[i-k] where 0<=i-k<n
            lo2 = max(0, i - n + 1)
            hi2 = min(2 * n - 2, i)
            if lo2 <= hi2:
                grad[i] += np.dot(w_padded[lo2:hi2+1], a[i-hi2:i-lo2+1][::-1])
        
        grad *= 2.0 * n
        
        # Projected gradient step with backtracking
        a_old = a.copy()
        obj_old = compute_obj(a)
        lr = 0.05 / n
        improved = False
        for _ in range(8):
            a_try = a_old - lr * grad
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 0.01:
                lr *= 0.5
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < obj_old - 1e-14:
                a = a_try
                improved = True
                break
            lr *= 0.5
        
        if not improved:
            # Try random perturbation
            rng_local = np.random.default_rng(it)
            a_try = a + rng_local.normal(0, 0.005, n)
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))

                a_try = a_try / s_try

                    a = a_try
                    improved = True

            obj_now = compute_obj(a)
            if obj_now < best_obj:

            best_score = obj_final
            best_seq = a_final.copy()
            print(f"Phase 6: obj={best_score:.8f} **")
    
    # === Phase 7: Final coordinate descent ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining - 5, 60.0)
        a_cd_final = coordinate_descent(best_seq, time_budget=alloc)
        obj_cd_final = compute_obj(a_cd_final)
        if obj_cd_final < best_score:
            best_score = obj_cd_final
            best_seq = a_cd_final.copy()
            print(f"Phase 7 Final CD: obj={best_score:.8f} **")
    
    # === Phase 8: One more LP if time ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining - 5, 40.0)
        a_last = lp_iteration(best_seq, time_budget=alloc, max_iters=20,
                             top_m_start=2 * len(best_seq) - 1,
                             top_m_end=2 * len(best_seq) - 1,
                             verbose=False, perturb_period=3)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"Phase 8 Last LP: obj={best_score:.8f} **")
    
    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
