import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.where(u * np.arange(1, n+1) > (cssv - 1))[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_direction(a, top_m=None, solver=cp.PDLP):
    """Solve LP to find descent direction."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    if top_m is None:
        top_m = 2*n - 1
    
    conv_a = compute_conv(a_norm)
    top_indices = np.argsort(conv_a)[::-1][:top_m]
    
    m = len(top_indices)
    # Build A matrix: A[k, i] = a_norm[k - i] if 0 <= k-i < n, else 0
    A = np.zeros((m, n))
    for idx, k in enumerate(top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[idx, lo:hi+1] = a_norm[k - hi:k - lo + 1][::-1]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for s in [solver, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv
        except Exception:
            continue
    return None

def lp_iterate(a, time_budget, top_m=500, verbose=True):
    """Iterative LP polishing."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        g = lp_direction(best_a, top_m=top_m)
        if g is None:
            break
        
        # Find best blend factor
        best_t = 0.0
        best_blend_obj = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1-t)*best_a + t*g
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:

            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_t = t
        
        if best_t > 0:
            a_new = (1-best_t)*best_a + best_t*g
            a_new = np.maximum(a_new, 0)

            obj_new = compute_obj(a_new)

    # d/d a[i] of sum_k w_k * conv[k]
    # conv[k] = sum_j a_j * a_{k-j}

        it += 1

        if stall >= 6:
            rng_local = np.random.default_rng(it * 104729 + 7)
            noise_scale = 0.001 * (1 + stall // 5)
            noise = rng_local.normal(0, noise_scale, n)
            a_pert = best_a + noise
            a_pert = np.maximum(a_pert, 0)
            sp = float(np.sum(a_pert))
            if sp > 0.01:
                best_a = a_pert / sp
                best_obj = compute_obj(best_a)
            stall = 0

    return best_a


def lse_smoothed_obj(a, T):
    """Log-sum-exp smoothed objective."""
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.minimum(T * (conv - cmax), 50.0)
    lse = cmax + float(np.log(np.sum(np.exp(shifted)))) / T
    return 2.0 * n * lse


def lse_gradient(a, T):
    """Gradient of 2n * LSE_T(conv(a,a)) for normalized a."""
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.minimum(T * (conv - cmax), 50.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / float(np.sum(exp_vals))

    grad = np.zeros(n)
    for i in range(n):
        lo1 = max(0, i)
        hi1 = min(2 * n - 2, i + n - 1)
        if lo1 <= hi1:
            grad[i] += float(np.dot(w[lo1:hi1 + 1], a[lo1 - i:hi1 - i + 1]))
        lo2 = max(0, i - n + 1)
        hi2 = min(2 * n - 2, i)
        if lo2 <= hi2:
            grad[i] += float(np.dot(w[lo2:hi2 + 1], a[i - hi2:i - lo2 + 1][::-1]))
    return 2.0 * n * grad


def lse_gd(a, time_budget, T_start=5.0, T_end=60.0, verbose=True):
    """LSE smoothed gradient descent with backtracking."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 3.0 / n
    it = 0
    best_it = 0
    patience = 800
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** frac

        grad = lse_gradient(a, T)
        grad = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        a_old = a.copy()
        obj_old = lse_smoothed_obj(a, T)
        t = lr
        improved = False
        for _ in range(15):
            a_try = a_old - t * grad
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 0.01:
                t *= 0.5
                continue
            a_try = a_try / s_try
            obj_try = lse_smoothed_obj(a_try, T)
            if obj_try < obj_old - 1e-12:
                a = a_try
                improved = True
                break
            t *= 0.5

        if not improved:
            for _ in range(5):
                noise = rng.normal(0, 0.02 / np.sqrt(n), n)
                a_try = np.maximum(a + noise, 0)
                s_try = float(np.sum(a_try))
                if s_try < 0.01:
                    continue
                a_try = a_try / s_try
                obj_try = lse_smoothed_obj(a_try, T)
                if obj_try < obj_old:
                    a = a_try
                    improved = True
                    break
            if not improved:
                break

        obj_true = compute_obj(a)
        if obj_true < best_obj:
            best_obj = obj_true
            best_a = a.copy()
            best_it = it

        it += 1
        if it - best_it > patience:
            a = best_a.copy()
            lr *= 0.5
            patience = max(300, patience // 2)
            best_it = it

        if verbose and it % 200 == 0:
            print(f"    LSE it{it}: obj={best_obj:.10f}, T={T:.1f}, lr={lr:.2e}, elapsed={time.monotonic()-start:.1f}s")

    return best_a


def coordinate_descent(a, time_budget, verbose=True):
    """Coordinate descent with multi-candidate search per coordinate."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 0.01:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    sweep = 0

    while time.monotonic() - start < time_budget:
        improved = False

        for i in range(n):
            if time.monotonic() - start >= time_budget:
                break

            old_val = a[i]
            if old_val < 1e-12:
                candidates = [1.0 / n, 3.0 / n, 10.0 / n]
            else:
                candidates = [0.0, old_val * 0.2, old_val * 0.5, old_val * 0.8,
                              old_val * 1.3, old_val * 1.8, old_val * 2.5]

            best_val = old_val
            best_local = best_obj

            for cv in candidates:
                if abs(cv - old_val) < 1e-15:
                    continue
                a[i] = cv
                s_new = float(np.sum(a))
                if s_new < 0.01:
                    continue
                obj_try = compute_obj(a)
                if obj_try < best_local:
                    best_local = obj_try
                    best_val = cv

            if best_val != old_val:
                a[i] = best_val
                improved = True
                if best_local < best_obj:
                    best_obj = best_local
                    best_a = a.copy()
            else:
                a[i] = old_val

        sweep += 1
        if verbose and sweep % 5 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, elapsed={time.monotonic()-start:.1f}s")

        if not improved:
            break

    return best_a


def u_shaped_init(n, sharpness):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.01 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def power_init(n, alpha):
    """Power-law edge concentration."""
    x = np.linspace(0, 1, n)
    d = np.abs(x - 0.5)
    a = np.maximum(d, 1.0 / (2 * n)) ** (-alpha) + 0.01
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def arcsine_init(n):
    """Arcsine distribution."""
    x = np.linspace(0.001, 0.999, n)
    a = 1.0 / np.sqrt(np.maximum(1e-10, 1 - (2 * x - 1) ** 2))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, mode):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = (2 * x - 1) ** 6 + 0.02
    elif mode == 1:
        a = np.exp(5.0 * (2 * x - 1) ** 2)
    elif mode == 2:
        a = 1.0 / np.cosh(4.0 * (2 * x - 1))
    elif mode == 3:
        a = 2.0 - (np.cos(np.pi * x) ** 4)
    elif mode == 4:
        a = np.abs(2 * x - 1) ** 0.7 + 0.05
    elif mode == 5:
        a = np.exp(-8.0 * (2 * x - 1) ** 4) + 0.5 * np.exp(4.0 * (2 * x - 1) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def try_n_value(best_seq, best_score, n_try, time_budget):
    """Try a different sequence length via interpolation + quick optimization."""
    n_old = len(best_seq)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_try)
    a_try = np.interp(x_new, x_old, best_seq)
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 0.01:
        return None, np.inf
    a_try = a_try / s

    # Quick LP
    t_lp = min(40.0, time_budget * 0.5)
    if t_lp > 5:
        a_try = lp_polish(a_try, time_budget=t_lp, verbose=False)

    # Quick CD
    t_cd = min(20.0, time_budget * 0.3)
    if t_cd > 3:
        a_try = coordinate_descent(a_try, time_budget=t_cd, verbose=False)

    obj_try = compute_obj(a_try)
    return a_try, obj_try


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization:
    1. Best initialization from multiple templates
    2. LP polishing (main workhorse with PDLP)
    3. LSE gradient descent for smoothing
    4. Additional LP rounds
    5. Coordinate descent for fine polishing
    6. n exploration
    7. Final combined polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # === Phase 0: Initialization (30s) ===
    print("=== Phase 0: Initialization ===")
    n_base = 800

    # Try parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.all(np.isfinite(parent_arr)):
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_base = parent.size
                print(f"  Parent (n={parent.size}): obj={obj:.10f}")
    except Exception:
        print("  Parent unavailable, using defaults")

    init_time = min(25.0, budget_s * 0.02)
    init_start = time.monotonic()

    inits = []
    inits.append(("u3", u_shaped_init(n_base, 3.0)))
    inits.append(("u5", u_shaped_init(n_base, 5.0)))
    inits.append(("u8", u_shaped_init(n_base, 8.0)))
    inits.append(("p1.5", power_init(n_base, 1.5)))
    inits.append(("p2.0", power_init(n_base, 2.0)))
    inits.append(("p2.5", power_init(n_base, 2.5)))
    inits.append(("p3.0", power_init(n_base, 3.0)))
    inits.append(("arc", arcsine_init(n_base)))
    for m in range(6):
        inits.append((f"div{m}", diverse_init(n_base, m)))

    for name, a_init in inits:
        if time.monotonic() - init_start > init_time:
            break
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Init '{name}': obj={obj_init:.10f} **")
        else:
            print(f"  Init '{name}': obj={obj_init:.10f}")

    print(f"  Best initial: obj={best_score:.10f}")

    if best_seq is None:
        best_seq = u_shaped_init(n_base, 5.0)
        best_score = compute_obj(best_seq)

    # === Phase 1: LP Polishing Main (400s) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 1: LP Polishing Main ===")
        alloc = min(350.0, remaining * 0.25)
        a_lp1 = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_lp1 = compute_obj(a_lp1)
        if obj_lp1 < best_score:
            best_score = obj_lp1
            best_seq = a_lp1.copy()
            print(f"  Phase 1 done: obj={best_score:.10f} **")
        else:
            print(f"  Phase 1 done: obj={obj_lp1:.10f}")

    # === Phase 2: LSE Gradient Descent (150s) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 2: LSE Gradient Descent ===")
        alloc = min(120.0, remaining * 0.12)
        a_lse = lse_gd(best_seq, time_budget=alloc, T_start=5.0, T_end=50.0, verbose=True)
        obj_lse = compute_obj(a_lse)
        if obj_lse < best_score:
            best_score = obj_lse
            best_seq = a_lse.copy()
            print(f"  Phase 2 done: obj={best_score:.10f} **")
        else:
            print(f"  Phase 2 done: obj={obj_lse:.10f}")

    # === Phase 3: LP Round 2 (200s) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 3: LP Round 2 ===")
        alloc = min(180.0, remaining * 0.15)
        a_lp2 = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase 3 done: obj={best_score:.10f} **")
        else:
            print(f"  Phase 3 done: obj={obj_lp2:.10f}")

    # === Phase 4: Coordinate Descent (250s) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 4: Coordinate Descent ===")
        alloc = min(200.0, remaining * 0.18)
        a_cd = coordinate_descent(best_seq, time_budget=alloc, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase 4 done: obj={best_score:.10f} **")
        else:
            print(f"  Phase 4 done: obj={obj_cd:.10f}")

    # === Phase 5: LP + CD cycle (150s) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 5: LP + CD Cycle ===")
        t_lp = min(80.0, remaining * 0.35)
        if t_lp > 10:
            a_lp3 = lp_polish(best_seq, time_budget=t_lp, verbose=False)
            obj_lp3 = compute_obj(a_lp3)
            if obj_lp3 < best_score:
                best_score = obj_lp3
                best_seq = a_lp3.copy()

        t_cd = min(60.0, remaining * 0.25)
        if t_cd > 10:
            a_cd2 = coordinate_descent(best_seq, time_budget=t_cd, verbose=False)
            obj_cd2 = compute_obj(a_cd2)
            if obj_cd2 < best_score:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                print(f"  Phase 5 CD: obj={best_score:.10f} **")

    # === Phase 6: n Exploration (200s) ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 6: n Exploration ===")
        n_cur = len(best_seq)
        n_options = []
        for delta in [-200, -100, 100, 200, 400]:
            n_try = n_cur + delta
            if 400 <= n_try <= 2500:
                n_options.append(n_try)
        n_options = sorted(set(n_options))[:4]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 80:
                break
            t_alloc = min(100.0, remaining * 0.4)
            a_try, obj_try = try_n_value(best_seq, best_score, n_try, t_alloc)
            if a_try is not None and obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.10f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 7: Final Intensive Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 7: Final Polish ===")
        # Quick LP
        t1 = min(remaining * 0.3, 60.0)
        if t1 > 10:
            a_f1 = lp_polish(best_seq, time_budget=t1, verbose=False)
            obj_f1 = compute_obj(a_f1)
            if obj_f1 < best_score:
                best_score = obj_f1
                best_seq = a_f1.copy()
                print(f"  Final LP: obj={best_score:.10f} **")

        # Quick CD
        t2 = min(remaining * 0.4, 50.0)
        if t2 > 10:
            a_f2 = coordinate_descent(best_seq, time_budget=t2, verbose=False)
            obj_f2 = compute_obj(a_f2)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final CD: obj={best_score:.10f} **")

        # Another LP pass
        t3 = min(remaining * 0.2, 30.0)
        if t3 > 8:
            a_f3 = lp_polish(best_seq, time_budget=t3, verbose=False)
            obj_f3 = compute_obj(a_f3)
            if obj_f3 < best_score:
                best_score = obj_f3
                best_seq = a_f3.copy()
                print(f"  Final LP2: obj={best_score:.10f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s")
    print(f"{'='*50}")
    return result
