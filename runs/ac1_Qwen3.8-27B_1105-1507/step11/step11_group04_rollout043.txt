def multiplicative_descent(a, time_budget, seed=42, verbose=True):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Adaptive step size
    eta = 1.0 / (2 * np.sqrt(n))  # initial
    eta_min = 1e-6
    eta_decay = 0.999
    
    # Momentum in dual space
    v = np.zeros(n)  # dual momentum
    mu = 0.5  # momentum coefficient
    
    iter_count = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Find active set (entries close to max)
        tol = 1e-8 * M
        active = conv >= M - tol
        active_idx = np.where(active)[0]
        
        # Compute subgradient
        subgrad = np.zeros(n)
        for k in active_idx:
            # part1: a[k-j] for valid j
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_hi >= j_lo:
                subgrad[j_lo:j_hi+1] += a[k - np.arange(j_lo, j_hi+1)]
            # part2: a[j+k] for valid j
            j_hi2 = n - 1 - k
            if j_hi2 >= 0:
                subgrad[:j_hi2+1] += a[np.arange(j_hi2+1) + k]
        
        # Normalize subgradient for stable updates
        sg_max = np.max(np.abs(subgrad))
        if sg_max > 0:
            subgrad_normed = subgrad / sg_max
        else:
            break
        
        # Momentum update in dual space
        v = mu * v + (1 - mu) * subgrad_normed
        
        # Multiplicative update
        # a_new = a * exp(-eta * v) / Z
        log_update = -eta * v
        # Numerical stability
        log_update -= np.max(log_update)
        update = np.exp(log_update)
        Z = float(np.sum(a * update))
        if Z < 1e-300:
            eta *= 0.5
            continue
        a_new = a * update / Z
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        # Evaluate
        obj_new = compute_obj_norm(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        eta *= eta_decay
        iter_count += 1
        
        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD iter {iter_count}: obj={best_obj:.10f}, eta={eta:.6f}, elapsed={elapsed:.1f}s")

            a = np.maximum(a, 0)
            a = a / a.sum()
            eta = 1.0 / (2 * np.sqrt(n))
            stall = 0
    
    return best_a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based smoothed multiplicative descent with Nesterov momentum,
adaptive tau annealing, subgradient polishing, and coordinate descent.
Key innovations: LSE smoothing for dense gradient signal, multiplicative
updates for natural non-negativity, and momentum in log-space for fast convergence.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_subgrad_direct(a, conv, tol=1e-10):
    """
    Compute subgradient of max(conv) w.r.t. a using direct summation.
    subgrad[j] = 2 * sum_{k in S} a[k-j] where S = {k: conv[k] ~ max}
    Returns (subgrad, M, active_indices).
    """
    n = len(a)
    M = float(np.max(conv))
    tol_abs = max(tol * M, 1e-12)
    active_indices = np.where(conv >= M - tol_abs)[0]
    subgrad = np.zeros(n)
    for k in active_indices:
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            subgrad[j_lo:j_hi + 1] += a[k - np.arange(j_lo, j_hi + 1)]
    subgrad *= 2.0
    return subgrad, M, active_indices


def compute_smoothed_subgrad_fft(a, conv, tau):
    """
    Compute subgradient of LSE(conv/tau)/tau w.r.t. a via FFT.
    This gives a dense gradient signal (all entries of conv contribute).
    Returns gradient vector of length n.
    """
    n = len(a)
    # Softmax weights
    conv_scaled = conv / tau
    conv_scaled = conv_scaled - np.max(conv_scaled)
    weights = np.exp(conv_scaled)
    weights /= np.sum(weights)

    # subgrad[j] = (2/tau) * sum_k weights[k] * a[k-j]
    # Use FFT: this is 2/tau * cross_correlation(a, weights)
    size = 1
    while size < 2 * n:
        size <<= 1

    fa = rfft(a, size)
    fw = rfft(weights, size)

    # cross_correlation: c[j] = sum_k a[k] * weights[k-j]
    # FFT: conj(fa) * fw gives cross-correlation of a and weights
    # But we want sum_k weights[k] * a[k-j]
    # = sum_k a[k-j] * weights[k]
    # Let m = k-j: sum_m a[m] * weights[m+j]
    # = cross_corr(a, weights)[-j] or equivalently corr of weights and a
    #
    # Actually: sum_k weights[k] * a[k-j]
    # = sum_m a[m] * weights[m+j]  (substituting m = k-j)
    # This is the cross-correlation of a and weights at lag -j
    # = (a * weights_reversed)[j + (2n-2)] in convolution terminology
    #
    # Simpler: use the fact that
    # sum_k weights[k] * a[k-j] = (weights * a_reflected)[j]
    # where a_reflected[m] = a[n-1-m]... this is getting circular.
    #
    # Let me just use: the DFT of c[j] = sum_k w[k] a[k-j] is:
    # C(f) = W(f) * conj(A(f))  (cross-correlation theorem)
    # So c = ifft(fw * conj(fa))

    c = irfft(fw * np.conj(fa), size)[:2 * n - 1]
    # c[j] should be sum_k weights[k] * a[k-j] for j = 0, ..., 2n-2
    # We need j = 0 to n-1

    subgrad = (2.0 / tau) * c[:n]
    return subgrad


def smoothed_multiplicative_descent(a, time_budget, seed=42, verbose=True):
    """
    Main optimizer: multiplicative descent with Nesterov momentum in log-space,
    LSE smoothing with adaptive annealing.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Hyperparameters
    tau_initial = 0.05
    tau_min = 1e-5
    tau_decay = 0.97
    momentum = 0.9
    eta_initial = 0.8
    eta_min = 0.01
    eta_adapt_factor = 0.7

    # State
    tau = tau_initial
    eta = eta_initial
    log_v = np.zeros(n)  # momentum in log-space
    stall = 0
    iter_count = 0

    # Precompute FFT size
    fft_size = 1
    while fft_size < 2 * n:
        fft_size <<= 1

    while time.monotonic() - start < time_budget:
        t_iter = time.monotonic()
        conv = compute_conv(a)
        M = float(np.max(conv))

        # Compute smoothed subgradient via FFT
        subgrad = compute_smoothed_subgrad_fft(a, conv, tau)

        # Momentum update in log-space
        log_v = momentum * log_v + (1.0 - momentum) * subgrad

        # Multiplicative update in log-space
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * log_v

        # Normalize (equivalent to projecting onto simplex in exp-space)
        log_a_new -= np.max(log_a_new)  # numerical stability
        a_new = np.exp(log_a_new)
        s_new = float(np.sum(a_new))
        if s_new < 1e-300:
            eta *= 0.5
            continue
        a_new = a_new / s_new
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()

        # Evaluate
        obj_new = compute_obj_norm(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive step size
        if stall > 0 and stall % 5 == 0:
            eta *= eta_adapt_factor
        elif stall == 0 and eta < eta_initial:
            eta = min(eta * 1.05, eta_initial)

        # Anneal tau
        tau = max(tau * tau_decay, tau_min)

        a = a_new
        iter_count += 1

        # Periodic reporting
        if verbose and iter_count % 3000 == 0:
            elapsed = time.monotonic() - start
            print(f"    SMD iter {iter_count}: obj={best_obj:.10f}, "
                  f"tau={tau:.6f}, eta={eta:.6f}, elapsed={elapsed:.1f}s")

        # Stall handling: reset and perturb
        if stall > 80:
            log_v = np.zeros(n)
            pert = rng.standard_normal(n) * 0.02
            a = best_a * np.exp(pert)
            a = np.maximum(a, 0)
            a = a / a.sum()
            eta = eta_initial
            tau = max(tau * 2, 0.01)
            stall = 0
            iter_count = 0

        # Time check per iteration (safety)
        if time.monotonic() - t_iter > 1.0:
            break

    if verbose:
        print(f"    SMD done: {iter_count} iters, best={best_obj:.10f}")
    return best_a


def subgradient_polish(a, time_budget, seed=42, verbose=False):
    """
    Projected subgradient method on the exact (non-smooth) objective.
    Uses adaptive step sizes and periodic resets.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    eta = 0.05
    eta_decay = 0.998
    stall = 0
    iter_count = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        subgrad, M, active_idx = compute_subgrad_direct(a, conv, tol=1e-10)

        # The gradient of the objective 2n*max_conv is 2n*subgrad
        # For the constrained problem (sum=1), we need to subtract the mean
        # to stay on the affine constraint
        direction = subgrad - np.mean(subgrad)
        direction_norm = np.linalg.norm(direction)
        if direction_norm < 1e-15:
            break

        direction = direction / direction_norm

        # Take step
        a_try = a + eta * (-direction)
        # Project onto simplex
        a_try = project_simplex(a_try)

        obj_try = compute_obj_norm(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            stall = 0
        else:
            stall += 1
            if stall % 10 == 0:
                eta *= 0.8

        eta = max(eta * eta_decay, 0.001)
        iter_count += 1

        if verbose and iter_count % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    SG iter {iter_count}: obj={best_obj:.10f}, "
                  f"eta={eta:.6f}, elapsed={elapsed:.1f}s")

        if stall > 200:
            # Perturb
            pert = rng.standard_normal(n) * 0.01
            a = best_a * np.exp(pert)
            a = np.maximum(a, 0)
            a = a / a.sum()
            eta = 0.05
            stall = 0

    if verbose:
        print(f"    SG done: {iter_count} iters, best={best_obj:.10f}")
    return best_a


def coordinate_descent_polish(a, time_budget, seed=42, verbose=False):
    """
    Coordinate descent on normalized sequence.
    Uses the O(n) per-coordinate update structure.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    sweep = 0
    no_improve = 0
    max_no_improve = 3
    mean_val = 1.0 / n

    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            a_j = a[j]
            k_start = max(0, j)
            k_end = min(j + n - 1, 2 * n - 2)

            # Unaffected max
            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))

            # Try a few candidate values
            candidates = []
            if a_j > 1e-12:
                candidates = [a_j * 0.5, a_j * 0.8, a_j * 1.2, a_j * 1.5, a_j * 2.0,
                              mean_val * 0.5, mean_val, mean_val * 2, mean_val * 4]
            else:
                candidates = [mean_val * 0.5, mean_val, mean_val * 2]
            candidates.append(0.0)
            candidates = [max(0.0, min(0.5, c)) for c in candidates]
            candidates = list(set(round(c, 12) for c in candidates))

            best_trial_obj = best_obj
            best_trial_val = a_j

            for val in candidates:
                delta = val - a_j
                if abs(delta) < 1e-15:
                    continue

                S_new = 1.0 + delta
                if S_new <= 1e-10:
                    continue

                # Compute new max of affected conv entries
                # conv[k] changes by 2*delta*a[k-j] for k in [j, j+n-1]
                # except at k=2j where change is delta^2 + 2*delta*a[j] = (val^2 - a_j^2)
                k_vals = np.arange(k_start, k_end + 1)
                # a[k-j] for k in k_vals
                a_vals = a[k_vals - j]
                new_conv_affected = conv[k_start:k_end + 1] + 2.0 * delta * a_vals
                # Fix k=2j
                if 2 * j >= k_start and 2 * j <= k_end:
                    idx_2j = 2 * j - k_start
                    new_conv_affected[idx_2j] = conv[2 * j] + (val * val - a_j * a_j)

                max_conv_new = max(max_unaffected, float(np.max(new_conv_affected)))
                obj_trial = 2.0 * n * max_conv_new / (S_new * S_new)

                if obj_trial < best_trial_obj - 1e-13:
                    best_trial_obj = obj_trial
                    best_trial_val = val

            if best_trial_val != a_j:
                delta = best_trial_val - a_j
                a[j] = best_trial_val
                S_new = 1.0 + delta

                # Update conv in-place
                k_vals = np.arange(k_start, k_end + 1)
                a_vals = a[k_vals - j]
                # Use old a[j] for the update (we already changed a[j])
                # delta * old_a[k-j] for k != 2j, and (val^2 - a_j^2) for k = 2j
                conv[k_start:k_end + 1] += 2.0 * delta * a_vals
                if 2 * j >= k_start and 2 * j <= k_end:
                    idx_2j = 2 * j - k_start
                    # Already added 2*delta*a[2j-j] = 2*delta*a_j, need to fix
                    conv[2 * j] = conv[2 * j] - 2.0 * delta * a_j + (best_trial_val * best_trial_val - a_j * a_j)

                # Normalize
                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S

                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")

    return best_a


def solve_lp_quick(a, warm_start=None, max_iters=50):
    """
    Quick LP solve: min M s.t. conv(a_norm, g) <= M, sum(g)=1, g>=0.
    Uses cvxpy with time limit.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))

    # Build convolution matrix (sparse)
    # A[k, j] = a_norm[k-j] if 0 <= k-j < n else 0
    # A has shape (2n-1, n)
    # For efficiency, use the structure
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a_norm

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass

    for solver in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0)
                s_g = float(np.sum(g_val))
                if s_g > 1e-10:
                    return g_val / s_g, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate_quick(a, time_budget, seed=42, verbose=False):

        if g_lp is None:
            break
        prev_g = g_lp
        iter_count += 1

        # Line search
        best_local_obj = best_obj
        best_local_a = a.copy()
        for t in [1.0, 0.7, 0.5, 0.3, 0.15]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)

    return best_a


def fourier_perturb(a, rng, strength=0.015):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 10):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

        a2 = subgradient_polish(best_a, time_budget=p2_budget, seed=seed + 100, verbose=False)
        obj2 = compute_obj_norm(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            print(f"  P2: {best_obj:.10f} **")
        else:
            print(f"  P2: {obj2:.10f}")

    # === Phase 3: Coordinate Descent (15% time) ===
    print("\n--- Phase 3: Coordinate Descent ---")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = min(remaining * 0.4, 300.0)
        a3 = coordinate_descent_polish(best_a, time_budget=p3_budget, seed=seed + 200, verbose=True)
        obj3 = compute_obj_norm(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            print(f"  P3: {best_obj:.10f} **")
        else:
            print(f"  P3: {obj3:.10f}")

    # === Phase 4: LP Alternating (10% time) ===
    print("\n--- Phase 4: LP Alternating ---")
    remaining = deadline - time.monotonic
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p4_budget = min(remaining * 0.3, 150.0)
        a4 = lp_alternate_quick(best_a, time_budget=p4_budget, seed=seed + 300, verbose=False)
        obj4 = compute_obj_norm(a4)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()
            print(f"  P4: {best_obj:.10f} **")
        else:
            print(f"  P4: {obj4:.10f}")

    # === Phase 5: SMD from perturbed state (5% time) ===
    print("\n--- Phase 5: SMD Restart ---")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        a_pert = fourier_perturb(best_a, rng, strength=0.01)
        p5_budget = min(remaining * 0.4, 100.0)
        a5 = smoothed_multiplicative_descent(a_pert, time_budget=p5_budget,
                                              seed=seed + 400, verbose=False)
        obj5 = compute_obj_norm(a5)
        if obj5 < best_obj:
            best_obj = obj5
            best_a = a5.copy()
            print(f"  P5: {best_obj:.10f} **")
        else:
            print(f"  P5: {obj5:.10f}")

    # === Phase 6: Length exploration ===
    print("\n--- Phase 6: Length Exploration ---")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    n_options = [n_best - 50, n_best - 25, n_best + 25, n_best + 50, n_best + 100]
    n_options = [n for n in n_options if 500 <= n <= 1500 and n != n_best]

    for n_try in n_options[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, n_try)
        a_try = np.interp(x_new, x_old, best_a)
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        alloc = min(remaining * 0.25, 120.0)
        if alloc > 40:
            a_try = smoothed_multiplicative_descent(a_try, time_budget=alloc * 0.6,
                                                     seed=n_try * 7, verbose=False)
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_try = coordinate_descent_polish(a_try,
                                                   time_budget=min(remaining * 0.3, 40.0),
                                                   seed=n_try * 13, verbose=False)

        obj_try = compute_obj_norm(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            print(f"  ** n={n_try}: {best_obj:.10f} **")
        else:
            print(f"  n={n_try}: {obj_try:.10f}")

    # === Phase 7: Final CD polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n--- Phase 7: Final Polish ---")
        a_fin = coordinate_descent_polish(best_a, time_budget=min(remaining * 0.7, 60.0),
                                          seed=seed + 42424, verbose=False)
        obj_fin = compute_obj_norm(a_fin)
        if obj_fin < best_obj:
            best_obj = obj_fin
            best_a = a_fin.copy()
            print(f"  Final: {best_obj:.10f} **")

    # Final validation and return
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
