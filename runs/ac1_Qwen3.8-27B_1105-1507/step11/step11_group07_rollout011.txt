    n = len(a)

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    valid = u * rho > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def make_shift_diff(a, i, j):
    """Compute d[k] = a[k-i] - a[k-j] for k in [0, 2n-2], with a[x]=0 for x out of [0,n-1]."""
