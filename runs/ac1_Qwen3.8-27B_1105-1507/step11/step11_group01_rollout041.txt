"""
Optimizes step function coefficients using a combination of:
1. Multiscale FISTA with adaptive softmax smoothing (coarse-to-fine)
2. Frank-Wolfe with away steps for sparse structure discovery
3. Exact subgradient line search for exploitation
4. Exact coordinate descent for final polishing

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_obj(a):
    """Compute the objective 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def compute_subgradient(a, conv=None):
    """Compute exact subgradient of max_k Q_k(a) at point a."""
    if conv is None:
        conv = compute_conv(a)
    max_val = float(np.max(conv))
    # Use near-max set for stability
    thresh = max_val * (1.0 - 1e-12) - 1e-15
    K = np.where(conv >= thresh)[0]
    n = len(a)
    # g[i] = (2/|K|) * sum_{k in K} a[k-i] * 1{0 <= k-i < n}
    # Use FFT: g = 2 * (1_K_normalized * a) [linear conv, take appropriate slice]
    h = np.zeros(len(conv))
    h[K] = 1.0 / len(K)
    # Linear convolution of h (length 2n-1) with a (length n) -> length 3n-2
    # But we want sum_k h[k] * a[k-i] for i=0..n-1
    # This is cross-correlation: (h ∘ a)[i] = sum_k h[k] * a[k-i]
    # = linear conv of h with flip(a), taken at appropriate positions
    # Actually: sum_k h[k] * a[k-i] where 0<=k-i<n means 0<=a-index<n
    # Let's just do it directly with FFT
    size = 1
    while size < len(conv) + n - 1:
        size <<= 1
    H = rfft(h, size)
    A = rfft(a, size)
    # We want c[i] = sum_k h[k] * a[k-i] for i = 0..n-1
    # This is (h * flip_a)[n-1+i] where flip_a[j] = a[n-1-j]
    flip_a = a[::-1]
    FA = rfft(flip_a, size)
    result = irfft(H * FA, size)
    # (h * flip_a)[m] = sum_k h[k] * flip_a[m-k] = sum_k h[k] * a[n-1-(m-k)]
    # = sum_k h[k] * a[n-1-m+k]
    # We want sum_k h[k] * a[k-i], so set n-1-m+k = k-i => n-1-m = -i => m = n-1+i
    g = 2.0 * result[n - 1:n - 1 + n]
    return g, K, max_val


def fista_smoothed(a, tau_schedule, time_budget, seed=42):
    """
    FISTA with adaptive softmax smoothing.
    tau_schedule: list of (tau, n_iterations) tuples.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    y = a.copy()
    t_kk = 1.0
    lr = 0.4  # Initial learning rate

    for tau, n_iters in tau_schedule:
        if time.monotonic() - start >= time_budget:
            break
        # Determine step size: for smoothed f with Lipschitz ~ 2*tau*max|a|^2 + base_L
        # Use backtracking
        for it in range(n_iters):
            elapsed = time.monotonic() - start
            if elapsed >= time_budget:
                break

            # Compute smoothed gradient at y
            conv_y = compute_conv(y)
            max_c = float(np.max(conv_y))
            exp_vals = np.exp(tau * (conv_y - max_c))
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv_y))
                p[np.argmax(conv_y)] = 1.0
            else:
                p = exp_vals / sum_exp

            # Gradient = 2 * (p * y) [convolution]
            size = 1
            while size < len(conv_y) + n - 1:
                size <<= 1
            P = rfft(p, size)
            Y = rfft(y, size)
            # grad[i] = 2 * sum_k p[k] * y[k-i]
            flip_y = y[::-1]
            FY = rfft(flip_y, size)
            g_full = irfft(P * FY, size)
            grad = 2.0 * g_full[n - 1:n - 1 + n]

            # FISTA update
            t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_kk * t_kk)) / 2.0
            y_new = a + ((t_kk - 1.0) / t_new) * (a - y)
            y_new = np.maximum(y_new, 0.0)
            s_y = float(np.sum(y_new))
            if s_y < 1e-10:
                y_new = a.copy()
                s_y = 1.0
            else:
                y_new = y_new / s_y

            # Gradient step with backtracking
            a_cand = proj_simplex(y_new - lr * grad)
            # Check if step is good (descent in smoothed objective)
            obj_cand = compute_obj(a_cand)
            if obj_cand > best_obj * 1.05:
                lr *= 0.5
                if lr < 0.01:
                    lr = 0.1
                    a_cand = a.copy()
            else:
                a = a_cand
                if obj_cand < best_obj:
                    best_obj = obj_cand
                    best_a = a.copy()

            y = y_new
            t_kk = t_new

    return best_a


def fw_away_steps(a, time_budget, seed=42):
    """
    Frank-Wolfe with away steps for the objective min max(Q(a)) on the simplex.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = elapsed / max(time_budget, 1e-10)

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        K = np.where(conv >= max_val * (1.0 - 1e-10))[0]

        # Subgradient
        g = np.zeros(n)
        for k in K:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                g[lo:hi + 1] += a[k - hi:k - lo + 1]
        g *= 2.0 / max(len(K), 1)

        # Towards step: e_{argmin g}
        i_towards = int(np.argmin(g))

        # Away step: e_{argmax g} (among coordinates with a[i] > 0)
        active = np.where(a > 1e-10)[0]
        if len(active) > 1:
            i_away = active[np.argmax(g[active])]
        else:
            i_away = -1

        # Line search
        best_t_obj = best_obj
        best_a_local = a.copy()
        gamma = 2.0 / (it + 2)  # Standard FW step size

        # Try towards step
        for t in [gamma, gamma * 0.5, gamma * 0.25, 0.1, 0.05]:
            a_try = (1.0 - t) * a + t * np.zeros(n)
            a_try[i_towards] += t
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_t_obj:
                    best_t_obj = obj_try
                    best_a_local = a_try.copy()

        # Try away step
        if i_away >= 0 and a[i_away] > 1e-10:
            denom = 1.0 - a[i_away]
            if denom > 1e-10:
                for t in [gamma, gamma * 0.5, gamma * 0.25]:
                    a_try = a.copy()
                    a_try[i_away] -= t * a[i_away]
                    a_try[i_towards] += t * a[i_away]
                    a_try = np.maximum(a_try, 0.0)
                    s_try = float(np.sum(a_try))
                    if s_try > 1e-10:
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_t_obj:
                            best_t_obj = obj_try
                            best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-15:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a

        it += 1

        # Periodic check
        if it % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"    FW-Away it{it}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def exact_subgrad_linesearch(a, time_budget, seed=42):
    """
    Subgradient method with Polyak-like step size and line search.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)  # Theoretical lower bound
    start = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)
    eta = 1.0 / math.sqrt(n)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed >= time_budget:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        K = np.where(conv >= max_val * (1.0 - 1e-10))[0]

        # Subgradient of max Q
        g = np.zeros(n)
        for k in K:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                g[lo:hi + 1] += a[k - hi:k - lo + 1]
        g *= 2.0 / max(len(K), 1)

        # Normalize gradient
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break

        # Polyak step size approximation
        gap = best_obj / (2.0 * n) - lower_bound / (2.0 * n)
        if gap > 1e-10:
            step_polyak = gap / (g_norm * g_norm)
            step = min(eta, max(0.01, step_polyak))
        else:
            step = eta / (1.0 + 0.01 * it)

        # Try several step sizes
        best_step_obj = best_obj
        best_step_a = a.copy()
        for t_factor in [1.0, 0.5, 0.25, 0.1, 2.0, 4.0]:
            t = step * t_factor
            a_try = proj_simplex(a - t * g)
            obj_try = compute_obj(a_try)
            if obj_try < best_step_obj:
                best_step_obj = obj_try
                best_step_a = a_try.copy()

        if best_step_obj < best_obj - 1e-15:
            best_obj = best_step_obj
            best_a = best_step_a.copy()
            a = best_a
            eta = min(eta * 1.01, 5.0 / math.sqrt(n))
        else:
            eta *= 0.8
            if eta < 0.01:
                # Restart with perturbation
                pert = rng.normal(0, 0.02, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                eta = 1.0 / math.sqrt(n)

        it += 1
        if it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    SubGrad it{it}: obj={best_obj:.8f}, eta={eta:.4f}, t={elapsed:.1f}s")

    return best_a


def coordinate_descent(a, time_budget=100.0, verbose=False, seed=42):
    """Exact coordinate descent with golden section search."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    prev_best = best_obj
    stall = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        sweep += 1

        # Periodic recompute
        if sweep % 3 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    a = a / S_check

        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask].copy()
            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            scale = max(v, S / n, 1e-6)
            candidates = [0.0, v * 0.5, v, v * 1.5, S / n, 2.0 * S / (3 * n)]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            def f_1d(x, v=v, S_other=S_other, c_k_lin=c_k_lin,
                     a_ki_lin=a_ki_lin, c_2i=c_2i):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section
            x_lo = max(0.0, best_x - scale * 0.3)
            x_hi = best_x + scale * 0.8
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-8
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)
            for _ in range(30):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            S += delta

            S_new = float(np.sum(a))
            if S_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if sweep % 10 == 0:
            if best_obj < prev_best - 1e-12:
                prev_best = best_obj
                stall = 0
            else:
                stall += 1
            if stall > 5:
                break
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def symmetrize(a):
    """Enforce palindromic symmetry: a[i] = a[n-1-i]."""
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    return a_sym


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 3:
        # Raised cosine window
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = a / a.sum()
    elif mode == 4:
        # Two bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Flat top with tapered edges
        edge = n // 6
        a = np.ones(n)
        taper = np.linspace(0, 1, edge)
        a[:edge] = taper
        a[-edge:] = taper[::-1]
        a = a / a.sum()
    elif mode == 6:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** 2
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

            print(f"  {label}: obj={best_score:.8f} **")
        else:
            print(f"  {label}: obj={obj:.8f}")
        return obj

    # === Phase 1: Multiscale FISTA (coarse to fine) ===
    print(f"\n=== Phase 1: Multiscale FISTA (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p1_budget = min(remaining * 0.3, 400.0)

        # Coarse: downsample to n/4, optimize, upsample
        n_coarse = max(100, n_best // 4)
        a_coarse = resample_sequence(best_seq, n_coarse)
        tau_sched_coarse = [(50, 200), (20, 300), (10, 400), (5, 500)]
        a_coarse = fista_smoothed(a_coarse, tau_sched_coarse,
                                  time_budget=p1_budget * 0.25, seed=seed)
        a_interp = resample_sequence(a_coarse, n_best)
        obj_coarse = try_update(a_interp, "Multiscale coarse->fine")

        # Fine: FISTA on full resolution
        tau_sched_fine = [(30, 300), (15, 400), (8, 500), (4, 600), (2, 800), (1, 1000)]
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_fista = fista_smoothed(best_seq, tau_sched_fine,
                                     time_budget=min(remaining * 0.5, 200.0),
                                     seed=seed + 10)
            try_update(a_fista, "FISTA fine")

    # === Phase 2: Frank-Wolfe with Away Steps ===
    print(f"\n=== Phase 2: FW with Away Steps ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p2_budget = min(remaining * 0.15, 200.0)
        a_fw = fw_away_steps(best_seq, time_budget=p2_budget, seed=seed + 100)
        try_update(a_fw, "FW-Away")
        # Follow up with CD
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_fw_cd = coordinate_descent(a_fw, time_budget=min(30, remaining * 0.3),
                                         seed=seed + 101)
            try_update(a_fw_cd, "FW-Away + CD")

    # === Phase 3: Exact Subgradient with Line Search ===
    print(f"\n=== Phase 3: Exact Subgradient Linesearch ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = min(remaining * 0.15, 180.0)
        a_sg = exact_subgrad_linesearch(best_seq, time_budget=p3_budget,
                                        seed=seed + 200)
        try_update(a_sg, "SubGrad")
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_sg_cd = coordinate_descent(a_sg, time_budget=min(30, remaining * 0.3),
                                         seed=seed + 201)
            try_update(a_sg_cd, "SubGrad + CD")

    # === Phase 4: Symmetry-enforced optimization ===
    print(f"\n=== Phase 4: Symmetry Enforced ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_sym = symmetrize(best_seq)
        a_sym = a_sym / a_sym.sum()
        # FISTA on symmetric sequence
        tau_sched_sym = [(20, 400), (10, 500), (5, 600)]
        a_sym = fista_smoothed(a_sym, tau_sched_sym,
                               time_budget=min(remaining * 0.2, 120.0),
                               seed=seed + 300)
        a_sym = symmetrize(a_sym)
        try_update(a_sym, "Symmetric FISTA")
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_sym_cd = coordinate_descent(a_sym, time_budget=min(25, remaining * 0.4),
                                          seed=seed + 301)
            try_update(a_sym_cd, "Symmetric CD")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Try n values: focus on range that might be optimal
    n_candidates = sorted(set([
        max(400, n_current - 200),
        max(400, n_current - 100),
        min(1500, n_current + 100),
        min(1500, n_current + 200),
        min(1500, n_current + 400),
        min(2000, n_current + 800),
        600, 700, 900, 1000, 1100, 1200
    ]))
    # Filter
    n_filtered = []
    for nc in n_candidates:
        if 400 <= nc <= 2000:
            if all(abs(nc - nf) > 80 for nf in n_filtered):
                n_filtered.append(nc)
    n_filtered = n_filtered[:7]
    print(f"  n candidates: {n_filtered}")

    for n_try in n_filtered:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick FISTA
        tau_try = [(20, 200), (10, 300), (5, 400)]
        alloc = min(remaining * 0.12, 50.0)
        if alloc > 15:
            a_try = fista_smoothed(a_try, tau_try, time_budget=alloc * 0.6,
                                   seed=n_try)
        # CD polish
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_try = coordinate_descent(a_try, time_budget=min(20, remaining * 0.3),
                                       seed=n_try + 500)

        try_update(a_try, f"n={n_try}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = 1 + r
        a_r = diverse_init(n_cur, rng, mode=mode)
        # Blend with best
        blend = 0.4 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.18, 80.0)
        if alloc > 20:
            tau_r = [(15, 200), (8, 300), (4, 400)]
            a_r = fista_smoothed(a_r, tau_r, time_budget=alloc * 0.5,
                                 seed=5000 + r * 100)
            a_r = coordinate_descent(a_r, time_budget=min(30, alloc * 0.4),
                                     seed=5000 + r * 100 + 1)
            try_update(a_r, f"Restart {r} (mode {mode})")

    # === Phase 7: Aggressive Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        for rnd in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            # CD with more time
            cd_alloc = min(remaining * 0.35, 100.0)
            if cd_alloc > 10:
                a_f = coordinate_descent(best_seq, time_budget=cd_alloc,
                                         verbose=(rnd == 0),
                                         seed=9000 + rnd * 7)
                try_update(a_f, f"Final CD {rnd}")

            # Subgradient burst
            remaining = deadline - time.monotonic()
            if remaining > 15:
                sg_alloc = min(remaining * 0.2, 50.0)
                if sg_alloc > 10:
                    a_f2 = exact_subgrad_linesearch(best_seq, time_budget=sg_alloc,
                                                    seed=9500 + rnd * 7)
                    a_f2 = coordinate_descent(a_f2, time_budget=min(15,
                                            deadline - time.monotonic() - 5),
                                              seed=9500 + rnd * 7 + 1)
                    try_update(a_f2, f"Final SG {rnd}")

    # === Final Validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = 2.0

    best_seq = np.maximum(best_seq, 0.0)
    result = [float(min(1000.0, x)) for x in best_seq]

    # Validate
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
        print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, "
              f"eval={val:.8f}, time={time.monotonic()-start_time:.1f}s ===")
    else:
        # Fallback
        result = [1.0] * 100
        best_score = 2.0
        print(f"\n=== FALLBACK: n=100, score=2.0 ===")

    return result
