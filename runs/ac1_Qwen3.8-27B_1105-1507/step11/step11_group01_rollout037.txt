"""
Optimizes step function coefficients using L-BFGS on smooth approximations,
coordinate descent, and multi-n exploration. Key innovation: exponential
surrogate optimization with temperature annealing for fast convergence to
high-quality local minima, combined with exact coordinate descent for
final polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def objective_normalized(a):
    """Compute 2*n*max(conv)/S^2 for normalized evaluation."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (S * S)


def smooth_obj_grad(x, t, n):
    """
    Smooth objective (exponential surrogate of 2n*max(conv)/S^2) and its gradient.
    x: sequence values (raw, not normalized), non-negative.
    t: temperature (larger = sharper approximation to max).
    Returns (objective_value, gradient_array).
    """
    a = np.maximum(x, 0.0)
    S = float(np.sum(a))
    if S < 1e-10:
        return 1e10, np.zeros(n)
    
    conv = compute_conv(a)
    vals = t * conv / (S * S)
    max_vals = float(np.max(vals))
    exp_vals = np.exp(vals - max_vals)
    sum_exp = float(np.sum(exp_vals))
    
    obj = (2.0 * n / t) * (max_vals + math.log(sum_exp))
    
    p = exp_vals / sum_exp
    
    # Compute c[i] = sum_j a[j] * p[j+i] via FFT
    # c[i] = conv(a, p_flipped)[2n-2-i]
    N = 1
    while N < 2 * (2 * n - 1):
        N <<= 1
    fa = fft(a, N)
    fp = fft(p[::-1], N)
    conv_ap = ifft(fa * fp).real[:2 * n - 1]
    c = conv_ap[n - 1:2 * n - 1][::-1]
    
    e_conv = float(np.dot(p, conv))
    
    grad = (4.0 * n / (S * S)) * (c - e_conv / S)
    
    return obj, grad


def run_lbfgs(a, t, maxiter=500):
    """Run L-BFGS-B on smooth objective. Returns (result_array, n_iterations)."""
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    
    def fun(x):
        val, _ = smooth_obj_grad(x, t, n)
        return val
    
    def grad(x):
        _, g = smooth_obj_grad(x, t, n)
        return g
    
    try:
        result = minimize(fun, a, jac=grad, method='L-BFGS-B',
                          bounds=[(0.0, None)] * n,
                          options={'maxiter': maxiter, 'ftol': 1e-14,
                                   'gtol': 1e-10, 'maxls': 40})
        return np.maximum(result.x, 0.0), result.nit
    except Exception:
        return a, 0


def coordinate_descent(a, time_budget=100.0, seed=42):
    """Exact coordinate descent with golden section search."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    
    start = time.monotonic()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    rng = np.random.default_rng(seed)
    
    stall = 0
    prev_best = best_obj
    
    for sweep in range(1000):
        if time.monotonic() - start >= time_budget:
            break
        
        order = np.random.default_rng(seed + sweep).permutation(n)
        S = float(np.sum(a))
        
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            
            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue
            
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            
            a_ki = np.zeros(len(ks))
            for idx, k in enumerate(ks):
                j = k - i
                if 0 <= j < n:
                    a_ki[idx] = a[j]
            
            c_k = conv[ks].copy()
            quad_mask = (ks == 2 * i)
            
            def _f1d(x, v=v, S_other=S_other, a_ki=a_ki, c_k=c_k,
                     quad_mask=quad_mask, n=n, conv=conv):
                new_c = c_k + 2.0 * (x - v) * a_ki
                if np.any(quad_mask):
                    new_c = new_c.copy()
                    new_c[quad_mask] += (x - v) ** 2
                new_S = S_other + x
                if new_S < 1e-10:
                    return 1e30
                return 2.0 * n * float(np.max(new_c)) / (new_S * new_S)
            
            # Quick candidate evaluation
            scale = max(v, S / n, 1e-8)
            candidates = [0.0, v * 0.5, v, v * 1.5, v * 2.0, S / n, 2.0 * S / n]
            candidates = sorted(set(max(0.0, c) for c in candidates))
            
            best_x = v
            best_f = _f1d(v)
            for x in candidates:
                fx = _f1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x
            
            # Golden section
            x_lo = max(0.0, best_x - scale * 0.5)
            x_hi = best_x + scale
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-10
            
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = _f1d(c_pt)
            fd = _f1d(d_pt)
            
            for _ in range(40):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = _f1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = _f1d(d_pt)
            
            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            
            if abs(x_opt - v) > 1e-14:
                delta = x_opt - v
                a[i] = x_opt
                S += delta
                conv[ks] += 2.0 * delta * a_ki
                if np.any(quad_mask):
                    conv[2 * i] += delta * delta
            
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj_check = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj_check < best_obj:
                    best_obj = obj_check
                    best_a = a.copy()
        
        # Stall detection
        if best_obj < prev_best - 1e-12:
            prev_best = best_obj
            stall = 0
        else:
            stall += 1
            if stall > 3:
                break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0.0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x * x + 1e-8)
        a = a / a.sum()

        x = np.linspace(0, 1, n)

        for f, p in zip(freqs, phases):
