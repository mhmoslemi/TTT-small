   import numpy as np
   from scipy.fft import rfft, irfft
   import time

   def compute_conv(a):
       """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
       n = len(a)
       size = 1
       while size < 2 * n - 1:
           size <<= 1
       fa = rfft(a, size)
       return irfft(fa * fa, size)[:2 * n - 1]

   def lse_gradient(a, T):
       """Compute gradient of LSE smoothed objective."""
       n = len(a)
       conv = compute_conv(a)  # length 2n-1
       cmax = np.max(conv)
       exp_vals = np.exp(T * (conv - cmax))
       w = exp_vals / exp_vals.sum()  # length 2n-1
       
       # p[i] = sum_j a[j] * w[i+j] for i=0,...,n-1
       # = conv(reverse(a), w)[i+n-1]
       ar = a[::-1]  # reverse
       # convolution of ar (length n) and w (length 2n-1) has length 3n-2
       size = 1
       while size < 3 * n - 2:
           size <<= 1
       fa = rfft(ar, size)
       fw = rfft(w, size)
       p_full = irfft(fa * fw, size)[:3*n-2]
       p = p_full[n-1:2*n-1]  # p[i] = p_full[i+n-1] for i=0,...,n-1
       
       return 2.0 * p

   def lse_objective(a, T):
       """LSE smoothed objective value."""
       n = len(a)
       conv = compute_conv(a)
       cmax = np.max(conv)
       lse = cmax + np.log(np.sum(np.exp(T * (conv - cmax)))) / T
       return 2.0 * n * lse  # for normalized a (sum=1), this is 2n * LSE(conv)

       """True objective: 2n * max(conv) for normalized a."""
       n = len(a)
       conv = compute_conv(a)
       return 2.0 * n * float(np.max(conv))

   def project_simplex(v):
       """Project v onto {x >= 0, sum(x) = 1}."""
       n = len(v)
       u = np.sort(v)[::-1]
       cssv = np.cumsum(u)
       cond = u * np.arange(1, n + 1) > (cssv - 1)
       rho_indices = np.nonzero(cond)[0]
       if len(rho_indices) == 0:
           return np.ones(n) / n
       rho = rho_indices[-1]
       theta = (cssv[rho] - 1.0) / (rho + 1.0)
       proj = np.maximum(v - theta, 0.0)
       s = proj.sum()
       if s > 1e-15:
           return proj / s
       return np.ones(n) / n

Key insight: minimize max of self-convolution for non-negative sequences.
The objective 2n*max(conv)/(sum)^2 has lower bound 2n/(2n-1) ~ 1 for large n.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import sys


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def true_objective(a):
    """True objective: 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def lse_objective(a, T):
    """LSE smoothed objective for normalized a."""
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)
    shifted = T * (conv - cmax)
    # Numerically stable log-sum-exp
    max_shifted = np.max(shifted)
    lse = cmax + (max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))) / T
    return 2.0 * n * lse


def lse_gradient(a, T):
    """
    Compute gradient of 2n * LSE_T(conv(a,a)) for normalized a.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    conv = compute_conv(a)  # length 2n-1
    cmax = np.max(conv)
    shifted = T * (conv - cmax)
    max_shifted = np.max(shifted)
    exp_vals = np.exp(shifted - max_shifted)
    w = exp_vals / exp_vals.sum()  # length 2n-1

    # p[i] = sum_{j=0}^{n-1} a[j] * w[i+j] for i=0,...,n-1
    # = conv(reverse(a), w)[i + n - 1]
    ar = a[::-1]
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = rfft(ar, size)
    fw = rfft(w, size)
    p_full = irfft(fa * fw, size)[:3 * n - 2]
    p = p_full[n - 1:2 * n - 1]

    return 2.0 * p


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} using Euler's algorithm."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n


def lse_gradient_descent(a, time_budget, T_start=3.0, T_end=200.0, verbose=False):
    """
    LSE smoothed gradient descent with backtracking line search and annealing.
    Optimizes 2n * max(conv(a,a)) for normalized non-negative a.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = true_objective(a)
    start = time.monotonic()
    lr = 2.0 / n
    it = 0
    best_it = 0
    patience = max(300, int(time_budget * 2))

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(time_budget, 1e-10))
        T = T_start * (T_end / T_start) ** frac

        grad = lse_gradient(a, T)
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-14:
            break

        # Backtracking line search
        a_old = a.copy()
        obj_old = lse_objective(a_old, T)
        t = lr
        improved = False
        for _ in range(25):
            a_try = a_old - t * grad_centered
            a_try = project_simplex(a_try)
            obj_try = lse_objective(a_try, T)
            if obj_try < obj_old - 1e-16:
                a = a_try
                improved = True
                break
            t *= 0.4

        if not improved:
            # Try random perturbations
            for _ in range(5):
                perturb = np.random.randn(n) * 0.01 / n
                a_try = project_simplex(a_old + perturb)
                obj_try = lse_objective(a_try, T)
                if obj_try < obj_old:
                    a = a_try
                    improved = True
                    break
            if not improved:
                break

        obj_true = true_objective(a)
        if obj_true < best_obj:
            best_obj = obj_true
            best_a = a.copy()
            best_it = it

        it += 1
        if it - best_it > patience:
            a = best_a.copy()
            lr *= 0.4
            patience = max(50, patience // 2)
            best_it = it

        if verbose and it % 200 == 0:
            print(f"        LSE-GD it{it}: obj={best_obj:.8f}, T={T:.1f}, "
                  f"elapsed={time.monotonic() - start:.1f}s")

    return best_a


def build_lp_matrix(a_norm, top_m):
    """Build constraint matrix for truncated LP at top-m peaks."""
    n = len(a_norm)
    conv_a = compute_conv(a_norm)
    top_indices = np.argsort(conv_a)[::-1][:min(top_m, 2 * n - 1)]

    A = np.zeros((len(top_indices), n))
    for idx, k in enumerate(top_indices):
        k = int(k)
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                A[idx, m] = a_norm[j]
    return A


def solve_lp(a, top_m):
    """Solve truncated LP: min M s.t. (a*g)[k] <= M (top m), sum(g)=1, g>=0."""
    try:
        import cvxpy as cp
    except ImportError:
        return None

    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_lp_matrix(a_norm, top_m)

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers = [cp.ECOS, cp.SCS, cp.CLARABEL]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def lp_polish(a, time_budget, top_m_start=25, top_m_end=10, max_iters=40, verbose=False):
    """Iterative LP polishing with line search blending."""
    n = len(a)
    best_a = a.copy()
    best_obj = true_objective(a)
    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        frac = min(1.0, it / max(1, max_iters - 1))
        top_m = int(round(top_m_start + frac * (top_m_end - top_m_start)))
        top_m = max(5, min(top_m, 2 * n - 2))

        g_lp = solve_lp(best_a, top_m)
        if g_lp is None:
            break

        # Line search for optimal blend
        best_t = 0.0
        best_obj_try = best_obj
        for t in np.linspace(0.1, 1.0, 10):
            a_try = (1 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj_try = true_objective(a_try)
            if obj_try < best_obj_try:
                best_obj_try = obj_try
                best_t = t

        # Refine
        if best_t > 0:
            for dt in [0.05, -0.05, 0.02, -0.02, 0.1]:
                t_try = best_t + dt
                if 0 < t_try < 1.5:
                    a_try = (1 - t_try) * best_a + t_try * g_lp
                    a_try = np.maximum(a_try, 0)
                    s = float(np.sum(a_try))
                    if s < 0.01:
                        continue
                    a_try = a_try / s
                    obj_try = true_objective(a_try)
                    if obj_try < best_obj_try:
                        best_obj_try = obj_try
                        best_t = t_try

        if best_t > 0 and best_obj_try < best_obj - 1e-14:
            a_new = (1 - best_t) * best_a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = true_objective(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 5 == 0:
            print(f"        LP it{it}: obj={best_obj:.8f}, top_m={top_m}, "
                  f"t={best_t:.3f}, elapsed={time.monotonic() - start:.1f}s")

        if stall >= 3:
            noise = np.random.randn(n) * 0.003 * stall
            a_pert = best_a + noise
            a_pert = np.maximum(a_pert, 0)
            s_p = float(np.sum(a_pert))
            if s_p > 0.01:
                best_a = a_pert / s_p
                best_obj = true_objective(best_a)
            stall = 0

    return best_a


def init_u_shaped(n, sharpness):
    """U-shaped initialization: high at edges, low in middle."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.01 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_power(n, alpha):
    """Power-law edge concentration."""
    x = np.linspace(0, 1, n)
    d = np.abs(x - 0.5)
    a = np.maximum(d, 1.0 / (2 * n)) ** (-alpha) + 0.001
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_arcsine(n):
    """Arcsine distribution - concentrated at edges."""
    x = np.linspace(0.001, 0.999, n)
    a = 1.0 / np.sqrt(np.maximum(1e-10, 1 - (2 * x - 1) ** 2))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_cosine(n, power):
    """Cosine bump - concentrated in middle."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * (x - 0.5)) ** (2 * power)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_triangle(n, p):
    """Generalized triangle/bump."""
    x = np.linspace(0, 1, n)
    a = np.maximum(0, 1 - np.abs(2 * x - 1) ** p)
    a = a + 0.001
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_random(n, rng):
    """Random non-negative initialization."""
    a = rng.random(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_interpolated(a_old, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a_old)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a_old)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return np.ones(n_new) / n_new
    return a_new / s


def multi_peak_subgradient(a, top_k=5):
    """Compute subgradient at top-k peaks (averaged)."""
    n = len(a)
    conv = compute_conv(a)
    k_peaks = np.argsort(conv)[::-1][:min(top_k, 2 * n - 1)]
    peak_vals = conv[k_peaks]
    weights = peak_vals / (peak_vals.sum() + 1e-15)

    grad = np.zeros(n)
    for ki, w in zip(k_peaks, weights):
        ki = int(ki)
        # d/d a[i] of (a*a)[ki] = 2*a[ki-i] if 0<=ki-i<n
        valid = (ki - np.arange(n) >= 0) & (ki - np.arange(n) < n)
        idx = np.where(valid)[0]
        if len(idx) > 0:
            grad[idx] += w * 2.0 * a[ki - idx]
    return grad


def subgradient_descent(a, time_budget, top_k=5, verbose=False):
    """Subgradient descent with momentum on top-k peaks."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = true_objective(a)
    start = time.monotonic()
    lr = 0.5 / n
    momentum = np.zeros(n)
    beta = 0.7
    it = 0
    best_it = 0
    patience = 500

    while time.monotonic() - start < time_budget:
        grad = multi_peak_subgradient(a, top_k)
        grad_centered = grad - np.mean(grad)

        a_old = a.copy()
        obj_old = true_objective(a_old)
        t = lr
        improved = False
        for _ in range(15):
            momentum = beta * momentum - t * grad_centered
            a_try = project_simplex(a_old + momentum)
            obj_try = true_objective(a_try)
            if obj_try < obj_old - 1e-14:
                a = a_try
                improved = True
                break
            t *= 0.5

        if not improved:
            momentum *= 0.5
            lr *= 0.8
        else:
            lr = min(t * 1.2, 5.0 / n)

        obj_true = true_objective(a)
        if obj_true < best_obj:
            best_obj = obj_true
            best_a = a.copy()
            best_it = it

        it += 1
        if it - best_it > patience:
            a = best_a.copy()
            momentum = np.zeros(n)
            lr = 0.2 / n
            beta = 0.5
            patience = max(100, patience // 2)
            best_it = it

        if verbose and it % 500 == 0:
            print(f"        SubGrad it{it}: obj={best_obj:.8f}, lr={lr:.6f}, "
                  f"elapsed={time.monotonic() - start:.1f}s")

    return best_a


def coordinate_descent(a, time_budget, max_sweeps=2, verbose=False):
    """Coordinate descent with local search on each element."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = true_objective(a)
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start >= time_budget:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - start >= time_budget:
                break
            ai_old = a[i]
            if ai_old < 1e-12:
                continue

            # Try scaling this coordinate
            best_local = best_obj
            best_val = ai_old
            for factor in [0.5, 0.75, 1.0, 1.33, 2.0, 0.0]:
                a[i] = ai_old * factor
                s_new = float(np.sum(a))
                if s_new < 0.01:
                    a[i] = ai_old
                    continue
                a_try = a / s_new
                obj_try = true_objective(a_try)
                if obj_try < best_local:
                    best_local = obj_try
                    best_val = ai_old * factor

            if best_val != ai_old:
                a[i] = best_val
                s_new = float(np.sum(a))
                if s_new > 0.01:
                    a = a / s_new
                    improved = True

        if sweep % 1 == 0 and verbose:
            print(f"        CD sweep {sweep}: obj={true_objective(a):.8f}, "
                  f"elapsed={time.monotonic() - start:.1f}s")

        if not improved:
            break

    obj_final = true_objective(a)
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = a.copy()
    return best_a

        obj_cd = true_objective(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase6 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase6 CD: obj={obj_cd:.8f}")

    # === Phase 7: n Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 7: n Exploration ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            n_best + 100, n_best + 200, n_best + 400,
            n_best - 100, n_best - 200,
        ]))
        n_options = [n for n in n_options if 300 <= n <= 4000]
        n_options = n_options[:3]
        print(f"  Trying n: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            a_try = init_interpolated(best_seq, n_try)
            # Quick optimization
            t1 = min(30.0, remaining * 0.3)
            a_try = lse_gradient_descent(a_try, time_budget=t1,
                                         T_start=5.0, T_end=50.0)
            t2 = min(25.0, remaining * 0.2)
            if t2 > 5:
                a_try = lp_polish(a_try, time_budget=t2,
                                  top_m_start=15, top_m_end=8, max_iters=10)
            obj_try = true_objective(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 8: Diverse Restarts ===")
        n_cur = len(best_seq)
        for r in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            # Create diverse initialization blended with current best
            a_r = init_random(n_cur, rng)
            blend = 0.3
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            s = a_r.sum()
            if s < 0.01:
                continue
            a_r = a_r / s

            t_arg = min(20.0, remaining * 0.3)
            a_r = lse_gradient_descent(a_r, time_budget=t_arg,
                                       T_start=5.0, T_end=40.0)
            t_lp = min(15.0, remaining * 0.2)
            if t_lp > 5:
                a_r = lp_polish(a_r, time_budget=t_lp,
                                top_m_start=15, top_m_end=8, max_iters=5)
            obj_r = true_objective(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 9: Final intensive LSE-GD ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 9: Final LSE-GD (budget={min(remaining-10, 100):.0f}s) ===")
        alloc = min(remaining - 10, 100.0)
        a_final = lse_gradient_descent(best_seq, time_budget=alloc,
                                       T_start=30.0, T_end=500.0, verbose=True)
        obj_final = true_objective(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LSE: obj={best_score:.8f} **")

    # === Final: CD pass ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_last_cd = coordinate_descent(best_seq, time_budget=min(remaining - 10, 30.0),
                                       max_sweeps=1)
        obj_last = true_objective(a_last_cd)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last_cd.copy()
            print(f"  Last CD: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = 2.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        # Fallback
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
