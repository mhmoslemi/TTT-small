def solve_lp_cuts(a, max_cuts=50, time_budget=5.0):
    """Solve the LP using row generation (cutting plane method)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    start = time.monotonic()
    
    # Build the convolution matrix A where (A @ g)[k] = (a_norm * g)[k]
    # A[k, j] = a_norm[k-j] if 0 <= k-j < n, else 0
    # This is a Toeplitz-like matrix
    
    # Start with all constraints
    # A is (2n-1) x n
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # A[k, j] = a_norm[k-j] for valid k-j
        # k-j in [0, n-1] => k in [j, j+n-1]
        r = min(n, 2*n-1-j)
        A[j:j+r, j] = a_norm[:r]
    
    # Row generation: start with a subset of rows
    # Initially, use all rows (since n=800 gives 1599 rows, manageable)
    all_rows = np.arange(2*n-1)
    active_rows = all_rows.copy()
    
    for iteration in range(max_cuts):
        if time.monotonic() - start > time_budget:
            break
            
        A_sub = A[active_rows, :]
        
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A_sub @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        prob.solve(solver=cp.PDLP)
        
        if g.value is None or prob.status not in ('optimal', 'optimal_inaccurate'):
            # Try fallback solvers
            for solver in [cp.CLARABEL, cp.SCS]:
                try:
                    prob.solve(solver=solver)
                    if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                        break
                except:
                    pass
            if g.value is None:
                return None, None
        
        g_val = np.maximum(g.value, 0)
        s = float(np.sum(g_val))
        if s < 1e-10:
            return None, None
        g_val = g_val / s
        
        # Check which constraints are violated
        conv_vals = A @ g_val  # (2n-1,)
        M_val = float(M.value) if M.value is not None else float(np.max(conv_vals))
        
        # Violated constraints: conv_vals[k] > M_val + tolerance
        violated = np.where(conv_vals > M_val + 1e-6)[0]
        
        if len(violated) == 0:
            return g_val, M_val
        
        if len(violated) > 0:
            # Add violated constraints to active set (keep at most max_cuts)
            if len(active_rows) + len(violated) > max_cuts:
                # Keep the most violated ones
                all_violated = np.concatenate([active_rows, violated])
                sort_idx = np.argsort(conv_vals[all_violated])[-max_cuts:]
                active_rows = all_violated[sort_idx]
            else:
                active_rows = np.concatenate([active_rows, violated])
    
    return g_val, float(M.value) if M.value is not None else float(np.max(A @ g_val))
