    1. Compute candidate `delta` values:

        # where B = flip(active_indicator)

    This can be computed in O(n log n) per iteration using FFT!

28. **Let me also add a "flattening" LP:**
    
    Instead of just minimizing the max, we can minimize a weighted sum where the top few values get more weight:
    - Sort conv values in decreasing order: `c_(1) >= c_(2) >= ... >= c_(2n-1)`

    - Or use the smooth approximation.
    
    Actually, the LP already kind of does this implicitly - by minimizing the max of the cross-convolution, it tends to flatten the response.

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Combines Multiplicative Weight Updates (MWM) with subgradient, LP alternating
minimization, analytical coordinate descent, and multi-restart exploration.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass

    for solver in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def mwm_optimize(a, time_budget, seed=42, verbose=True):
    """
    Multiplicative Weight Updates (MWM) with subgradient.
    Uses the subgradient of the max-convolution objective.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    eta = 0.1  # Initial step size
    eta_min = 1e-6
    iter_count = 0
    stall = 0
    best_eta = eta

    while time.monotonic() - start < time_budget - 2:
        t0 = time.monotonic()
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        current_obj = 2.0 * n * max_conv

        # Find active set
        eps = 1e-4 * max_conv
        active = np.where(conv > max_conv - eps)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(conv)])
        
        # Compute subgradient using FFT
        # grad_j = 2 * (mean_{k in active} a_{k-j} - max_conv)
        # mean_{k in active} a_{k-j} = (1/|active|) * sum_{k in active} a_{k-j}
        # sum_{k in active} a_{k-j} = (a * B)[2n-2-j] where B = flip(indicator(active))
        
        active_indicator = np.zeros(2 * n - 1)
        active_indicator[active] = 1.0
        B = active_indicator[::-1]  # flip
        
        # Compute a * B via FFT
        size = 1
        while size < n + (2 * n - 1) - 1:
            size <<= 1
        fa = rfft(a, size)
        fB = rfft(B, size)
        conv_ab = irfft(fa * fB, size)[:n + (2 * n - 1) - 1]
        
        # grad_j = 2 * (conv_ab[2n-2-j] / |active| - max_conv)
        # Note: j goes from 0 to n-1, so 2n-2-j goes from 2n-2 down to n-1
        grad = 2.0 * (conv_ab[2 * n - 2 : n - 1 : -1] / len(active) - max_conv)
        
        # MWM update with backtracking
        improved = False
        for attempt in range(10):
            try_eta = best_eta * (1.0 if attempt == 0 else 0.5 ** attempt)
            a_new = a * np.exp(-try_eta * grad)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj_norm(a_new)
            
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                best_eta = try_eta
                improved = True
                break
            elif obj_new < current_obj:
                # Accept even if not improving globally (for progress)
                a = a_new
                best_eta = try_eta
                break
        
        if not improved:
            stall += 1
            # Reduce step size
            best_eta = max(eta_min, best_eta * 0.7)
            if stall > 20:
                # Perturb to escape
                strength = 0.01 * min(stall / 20, 3.0)
                a = best_a * np.exp(rng.standard_normal(n) * strength)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0
        else:
            stall = 0
            # Slowly increase step size
            best_eta = min(0.5, best_eta * 1.02)
        
        iter_count += 1
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    MWM iter {iter_count}: obj={best_obj:.10f}, eta={best_eta:.6f}, elapsed={elapsed:.0f}s")

    if verbose:
        print(f"    MWM done: {iter_count} iters, best={best_obj:.10f}")
    return best_a


def analytical_cd_polish(a, time_budget, seed=42, verbose=True):
    """
    Coordinate descent with analytical optimal step for each coordinate.
    For each coordinate j, finds the delta that minimizes the objective
    by considering the analytical optimum and key breakpoints.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    sweep = 0
    no_improve = 0
    max_no_improve = 3

    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            a_j = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)
            
            if k_start > 2 * n - 2 or k_end < 0:
                continue

            # Compute unaffected max
            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))

            # Candidate deltas
            candidates = set()
            candidates.add(0.0)
            candidates.add(-a_j)  # set to zero
            
            # Analytical optimum for each affected term being the max
            affected_indices = np.arange(k_start, k_end + 1)
            for k in affected_indices:
                if k == 2 * j:
                    # Quadratic case: delta_opt = 1 - conv_{2j} / a_j
                    if a_j > 1e-15:
                        delta_opt = 1.0 - conv[2 * j] / a_j
                        if delta_opt >= -a_j:
                            candidates.add(delta_opt)
                else:
                    # Linear case: delta_opt = 1 - conv_k / a_{k-j}
                    idx = k - j
                    if 0 <= idx < n and a[idx] > 1e-15:
                        delta_opt = 1.0 - conv[k] / a[idx]
                        if delta_opt >= -a_j:
                            candidates.add(delta_opt)
            
            # Also try to match unaffected max
            # Find delta where max_affected(delta) = max_unaffected
            # For the currently max affected term:
            current_max_k = k_start + int(np.argmax(conv[k_start:k_end + 1]))
            if current_max_k == 2 * j:
                if a_j > 1e-15:
                    # conv_{2j} + 2*a_j*d + d^2 = max_unaffected
                    # d^2 + 2*a_j*d + (conv_{2j} - max_unaffected) = 0
                    disc = 4 * a_j * a_j - 4 * (conv[2 * j] - max_unaffected)
                    if disc >= 0:
                        d1 = (-2 * a_j + np.sqrt(disc)) / 2
                        d2 = (-2 * a_j - np.sqrt(disc)) / 2
                        for d in [d1, d2]:
                            if d >= -a_j:
                                candidates.add(d)
            else:
                idx = current_max_k - j
                if 0 <= idx < n and a[idx] > 1e-15:
                    # conv_k + 2*d*a_{k-j} = max_unaffected
                    d = (max_unaffected - conv[current_max_k]) / (2 * a[idx])
                    if d >= -a_j:
                        candidates.add(d)
            
            # Heuristic candidates
            mean_val = 1.0 / n
            for factor in [0.1, 0.3, 0.7, 1.5, 3.0]:
                candidates.add(mean_val * factor - a_j)
            
            # Evaluate all candidates
            best_trial_obj = best_obj
            best_trial_delta = 0.0
            
            for delta in candidates:
                if delta < -a_j - 1e-12:
                    continue
                if delta > 1.0:  # Can't increase too much
                    continue
                delta = max(delta, -a_j)
                
                new_val = a_j + delta
                S_new = 1.0 + delta
                if S_new <= 1e-10:
                    continue
                
                # Compute max of affected conv entries
                max_affected = 0.0
                for k in affected_indices:
                    if k == 2 * j:
                        val = conv[k] + 2 * a_j * delta + delta * delta
                    else:
                        idx = k - j
                        if 0 <= idx < n:
                            val = conv[k] + 2 * delta * a[idx]
                        else:
                            val = conv[k]
                    max_affected = max(max_affected, val)
                
                max_conv_trial = max(max_unaffected, max_affected)
                obj_trial = 2.0 * n * max_conv_trial / (S_new * S_new)
                
                if obj_trial < best_trial_obj - 1e-14:
                    best_trial_obj = obj_trial
                    best_trial_delta = delta
            
            if best_trial_delta != 0.0:
                delta = best_trial_delta
                new_val = a_j + delta
                S_new = 1.0 + delta
                
                # Update conv in-place
                for k in affected_indices:
                    if k == 2 * j:
                        conv[k] += 2 * a_j * delta + delta * delta
                    else:
                        idx = k - j
                        if 0 <= idx < n:
                            conv[k] += 2 * delta * a[idx]
                
                # Normalize
                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S
                a[j] = new_val * inv_S  # Wait, this is wrong after normalization
                
                # Actually let me redo the update more carefully
                # a[j] was a_j, now should be new_val * inv_S (after normalizing everything)
                # But we already multiplied a by inv_S, so a[j] is now (a_j + delta) * inv_S? No...
                # We set a[j] = a_j initially, then the "virtual" new value is a_j + delta.
                # After multiplying all of a by inv_S, a[j] = a_j * inv_S.
                # But we want a[j] = (a_j + delta) * inv_S.
                # So we need to add delta * inv_S to a[j].
                a[j] += delta * inv_S
                
                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and sweep % 5 == 0:

            print(f"    ACd sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")

    return best_a


def lp_alternate_improved(a, time_budget, seed=42, verbose=True):
    """LP alternating minimization with adaptive step size and better warm starts."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    iter_count = 0
    stall = 0

    while time.monotonic() - start < time_budget - 3:
        t0 = time.monotonic()
        g_lp, M_lp = solve_lp(a, warm_start=prev_g)
        if g_lp is None:
            break
        prev_g = g_lp
        iter_count += 1

        # Adaptive line search
        best_local_obj = best_obj
        best_local_a = a.copy()
        best_t = 0.0

        t_values = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]
        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_t = t

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall = 0
            if verbose and iter_count % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP {iter_count}: obj={best_obj:.10f}, t={best_t:.3f}, elapsed={elapsed:.0f}s")
        else:
            stall += 1
            if stall >= 5:
                # More aggressive perturbation
                strength = 0.01 + 0.005 * min(stall, 15)
                a = best_a + strength * rng.standard_normal(n) * best_a
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                prev_g = None
                stall = 0

    if verbose:
        print(f"    LP done: {iter_count} iters, best={best_obj:.10f}")
    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 9):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_perturb(a, rng, strength=0.05):
    """Add random multiplicative perturbation."""
    n = len(a)
    a_new = a * np.exp(rng.standard_normal(n) * strength)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s
