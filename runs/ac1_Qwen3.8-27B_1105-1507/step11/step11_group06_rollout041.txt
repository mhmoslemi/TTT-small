def build_conv_matrix(a):
    """Build matrix A such that (A @ g)_k = (a*g)_k for all k."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_self_conv_aware(a, t_target=0.3):
    """
    Solve LP that accounts for self-convolution of the mixture.
    min M
    s.t. ((1-t)f + t*g) * ((1-t)f + t*g))_k <= M (linearized)
         sum(g) = 1, g >= 0
    Linearization: ((1-t)^2 + t^2) * (a*a)_k + 2*t*(1-t) * (a*g)_k <= M
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    t = t_target
    coeff_aa = (1 - t) ** 2 + t ** 2
    coeff_ag = 2 * t * (1 - t)
    
    aa = compute_conv(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [
        coeff_aa * cp.multiply(aa, 1) + A @ (coeff_ag * g) <= M,
        cp.sum(g) == 1
    ]
    # Wait, this isn't right. The constraint should be:
    # coeff_aa * aa_k + coeff_ag * (A @ g)_k <= M for all k
    # which is: A @ (coeff_ag * g) + coeff_aa * aa <= M
    
    constraints = [
        A @ (coeff_ag * g) + coeff_aa * aa <= M,
        cp.sum(g) == 1
    ]
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_standard_lp(a):
    """Standard LP: min M s.t. (a*g)_k <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, direction, t_values=None):
    """Line search for best t in a + t*(direction - a)."""
    if t_values is None:
        t_values = np.concatenate([
            np.linspace(0, 0.5, 20),
            np.linspace(0.5, 1.0, 30),
            np.linspace(1.0, 1.3, 15)
        ])
    best_t = 0.0
    best_obj = compute_obj(a)
    for t in t_values:
        a_try = (1 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    return best_t, best_obj


def flatten_iteration(a, n_iters=100, verbose=False):
    """
    Direct convolution flattening:
    - Compute b = a*a
    - For each j, the "contribution" to the max of b is measured
    - Reduce a_j where it contributes to hot spots, increase where cold
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(n_iters):
        b = compute_conv(a)
        b_max = np.max(b)
        # Compute gradient of max(b) w.r.t. a
        # Use the top few active constraints
        top_k = max(1, min(5, len(b)))
        active = np.argsort(b)[-top_k:]
        
        # Gradient at active constraints
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j] / top_k
        
        # Negative gradient step with projection
        lr = 0.05 / (1.0 + 0.05 * it)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        # Use best as current if not improving
        a = best_a if obj >= best_obj else a_new
        
    return best_a, best_obj


def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.02 + 5.0 * (2 * x - 1) ** 4)
    elif mode == 6:
        # W-shaped
        a = np.abs(4 * (x - 0.5) ** 2 - 0.25) + 0.05
    elif mode == 7:
        # Bimodal with gap
        a = np.exp(-40 * x ** 2) + np.exp(-40 * (1 - x) ** 2) + 0.001
    else:
        a = 1.0 / (0.1 + 2.0 * np.abs(2 * x - 1))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s

"""
Multi-phase step function optimization combining:
- Self-convolution-aware LP direction finding
- Nesterov-accelerated exact subgradient descent
- FFT-based efficient gradient computation
- Shape parameter optimization (U-shaped family)
- Aggressive n exploration with diverse restarts
Key improvement: LP formulation accounts for self-convolution of the mixture,
and accelerated subgradient method provides cheap iterations between expensive LPs.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(x):
    n = len(x)
    if n == 0:
        return x
    lo = -1e10
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(x - mid, 0.0))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.maximum(x - tau, 0.0)


def grad_of_max_conv(a, p_weights=None):
    """Compute gradient of sum_k p_k * (a*a)_k w.r.t. a.
    If p_weights is None, use uniform weights on the top-1 active constraint.
    """
    n = len(a)
    if p_weights is None:
        conv = compute_conv(a)
        max_val = np.max(conv)
        active = (conv >= max_val - 1e-12)
        p_weights = active.astype(np.float64)
        total = np.sum(p_weights)
        if total > 0:
            p_weights = p_weights / total
        else:
            p_weights = np.zeros(2 * n - 1)
            p_weights[np.argmax(conv)] = 1.0

    # grad_i = 2 * sum_k p_k * a[k-i] for valid k-i
    # = 2 * (a correlated with p)[i]
    # Using: c[i] = sum_j a[j] * p[i+j] = conv(a, p_rev)[n-1 : 2n-1][::-1]
    p_rev = p_weights[::-1]
    conv_full = np.convolve(a, p_rev, mode='full')
    c = conv_full[n - 1: 2 * n - 1][::-1]
    return 2.0 * c


def nesterov_subgradient(a, time_budget=30.0, top_k=3, verbose=False):
    """Nesterov-accelerated subgradient descent on max_k (a*a)_k with simplex constraint."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    y = a.copy()  # Nesterov momentum variable
    t_k = 1.0
    lr0 = 0.03
    start = time.monotonic()
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        # Compute subgradient at y using top_k active constraints
        conv_y = compute_conv(y)
        top_indices = np.argsort(conv_y)[-top_k:]
        vals = conv_y[top_indices]
        max_val = np.max(vals)
        # Soft weights: emphasize the maximum
        weights = np.exp(50.0 * (vals - max_val))
        weights = weights / np.sum(weights)

        # Build combined p_weights
        p_w = np.zeros(2 * n - 1)
        p_w[top_indices] = weights

        grad = grad_of_max_conv(y, p_w)

        # Step
        lr = lr0 / (1.0 + 0.01 * it)
        x_new = project_simplex(y - lr * grad)

        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        a = x_new
        t_k = t_new

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            best_it = it

        it += 1
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    NS it{it}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

        if it - best_it > 3000:
            a = best_a.copy()
            y = best_a.copy()
            t_k = 1.0
            lr0 = max(0.005, lr0 * 0.5)
            best_it = it

    return best_a, best_obj


def build_conv_matrix(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_standard(a):
    """min M s.t. (a*g)_k <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def solve_lp_self_aware(a, t_val=0.4):
    """LP with self-convolution linearization for mixture (1-t)*a + t*g."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    aa = compute_conv(a_norm)

    coeff_aa = (1 - t_val) ** 2 + t_val ** 2
    coeff_ag = 2.0 * t_val * (1 - t_val)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [
        coeff_aa * aa + A @ (coeff_ag * g) <= M,
        cp.sum(g) == 1
    ]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def solve_lp_multi_t(a, t_vals=None):
    """Solve LP for multiple t values, return best direction."""
    if t_vals is None:
        t_vals = [0.2, 0.4, 0.6, 0.8]
    best_dir = None
    best_obj = np.inf
    for t in t_vals:

        if obj_g < best_obj:
            best_obj = obj_g
            best_dir = g
    if best_dir is not None:
        return best_dir
    return solve_lp_standard(a)


def line_search_full(a, direction, n_coarse=50, n_fine=30):
    """Two-stage line search."""
    best_t = 0.0
    best_obj = compute_obj(a)
    ts = np.linspace(0.0, 1.3, n_coarse)
    for t in ts:
        a_try = (1.0 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t

    if 0.0 < best_t < 1.3:
        lo = max(0.0, best_t - 0.06)
        hi = min(1.3, best_t + 0.06)
        ts_fine = np.linspace(lo, hi, n_fine)
        for t in ts_fine:
            a_try = (1.0 - t) * a + t * direction
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t
    return best_t, best_obj


def u_shaped_param(n, sharpness=3.0, offset=0.05):
    x = np.linspace(0, 1, n)
    a = 1.0 / (offset + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def optimize_shape_params(n, n_sharp=20, n_offset=10):
    """Grid search over U-shaped family parameters."""
    best_a = None
    best_obj = np.inf
    sharpnesses = np.logspace(-2, 2.5, n_sharp)
    offsets = np.logspace(-3, 0.5, n_offset)
    for sh in sharpnesses:
        for off in offsets:
            a = u_shaped_param(n, sharpness=sh, offset=off)
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    return best_a, best_obj


def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.02 + 5.0 * (2 * x - 1) ** 4)
    elif mode == 6:
        a = np.abs(4 * (x - 0.5) ** 2 - 0.25) + 0.05
    elif mode == 7:
        a = np.exp(-50 * x ** 2) + np.exp(-50 * (1 - x) ** 2) + 0.001
    elif mode == 8:
        # Cosine-tapered U
        a = 0.5 + 0.5 * np.cos(np.pi * (2 * x - 1))
        a = a ** 2 + 0.01
    else:
        a = 1.0 / (0.1 + 2.0 * np.abs(2 * x - 1))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    if n_src == 0:
        return None
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s


def main_optimization_loop(a, time_budget=100.0, max_iters=80, verbose=False):
    """
    Main optimization loop alternating between:
    - LP direction finding (standard and self-aware)
    - Nesterov subgradient steps
    - Periodic diverse restarts on stall
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(777)
    consecutive_sg = 0

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        # Decide strategy based on iteration count and stall
        if stall > 4:
            # Do subgradient for a few iterations
            sg_budget = min(remaining * 0.15, 15.0)
            a_sg, obj_sg = nesterov_subgradient(a, time_budget=sg_budget, top_k=5)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg.copy()
                a = a_sg
                stall = 0
            consecutive_sg += 1
            it += 1
            if consecutive_sg >= 2:
                # Diverse restart
                mode = it % 9
                a_div = diverse_init(n, rng, mode=mode)
                blend = min(0.4, 0.15 * stall)
                a = (1.0 - blend) * best_a + blend * a_div
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                consecutive_sg = 0
            continue

        # LP-based direction finding
        T0 = time.monotonic()
        if it % 4 == 0:
            g_dir = solve_lp_standard(a)
        elif it % 4 == 1:
            g_dir = solve_lp_self_aware(a, t_val=0.3)
        elif it % 4 == 2:
            g_dir = solve_lp_self_aware(a, t_val=0.5)
        else:
            g_dir = solve_lp_multi_t(a, t_vals=[0.3, 0.5, 0.7])
        lp_time = time.monotonic() - T0

        if g_dir is None:
            stall += 1
            it += 1
            continue

        # Line search
        t_opt, obj_opt = line_search_full(a, g_dir)

        if t_opt > 1e-6 and obj_opt < best_obj - 1e-14:
            a_new = (1.0 - t_opt) * a + t_opt * g_dir
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
                consecutive_sg = 0
            else:
                stall += 1
        else:
            # Try direct replacement
            obj_direct = compute_obj(g_dir)
            if obj_direct < best_obj:
                best_obj = obj_direct
                best_a = g_dir.copy()
                a = g_dir.copy()
                stall = 0
                consecutive_sg = 0
            else:
                stall += 1

        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    Opt it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                  f"lp={lp_time:.2f}s, elapsed={elapsed:.1f}s, stall={stall}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization:
    1. Load parent + shape parameter optimization
    2. Intensive main optimization loop (LP + Nesterov subgradient)
    3. n exploration with interpolation
    4. Diverse restarts
    5. Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        s = float(np.sum(parent_arr))
        if s > 0.01:
            best_seq = parent_arr / s
            best_score = compute_obj(best_seq)
            print(f"Parent loaded: n={parent_arr.size}, obj={best_score:.8f}")
        else:
            raise ValueError("Sum too small")
    except Exception:
        best_seq = u_shaped_param(800, sharpness=3.0, offset=0.05)
        best_score = compute_obj(best_seq)
        print(f"Default init: n={len(best_seq)}, obj={best_score:.8f}")

    n_best = len(best_seq)

    # Shape parameter optimization at current n
    print("\n=== Shape Parameter Optimization ===")
    a_shape, obj_shape = optimize_shape_params(n_best, n_sharp=25, n_offset=12)
    if obj_shape < best_score:
        best_score = obj_shape
        best_seq = a_shape.copy()
        print(f"  Shape opt: obj={best_score:.8f} **")
    else:
        print(f"  Shape opt: obj={obj_shape:.8f} (parent better)")

    # Diverse inits
    for mode in range(9):
        a_alt = diverse_init(n_best, rng, mode=mode)
        obj_alt = compute_obj(a_alt)
        if obj_alt < best_score:
            best_score = obj_alt
            best_seq = a_alt.copy()
            print(f"  Div init mode={mode}: obj={best_score:.8f} **")

    # === Phase 1: Main optimization loop ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        alloc = min(remaining * 0.50, 600.0)
        print(f"\n=== Phase 1: Main Optimization (n={n_best}, budget={alloc:.0f}s) ===")
        a_lp, obj_lp = main_optimization_loop(best_seq, time_budget=alloc, max_iters=150, verbose=True)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_lp:.8f}")

    # === Phase 2: Nesterov subgradient refinement ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 2: Nesterov Subgradient ===")
        for top_k in [1, 3, 5, 10]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(remaining * 0.15, 30.0)
            a_ns, obj_ns = nesterov_subgradient(best_seq, time_budget=alloc, top_k=top_k, verbose=True)
            if obj_ns < best_score:
                best_score = obj_ns
                best_seq = a_ns.copy()
                print(f"  NS top_k={top_k}: obj={best_score:.8f} **")
            else:
                print(f"  NS top_k={top_k}: obj={obj_ns:.8f}")

    # === Phase 3: n Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 3: n Exploration ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            n_best - 150, n_best - 100, n_best - 50,
            n_best + 50, n_best + 100, n_best + 150, n_best + 300,
            500, 600, 700, 800, 900, 1000, 1100, 1200, 1300,
            1400, 1500, 1600, 1800, 2000
        ]))
        n_options = [n for n in n_options if 400 <= n <= 2200 and abs(n - n_best) >= 40]
        n_options.sort(key=lambda x: abs(x - n_best))
        n_options = n_options[:8]
        print(f"  n candidates: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            a_try = interpolate_sequence(best_seq, n_try)
            if a_try is None:
                continue
            # Also try shape-optimized at this n
            a_shape_n, obj_shape_n = optimize_shape_params(n_try, n_sharp=15, n_offset=8)
            if obj_shape_n < compute_obj(a_try):
                a_try = a_shape_n

            alloc = min(remaining * 0.25, 50.0)
            a_try2, obj_try = main_optimization_loop(a_try, time_budget=alloc, max_iters=40, verbose=False)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try2.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Diverse restarts with blending ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: Diverse Restarts ===")
        n_cur = len(best_seq)
        for r in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 35:
                break
            mode = (r * 2 + 3) % 9
            a_r = diverse_init(n_cur, rng, mode=mode)
            blend = 0.15 + 0.08 * r
            a_r = (1.0 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
            alloc = min(remaining * 0.25, 40.0)
            if alloc > 8:
                a_r, obj_r = main_optimization_loop(a_r, time_budget=alloc, max_iters=25, verbose=False)
            else:
                obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} mode={mode}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} mode={mode}: obj={obj_r:.8f}")

    # === Phase 5: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 5: Final Polish ===")
        alloc = min(remaining - 10, 120.0)
        if alloc > 10:
            a_final, obj_final = main_optimization_loop(best_seq, time_budget=alloc, max_iters=80, verbose=True)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final: obj={best_score:.8f} **")
            else:
                print(f"  Final: obj={obj_final:.8f}")

    # === Phase 6: Ultimate squeeze ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 6: Squeeze ===")
        # Subgradient with fine top_k
        alloc_sg = min(remaining * 0.35, 20.0)
        a_sq, obj_sq = nesterov_subgradient(best_seq, time_budget=alloc_sg, top_k=2, verbose=True)
        if obj_sq < best_score:
            best_score = obj_sq
            best_seq = a_sq.copy()
            print(f"  Squeeze NS: obj={best_score:.8f} **")

        remaining2 = deadline - time.monotonic()
        if remaining2 > 15:
            a_sq2, obj_sq2 = main_optimization_loop(best_seq, time_budget=min(remaining2 - 5, 20.0), max_iters=15)
            if obj_sq2 < best_score:
                best_score = obj_sq2
                best_seq = a_sq2.copy()
                print(f"  Squeeze LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
