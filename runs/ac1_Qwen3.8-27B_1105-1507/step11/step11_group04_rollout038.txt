```python
import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def compute_obj_norm(a):
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))
```

```python
def project_simplex(y):
    """Project y onto the simplex {a >= 0, sum(a) = 1}."""
    n = len(y)
    # Bisection for tau
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))
    for _ in range(100):
        mid = (lo + hi) / 2
        s = float(np.sum(np.maximum(y - mid, 0)))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.maximum(y - tau, 0)
```

```python
import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def project_simplex(y):
    """Project y onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(y)
    if n == 0:
        return np.array([])
    # Use sorting-based projection (faster than bisection for moderate n)
    u = np.sort(y)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(y - theta, 0)

    """
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    
    # Find all positions where conv is close to max
    threshold = max_val - 1e-10 * max_val
    max_positions = np.where(conv >= threshold)[0]
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in max_positions:
        # d(conv[k])/d(a[j]) = 2*a[k-j] if 0 <= k-j < n, else 0
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            grad[lo:hi+1] += 2.0 * a[k - lo:k - hi + 1][::-1]
    
    # Normalize gradient by number of max positions
    grad /= len(max_positions)
    
    # Gradient step
    y = a - step_size * grad
    
    # Project onto simplex
    a_new = project_simplex(y)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def accelerated_subgradient(a, time_budget, verbose=False):
    """
    Nesterov-accelerated subgradient method.
    """
    n = len(a)
    a = a / float(np.sum(a))
    start = time.monotonic()
    
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    
    y = a.copy()
    t = 1.0
    base_lr = 2.0 / (n ** 0.5 + 1)
    
    for it in range(5000):
        if time.monotonic() - start > time_budget - 1:
            break
        
        # Compute gradient at y
        conv_y = compute_conv(y)
        max_val = float(np.max(conv_y))
        threshold = max_val - 1e-10
        max_positions = np.where(conv_y >= threshold)[0]
        
        grad = np.zeros(n)
        for k in max_positions:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                grad[lo:hi+1] += 2.0 * a[k - lo:k - hi + 1][::-1]
        grad /= len(max_positions)
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        lr = base_lr / (1.0 + t)
        
        y_new = project_simplex(y - lr * grad)
        # Momentum
        momentum = (t - 1.0) / t_new
        y = (1.0 - momentum) * y_new + momentum * y
        y = project_simplex(y)
        
        # Check improvement
        obj_y = compute_obj_norm(y)
        if obj_y < best_obj - 1e-12:
            best_obj = obj_y
            best_a = y.copy()
        
        t = t_new
    
    return best_a


def fourier_init(n, rng, n_harmonics=8):
    """Generate a Fourier-based random initialization."""
    x = np.arange(n)
    a = np.ones(n)
    for k in range(1, n_harmonics + 1):
        amp = rng.uniform(0.1, 0.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        a *= (1.0 + amp * np.cos(2.0 * np.pi * k * x / n + phase))
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def generate_init(n, method, rng, parent=None):
    """Generate initial sequence."""
    if method == 'uniform':
        return np.ones(n) / n
    elif method == 'random':
        a = rng.dirichlet(np.ones(n))
        return a
    elif method == 'fourier':
        return fourier_init(n, rng)
    elif method == 'gaussian':
        x = np.arange(n)
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / a.sum()
    elif method == 'parent' and parent is not None:
        if len(parent) == n:
            return parent / parent.sum()
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, parent)
            a = np.maximum(a, 0)
            s = a.sum()
            return a / s if s > 1e-10 else np.ones(n) / n
    else:
        return np.ones(n) / n


def perturb_sequence(a, rng, strength=0.02):
    """Perturb a sequence while maintaining non-negativity and normalization."""
    n = len(a)
    pert = rng.standard_normal(n) * strength
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s
```
