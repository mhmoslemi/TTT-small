"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses enhanced LP fixed-point with golden-section line search,
convolution-targeted subgradient steps, and multiplicative updates.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lp_solve(a, solver_list=None):
    """Solve min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1. Returns (g, M)."""
    n = len(a)
    # Build toeplitz matrix A where (A @ g)[k] = conv(a, g)[k]
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solver_list:
        try:

            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:

    return None, None


def golden_section_search(a, g, time_budget, max_evals=30):
    """
    Golden section search over t in [0, 1] to minimize obj((1-t)*a + t*g).
    Returns best t and best value.
    """

    lo = 0.0
    hi = 1.0

def multiplicative_update(a, g, t):
    """Multiplicative (geometric) update: a_new = a * (g/mean(g))^t."""
    g_mean = float(np.mean(g))
    if g_mean < 1e-15:
        return a.copy()
    ratio = g / g_mean
    # Clip to avoid log of negative
    ratio = np.maximum(ratio, 1e-15)
    a_new = a * (ratio ** t)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def subgradient_step(a, step_size, top_k=5):
    """
    Subgradient step targeting the top-k convolution entries.
    Returns updated a.
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    
    # Find top k lag positions
    top_indices = np.argsort(conv)[::-1][:top_k]
    
    # Compute subgradient (average gradient over active lags)
    subgrad = np.zeros(n)
    for lag in top_indices:
        for i in range(n):
            j = lag - i
            if 0 <= j < n:
                subgrad[i] += 2.0 * a[j]
    subgrad /= len(top_indices)
    
    # Normalize
    gn = np.linalg.norm(subgrad)
    if gn < 1e-15:
        return a.copy()
    subgrad = subgrad / gn
    
    # Step
    a_new = a - step_size * subgrad
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

def convex_gd(a, time_budget, verbose=False, seed=42):
    """
    Gradient descent with log-sum-exp smoothing of the max-convolution.
    Uses Nesterov acceleration.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau_levels = [10, 30, 100, 300, 1000, 3000]

    for tau in tau_levels:
        if time.monotonic() - start >= time_budget:
            break
        lr = 1.0 / np.sqrt(n)
        beta = 0.9
        v = np.zeros(n)
        iters = 0
        max_iters = 3000

        while iters < max_iters and time.monotonic() - start < time_budget:
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            # Compute smoothed gradient
            conv = compute_conv(y)
            exact_max = float(np.max(conv))
            shifted = tau * (conv - exact_max)
            shifted = np.clip(shifted, -700, 700)
            exp_vals = np.exp(shifted)
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = exp_vals / sum_exp

            # Gradient: 2 * ifft(fft(p) * conj(fft(y)))[:n]
            N = 1
            while N < 2 * n:
                N <<= 1
            P = fft(p, N)
            Y = fft(y, N)
            cross = ifft(P * np.conj(Y)).real
            g = 2.0 * cross[:n]

            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new
            iters += 1
            lr = max(1e-7, lr * 0.99995)

        if verbose:
            print(f"    GD tau={tau:5d}: obj={best_obj:.10f}, iters={iters}")

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

If such `h` exists, we can take a step in that direction. This is essentially a "feasible direction" search.

Actually, this is exactly what the LP fixed-point is doing! The LP finds `g` such that `conv(a*, g)_l <= M_lp` for all l. The perturbation direction is `h = g - a*` (since `sum(g) = 1 = sum(a*)`, so `sum(h) = 0`).

At a fixed point, `g = a*`, so `h = 0`, meaning no feasible direction exists. To escape, we need to either:
1. Accept a slightly worse first-order improvement (the LP solution might not be exactly `a*` if we're not at a fixed point)
2. Use second-order information
3. Use a different direction

**Second-order information**: The change in the objective is:
`max_l [conv(a*,a*)_l + 2*eps*conv(a*,h)_l + eps^2*conv(h,h)_l]`

The `eps^2` term is the "curvature." If we can find a direction `h` where the first-order term is zero (or slightly negative) but the second-order term is negative, we can still make progress.

This suggests using a **Newton-type step**: solve for `h` that minimizes the quadratic approximation:
`min_h max_l [c_l + 2*eps*d_l + eps^2*e_l]`
where `c_l = conv(a*,a*)_l`, `d_l = conv(a*,h)_l`, `e_l = conv(h,h)_l`.

This is a non-convex problem, but we can approximate it.
