"""
Convex-aware optimization for autocorrelation minimization.
Uses: (1) Nesterov-accelerated mirror descent with multiplicative weights,
(2) LP-based direction finding with multiple solvers,
(3) Direct QP for small n, (4) diverse n exploration.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v, c):
    """Project v onto {x: sum(x)=c, x>=0}."""
    n = len(v)
    # First project onto simplex {x: sum(x)=1, x>=0}, then scale
    u = v / c if c != 0 else v
    # Water-filling
    mu = 0.0
    for _ in range(50):
        x = np.maximum(u - mu, 0)
        s = np.sum(x)
        if abs(s - 1.0) < 1e-12:
            break
        mu += (s - 1.0) / np.sum(x > 0) if np.sum(x > 0) > 0 else 0
    return x * c


def project_simplex_fast(v, c):
    """Faster projection using sorting."""
    n = len(v)
    if c <= 0:
        return np.zeros(n)
    u = v / c
    # Sort in decreasing order
    u_sorted = np.sort(u)[::-1]
    cssv = np.cumsum(u_sorted)
    rho = np.where(u_sorted * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    x = np.maximum(u - theta, 0)
    return x * c


def subgradient_of_max_conv(a):
    """Compute subgradient of max_k conv(a,a)[k] with respect to a."""
    n = len(a)
    conv = compute_conv(a)
    k_max = int(np.argmax(conv))
    grad = np.zeros(n)
    for i in range(n):
        j1 = k_max - i
        j2 = i - k_max
        if 0 <= j1 < n:
            grad[i] += a[j1]
        if 0 <= j2 < n:
            grad[i] += a[j2]
    return grad * 2.0  # factor of 2 from the quadratic form


def nesterov_mirror_descent(a, time_budget=60.0, verbose=False):
    """
    Nesterov-accelerated mirror descent for minimizing max_conv(a)
    subject to sum(a) = c, a >= 0.
    Uses multiplicative weight updates.
    """
    n = len(a)
    c = float(np.sum(a))
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a / c)))
    
    start = time.monotonic()
    # Nesterov state
    a_curr = a.copy()  # x_t
    a_prev = a.copy()  # x_{t-1}
    
    eta = 0.5  # learning rate
    it = 0
    patience = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        # Nesterov momentum
        t_curr = 1.0 + it
        t_prev = 1.0 + (it - 1) if it > 0 else 1.0
        beta = (t_prev - 1) / t_curr if it > 0 else 0.0
        a_look = a_curr + beta * (a_curr - a_prev)
        a_look = np.maximum(a_look, 1e-15)
        
        # Subgradient at look-ahead point
        grad = subgradient_of_max_conv(a_look)
        
        # Multiplicative weight update (mirror descent)
        # a_new_i = a_look_i * exp(-eta * grad_i / c)
        # Scale eta by 1/sqrt(t) for sublinear convergence
        eta_t = eta / np.sqrt(1.0 + it)
        a_new = a_look * np.exp(-eta_t * grad / c)
        a_new = np.maximum(a_new, 1e-15)
        
        # Project onto sum(a) = c
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new * (c / s)
        
        a_prev = a_curr.copy()
        a_curr = a_new
        
        # Evaluate
        obj = 2.0 * n * float(np.max(compute_conv(a_curr / c)))
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_curr.copy()
            patience = 0
            best_it = it
        else:
            patience += 1
        
        it += 1
        if verbose and it % 100 == 0:
            print(f"    NMD it{it}: obj={best_obj:.8f}, patience={patience}")
        
        # Restart if stuck too long
        if patience > 2000:
            a_curr = best_a.copy()
            a_prev = best_a.copy()
            patience = 0
            eta = max(0.05, eta * 0.5)
        
        # Adaptive learning rate
        if it % 500 == 0 and it > 100:
            eta *= 0.95
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A such that A @ g = conv(a, g)[:n] (first n lags).
    Actually we need all 2n-1 lags."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # (a * g)[k] has contribution a[j] * g[k-j] for valid k-j
        # So A[k, j] = a[j] if 0 <= k-j < n, i.e., j <= k <= j+n-1
        lo = j
        hi = min(2 * n - 1, j + n)
        A[lo:hi, j] = a[j]
    return A


def solve_lp_direction(a, solvers=None):
    """Solve LP to find best direction g minimizing max conv(a, g)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    # Use sparse representation for efficiency
    from scipy.sparse import csr_matrix
    
    # Build sparse convolution matrix
    rows = []
    cols = []
    vals = []
    for j in range(n):
        lo = j
        hi = min(2 * n - 1, j + n)
        rows.extend(range(lo, hi))
        cols.extend([j] * (hi - lo))
        vals.extend([a_norm[j]] * (hi - lo))
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    if solvers is None:
        solvers = [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, direction, t_max=1.5, n_coarse=50, n_fine=30):
    """Exhaustive line search along a direction."""
    n = len(a)
    c = float(np.sum(a))
    best_t = 0.0
    best_obj = 2.0 * n * float(np.max(compute_conv(a / c)))
    
    ts = np.linspace(0.0, t_max, n_coarse)
    for t in ts:
        a_try = (1.0 - t) * a + t * direction * c
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try * (c / s)
        obj = 2.0 * n * float(np.max(compute_conv(a_try / c)))
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    if 0.0 < best_t < t_max:
        lo = max(0.0, best_t - 0.05)
        hi = min(t_max, best_t + 0.05)
        ts_fine = np.linspace(lo, hi, n_fine)
        for t in ts_fine:
            a_try = (1.0 - t) * a + t * direction * c
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try * (c / s)
            obj = 2.0 * n * float(np.max(compute_conv(a_try / c)))
            if obj < best_obj:
                best_obj = obj
                best_t = t
    
    return best_t, best_obj


def lp_polish(a, time_budget=60.0, max_iters=50, verbose=False):

    n = len(a)
    c = float(np.sum(a))
    a = (a / c * c).astype(np.float64)  # ensure float64

    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break
        
        # Line search
        t_opt, obj_opt = line_search(a, g_lp, t_max=1.2, n_coarse=30, n_fine=20)
        
        if t_opt > 1e-6 and obj_opt < best_obj - 1e-14:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp * c
            a_new = np.maximum(a_new, 0)
            s = float(np.sum(a_new))
            if s > 1e-10:
                a_new = a_new * (c / s)
                best_obj = 2.0 * n * float(np.max(compute_conv(a_new / c)))
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            # Try direct replacement
            obj_direct = 2.0 * n * float(np.max(compute_conv(g_lp)))
            if obj_direct < best_obj:
                best_obj = obj_direct
                best_a = (g_lp * c).copy()
                a = best_a.copy()
                stall = 0
            else:
                stall += 1
        
        it += 1
        if verbose and it % 3 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}")
        
        if stall >= 3:
            # Perturb

            a = best_a * np.exp(pert)
            a = np.maximum(a, 1e-10)
            a = a * (c / a.sum())
            stall = 0
    
    return best_a, best_obj

def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.02 + 5.0 * (2 * x - 1) ** 4)
    elif mode == 6:
        a = 1.0 / (0.05 + 2.0 * (2 * x - 1) ** 2)  # U-shaped
    elif mode == 7:
        a = np.cos(np.pi * x) ** 2 + 0.01
    else:
        a = 1.0 / (0.1 + 3.0 * np.abs(2 * x - 1))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s

"""
Convex-inspired heuristic optimization for autocorrelation minimization.
Uses: (1) LP-based direction finding with PDLP/Clarabel solvers,
(2) Multiplicative weight (mirror descent) subgradient heuristic,
(3) Adaptive perturbation for escaping local minima,

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def subgradient_max_conv(a):
    """Subgradient of max_k conv(a,a)[k] w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    k_max = int(np.argmax(conv))
    grad = np.zeros(n)
    for i in range(n):
        j1 = k_max - i
        j2 = i - k_max
        if 0 <= j1 < n:
            grad[i] += a[j1]
        if 0 <= j2 < n:
            grad[i] += a[j2]
    return grad * 2.0


def mirror_descent_subgrad(a, time_budget=60.0, verbose=False):
    """
    Multiplicative weight (mirror descent) heuristic for reducing max autocorrelation.
    Works on normalized sequences (sum=1).

def build_sparse_conv_matrix(a_norm):
    """Build sparse matrix A where A[k, j] = a_norm[j] if k-j is valid index."""
    n = len(a_norm)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        lo = j
        hi = min(2 * n - 1, j + n)
        rows.extend(range(lo, hi))
        cols.extend([j] * (hi - lo))
        vals.extend([a_norm[j]] * (hi - lo))
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_direction(a, solvers=None):
    """Find direction g minimizing max conv(a_norm, g) via LP."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_sparse_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

    if solvers is None:
        solvers = [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, direction, t_max=1.5, n_coarse=40, n_fine=25):
    """Exhaustive line search: minimize obj along a + t*(direction - a)."""
    n = len(a)
    best_t = 0.0
    best_obj = compute_obj(a)

    ts = np.linspace(0.0, t_max, n_coarse)
    for t in ts:
        a_try = (1.0 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t

    if 0.0 < best_t < t_max:
        lo = max(0.0, best_t - 0.04)
        hi = min(t_max, best_t + 0.04)
        ts_fine = np.linspace(lo, hi, n_fine)
        for t in ts_fine:
            a_try = (1.0 - t) * a + t * direction
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t

    return best_t, best_obj


def lp_polish(a, time_budget=60.0, max_iters=80, verbose=False, seed=42):
    """LP-based iterative polishing with adaptive perturbation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)

    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break

        t_opt, obj_opt = line_search(a, g_lp, t_max=1.3, n_coarse=35, n_fine=25)

        improved = False
        if t_opt > 1e-6 and obj_opt < best_obj - 1e-14:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    stall = 0
                    improved = True

        if not improved:
            # Try direct
            obj_direct = compute_obj(g_lp)
            if obj_direct < best_obj - 1e-14:
                best_obj = obj_direct
                best_a = g_lp.copy()
                a = g_lp.copy()
                stall = 0
            else:
                stall += 1

        it += 1
        if verbose and it % 5 == 0:
            print(f"      LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, stall={stall}")

        if stall >= 3:
            # Adaptive perturbation
            if stall <= 5:
                freq = rng.integers(1, max(2, n // 60))
                phase = rng.uniform(0, 2 * np.pi)
                idx = np.arange(n, dtype=np.float64)
                amp = 0.015 * stall
                pert = amp * np.cos(2 * np.pi * freq * idx / n + phase)
                a = best_a * np.exp(pert)
            else:
                # More aggressive: blend with diverse init
                mode = it % 5
                a_div = _diverse_init(n, rng, mode=mode)
                blend = min(0.3 + 0.05 * (stall - 5), 0.6)
                a = (1.0 - blend) * best_a + blend * a_div
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            stall = 0

    return best_a, best_obj


def _diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = np.exp(4 * (2 * x - 1) ** 2)
    else:
        a = 1.0 / (0.05 + 3.0 * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s


def propose_candidate(seed=42, budget_s=1500):
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        s = float(np.sum(parent_arr))
        if s > 0.01:
            best_seq = parent_arr / s
            best_score = compute_obj(best_seq)
            print(f"Parent loaded: n={parent_arr.size}, obj={best_score:.8f}")
        else:
            raise ValueError("Sum too small")
    except Exception:
        best_seq = _diverse_init(800, rng, mode=4)
        best_score = compute_obj(best_seq)
        print(f"Default init: n=800, obj={best_score:.8f}")

    n_best = len(best_seq)

    # Try alt inits at same n
    for mode in range(5):
        a_alt = _diverse_init(n_best, rng, mode=mode)
        obj_alt = compute_obj(a_alt)
        if obj_alt < best_score:
            best_score = obj_alt
            best_seq = a_alt.copy()
            print(f"  Alt init mode={mode}: obj={best_score:.8f} **")

    # === Phase 1: LP Polish (main workhorse) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.25, 250.0)
        print(f"\n=== Phase 1: LP Polish (n={n_best}, budget={alloc:.0f}s) ===")
        a_lp, obj_lp = lp_polish(best_seq, time_budget=alloc, max_iters=120, verbose=True, seed=1)
        if obj_lp < best_score - 1e-12:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: Mirror Descent Subgradient ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.1, 80.0)
        print(f"\n=== Phase 2: Mirror Descent (budget={alloc:.0f}s) ===")
        a_md, obj_md = mirror_descent_subgrad(best_seq, time_budget=alloc, verbose=True)
        if obj_md < best_score - 1e-12:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj_md:.8f}")

    # === Phase 3: LP Polish Round 2 ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(remaining * 0.15, 150.0)
        print(f"\n=== Phase 3: LP Polish R2 (budget={alloc:.0f}s) ===")
        a_lp2, obj_lp2 = lp_polish(best_seq, time_budget=alloc, max_iters=80, verbose=True, seed=2)
        if obj_lp2 < best_score - 1e-12:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj_lp2:.8f}")

    # === Phase 4: Mirror Descent Round 2 ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.08, 50.0)
        print(f"\n=== Phase 4: MD R2 (budget={alloc:.0f}s) ===")
        a_md2, obj_md2 = mirror_descent_subgrad(best_seq, time_budget=alloc, verbose=False)
        if obj_md2 < best_score - 1e-12:
            best_score = obj_md2
            best_seq = a_md2.copy()
            print(f"  Phase 4: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4: obj={obj_md2:.8f}")

    # === Phase 5: n Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        print(f"\n=== Phase 5: n Exploration ===")
        n_candidates = [650, 720, 750, 850, 900, 1000, 1100, 1200]
        n_candidates = [n for n in n_candidates if abs(n - n_best) >= 20]
        n_candidates.sort(key=lambda x: abs(x - n_best))
        n_candidates = n_candidates[:5]
        print(f"  Trying: {n_candidates}")

        for n_try in n_candidates:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break

            a_try = interpolate_sequence(best_seq, n_try)
            if a_try is None:
                continue
            obj_init = compute_obj(a_try)

            # Short LP polish
            alloc_lp = min(remaining * 0.3, 60.0)
            a_opt, obj_opt = lp_polish(a_try, time_budget=alloc_lp, max_iters=30, verbose=False, seed=n_try)

            # Short MD
            remaining2 = deadline - time.monotonic()
            alloc_md = min(remaining2 * 0.3, 30.0)
            if alloc_md > 10:
                a_opt2, obj_opt2 = mirror_descent_subgrad(a_opt, time_budget=alloc_md, verbose=False)
                if obj_opt2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj_opt2

            if obj_opt < best_score - 1e-10:
                best_score = obj_opt
                best_seq = a_opt.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_opt:.8f} (init={obj_init:.8f})")

    # === Phase 6: LP Polish on best (if n changed) ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(remaining * 0.2, 120.0)
        print(f"\n=== Phase 6: LP Polish (n={n_best}, budget={alloc:.0f}s) ===")
        a_lp3, obj_lp3 = lp_polish(best_seq, time_budget=alloc, max_iters=60, verbose=True, seed=3)
        if obj_lp3 < best_score - 1e-12:
            best_score = obj_lp3
            best_seq = a_lp3.copy()
            print(f"  Phase 6: obj={best_score:.8f} **")
        else:
            print(f"  Phase 6: obj={obj_lp3:.8f}")

    # === Phase 7: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 7: Diverse Restarts ===")
        for r in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            mode = (r + 1) % 5
            a_r = _diverse_init(n_best, rng, mode=mode)
            blend = 0.35
            a_r = (1.0 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
            alloc = min(remaining * 0.35, 50.0)
            if alloc > 15:
                a_r2, obj_r = lp_polish(a_r, time_budget=alloc, max_iters=25, verbose=False, seed=100 + r)
            else:
                a_r2, obj_r = a_r, compute_obj(a_r)
            if obj_r < best_score - 1e-10:
                best_score = obj_r
                best_seq = a_r2.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 8: Final squeeze ===
    remaining = deadline - time.monotonic()
    if remaining > 25:
        print(f"\n=== Phase 8: Final Squeeze ===")
        # Short MD
        alloc1 = min(remaining * 0.4, 35.0)
        if alloc1 > 10:
            a_f, obj_f = mirror_descent_subgrad(best_seq, time_budget=alloc1, verbose=False)
            if obj_f < best_score - 1e-12:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  MD: obj={best_score:.8f} **")

        remaining2 = deadline - time.monotonic()
        if remaining2 > 15:
            alloc2 = min(remaining2 - 5, 40.0)
            a_f2, obj_f2 = lp_polish(best_seq, time_budget=alloc2, max_iters=30, verbose=False, seed=999)
            if obj_f2 < best_score - 1e-12:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final LP: obj={best_score:.8f} **")
            else:
                print(f"  Final LP: obj={obj_f2:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
