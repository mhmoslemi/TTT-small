"""

to minimize 2*n*max(conv(a,a))/(sum(a))^2 using aggressive LP fixed-point

import numpy as np

import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a*a)."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    shifted = tau * (conv - exact_max)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def lp_solve(a, solver_list=None):
    """Solve min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1. Returns (g, M)."""
    n = len(a)
    # Build constraint matrix: (a*g)[k] = sum_j A[k,j] * g_j
    # (a*g)[k] = sum_{j=0}^{n-1} a_{k-j} * g_j where 0 <= k-j < n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = min(2 * n - 1, j + n)
        A[start:end, j] = a[:end - start]
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None


def lp_solve_top_k(a, k=20, solver_list=None):
    """Solve LP constraining only top-k active lags. Faster than full LP."""
    n = len(a)
    conv_a = compute_conv(a)
    max_val = float(np.max(conv_a))
    # Find active lags (top k)
    active_idx = np.argsort(conv_a)[-k:]
    
    A = np.zeros((len(active_idx), n))
    for row, lag in enumerate(active_idx):
        for j in range(n):
            i = lag - j
            if 0 <= i < n:
                A[row, j] = a[i]
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    # Verify full constraint

                    full_conv = np.convolve(a, gv)
                    M_full = float(np.max(full_conv))

    return None, None

        lr = 3.0 / np.sqrt(n)
        beta = 0.9
        v = np.zeros(n)
        iters = 0

        while time.monotonic() - start < time_budget and iters < 3000:
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new
            iters += 1
            lr = max(1e-8, lr * 0.9999)

        if verbose:
            print(f"    GD tau={tau:5d}: obj={best_obj:.10f}, iters={iters}")

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_structured_init(n, rng, mode=0):
    """Create structured initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.maximum(np.sinc(x), 0.001)
        return a / a.sum()
    elif mode == 2:
        a = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / max(1, n - 1))
        return a / a.sum()
    elif mode == 3:
        x = np.arange(n) / max(1, n - 1)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        return np.maximum(a, 0.001) / a.sum()
    elif mode == 4:
        x = np.linspace(-3, 3, n)
        a = np.exp(-x ** 2)
        return a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
        return a / a.sum()
    elif mode == 6:
        # Optimal-ish: raised cosine with extra smoothing
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5))) ** 2
        return a / a.sum()
    elif mode == 7:
        # Parabolic
        x = np.linspace(-1, 1, n)
        a = np.maximum(1 - x ** 2, 0.001)
        return a / a.sum()
    else:
        a = rng.uniform(0.1, 1.0, n)
        return a / a.sum()


def active_set_subgradient(a, time_budget, verbose=False, seed=42, max_iters=300):
    """Subgradient method targeting active lags with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1.0:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        threshold = max_val * 0.998
        max_lags = np.where(conv >= threshold)[0]

        # Compute subgradient (average over active lags)
        subgrad = np.zeros(n)
        for lag in max_lags:
            for i in range(n):
                j = lag - i
                if 0 <= j < n:
                    subgrad[i] += 2.0 * a[j]
        subgrad /= len(max_lags)

        gn = np.linalg.norm(subgrad)
        if gn < 1e-15:
            break
        subgrad = subgrad / gn

        # Line search
        improved = False
        for step in [0.0002, 0.0005, 0.001, 0.002, 0.005, 0.01, 0.02]:
            a_try = a - step * subgrad
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            # Small random nudge
            a = a + 0.005 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            if it % 50 == 0 and it > 0:
                # Recover if drifted too far
                if compute_obj(a) > best_obj * 1.01:
                    a = best_a.copy()

        if verbose and it % 50 == 0:
            print(f"    ActiveSet it{it:3d}: obj={best_obj:.10f}, active={len(max_lags)}")

    return best_a
