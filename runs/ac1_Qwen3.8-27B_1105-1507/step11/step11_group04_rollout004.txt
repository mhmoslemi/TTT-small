"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses improved LP alternating minimization with continuous coordinate descent,
subgradient polishing, and multi-restart exploration with adaptive length.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None, verbose=False):
    """
    Solve LP: min M s.t. (a_norm*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
            M.value = float(np.max(A @ warm_start))
        except Exception:
            pass

    solvers = ['CLARABEL', 'SCS', 'ECOS', 'MOSEK', 'PDLP']
    for solver_name in solvers:
        try:
            solver = getattr(cp, solver_name, None)
            if solver is None:
                continue
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s_g = float(np.sum(g_val))
                if s_g > 1e-10:
                    g_val = g_val / s_g
                    M_val = float(np.max(A @ g_val))
                    return g_val, M_val
        except Exception:
            continue
    return None, None


def coordinate_descent_continuous(a, time_budget, seed=42, verbose=False):
    """
    Continuous coordinate descent on normalized sequence (sum=1).
    For each coordinate, performs 1D minimization using sampling.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)
    mean_val = 1.0 / n

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    sweep = 0
    no_improve = 0
    max_no_improve = 3

    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            a_j = a[j]
            # Determine range of k affected
            k_start = j  # minimum k where a[j] participates
            k_end = min(j + n - 1, 2 * n - 2)  # maximum k

            # Unaffected max
            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))

            # Sample candidate x values
            # The optimal x is likely near the current value or near mean
            candidates = np.concatenate([
                np.linspace(0, min(0.1, max(a_j * 3, mean_val * 5)), 15),
                np.linspace(max(0, a_j * 0.3), max(a_j * 3, mean_val * 4), 20),
                np.array([0, mean_val, a_j])
            ])
            candidates = np.unique(candidates[(candidates >= 0) & (candidates <= 0.5)])
            
            if len(candidates) == 0:
                candidates = np.array([0.0, mean_val])

            S_prime = 1.0 - a_j  # sum of all other coordinates
            best_trial_obj = best_obj
            best_trial_val = a_j

            for x in candidates:
                S_new = S_prime + x
                if S_new <= 1e-10:
                    continue
                # Compute affected conv entries
                # conv_new[k] = (conv[k] + delta_k(x)) / S_new^2
                # delta_k(x) = 2*(x - a_j)*a[k-j] for k-j in [0,n-1], 2j!=k
                #            = (x^2 - a_j^2) for k = 2j
                k_indices = np.arange(k_start, k_end + 1)
                # For each k, the offset is k - j which is in [0, n-1]
                offsets = k_indices - j  # these are in [0, n-1]
                affected_conv = conv[k_start:k_end + 1].copy()
                
                # Delta for each affected k
                delta = 2.0 * (x - a_j) * a[offsets]
                # Handle the case where 2j = k (i.e., offset = j)
                mask_diag = (offsets == j)
                if np.any(mask_diag):
                    delta[mask_diag] = x * x - a_j * a_j
                
                affected_new = (affected_conv + delta) / (S_new * S_new)
                max_affected = float(np.max(affected_new))
                max_total = max(max_unaffected / (S_new * S_new) if S_new != 1.0 else max_unaffected, max_affected)
                
                # Wait, the unaffected entries also need to be divided by S_new^2
                # Because the whole conv is scaled by 1/S_new^2
                max_total = max(max_unaffected, float(np.max(affected_conv + delta))) / (S_new * S_new)
                
                obj_trial = 2.0 * n * max_total

                if obj_trial < best_trial_obj - 1e-14:
                    best_trial_obj = obj_trial
                    best_trial_val = x

            if best_trial_val != a_j:
                # Apply update
                x = best_trial_val
                S_new = S_prime + x
                delta = x - a_j
                
                # Update conv in place (unnormalized)
                k_indices = np.arange(k_start, k_end + 1)
                offsets = k_indices - j
                delta_conv = 2.0 * delta * a[offsets]
                mask_diag = (offsets == j)
                if np.any(mask_diag):
                    delta_conv[mask_diag] = x * x - a_j * a_j
                conv[k_start:k_end + 1] += delta_conv
                
                # Normalize
                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S
                a[j] = x * inv_S  # Make sure a[j] is correct
                
                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")

    return best_a


def subgradient_polish(a, time_budget, seed=42, verbose=False):
    """
    Projected subgradient method to minimize max(conv) on the simplex.
    Used as a polishing step to escape CD stagnation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    step_size = 0.01 / np.sqrt(n)
    decay = 0.999
    momentum = np.zeros(n)
    mu = 0.5  # momentum parameter
    
    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        if step_size < 1e-10:
            break
            
        # Compute max and its location(s)
        max_val = float(np.max(conv))
        active = np.where(conv >= max_val - 1e-8 * max_val)[0]
        
        # Subgradient: average gradient over active set
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    grad[i] += 2.0 * a[ki]
        grad /= len(active)
        
        # Add small random perturbation to escape plateaus
        grad += rng.standard_normal(n) * step_size * 0.1
        
        # Momentum update
        momentum = mu * momentum + (1 - mu) * grad
        a_new = a - step_size * momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()
        
        conv = compute_conv(a_new)
        obj_new = 2.0 * n * float(np.max(conv))
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            step_size *= 1.01  # Slightly increase on improvement
        else:
            step_size *= decay
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"    SG iter {it}: obj={best_obj:.10f}, step={step_size:.6f}, time={elapsed:.1f}s")
    
    return best_a


def lp_alternate_improved(a, time_budget, seed=42, verbose=True):
    """
    LP alternating minimization with adaptive step size, warm starts,
    and periodic CD micro-steps.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    iter_count = 0
    stall = 0
    lp_times = []

    while time.monotonic() - start < time_budget - 5:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        t0 = time.monotonic()
        g_lp, M_lp = solve_lp(a, warm_start=prev_g)
        lp_time = time.monotonic() - t0
        lp_times.append(lp_time)
        
        if g_lp is None:
            break
        prev_g = g_lp
        iter_count += 1

        # Check if we're at a fixed point
        similarity = float(np.dot(a, g_lp))
        if similarity > 1.0 - 1e-8:
            if verbose:
                print(f"    LP {iter_count}: fixed point reached, obj={best_obj:.10f}")
            break

        # Adaptive line search
        best_local_obj = best_obj
        best_local_a = a.copy()

        # Generate t values based on LP time and current state
        base_t = min(1.0, max(0.01, 5.0 / (iter_count + 10)))
        t_values = sorted(set([
            1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01,
            base_t, base_t * 0.5, base_t * 2.0
        ]))
        t_values = [t for t in t_values if 0 < t <= 1.0]

        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a.copy()
            stall = 0
            if verbose and iter_count % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP {iter_count}: obj={best_obj:.10f}, sim={similarity:.6f}, t_lp={lp_time:.2f}s, elapsed={elapsed:.0f}s")
        else:
            stall += 1
            if stall >= 5:
                # Try perturbation strategies
                strategies = [
                    # Fourier perturbation
                    lambda: _fourier_perturb(best_a, rng, 0.008),
                    # Gaussian perturbation
                    lambda: _gaussian_perturb(best_a, rng, 0.005),
                    # Random coordinate shuffle
                    lambda: _shuffle_perturb(best_a, rng),
                ]
                for strat in strategies:
                    a_pert = strat()
                    s_pert = float(np.sum(a_pert))
                    if s_pert > 1e-10:
                        a_pert = a_pert / s_pert
                        obj_pert = compute_obj_norm(a_pert)
                        if obj_pert < best_obj:
                            a = a_pert
                            best_a = a_pert
                            best_obj = obj_pert
                            break
                        # Even if not better, use as new starting point
                        a = a_pert
                    prev_g = None
                stall = 0

        # Periodic micro-CD (every 10 LP iterations, spend up to 2s)
        if iter_count % 10 == 0:
            cd_budget = min(2.0, (time_budget - (time.monotonic() - start)) * 0.02)
            if cd_budget > 0.5:
                a_cd = coordinate_descent_continuous(a, time_budget=cd_budget, seed=seed + iter_count, verbose=False)
                obj_cd = compute_obj_norm(a_cd)
                if obj_cd < best_obj - 1e-12:
                    best_obj = obj_cd
                    best_a = a_cd.copy()
                    a = a_cd.copy()
                    if verbose:
                        print(f"    LP {iter_count} + CD: obj={best_obj:.10f}")

    if verbose:
        avg_lp = np.mean(lp_times[-20:]) if lp_times else 0
        print(f"    LP done: {iter_count} iters, best={best_obj:.10f}, avg_lp_time={avg_lp:.2f}s")
    return best_a


def _fourier_perturb(a, rng, strength):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 10):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _gaussian_perturb(a, rng, strength):
    """Add Gaussian perturbation."""
    n = len(a)
    pert = rng.standard_normal(n) * strength
    a_new = a + pert * a
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _shuffle_perturb(a, rng):
    """Random coordinate exchange perturbation."""
    n = len(a)
    a_new = a.copy()
    n_swaps = max(5, n // 100)
    for _ in range(n_swaps):
        i, j = rng.integers(0, n, size=2)
        if i != j:
            diff = (a_new[j] - a_new[i]) * 0.1
            a_new[i] += diff
            a_new[j] -= diff
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def structured_init(n, seed=0):
    """Generate structured initial sequences for exploration."""
    rng = np.random.default_rng(seed)
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Quadratic residue inspired
    x = np.arange(n)
    a = 1.0 + 0.3 * np.sin(2 * np.pi * x / n) * np.cos(4 * np.pi * x / n)
    a = np.maximum(a, 0)
    inits.append(a / a.sum())
    
    # Periodic pattern
    period = max(n // 7, 3)
    pattern = np.ones(period)
    pattern[period // 2] = 0.5
    a = np.tile(pattern, n // period + 1)[:n]
    inits.append(a / a.sum())
    
    # Random smooth
    a = np.exp(rng.standard_normal(n) * 0.3)
    # Smooth it
    kernel = np.ones(5) / 5
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    inits.append(a / a.sum())
    
    return inits
