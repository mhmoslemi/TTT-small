        #   If k-i is in the valid range for j: a_{k-i} (this gives a_i for j=k-i)
        #   So total: a_{k-i} + a_{k-i} = 2*a_{k-i} if i != k-i
        #   Or a_i + a_i = 2*a_i if i == k-i
        #   Both can be written as 2*a_{k-i} (since when i=k-i, a_{k-i}=a_i)
        #   But only if 0 <= k-i < n
        # So: d/d(a_i) conv(a,a)_k = 2*a_{k-i} if 0 <= k-i < n, else 0
        
        # Wait, that's not right either. Let me be very explicit.
        # conv(a,a)_k = sum_{j} a_j * a_{k-j} where 0 <= j < n and 0 <= k-j < n
        # = sum_{j: max(0, k-n+1) <= j <= min(k, n-1)} a_j * a_{k-j}
        # 
        # d/d(a_i) of this sum:
        # For each term a_j * a_{k-j}:
        #   d/d(a_i) [a_j * a_{k-j}] = delta(j,i)*a_{k-j} + a_j*delta(k-j,i)
        #   = delta(j,i)*a_{k-j} + a_j*delta(k-i,j)
        # Summing over j:
        #   = a_{k-i} (if 0<=k-i<n and i is valid j) + a_{k-i} (if 0<=k-i<n and k-i is valid j)
        #   Wait, I'm getting confused. Let me just do it directly.
        
        # For a specific i, the terms in conv(a,a)_k that involve a_i are:
        # 1. When j = i: term is a_i * a_{k-i} (requires 0 <= k-i < n)
        # 2. When k-j = i, i.e., j = k-i: term is a_{k-i} * a_i (requires 0 <= k-i < n)
        # If i != k-i (k != 2i): both terms exist and are distinct, total = 2 * a_i * a_{k-i}
        #   d/d(a_i) = a_{k-i} + a_{k-i} = 2 * a_{k-i}
        # If i == k-i (k = 2i): only one term, a_i * a_i = a_i^2
        #   d/d(a_i) = 2 * a_i
        # In both cases, if 0 <= k-i < n: d/d(a_i) = 2 * a_{k-i}
        # (Since when k=2i, a_{k-i} = a_i, so 2*a_{k-i} = 2*a_i)
        # If k-i < 0 or k-i >= n: d/d(a_i) = 0
        
        # Great, so the formula is simply:
        # d/d(a_i) conv(a,a)_k = 2 * a_{k-i} if 0 <= k-i < n, else 0
        
        # Let me redo the gradient computation more efficiently
        grad = np.zeros(n)
        for k in active:
            # For each i in [0, n-1], if 0 <= k-i < n, add 2*a[k-i]
            # This means i ranges from max(0, k-n+1) to min(k, n-1)
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_start <= i_end:
                # indices k-i for i in [i_start, i_end] are k-i_start to k-i_end
                # = k - max(0,k-n+1) to k - min(k,n-1)
                # Let's just compute directly
                idx = i_start
                while idx <= i_end:
                    grad[idx] += 2.0 * a[k - idx]
                    idx += 1
        grad /= len(active)
        
        # The full objective subgradient (accounting for normalization)
        # f(a) = 2n * max_k conv(a,a)_k / (sum(a))^2
        # df/da_i = 2n * [d(max_k conv_k)/da_i * (sum a)^2 - max_k conv_k * 2*(sum a)] / (sum a)^4
        # = 2n * [g_i / (sum a) - 2 * max_k conv_k * a_i / (sum a)^2]
        # where g_i is the subgradient of max_k conv_k
        
        # Actually, since we normalize after each step, let's work with the
        # simplified objective F(a) = max_k conv(a,a)_k subject to sum(a)=1, a>=0
        # and use the constraint qualification to get the effective subgradient.
        # The projected subgradient method handles this via the projection.
        
        # For the projected method, we just need the gradient of max_k conv_k
        # (the constraint is handled by projection)
        
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break
        
        # Step size: Barzilai-Borwein
        if prev_g is not None and it % 5 == 0:
            s_vec = a - prev_a
            y_vec = grad - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-6), 1.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb
        
        # Decay learning rate
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        lr = lr * (1.0 - 0.3 * progress)
        lr = max(lr, 1e-8 / np.sqrt(n))
        
        # Take step
        a_new = project_simplex(a - lr * grad)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            patience = 0
        else:
            patience += 1
            # Backtracking
            t = lr
            found = False
            for _ in range(10):
                t *= 0.5
                a_try = project_simplex(a - t * grad)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    best_it = it
                    a_new = a_try
                    found = True
                    break
            if found:
                patience = 0
        
        prev_a = a.copy()
        prev_g = grad.copy()
        a = a_new
        it += 1
        
        if patience > 5000:
            # Reset with perturbation
            rng = np.random.default_rng(it)
            a = project_simplex(a + 0.01 * rng.standard_normal(n))
            prev_a = a.copy()
            prev_g = None
            lr = 1.0 / np.sqrt(n)
            patience = 0
        
        if verbose and (it % 200 == 0 or it == 1):
            elapsed = time.monotonic() - start
            print(f"    Psubgrad it{it}: obj={best_obj:.8f}, lr={lr:.6f}, "
                  f"|active|={len(active)}, time={elapsed:.0f}s")
    
    return best_a


def lp_direction(a):
    """Solve LP to find direction g that reduces max conv(a,g)."""
    n = len(a)
    a_norm = a / np.sum(a)
    conv_aa = compute_conv(a_norm)
    
    # Build matrix A: (A @ g)[k] = conv(a,g)[k]
    # conv(a,g)[k] = sum_j a[k-j] * g[j] for valid j
    # So A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A[k, j] = a_norm[k - j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def coordinate_descent_polish(a, time_budget=60.0, n_samples_per_coord=20):
    """Coordinate descent: for each coordinate, try several values."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Random order
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - start > time_budget - 0.5:
                break
            
            a_save = a[i]
            a_save_norm = float(np.sum(a))
            
            # Try several multipliers
            for mult in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.7, 2.0]:
                a_try = a.copy()
                a_try[i] = a_save * mult
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            
            it += 1
        
        if not improved and it > n * 3:
            break
    
    return best_a


def mirror_descent_optimize(a, time_budget=200.0, tau_init=50.0, tau_end=2000.0, verbose=False):
    """Mirror descent with adaptive temperature."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break
        
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.5
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = tau * (conv - max_conv)
        shifted = shifted - shifted.max()
        exp_vals = np.exp(shifted)
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(2 * n - 1)
            p[int(np.argmax(conv))] = 1.0
        else:
            p = exp_vals / sum_exp
        
        # Compute gradient from p
        grad = np.zeros(n)
        for i in range(n):
            # grad[i] = 2 * sum_k p[k] * a[k-i] (if 0<=k-i<n)
            k_start = max(0, i)  # k-i >= 0 => k >= i
            k_end = min(2 * n - 2, i + n - 1)  # k-i < n => k < i+n
            if k_start <= k_end:
                grad[i] = 2.0 * float(np.dot(p[k_start:k_end+1], a[k_start-i:k_end+1-i]))
        
        g_max = float(np.max(np.abs(grad)))
        if g_max < 1e-15:
            break
        
        eta = 2.0 / (g_max + 1e-10)
        eta *= (1.0 + 0.5 * (1.0 - progress))
        
        exponent = -eta * grad
        exponent = np.clip(exponent, -50, 50)
        a_new = a * np.exp(exponent)
        s_new = float(np.sum(a_new))
        if s_new < 1e-300:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 50:
                a_pert = a * np.exp(0.02 * rng.standard_normal(n))
                a_pert = np.maximum(a_pert, 0)
                s_pert = float(np.sum(a_pert))
                if s_pert > 1e-10:
                    a = a_pert / s_pert
                else:
                    a = best_a.copy()
                stall = 0
        
        a = a_new
        it += 1
        
        if verbose and (it % 100 == 0):
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, time={time.monotonic()-start:.0f}s")
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fast_subgradient(a, conv, active_set):
    """Fast computation of subgradient from active set using convolution."""
    n = len(a)
    # The subgradient of max_k conv(a,a)_k at a is (1/|S|) * sum_{k in S} d conv_k / da
    # d conv_k / da_i = 2 * a[k-i] if 0 <= k-i < n
    # This is equivalent to: (1/|S|) * 2 * (rev_a * indicator_S)[i]
    # where indicator_S is the indicator of the active set (as a vector of length 2n-1)
    
    # More precisely: grad[i] = (2/|S|) * sum_{k in S, 0<=k-i<n} a[k-i]
    # Let j = k-i, so k = j+i, and we need j in [0,n) and k=j+i in S
    # grad[i] = (2/|S|) * sum_{j: 0<=j<n, j+i in S} a[j]
    # = (2/|S|) * (a * indicator_S_shifted)[i]
    
    # indicator_S as a vector of length 2n-1
    ind = np.zeros(2 * n - 1)
    for k in active_set:
        ind[k] = 1.0
    
    # conv(a, ind) gives sum_j a[j] * ind[k-j]
    # We want sum_j a[j] * ind[j+i] for each i
    # This is cross-correlation of a with ind, evaluated at shift i
    # = conv(a, ind_reversed)[i + (2n-2)] ... hmm, let me think
    
    # Actually: sum_{j=0}^{n-1} a[j] * ind[j+i]
    # Let's compute conv(a, ind) = sum_j a[j] * ind[k-j]
    # At k = i: sum_j a[j] * ind[i-j] -- not quite what we want
    # We want sum_j a[j] * ind[j+i]
    # = sum_j a[j] * ind[(2n-2) - (2n-2-j-i)]
    # = (conv(a, ind_rev))[(2n-2) - (2n-2-j-i)]... this is getting complicated
    
    # Let me just use the direct approach with FFT
    # grad[i] = (2/|S|) * sum_{j: j+i in [0,2n-2]} a[j] * ind[j+i]
    # = (2/|S|) * (cross_correlation of a and ind)[i]
    # where cross_correlation[x] = sum_j a[j] * ind[j+x]
    
    # cross_correlation can be computed as:
    # = convexolution of a and reversed ind
    # If ind_rev = ind[::-1], then conv(a, ind_rev)[k] = sum_j a[j] * ind_rev[k-j] = sum_j a[j] * ind[(2n-2)-(k-j)] = sum_j a[j] * ind[2n-2-k+j]
    # We want sum_j a[j] * ind[j+i], so 2n-2-k+j = j+i => k = 2n-2-i
    # So cross_correlation[i] = conv(a, ind_rev)[2n-2-i]
    
    ind_rev = ind[::-1]
    size = 1
    while size < 4 * n:
        size <<= 1
    fa = rfft(a, size)
    fi = rfft(ind_rev, size)
    c = irfft(fa * fi, size)
    # cross_correlation[i] = c[2n-2-i] for i in [0, n-1]
    grad = (2.0 / len(active_set)) * c[2*n-2-n+1:2*n-1][::-1]
    # Let me double check: for i=0, we need c[2n-2], for i=n-1, we need c[n-1]
    # So grad[i] = c[2n-2-i]
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = (2.0 / len(active_set)) * c[2*n-2-i]
    
    return grad


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: projected subgradient + LP + coordinate descent + mirror descent + multi-n.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Initialize from parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")
    
    n_current = len(best_seq)
    
    # === Phase 1: Projected Subgradient (main) ===
    print(f"\n=== Phase 1: Projected Subgradient (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 400.0)
    if p1_budget > 30:
        a_ps = projected_subgradient_optimize(best_seq, time_budget=p1_budget, verbose=True)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  Phase1 Psubgrad: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 Psubgrad: obj={obj_ps:.8f}")
    
    # === Phase 2: LP-Driven ===
    print(f"\n=== Phase 2: LP-Driven ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 250.0)
    if p2_budget > 20:
        n_lp = len(best_seq)
        a_lp = best_seq.copy()
        lp_start = time.monotonic()
        lp_it = 0
        while time.monotonic() - lp_start < p2_budget:
            if time.monotonic() - lp_start > p2_budget - 5:
                break
            g_dir = lp_direction(a_lp)
            if g_dir is None:
                break
            
            # Line search
            best_t = 0.0
            best_obj_t = compute_obj(a_lp)
            for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
                a_try = np.maximum((1-t) * a_lp + t * g_dir, 0)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    obj_try = compute_obj(a_try / s_try)
                    if obj_try < best_obj_t:
                        best_obj_t = obj_try
                        best_t = t
            
            if best_t > 0:
                a_try = np.maximum((1-best_t) * a_lp + best_t * g_dir, 0)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    if compute_obj(a_try) < best_score:
                        best_score = compute_obj(a_try)
                        best_seq = a_try.copy()
                    a_lp = a_try
                    lp_it += 1
                    if lp_it % 5 == 0:
                        print(f"    LP it{lp_it}: obj={best_score:.8f}, t={best_t:.3f}")
            else:
                # No improvement, perturb
                a_lp = a_lp + 0.005 * rng.standard_normal(n_lp)
                a_lp = np.maximum(a_lp, 0)
                s_lp = float(np.sum(a_lp))
                if s_lp > 1e-10:
                    a_lp = a_lp / s_lp
        
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f}")
    
    # === Phase 3: Mirror Descent ===
    print(f"\n=== Phase 3: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    if p3_budget > 20:
        a_md = mirror_descent_optimize(best_seq, time_budget=p3_budget,
                                       tau_init=100.0, tau_end=3000.0, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
