import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    """Compute linear convolution a * a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute the objective 2n * max(conv) / sum^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    # Algorithm from Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def grad_active(a, k_max):
    """Compute gradient of (a*a)(k_max) with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        j1 = k_max - i
        j2 = i - k_max
        if 0 <= j1 < n:
            grad[i] += a[j1]
        if 0 <= j2 < n:
            grad[i] += a[j2]
    return grad


def nesterov_optimize(a_init, time_budget=60.0, verbose=False):
    """Nesterov accelerated gradient on the active convolution constraint."""
    n = len(a_init)
    a = project_simplex(a_init.copy())
    y = a.copy()  # Nesterov momentum variable
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adaptive learning rate
    # The gradient of (a*a)(k) has norm bounded by ~2*sum(a) = 2 (when sum=1)
    # More precisely, ||grad||_2 <= 2*||a||_2 <= 2 (since ||a||_2 <= ||a||_1 = 1 on simplex)
    # So Lipschitz constant L ~ 4 (from Hessian eigenvalues)
    L = 4.0  # Initial estimate
    t_k = 1.0  # Nesterov step counter
    
    it = 0
    stale_count = 0
    while time.monotonic() - start < time_budget:
        batch_check = (it % 50 == 0) or (time.monotonic() - start > time_budget - 2)
        
        conv = compute_conv(y)
        k_max = int(np.argmax(conv))
        grad = grad_active(y, k_max)
        
        # Step
        a_new = y - grad / L
        a_new = project_simplex(a_new)
        
        # Nesterov update
        t_prev = t_k
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_prev - 1.0) / t_k
        y = a_new + beta * (a_new - a)
        y = project_simplex(y)
        a = a_new
        
        if batch_check:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                stale_count = 0
            else:
                stale_count += 1
                if stale_count > 5:
                    # Adaptive L adjustment
                    L *= 0.9
                    if L < 0.5:
                        L = 0.5
                    y = best_a.copy()
                    a = best_a.copy()
                    t_k = 1.0
                    stale_count = 0
        it += 1
        
        if verbose and it % 500 == 0:
            obj = compute_obj(a)
            elapsed = time.monotonic() - start
            print(f"    Nesterov it={it}: obj={obj:.8f}, best={best_obj:.8f}, L={L:.3f}, elapsed={elapsed:.1f}s")
    
    # Final evaluation
    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()
    
    return best_a, best_obj

def build_conv_matrix_for_lp(a):
    """Build matrix A such that A @ g = a * g (cross-convolution)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # (a*g)[k] = sum_j a[j] * g[k-j]
        # So A[k, j] = a[j] if 0 <= k-j < n, i.e., j <= k < j+n
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix_for_lp(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def interpolate_sequence(a_src, n_dst):

    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:

    return project_simplex(a_dst)

Improved step function optimization with corrected gradient, Nesterov acceleration,
LP refinement, diverse asymmetric initializations, and systematic n exploration.
Key fix: correct gradient of convolution peak (was wrong in previous version).
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Euclidean projection onto probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cssv[rho[-1]] - 1) / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0)


def grad_conv_at(x, k):
    """Correct gradient of (x*x)(k) w.r.t. x. Result: 2*x[k-i] for valid i."""
    n = len(x)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    grad = np.zeros(n)
    if lo <= hi:
        grad[lo:hi + 1] = 2.0 * x[k - hi:k - lo + 1][::-1]
    return grad


def fixed_gradient_descent(a_init, time_budget=60.0, step_size=0.3, momentum=0.85, verbose=False):
    """Projected gradient with momentum using CORRECT gradient."""
    n = len(a_init)
    x = project_simplex(np.maximum(a_init, 0).astype(np.float64))
    v = np.zeros(n)
    best_x = x.copy()
    best_obj = compute_obj(x)
    start = time.monotonic()
    it = 0
    stale = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(x)
        k_max = int(np.argmax(conv))
        grad = grad_conv_at(x, k_max)

        # Momentum update
        v = momentum * v - step_size * grad
        x_new = x + v
        x_new = project_simplex(x_new)

        # Periodic evaluation
        if it % 10 == 0:
            obj = compute_obj(x_new)
            if obj < best_obj:
                best_obj = obj
                best_x = x_new.copy()
                x = x_new.copy()
                stale = 0
            else:
                stale += 1
                if stale > 20:
                    x = best_x.copy()
                    v = np.zeros(n)
                    step_size *= 0.85
                    if step_size < 0.01:
                        step_size = 0.3
                        momentum = 0.5
                    stale = 0
        else:
            x = x_new.copy()

        it += 1
        if verbose and it % 500 == 0:
            obj = compute_obj(x)
            elapsed = time.monotonic() - start
            print(f"    GD it={it}: obj={obj:.8f}, best={best_obj:.8f}, ss={step_size:.4f}, t={elapsed:.1f}s")

    obj = compute_obj(x)
    if obj < best_obj:
        best_obj = obj
        best_x = x.copy()
    return best_x, best_obj


def nesterov_descent(a_init, time_budget=60.0, L=2.5, verbose=False):
    """Nesterov accelerated gradient with correct gradient and simplex projection."""
    n = len(a_init)
    x = project_simplex(np.maximum(a_init, 0).astype(np.float64))
    y = x.copy()
    best_x = x.copy()
    best_obj = compute_obj(x)
    start = time.monotonic()
    t_k = 1.0
    it = 0
    stale = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        k_max = int(np.argmax(conv))
        grad = grad_conv_at(y, k_max)

        # Gradient step
        x_new = project_simplex(y - grad / L)

        # Nesterov momentum
        t_prev = t_k
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_prev - 1.0) / t_k
        y = project_simplex(x_new + beta * (x_new - x))
        x = x_new

        if it % 15 == 0:
            obj = compute_obj(x)
            if obj < best_obj:
                best_obj = obj
                best_x = x.copy()
                stale = 0
            else:
                stale += 1
                if stale > 8:
                    L *= 0.9
                    if L < 0.8:
                        L = 2.5
                        y = best_x.copy()
                        x = best_x.copy()
                        t_k = 1.0
                    else:
                        y = best_x.copy()
                stale = 0
        it += 1

        if verbose and it % 200 == 0:
            obj = compute_obj(x)
            elapsed = time.monotonic() - start
            print(f"    Nesterov it={it}: obj={obj:.8f}, best={best_obj:.8f}, L={L:.3f}, t={elapsed:.1f}s")

    obj = compute_obj(x)
    if obj < best_obj:
        best_obj = obj
        best_x = x.copy()
    return best_x, best_obj


def multi_peak_descent(a_init, time_budget=60.0, num_peaks=3, step_size=0.15, verbose=False):
    """Gradient descent using average of top-k peak gradients for stability."""
    n = len(a_init)
    x = project_simplex(np.maximum(a_init, 0).astype(np.float64))
    v = np.zeros(n)
    momentum = 0.8
    best_x = x.copy()
    best_obj = compute_obj(x)
    start = time.monotonic()
    it = 0
    stale = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(x)
        top_k = np.argsort(conv)[::-1][:num_peaks]

        grad = np.zeros(n)
        for k_act in top_k:
            grad += grad_conv_at(x, int(k_act))
        grad /= num_peaks

        lr = step_size / (1.0 + 0.005 * it)
        v = momentum * v - lr * grad
        x_new = project_simplex(x + v)

        if it % 10 == 0:
            obj = compute_obj(x_new)
            if obj < best_obj:
                best_obj = obj
                best_x = x_new.copy()
                x = x_new.copy()
                stale = 0
            else:
                stale += 1
                if stale > 15:
                    x = best_x.copy()
                    v = np.zeros(n)
                    stale = 0
        else:
            x = x_new.copy()

        it += 1

    obj = compute_obj(x)
    if obj < best_obj:
        best_obj = obj
        best_x = x.copy()
    return best_x, best_obj


def build_cross_conv_matrix(a):
    """Build matrix A such that A @ g gives the cross-convolution a * g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_cross_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_refine(a_init, time_budget=100.0, max_iters=100, verbose=False):
    """LP-based iterative refinement."""
    n = len(a_init)
    a = project_simplex(np.maximum(a_init, 0).astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        # Line search
        best_t = 0.0
        best_obj_try = best_obj
        for t in np.concatenate([np.linspace(0, 1.0, 15), np.linspace(0.5, 1.5, 10)]):
            a_try = project_simplex((1 - t) * a + t * g_lp)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_try:
                best_obj_try = obj_try
                best_t = t

        if best_t > 1e-6 and best_obj_try < best_obj - 1e-14:
            a_new = project_simplex((1 - best_t) * a + best_t * g_lp)
            best_obj = compute_obj(a_new)
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            obj_direct = compute_obj(g_lp)
            if obj_direct < best_obj:
                best_obj = obj_direct
                best_a = g_lp.copy()
                a = g_lp.copy()
                stall = 0
            elif stall >= 3:
                freq = rng.integers(1, max(2, n // 80))
                phase = rng.uniform(0, 2 * np.pi)
                idx = np.arange(n, dtype=np.float64)
                pert = 0.015 * stall * np.cos(2 * np.pi * freq * idx / n + phase)
                a = project_simplex(best_a * np.exp(pert))
                stall = 0
        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"      LP it{it}: obj={best_obj:.8f}, stall={stall}, t={elapsed:.1f}s")

    return best_a, best_obj


def make_monotone_init(n, shape):
    """Create monotone asymmetric initializations."""
    if shape == 'exp1':
        a = np.exp(np.linspace(0, 1.5, n))
    elif shape == 'exp2':
        a = np.exp(np.linspace(0, 3.0, n))
    elif shape == 'exp3':
        a = np.exp(np.linspace(0, 5.0, n))
    elif shape == 'poly2':
        x = np.linspace(0, 1, n)
        a = x ** 2 + 0.01
    elif shape == 'poly3':
        x = np.linspace(0, 1, n)
        a = x ** 3 + 0.005
    elif shape == 'sqrt':
        x = np.linspace(0, 1, n)
        a = np.sqrt(x) + 0.01
    elif shape == 'beta_r':
        x = np.linspace(0.005, 0.995, n)
        a = x ** 2 * (1 - x) ** 4 + 0.001
    elif shape == 'beta_l':
        x = np.linspace(0.005, 0.995, n)
        a = x ** 4 * (1 - x) ** 2 + 0.001
    elif shape == 'linear':
        a = np.linspace(0.01, 1.0, n)
    elif shape == 'cos':
        x = np.linspace(0, 1, n)
        a = np.cos(x * np.pi / 2) ** 1.5 + 0.01
    elif shape == 'recip':
        x = np.linspace(0.05, 1.0, n)
        a = 1.0 / x
    else:
        a = np.exp(np.linspace(0, 2.0, n))
    a = np.maximum(a, 0)
    return project_simplex(a)

def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    if n_src == n_dst:
        return a_src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return project_simplex(a_dst)
