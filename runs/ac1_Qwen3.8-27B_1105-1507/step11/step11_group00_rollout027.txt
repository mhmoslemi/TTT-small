def compute_conv(a):
    n = len(a)

        return np.array([])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):

    if n == 0:

    if s < 0.01:

    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def subgradient_exact(a, k_star):
    """Exact subgradient when max is at index k_star."""
    n = len(a)
    # d(conv[k_star])/da[i] = 2*a[k_star - i] if 0 <= k_star - i < n
    grad = np.zeros(n)
    # For i in [0, n-1], k_star - i is in [k_star-(n-1), k_star]
    # We need 0 <= k_star - i < n, i.e., k_star - n + 1 <= i <= k_star
    i_low = max(0, k_star - n + 1)

    if i_low <= i_high:
        grad[i_low:i_high+1] = 2.0 * a[k_star - i_high : k_star - i_low - 1 : -1]
    return grad

    return grad

    v = np.zeros(n)  # velocity
    lr = 1.0 / np.sqrt(n)
    best_it = 0
    it = 0

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        tau = tau_init + (tau_end - tau_init) * progress ** 2

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:

            v = np.zeros(n)

        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1

def lp_direction(a, time_per_lp=5.0):
    """LP-based direction finding."""
    import cvxpy as cp
    n = len(a)
    a_norm = a / np.sum(a)
    conv_aa = compute_conv(a_norm)
    
    # Build constraint matrix
    A = np.zeros((2*n-1, n))
    k_idx = np.arange(2*n-1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A[valid] = a_norm[diff[valid]]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [2.0 * (A @ g) - M <= conv_aa, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None

def lp_refine(a, time_budget=100.0, verbose=False):
    """LP-based refinement with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    it = 0
    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 3:
            break
        
        g = lp_direction(a)
        if g is None:
            break
        
        # Line search
        best_t = 0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            a_try = np.maximum((1-t)*a + t*g, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try_norm = a_try / s_try
                obj_try = compute_obj(a_try_norm)
                if obj_try < best_obj_ls:
                    best_obj_ls = obj_try
                    best_t = t
        
        if best_t > 0:
            a_new = np.maximum((1-best_t)*a + best_t*g, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                best_obj = best_obj_ls
                best_a = a_new.copy()
                a = a_new
        
        it += 1
        if verbose and it % 5 == 0:
            print(f"  LP ref it{it}: obj={best_obj:.8f}, time={time.monotonic()-start:.0f}s")
        
        if best_t == 0:
            break
    
    return best_a

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a, left=0, right=0)

"""
Optimizes step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov accelerated projected subgradient with Polyak step size,
LSE smoothing, LP-based direction finding, and multi-n exploration.
Key improvements: exact subgradient at argmax for aggressive early steps,
FFT-accelerated LSE gradients, adaptive temperature scheduling,
asymmetric U-shaped restarts, and careful time management.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def subgradient_exact(a, k_star):
    """Exact subgradient of max conv when max is at index k_star."""
    n = len(a)
    grad = np.zeros(n)
    i_low = max(0, k_star - n + 1)
    i_high = min(n - 1, k_star)
    if i_low <= i_high:
        a_slice = a[k_star - i_high:k_star - i_low + 1][::-1]
        grad[i_low:i_high + 1] = 2.0 * a_slice
    return grad


def lse_gradient_fast(a, conv, tau):
    """Fast LSE gradient using FFT convolution trick."""
    n = len(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    shifted = shifted - shifted.max()
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()

    b = a[::-1]
    N_fft = 1
    while N_fft < 3 * n - 1:
        N_fft <<= 1
    c = irfft(rfft(b, N_fft) * rfft(p, N_fft), N_fft)
    return 2.0 * c[n - 1:2 * n - 1]


def subgradient_topk(a, conv, k=5):
    """Weighted subgradient at top-k peaks."""
    n = len(a)
    idx = np.argsort(conv)[-k:]
    weights = np.exp(100.0 * (conv[idx] - conv[idx].max()))
    total_w = float(np.sum(weights))
    if total_w < 1e-300:
        weights = np.ones(k) / k
    else:
        weights = weights / total_w
    grad = np.zeros(n)
    for w, idx_i in zip(weights, idx):
        if w > 1e-10:
            grad += w * subgradient_exact(a, int(idx_i))
    return grad


def psa_optimize(a, time_budget=300.0, tau_init=50.0, tau_end=3000.0,
                 verbose=True, seed=42):
    """
    Nesterov accelerated projected subgradient with adaptive LSE temperature
    and Polyak step size.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    beta = 0.8
    v = np.zeros(n)
    lr = 1.0 / np.sqrt(n)
    init_lr = lr
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        tau = tau_init + (tau_end - tau_init) * progress ** 1.8

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            v = np.zeros(n)
        else:
            y = y / s_y

        conv_y = compute_conv(y)

        # Choose gradient strategy based on progress
        if progress < 0.2:
            # Exact subgradient at argmax (aggressive)
            k_star = int(np.argmax(conv_y))
            g = subgradient_exact(y, k_star)
        elif progress < 0.5:
            # Blend exact and LSE
            k_star = int(np.argmax(conv_y))
            g_exact = subgradient_exact(y, k_star)
            g_lse = lse_gradient_fast(y, conv_y, tau)
            g = 0.6 * g_exact + 0.4 * g_lse
        elif progress < 0.8:
            # LSE with wider support
            g = lse_gradient_fast(y, conv_y, tau)
        else:
            # Very wide LSE (near-exact max)
            g = lse_gradient_fast(y, conv_y, max(tau, 5000.0))

        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break

        # Step size: blend Polyak and standard
        current_obj = compute_obj(y)
        f_lower = 2.0 * n / (2.0 * n - 1.0)
        eta_polyak = max(0.0, current_obj - f_lower) / (g_norm ** 2 + 1e-10)
        eta_std = lr / (g_norm + 1e-10)
        # Clip Polyak to avoid too-large steps
        eta = 0.4 * min(eta_polyak, 10.0 * eta_std) + 0.6 * eta_std
        eta = min(eta, 50.0 / (float(np.max(np.abs(g))) + 1e-10))

        # Update
        v = beta * v - eta * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
            lr = init_lr * 0.5
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        # Decay learning rate slowly
        lr = init_lr * (1.0 - 0.3 * progress)

        # Periodic perturbation to escape local minima
        gap = it - best_it
        if gap > 3000 and progress < 0.9:
            strength = 0.005 + 0.01 * progress
            a = a * np.exp(strength * rng.standard_normal(n))
            a = np.maximum(a, 0)
            s_p = float(np.sum(a))
            if s_p > 1e-10:
                a = a / s_p
            else:
                a = best_a.copy()
            v = np.zeros(n)
            lr = init_lr * 0.5
            best_it = it

        if verbose and (it == 1 or it % 500 == 0):
            print(f"    PSA it{it}: obj={best_obj:.8f}, tau={tau:.0f}, "
                  f"eta={eta:.6f}, gap={gap}, t={time.monotonic()-start:.0f}s")

    return best_a


def mirror_descent_opt(a, time_budget=200.0, tau_init=100.0, tau_end=5000.0,
                       verbose=False, seed=7):
    """Mirror descent (multiplicative weights) with LSE gradient."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.5

        conv = compute_conv(a)
        g = lse_gradient_fast(a, conv, tau)
        g_max = float(np.max(np.abs(g)))
        if g_max < 1e-15:
            break

        eta = 3.0 / (g_max + 1e-10)
        eta *= (1.0 + 0.5 * (1.0 - progress))

        # Multiplicative update
        exponent = -eta * g
        exponent = np.clip(exponent, -30, 30)
        a_new = a * np.exp(exponent)
        s_new = float(np.sum(a_new))
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Occasional perturbation
        if it % 100 == 50 and progress < 0.8:
            temp = 0.005 * (1.0 - progress)
            a_new = a_new + temp * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_n = float(np.sum(a_new))
            if s_n > 1e-10:
                a_new = a_new / s_n

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        else:
            # Try smaller step
            a_half = a * np.exp(-eta * 0.3 * g)
            a_half = np.maximum(a_half, 0)
            s_h = float(np.sum(a_half))
            if s_h > 1e-10:
                a_half = a_half / s_h
                obj_half = compute_obj(a_half)
                if obj_half < best_obj - 1e-14:
                    best_obj = obj_half
                    best_a = a_half.copy()
                    best_it = it
                    a = a_half
                else:
                    a = a_new
            else:
                a = a_new

        if it - best_it > 5000 and progress < 0.9:
            a = best_a * np.exp(0.01 * rng.standard_normal(n))
            a = np.maximum(a, 0)
            s_r = float(np.sum(a))
            if s_r > 1e-10:
                a = a / s_r
            best_it = it

        a = a_new
        it += 1

        if verbose and (it == 1 or it % 200 == 0):
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, "
                  f"t={time.monotonic()-start:.0f}s")

    return best_a


def lp_direction(a):
    """Solve LP for direction: min M s.t. 2*conv(a,g)_k <= conv(a,a)_k + M, sum(g)=1, g>=0."""
    import cvxpy as cp
    n = len(a)
    a_norm = a / np.sum(a)
    conv_aa = compute_conv(a_norm)

    A = np.zeros((2 * n - 1, n))
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A[valid] = a_norm[diff[valid]]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [2.0 * (A @ g) - M <= conv_aa, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def lp_refine(a, time_budget=100.0, verbose=False):
    """LP-based refinement with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget - 3.0:
        g = lp_direction(a)
        if g is None:
            break

        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            a_try = np.maximum((1 - t) * a + t * g, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try_n = a_try / s_try
                obj_try = compute_obj(a_try_n)
                if obj_try < best_obj_ls:
                    best_obj_ls = obj_try
                    best_t = t

        if best_t > 0:
            a_new = np.maximum((1 - best_t) * a + best_t * g, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                best_obj = best_obj_ls
                best_a = a_new.copy()
                a = a_new
        else:
            # Try with smaller learning rate on gradient
            conv = compute_conv(a)
            g_grad = lse_gradient_fast(a, conv, 500.0)
            g_grad = np.maximum(g_grad, 0)
            s_gg = float(np.sum(g_grad))
            if s_gg > 1e-10:
                g_grad = g_grad / s_gg
                for t in [0.02, 0.05, 0.1]:
                    a_try = np.maximum((1 - t) * a + t * g_grad, 0)
                    s_try = float(np.sum(a_try))
                    if s_try > 1e-10:
                        a_try_n = a_try / s_try
                        obj_try = compute_obj(a_try_n)
                        if obj_try < best_obj_ls:
                            best_obj_ls = obj_try
                            best_t = t
                            g = g_grad
                if best_t > 0:
                    a_new = np.maximum((1 - best_t) * a + best_t * g, 0)
                    s_new = float(np.sum(a_new))
                    if s_new > 1e-10:
                        a_new = a_new / s_new
                        best_obj = best_obj_ls
                        best_a = a_new.copy()
                        a = a_new
                    else:
                        break
                else:
                    break

        it += 1
        if verbose and it % 3 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, "
                  f"time={time.monotonic()-start:.0f}s")

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a, left=0, right=0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines PSA, mirror descent, LP refinement,
    multi-n exploration, and diverse restarts with careful time management.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Initialize from parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 1000
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * (x - 0.5)) ** 2 + 0.1
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = compute_obj(best_seq)
        print(f"U-shaped init (n={n_init}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Primary PSA ===
    print(f"\n=== Phase 1: PSA (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1 = min(remaining * 0.30, 450.0)
    if p1 > 50:
        a1 = psa_optimize(best_seq, time_budget=p1, tau_init=30, tau_end=5000,
                          verbose=True, seed=42)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1 PSA: {best_score:.8f} **")
        else:
            print(f"  Phase1 PSA: {obj1:.8f}")

    # === Phase 2: Mirror Descent ===
    print(f"\n=== Phase 2: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p2 = min(remaining * 0.20, 250.0)
    if p2 > 30:
        a2 = mirror_descent_opt(best_seq, time_budget=p2, tau_init=50, tau_end=3000,
                                verbose=True, seed=7)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 MD: {best_score:.8f} **")
        else:
            print(f"  Phase2 MD: {obj2:.8f}")

    # === Phase 3: LP Refinement ===
    print(f"\n=== Phase 3: LP Refinement ===")
    remaining = deadline - time.monotonic()
    p3 = min(remaining * 0.18, 200.0)
    if p3 > 20 and len(best_seq) <= 1500:
        a3 = lp_refine(best_seq, time_budget=p3, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3 LP: {best_score:.8f} **")
        else:
            print(f"  Phase3 LP: {obj3:.8f}")
    else:
        print(f"  Skipping LP (n too large or time insufficient)")

    # === Phase 4: PSA Round 2 (high tau) ===
    print(f"\n=== Phase 4: PSA Round 2 ===")
    remaining = deadline - time.monotonic()
    p4 = min(remaining * 0.18, 250.0)
    if p4 > 30:
        a4 = psa_optimize(best_seq, time_budget=p4, tau_init=500, tau_end=20000,
                          verbose=True, seed=123)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4 PSA2: {best_score:.8f} **")
        else:
            print(f"  Phase4 PSA2: {obj4:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    n_options = []
    for delta in [-200, -100, 100, 200, 400, 600]:
        n_try = n_cur + delta
        if 500 <= n_try <= 3000:
            n_options.append(n_try)
    # Add some absolute values
    for n_abs in [800, 1000, 1200, 1400, 1600, 2000]:
        if n_abs not in n_options:
            n_options.append(n_abs)
    n_options = sorted(set(n_options))
    # Pick at most 5 diverse ones
    if len(n_options) > 5:
        n_options = [n_options[i] for i in range(0, len(n_options), max(1, len(n_options) // 5))][:5]
    print(f"  Trying n: {n_options}")

    for n_try in n_options:

        if remaining < 70:
            break

        a_try = resample(best_seq, n_try)

        alloc_psa = min(remaining * 0.4, 60.0)
        if alloc_psa > 15:
            a_try = psa_optimize(a_try, time_budget=alloc_psa, tau_init=50,
                                  tau_end=2000, verbose=False, seed=n_try % 100)

        remaining = deadline - time.monotonic()
        alloc_md = min(remaining * 0.25, 30.0)
        if alloc_md > 10:
            a_try = mirror_descent_opt(a_try, time_budget=alloc_md, tau_init=100,
                                        tau_end=2000, verbose=False, seed=n_try % 100 + 50)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()

        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    restarts = []

    # U-shaped

    r1 = np.maximum(0.5 * best_seq + 0.5 * u_shape, 0)
    r1 = r1 / max(float(np.sum(r1)), 1e-10)
    restarts.append(("u_shape", r1))

    # Asymmetric tilt
    tilt = np.exp(0.15 * np.linspace(-1, 1, n_cur))
    r2 = best_seq * tilt
    r2 = np.maximum(r2, 0)
    r2 = r2 / max(float(np.sum(r2)), 1e-10)
    restarts.append(("tilt", r2))

    # W-shaped (two peaks)
    w_shape = 0.5 + 0.4 * np.cos(2 * np.pi * (x - 0.5))
    w_shape = np.maximum(w_shape, 0.01)
    w_shape = w_shape / max(float(np.sum(w_shape)), 1e-10)
    r3 = np.maximum(0.4 * best_seq + 0.6 * w_shape, 0)
    r3 = r3 / max(float(np.sum(r3)), 1e-10)
    restarts.append(("w_shape", r3))

    for name, a_r in restarts:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        alloc_psa = min(remaining * 0.3, 60.0)
        if alloc_psa > 15:
            a_r = psa_optimize(a_r, time_budget=alloc_psa, tau_init=50,
                                tau_end=2000, verbose=False, seed=hash(name) % 1000)

        remaining = deadline - time.monotonic()
        alloc_md = min(remaining * 0.15, 30.0)
        if alloc_md > 10:
            a_r = mirror_descent_opt(a_r, time_budget=alloc_md, tau_init=100,
                                      tau_end=1500, verbose=False, seed=hash(name) % 1000 + 500)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart ({name}): {best_score:.8f} **")
        else:
            print(f"  Restart ({name}): {obj_r:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Final PSA with very high tau (near-exact max)
    if remaining > 40:
        alloc = min(remaining * 0.45, 180.0)
        a_f = psa_optimize(best_seq, time_budget=alloc, tau_init=1000, tau_end=50000,
                           verbose=True, seed=999)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final PSA: {best_score:.8f} **")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 30 and len(best_seq) <= 1200:
        alloc = min(remaining * 0.35, 80.0)
        a_f2 = lp_refine(best_seq, time_budget=alloc, verbose=True)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final LP: {best_score:.8f} **")

    # Final MD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 60.0)
        a_f3 = mirror_descent_opt(best_seq, time_budget=alloc, tau_init=2000,
                                   tau_end=50000, verbose=False, seed=777)

        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
