    """Full linear convolution a*a via FFT."""
    n = len(a)

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def build_conv_matrix(a):

def parametric_init(n, mode, **params):
    """Generate parametric initializations."""
    x = np.linspace(0, 1, n)
    xm = x - 0.5  # centered
    
    if mode == 0:  # uniform
        a = np.ones(n)
    elif mode == 1:  # raised cosine^p
        p = params.get('p', 1.0)
        a = np.cos(np.pi * xm) ** (2 * p)
        a = np.maximum(a, 0)
    elif mode == 2:  # Gaussian
        sigma = params.get('sigma', 0.25)
        a = np.exp(-xm**2 / (2 * sigma**2))
    elif mode == 3:  # Parabolic
        a = 1 - 4 * xm**2
        a = np.maximum(a, 0)
    elif mode == 4:  # Quartic
        a = (1 - 4 * xm**2) ** 2
        a = np.maximum(a, 0)
    elif mode == 5:  # Beta distribution shape
        alpha = params.get('alpha', 2.0)
        # a[i] proportional to (i+1)^{alpha-1} * (n-i)^{alpha-1}
        i = np.arange(1, n+1)
        a = i**(alpha-1) * (n-i+1)**(alpha-1)
    elif mode == 6:  # Power law
        p = params.get('p', 2.0)
        a = np.cos(np.pi * xm) ** p
        a = np.maximum(a, 0)
    elif mode == 7:  # B-spline (cubic)
        # Cubic B-spline support on [-0.5, 0.5]
        t = np.abs(4 * xm)  # scaled to [-2, 2]
        a = np.zeros(n)
        mask1 = t < 1
        mask2 = (t >= 1) & (t < 2)
        a[mask1] = 1 - 1.5*t[mask1]**2 + 0.5*t[mask1]**3
        a[mask2] = 0.25 * (2 - t[mask2])**3
    elif mode == 8:  # Triangular
        a = 1 - 2 * np.abs(xm)
    elif mode == 9:  # Super-Gaussian
        p = params.get('p', 4.0)
        a = np.exp(-np.abs(xm/p)**p)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses a multi-phase strategy: (1) parametric family optimization with parameter
tuning, (2) LP-based iteration with optimal line search, (3) block coordinate LP
descent, (4) exact golden-section coordinate descent, (5) multi-n exploration with
resampling, (6) diverse restarts with local search, and (7) final CD polish.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def obj_value(a):
    """Compute 2n*max(conv)/sum^2 for a non-negative sequence."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n - j, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=30.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns normalized g."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15 or n < 2:
        return None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    problem = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            problem.solve(solver=solver, timeout=timeout)
            if g.value is not None and problem.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:
                    g_val = g_val / g_sum
                    return g_val
        except Exception:
            continue
    return None

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    num_blocks = max(1, (n + block_size - 1) // block_size)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        improved_any = False
        block_order = rng.permutation(num_blocks)

        for bi in block_order:
            if time.monotonic() - start >= time_budget:
                break
            rem = time_budget - (time.monotonic() - start)
            if rem < 4:
                break

            start_idx = bi * block_size
            end_idx = min(n, start_idx + block_size)
            bsize = end_idx - start_idx
            if bsize < 3:
                continue

            conv_a = compute_conv(a)
            A_full = build_conv_matrix(a)
            A_sub = A_full[:, start_idx:end_idx]
            base = conv_a - A_sub @ a[start_idx:end_idx]

            g_block = cp.Variable(bsize, nonneg=True)
            M = cp.Variable()
            target_sum = float(np.sum(a[start_idx:end_idx]))
            constraints = [base + A_sub @ g_block <= M, cp.sum(g_block) == target_sum]
            prob = cp.Problem(cp.Minimize(M), constraints)

            solved = False
            for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP]:
                try:
                    prob.solve(solver=solver, timeout=min(4.0, rem / 5.0))
                    if g_block.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                        g_new = np.maximum(g_block.value, 0)
                        gs = float(np.sum(g_new))
                        if gs > 1e-10 and abs(gs - target_sum) > 1e-8:
                            g_new = g_new * target_sum / gs
                        a[start_idx:end_idx] = g_new
                        solved = True
                        break
                except Exception:
                    continue

            if solved:
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new
                    obj_new = 2.0 * n * float(np.max(compute_conv(a)))
                    if obj_new < best_obj - 1e-14:
                        best_obj = obj_new
                        best_a = a.copy()
                        improved_any = True

        if not improved_any:
            break

    return best_a


def parametric_seq(n, mode, p=1.0, sigma=0.25, alpha=2.0):
    """Generate parametric initializations."""
    x = np.linspace(0, 1, n)
    xm = x - 0.5

    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = np.cos(np.pi * xm) ** (2.0 * p)
        a = np.maximum(a, 0)
    elif mode == 2:
        a = np.exp(-xm ** 2 / (2.0 * sigma ** 2))
    elif mode == 3:
        a = np.maximum(1.0 - 4.0 * xm ** 2, 0)
    elif mode == 4:
        a = np.maximum(1.0 - 4.0 * xm ** 2, 0) ** 2
    elif mode == 5:
        i = np.arange(1, n + 1, dtype=np.float64)
        a = i ** (alpha - 1) * (n - i + 1) ** (alpha - 1)
    elif mode == 6:
        a = np.cos(np.pi * xm) ** p
        a = np.maximum(a, 0)
    elif mode == 7:
        t = np.abs(4.0 * xm)
        a = np.zeros(n)
        m1 = t < 1.0
        m2 = (t >= 1.0) & (t < 2.0)
        a[m1] = 1.0 - 1.5 * t[m1] ** 2 + 0.5 * t[m1] ** 3
        a[m2] = 0.25 * (2.0 - t[m2]) ** 3
    elif mode == 8:
        a = np.maximum(1.0 - 2.0 * np.abs(xm), 0)
    elif mode == 9:
        a = np.exp(-np.abs(xm / p) ** p)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def smoothed_gd(a, time_budget=80.0, tau_init=50.0, tau_end=800.0, seed=42):
    """Nesterov-accelerated smoothed gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / math.sqrt(n)
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            s_y = 1.0
        else:
            y = y / s_y

        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p_vec = np.zeros(len(conv))
            p_vec[np.argmax(conv)] = 1.0
        else:
            p_vec = exp_vals / sum_exp

        N = 1
        while N < 3 * n:
            N <<= 1

        P_fft = fft(p_vec, N)
        A_fft = fft(y, N)
        g_full = ifft(P_fft * np.conj(A_fft)).real
        grad = 2.0 * g_full[:n]

        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-15:
            lr_eff = min(lr, 2.5 / (math.sqrt(n) * (g_norm / math.sqrt(n) + 1e-10)))
        else:
            lr_eff = lr

        v = beta * v - lr_eff * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        it += 1

    return best_a
