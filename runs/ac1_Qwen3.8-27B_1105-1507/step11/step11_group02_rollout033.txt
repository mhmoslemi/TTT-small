"""
Optimizes step function coefficients using targeted coordinate descent with
peak-driven coordinate selection, LP-based alternating minimization, asymmetric
initialization strategies, and multi-length exploration.

Key improvements over previous approach:
1. Peak-targeted CD: prioritizes coordinates contributing to the current max convolution
2. Asymmetric initializations (power-law ramps, exponential decays) that exploit the
   non-convexity of the problem
3. Pairwise coordinate updates to escape 1D local minima
4. More aggressive LP polishing with log-sum-exp smooth approximation
5. Wider n exploration with efficient resampling
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, time_budget=20.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None) on failure.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    start = time.monotonic()
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS]
    for solver in solvers:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_g = float(np.sum(g_val))
                if s_g > 1e-10:
                    return g_val / s_g, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_smoothed(a, tau=10.0, time_budget=20.0):
    """
    Use log-sum-exp smoothing to convert max constraint to a smooth objective.
    Solves: min log-sum-exp(tau * conv(a,g)) subject to sum(g)=1, g>=0.
    Uses gradient projection as a fallback.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    # Start from a and do projected gradient descent on the smoothed objective
    g = a_norm.copy()
    best_g = g.copy()
    best_val = compute_obj(g)

    start = time.monotonic()
    lr = 0.1 / tau
    it = 0

    while time.monotonic() - start < time_budget:
        # Compute conv(a, g)
        conv_ag = A @ g
        max_val = float(np.max(conv_ag))
        exp_vals = np.exp(tau * (conv_ag - max_val))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv_ag))
            p[np.argmax(conv_ag)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient of log-sum-exp w.r.t. g is A^T
        grad = A.T @ (tau * p)
        grad = grad - np.mean(grad)  # project to maintain sum

        g_new = g - lr * grad
        # Project to simplex (sum=1, g>=0)
        g_new = np.maximum(g_new, 0)
        s_g = float(np.sum(g_new))
        if s_g < 1e-10:
            break
        g_new = g_new / s_g

        obj_new = compute_obj(g_new)
        if obj_new < best_val:
            best_val = obj_new
            best_g = g_new.copy()

        g = g_new
        lr *= 0.995
        it += 1

    return best_g


def targeted_coordinate_descent(a, time_budget=300.0, verbose=False, seed=42):
    """
    Coordinate descent with peak-targeted coordinate selection.
    Coordinates that contribute most to the current convolution peak are
    updated first.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    peak_idx = int(np.argmax(conv))

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        sweep += 1
        peak_idx = int(np.argmax(conv))
        S = float(np.sum(a))

        # Determine coordinate priority: coordinates contributing to the peak
        # are updated first
        if sweep % 2 == 1 or peak_idx < n:
            # Build priority: coordinate i contributes to conv[peak_idx]
            # if 0 <= peak_idx - i <= n - 1, i.e., i in [max(0, peak_idx-n+1), min(n-1, peak_idx)]
            i_lo = max(0, peak_idx - n + 1)
            i_hi = min(n - 1, peak_idx)
            if i_lo <= i_hi:
                priority_coords = list(range(i_lo, i_hi + 1))
                random_coords = [i for i in range(n) if i not in set(priority_coords)]
            else:
                priority_coords = []
                random_coords = list(range(n))
        else:
            priority_coords = []
            random_coords = list(range(n))

        rng.shuffle(priority_coords)
        rng.shuffle(random_coords)
        order = priority_coords + random_coords

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            def f_1d(x, v=v, S_other=S_other, n=n, c_k_lin=c_k_lin,
                     a_ki_lin=a_ki_lin, c_2i=c_2i):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Find best candidate
            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, v * 0.3, v * 0.5, v * 0.7, v * 1.0,
                          v * 1.3, v * 1.5, v * 2.0, 1.0 / n, 2.0 / n,
                          max(0.0, v - scale * 0.5), v + scale * 0.5]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section refinement
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.0
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(35):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-13:
                continue

            delta = x_opt - v
            a[i] = x_opt
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            S += delta

            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, peak_idx={peak_idx}, t={elapsed:.1f}s")

    return best_a


def pairwise_cd(a, n_pairs_per_sweep=20, time_budget=60.0, seed=42):
    """
    Pairwise coordinate descent: update pairs of coordinates simultaneously.
    This helps escape 1D local minima.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        # Pick random pairs
        for _ in range(n_pairs_per_sweep):
            if time.monotonic() - start >= time_budget:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            vi = a[i]
            vj = a[j]
            S_other = S - vi - vj

            # Joint 2D search over (a[i], a[j])
            # Try several candidates
            candidates_i = [0.0, vi * 0.5, vi, vi * 1.5]
            candidates_j = [0.0, vj * 0.5, vj, vj * 1.5]

            best_xi, best_xj = vi, vj
            best_f = None

            for xi in candidates_i:
                for xj in candidates_j:
                    S_try = S_other + xi + xj
                    if S_try < 1e-10:
                        continue
                    # Estimate conv change
                    # conv change at positions affected by i and j
                    max_est = 0.0
                    for k in range(2 * n - 1):
                        # conv[k] = sum_l a[l]*a[k-l]
                        # Change from modifying a[i] and a[j]:
                        # delta at k involves terms with i or j
                        old_val = conv[k]
                        new_val = old_val
                        # Terms involving i: a[i]*a[k-i] for valid k-i
                        ki = k - i
                        if 0 <= ki < n:
                            new_val = new_val - a[i] * a[ki] + xi * a[ki]
                        kj = k - j
                        if 0 <= kj < n:
                            if kj != i:  # avoid double counting if i==k-j and j==k-i
                                new_val = new_val - a[j] * a[kj] + xj * a[kj]
                            else:
                                new_val = new_val - a[j] * a[i] + xj * xi
                        if new_val > max_est:
                            max_est = new_val
                    obj_try = 2.0 * n * max_est / (S_try * S_try)
                    if best_f is None or obj_try < best_f:
                        best_f = obj_try
                        best_xi, best_xj = xi, xj

            if best_f is None:
                continue

            if best_f < best_obj - 1e-12 or (best_xi != vi or best_xj != vj):
                a[i] = best_xi
                a[j] = best_xj
                conv = compute_conv(a)  # Recompute for accuracy
                S = float(np.sum(a))
                obj = 2.0 * n * float(np.max(conv)) / (S * S) if S > 1e-10 else np.inf
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

    return best_a


def multiplicative_descent(a, time_budget=60.0, tau_init=30.0, tau_end=300.0, verbose=False):
    """Multiplicative gradient descent with adaptive learning rate."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    lr_base = 6.0 / np.sqrt(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress
        lr = lr_base * (1.0 - 0.6 * progress)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient via FFT
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        grad = 2.0 * g_full[:n]

        # Multiplicative update
        update = np.exp(-lr * grad)
        update = np.clip(update, 0.05, 20.0)
        a_new = a * update
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(np.sum(a_new))

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 500 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.0f}")

    return best_a


def lp_polish_cycle(a, time_budget=60.0, seed=42):
    """LP-based polishing with multiple blending factors."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp_minpeak(a, time_budget=min(remaining * 0.4, 15.0))
        if g_lp is None:
            # Try smoothed approach
            g_ls = solve_lp_smoothed(a, tau=5.0, time_budget=min(remaining * 0.3, 10.0))
            if g_ls is not None:
                g_lp = g_ls

        if g_lp is None:
            break

        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 2:
                pert_scale = 0.02 * stall
                pert = rng.normal(0, pert_scale, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                s_a = float(a.sum())
                if s_a > 1e-10:
                    a = a / s_a
                stall = 0

    return best_a


def asymmetric_init(n, mode=0, rng=None):
    """Generate asymmetric initializations."""
    if rng is None:
        rng = np.random.default_rng(0)

    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)

    if mode == 0:
        # Power law ramp (right-heavy)
        a = (x + 0.01) ** rng.uniform(2, 5)
    elif mode == 1:
        # Exponential decay
        beta = rng.uniform(2, 6)
        a = np.exp(-beta * x)
    elif mode == 2:
        # Sigmoid
        center = rng.uniform(0.2, 0.8)
        sharp = rng.uniform(3, 10)
        a = 1.0 / (1.0 + np.exp(sharp * (x - center)))
    elif mode == 3:
        # Power law (left-heavy)
        a = (1.0 - x + 0.01) ** rng.uniform(2, 5)
    elif mode == 4:
        # Gaussian off-center
        center = rng.uniform(0.1, 0.4)
        width = rng.uniform(0.1, 0.3)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    elif mode == 5:
        # Bump + tail
        center = rng.uniform(0.3, 0.6)
        width = rng.uniform(0.05, 0.15)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        a += 0.3 * np.exp(-3 * (1 - x))
    elif mode == 6:
        # Smooth ramp with curvature
        a = x ** rng.uniform(1.5, 3.0) + 0.1 * np.sin(np.pi * x)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fast_cd(a, time_budget=100.0, seed=42):
    """
    Fast coordinate descent with simplified line search.
    Uses fewer golden section iterations for speed.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    peak_idx = int(np.argmax(conv))

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        peak_idx = int(np.argmax(conv))
        S = float(np.sum(a))

        # Targeted order
        i_lo = max(0, peak_idx - n + 1)
        i_hi = min(n - 1, peak_idx)
        order = list(range(n))
        rng.shuffle(order)
        # Move peak-contributing coords to front
        front = [i for i in order if i_lo <= i <= i_hi]
        back = [i for i in order if not (i_lo <= i <= i_hi)]
        order = front + back

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            # Quick 1D optimization
            scale = max(v, 1.0 / n, 1e-6)
            best_x = v
            best_f = np.inf

            # Try a few candidates
            for frac in [0.0, 0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
                x = v * frac
                x = max(0.0, x)
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    continue
                f = 2.0 * n * max_val / denom
                if f < best_f:
                    best_f = f
                    best_x = x

            # Golden section around best
            x_lo = max(0.0, best_x - scale * 0.5)
            x_hi = best_x + scale * 0.8
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)

            def f_quick(x):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            fc = f_quick(c_pt)
            fd = f_quick(d_pt)

            for _ in range(25):
                if x_hi - x_lo < 1e-12 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_quick(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_quick(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-12:
                continue

            delta = x_opt - v
            a[i] = x_opt
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            S += delta

            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: multi-phase optimization combining targeted CD,
    LP polishing, multiplicative descent, asymmetric inits, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    # === Phase 1: Quick CD from parent ===
    print(f"\n=== Phase 1: Quick CD (parent) ===")
    remaining = deadline - time.monotonic()
    p1 = min(remaining * 0.25, 300.0)
    if p1 > 30:
        a_cd = targeted_coordinate_descent(best_seq, time_budget=p1, verbose=True, seed=seed)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 2: LP Polishing ===
    print(f"\n=== Phase 2: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2 = min(remaining * 0.2, 200.0)
    if p2 > 15:
        for rnd in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(p2 / (5 - rnd), remaining * 0.4)
            if alloc < 8:
                break
            a_lp = lp_polish_cycle(best_seq, time_budget=alloc, seed=3000 + rnd * 17)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")

    # === Phase 3: Post-LP CD ===
    print(f"\n=== Phase 3: Post-LP CD ===")
    remaining = deadline - time.monotonic()
    p3 = min(remaining * 0.12, 100.0)
    if p3 > 15:
        a_cd2 = targeted_coordinate_descent(best_seq, time_budget=p3, verbose=False, seed=seed + 777)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  CD2: obj={best_score:.8f} **")
        else:
            print(f"  CD2: obj={obj_cd2:.8f}")

    # === Phase 4: Multiplicative Descent ===
    print(f"\n=== Phase 4: Multiplicative Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.08, 60.0)
        a_md = multiplicative_descent(best_seq, time_budget=alloc, verbose=False)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  MD: obj={best_score:.8f} **")
        else:
            print(f"  MD: obj={obj_md:.8f}")

    # === Phase 5: Asymmetric Initializations ===
    print(f"\n=== Phase 5: Asymmetric Initializations ===")
    n_current = len(best_seq)
    remaining = deadline - time.monotonic()

    n_attempts = min(6, max(2, int(remaining / 80)))
    print(f"  Trying {n_attempts} asymmetric inits (n={n_current})")

    for attempt in range(n_attempts):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        mode = attempt % 7
        a_init = asymmetric_init(n_current, mode=mode, rng=rng)

        # Blend slightly with best to maintain quality
        blend = 0.2 + 0.1 * attempt
        a_init = (1 - blend) * a_init + blend * best_seq
        a_init = np.maximum(a_init, 0)
        s_init = float(np.sum(a_init))
        if s_init > 1e-10:
            a_init = a_init / s_init

        alloc_cd = min(remaining * 0.15, 50.0)
        if alloc_cd > 15:
            a_opt = fast_cd(a_init, time_budget=alloc_cd, seed=5000 + attempt * 13)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Asym init {mode}: obj={best_score:.8f} **")
            else:
                print(f"  Asym init {mode}: obj={obj_opt:.8f}")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(400, n_best - 100), max(400, n_best - 200),
        n_best + 100, n_best + 200, n_best + 300,
        600, 700, 900, 1000, 1200, 1400, 1600
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1800 and abs(n - n_best) >= 50]
    n_options = n_options[:8]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        alloc_cd = min(remaining * 0.12, 40.0)
        if alloc_cd > 10:
            a_try = fast_cd(a_try, time_budget=alloc_cd, seed=n_try)

        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.05, 15.0)
        if alloc_lp > 5:
            a_try = lp_polish_cycle(a_try, time_budget=alloc_lp, seed=n_try + 2000)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Pairwise CD (escape local minima) ===
    print(f"\n=== Phase 7: Pairwise CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.08, 50.0)
        a_pw = pairwise_cd(best_seq, n_pairs_per_sweep=30, time_budget=alloc, seed=seed + 9999)
        obj_pw = compute_obj(a_pw)
        if obj_pw < best_score:
            best_score = obj_pw
            best_seq = a_pw.copy()
            print(f"  Pairwise: obj={best_score:.8f} **")
        else:
            print(f"  Pairwise: obj={obj_pw:.8f}")

    # === Phase 8: Final Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 120.0)
        a_final = targeted_coordinate_descent(best_seq, time_budget=alloc, verbose=False, seed=seed + 19999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_lp_final = lp_polish_cycle(best_seq, time_budget=min(remaining * 0.5, 40.0), seed=8888)
        obj_lp_final = compute_obj(a_lp_final)
        if obj_lp_final < best_score:
            best_score = obj_lp_final
            best_seq = a_lp_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # One more targeted CD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_cd_final = targeted_coordinate_descent(best_seq, time_budget=min(remaining * 0.6, 60.0),
                                                 verbose=False, seed=seed + 25000)
        obj_cd_final = compute_obj(a_cd_final)
        if obj_cd_final < best_score:
            best_score = obj_cd_final
            best_seq = a_cd_final.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
