import time

    n = len(a)

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad(a, tau):
    """Smoothed gradient of max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = compute_conv(a)
    mc = float(np.max(conv))
    ev = np.exp(tau * (conv - mc))
    se = float(np.sum(ev))
    if se < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = ev / se
    
    N = 1
    while N < 3 * n:
        N <<= 1
    A = fft(a, N)
    P = fft(p, N)
    corr = ifft(np.conj(A) * P).real[:n]
    convap = ifft(A * P).real[:n]
    grad = corr + convap
    return grad, conv, p

            break

        y = np.maximum(y, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses a multi-phase strategy: accelerated gradient flow with log-sum-exp smoothing,
exact coordinate descent, LP-based direction finding, and multi-n exploration.
Key innovations: adaptive tau curriculum, Nesterov momentum with restarts,
and integrated LP-CD polishing loops.
"""

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via real FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad(a, tau):
    """
    Compute smoothed gradient of max(conv(a,a)) for a with sum(a)=1.
    Returns (gradient, conv, weights).
    """
    n = len(a)
    conv = compute_conv(a)
    mc = float(np.max(conv))
    ev = np.exp(tau * (conv - mc))
    se = float(np.sum(ev))
    if se < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = ev / se

    N = 1
    while N < 3 * n:
        N <<= 1
    A = fft(a, N)
    P = fft(p, N)
    corr = ifft(np.conj(A) * P).real[:n]
    convap = ifft(A * P).real[:n]
    grad = corr + convap
    return grad, conv, p


def gradient_flow(a, time_budget, tau_start=15.0, tau_end=400.0, seed=42):
    """
    Nesterov-accelerated gradient flow with adaptive step size and momentum restarts.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    beta = 0.87
    v = np.zeros(n)
    base_lr = 3.0 / np.sqrt(n)
    it = 0
    stall = 0
    best_stall_obj = best_obj

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1.0))
        tau = tau_start + (tau_end - tau_start) * (progress ** 0.7)
        lr = base_lr * (1.0 - 0.5 * progress)

        try:
            grad, conv, p = smoothed_grad(a, tau)
        except Exception:
            break

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-16:
            break

        y = a + beta * v
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy < 1e-10:
            y = a.copy()
            sy = float(np.sum(y))
        y = y / sy

        try:
            grad_y, _, _ = smoothed_grad(y, tau)
        except Exception:
            grad_y = grad

        # Adaptive learning rate
        gy_norm = float(np.linalg.norm(grad_y))
        if gy_norm > 1e-15:
            lr_eff = min(lr * 1.5 / (gy_norm + 0.1), lr * 3.0)
        else:
            lr_eff = lr

        v = beta * v - lr_eff * grad_y
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
            v = np.zeros(n)
            s_new = float(np.sum(a_new))
        a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            if best_obj < best_stall_obj:
                best_stall_obj = best_obj
        else:
            stall += 1

        a = a_new
        it += 1

        if stall > 800:
            beta = max(0.5, beta * 0.9)
            v = np.zeros(n)
            stall = 0
            if stall > 3000:
                break

        if it % 10000 == 0:
            print(f"    GF it{it}: obj={best_obj:.8f}, tau={tau:.0f}")

    return best_a


def f1d_cd(x, conv_vals, ks, d_coeffs, v, S_ot, quad_k, k_lo, has_quad, n, max_unaff):
    """1D objective for coordinate descent at value x."""
    new_vals = conv_vals + d_coeffs * (x - v)
    if has_quad:
        qidx = quad_k - k_lo
        new_vals = new_vals.copy()
        new_vals[qidx] += x * x - v * v
    m_aff = float(np.max(new_vals))
    m = max(m_aff, max_unaff)
    denom = (S_ot + x) ** 2
    if denom < 1e-30:
        return 1e30
    return 2.0 * n * m / denom


def cd_polish(a, time_budget, seed=42):
    """
    Exact coordinate descent with golden section search.
    Multiple sweeps until time runs out or convergence.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        any_improved = False

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_ot = float(np.sum(a)) - v
            if S_ot < 1e-12:
                continue

            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_lo, k_hi + 1)

            d_coeffs = np.zeros(len(ks))
            for idx in range(len(ks)):
                k = ks[idx]
                c = 0.0
                if 0 <= k - i < n:
                    c += a[k - i]
                if 0 <= i - k < n:
                    c += a[i - k]
                d_coeffs[idx] = c

            quad_k = 2 * i
            has_quad = (0 <= quad_k <= 2 * n - 2)

            mask = np.ones(2 * n - 1, dtype=bool)
            mask[k_lo:k_hi + 1] = False
            if mask.any():
                max_unaff = float(np.max(conv[mask]))
            else:
                max_unaff = 0.0

            conv_vals = conv[ks].copy()

            best_x = v
            best_f = f1d_cd(v, conv_vals, ks, d_coeffs, v, S_ot, quad_k, k_lo, has_quad, n, max_unaff)

            candidates = [0.0, v * 0.3, v * 0.7, v * 1.3, v * 2.0, 1.0 / n, 2.0 / n, 3.0 / n]
            for cx in candidates:
                cx = max(0.0, cx)
                fx = f1d_cd(cx, conv_vals, ks, d_coeffs, v, S_ot, quad_k, k_lo, has_quad, n, max_unaff)
                if fx < best_f:
                    best_f = fx
                    best_x = cx

            scale = max(best_x, 1.0 / n, 1e-6)
            x_lo = max(0.0, best_x - 0.7 * scale)
            x_hi = best_x + 1.3 * scale
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-10

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f1d_cd(c_pt, conv_vals, ks, d_coeffs, v, S_ot, quad_k, k_lo, has_quad, n, max_unaff)
            fd = f1d_cd(d_pt, conv_vals, ks, d_coeffs, v, S_ot, quad_k, k_lo, has_quad, n, max_unaff)

            for _ in range(35):
                if x_hi - x_lo < 1e-13 * max(1.0, abs(x_hi)):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f1d_cd(c_pt, conv_vals, ks, d_coeffs, v, S_ot, quad_k, k_lo, has_quad, n, max_unaff)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f1d_cd(d_pt, conv_vals, ks, d_coeffs, v, S_ot, quad_k, k_lo, has_quad, n, max_unaff)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-12:
                continue

            delta = x_opt - v
            a[i] = x_opt
            conv[ks] += d_coeffs * delta
            if has_quad:
                conv[quad_k] += x_opt * x_opt - v * v

            S_chk = float(np.sum(a))
            if S_chk > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_chk * S_chk)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    any_improved = True

        if not any_improved:
            break

    return best_a


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def lp_direction(a, time_budget=20.0):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    start = time.monotonic()
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M_var.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget, seed=42):

        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:

            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:

            a_try = a_try / st
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.1) ** 2) / 0.02) + np.exp(-((x - 0.9) ** 2) / 0.02)
        return a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x ** 2 + 1e-4)
        return a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for _ in range(5):
            f = rng.integers(1, 8)

    elif mode == 5:

        a += 0.1 * rng.uniform(0, 1, n)
        return a / a.sum()
    else:
        a = rng.exponential(1.0, n)
        return a / a.sum()


def resample_seq(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
