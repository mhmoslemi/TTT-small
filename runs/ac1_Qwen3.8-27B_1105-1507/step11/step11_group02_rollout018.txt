    while size < 2*n - 1:

    if s < 0.01:

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses: FFT-based smoothed gradient descent, exact coordinate descent,
LP direction finding, peak-targeted penalty, and multi-n exploration.
Key innovations: efficient cross-correlation gradient, adaptive tau scheduling,
peak equalization penalty, and aggressive CD with rapid convergence detection.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    c = compute_conv(a_n)
    return 2.0 * n * float(np.max(c))


def compute_smoothed_grad(a, tau):
    """
    Compute smoothed gradient of max(conv(a,a)) via softmax weights + FFT cross-correlation.
    Returns (grad, conv, weights).
    """
    n = len(a)
    c = compute_conv(a)
    max_c = float(np.max(c))
    shifted = tau * (c - max_c)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    w = exp_vals / float(np.sum(exp_vals))

    # grad[i] = 2 * sum_k w[k] * a[k-i] for valid indices
    # = 2 * cross_correlation(w, a)[i]
    N = 1
    while N < 3 * n:
        N <<= 1
    W = fft(w, N)
    A = fft(a, N)
    R = ifft(W * np.conj(A)).real
    grad = 2.0 * R[:n]
    return grad, c, w


def project_simplex(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    rho_candidate = (mu - (cv - 1.0) / np.arange(1, n + 1)) > 0
    rho = int(np.sum(rho_candidate))
    if rho == 0:
        return np.ones(n) / n
    theta = (float(cv[rho - 1]) - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def coordinate_descent(a, time_budget=300.0, verbose=False, seed=42, tol=1e-12):
    """
    Exact coordinate descent with golden section search.
    Optimizes one coordinate at a time with exact 1D minimization.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    no_improve_sweeps = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        obj_before = best_obj
        order = rng.permutation(n)
        S_cur = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S_cur - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            def f_1d(x):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Candidate points for initial bracket
            candidates = [0.0, v * 0.3, v * 0.6, v * 1.4, v * 2.0,
                          1.0 / n, 2.0 / n, 3.0 / n, v]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section around best candidate
            scale = max(best_x, 1.0 / n, 1e-6)
            x_lo = max(0.0, best_x - 1.5 * scale)
            x_hi = best_x + 2.0 * scale
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-10

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(50):
                if x_hi - x_lo < tol * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-15:
                continue

            delta = x_opt - v
            a[i] = x_opt
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            S_cur += delta

            if S_cur > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_cur * S_cur)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if best_obj < obj_before - 1e-10:
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
            if no_improve_sweeps > 8:
                # Perturb to escape
                pert_scale = 0.02 * no_improve_sweeps
                pert = rng.normal(0, pert_scale, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / float(a.sum())
                conv = compute_conv(a)
                no_improve_sweeps = 0

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def subgradient_descent(a, time_budget=120.0, tau_init=20.0, tau_end=300.0, verbose=False):
    """
    Smoothed subgradient descent with adaptive tau and backtracking line search.
    Projects onto simplex after each step.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    lr = 0.05
    it = 0
    consecutive_bad = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init * (tau_end / tau_init) ** progress

        try:
            grad, conv, w = compute_smoothed_grad(a, tau)
        except Exception:
            break

        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break

        # Adaptive learning rate: smaller when gradient is large
        eff_lr = lr / (1.0 + 0.5 * progress)
        g_scale = float(np.max(np.abs(grad)))
        if g_scale > 1e-10:
            eff_lr = min(eff_lr, 0.1 / g_scale)

        # Try multiple step sizes via backtracking
        found = False
        for attempt in range(6):
            step = eff_lr * (0.5 ** attempt)
            a_try = a - step * grad
            a_try = project_simplex(a_try)

            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                consecutive_bad = 0
            if obj_try < 2.0 * n * float(np.max(conv)):
                a = a_try
                found = True
                break

        if not found:
            a = best_a.copy()
            consecutive_bad += 1
            if consecutive_bad > 10:
                # Restart from best with perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = project_simplex(a)
                consecutive_bad = 0
            # Reduce learning rate
            eff_lr *= 0.8

        it += 1
        if verbose and it % 500 == 0:
            print(f"    SGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={eff_lr:.4f}")

    return best_a


def peak_penalty_descent(a, time_budget=100.0, num_peaks=10, verbose=False):
    """
    Targeted peak reduction: penalize the top peaks in the convolution.
    Uses gradient of the penalty function to drive peaks down.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    lr = 0.03
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        c = compute_conv(a)
        max_c = float(np.max(c))
        target = float(np.mean(c))

        # Identify top peaks
        peak_indices = np.argsort(c)[::-1][:num_peaks]
        # Penalty: sum of excess^2 for peaks above target
        excess = np.maximum(c - target, 0)
        # Gradient of penalty: for peak at position k, grad_a[i] += 2*excess[k]*a[k-i]
        pen_grad = np.zeros(n)
        for k in peak_indices:
            e = excess[k]
            if e < 1e-15:
                continue
            # d c[k] / d a[i] = 2*a[k-i] if 0<=k-i<n
            idx = k - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            pen_grad[valid] += 2.0 * e * a[idx[valid]]

        # Also add small gradient toward uniformization
        total_pen_grad = pen_grad

        # Projected step
        a_try = a - lr * total_pen_grad
        a_try = project_simplex(a_try)

        obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            stall = 0
        else:
            a = best_a.copy()
            stall += 1
            if stall % 5 == 0:
                lr *= 0.9
            if stall > 50:
                # Perturb
                rng = np.random.default_rng(hash(str(time.monotonic())) % 100000)
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = project_simplex(a)
                stall = 0
                lr = 0.03

        it += 1
        if verbose and it % 200 == 0:
            print(f"    PeakPen it{it}: obj={best_obj:.8f}")

    return best_a


def solve_lp_direction(a, time_budget=15.0):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0.
    Returns direction g or None.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))

    # Build A matrix: A[k, i] = a_norm[k-i] if 0<=k-i<n else 0
    # This is the convolution matrix
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 2, i + n - 1)
        A[k_start:k_end + 1, i] = a_norm[0:n - (k_end - k_start)]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    start = time.monotonic()
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def lp_blend_iteration(a, time_budget=120.0, verbose=False, seed=42):
    """
    Iterative LP-based blending: solve LP direction, blend with current, repeat.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        g_lp = solve_lp_direction(a, time_budget=min(remaining * 0.3, 12.0))
        if g_lp is None:
            break

        best_blend_obj = best_obj
        best_blend_a = best_a.copy()
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_blend_a = a_try.copy()

        if best_blend_obj < best_obj - 1e-12:
            best_obj = best_blend_obj
            best_a = best_blend_a.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 3:
                pert = rng.normal(0, 0.02 * stall, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / float(a.sum())
                stall = 0

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] * n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 0.01)
        a = a / a.sum()
    elif mode == 3:
        centers = rng.uniform(0.2, 0.8, size=2)
        widths = rng.uniform(0.05, 0.2, size=2)
        x = np.linspace(0, 1, n)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        freqs = rng.integers(2, 8, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Cosine-tapered
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 6:
        # Random block structure
        num_blocks = rng.integers(2, 6)
        a = np.zeros(n)
        block_size = n // num_blocks
        for i in range(num_blocks):
            start = i * block_size
            end = (i + 1) * block_size if i < num_blocks - 1 else n
            a[start:end] = rng.uniform(0.5, 1.5)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase optimization:
    Phase 1: CD polish on parent
    Phase 2: Subgradient descent with adaptive tau
    Phase 3: CD after SGD
    Phase 4: Peak penalty optimization
    Phase 5: LP blending
    Phase 6: Multi-n exploration
    Phase 7: Diverse restarts
    Phase 8: Final polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    def remaining():
        return deadline - time.monotonic()

    # === Phase 1: Coordinate Descent (main workhorse) ===
    print(f"\n=== Phase 1: CD ===")
    if remaining() > 60:
        p1_budget = min(remaining() * 0.25, 300.0)
        a_cd = coordinate_descent(best_seq, time_budget=p1_budget,
                                  verbose=True, seed=seed)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 2: Subgradient Descent ===
    print(f"\n=== Phase 2: Subgradient Descent ===")
    if remaining() > 60:
        p2_budget = min(remaining() * 0.12, 120.0)
        a_sgd = subgradient_descent(best_seq, time_budget=p2_budget,
                                    tau_init=15.0, tau_end=200.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  SGD: obj={best_score:.8f} **")
        else:
            print(f"  SGD: obj={obj_sgd:.8f}")
            # Brief CD after SGD
            if remaining() > 30:
                a_cd2 = coordinate_descent(a_sgd, time_budget=min(remaining() * 0.08, 60.0),
                                           verbose=False, seed=seed + 100)
                obj_cd2 = compute_obj(a_cd2)
                if obj_cd2 < best_score:
                    best_score = obj_cd2
                    best_seq = a_cd2.copy()
                    print(f"  SGD+CD: obj={best_score:.8f} **")

    # === Phase 3: CD polish ===
    print(f"\n=== Phase 3: CD Polish ===")
    if remaining() > 40:
        p3_budget = min(remaining() * 0.15, 150.0)
        a_cd = coordinate_descent(best_seq, time_budget=p3_budget,
                                  verbose=True, seed=seed + 500)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD2: obj={best_score:.8f} **")
        else:
            print(f"  CD2: obj={obj_cd:.8f}")

    # === Phase 4: Peak Penalty ===
    print(f"\n=== Phase 4: Peak Penalty ===")
    if remaining() > 50:
        p4_budget = min(remaining() * 0.1, 80.0)
        a_pp = peak_penalty_descent(best_seq, time_budget=p4_budget,
                                    num_peaks=8, verbose=True)
        obj_pp = compute_obj(a_pp)
        if obj_pp < best_score:
            best_score = obj_pp
            best_seq = a_pp.copy()
            print(f"  PeakPen: obj={best_score:.8f} **")
            # Follow with CD
            if remaining() > 30:
                a_cd3 = coordinate_descent(a_pp, time_budget=min(remaining() * 0.06, 50.0),
                                           verbose=False, seed=seed + 900)
                obj_cd3 = compute_obj(a_cd3)
                if obj_cd3 < best_score:
                    best_score = obj_cd3
                    best_seq = a_cd3.copy()
                    print(f"  PeakPen+CD: obj={best_score:.8f} **")
        else:
            print(f"  PeakPen: obj={obj_pp:.8f}")

    # === Phase 5: LP Blending ===
    print(f"\n=== Phase 5: LP Blending ===")
    if remaining() > 40:
        p5_budget = min(remaining() * 0.1, 80.0)
        a_lp = lp_blend_iteration(best_seq, time_budget=p5_budget,
                                  verbose=False, seed=seed + 2000)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
            # Follow with CD
            if remaining() > 30:
                a_cd4 = coordinate_descent(a_lp, time_budget=min(remaining() * 0.06, 50.0),
                                           verbose=False, seed=seed + 3000)
                obj_cd4 = compute_obj(a_cd4)
                if obj_cd4 < best_score:
                    best_score = obj_cd4
                    best_seq = a_cd4.copy()
                    print(f"  LP+CD: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_current = len(best_seq)
    n_options = sorted(set([
        max(400, n_current - 100),
        n_current,
        n_current + 100,
        n_current + 200,
        600, 900, 1000, 1200, 1500
    ]))
    # Filter to reasonable range and diversity
    n_options = [n for n in n_options if 400 <= n <= 2000]
    # Remove duplicates and sort
    n_options = sorted(set(n_options))
    # Limit to 5 options spread across the range
    if len(n_options) > 5:
        indices = np.linspace(0, len(n_options) - 1, 5, dtype=int)
        n_options = [n_options[i] for i in indices]
    # Always include current best n
    if n_current not in n_options:
        n_options.append(n_current)
    n_options = sorted(set(n_options))

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        if remaining() < 50:
            break

        if n_try == n_current:
            continue  # Already optimized

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # CD on new length
        if remaining() > 40:
            alloc_cd = min(remaining() * 0.12, 60.0)
            a_try = coordinate_descent(a_try, time_budget=alloc_cd,
                                       verbose=False, seed=n_try)
            # Brief SGD
            if remaining() > 40:
                alloc_sgd = min(remaining() * 0.05, 30.0)
                a_try = subgradient_descent(a_try, time_budget=alloc_sgd,
                                            tau_init=20.0, tau_end=150.0, verbose=False)
                # Final CD
                if remaining() > 30:
                    a_try = coordinate_descent(a_try, time_budget=min(remaining() * 0.05, 30.0),
                                               verbose=False, seed=n_try + 777)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse Restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    n_use = len(best_seq)
    for r in range(4):
        if remaining() < 40:
            break

        mode = (r + 1) % 7
        a_r = diverse_init(n_use, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = project_simplex(a_r)

        alloc = min(remaining() * 0.1, 50.0)
        if alloc > 20:
            a_r = coordinate_descent(a_r, time_budget=alloc, verbose=False,
                                     seed=8000 + r * 111)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 8: Final Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    if remaining() > 20:
        alloc = min(remaining() * 0.4, 120.0)
        a_final = coordinate_descent(best_seq, time_budget=alloc,
                                     verbose=False, seed=seed + 19999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    if remaining() > 15:
        alloc = min(remaining() * 0.3, 60.0)
        a_sgd_final = subgradient_descent(best_seq, time_budget=alloc,
                                          tau_init=50.0, tau_end=400.0, verbose=False)
        obj_sgd_f = compute_obj(a_sgd_final)
        if obj_sgd_f < best_score:
            best_score = obj_sgd_f
            best_seq = a_sgd_final.copy()
            print(f"  Final SGD: obj={best_score:.8f} **")

    # One more CD if time
    if remaining() > 10:
        a_cd_f = coordinate_descent(best_seq, time_budget=min(remaining() * 0.5, 40.0),
                                    verbose=False, seed=seed + 25000)
        obj_cd_f = compute_obj(a_cd_f)
        if obj_cd_f < best_score:
            best_score = obj_cd_f
            best_seq = a_cd_f.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
