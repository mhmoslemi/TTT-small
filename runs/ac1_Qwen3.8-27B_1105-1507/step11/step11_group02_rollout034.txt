import numpy as np
from scipy.fft import rfft, irfft
import time

def convolve_auto(a):
    """Compute auto-convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    conv = convolve_auto(a)
    return 2.0 * n * np.max(conv) / (s * s)

def objective_normalized(a):
    """Assume sum(a)=1, return 2*n*max(conv(a,a))"""
    n = len(a)
    conv = convolve_auto(a)
    return 2.0 * n * np.max(conv)

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Compute full linear auto-convolution a*a via FFT. Returns array of length 2n-1."""

    return irfft(fa * fa, size)[:2 * n - 1]


def obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def obj_raw(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return float(2.0 * n * np.max(conv) / (s * s))


def cd_single_sweep(a, n, order=None):
    """
    One sweep of coordinate descent with exact 1D minimization.
    Exploits the analytical structure: for coordinate i, the objective is
    2n * max(x^2 + C, max_k(D_k + b_k*x)) / (S_other + x)^2
    which has closed-form optimal candidates.
    Returns updated a and conv.
    """
    a = a.astype(np.float64).copy()
    conv = compute_conv(a)
    S = float(np.sum(a))
    
    if order is None:
        order = list(range(n))
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for i in order:
        v = a[i]
        S_other = S - v
        if S_other < 1e-14:
            continue
        
        # Quadratic term coefficient and constant
        k2i = 2 * i
        if 0 <= k2i <= 2 * n - 2:
            C_quad = conv[k2i] - v * v
        else:
            C_quad = 0.0
        
        # Linear terms: k in [i, i+n-1] intersect [0, 2n-2], excluding k=2i
        k_min = max(0, i)
        k_max = min(2 * n - 2, i + n - 1)
        
        if k_min > k_max:
            x_opt = max(0.0, v * 0.5)
            a[i] = x_opt
            S = S - v + x_opt
            continue
        
        ks = np.arange(k_min, k_max + 1)
        mask = ks != k2i
        ks_lin = ks[mask]
        
        if len(ks_lin) > 0:
            coeffs = 2.0 * a[ks_lin - i]  # b_k = 2*a[k-i]
            consts = conv[ks_lin] - 2.0 * v * a[ks_lin - i]  # D_k
        else:
            coeffs = np.array([])
            consts = np.array([])
        
        # Define 1D objective
        def f1d(x, C_quad=C_quad, coeffs=coeffs, consts=consts,
                S_other=S_other, n=n):
            quad_val = x * x + C_quad
            if len(consts) > 0:
                lin_max = float(np.max(consts + coeffs * x))
            else:
                lin_max = -1e30
            m = quad_val if quad_val > lin_max else lin_max
            denom = (S_other + x) * (S_other + x)
            if denom < 1e-30:
                return 1e30
            return 2.0 * n * m / denom
        
        # Analytical candidates
        candidates = [0.0, v]
        
        # Quadratic optimum: x* = C_quad / S_other
        if S_other > 1e-10 and C_quad > 1e-15:
            xq = C_quad / S_other
            candidates.append(xq)
        
        # Linear optimum for the most active constraint
        if len(consts) > 0:
            active_k = int(np.argmax(consts + coeffs * v))
            b_act = coeffs[active_k]
            D_act = consts[active_k]
            if b_act > 1e-12:
                xl = S_other - 2.0 * D_act / b_act
                if xl > 0:
                    candidates.append(xl)
            # Also try the second most active
            if len(consts) > 1:
                sorted_idx = np.argsort(consts + coeffs * v)[::-1]
                for si in sorted_idx[1:3]:
                    b_s = coeffs[si]
                    D_s = consts[si]
                    if b_s > 1e-12:
                        xs = S_other - 2.0 * D_s / b_s
                        if xs > 0:
                            candidates.append(xs)
        
        # Additional heuristics
        candidates.append(1.0 / n)
        candidates.append(2.0 / n)
        candidates.append(v * 0.7)
        candidates.append(v * 1.3)
        
        candidates = sorted(set(max(0.0, float(c)) for c in candidates if np.isfinite(c)))
        
        # Evaluate candidates
        best_x = v
        best_f = f1d(v)
        for c in candidates:
            fc = f1d(c)
            if fc < best_f:
                best_f = fc
                best_x = c
        
        # Golden section refinement
        scale = max(best_x, 1.0 / n, 1e-6)
        x_lo = max(0.0, best_x - 0.5 * scale)
        x_hi = best_x + 1.5 * scale
        if x_hi - x_lo < 1e-12:
            x_hi = x_lo + 1e-8
        
        c_pt = x_hi - gr * (x_hi - x_lo)
        d_pt = x_lo + gr * (x_hi - x_lo)
        fc = f1d(c_pt)
        fd = f1d(d_pt)
        
        for _ in range(35):
            if x_hi - x_lo < 1e-12 * max(1.0, x_hi):
                break
            if fc < fd:
                x_hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = x_hi - gr * (x_hi - x_lo)
                fc = f1d(c_pt)
            else:
                x_lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = x_lo + gr * (x_hi - x_lo)
                fd = f1d(d_pt)
        
        x_opt = max(0.0, (x_lo + x_hi) / 2.0)
        
        if abs(x_opt - v) > 1e-13:
            delta = x_opt - v
            a[i] = x_opt
            S += delta
            # Update conv
            if 0 <= k2i <= 2 * n - 2:
                conv[k2i] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a[ks_lin - i]
    
    return a, conv


def coordinate_descent(a, time_budget=300.0, seed=42, verbose=False):
    """Multiple sweeps of coordinate descent with periodic reconvolution."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    stall_count = 0
    
    while time.monotonic() - start < time_budget:
        sweep += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # Decide order: mostly sequential, sometimes random
        if sweep % 10 == 0:
            order = list(rng.permutation(n))
        else:
            order = list(range(n))
        
        a, conv = cd_single_sweep(a, n, order)
        
        # Recompute objective periodically
        if sweep % 5 == 0 or remaining < 10:
            s = float(np.sum(a))
            if s > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (s * s)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    stall_count = 0
                elif best_obj - obj > 1e-10:
                    # We got worse, revert
                    a = best_a.copy()
                    stall_count += 1
                else:
                    stall_count += 1
        
        # If stalled, perturb and restart
        if stall_count >= 8:
            pert = rng.normal(0, 0.02 * stall_count, n)
            a = best_a * np.exp(pert)
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            stall_count = 0
        
        if verbose and sweep % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_a


def mirror_descent_fft(a, time_budget=100.0, tau_init=50.0, tau_end=500.0,
                       seed=42, verbose=False):
    """
    Mirror descent with FFT-based gradient computation.
    Uses log-sum-exp smoothing of the max-conv objective.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    lr_base = 6.0 / np.sqrt(n)
    it = 0
    stall = 0
    
    # Precompute FFT size for gradient
    N_grad = 1
    while N_grad < 3 * n - 2:
        N_grad <<= 1
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress
        lr = lr_base * (1.0 - 0.6 * progress)
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient via cross-correlation: g[i] = sum_j p[i+j] * a_j
        P = fft(p, N_grad)
        A = fft(a, N_grad)
        g = ifft(P * np.conj(A)).real[:n]
        g = 4.0 * n * g  # Scale by 4n
        
        # Mirror descent update
        g_norm = float(np.max(np.abs(g)))
        if g_norm > 1e-15:
            g_scaled = g / (g_norm + 1e-15)
        else:
            g_scaled = g
        
        update = np.exp(-lr * g_scaled)
        update = np.clip(update, 0.001, 1000.0)
        a_new = a * update
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        conv_new = compute_conv(a_new)
        obj_new = 2.0 * n * float(np.max(conv_new))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        it += 1
        
        # Perturb if stalled
        if stall > 30:
            pert = rng.normal(0, 0.01, n)
            a = best_a * np.exp(pert)
            a = np.maximum(a, 0)
            a = a / float(np.sum(a))
            stall = 0
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.1f}s")
    
    return best_a


def alternating_cd_md(a, time_budget=300.0, seed=42, verbose=False):
    """
    Alternate between CD sweeps and mirror descent steps.
    CD provides fast local convergence; MD helps escape local minima.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Nesterov-like momentum for the overall sequence
    v_mom = np.zeros(n)
    beta = 0.3
    
    it = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        # Phase: CD for a bit
        cd_time = min(remaining * 0.4, 60.0)
        if cd_time > 5:
            a_cd = coordinate_descent(a, time_budget=cd_time, seed=seed + it,
                                      verbose=False)
            obj_cd = obj_normalized(a_cd)
            if obj_cd < best_obj:
                best_obj = obj_cd
                best_a = a_cd.copy()
            a = best_a
        
        # Phase: Mirror descent for a bit
        remaining = time_budget - (time.monotonic() - start)
        md_time = min(remaining * 0.3, 40.0)
        if md_time > 5:
            a_md = mirror_descent_fft(a, time_budget=md_time,
                                      tau_init=50.0, tau_end=300.0,
                                      seed=seed + 1000 + it, verbose=False)
            obj_md = obj_normalized(a_md)
            if obj_md < best_obj:
                best_obj = obj_md
                best_a = a_md.copy()
            a = best_a
        
        # Blend with momentum
        v_mom = beta * v_mom + (1 - beta) * (a - best_a)
        a_candidate = best_a + v_mom
        a_candidate = np.maximum(a_candidate, 0)
        s_cand = float(np.sum(a_candidate))
        if s_cand > 1e-10:
            a_candidate = a_candidate / s_cand
            obj_cand = obj_normalized(a_candidate)
            if obj_cand < best_obj:
                best_obj = obj_cand
                best_a = a_candidate.copy()
        
        it += 1
        
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    ALT iter{it}: obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_a


def adversarial_cd(a, time_budget=100.0, seed=42, verbose=False):
    """
    Coordinate descent that prioritizes coordinates involved in the worst lags.
    Identifies the k's with maximum (a*a)_k and updates coordinates that
    contribute to those lags.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    sweep = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        sweep += 1
        conv = compute_conv(a)
        S = float(np.sum(a))
        
        # Find top-K worst lags
        K = max(5, n // 20)
        top_k = np.argsort(conv)[::-1][:K]
        
        # Build priority order: coordinates involved in top lags first
        priority = np.zeros(n)
        for k in top_k:
            weight = conv[k] - np.median(conv) + 1
            for j in range(n):
                if 0 <= k - j <= n - 1:
                    priority[j] += weight * a[j] * a[k - j]
        
        # Sort by priority (descending)
        order = list(np.argsort(priority)[::-1])
        
        a, conv = cd_single_sweep(a, n, order)
        
        if sweep % 10 == 0:
            s = float(np.sum(a))
            if s > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (s * s)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
        
        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    AdvCD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        x = np.arange(n)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Semicaircular (beta-like)
        x = np.linspace(-0.999, 0.999, n)
        a = 1.0 / np.sqrt(1 - x ** 2 + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 5:
        # Multi-peak sinusoidal
        freqs = rng.integers(2, 15, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.abs(np.sin(np.pi * f * x + p))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Tent
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2 * x - 1)
        a += 0.1 * rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 7:
        # Rectangular window
        w = max(1, n // 3)
        start_idx = rng.integers(0, n - w)
        a = np.zeros(n)
        a[start_idx:start_idx + w] = 1.0
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining fast CD, FFT mirror descent,
    adversarial CD, alternating schemes, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent if available
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = obj_normalized(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
            else:
                raise ValueError("Invalid parent")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = obj_normalized(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")
    
    n_best = len(best_seq)
    
    # === Phase 1: Fast CD (main workhorse) ===
    print(f"\n=== Phase 1: Fast Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 350.0)
    if p1_budget > 30:
        a_cd = coordinate_descent(best_seq, time_budget=p1_budget,
                                  verbose=True, seed=seed)
        obj_cd = obj_normalized(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)")
    
    # === Phase 2: FFT Mirror Descent ===
    print(f"\n=== Phase 2: Mirror Descent with FFT ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 200.0)
    if p2_budget > 20:
        a_md = mirror_descent_fft(best_seq, time_budget=p2_budget,
                                  tau_init=50.0, tau_end=400.0,
                                  seed=seed + 100, verbose=True)
        obj_md = obj_normalized(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  MD: obj={best_score:.8f} **")
            # Follow up with CD
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_cd2 = coordinate_descent(a_md, time_budget=min(remaining * 0.15, 80.0),
                                           seed=seed + 200, verbose=False)
                obj_cd2 = obj_normalized(a_cd2)
                if obj_cd2 < best_score:
                    best_score = obj_cd2
                    best_seq = a_cd2.copy()
                    print(f"  MD+CD: obj={best_score:.8f} **")
        else:
            print(f"  MD: obj={obj_md:.8f}")
    
    # === Phase 3: Adversarial CD ===
    print(f"\n=== Phase 3: Adversarial CD ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 150.0)
    if p3_budget > 15:
        a_adv = adversarial_cd(best_seq, time_budget=p3_budget,
                               seed=seed + 300, verbose=True)
        obj_adv = obj_normalized(a_adv)
        if obj_adv < best_score:
            best_score = obj_adv
            best_seq = a_adv.copy()
            print(f"  AdvCD: obj={best_score:.8f} **")
            # Follow up with standard CD
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_cd3 = coordinate_descent(a_adv, time_budget=min(remaining * 0.1, 60.0),
                                           seed=seed + 400, verbose=False)
                obj_cd3 = obj_normalized(a_cd3)
                if obj_cd3 < best_score:
                    best_score = obj_cd3
                    best_seq = a_cd3.copy()
                    print(f"  AdvCD+CD: obj={best_score:.8f} **")
        else:
            print(f"  AdvCD: obj={obj_adv:.8f}")
    
    # === Phase 4: Alternating CD + MD ===
    print(f"\n=== Phase 4: Alternating CD + Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 200.0)
    if p4_budget > 30:
        a_alt = alternating_cd_md(best_seq, time_budget=p4_budget,
                                  seed=seed + 500, verbose=True)
        obj_alt = obj_normalized(a_alt)
        if obj_alt < best_score:
            best_score = obj_alt
            best_seq = a_alt.copy()
            print(f"  ALT: obj={best_score:.8f} **")
        else:
            print(f"  ALT: obj={obj_alt:.8f}")
    
    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    
    n_options = sorted(set([
        n_current - 100, n_current - 50,
        n_current + 50, n_current + 100, n_current + 200,
        500, 600, 700, 900, 1000, 1100, 1200, 1500
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000 and abs(n - n_current) > 30]
    n_options = n_options[:5]
    
    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        
        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        
        # CD on new length
        alloc_cd = min(remaining * 0.25, 80.0)
        if alloc_cd > 15:
            a_try = coordinate_descent(a_try, time_budget=alloc_cd,
                                       verbose=False, seed=n_try)
        
        # Brief mirror descent
        remaining = deadline - time.monotonic()
        alloc_md = min(remaining * 0.08, 30.0)
        if alloc_md > 10:
            a_try = mirror_descent_fft(a_try, time_budget=alloc_md,
                                       tau_init=50.0, tau_end=200.0,
                                       seed=n_try + 2000, verbose=False)
        
        # Final CD
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_try = coordinate_descent(a_try, time_budget=min(remaining * 0.1, 30.0),
                                       verbose=False, seed=n_try + 3000)
        
        obj_try = obj_normalized(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")
    
    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    
    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        
        mode = (r % 8) + 1
        a_r = diverse_init(n_cur, rng, mode=mode)
        blend = 0.5 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r < 1e-10:
            continue
        a_r = a_r / s_r
        
        alloc = min(remaining * 0.2, 80.0)
        if alloc > 20:
            a_r = coordinate_descent(a_r, time_budget=alloc,
                                     verbose=False, seed=9000 + r * 111)
            obj_r = obj_normalized(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")
    
    # === Phase 7: Final intensive polishing ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Extended CD
        alloc = min(remaining * 0.4, 200.0)
        a_final = coordinate_descent(best_seq, time_budget=alloc,
                                     verbose=True, seed=seed + 19999)
        obj_final = obj_normalized(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")
    
    # One more round of MD + CD if time
    remaining = deadline - time.monotonic()
    if remaining > 40:
        a_md_f = mirror_descent_fft(best_seq, time_budget=min(remaining * 0.3, 60.0),
                                    tau_init=100.0, tau_end=500.0,
                                    seed=seed + 25000, verbose=False)
        obj_md_f = obj_normalized(a_md_f)
        if obj_md_f < best_score:
            best_score = obj_md_f
            best_seq = a_md_f.copy()
            print(f"  Final MD: obj={best_score:.8f} **")
        
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_cd_f = coordinate_descent(best_seq, time_budget=min(remaining * 0.5, 50.0),
                                        verbose=False, seed=seed + 30000)
            obj_cd_f = obj_normalized(a_cd_f)
            if obj_cd_f < best_score:
                best_score = obj_cd_f
                best_seq = a_cd_f.copy()
                print(f"  Final CD2: obj={best_score:.8f} **")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = obj_normalized(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
