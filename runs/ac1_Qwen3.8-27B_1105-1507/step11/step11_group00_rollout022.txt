import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Assumes a >= 0."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def conv_gradient(a, p):
    """
    Compute gradient of sum_k p[k] * conv(a,a)[k] with respect to a.
    This equals 2 * (reverse(a) * p) centered, i.e., 2 * conv_reverse.
    Specifically: grad[i] = 2 * sum_k p[k] * (contribution of a[i] to conv(a,a)[k])
                 = 2 * sum_k p[k] * sum_j a[j]*delta_{j+i=k or j=k-i}
    More simply: grad = 2 * conv(a, p) shifted appropriately.
    Actually: d/dx_i sum_k p[k] * sum_j a[j]*a[k-j] = 2 * sum_k p[k] * a_at_k_minus_i
    which is 2 * (a * p)_i where (a*p) is cross-correlation with p reversed.
    Using FFT: grad[i] = 2 * ifft(fft(a) * conj(fft(p_rev)))... 
    Let me just do it directly with the cross-correlation interpretation.
    
    d conv(a,a)[k] / d a[i] = a[k-i] + a[i-i+k]... 
    
    Actually the cleanest: conv(a,a)[k] = sum_j a[j]*a[k-j]
    d/d a[i] conv(a,a)[k] = a[k-i] (if 0<=k-i<n) + a[i] * (1 if i=k-i, i.e., k=2i)
    
    So grad[i] = 2 * sum_k p[k] * a[k-i] (for valid k-i in [0,n-1])
               (the factor of 2 accounts for both a[i] appearing as a[j] and a[k-j])
    
    This is 2 * cross_correlation(a, p) where cross_correlation[i] = sum_k a[k-i]*p[k]
    = 2 * (reverse(a) * p)[n-1-i] ... let me just use FFT.
    
    grad[i] = 2 * sum_k p[k] * a[k-i] for 0 <= k-i < n
    Let b = a reversed. Then sum_k p[k]*a[k-i] = sum_k p[k]*b[n-1-(k-i)]
    = sum_k p[k]*b[n-1-k+i] = (p * b)[n-1+i] where * is convolution.
    So grad = 2 * (convolve(p, b))[n-1 : n-1+n]
    """
    n = len(a)
    b = a[::-1]
    N_fft = 1
    while N_fft < 3 * n - 1:
        N_fft <<= 1
    c = irfft(rfft(b, N_fft) * rfft(p, N_fft), N_fft)
    return 2.0 * c[n - 1:2 * n - 1]


def lse_gradient(a, conv, tau):
    """Gradient of logsumexp(tau * conv(a,a))/tau with respect to a."""
    n = len(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    if shifted.max() > 500:
        shifted = shifted - shifted.max()
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    return conv_gradient(a, p)


def topk_gradient(a, conv, p_vec, k=5):
    """Subgradient using top-k lags with water-filling weights."""
    n = len(a)
    idx = np.argsort(conv)[-k:]
    p = np.zeros(len(conv))
    p[idx] = 1.0 / k
    return conv_gradient(a, p)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = float(np.sum(w))
    if s < 1e-300:
        return np.ones(n) / n
    return w / s


def spectral_precondition(a, conv, tau):
    """
    Compute a spectral preconditioner using the diagonal of the local Hessian
    approximated via the power spectrum.
    """
    n = len(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    if shifted.max() > 500:
        shifted = shifted - shifted.max()
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    # The "diagonal" of the Hessian of the smoothed objective is:
    # H_ii ≈ 2 * sum_k p[k] * a[k-i]^2 ... not quite, but we can approximate
    # the curvature at each coordinate via 2*sum_k p[k]*a[k-i]
    # Actually the Hessian entry H[i,j] = 2*sum_k p[k]*delta_{i+j,k} (for i≠j)
    # The "diagonal" (i=j) is 2*sum_k p[k]*delta_{2i,k} which is sparse
    # A good preconditioner is the row sum of |H|: d[i] = 2*sum_k p[k]*|a[k-i]|
    # But since a >= 0, this is 2*sum_k p[k]*a[k-i] = the gradient itself (up to factor)
    # So the gradient magnitude IS a good preconditioner
    return np.maximum(np.abs(conv_gradient(a, p)), 1e-10)


def active_set_compute(a, conv, n_lags=10):
    """Return indices of the top n_lags peaks in the convolution."""
    return np.argsort(conv)[-n_lags:]


def active_set_gradient(a, conv, active_idx):
    """Compute gradient using only active lags (uniform weights)."""
    n = len(a)
    p = np.zeros(len(conv))
    p[active_idx] = 1.0 / len(active_idx)
    return conv_gradient(a, p)


def subgradient_optimize(a, time_budget=300.0, tau_start=30.0, tau_end=3000.0,
                         verbose=True, use_active=False, n_active=8,
                         use_precond=False, nesterov_beta=0.0):
    """
    Projected subgradient (or Nesterov-accelerated) method with adaptive smoothing.
    Projects onto the simplex at each step.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    # Lower bound for Polyak step size: 2n/(2n-1) is the theoretical min
    lb = 2.0 * n / (2.0 * n - 1.0)
    # A practical lower bound (slightly above theoretical)
    lb_practical = lb + 0.05  # conservative

    it = 0
    best_it = 0
    stall = 0
    v = np.zeros(n)  # for Nesterov momentum

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1.0:
            break

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Adaptive tau: smooth early, sharp late
        tau = tau_start * (tau_end / tau_start) ** progress

        # Compute y (Nesterov lookahead)
        if nesterov_beta > 0:
            y = a + nesterov_beta * v
            y = project_simplex(y)
        else:
            y = a.copy()

        conv = compute_conv(y)
        max_c = float(np.max(conv))
        current_obj = 2.0 * n * max_c

        # Compute gradient
        if use_active and progress > 0.3:
            active_idx = active_set_compute(y, conv, n_active)
            grad = active_set_gradient(y, conv, active_idx)
        else:
            grad = lse_gradient(y, conv, tau)

        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break

        # Step size
        if use_precond:
            # Normalized step
            step = grad / (grad_norm + 1e-10)
            # Polyak-like step size
            polyak_step = max((current_obj - lb_practical) / (grad_norm + 1e-10), 1e-8)
            eta = min(polyak_step, 2.0 / np.sqrt(n))
            a_new = y - eta * step
        else:
            # Standard diminishing step with Polyak averaging
            # Step size: c / sqrt(t) with adaptive c
            c = 2.0 / np.sqrt(n)
            eta = c / (np.sqrt(it + 1) * 0.1 + 1.0)
            # Scale by current gap
            gap = max(current_obj - lb_practical, 0.1)
            eta *= min(gap, 2.0) / max(grad_norm / np.sqrt(n), 0.1)
            eta = min(eta, 5.0 / np.sqrt(n))
            a_new = y - eta * grad

        # Project onto simplex
        a_new = project_simplex(a_new)

        # Nesterov momentum update
        if nesterov_beta > 0:
            v = (a_new - a) / max(nesterov_beta, 1e-10)
            # Dampen momentum
            v *= 0.95

        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall > 500 and progress > 0.5:
                # Reset to best and try smaller steps
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0

        it += 1
        if verbose and (it % 200 == 0 or it == 1):
            print(f"    SubGrad it{it}: obj={best_obj:.8f}, tau={tau:.0f}, "
                  f"stall={stall}, t={time.monotonic()-start:.0f}s")

    return best_a


def nesterov_high_tau(a, time_budget=100.0, tau_start=200.0, tau_end=10000.0,
                      verbose=False):
    """Nesterov-accelerated gradient descent with high smoothing for polishing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.75
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0
    prev_t = None

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start * (tau_end / tau_start) ** (progress ** 1.3)

        # Nesterov lookahead
        if it > 0:
            t_k = 0.5 + np.sqrt(it * it + 4 * t_k * t_k) / 2.0
        else:
            t_k = 1.0
        if prev_t is None:
            prev_t = 1.0
        beta_k = (1.0 - prev_t) / t_k
        prev_t = t_k

        y = a + beta_k * (a - (a - v))  # simplified Nesterov
        # Actually standard Nesterov: y = x_k + beta_k * (x_k - x_{k-1})
        # Let's use the simpler form: y = x + momentum
        y = a + beta * v
        y = project_simplex(y)

        conv_y = compute_conv(y)
        grad = lse_gradient(y, conv_y, tau)
        g_norm = float(np.linalg.norm(grad))

        if g_norm < 1e-15:
            break

        # Adaptive learning rate
        adapt = 1.0 / (g_norm + 1e-10)
        lr = 0.9 * lr + 0.1 * min(max(adapt * 0.5, lr * 0.3), lr * 3.0)

        v = beta * v - lr * grad
        a_new = y + v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = lr * 0.5
            best_it = it

    if verbose:
        print(f"    NesterovHighTau: obj={best_obj:.8f}, iters={it}")
    return best_a


def mirror_descent_fast(a, time_budget=200.0, tau_init=50.0, tau_end=2000.0,
                        verbose=False):
    """
    Fast mirror descent (multiplicative weights) with simplex projection fallback.
    Uses adaptive step sizes and periodic restart.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)

    it = 0
    best_it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init * (tau_end / tau_init) ** progress

        conv = compute_conv(a)
        grad = lse_gradient(a, conv, tau)
        g_max = float(np.max(np.abs(grad)))
        if g_max < 1e-15:
            break

        # Multiplicative weights step
        eta = 1.5 / (g_max * np.sqrt(n) + 1e-10)
        eta *= (1.0 + 0.3 * (1.0 - progress))  # larger steps early

        exponent = -eta * grad
        exponent = np.clip(exponent, -30, 30)
        a_new = a * np.exp(exponent)

        # Add small random perturbation periodically
        if it % 50 == 25 and progress < 0.7:
            noise = 0.005 * (1.0 - progress)
            a_new += noise * rng.standard_normal(n)

        # Project onto simplex
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall > 100:
                # Try a smaller step
                a_half = a * np.exp(-eta * 0.2 * grad)
                a_half = project_simplex(np.maximum(a_half, 0))
                obj_half = compute_obj(a_half)
                if obj_half < best_obj:
                    best_obj = obj_half
                    best_a = a_half.copy()
                    a = a_half
                    stall = 0
                elif stall > 300:
                    # Perturb
                    a = best_a * np.exp(0.03 * (1.0 + progress) * rng.standard_normal(n))
                    a = project_simplex(np.maximum(a, 0))
                    stall = 0

        a = a_new
        it += 1

        if verbose and (it % 100 == 0):
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, "
                  f"stall={stall}, t={time.monotonic()-start:.0f}s")

    return best_a


def lp_refined_direction(a):
    """
    Solve a simplified LP to find a good search direction.
    min sum(g) subject to: a^T C_k g <= M_k for all k, sum(g)=1, g>=0
    where M_k is the current conv value at each k.
    This finds the direction that most reduces the peak.
    Uses a simplified approach: for each active lag, find the direction
    that most reduces it, then combine.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv = compute_conv(a_norm)
    max_c = float(np.max(conv))

    # For each top lag, compute the "pressure" direction
    top_idx = np.argsort(conv)[-5:]
    directions = []
    for k in top_idx:
        # Direction that reduces conv[k]: reduce a[i]*a[k-i] for all valid i
        # The "easiest" reduction is to reduce the largest product
        # Approximate direction: reduce positions that contribute most to lag k
        contrib = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                contrib[i] = a_norm[j]
            j2 = i + k  # also consider the other direction
        # Actually for conv[k] = sum_j a[j]*a[k-j], the contribution of a[i]
        # is a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n) for i in 0..n-1
        # The gradient of conv[k] wrt a is 2*a[k-i] (for valid k-i)
        # Direction to reduce conv[k]: -gradient_normalized
        grad_k = conv_gradient(a_norm, np.eye(len(conv))[k])
        # This is 2*a[k-i] for valid i
        norm_k = float(np.linalg.norm(grad_k))
        if norm_k > 1e-10:
            directions.append(-grad_k / norm_k)

    if not directions:
        return None

    # Combine: average of top directions, keep non-negative part
    d = np.mean(directions, axis=0)
    d = np.maximum(d, 0)
    s_d = float(np.sum(d))
    if s_d < 1e-10:
        # Fallback: uniform direction (which is zero change)
        return None
    return d / s_d


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def chi_squared_init(n, seed=42):
    """Initialize using a chi-squared-likedistribution for good spectral properties."""
    rng = np.random.default_rng(seed)
    # Create a sequence with good autocorrelation properties
    # Start with a smoothed version of a Barker-like pattern
    a = rng.uniform(0.5, 1.5, n)
    # Apply a smooth window
    window = 0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n) / n)
    a = a * (0.5 + 0.5 * window)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s


def fourier_optimized_init(n, seed=42):
    """
    Initialize by optimizing in the frequency domain.
    We want |FFT(a)|^2 to be as flat as possible (for flat autocorrelation).
    This is equivalent to making a as "white" as possible in frequency.
    """
    rng = np.random.default_rng(seed)
    # Start with random phase in frequency domain
    freq = np.zeros(n, dtype=complex)
    freq[0] = 1.0  # DC component
    for k in range(1, n):
        freq[k] = rng.uniform(0.8, 1.2) * np.exp(1j * rng.uniform(0, 2 * np.pi))
    # Enforce Hermitian symmetry for real output
    if n % 2 == 0:
        for k in range(1, n // 2):
            freq[n - k] = np.conj(freq[k])
    else:
        for k in range(1, n // 2 + 1):
            freq[n - k] = np.conj(freq[k])
    a = np.real(ifft(freq))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining multiple optimization strategies with different
    initialization and n values to find the best step function coefficients.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # === Initialize from parent or default ===
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default uniform init (n={n0}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Subgradient with simplex projection (coarse) ===
    print(f"\n=== Phase 1: Subgradient (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 30:
        a_sub = subgradient_optimize(best_seq, time_budget=p1_budget,
                                     tau_start=20.0, tau_end=500.0,
                                     verbose=True, nesterov_beta=0.7)
        obj_sub = compute_obj(a_sub)
        if obj_sub < best_score:
            best_score = obj_sub
            best_seq = a_sub.copy()
            print(f"  Phase1 SubGrad: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 SubGrad: obj={obj_sub:.8f} (no improvement)")

    # === Phase 2: Mirror Descent (refined) ===
    print(f"\n=== Phase 2: Mirror Descent (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 250.0)
    if p2_budget > 30:
        a_md = mirror_descent_fast(best_seq, time_budget=p2_budget,
                                   tau_init=50.0, tau_end=2000.0, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Phase2 MD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 MD: obj={obj_md:.8f} (no improvement)")

    # === Phase 3: Nesterov high-tau polish ===
    print(f"\n=== Phase 3: Nesterov High-Tau Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 200.0)
    if p3_budget > 20:
        a_nes = nesterov_high_tau(best_seq, time_budget=p3_budget,
                                  tau_start=200.0, tau_end=8000.0, verbose=True)
        obj_nes = compute_obj(a_nes)
        if obj_nes < best_score:
            best_score = obj_nes
            best_seq = a_nes.copy()
            print(f"  Phase3 Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 Nesterov: obj={obj_nes:.8f} (no improvement)")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Try a range of n values around the current best
    n_options = sorted(set([
        n_best - 50, n_best - 20, n_best + 20, n_best + 50,
        n_best + 100, n_best - 100,
        600, 900, 1100, 1400, 1800, 2200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 3000]
    # Limit to 6 best candidates
    n_options = n_options[:6]
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Optimize: subgradient + MD
        alloc1 = min(remaining * 0.3, 60.0)
        if alloc1 > 15:
            a_try = subgradient_optimize(a_try, time_budget=alloc1,
                                         tau_start=30.0, tau_end=500.0,
                                         verbose=False, nesterov_beta=0.6)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.25, 50.0)
        if alloc2 > 10:
            a_try = mirror_descent_fast(a_try, time_budget=alloc2,
                                        tau_init=100.0, tau_end=1000.0, verbose=False)

        remaining = deadline - time.monotonic()

    # Mirrored + smoothed

    a_mir = project_simplex(np.maximum(a_mir, 0))
    inits.append(("mirror_sym", a_mir))
    # Perturbed best
    a_pert = best_seq * np.exp(0.05 * rng.standard_normal(n_cur))
    a_pert = project_simplex(np.maximum(a_pert, 0))
    inits.append(("perturbed", a_pert))

    for name, a_init in inits:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        # Quick optimization
        alloc1 = min(remaining * 0.3, 50.0)
        if alloc1 > 15:
            a_opt = subgradient_optimize(a_init, time_budget=alloc1,
                                         tau_start=50.0, tau_end=400.0,
                                         verbose=False, nesterov_beta=0.7)
        else:
            a_opt = a_init

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.2, 30.0)

            a_opt = mirror_descent_fast(a_opt, time_budget=alloc2,

    print(f"\n=== Phase 6: Final Polish ===")

        a_final = nesterov_high_tau(best_seq, time_budget=alloc,

        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final Nesterov: obj={best_score:.8f} **")

    if remaining > 30:
        alloc = min(remaining * 0.3, 100.0)
        a_final2 = subgradient_optimize(best_seq, time_budget=alloc,

        obj_final2 = compute_obj(a_final2)

    return result
