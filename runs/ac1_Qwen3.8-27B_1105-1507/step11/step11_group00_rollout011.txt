Optimizes step function coefficients to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses a combination of structured initialization, adaptive smoothed-max gradient
descent with heavy-ball momentum, peak-shaving, and simulated annealing.

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

import numpy as np

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smoothing_gradient(a, conv, tau):
    """Gradient of soft-max tau w.r.t. a, times tau (so it's the gradient of tau * soft_max)."""
    n = len(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    if shifted.max() > 500:
        shifted = shifted - shifted.max()
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(2 * n - 1)
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    
    N_fft = 1
    while N_fft < 3 * n:
        N_fft <<= 1
    flip_a = a[::-1]
    c = irfft(rfft(flip_a, N_fft) * rfft(p, N_fft), N_fft)
    return 2.0 * c[n-1:2*n-1]

def topk_gradient(a, conv, k=5):

    n = len(a)
    idx = np.argsort(conv)[-k:]

    p[idx] = 1.0 / k
    
    N_fft = 1

        N_fft <<= 1

def project_simplex(v):

    v = np.maximum(v, 0)
    s = float(np.sum(v))
    if s < 1e-300:
        return np.ones(len(v)) / len(v)

def peak_shaving(a, n_peaks=5, n_steps=10):

def make_raised_cosine(n, width=1.0):
    """Raised cosine window."""
    x = np.linspace(0, width, n)

def make_two_sided(n, width_frac=0.3):

    a = np.zeros(n)
    w = max(1, int(n * width_frac))
    x = np.linspace(-1, 1, w)
    a[:w] = np.exp(-x**2 * 3)
    a[-w:] = np.exp(-x**2 * 3)
    return project_simplex(a)

def make_fourier(n, seed=42):
    """Random Fourier-based initialization."""
    rng = np.random.default_rng(seed)
    # Random real and imaginary parts
    freqs = rfft(np.zeros(n), n)
    a = np.zeros(n)

        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    return project_simplex(a)

def resample(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_simplex(a_new)
