"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of: aggressive LP fixed-point with full replacement (t=1),
multiplicative (mirror descent) updates, dual-based optimization, adaptive
smoothed gradient descent, and multi-n exploration.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    shifted = tau * (conv - exact_max)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def lp_solve(a, solver_list=None):
    """Solve min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1. Returns (g, M)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None


def lp_fixed_point_aggressive(a, time_budget, verbose=False, seed=42, max_iters=300):
    """
    Aggressive LP fixed-point: try t=1 first, then decreasing t.
    Also uses dual information to guide perturbations.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall_count = 0
    consecutive_fails = 0

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, M_lp = lp_solve(a)
        if g_lp is None:
            consecutive_fails += 1
            if consecutive_fails > 8:
                break
            # Perturb and retry
            a = best_a + 0.005 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            continue
        consecutive_fails = 0

        # Try t=1 first (full replacement), then decreasing
        t_candidates = [1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]
        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = 0.0

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a
            stall_count = 0
            if verbose and (it % 5 == 0 or best_local_t == 1.0):
                print(f"    LP-A it{it:3d}: obj={best_obj:.10f}, t={best_local_t:.3f}, M={M_lp:.8f}")
        else:
            stall_count += 1
            if stall_count % 5 == 0:
                # Perturb based on worst lags
                conv = compute_conv(a)
                max_val = float(np.max(conv))
                worst_lags = np.where(conv >= max_val * 0.995)[0]
                if len(worst_lags) > 0:
                    # Compute gradient to push down worst lags
                    subgrad = np.zeros(n)
                    for lag in worst_lags[:5]:
                        for i in range(n):
                            j = lag - i
                            if 0 <= j < n:
                                subgrad[i] += a[j]
                    subgrad /= len(worst_lags[:5])
                    gn = np.linalg.norm(subgrad)
                    if gn > 1e-15:
                        a = a - 0.002 * subgrad / gn
                else:
                    a = a + 0.005 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                sa = float(np.sum(a))
                if sa > 1e-10:
                    a = a / sa
                else:
                    a = best_a.copy()

            if stall_count > 50:
                a = best_a.copy()
                a = a + 0.03 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_count = 0

    return best_a


def multiplicative_descent(a, time_budget, verbose=False, seed=42, max_iters=500):
    """
    Multiplicative (mirror descent) updates on log(a).
    a_i <- a_i * exp(-lr * grad_i) naturally maintains non-negativity.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).copy().astype(np.float64)
    s = float(np.sum(a))
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    lr = 2.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))

        # Find active lags (near-max)
        threshold = max_val * 0.998
        active = np.where(conv >= threshold)[0]

        # Subgradient from active lags
        subgrad = np.zeros(n)
        for lag in active:
            for i in range(n):
                j = lag - i
                if 0 <= j < n:
                    subgrad[i] += 2.0 * a[j]
        subgrad /= len(active)

        # Momentum
        v = momentum * v + (1 - momentum) * subgrad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-20:
            break
        v = v / v_norm

        # Multiplicative update
        a_new = a * np.exp(-lr * v)
        a_new = np.maximum(a_new, 1e-15)
        sn = float(np.sum(a_new))
        if sn < 1e-10:
            a_new = best_a.copy()
            continue
        a_new = a_new / sn

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if verbose and it % 20 == 0:
                print(f"    MD it{it:3d}: obj={best_obj:.10f}, lr={lr:.5f}")
        else:
            # Reduce learning rate
            lr *= 0.9
            if it % 30 == 0 and it > 0:
                # Perturb
                a = best_a + lr * 0.5 * rng.standard_normal(n)
                a = np.maximum(a, 1e-10)
                a = a / a.sum()

        # Adapt lr
        lr = max(0.01, min(5.0, lr))

    return best_a


def dual_ascent(a, time_budget, verbose=False, seed=42, max_iters=200):

    rng = np.random.default_rng(seed)

        remaining = time_budget - (time.monotonic() - start)

            break

def penalty_optimize(a, time_budget, verbose=False, seed=42, max_iters=300):
    """
    Penalty/regularization approach: minimize sum(conv^2) + mu * max(conv)^2
    using smoothed gradient. This smooths the max and allows better optimization.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Use L-BFGS-B style via scipy
    from scipy.optimize import minimize

    def obj_func(x):
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx < 0.01:
            return 1e10
        x = x / sx
        conv = compute_conv(x)
        return 2.0 * n * float(np.max(conv))

    for attempt in range(3):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 30:
            break

        x0 = a + 0.01 * rng.standard_normal(n)
        x0 = np.maximum(x0, 0)
        x0 = x0 / x0.sum()

        try:
            # Use BFGS with bounds
            res = minimize(
                obj_func, x0, method='L-BFGS-B',
                bounds=[(0, None)] * n,
                options={'maxiter': 500, 'maxfun': 2000, 'ftol': 1e-12, 'gtol': 1e-8}
            )
            if res.x is not None:
                a_try = np.maximum(res.x, 0)
                st = float(np.sum(a_try))
                if st > 0.01:
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        if verbose:
                            print(f"    L-BFGS attempt {attempt}: obj={best_obj:.10f}")
        except Exception:
            continue

    return best_a


def coordinate_descent(a, time_budget, verbose=False, seed=42):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate i, the contribution to conv is a_i appearing in conv[i] and conv[lag] for various lags.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Precompute which lags each coordinate affects
    # a_i contributes to conv[k] for k where a_{k-i} exists
    # conv[k] = sum_j a_j * a_{k-j}, so a_i appears when j=i (term a_i * a_{k-i})
    # and when k-j=i, i.e., j=k-i (term a_{k-i} * a_i)
    # So a_i contributes 2*a_i*a_{k-i} to conv[k] (for k-i in [0,n-1])
    # plus a_i^2 to conv[2i] (the self-term)

    sweeps = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        improved = False
        # Random order for diversity
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget - 5:
                break

            # Current value
            ai = a[i]
            if ai < 1e-15:
                continue

            # Compute gradient w.r.t. a_i
            # d(conv[k])/da_i = a_{k-i} + a_{k-i} = 2*a_{k-i} (for k-i in [0,n-1], k != 2i)
            # d(conv[2i])/da_i = 2*a_i
            # The subgradient of max is the gradient at the max lags
            conv = compute_conv(a)
            max_val = float(np.max(conv))
            active = np.where(conv >= max_val * 0.999)[0]

            grad_i = 0.0
            for lag in active:
                j = lag - i
                if 0 <= j < n:
                    if lag == 2 * i:
                        grad_i += 2.0 * a[j]
                    else:
                        grad_i += 2.0 * a[j]  # symmetric contribution

            # Optimal a_i: if we increase a_i, it affects conv linearly
            # For small step: a_i_new = a_i - step * grad_i / (2 * s * n) * 2*n
            # Simplified: try a few values
            # The objective contribution from a_i is complex, so use line search
            candidates = [ai, ai * 0.9, ai * 0.8, ai * 0.7, ai * 1.1, ai * 1.2, 0.0]
            best_ai = ai
            best_local_obj = best_obj

            for cand_ai in candidates:
                a[i] = cand_ai
                s_new = float(np.sum(a))
                if s_new < 0.01:
                    a[i] = ai
                    continue
                # Quick eval
                a_norm = a / s_new
                conv_new = compute_conv(a_norm)
                obj_new = 2.0 * n * float(np.max(conv_new))
                if obj_new < best_local_obj - 1e-14:
                    best_local_obj = obj_new
                    best_ai = cand_ai
                    a[i] = cand_ai
                else:
                    a[i] = ai

            a[i] = best_ai
            if best_ai != ai:
                improved = True

        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s
        obj_now = compute_obj(a)
        if obj_now < best_obj - 1e-14:
            best_obj = obj_now
            best_a = a.copy()
            if verbose and sweeps % 5 == 0:
                print(f"    CD sweep {sweeps}: obj={best_obj:.10f}")
        sweeps += 1

        if not improved and sweeps > 10:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_structured_init(n, mode=0):
    """Create structured initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.maximum(np.sinc(x), 0.001)
        return a / a.sum()
    elif mode == 2:
        a = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
        return a / a.sum()
    elif mode == 3:
        x = np.arange(n) / (n - 1)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        return a / a.sum()
    elif mode == 4:
        x = np.linspace(-3, 3, n)
        a = np.exp(-x ** 2)
        return a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
        return a / a.sum()
    elif mode == 6:
        # Fejer kernel (triangular)
        x = np.linspace(-1, 1, n)
        a = np.maximum(1 - np.abs(x), 0.001)
        return a / a.sum()
    else:
        return np.ones(n) / n


def smoothed_gd(a, time_budget, verbose=False, seed=42):
    """Smoothed gradient descent with adaptive tau and step size."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    tau = 10.0
    lr = 1.0 / np.sqrt(n)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        g, current_max = smoothed_grad(a, tau)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Step size: try a few
        best_step_obj = best_obj
        best_step_a = a.copy()
        for step in [lr, lr * 0.5, lr * 0.2, lr * 2.0]:
            a_try = a - step * g / g_norm
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_step_obj - 1e-14:
                best_step_obj = obj_try
                best_step_a = a_try.copy()

        if best_step_obj < best_obj - 1e-14:
            best_obj = best_step_obj
            best_a = best_step_a.copy()
            a = best_a
            stall = 0
            tau = tau * 1.05  # slowly increase tau
            if verbose and int(tau) % 20 == 0:
                print(f"    GD tau={tau:.0f}: obj={best_obj:.10f}")
        else:
            stall += 1
            lr *= 0.9
            if stall % 20 == 0:
                tau *= 0.8
                a = best_a + 0.005 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()

        tau = min(tau, 5000.0)
        lr = max(lr, 1e-6)

    return best_a
