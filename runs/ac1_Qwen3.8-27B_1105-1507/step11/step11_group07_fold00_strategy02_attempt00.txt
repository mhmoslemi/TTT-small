def soft_max_gradient_descent_with_constraint(a, time_budget=10.0, beta_start=5.0, beta_end=20.0,
                                           momentum=0.9, lr_start=0.005, sharpness_factor=0.7):
    """
    Gradient descent that minimizes the score with constraints.
    """
    a_norm = a / np.sum(a)
    beta_values = np.linspace(beta_start, beta_end, int(time_budget)+1)
    
    # Update candidates using damped gradients
    for beta_val in beta_values:
        # --- Caution: ensure non-negative values ---
        conv = np.convolve(a_norm, a_norm, mode='full')
        max_conv = np.max(conv)
        # --- Set gradient constraint ---
        if max_conv > np.inf:  # Safety check
