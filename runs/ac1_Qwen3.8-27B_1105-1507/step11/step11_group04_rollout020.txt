"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a hybrid approach: mirror descent for fast exploration, LP alternating for
precise local optimization, and coordinate descent with analytic steps for
final polishing. Multi-restart with structured perturbations.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def simplex_projection(v, target_sum=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def mirror_descent(a, time_budget, seed=42, verbose=False):
    """
    Mirror descent (exponential weights) on the simplex.
    Fast exploration phase: O(n) per iteration.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Adaptive step size schedule
    t = 0
    eta0 = 0.5  # initial step size

    while time.monotonic() - start < time_budget:
        t += 1
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        obj = 2.0 * n * max_c

        # Find maximizer(s)
        max_idx = np.where(np.abs(conv - max_c) < 1e-10 * max_c)[0]
        if len(max_idx) == 0:
            max_idx = np.array([np.argmax(conv)])

        # Compute subgradient
        # sg[i] = (4n / |S|) * sum_{k in S} a[k-i] * 1{0 <= k-i <= n-1}
        sg = np.zeros(n)
        for k in max_idx:
            # For this k: contributes 2*a[k-i] to dc[k]/da[i] for valid i
            # i ranges from max(0, k-(n-1)) to min(n-1, k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_end >= i_start:
                indices = np.arange(i_start, i_end + 1)
                sg[indices] += 2.0 * a[k - indices]

        sg = sg * (2.0 * n / len(max_idx))

        # Step size: eta_t = eta0 / sqrt(t)
        eta = eta0 / np.sqrt(t)

        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - eta * sg
        # Normalize (softmax-like)
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / np.sum(a_new)

        new_obj = compute_obj_norm(a_new)
        if new_obj < best_obj:
            best_obj = new_obj
            best_a = a_new.copy()

        if new_obj < obj:
            a = a_new
        else:
            # Halve step size
            eta0 = max(eta0 * 0.5, 0.01)
            a = a_new  # still take the step for exploration
            # or: a stays the same (more conservative)

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD iter {t}: obj={best_obj:.10f}, eta0={eta0:.4f}, time={elapsed:.1f}s")

    return best_a


def solve_lp_fast(a, warm_start=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses ECOS for speed.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass

    for solver in ['ECOS', 'SCS', 'CLARABEL']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 10):

        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)

    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:

    return a_new / s


def resample_sequence(a, n_new):
    """Resample sequence to new length via interpolation."""
    n_old = len(a)

    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(n_new) / n_new
    return a_new / s

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Hybrid approach: mirror descent for fast exploration, LP alternating for precise
local optimization, coordinate descent with structured candidates for polishing,
multi-restart with Fourier perturbations, and systematic length exploration.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def mirror_descent(a, time_budget, seed=42, verbose=False):
    """
    Mirror descent (exponential weights) on the simplex.
    Fast exploration: O(n log n) per iteration via FFT-based conv.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_obj = compute_obj_norm(a)
    start = time.monotonic()

    t = 0
    eta0 = 1.0
    patience = 0
    max_patience = 500

    while time.monotonic() - start < time_budget:
        t += 1
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        obj = 2.0 * n * max_c

        # Find maximizer set
        tol = 1e-8 * max_c
        max_idx = np.where(np.abs(conv - max_c) < tol)[0]
        if len(max_idx) == 0:

        for k in max_idx:

            patience = 0

    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass

    for solver_name in ['ECOS', 'SCS', 'CLARABEL']:
        try:
            solver = getattr(cp, solver_name)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate_fast(a, time_budget, seed=42, verbose=False):
    """
    LP alternating minimization with adaptive line search and warm starts.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    iter_count = 0
    stall = 0

    while time.monotonic() - start < time_budget - 2:
        t0 = time.monotonic()
        g_lp, M_lp = solve_lp_fast(a, warm_start=prev_g)
        if g_lp is None:
            break
        prev_g = g_lp
        iter_count += 1

        # Line search with multiple candidates
        best_local_obj = best_obj
        best_local_a = a.copy()
        t_values = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]
        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall = 0
            if verbose and iter_count % 15 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP iter {iter_count}: obj={best_obj:.10f}, elapsed={elapsed:.0f}s")
        else:
            stall += 1
            if stall >= 8:
                strength = 0.008 + 0.004 * min(stall, 12)
                pert = rng.standard_normal(n) * strength
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                prev_g = None
                stall = 0

        # Safety: if LP is too slow, break
        if time.monotonic() - t0 > time_budget * 0.3:
            break

    if verbose:
        print(f"    LP done: {iter_count} iters, best={best_obj:.10f}")
    return best_a


def coord_descent_polish(a, time_budget, seed=42, verbose=False):
    """
    Coordinate descent with structured candidate generation.
    Tracks convolution updates incrementally for speed.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    sweep = 0
    no_improve = 0
    mean_val = 1.0 / n

    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            a_j = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)

            # Unaffected max
            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))

            # Generate candidates
            if a_j < 1e-14:
                candidates = np.array([0.0, mean_val * 0.1, mean_val * 0.3, mean_val * 0.7, mean_val, mean_val * 1.5, mean_val * 3])
            else:
                # Include local context
                left = a[j - 1] if j > 0 else a[0]
                right = a[j + 1] if j < n - 1 else a[-1]
                candidates = np.array([
                    0.0,
                    a_j * 0.1, a_j * 0.3, a_j * 0.5, a_j * 0.7, a_j,
                    a_j * 1.3, a_j * 1.7, a_j * 2.5, a_j * 4.0,
                    mean_val * 0.2, mean_val * 0.5, mean_val * 0.8, mean_val, mean_val * 1.5, mean_val * 2.5,
                    left * 0.5, left * 0.8, left * 1.2, left * 1.5,
                    right * 0.5, right * 0.8, right * 1.2, right * 1.5,
                    (left + right) * 0.4, (left + right) * 0.8, (left + right) * 1.2,
                ])
                candidates = np.unique(candidates[(candidates >= 0) & (candidates <= 0.5)])

            best_trial = best_obj
            best_val = a_j

            for val in candidates:
                delta = val - a_j
                if abs(delta) < 1e-15:
                    continue
                S_new = 1.0 + delta
                if S_new <= 1e-10:
                    continue

                # Compute delta_conv for affected region
                a_vals = a[k_start - j: k_end - j + 1]
                delta_conv = 2.0 * delta * a_vals
                if k_start <= 2 * j <= k_end:
                    pos = 2 * j - k_start
                    delta_conv[pos] = 2.0 * a_j * delta + delta * delta

                affected = conv[k_start:k_end + 1] + delta_conv
                max_affected = float(np.max(affected))
                max_total = max(max_unaffected, max_affected)
                obj_trial = 2.0 * n * max_total / (S_new * S_new)

                if obj_trial < best_trial - 1e-13:
                    best_trial = obj_trial
                    best_val = val

            if best_val != a_j:
                delta = best_val - a_j
                a[j] = best_val
                S_new = 1.0 + delta

                # Update conv in-place
                # Need old a values for cross terms (before update)
                a_vals_old = a[k_start - j: k_end - j + 1].copy()
                if k_start <= 2 * j <= k_end:
                    a_vals_old[2 * j - k_start] = a_j

                delta_conv = 2.0 * delta * a_vals_old
                if k_start <= 2 * j <= k_end:
                    pos = 2 * j - k_start
                    delta_conv[pos] = 2.0 * a_j * delta + delta * delta

                conv[k_start:k_end + 1] += delta_conv

                # Renormalize
                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S

                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                break

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")

    return best_a


def fourier_perturb(a, rng, strength=0.015):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 12):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_sequence(a, n_new):
    """Resample sequence to new length via linear interpolation."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Mirror descent + LP alternating + CD polish +
    multi-restart + length exploration.
    Target: minimize 2*n*max(conv(a,a))/(sum(a))^2 below 1.5027.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 10.0

    # Load parent
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0)
        if a.size == 0 or a.sum() < 0.01:
            raise ValueError("Invalid parent")
        a = a / a.sum()
        print(f"Parent: n={a.size}, obj={compute_obj_norm(a):.10f}")
    except Exception:
        a = np.ones(800) / 800.0
        print(f"Default: n=800, obj={compute_obj_norm(a):.10f}")

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    rng = np.random.default_rng(seed)

    # === Phase 1: Mirror Descent (fast exploration) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n--- Phase 1: Mirror Descent ---")
        md_budget = min(remaining * 0.12, 150.0)
        a_md = mirror_descent(best_a, time_budget=md_budget, seed=seed, verbose=True)
        obj_md = compute_obj_norm(a_md)
        if obj_md < best_obj:
            best_obj = obj_md
            best_a = a_md.copy()
            print(f"  MD: {best_obj:.10f} **")
        else:
            print(f"  MD: {obj_md:.10f} (no improvement)")

    # === Phase 2: LP Alternating (main workhorse) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n--- Phase 2: LP Alternating ---")
        lp_budget = min(remaining * 0.45, 550.0)
        a_lp = lp_alternate_fast(best_a, time_budget=lp_budget, seed=seed, verbose=True)
        obj_lp = compute_obj_norm(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            print(f"  LP: {best_obj:.10f} **")
        else:
            print(f"  LP: {obj_lp:.10f}")

    # === Phase 3: Coordinate Descent Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n--- Phase 3: CD Polish ---")
        cd_budget = min(remaining * 0.2, 250.0)
        a_cd = coord_descent_polish(best_a, time_budget=cd_budget, seed=seed, verbose=True)
        obj_cd = compute_obj_norm(a_cd)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()
            print(f"  CD: {best_obj:.10f} **")
        else:
            print(f"  CD: {obj_cd:.10f}")

    # === Phase 4: LP from perturbed states ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print("\n--- Phase 4: LP Restarts (perturbed) ---")
        for restart_i in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            strengths = [0.006, 0.012, 0.02, 0.03]
            a_pert = fourier_perturb(best_a, rng, strength=strengths[restart_i % 4])
            p_budget = min(remaining * 0.35, 120.0)
            a_r = lp_alternate_fast(a_pert, time_budget=p_budget, seed=seed + 1000 * (restart_i + 1), verbose=False)
            # Quick CD after LP
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_r = coord_descent_polish(a_r, time_budget=min(remaining * 0.3, 30.0),
                                           seed=seed + 5000 * (restart_i + 1), verbose=False)
            obj_r = compute_obj_norm(a_r)
            if obj_r < best_obj:
                best_obj = obj_r
                best_a = a_r.copy()
                print(f"  Restart {restart_i}: {best_obj:.10f} **")
            else:
                print(f"  Restart {restart_i}: {obj_r:.10f}")

    # === Phase 5: Length Exploration ===
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    print(f"\n--- Phase 5: Length Exploration (current n={n_best}) ---")
    n_options = []
    for delta in [-100, -50, -25, -10, 10, 25, 50, 100, 150, 200]:
        n_try = n_best + delta
        if 300 <= n_try <= 2000 and n_try != n_best:
            n_options.append(n_try)
    n_options = sorted(set(n_options))[:8]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        a_try = resample_sequence(best_a, n_try)
        alloc = min(remaining * 0.3, 100.0)

        if alloc > 20:
            a_try = lp_alternate_fast(a_try, time_budget=alloc * 0.65, seed=n_try * 37, verbose=False)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_try = coord_descent_polish(a_try, time_budget=min(remaining * 0.35, 35.0),
                                             seed=n_try * 19, verbose=False)

        obj_try = compute_obj_norm(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            print(f"  ** n={n_try}: {best_obj:.10f} **")
        else:
            print(f"  n={n_try}: {obj_try:.10f}")

    # === Phase 6: Final LP + CD polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n--- Phase 6: Final Polish ---")
        alloc_lp = min(remaining * 0.4, 80.0)
        if alloc_lp > 10:
            a_fin = lp_alternate_fast(best_a, time_budget=alloc_lp, seed=seed + 99999, verbose=False)
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_fin = coord_descent_polish(a_fin, time_budget=min(remaining * 0.8, 50.0),
                                             seed=seed + 888888, verbose=False)
            obj_fin = compute_obj_norm(a_fin)
            if obj_fin < best_obj:
                best_obj = obj_fin
                best_a = a_fin.copy()
                print(f"  Final: {best_obj:.10f} **")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        final_obj = val
    else:
        result = [1.0] * 100
        final_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={final_obj:.10f}, time={elapsed:.1f}s ===")
    return result
