import numpy as np
from scipy.fft import rfft, irfft
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def gradient_from_p(a, p):
    """
    Compute 2 * sum_j a[j] * p[i+j] for i=0..n-1.
    p has length 2n-1 (distribution over convolution lags).
    """
    n = len(a)
    b = a[::-1]
    N_fft = 1
    while N_fft < 3 * n - 1:
        N_fft <<= 1
    c = irfft(rfft(b, N_fft) * rfft(p, N_fft), N_fft)
    return 2.0 * c[n - 1:2 * n - 1]


def lse_gradient(a, conv, tau):
    """Compute gradient of LSE(tau, conv) with respect to a."""
    n = len(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    if shifted.max() > 500:
        shifted = shifted - shifted.max()
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(2 * n - 1)
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    return gradient_from_p(a, p)


def topk_gradient(a, conv, k=3):
    """Subgradient using top-k peaks."""
    n = len(a)
    idx = np.argsort(conv)[-k:]
    p = np.zeros(2 * n - 1)
    p[idx] = 1.0 / k
    return gradient_from_p(a, p)


def mirror_descent_step(a, grad, eta):
    """Multiplicative weights update with normalization."""
    exponent = -eta * grad

    a_new = a * np.exp(exponent)
    s = float(np.sum(a_new))
    if s < 1e-300:
        return a.copy()
    return a_new / s

            # Try scaling a[i] by various factors
            best_factor = 1.0
            best_local = best_obj
            orig_ai = a[i]
            for factor in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:
                a[i] = orig_ai * factor
                s_new = float(np.sum(a))
                if s_new < 1e-10:
                    a[i] = orig_ai
                    continue
                a_try = a / s_new
                obj_try = compute_obj(a_try)
                if obj_try < best_local - tolerance:
                    best_local = obj_try
                    best_factor = factor
                    best_a_try = a_try.copy()
            if best_factor != 1.0:
                a = best_a_try.copy()
                best_obj = best_local
                improved = True
            else:
                a[i] = orig_ai
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new

        if not improved:
            if verbose:
                print(f"      CD: converged at round {rnd}, obj={best_obj:.10f}")
            break

    if verbose:
        print(f"      CD: done after {rnd+1} rounds, obj={best_obj:.10f}")
    return a


def peak_surgery(a, n_peaks=5, max_rounds=30, verbose=False):
    """
    Targeted reduction of top autocorrelation peaks.
    For each top peak at lag k*, reduce the contributing a_i values
    and redistribute mass to positions that don't contribute to any top peak.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)

    for rnd in range(max_rounds):
        if time.monotonic() - start > 45.0:
            break

        conv = compute_conv(a)
        top_lags = np.argsort(conv)[-n_peaks:]
        top_vals = conv[top_lags]

        if top_vals[0] < 1.0 / (2 * n - 1) * 1.01:
            break  # Already near optimal

        # Find which a_i contribute most to the top peaks
        contrib = np.zeros(n)
        for k in top_lags:
            for m in range(n):
                km = k - m
                if 0 <= km < n:
                    contrib[m] += a[m] * a[km] * (conv[k] - 1.0/n)

        # Positions to reduce (high contributors)
        top_contrib_idx = np.argsort(contrib)[-min(10, n // 10):]

        # Positions to move mass to (low contribution to any top peak)
        safe_score = np.zeros(n)
        for k in top_lags:
            for m in range(n):
                km = k - m
                if 0 <= km < n:
                    safe_score[m] += a[km]

        # Combine: prefer positions that are low contributors AND safe
        move_source = top_contrib_idx
        move_dest = np.argsort(safe_score + contrib * 0.5)[:min(10, n // 10)]

        for src in move_source:
            if a[src] < 1e-8:
                continue
            # Find best destination
            best_dest = -1
            best_obj_dest = best_obj
            dest_candidates = [d for d in move_dest if d != src and a[d] < a[src]]
            if not dest_candidates:
                dest_candidates = [d for d in range(n) if d != src][:20]

            for dst in dest_candidates[:8]:
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    delta = a[src] * frac
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[dst] += delta
                    if a_try[src] < 0:
                        continue
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj_dest - 1e-12:
                        best_obj_dest = obj_try
                        best_dest = (dst, frac, a_try.copy())

            if best_dest != -1:
                a = best_dest[2]
                best_obj = best_obj_dest
                if verbose and rnd % 10 == 0:
                    print(f"      PS rnd{rnd}: obj={best_obj:.10f}")
                break  # Restart peak finding after one successful move

    if verbose:
        print(f"      PS: done, obj={best_obj:.10f}")
    return a


def adaptive_mirror_descent(a, time_budget=200.0, verbose=False):
    """
    Improved mirror descent with cosine annealing, adaptive top-k,
    and periodic diversity injection.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42 + n)

    it = 0
    stall = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.3:
            break

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Cosine annealing for temperature
        tau = 20.0 + 980.0 * (1.0 + np.cos(np.pi * progress)) / 2.0
        # Adaptive top-k: start with few, increase over time
        k_top = max(1, int(2 + 4 * progress))

        conv = compute_conv(a)

        # Blend of gradients
        alpha = 0.3 + 0.3 * progress  # More top-k weight later
        g_smooth = lse_gradient(a, conv, tau)
        g_topk = topk_gradient(a, conv, k_top)
        g = alpha * g_topk + (1.0 - alpha) * g_smooth

        g_max = float(np.max(np.abs(g)))
        if g_max < 1e-15:
            break

        # Adaptive step size with backtracking
        eta = 1.5 / (g_max + 1e-10)
        eta *= (1.0 + 0.5 * (1.0 - progress))

        a_new = mirror_descent_step(a, g, eta)

        # Diversity injection every 50 iterations
        if it % 50 == 25 and progress < 0.9:
            temp = 0.005 * (1.0 - progress)
            a_new = a_new + temp * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            # Backtracking
            for shrink in [0.5, 0.2, 0.1]:
                a_bt = mirror_descent_step(a, g, eta * shrink)
                obj_bt = compute_obj(a_bt)
                if obj_bt < best_obj - 1e-13:
                    best_obj = obj_bt
                    best_a = a_bt.copy()
                    best_it = it
                    a = a_bt
                    stall = 0
                    break
            else:
                if stall >= 30:
                    # Larger perturbation to escape
                    strength = 0.03 * (1.0 + progress)
                    a_pert = a * np.exp(strength * rng.standard_normal(n))
                    a_pert = np.maximum(a_pert, 0)
                    s_pert = float(np.sum(a_pert))
                    if s_pert > 1e-10:
                        a = a_pert / s_pert
                    else:
                        a = best_a.copy()
                    # Also try time-reversal symmetry
                    if rng.random() < 0.3:
                        a_rev = a[::-1].copy()
                        a_sym = (a + a_rev) / 2.0
                        s_sym = float(np.sum(a_sym))
                        if s_sym > 1e-10:
                            a = a_sym / s_sym
                    stall = 0

        a = a_new
        it += 1

        if verbose and (it % 50 == 0 or it == 1):
            print(f"      MD it{it}: obj={best_obj:.10f}, tau={tau:.1f}, "
                  f"k={k_top}, stall={stall}, t={time.monotonic()-start:.0f}s")

    return best_a


def structured_initialization(n, method="cosine", rng=None):
    """Generate structured initial sequences."""
    if rng is None:
        rng = np.random.default_rng(0)

    if method == "uniform":
        return np.ones(n) / n
    elif method == "gaussian":
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - n/2) / sigma) ** 2)
        return a / a.sum()
    elif method == "cosine":
        x = np.arange(n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x / n))
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    elif method == "raised_cosine_window":
        x = np.arange(n)
        a = 0.5 * (1 + np.cos(np.pi * (2 * x / (n - 1) - 1)))
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    elif method == "bimodal":
        x = np.arange(n)
        sigma = n / 6.0
        a = np.exp(-0.5 * ((x - n*0.25) / sigma) ** 2) + np.exp(-0.5 * ((x - n*0.75) / sigma) ** 2)
        return a / a.sum()
    elif method == "smoothed_uniform":
        a = np.ones(n)
        # Add small smooth perturbation
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        a *= (1 + 0.1 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase))
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a /= s
        return a
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    from scipy.interpolate import interp1d
    f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
    a_new = f(x_new)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-n structured initialization, adaptive mirror descent,
    coordinate descent polishing, and peak surgery.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Initialize from parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                n_parent = parent.size
                print(f"Parent (n={n_parent}): obj={best_score:.10f}")
    except Exception:
        n_parent = 800
        print("No parent, starting fresh")

    n_parent = len(best_seq) if best_seq is not None else 800

    # === Phase 1: Multi-n exploration with structured inits ===
    print(f"\n=== Phase 1: Multi-n Exploration ===")
    n_candidates = []
    # Around parent
    for d in [-300, -200, -100, 0, 100, 200, 300, 400, 500, 700, 1000]:
        nc = n_parent + d
        if 400 <= nc <= 2500:
            n_candidates.append(nc)
    # Some fixed values
    for nf in [500, 600, 800, 1000, 1200, 1400, 1600, 2000]:
        if nf not in n_candidates:
            n_candidates.append(nf)
    n_candidates = sorted(set(n_candidates))

    # Quick MD on each n with different inits
    phase1_results = []
    for nc in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        # Try 2 different inits for each n
        inits = []
        # Resampled parent
        if parent is not None:
            inits.append(("resample", resample_sequence(parent, nc)))
        # Structured
        inits.append(("cosine", structured_initialization(nc, "cosine", rng)))
        inits.append(("bimodal", structured_initialization(nc, "bimodal", rng)))

        for init_name, a_init in inits[:2]:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(remaining * 0.15, 40.0)
            if alloc < 10:
                continue

            a_try = adaptive_mirror_descent(a_init.copy(), time_budget=alloc, verbose=False)
            obj_try = compute_obj(a_try)
            phase1_results.append((obj_try, a_try, nc, init_name))

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  P1 n={nc} ({init_name}): obj={best_score:.10f} **")
            else:
                print(f"  P1 n={nc} ({init_name}): obj={obj_try:.10f}")

    # Sort results and keep top ones
    phase1_results.sort(key=lambda x: x[0])
    print(f"  Phase 1 best: obj={phase1_results[0][0]:.10f} (n={phase1_results[0][2]})")

    # === Phase 2: Intensive optimization on top candidates ===
    print(f"\n=== Phase 2: Intensive Optimization ===")
    remaining = deadline - time.monotonic()

    # Take top 3 from phase 1 (deduplicated by n)
    top_candidates = []
    seen_ns = set()
    for obj_c, a_c, nc, name_c in phase1_results:
        if nc not in seen_ns:
            top_candidates.append((obj_c, a_c, nc))
            seen_ns.add(nc)
        if len(top_candidates) >= 3:
            break
    # Also include current best
    if best_seq is not None and len(best_seq) not in seen_ns:
        top_candidates.append((best_score, best_seq, len(best_seq)))

    for rank, (obj_c, a_c, nc) in enumerate(top_candidates[:3]):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        alloc = min(remaining * 0.3, 200.0)

        print(f"  P2 candidate {rank} (n={nc}, obj={obj_c:.10f}): MD alloc={alloc:.0f}s")
        a_opt = adaptive_mirror_descent(a_c.copy(), time_budget=alloc, verbose=(rank == 0))
        obj_opt = compute_obj(a_opt)

        # Coordinate descent polish
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"  P2 candidate {rank}: CD polish")
            a_opt = coordinate_descent_polish(a_opt, max_rounds=30, verbose=(rank == 0))
            obj_opt = compute_obj(a_opt)

        # Peak surgery
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"  P2 candidate {rank}: Peak surgery")
            a_opt = peak_surgery(a_opt, n_peaks=5, max_rounds=20, verbose=(rank == 0))
            obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  P2 candidate {rank}: ** NEW BEST obj={best_score:.10f} **")
        else:
            print(f"  P2 candidate {rank}: obj={obj_opt:.10f} (no improvement)")

    # === Phase 3: Final intensive polish on best ===
    print(f"\n=== Phase 3: Final Polish (n={len(best_seq)}) ===")

        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final MD: ** obj={best_score:.10f} **")
        else:
            print(f"  Final MD: obj={obj_f:.10f}")

    # Round 2: Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"  Final CD:")
        a_f = coordinate_descent_polish(best_seq.copy(), max_rounds=80, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final CD: ** obj={best_score:.10f} **")
        else:
            print(f"  Final CD: obj={obj_f:.10f}")

    # Round 3: Peak surgery
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"  Final PS:")
        a_f = peak_surgery(best_seq.copy(), n_peaks=8, max_rounds=40, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final PS: ** obj={best_score:.10f} **")
        else:
            print(f"  Final PS: obj={obj_f:.10f}")

    # Round 4: Another CD after PS
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_f = coordinate_descent_polish(best_seq.copy(), max_rounds=50)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:

            best_seq = a_f.copy()

    # === Phase 4: Try slightly different n (fine-tune) ===
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 4: Fine n-tuning ===")
        for dn in [-50, 50, -100, 100, -200, 200]:

            a_try = resample_sequence(best_seq, nc)
            alloc1 = min(remaining * 0.3, 40.0)
            a_try = adaptive_mirror_descent(a_try, time_budget=alloc1, verbose=False)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_try = coordinate_descent_polish(a_try, max_rounds=30)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  n={nc}: ** obj={best_score:.10f} **")

    # === Phase 5: Symmetry exploitation ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 5: Symmetry ===")
        # Time-reversal symmetric version
        a_sym = (best_seq + best_seq[::-1]) / 2.0
        s_sym = float(np.sum(a_sym))
        if s_sym > 1e-10:

    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
