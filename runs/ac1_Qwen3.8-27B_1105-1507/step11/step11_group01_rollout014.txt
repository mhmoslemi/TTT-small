import numpy as np
from scipy.fft import rfft, irfft
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def obj_from_conv(conv, n, S):
    """Compute objective from precomputed convolution and sum."""
    if S < 1e-30:
        return np.inf
    return 2.0 * n * float(np.max(conv)) / (S * S)

def ils_search(a, time_budget=600.0, verbose=True, seed=42):
    """
    Iterated Local Search: CD -> perturb -> CD -> ...
    Adaptive perturbation size based on acceptance rate.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    perturb_scale = 0.05
    accept_count = 0
    total_count = 0
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 30:
            break

        # Perturb
        noise = rng.normal(0, perturb_scale, n)
        a_pert = best_a * np.exp(noise)
        a_pert = np.maximum(a_pert, 0)
        s_pert = float(np.sum(a_pert))
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert

        # CD from perturbed
        cd_time = min(remaining * 0.3, 50.0)
        if cd_time < 5:
            break
        a_new = fast_cd(a_pert, time_budget=cd_time, seed=seed + it * 1000)
        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))

        total_count += 1
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            accept_count += 1
            if verbose and it % 5 == 0:
                print(f"    ILS it{it}: obj={best_obj:.8f}, scale={perturb_scale:.4f}, t={time.monotonic()-start:.1f}s **")
        else:
            if verbose and it % 10 == 0:
                print(f"    ILS it{it}: rejected (obj={obj_new:.8f}), scale={perturb_scale:.4f}, t={time.monotonic()-start:.1f}s")

        # Adapt perturbation scale
        if total_count >= 10:
            accept_rate = accept_count / total_count
            if accept_rate > 0.5:
                perturb_scale *= 1.2
            elif accept_rate < 0.15:
                perturb_scale *= 0.7
            perturb_scale = np.clip(perturb_scale, 0.001, 0.3)
            # Reset counter
            accept_count = 0
            total_count = 0

        it += 1

    return best_a

            eta *= 0.5

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # triangular
        x = np.arange(n) / n
        a = np.abs(1 - 2*x)
        a = a / a.sum()
    elif mode == 3:
        # bell
        x = np.linspace(-1, 1, n)
        a = np.exp(-x*x*5)
        a = a / a.sum()
    elif mode == 4:
        # two peaks
        x = np.linspace(0, 1, n)
        a = np.exp(-((x-0.3)**2)/0.01) + np.exp(-((x-0.7)**2)/0.01)
        a = a / a.sum()
    elif mode == 5:
        # sawtooth
        x = np.linspace(0, 1, n)
        a = x % 0.2
        a = a + 0.1
        a = a / a.sum()
    elif mode == 6:
        # cosine window
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2*np.pi*x))
        a = a + 0.01
        a = a / a.sum()
    elif mode == 7:
        # random smooth
        freqs = rng.uniform(1, 5, size=3)
        phases = rng.uniform(0, 2*np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.sin(2*np.pi*f*x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

"""
Optimizes step function coefficients using Iterated Local Search (ILS) with
coordinate descent, subgradient refinement targeting top lags, adaptive
perturbation, and n-swap warm starts. Avoids closures per requirements.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective 2n*max(conv)/(sum^2) for non-negative a."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-30:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def f_1d_eval(x, v, S, n, c_lin, p_lin, c_quad, has_quad):
    """Evaluate 1D subproblem at point x. Top-level to avoid closure."""
    if len(p_lin) > 0:
        lin_new = c_lin + 2.0 * p_lin * (x - v)
        max_lin = float(np.max(lin_new))
    else:
        max_lin = 0.0
    if has_quad:
        quad_new = c_quad + x * x - v * v
        max_val = max(max_lin, quad_new)
    else:
        max_val = max_lin
    if max_val < 0:
        max_val = 0.0
    denom = (S - v + x) * (S - v + x)
    if denom < 1e-30:
        return 1e30
    return 2.0 * n * max_val / denom


def fast_cd(a, time_budget=60.0, verbose=False, seed=42, max_sweeps=200):
    """
    Coordinate descent with incremental conv updates.
    No closures - uses top-level f_1d_eval.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    prev_obj = best_obj
    stall = 0

    while time.monotonic() - start < time_budget and sweep < max_sweeps:
        sweep += 1
        order = rng.permutation(n)
        S_local = 1.0

        if sweep > 1 and sweep % 3 == 0:
            S_check = float(np.sum(a))
            if abs(S_check - 1.0) > 1e-6:
                a = a / S_check
                S_local = 1.0
                conv = compute_conv(a)

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)

            if k_min > k_max:
                continue

            ks = np.arange(k_min, k_max + 1)
            partners = ks - i
            valid = (partners >= 0) & (partners < n)
            ks_valid = ks[valid]
            partners_valid = partners[valid]

            k_quad = 2 * i
            has_quad = (k_quad >= k_min) and (k_quad <= k_max)

            # Separate quad and linear
            if has_quad:
                quad_mask = (ks_valid == k_quad)
                if np.any(quad_mask):
                    lin_mask = ~quad_mask
                    c_lin = conv[ks_valid[lin_mask]]
                    p_lin = a[partners_valid[lin_mask]]
                    c_quad = float(conv[k_quad])
                else:
                    c_lin = np.array([])
                    p_lin = np.array([])
                    c_quad = 0.0
            else:
                c_lin = conv[ks_valid]
                p_lin = a[partners_valid]
                c_quad = 0.0

            S_other = S_local - v

            # Adaptive initial interval
            scale = max(v, 1.0 / n, 1e-8)
            # Check a few candidates
            candidates = [0.0, v * 0.5, v * 1.5, 1.0 / n, 2.0 / n, 0.5 / n]
            candidates = [max(0.0, c) for c in candidates]

            best_x = v
            best_f = f_1d_eval(v, v, S_local, n, c_lin, p_lin, c_quad, has_quad)
            for x in candidates:
                fx = f_1d_eval(x, v, S_local, n, c_lin, p_lin, c_quad, has_quad)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section
            x_lo = max(0.0, best_x - scale * 1.5)
            x_hi = best_x + scale * 1.5
            if x_hi <= x_lo:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d_eval(c_pt, v, S_local, n, c_lin, p_lin, c_quad, has_quad)
            fd = f_1d_eval(d_pt, v, S_local, n, c_lin, p_lin, c_quad, has_quad)

            for _ in range(35):
                if x_hi - x_lo < 1e-14:
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d_eval(c_pt, v, S_local, n, c_lin, p_lin, c_quad, has_quad)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d_eval(d_pt, v, S_local, n, c_lin, p_lin, c_quad, has_quad)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-15:
                continue

            delta = x_opt - v
            a[i] = x_opt

            # Update conv
            for idx in range(len(ks_valid)):
                k = ks_valid[idx]
                p = partners_valid[idx]
                if k == k_quad and has_quad:
                    conv[k] += x_opt * x_opt - v * v
                else:
                    conv[k] += 2.0 * delta * a[p]

            S_local += delta

        # Track best
        if S_local > 1e-10:
            obj_now = 2.0 * n * float(np.max(conv)) / (S_local * S_local)
            if obj_now < best_obj:
                best_obj = obj_now
                best_a = a.copy()

        if best_obj < prev_obj - 1e-14:
            prev_obj = best_obj
            stall = 0
        else:
            stall += 1
            if stall > 6:
                break

        if verbose and sweep % 10 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

    return best_a


def subgradient_refine(a, time_budget=80.0, verbose=False, seed=42):
    """
    Subgradient descent targeting top lags with periodic CD boost.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    eta = 0.05
    it = 0
    patience = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        # Top lags within 2% of max
        threshold = max_conv * 0.98
        top_lags = np.where(conv >= threshold)[0]

        # Subgradient
        grad = np.zeros(n)
        for k in top_lags:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += a[k - hi:k - lo + 1]
        grad /= max(len(top_lags), 1)

        # Center and apply
        grad_c = grad - np.mean(grad)
        a_new = a - eta * grad_c
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            # Slightly increase eta on success
            eta = min(eta * 1.05, 0.2)
        else:
            patience += 1
            if patience % 30 == 0:
                eta *= 0.7
                if eta < 0.003:
                    noise = rng.normal(0, 0.04, n)
                    a_p = a * np.exp(noise)
                    a_p = np.maximum(a_p, 0)
                    a_p = a_p / a_p.sum()
                    a = a_p
                    eta = 0.05
                    patience = 0

        # Periodic CD
        if it % 40 == 39 and remaining > 20:
            a_cd = fast_cd(a, time_budget=min(12, remaining * 0.15),
                           verbose=False, seed=seed + it)
            obj_cd = 2.0 * n * float(np.max(compute_conv(a_cd)))
            if obj_cd < best_obj:
                best_obj = obj_cd
                best_a = a_cd.copy()
                a = a_cd

        it += 1

    return best_a


def ils_search(a, time_budget=400.0, verbose=True, seed=42):
    """
    Iterated Local Search with adaptive perturbation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    perturb_scale = 0.04
    accept_count = 0
    total_count = 0
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 30:
            break

        # Adaptive perturbation: mix of global and local
        if it % 7 == 6:
            # Occasional large jump
            noise = rng.normal(0, perturb_scale * 3, n)
        else:
            noise = rng.normal(0, perturb_scale, n)

        a_pert = best_a * np.exp(noise)
        a_pert = np.maximum(a_pert, 0)
        s_pert = float(np.sum(a_pert))
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert

        cd_time = min(remaining * 0.25, 45.0)
        if cd_time < 5:
            break
        a_new = fast_cd(a_pert, time_budget=cd_time, seed=seed + it * 1000)
        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))

        total_count += 1
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            accept_count += 1
            no_improve = 0
            if verbose and it % 3 == 0:

        else:
            no_improve += 1
            if verbose and it % 8 == 0:
                print(f"    ILS it{it}: reject obj={obj_new:.8f}, scale={perturb_scale:.4f}, t={time.monotonic()-start:.1f}s")

        # Adapt scale
        if total_count >= 8:
            rate = accept_count / total_count
            if rate > 0.5:
                perturb_scale = min(perturb_scale * 1.15, 0.25)
            elif rate < 0.15:
                perturb_scale = max(perturb_scale * 0.75, 0.002)
            accept_count = 0
            total_count = 0

        # If stuck, try subgradient
        if no_improve > 15:
            sub_time = min(remaining * 0.1, 30.0)
            if sub_time > 5:
                a_sub = subgradient_refine(best_a, time_budget=sub_time,
                                           verbose=False, seed=seed + 50000 + it)
                obj_sub = 2.0 * n * float(np.max(compute_conv(a_sub)))
                if obj_sub < best_obj:
                    best_obj = obj_sub
                    best_a = a_sub.copy()
                    no_improve = 0
                    if verbose:
                        print(f"    ILS it{it}: subgradient rescue obj={best_obj:.8f} **")

        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.arange(n) / n
        a = np.abs(1 - 2 * x) + 0.1
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x * x * 4)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.25) ** 2) / 0.01) + np.exp(-((x - 0.75) ** 2) / 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x * 3)) + 0.1
        a = a / a.sum()
    elif mode == 6:
        freqs = rng.uniform(1, 6, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 7:
        # Asymmetric power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase ILS-based search: CD init -> subgradient -> ILS -> n-sweep ->
    diverse restarts -> final polish. All helper calls use top-level functions.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        n_def = 800
        best_seq = np.ones(n_def) / n_def
        best_score = compute_obj(best_seq)
        print(f"Default (n={n_def}): obj={best_score:.8f}")

    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)

    n_best = len(best_seq)
    print(f"Starting optimization from n={n_best}, score={best_score:.8f}")

    # === Phase 1: Initial CD ===
    print(f"\n=== Phase 1: Initial CD ===")
    remaining = deadline - time.monotonic()
    p1 = min(remaining * 0.15, 120.0)
    if p1 > 20:
        a1 = fast_cd(best_seq, time_budget=p1, verbose=True, seed=seed)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj1:.8f} (no improvement)")

    # === Phase 2: Subgradient refinement ===
    print(f"\n=== Phase 2: Subgradient Refinement ===")
    remaining = deadline - time.monotonic()
    p2 = min(remaining * 0.1, 90.0)
    if p2 > 15:
        a2 = subgradient_refine(best_seq, time_budget=p2, verbose=False, seed=seed + 100)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Subgrad: obj={best_score:.8f} **")
        # Follow-up CD
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a2b = fast_cd(best_seq, time_budget=min(25, remaining * 0.1),
                          verbose=False, seed=seed + 200)
            obj2b = compute_obj(a2b)
            if obj2b < best_score:
                best_score = obj2b
                best_seq = a2b.copy()
                print(f"  CD after subgrad: obj={best_score:.8f} **")

    # === Phase 3: Main ILS ===
    print(f"\n=== Phase 3: Iterated Local Search (main) ===")
    remaining = deadline - time.monotonic()
    p3 = min(remaining * 0.35, 450.0)
    if p3 > 60:
        a3 = ils_search(best_seq, time_budget=p3, verbose=True, seed=seed + 300)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  ILS: obj={best_score:.8f} **")
        else:
            print(f"  ILS: obj={obj3:.8f}")

    # === Phase 4: n-sweep ===
    print(f"\n=== Phase 4: n-sweep ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    n_candidates = [600, 700, 750, 800, 850, 900, 950, 1000, 1100, 1200]
    # Sort by distance to current n, try closest first
    n_candidates.sort(key=lambda x: abs(x - n_cur))
    n_filtered = []
    for nc in n_candidates:
        if 400 <= nc <= 1500:
            if not n_filtered or all(abs(nc - nf) > 40 for nf in n_filtered):
                n_filtered.append(nc)
    n_filtered = n_filtered[:6]
    print(f"  Trying n values: {n_filtered}")

    for n_try in n_filtered:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        if n_try == n_cur:
            continue

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        alloc = min(remaining * 0.12, 45.0)
        if alloc > 10:
            a_try = fast_cd(a_try, time_budget=alloc, verbose=False, seed=n_try + 5000)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_cur = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = 1 + r
        a_r = diverse_init(n_cur, rng, mode=mode)
        blend = 0.15 + 0.08 * r
        a_r = (1 - blend) * best_seq + blend * a_r
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r < 1e-10:
            continue
        a_r = a_r / s_r

        alloc = min(remaining * 0.18, 70.0)
        if alloc > 20:
            # CD first, then short ILS
            a_r = fast_cd(a_r, time_budget=alloc * 0.5, verbose=False, seed=8000 + r * 100)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
            else:
                # Try ILS from here
                remaining2 = deadline - time.monotonic()
                if remaining2 > 30:
                    a_ils = ils_search(a_r, time_budget=min(25, remaining2 * 0.3),
                                       verbose=False, seed=9000 + r * 100)
                    obj_ils = compute_obj(a_ils)
                    if obj_ils < best_score:
                        best_score = obj_ils
                        best_seq = a_ils.copy()
                        print(f"  Restart {r} ILS: obj={best_score:.8f} **")
                    else:
                        print(f"  Restart {r}: obj={obj_r:.8f} (ILS: {obj_ils:.8f})")
                else:
                    print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 6: Final ILS polish ===
    print(f"\n=== Phase 6: Final ILS Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        p6 = min(remaining * 0.5, 180.0)
        a6 = ils_search(best_seq, time_budget=p6, verbose=True, seed=seed + 9000)
        obj6 = compute_obj(a6)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Final ILS: obj={best_score:.8f} **")

    # === Phase 7: Ultimate CD ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        p7 = min(remaining * 0.85, 100.0)
        a7 = fast_cd(best_seq, time_budget=p7, verbose=True, seed=seed + 9999)
        obj7 = compute_obj(a7)
        if obj7 < best_score:
            best_score = obj7
            best_seq = a7.copy()
            print(f"  Ultimate CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = 2.0

    best_seq = np.maximum(best_seq, 0.0)
    result = [float(min(1000.0, max(0.0, x))) for x in best_seq]

    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
        print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f} (eval={val:.8f}), "
              f"time={time.monotonic()-start_time:.1f}s ===")
    else:
        print(f"\n=== WARNING: evaluate returned inf, using fallback ===")
        result = [1.0] * 100
        best_score = 2.0

    return result
