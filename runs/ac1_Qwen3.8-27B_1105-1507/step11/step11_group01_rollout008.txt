def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def subgradient(a, conv):
    """Compute subgradient of max(conv) w.r.t. a.
    Returns gradient vector of length n."""
    n = len(a)

    # Find all k where conv[k] >= max_val * (1 - 1e-10)
    threshold = max_val * (1 - 1e-9)
    active = np.where(conv >= threshold)[0]
    
    grad = np.zeros(n)
    for k in active:
        # ∂g_k/∂a_m = 2*a[k-m] for valid m
        # m ranges from max(0, k-n+1) to min(k, n-1)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            indices = np.arange(lo, hi + 1)
            grad[indices] += 2.0 * a[k - indices]
    
    grad /= len(active)
    return grad

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
