        p = exp_vals / sum_exp
    return gradient_from_p(a, p)


def topk_gradient(a, connv, k=5):
    """Subgradient using top-k peaks with proportional weights."""
    n = len(a)
    # Get top k
    top_idx = np.argpartition(connv, -k)[-k:]
    top_vals = connv[top_idx]
    # Weight proportional to value (emphasize the largest)
    weights = top_vals ** 2  # Square to emphasize larger peaks
    weights = weights / weights.sum()
    p = np.zeros(2 * n - 1)
    p[top_idx] = weights
    return gradient_from_p(a, p)


def mirror_step(a, grad, eta):
    """Multiplicative weights: a_i <- a_i * exp(-eta * grad_i) / sum(...)"""
    exponent = -eta * grad
    exponent = np.clip(exponent, -50, 50)
    a_new = a * np.exp(exponent)
    s = float(np.sum(a_new))
    if s < 1e-300:
        return a.copy()
    return a_new / s

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""

Uses peak-equalizing mirror descent with annealed log-sum-exp, multiple structured

import numpy as np
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def gradient_from_p(a, p):
    """
    Compute gradient of sum_k p[k] * conv(a,a)[k] with respect to a.
    grad[i] = 2 * sum_k p[k] * a[k-i] (where valid)
    """
    n = len(a)
    b = a[::-1]
    N_fft = 1
    while N_fft < 3 * n - 1:
        N_fft <<= 1
    c = irfft(rfft(b, N_fft) * rfft(p, N_fft), N_fft)
    return 2.0 * c[n - 1:2 * n - 1]

    # 4. Gaussian (center)
    a = np.exp(-6.0 * (x - 0.5) ** 2)
    candidates.append(a / a.sum())

    # 5. Sinusoidal modulation
    a = 1.0 + 0.4 * np.sin(2 * np.pi * x * 3 + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0.01)
    candidates.append(a / a.sum())

    # 6. Double sinusoidal
    a = 1.0 + 0.3 * np.sin(2 * np.pi * x * 2) + 0.2 * np.sin(2 * np.pi * x * 5 + 0.7)
    a = np.maximum(a, 0.01)
    candidates.append(a / a.sum())

    # 7. Sinc-like
    W = max(2, n // 6)
    idx = np.arange(n) - n // 2
    a = np.sinc(idx / W)
    a = np.maximum(a, 0)
    if a.sum() > 0.01:
        candidates.append(a / a.sum())

    # 8. Random with smoothing
    a_raw = rng.uniform(0.3, 1.7, n)
    a = uniform_filter1d(a_raw, size=max(5, n // 40))
    a = np.maximum(a, 0.01)
    candidates.append(a / a.sum())

    # 9. Two-bump
    a = (np.exp(-30.0 * (x - 0.3) ** 2) + np.exp(-30.0 * (x - 0.7) ** 2))
    candidates.append(a / a.sum())

    # 10. Power law
    a = (x + 0.05) ** 1.5
    candidates.append(a / a.sum())

    # 11. Uniform (baseline)
    candidates.append(np.ones(n) / n)

    # 12. i.i.d. exponential
    a = rng.exponential(1.0, n)
    candidates.append(a / a.sum())

    # 13. Low-frequency random
    a = rng.uniform(0.5, 1.5, n)
    a = uniform_filter1d(a, size=max(3, n // 20))
    a = np.maximum(a, 0.01)
    candidates.append(a / a.sum())

    # 14. Anti-symmetric bias (high at start, low at end)
    a = np.linspace(2.0, 0.1, n)
    candidates.append(a / a.sum())

    # 15. Piecewise: high-low-high
    a = np.concatenate([np.ones(n//3)*2, np.ones(n//3)*0.3, np.ones(n - 2*(n//3))*2])
    a = uniform_filter1d(a, size=max(3, n//30))
    candidates.append(a / a.sum())

    best = None
    best_obj = np.inf
    for c in candidates:
        obj = compute_obj(c)
        if obj < best_obj:
            best_obj = obj
            best = c.copy()

    return best, best_obj


def optimize_hybrid(a, time_budget, tau_start=30.0, tau_end=10000.0,
                    topk=3, verbose=False, seed=0):
    """
    Hybrid mirror descent: blends exact subgradient, top-k, and LSE.
    Uses adaptive step size with backtracking line search.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    best_it = 0
    stall = 0
    peak_cycle = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.05:
            break

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Temperature: slower ramp, plateau at high value
        if progress < 0.3:
            tau = tau_start * (1 + 10 * progress / 0.3)
        elif progress < 0.8:
            tau = tau_start * 11 + (tau_end - tau_start * 11) * (progress - 0.3) / 0.5
        else:
            tau = tau_end

        conv = compute_conv(a)

        # Hybrid gradient selection based on iteration phase
        if it < 50:
            # Warm-up: LSE for smooth exploration
            g = lse_gradient(a, conv, max(tau, 30.0))
        elif it % 20 == 0:
            # Every 20th iteration: exact single-peak subgradient
            g = single_peak_gradient(a, conv)
        elif it % 20 == 5:
            # Top-k with high power (emphasize worst peak)
            g = topk_gradient(a, conv, k=topk, power=3.0)
        else:
            # Blend: 70% LSE + 30% top-k
            g_lse = lse_gradient(a, conv, tau)
            g_top = topk_gradient(a, conv, k=topk, power=2.0)
            alpha = 0.2 + 0.3 * progress
            g = (1 - alpha) * g_lse + alpha * g_top

        g_max = float(np.max(np.abs(g)))
        if g_max < 1e-15:
            break

        # Adaptive step size: decrease with progress
        eta_base = 2.0 / (g_max + 1e-10)
        # Schedule: start moderate, end very small
        eta_scale = 1.0 - 0.85 * progress ** 1.5
        eta = eta_base * eta_scale
        eta = max(eta, 0.005 * eta_base)

        # Mirror descent step
        a_new = mirror_step(a, g, eta)

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            # Backtracking: try smaller step
            improved = False
            for factor in [0.5, 0.25, 0.1]:
                a_try = mirror_step(a, g, eta * factor)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    best_it = it
                    a = a_try
                    stall = 0
                    improved = True
                    break
            if not improved:
                stall += 1
                if stall >= 30:
                    # Escape: multiplicative perturbation
                    strength = 0.005 + 0.015 * progress
                    a = a * np.exp(strength * rng.standard_normal(n))
                    a = np.maximum(a, 0)
                    s_new = float(np.sum(a))
                    if s_new > 1e-10:
                        a = a / s_new
                    else:
                        a = best_a.copy()
                    stall = 0
                    # Also try a smoothing step
                    if progress > 0.5:
                        k_smooth = max(3, n // 100)
                        a_smooth = uniform_filter1d(a, size=k_smooth)
                        a = 0.9 * a + 0.1 * a_smooth
                        a = np.maximum(a, 0)
                        s_new = float(np.sum(a))
                        if s_new > 1e-10:
                            a = a / s_new

        a = a_new
        it += 1
        peak_cycle += 1

        if verbose and (it % 500 == 0 or it == 1):
            print(f"    Hybrid it{it}: obj={best_obj:.8f}, tau={tau:.0f}, "
                  f"eta={eta:.5f}, stall={stall}, time={time.monotonic()-start:.0f}s")

    return best_a


def peak_equalization(a, n_peaks=5, n_steps=300, verbose=False):
    """
    Direct peak equalization: iteratively reduce the worst peaks with
    adaptive weights that emphasize the most severe peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    for step in range(n_steps):
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        if max_val < 1e-15:
            break

        # Weight peaks by their excess over the mean
        mean_val = float(np.mean(conv))
        excess = np.maximum(conv - mean_val, 0)
        
        # Get top peaks weighted by squared excess
        n_actual = min(n_peaks, 2 * n - 1)
        top_idx = np.argpartition(conv, -n_actual)[-n_actual:]
        weights = conv[top_idx] ** 2.5  # High power to emphasize worst
        wsum = float(np.sum(weights))
        if wsum < 1e-300:
            break

        p = np.zeros(2 * n - 1)
        p[top_idx] = weights / wsum

        g = gradient_from_p(a, p)

        # Very small step for peak equalization
        g_norm = float(np.max(np.abs(g)))
        if g_norm < 1e-15:
            break
        eta = 0.3 / (g_norm + 1e-10)

        a_new = mirror_step(a, g, eta)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

        if verbose and (step % 50 == 0):
            print(f"    PeakEq step{step}: obj={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fine_polish(a, time_budget, verbose=False):
    """
    Very fine polish: tiny steps with very high temperature LSE.
    Focuses on the single worst peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    it = 0
    while time.monotonic() - start < time_budget:
        if time.monotonic() - start < 0.05:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        p = np.zeros(2 * n - 1)
        p[k_star] = 1.0
        g = gradient_from_p(a, p)

        g_max = float(np.max(np.abs(g)))
        if g_max < 1e-15:
            break

        # Very small step
        eta = 0.1 / (g_max + 1e-10)
        a_new = mirror_step(a, g, eta)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and (it % 2000 == 0):
            print(f"    Polish it{it}: obj={best_obj:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-initialization + hybrid mirror descent + peak equalization
    + fine polish. Uses diverse hyperparameter settings across runs.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    best_seq = None
    best_score = np.inf

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                n_target = len(best_seq)
                print(f"Parent (n={n_target}): obj={best_score:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    # Phase 1: Structured initializations
    print(f"\n=== Phase 1: Structured Initializations (n={n_target}) ===")
    init_seq, init_score = structured_init(n_target, rng)
    print(f"  Best init: obj={init_score:.8f}")

    if init_score < best_score:
        best_score = init_score
        best_seq = init_seq.copy()

    # Phase 2: Main optimization (run 1 - moderate temperature)
    print(f"\n=== Phase 2: Main Optimization Run 1 ===")
    remaining = deadline - time.monotonic()
    opt_budget = min(remaining * 0.3, 400.0)
    if opt_budget > 30:
        a_opt = optimize_hybrid(best_seq, time_budget=opt_budget,
                                tau_start=30.0, tau_end=5000.0,
                                topk=3, verbose=True, seed=100)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Run 1: obj={best_score:.8f} **")
        else:
            print(f"  Run 1: obj={obj_opt:.8f} (no improvement)")

    # Phase 3: Peak equalization
    print(f"\n=== Phase 3: Peak Equalization ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_pe = peak_equalization(best_seq, n_peaks=5, n_steps=300, verbose=True)
        obj_pe = compute_obj(a_pe)
        if obj_pe < best_score:
            best_score = obj_pe
            best_seq = a_pe.copy()
            print(f"  PeakEq: obj={best_score:.8f} **")
        else:
            print(f"  PeakEq: obj={obj_pe:.8f} (no improvement)")

    # Phase 4: Main optimization (run 2 - high temperature, different seed)
    print(f"\n=== Phase 4: Main Optimization Run 2 ===")
    remaining = deadline - time.monotonic()
    opt_budget2 = min(remaining * 0.25, 350.0)
    if opt_budget2 > 20:
        a_opt2 = optimize_hybrid(best_seq, time_budget=opt_budget2,
                                 tau_start=100.0, tau_end=20000.0,
                                 topk=2, verbose=True, seed=200)
        obj_opt2 = compute_obj(a_opt2)
        if obj_opt2 < best_score:
            best_score = obj_opt2
            best_seq = a_opt2.copy()
            print(f"  Run 2: obj={best_score:.8f} **")
        else:
            print(f"  Run 2: obj={obj_opt2:.8f} (no improvement)")

    # Phase 5: Another peak equalization with different params
    print(f"\n=== Phase 5: Peak Equalization 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_pe2 = peak_equalization(best_seq, n_peaks=3, n_steps=200)
        obj_pe2 = compute_obj(a_pe2)
        if obj_pe2 < best_score:
            best_score = obj_pe2
            best_seq = a_pe2.copy()
            print(f"  PeakEq2: obj={best_score:.8f} **")
        else:
            print(f"  PeakEq2: obj={obj_pe2:.8f} (no improvement)")

    # Phase 6: Multi-n exploration
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        n_best - 100, n_best - 50, n_best + 50, n_best + 100,
        n_best + 200, n_best - 200,
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000][:4]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.2, 50.0)
        if alloc > 15:
            a_try = optimize_hybrid(a_try, time_budget=alloc,
                                    tau_start=50.0, tau_end=3000.0,
                                    topk=3, verbose=False, seed=n_try)
            # Quick polish
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                a_try = fine_polish(a_try, time_budget=min(remaining2 * 0.3, 20.0))

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()

    for name, transform in restart_configs:

        a_r = transform(best_seq.copy())
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r
        else:
            a_r = best_seq.copy()

        alloc = min(remaining * 0.25, 60.0)
        if alloc > 10:
            a_r = optimize_hybrid(a_r, time_budget=alloc,
                                  tau_start=100.0, tau_end=5000.0,
                                  topk=3, verbose=False, seed=hash(name) % 10000)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart ({name}): obj={best_score:.8f} **")
        else:
            print(f"  Restart ({name}): obj={obj_r:.8f}")

    # Phase 8: Final intensive fine polish
    print(f"\n=== Phase 8: Final Fine Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.6, 200.0)
        a_final = fine_polish(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  FinePolish: obj={best_score:.8f} **")
        else:
            print(f"  FinePolish: obj={obj_final:.8f}")

    # Phase 9: One more peak equalization
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_final2 = peak_equalization(best_seq, n_peaks=3, n_steps=100)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  FinalPeakEq: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
