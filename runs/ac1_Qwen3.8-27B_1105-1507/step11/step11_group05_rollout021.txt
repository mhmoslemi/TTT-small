"""
Minimax LP-based optimization for step function auto-correlation peak minimization.
Key innovations:
1. Successive linearization of the quadratic minimax problem -> LP subproblems
2. Full constraint LP (all 2n-1 peaks) for accurate direction finding
3. Proper 1D coordinate descent with adaptive grid
4. Diverse non-symmetric initializations (monotone, exponential, etc.)
5. Adaptive n exploration with LP re-solve at each n
6. Multi-restart with structural diversity

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear auto-convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def normalize(a):
    """Normalize to sum=1."""
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    ss = proj.sum()
    if ss < 1e-15:
        return np.ones(n) / n
    return proj / ss

        if j_start <= j_end:

        for t in np.linspace(lo, hi, n_fine):
            a_try = (1 - t) * a0 + t * a1
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t

    return best_t, best_obj


def coord_descent(a, time_budget, max_sweeps=5, verbose=False):
    """Coordinate descent with adaptive 1D search per coordinate."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start >= time_budget:
            break
        sweep_improved = False

        # Randomize coordinate order for diversity
        order = np.random.permutation(n)

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            ai_old = a[i]
            s_excl = 1.0 - ai_old  # sum of other elements (since sum=1)

            # Build candidate values adaptively
            if ai_old < 1e-12:
                candidates = np.array([0.0, 0.001, 0.005, 0.01, 0.02])
            else:
                scale = ai_old
                candidates = np.array([0.0, scale * 0.3, scale * 0.6, scale * 0.8,
                                       scale * 1.0, scale * 1.2, scale * 1.5,
                                       scale * 2.0, scale * 3.0])

            best_val = ai_old
            best_local_obj = best_obj
            a_work = a.copy()

            for cv in candidates:
                a_work[i] = cv
                s_new = float(np.sum(a_work))
                if s_new < 1e-10:
                    continue
                a_norm = a_work / s_new
                obj = compute_obj(a_norm)
                if obj < best_local_obj:
                    best_local_obj = obj
                    best_val = cv

            if best_val != ai_old:
                a[i] = best_val
                a = normalize(a)
                sweep_improved = True

        obj_now = compute_obj(a)
        if obj_now < best_obj:
            best_obj = obj_now
            best_a = a.copy()

        if verbose:
            print(f"        CD sweep {sweep}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

        if not sweep_improved:
            break

    return best_a


def lse_gradient_descent(a, time_budget, T_start=10.0, T_end=80.0, verbose=False):
    """Log-sum-exp smoothed gradient descent on the simplex."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.5 / n

    def lse_smoothed(a_cur, T):
        conv = compute_conv(a_cur)
        cmax = np.max(conv)
        shifted = T * (conv - cmax)
        exp_vals = np.exp(shifted)
        lse = cmax + np.log(np.sum(exp_vals)) / T
        return 2.0 * n * lse, exp_vals / np.sum(exp_vals)

    def lse_grad(a_cur, T):
        conv = compute_conv(a_cur)
        cmax = np.max(conv)
        exp_vals = np.exp(T * (conv - cmax))
        w = exp_vals / np.sum(exp_vals)
        grad = np.zeros(n)
        for i in range(n):
            # (w * a_cur) contribution from term 1
            lo1 = max(0, i)
            hi1 = min(2 * n - 2, i + n - 1)
            if lo1 <= hi1:
                grad[i] += np.dot(w[lo1:hi1 + 1], a_cur[lo1 - i:hi1 - i + 1])
            lo2 = max(0, i - n + 1)
            hi2 = min(2 * n - 2, i)
            if lo2 <= hi2:
                grad[i] += np.dot(w[lo2:hi2 + 1], a_cur[i - hi2:i - lo2 + 1][::-1])
        return 2.0 * n * grad

    it = 0
    patience = 200
    best_it = 0

    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** frac

        grad = lse_grad(a, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        a_old = a.copy()
        obj_smooth_old = lse_smoothed(a_old, T)[0]
        t_lr = lr
        improved = False

        for _ in range(12):
            a_try = project_simplex(a_old - t_lr * grad)
            obj_smooth_try = lse_smoothed(a_try, T)[0]
            if obj_smooth_try < obj_smooth_old - 1e-13:
                a = a_try
                improved = True
                break
            t_lr *= 0.5

        if not improved:
            # Random perturbation
            direction = a + np.random.randn(n) * (0.005 + 0.002 * (it // 100))
            a_try = project_simplex(direction)
            obj_smooth_try = lse_smoothed(a_try, T)[0]
            if obj_smooth_try < obj_smooth_old:
                a = a_try
                improved = True
            else:
                break

        obj_true = compute_obj(a)
        if obj_true < best_obj:
            best_obj = obj_true
            best_a = a.copy()
            best_it = it
            lr = t_lr * 1.05

        it += 1
        if it - best_it > patience:
            a = best_a.copy()
            lr *= 0.5
            patience = max(50, patience // 2)
            best_it = it

        if verbose and it % 200 == 0:
            print(f"        LSE it{it}: obj={best_obj:.8f}, T={T:.1f}, t={time.monotonic()-start:.1f}s")

    return best_a


def lp_iteration_loop(a, time_budget, top_m_start=30, top_m_end=60, max_iters=100, verbose=False):
    """Main LP iteration loop with successive linearization."""
    n = len(a)
    best_a = normalize(a.copy())
    best_obj = compute_obj(best_a)
    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        frac = min(1.0, it / max(1, max_iters - 1))
        top_m = int(round(top_m_start + frac * (top_m_end - top_m_start)))
        top_m = max(10, min(top_m, 2 * n - 1))

        # Solve linearized LP
        a_lp, M_val = solve_linearized_lp(best_a, top_m=top_m)

        if a_lp is None:
            # Fallback to max-sum LP
            c = compute_conv(best_a)
            M_bound = float(np.max(c))
            a_lp = solve_lp_max_sum(best_a, M_bound, top_m=min(top_m, 40))

        if a_lp is None:
            break

        # Blend line search
        t_opt, obj_opt = blend_line_search(best_a, a_lp)

        if t_opt > 0.01 and obj_opt < best_obj - 1e-12:
            a_new = (1 - t_opt) * best_a + t_opt * a_lp
            a_new = normalize(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 5 == 0:
            print(f"        LP it{it}: obj={best_obj:.8f}, top_m={top_m}, "
                  f"t_opt={t_opt:.3f}, M_lp={M_val:.6f}, elapsed={time.monotonic()-start:.1f}s")

        if stall >= 5:
            # Perturb to escape
            noise_scale = 0.002 * stall
            noise = np.random.randn(n) * noise_scale
            a_pert = best_a + noise
            a_pert = normalize(a_pert)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_obj + 0.01:
                best_a = a_pert
                best_obj = obj_pert
            stall = 0

    return best_a


def diverse_init(n, rng, mode):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n, dtype=float)

    if mode == 0:
        a = np.exp(-3.0 * x)  # Monotone exponential decay
    elif mode == 1:
        a = np.exp(-5.0 * (1 - x))  # Monotone exponential growth
    elif mode == 2:
        a = (x ** 0.3) * np.exp(-2.0 * x)  # Power times exponential
    elif mode == 3:
        a = 1.0 / (x + 0.1) ** 1.5  # Power law decay
    elif mode == 4:
        a = np.exp(-2.0 * x) + 0.3 * np.exp(-2.0 * (1 - x))  # Asymmetric U
    elif mode == 5:
        a = np.exp(-1.5 * x) + 0.5 * np.exp(-1.5 * (1 - x))  # Mild asymmetric U
    elif mode == 6:
        a = np.ones(n)  # Uniform
    elif mode == 7:
        a = (2 * x - 1) ** 2 + 0.1  # Symmetric U
    elif mode == 8:
        a = np.exp(-((idx - n * 0.2) ** 2) / (2 * (n * 0.15) ** 2)) + \
            0.3 * np.exp(-((idx - n * 0.8) ** 2) / (2 * (n * 0.1) ** 2)) + 0.01
    elif mode == 9:
        a = np.exp(-8.0 * x) + 0.01  # Very sharp decay
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def try_n_value(a_best, n_try, time_budget):
    """Resample best sequence to new n and do quick optimization."""
    n_old = len(a_best)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_try)
    a_try = np.interp(x_new, x_old, a_best)
    a_try = normalize(a_try)

    # Quick LSE GD
    t1 = min(20.0, time_budget * 0.3)
    if t1 > 5:
        a_try = lse_gradient_descent(a_try, time_budget=t1, T_start=10.0, T_end=50.0)

    # Quick LP
    t2 = min(30.0, time_budget * 0.5)
    if t2 > 5:
        a_try = lp_iteration_loop(a_try, time_budget=t2, top_m_start=20, top_m_end=40, max_iters=15)

    obj_try = compute_obj(a_try)
    return a_try, obj_try


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-method minimax optimization:
    1. Diverse initialization (especially non-symmetric)
    2. Successive linearization LP (main driver)
    3. LSE smoothed gradient descent
    4. Coordinate descent polish
    5. Adaptive n exploration
    6. Multi-restart with perturbation
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    rng = np.random.default_rng(seed)
    best_seq = None
    best_score = np.inf

    # === Load parent if available ===
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.all(np.isfinite(parent_arr)):
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={n_current}): obj={obj:.8f}")
    except Exception:
        print("No parent available, using fresh initialization")

    # === Phase 1: Diverse initialization ===
    print("\n=== Phase 1: Diverse Initialization ===")
    init_results = []
    for mode in range(10):
        a_init = diverse_init(n_current, rng, mode)
        obj_init = compute_obj(a_init)
        init_results.append((obj_init, mode, a_init.copy()))
        print(f"  Init mode {mode}: obj={obj_init:.8f}")

    init_results.sort(key=lambda x: x[0])
    if init_results[0][0] < best_score:
        best_score = init_results[0][0]
        best_seq = init_results[0][2].copy()
        print(f"  ** Best init: mode {init_results[0][1]}, obj={best_score:.8f}")

    # Also try a few more n values for initialization
    for n_try in [500, 1000, 1200]:
        for mode in [0, 3, 4]:
            a_init = diverse_init(n_try, rng, mode)
            obj_init = compute_obj(a_init)
            if obj_init < best_score:
                best_score = obj_init
                best_seq = a_init.copy()
                n_current = n_try
                print(f"  ** Init n={n_try} mode {mode}: obj={best_score:.8f}")

    print(f"  Best after init: obj={best_score:.8f}, n={len(best_seq)}")

    # === Phase 2: LP iteration on best sequence ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 2: LP Iteration (budget={remaining:.0f}s) ===")
        alloc = min(400.0, remaining * 0.35)
        a_lp = lp_iteration_loop(best_seq, time_budget=alloc,
                                  top_m_start=30, top_m_end=80, max_iters=80, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** LP: obj={best_score:.8f}")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 3: LSE gradient descent ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 3: LSE Gradient Descent ===")
        alloc = min(60.0, remaining * 0.08)
        a_lse = lse_gradient_descent(best_seq, time_budget=alloc, T_start=10.0, T_end=60.0)
        obj_lse = compute_obj(a_lse)
        if obj_lse < best_score:
            best_score = obj_lse
            best_seq = a_lse.copy()
            print(f"  ** LSE: obj={best_score:.8f}")
        else:
            print(f"  LSE: obj={obj_lse:.8f}")

    # === Phase 4: Second LP round ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: LP Round 2 ===")
        alloc = min(300.0, remaining * 0.25)
        a_lp2 = lp_iteration_loop(best_seq, time_budget=alloc,
                                   top_m_start=50, top_m_end=100, max_iters=60, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  ** LP2: obj={best_score:.8f}")
        else:
            print(f"  LP2: obj={obj_lp2:.8f}")

    # === Phase 5: Coordinate descent ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 5: Coordinate Descent ===")
        alloc = min(80.0, remaining * 0.1)
        a_cd = coord_descent(best_seq, time_budget=alloc, max_sweeps=3, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  ** CD: obj={best_score:.8f}")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 6: N exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 6: N Exploration ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            n_best - 100, n_best - 50,
            n_best + 50, n_best + 100, n_best + 200,
            int(n_best * 0.8), int(n_best * 1.2),
        ]))
        n_options = [n for n in n_options if 300 <= n <= 2500 and abs(n - n_best) >= 40]
        n_options = n_options[:5]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            t_alloc = min(50.0, remaining * 0.3)
            a_try, obj_try = try_n_value(best_seq, n_try, t_alloc)
            if a_try is not None and obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Intensive LP on best n ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 7: Intensive LP Polish ===")
        alloc = min(200.0, remaining * 0.3)
        a_polish = lp_iteration_loop(best_seq, time_budget=alloc,
                                      top_m_start=60, top_m_end=120, max_iters=50, verbose=False)
        obj_polish = compute_obj(a_polish)
        if obj_polish < best_score:
            best_score = obj_polish
            best_seq = a_polish.copy()
            print(f"  ** Polish LP: obj={best_score:.8f}")
        else:
            print(f"  Polish LP: obj={obj_polish:.8f}")

    # === Phase 8: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 8: Diverse Restarts ===")
        n_cur = len(best_seq)
        for r in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            mode = (r * 3) % 10
            a_r = diverse_init(n_cur, rng, mode=mode)
            # Blend with best
            for blend in [0.3, 0.5, 0.7]:
                a_mix = (1 - blend) * a_r + blend * best_seq
                a_mix = normalize(a_mix)
                obj_mix = compute_obj(a_mix)
                if obj_mix < best_score:
                    best_score = obj_mix
                    best_seq = a_mix.copy()
                    print(f"  Restart {r} (blend={blend}): obj={best_score:.8f} **")
                    break

        # LP on perturbed best
        remaining = deadline - time.monotonic()
        if remaining > 40:
            noise = rng.normal(0, 0.003, n_cur)
            a_pert = normalize(best_seq + noise)
            alloc = min(60.0, remaining * 0.4)
            a_pert = lp_iteration_loop(a_pert, time_budget=alloc,
                                        top_m_start=40, top_m_end=80, max_iters=20, verbose=False)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  Perturbed LP: obj={best_score:.8f} **")
            else:
                print(f"  Perturbed LP: obj={obj_pert:.8f}")

    # === Phase 9: Final coordinate descent ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 9: Final CD ===")
        alloc = min(remaining - 5, 40.0)
        a_final = coord_descent(best_seq, time_budget=alloc, max_sweeps=2)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Final CD: obj={best_score:.8f}")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    # === Phase 10: Final LSE polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining - 5, 30.0)
        a_last = lse_gradient_descent(best_seq, time_budget=alloc, T_start=30.0, T_end=100.0)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  ** Final LSE: obj={best_score:.8f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
