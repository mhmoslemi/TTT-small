def normalize(a):
    s = float(np.sum(a))
    if s < 1e-15:
        return a
    return a / s

def build_conv_matrix(a):
    """A[k,j] such that (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_lo = j
        k_hi = min(j + n, 2 * n - 1)
        A[k_lo:k_hi, j] = a[:k_hi - k_lo]
    return A

def solve_lp_minpeak(a, delta=None):
    """Solve: min M s.t. A@g <= M, sum(g)=1, g>=0, (|g-a|_2 <= delta)."""
    n = len(a)
    a = normalize(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    if delta is not None and delta > 1e-10:
        constraints.append(cp.norm(g - a, 2) <= delta)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(np.max(A @ g_val))
        except Exception:
            continue
    return None, None

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) Nesterov-accelerated smoothed gradient descent with adaptive tau,
(2) LP polishing with trust region, (3) exact coordinate descent,
(4) systematic multi-n exploration with resampling.
Key improvements: FFT-based gradient computation, proper simplex projection,
adaptive smoothing schedule, and aggressive n search.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def obj_normalized(a):
    """Compute objective assuming a is normalized (sum=1). Returns 2n*max(conv)."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def normalize(a):
    """Normalize to sum=1, maintaining non-negativity."""
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-15:
        return a
    return a / s


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_lo = j
        k_hi = min(j + n, 2 * n - 1)
        A[k_lo:k_hi, j] = a[:k_hi - k_lo]
    return A


def solve_lp_minpeak(a, delta=None):
    """Solve LP/SOCP: min M s.t. A@g <= M, sum(g)=1, g>=0, (|g-a|_2 <= delta)."""
    n = len(a)
    a = normalize(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

    if delta is not None and delta > 1e-10:
        constraints.append(cp.norm(g - a, 2) <= delta)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(np.max(A @ g_val))
        except Exception:
            continue
    return None, None


def compute_gradient(a, p):
    """Compute gradient of smoothed objective: 2 * (p cross-corr a)."""
    n = len(a)
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    P_f = rfft(p, N)
    A_f = rfft(a, N)
    c = irfft(P_f * np.conj(A_f), N)[:n]
    return 2.0 * c


def nesterov_gd(a, time_budget=100.0, tau_init=15.0, tau_end=800.0,
                 lr_init=None, verbose=False, seed=42):
    """Nesterov-accelerated smoothed gradient descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    if lr_init is None:
        lr_init = 5.0 / np.sqrt(n)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    momentum = 0.75
    v_mom = np.zeros(n)
    it = 0
    stall = 0
    tau = tau_init

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * (progress ** 0.5)
        lr = lr_init / (1.0 + progress * 2.0)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Softmax weights (numerically stable)
        shifted = tau * (conv - max_conv)
        exp_vals = np.exp(shifted)
        p = exp_vals / float(np.sum(exp_vals))

        # Gradient via FFT cross-correlation
        grad = compute_gradient(a, p)

        # Project to simplex tangent (subtract mean)
        grad_mean = float(np.mean(grad))
        grad_tan = grad - grad_mean

        # Nesterov momentum
        v_mom = momentum * v_mom - lr * grad_tan
        a_try = a + v_mom
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            v_mom = np.zeros(n)
            a_try = a.copy()
        else:
            a_try = a_try / s_try

        obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            stall = 0
        elif obj_try < best_obj * 1.01:
            # Accept if close (for exploration)
            a = a_try
            stall += 1
        else:
            stall += 1

        if stall > 40:
            # Reset with perturbation from best
            sig = 0.01 + 0.02 * (stall / 100.0)
            pert = rng.normal(0, sig, n)
            a = best_a * np.exp(pert)
            a = normalize(a)
            v_mom = np.zeros(n)
            stall = 0

        it += 1
        if verbose and it % 500 == 0:
            print(f"    NGD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.4f}, t={elapsed:.1f}s")

    return best_a


def coordinate_descent(a, time_budget=100.0, verbose=False, seed=42):
    """Exact coordinate descent with FFT-based convolution updates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a[ks_lin - i]
            c_k_lin = conv[ks_lin].copy()

            k_2i = 2 * i
            c_2i = float(conv[k_2i]) if 0 <= k_2i <= 2 * n - 2 else 0.0

            # 1D objective function
            def f_1d(x, v=v, S_other=S_other, n=n, c_k_lin=c_k_lin,
                     a_ki_lin=a_ki_lin, c_2i=c_2i):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) ** 2
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Candidate points
            scale = max(v, S / n, 1e-6)
            candidates = [0.0, v * 0.25, v * 0.5, v * 0.75, v,
                          v * 1.5, v * 2.0, v * 3.0, S / n]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section refinement
            x_lo = max(0.0, best_x - scale * 0.5)
            x_hi = best_x + scale
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(35):
                if x_hi - x_lo < 1e-14 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-15:
                continue

            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_2i <= 2 * n - 2:
                conv[k_2i] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta

        # Periodic re-normalization and drift correction
        if sweep % 10 == 0:
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                a = a / S_check
                conv = compute_conv(a)
                obj = 2.0 * n * float(np.max(conv))
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def lp_polish(a, time_budget=60.0, verbose=False, seed=42):
    """LP-based polishing with adaptive trust region shrinking."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    delta = 1.5  # Initial trust region radius
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        g_lp, M_lp = solve_lp_minpeak(a, delta=delta)
        if g_lp is None:
            delta *= 0.7
            if delta < 0.01:
                break
            continue

        # Try multiple blend factors
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = normalize(a_try)
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
            delta *= 0.85  # Shrink trust region on success
        else:
            stall += 1
            delta *= 0.7
            if stall > 3:
                # Perturb and retry
                pert = rng.normal(0, 0.02, n)
                a = best_a * np.exp(pert)
                a = normalize(a)
                stall = 0

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations for exploration."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        a = np.zeros(n)
        for c in [n // 4, 3 * n // 4]:
            a += np.exp(-((np.arange(n, dtype=float) - c) ** 2) / (2 * (n // 8) ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x * x + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 5:
        # Periodic
        x = np.linspace(0, 1, n)
        f = rng.integers(2, 8)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

    print(f"  n candidates: {n_filtered}")

    for n_try in n_filtered:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = normalize(a_try)

        # Quick NGD
        alloc = min(remaining * 0.12, 50.0)
        if alloc > 15:
            a_try = nesterov_gd(a_try, time_budget=alloc,
                                tau_init=10.0, tau_end=400.0, seed=n_try)

        # Quick CD
        remaining = deadline - time.monotonic()
        cd_alloc = min(remaining * 0.08, 30.0)
        if cd_alloc > 10:
            a_try = coordinate_descent(a_try, time_budget=cd_alloc,
                                       verbose=False, seed=2000 + n_try)

        obj_try = 2.0 * a_try.size * float(np.max(compute_conv(normalize(a_try))))
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_cur = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse Restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = 1 + r
        a_r = diverse_init(n_final, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = normalize(a_r)

        alloc = min(remaining * 0.15, 60.0)
        if alloc > 15:
            a_r = nesterov_gd(a_r, time_budget=alloc,
                              tau_init=15.0, tau_end=500.0, seed=3000 + r * 100)
            obj_r = 2.0 * a_r.size * float(np.max(compute_conv(normalize(a_r))))
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode {mode}): obj={obj_r:.8f}")

    # === Phase 6: Final Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        for final_round in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            # CD
            cd_alloc = min(remaining * 0.35, 80.0)
            if cd_alloc > 10:
                a_f = coordinate_descent(best_seq, time_budget=cd_alloc,
                                         verbose=False, seed=9000 + final_round)
                obj_f = 2.0 * a_f.size * float(np.max(compute_conv(normalize(a_f))))
                if obj_f < best_score:
                    best_score = obj_f
                    best_seq = a_f.copy()
                    print(f"  Final CD {final_round}: obj={best_score:.8f} **")

            # LP
            remaining = deadline - time.monotonic()
            if remaining > 12:
                lp_alloc = min(remaining * 0.3, 40.0)
                if lp_alloc > 8:
                    a_f2 = lp_polish(best_seq, time_budget=lp_alloc,
                                     verbose=False, seed=9500 + final_round)
                    obj_f2 = 2.0 * a_f2.size * float(np.max(compute_conv(normalize(a_f2))))
                    if obj_f2 < best_score:
                        best_score = obj_f2
                        best_seq = a_f2.copy()
                        print(f"  Final LP {final_round}: obj={best_score:.8f} **")

            # Short NGD
            remaining = deadline - time.monotonic()
            if remaining > 10:
                gd_alloc = min(remaining * 0.25, 30.0)
                if gd_alloc > 8:
                    a_f3 = nesterov_gd(best_seq, time_budget=gd_alloc,
                                       tau_init=50.0, tau_end=1000.0,
                                       seed=9800 + final_round)
                    obj_f3 = 2.0 * a_f3.size * float(np.max(compute_conv(normalize(a_f3))))
                    if obj_f3 < best_score:
                        best_score = obj_f3
                        best_seq = a_f3.copy()
                        print(f"  Final NGD {final_round}: obj={best_score:.8f} **")

    # === Final Validation ===
    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(100)
        best_score = 2.0

    best_seq = np.maximum(best_seq, 0.0)
    result = [float(min(1000.0, x)) for x in best_seq]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
