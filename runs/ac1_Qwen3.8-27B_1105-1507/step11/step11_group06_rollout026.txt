    a = np.maximum(a, 0).astype(np.float64)

        return a, np.inf

        k_max = int(np.argmax(conv))

        for i in range(n):

            if 0 <= j2 < n:
                grad[i] += a[j2]

        a_new = np.maximum(a_new, 1e-15)

            a_new = a.copy()
        else:

        best_obj = obj
        best_a = a.copy()
    return best_a, best_obj

"""
Hybrid optimizer: LP-based direction finding with simulated annealing,
multiplicative weights (mirror descent) for local optima escape, and
systematic n exploration. Improves on previous LP-only approach.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp(a):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def line_search(a, direction, t_max=2.0, n_coarse=60, n_fine=40):
    best_t = 0.0
    best_obj = compute_obj(a)
    ts = np.linspace(0.0, t_max, n_coarse)
    for t in ts:
        a_try = (1.0 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    if 0.0 < best_t < t_max:
        lo = max(0.0, best_t - 0.06)
        hi = min(t_max, best_t + 0.06)
        ts_fine = np.linspace(lo, hi, n_fine)
        for t in ts_fine:
            a_try = (1.0 - t) * a + t * direction
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t
    return best_t, best_obj

def u_shaped_init(n, sharpness=3.0):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.02 + 5.0 * (2 * x - 1) ** 4)
    elif mode == 6:
        a = np.cos(np.pi * x) ** 2 + 0.01
    elif mode == 7:
        a = 1.0 + np.sin(2 * np.pi * x) * 0.3
    else:
        a = 1.0 / (0.1 + 2.0 * np.abs(2 * x - 1))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def multiplicative_weights(a, time_budget=30.0, lr0=0.08, decay=0.9998):
    """Mirror descent with multiplicative weights."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr0
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_max - i
            j2 = i - k_max
            if 0 <= j1 < n:
                grad[i] += a[j1]
            if 0 <= j2 < n:
                grad[i] += a[j2]
        
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new
        lr *= decay
        it += 1
        
        if it % 5 == 0 or it < 5:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                best_it = it
        
        if it - best_it > 400:
            a = best_a.copy()
            lr = lr0 * (0.3 + 0.7 * rng.random())
            it = best_it
    
    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()
    return best_a, best_obj

                a = a_new
                stall = 0

                if obj_direct < best_obj:
                    best_obj = obj_direct

                    a = g_lp.copy()

            print(f"      LP-Anneal it{it}: obj={best_obj:.8f}, stall={stall}, t={t_opt:.4f}, elapsed={elapsed:.1f}s")

"""
Hybrid optimizer: LP-based direction finding with simulated annealing restarts,
multiplicative weights (mirror descent) for local optima escape, and systematic
n exploration. Key improvements: larger line search range, annealed LP restarts,
complementary mirror descent, and more aggressive multi-n search.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, direction, t_max=2.0, n_coarse=60, n_fine=40):
    best_t = 0.0
    best_obj = compute_obj(a)
    ts = np.linspace(0.0, t_max, n_coarse)
    for t in ts:
        a_try = (1.0 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    if 0.0 < best_t < t_max:
        lo = max(0.0, best_t - 0.06)
        hi = min(t_max, best_t + 0.06)
        ts_fine = np.linspace(lo, hi, n_fine)
        for t in ts_fine:
            a_try = (1.0 - t) * a + t * direction
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t
    return best_t, best_obj


def u_shaped_init(n, sharpness=3.0):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = (np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) +
             np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01)
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.02 + 5.0 * (2 * x - 1) ** 4)
    elif mode == 6:
        a = np.cos(np.pi * x) ** 2 + 0.01
    elif mode == 7:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * x)
    elif mode == 8:
        a = (1 - np.abs(2 * x - 1)) ** 3 + 0.02
    else:
        a = 1.0 / (0.1 + 2.0 * np.abs(2 * x - 1))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def multiplicative_weights(a, time_budget=30.0, lr0=0.08, decay=0.9998, rng=None):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr0
    it = 0
    best_it = 0
    rng = rng if rng is not None else np.random.default_rng(0)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_max - i
            j2 = i - k_max
            if 0 <= j1 < n:
                grad[i] += a[j1]
            if 0 <= j2 < n:
                grad[i] += a[j2]

        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new
        lr *= decay
        it += 1

        if it % 5 == 0 or it < 5:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                best_it = it

        if it - best_it > 400:
            a = best_a.copy()
            lr = lr0 * (0.3 + 0.7 * rng.random())
            it = best_it

    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()
    return best_a, best_obj


def lp_polish_annealed(a, time_budget=100.0, max_iters=150, rng=None):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = rng if rng is not None else np.random.default_rng(123)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 12:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        t_opt, obj_opt = line_search(a, g_lp, t_max=2.0, n_coarse=50, n_fine=30)

        if t_opt > 1e-6 and obj_opt < best_obj - 1e-14:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            if stall >= 3:
                pert_scale = 0.008 * min(stall, 6)
                pert = rng.normal(0, pert_scale, n)
                a_pert = a * np.exp(pert)
                a_pert = np.maximum(a_pert, 0)
                s_p = float(np.sum(a_pert))
                if s_p > 0.01:
                    a = a_pert / s_p
                else:
                    a = best_a.copy()
                stall = 0
            elif stall >= 2:
                obj_direct = compute_obj(g_lp)
                if obj_direct < best_obj:
                    best_obj = obj_direct
                    best_a = g_lp.copy()
                    a = g_lp.copy()
                    stall = 0

        it += 1
        if it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"      LP-Anneal it{it}: obj={best_obj:.8f}, stall={stall}, t={t_opt:.4f}, el={elapsed:.0f}s")

    return best_a, best_obj


def interpolate_seq(a_src, n_dst):
    n_src = len(a_src)
    if n_src == 1:
        return np.ones(n_dst) / n_dst
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s
