    """Compute smoothed subgradient of max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    # Soft weights
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    # g[i] = 2 * sum_k p[k] * a[k-i] (if 0<=k-i<n)
    # This is 2 * cross_correlation(p, a)
    N = 1
    while N < max(2*n, len(conv)):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real

    # We want linear cross-correlation, so we need to be careful about wrap-around
    # For our purposes, the circular version is a good approximation when N >= 2n-1
    g = 2.0 * g_full[:n]
    
    return g, conv, p, max_conv

Optimizes step function coefficients using a hybrid strategy:
1. Fast multiplicative subgradient (spectral, O(n log n) per iteration) for rapid convergence
2. Precise coordinate descent with golden section for fine-tuning
3. LP-based direction finding for escaping local minima
4. Multi-n exploration for finding the best sequence length

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def compute_smoothed_gradient(a, tau):
    """
    Compute smoothed subgradient of max(conv(a,a)) using log-sum-exp.
    Returns (gradient, conv, weights, max_conv).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n], conv, p, max_conv

    Uses adaptive step size and tau scheduling.
    """

    s = float(np.sum(a))

        return a

    start = time.monotonic()

    v = np.zeros(n)

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * (progress ** 0.7)
        eta_eff = eta_init * (1.0 - 0.6 * progress)

        # Nesterov lookahead

                else:
                    f_val = 2.0 * n * max_val / denom
                if f_val < best_f:
                    best_f = f_val
                    best_x = x_c

            # Golden section around best_x
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.0

            # Ensure best_x is between x_lo and x_hi
            if x_lo >= x_hi:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)

            # Evaluate at c and d
            def eval_f(x):
                if len(ks_lin) > 0:
                    max_lin = float(np.max(c_k + two_a_ki * (x - v)))
                else:
                    max_lin = 0.0
                if has_quad:
                    quad_val = c_2i + x * x - v_sq
                    max_val = max(max_lin, quad_val)
                else:
                    max_val = max_lin
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            fc = eval_f(c_pt)
            fd = eval_f(d_pt)

            # Check if current v is better
            fv = eval_f(v)
            if fv < fc:
                x_hi = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = x_lo + gr * (x_hi - x_lo)
                fd = eval_f(d_pt)
            elif fv < fd:
                x_lo = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = x_hi - gr * (x_hi - x_lo)
                fc = eval_f(c_pt)

            for _ in range(35):
                if x_hi - x_lo < 1e-13 * max(1.0, abs(x_hi)):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = eval_f(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = eval_f(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            f_opt = eval_f(x_opt)
            f_cur = eval_f(v)

            if f_opt < f_cur - 1e-13:
                delta = x_opt - v
                a[i] = x_opt
                if len(ks_lin) > 0:
                    conv[ks_lin] += 2.0 * delta * a_ki
                if has_quad:
                    conv[2 * i] = c_2i + x_opt * x_opt - v_sq
                S += delta

                if S > 1e-10:
                    obj = 2.0 * n * float(np.max(conv)) / (S * S)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()

        obj_after = best_obj
        if obj_after > obj_before - 1e-10:
            stall_count += 1
        else:
            stall_count = 0

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, time_budget=20.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    start = time.monotonic()
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_proximity(a, lam=0.05, time_budget=20.0):
    """LP with proximity: min M + lam*||g-a||_1 s.t. conv(a,g)<=M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    objective = cp.Minimize(M + lam * cp.norm1(g - a_norm))
    prob = cp.Problem(objective, constraints)

    start = time.monotonic()
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS]
    for solver in solvers:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_direction_polish(a, time_budget=60.0, verbose=False, seed=42):
    """LP-based direction finding and blending."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        use_prox = (stall > 1) or (rng.random() < 0.3)
        alloc = min(remaining * 0.4, 18.0)

        if use_prox:
            g_lp, _ = solve_lp_proximity(a, lam=0.05, time_budget=alloc)
        else:
            g_lp, _ = solve_lp_minpeak(a, time_budget=alloc)

        if g_lp is None:
            break

        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 2:
                pert = rng.normal(0, 0.01 * stall, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / float(a.sum())
                stall = 0

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        # Spectral random
        N = n
        X = rng.standard_normal(N) + 1j * rng.standard_normal(N)
        a = np.real(ifft(X))
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    elif mode == 3:
        # Smooth random (low-pass filtered)
        a = rng.standard_normal(n)
        # Apply simple moving average for smoothness
        kernel = np.ones(5) / 5.0
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a = a / (a.sum() + 1e-15)
    elif mode == 4:
        # Bimodal
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.3) ** 2) / 0.02) + np.exp(-((x - 0.7) ** 2) / 0.02)
        a += 0.1 * rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 5:
        # Cosine window with modulation
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a += 0.2 * np.abs(np.sin(2 * np.pi * 3 * x))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining multiplicative subgradient (fast), coordinate descent (precise),
    LP polishing, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 12.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    parent_n = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                parent_n = parent.size
                best_score = compute_obj_normalized(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent_n}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(parent_n) / parent_n
        best_score = compute_obj_normalized(best_seq)
        print(f"Default init (n={parent_n}): obj={best_score:.8f}")

    # === Phase 1: Fast Multiplicative Subgradient (get close to optimum quickly) ===
    print(f"\n=== Phase 1: Multiplicative Subgradient ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.15, 120.0)
    if p1_budget > 20:
        a_ms = multiplicative_subgradient(best_seq, time_budget=p1_budget,
                                          tau_init=30.0, tau_end=400.0,
                                          eta_init=0.8, verbose=True, seed=seed)
        obj_ms = compute_obj_normalized(a_ms)
        if obj_ms < best_score:
            best_score = obj_ms
            best_seq = a_ms.copy()
            print(f"  MS: obj={best_score:.8f} **")
        else:
            print(f"  MS: obj={obj_ms:.8f} (no improvement)")

    # === Phase 2: Coordinate Descent (main precision tool) ===
    print(f"\n=== Phase 2: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.30, 400.0)
    if p2_budget > 30:
        a_cd = coordinate_descent_fast(best_seq, time_budget=p2_budget,
                                       verbose=True, seed=seed)
        obj_cd = compute_obj_normalized(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 3: Second MS pass (refine after CD) ===
    print(f"\n=== Phase 3: Second MS Pass ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = min(remaining * 0.08, 60.0)
        a_ms2 = multiplicative_subgradient(best_seq, time_budget=p3_budget,
                                           tau_init=80.0, tau_end=600.0,
                                           eta_init=0.3, verbose=False, seed=seed + 100)
        obj_ms2 = compute_obj_normalized(a_ms2)
        if obj_ms2 < best_score:
            best_score = obj_ms2
            best_seq = a_ms2.copy()
            print(f"  MS2: obj={best_score:.8f} **")
        else:
            print(f"  MS2: obj={obj_ms2:.8f}")

    # === Phase 4: CD again (precision after MS2) ===
    print(f"\n=== Phase 4: Second CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = min(remaining * 0.20, 200.0)
        a_cd2 = coordinate_descent_fast(best_seq, time_budget=p4_budget,
                                        verbose=True, seed=seed + 777)
        obj_cd2 = compute_obj_normalized(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  CD2: obj={best_score:.8f} **")
        else:
            print(f"  CD2: obj={obj_cd2:.8f}")

    # === Phase 5: LP Polishing ===
    print(f"\n=== Phase 5: LP Polishing ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p5_budget = min(remaining * 0.15, 120.0)
        a_lp = lp_direction_polish(best_seq, time_budget=p5_budget,
                                   verbose=False, seed=3000)
        obj_lp = compute_obj_normalized(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
            # Follow up with CD
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_cd_lp = coordinate_descent_fast(best_seq, time_budget=min(remaining * 0.2, 80.0),
                                                  verbose=False, seed=seed + 1111)
                obj_cd_lp = compute_obj_normalized(a_cd_lp)
                if obj_cd_lp < best_score:
                    best_score = obj_cd_lp
                    best_seq = a_cd_lp.copy()
                    print(f"  LP+CD: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        n_best - 50, n_best - 100, n_best + 50, n_best + 100,
        n_best + 150, n_best + 200, n_best + 300,
        600, 700, 850, 900, 950, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500 and abs(n - n_best) > 40]
    n_options = n_options[:8]

    print(f"  Current best n={n_best}, trying: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # Fast MS + CD on new length
        alloc_ms = min(remaining * 0.05, 30.0)
        if alloc_ms > 10:
            a_try = multiplicative_subgradient(a_try, time_budget=alloc_ms,
                                               tau_init=30.0, tau_end=200.0,
                                               eta_init=0.5, verbose=False, seed=n_try)

        alloc_cd = min(remaining * 0.08, 50.0)
        if alloc_cd > 15:
            a_try = coordinate_descent_fast(a_try, time_budget=alloc_cd,
                                            verbose=False, seed=n_try + 500)

        obj_try = compute_obj_normalized(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse Restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = 1 + r
        a_r = diverse_init(n_current, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / float(a_r.sum())

        # Quick MS + CD
        alloc = min(remaining * 0.15, 50.0)
        if alloc > 15:
            a_r = multiplicative_subgradient(a_r, time_budget=alloc * 0.4,
                                             tau_init=30.0, tau_end=150.0,
                                             eta_init=0.5, verbose=False, seed=9000 + r * 100)
            a_r = coordinate_descent_fast(a_r, time_budget=alloc * 0.6,
                                          verbose=False, seed=9000 + r * 100 + 1)
            obj_r = compute_obj_normalized(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode {mode}): obj={obj_r:.8f}")

    # === Phase 8: Final Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        alloc = min(remaining * 0.35, 120.0)
        a_final = coordinate_descent_fast(best_seq, time_budget=alloc,
                                          verbose=False, seed=seed + 19999)
        obj_final = compute_obj_normalized(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.25, 60.0)
        a_final2 = multiplicative_subgradient(best_seq, time_budget=alloc,
                                              tau_init=100.0, tau_end=800.0,
                                              eta_init=0.2, verbose=False, seed=seed + 22222)
        obj_final2 = compute_obj_normalized(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final MS: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_final3 = coordinate_descent_fast(best_seq, time_budget=min(remaining * 0.5, 50.0),
                                           verbose=False, seed=seed + 25000)
        obj_final3 = compute_obj_normalized(a_final3)
        if obj_final3 < best_score:

            best_seq = a_final3.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:

    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
