"""
Optimized step function search using Adam-based soft-max gradient descent,
peak-deflation QP polishing, and coordinate descent refinement.
Key innovations:
- Adam optimizer with adaptive per-coordinate learning rates
- Hybrid gradient (log-sum-exp + exact peak subgradient)
- Iterative peak-deflation via QP (targets the maximum lag directly)
- Coordinate descent for final fine-tuning
- Parametric family grid search for initialization
- Aggressive length exploration
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return float(2.0 * n * float(np.max(conv)))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_gradient_lse(a, conv, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*conv)))*n using softmax weights."""
    n = len(a)
    c_max = float(np.max(conv))
    exp_vals = np.exp(beta * (conv - c_max))
    p = exp_vals / float(np.sum(exp_vals))
    # grad_i = 2 * sum_j a_j * p_{i+j} (cross-correlation of a and p)
    a_rev = a[::-1]
    size = 1
    while size < n + len(p) - 1:
        size <<= 1
    A = rfft(a_rev, size)
    P = rfft(p, size)
    full = irfft(A * P, size)
    q = full[n - 1:n - 1 + n]
    return 2.0 * q


def compute_peak_subgradient(a, k_peak):
    """Exact subgradient of conv(a,a)_{k_peak} w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for m in range(n):
        j = k_peak - m
        if 0 <= j < n:
            grad[m] = 2.0 * a[j]
    return grad


def adam_softmax_gd(a, time_budget=300.0, beta_start=15.0, beta_end=60.0,
                     lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                     peak_weight=0.5, verbose=True, seed=42):
    """Adam-based soft-max gradient descent with peak subgradient hybrid."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    it = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress
        # Decay learning rate
        lr_cur = lr * (1.0 - 0.5 * progress ** 2)
        
        conv = compute_conv(a)
        k_peak = int(np.argmax(conv))
        
        # Log-sum-exp gradient
        grad_lse = compute_gradient_lse(a, conv, beta)
        
        # Peak subgradient
        grad_peak = compute_peak_subgradient(a, k_peak)
        # Normalize peak gradient to have similar scale
        scale_peak = float(np.sum(grad_peak))
        if scale_peak > 1e-10:
            grad_peak = grad_peak / scale_peak * float(np.sum(grad_lse))
        
        # Hybrid gradient
        grad = (1 - peak_weight) * grad_lse + peak_weight * grad_peak
        
        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        
        a_new = a - lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Perturbation restart
                pert = rng.normal(0, 0.003, n)
                a = project_simplex(best_a + pert)
                m = np.zeros(n)
                v = np.zeros(n)
                stall = 0
        
        a = a_new
        it += 1
        
        if verbose and it % 300 == 0:

def peak_deflation_qp(a, M_target, time_per_qp=3.0):
    """Solve QP to reduce the peak lag: min 0.5*||g-a||^2 s.t. conv(a,g)_{k*} <= M_target, sum(g)=1, g>=0."""
    n = len(a)
    conv_a = compute_conv(a)
    k_peak = int(np.argmax(conv_a))
    
    # conv(a,g)_{k_peak} = sum_i a_i * g_{k_peak - i} (linear in g)
    c = np.zeros(n)
    for i in range(n):
        j = k_peak - i
        if 0 <= j < n:
            c[i] = a[j]
    
    g = cp.Variable(n, nonneg=True)
    constraints = [c @ g <= M_target, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(cp.sum_squares(g - a)), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def iterative_peak_deflation(a, time_budget=60.0, verbose=True):
    """Iteratively apply peak deflation QP."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget and it < 100:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        conv = compute_conv(a)
        M_current = float(np.max(conv))
        # Try to reduce by 1%
        M_target = M_current * 0.995
        
        g_new = peak_deflation_qp(a, M_target, time_per_qp=min(2.0, remaining * 0.1))
        
        if g_new is None:
            stall += 1
        else:
            # Blend
            for t in [0.5, 0.8, 1.0, 1.2]:
                a_try = project_simplex((1 - t) * a + t * g_new)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    stall = 0
                    break
            else:
                stall += 1
            it += 1
            
            if verbose and it % 5 == 0:
                print(f"    PeakDefl it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
        
        if stall >= 5:
            break
    
    return best_a, best_obj


def coordinate_descent(a, time_budget=30.0, step_sizes=None):
    """Simple coordinate descent: try perturbing each coordinate."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if step_sizes is None:
        step_sizes = [0.001, 0.005, 0.01, 0.02, 0.05]
    
    rng = np.random.default_rng(123)
    
    while time.monotonic() - start < time_budget:
        improved = False
        # Random shuffle of coordinates
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            for ds in step_sizes:
                for sign in [1, -1]:
                    a_try = a.copy()
                    a_try[i] += sign * ds * a.mean()
                    a_try = np.maximum(a_try, 0)
                    s = float(np.sum(a_try))
                    if s < 0.01:
                        continue
                    a_try = a_try / s
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-10:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            # Try smaller steps
            step_sizes = [s * 0.5 for s in step_sizes]
            if step_sizes[0] < 1e-6:
                break
    
    return best_a, best_obj


def parametric_init(n, eps=0.01, gamma=1.0):
    """U-shaped parametric initialization: 1/(eps + x^2)^gamma."""
    x = 2 * np.linspace(0, 1, n) - 1  # in [-1, 1]
    a = 1.0 / (eps + x ** 2) ** gamma
    return project_simplex(a)


def power_law_init(n, beta=0.5, eps=0.01):
    """Power-law U-shape: 1/|x|^beta + eps."""
    x = 2 * np.linspace(0, 1, n) - 1
    a = 1.0 / (np.abs(x) ** beta + eps)
    return project_simplex(a)


def cos_u_init(n, alpha=2.0, beta=1.0):
    """Cosine U-shape: (1 + alpha*cos(pi*x))^beta."""
    x = np.linspace(0, 1, n)
    a = (1.0 + alpha * np.cos(np.pi * (2 * x - 1))) ** beta
    a = np.maximum(a, 0)
    return project_simplex(a)


def diverse_initializations(n, rng):
    """Generate multiple diverse initializations and return the best."""
    candidates = []
    
    # Parametric family
    for eps in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
        for gamma in [0.5, 1.0, 1.5, 2.0, 3.0]:
            candidates.append(parametric_init(n, eps, gamma))
    
    for beta in [0.3, 0.5, 0.7, 1.0, 1.5]:
        candidates.append(power_law_init(n, beta))
    
    for alpha in [1.0, 2.0, 3.0, 5.0]:
        for beta in [0.5, 1.0, 2.0, 3.0]:
            candidates.append(cos_u_init(n, alpha, beta))
    
    # Random smooth
    for _ in range(5):
        a = rng.uniform(0.1, 1.0, n)
        kernel_size = max(3, n // 20)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        candidates.append(project_simplex(a))
    
    # Evaluate all
    best_a = None
    best_obj = np.inf
    for c in candidates:
        obj = compute_obj(c)
        if obj < best_obj:
            best_obj = obj
            best_a = c
    
    return best_a, best_obj

- Hybrid gradient (log-sum-exp + exact peak subgradient)
- Iterative peak-deflation via QP (targets the maximum lag directly)
- Fast peak-reduction step (water-filling) as QP alternative
- Coordinate descent for final fine-tuning
- Parametric family grid search for initialization
- Aggressive length exploration
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return float(2.0 * n * float(np.max(conv)))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_gradient_lse(a, conv, beta):
    """Gradient of soft-max objective using softmax weights and FFT cross-correlation."""
    n = len(a)
    c_max = float(np.max(conv))
    exp_vals = np.exp(beta * (conv - c_max))
    p = exp_vals / float(np.sum(exp_vals))
    a_rev = a[::-1]
    size = 1
    while size < n + len(p) - 1:
        size <<= 1
    A = rfft(a_rev, size)
    P = rfft(p, size)
    full = irfft(A * P, size)
    q = full[n - 1:n - 1 + n]
    return 2.0 * q


def compute_peak_subgradient(a, k_peak):
    """Exact subgradient of conv(a,a)_{k_peak} w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for m in range(n):
        j = k_peak - m
        if 0 <= j < n:
            grad[m] = 2.0 * a[j]
    return grad


def adam_softmax_gd(a, time_budget=300.0, beta_start=15.0, beta_end=60.0,
                    lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                    peak_weight=0.5, verbose=True, seed=42):
    """Adam-based soft-max gradient descent with peak subgradient hybrid."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    it = 0
    stall = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress
        lr_cur = lr * (1.0 - 0.5 * progress ** 2)

        conv = compute_conv(a)
        k_peak = int(np.argmax(conv))

        grad_lse = compute_gradient_lse(a, conv, beta)
        grad_peak = compute_peak_subgradient(a, k_peak)

        # Normalize peak gradient to match LSE gradient scale
        sum_lse = float(np.sum(grad_lse))
        sum_peak = float(np.sum(grad_peak))
        if sum_peak > 1e-10 and sum_lse > 1e-10:
            grad_peak = grad_peak * (sum_lse / sum_peak)

        grad = (1.0 - peak_weight) * grad_lse + peak_weight * grad_peak

        it += 1
        m_adam = beta1 * m_adam + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * (grad * grad)
        m_hat = m_adam / (1.0 - beta1 ** it)
        v_hat = v_adam / (1.0 - beta2 ** it)

        a_new = a - lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                pert = rng.normal(0, 0.004, n)
                a = project_simplex(best_a + pert)
                m_adam = np.zeros(n)
                v_adam = np.zeros(n)
                stall = 0

        a = a_new

        if verbose and it % 200 == 0:
            print(f"    AdamGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, "
                  f"lr={lr_cur:.5f}, t={elapsed:.1f}s")

    return best_a, best_obj


def fast_peak_reduction(a, alpha=0.02):
    """Fast water-filling step to reduce the peak lag without QP."""
    n = len(a)
    conv = compute_conv(a)
    k_peak = int(np.argmax(conv))

    # Pressure vector
    c = np.zeros(n)
    for m in range(n):
        j = k_peak - m
        if 0 <= j < n:
            c[m] = a[j]

    # Reduce coordinates with high pressure
    c_centered = c - np.mean(c)
    a_new = a - alpha * c_centered
    a_new = project_simplex(a_new)
    return a_new


def peak_deflation_qp(a, M_target):
    """Solve QP: min 0.5*||g-a||^2 s.t. conv(a,g)_{k*} <= M_target, sum(g)=1, g>=0."""
    n = len(a)
    conv_a = compute_conv(a)
    k_peak = int(np.argmax(conv_a))

    c = np.zeros(n)
    for i in range(n):
        j = k_peak - i
        if 0 <= j < n:
            c[i] = a[j]

    g = cp.Variable(n, nonneg=True)
    constraints = [c @ g <= M_target, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(cp.sum_squares(g - a)), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def iterative_peak_deflation(a, time_budget=60.0, verbose=True, use_qp=True):
    """Iteratively apply peak deflation (QP or fast step)."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget and it < 200:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        conv = compute_conv(a)
        M_current = float(np.max(conv))

        if use_qp and remaining > 5:
            M_target = M_current * 0.99
            g_new = peak_deflation_qp(a, M_target)
            if g_new is not None:
                for t in [0.5, 0.7, 0.9, 1.0, 1.2]:
                    a_try = project_simplex((1 - t) * a + t * g_new)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        it += 1
                        stall = 0
                        break
                else:
                    # Fallback to fast step
                    a_try = fast_peak_reduction(a, alpha=0.01)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        it += 1
                        stall = 0
                    else:
                        stall += 1
            else:
                stall += 1
        else:
            # Fast peak reduction
            a_try = fast_peak_reduction(a, alpha=0.01)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                it += 1
                stall = 0
            else:
                # Try smaller alpha
                a_try = fast_peak_reduction(a, alpha=0.005)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    it += 1
                    stall = 0
                else:
                    stall += 1

        it += 1
        if verbose and it % 10 == 0:
            print(f"    PeakDefl it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

        if stall >= 8:
            break

    return best_a, best_obj


def coordinate_descent(a, time_budget=30.0):
    """Coordinate descent: try perturbing pairs of coordinates (mass transfer)."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(456)

    step_sizes = [0.001, 0.003, 0.005, 0.01]

    while time.monotonic() - start < time_budget:
        improved = False
        # Try random coordinate pairs (mass transfer)
        n_tries = min(500, n * 2)
        for _ in range(n_tries):
            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for ds in step_sizes:
                for sign in [1, -1]:
                    a_try = a.copy()
                    amount = sign * ds * a.mean()
                    a_try[i] -= amount
                    a_try[j] += amount
                    if a_try[i] < 0 or a_try[j] < 0:
                        continue
                    a_try = a_try / float(np.sum(a_try))
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-10:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            step_sizes = [s * 0.5 for s in step_sizes]
            if step_sizes[0] < 1e-6:
                break

    return best_a, best_obj


def parametric_init(n, eps, gamma):
    """U-shaped: 1/(eps + x^2)^gamma."""
    x = 2 * np.linspace(0, 1, n) - 1
    a = 1.0 / (eps + x * x) ** gamma
    return project_simplex(a)


def power_law_init(n, beta, eps=0.01):
    """Power-law U-shape."""
    x = 2 * np.linspace(0, 1, n) - 1
    a = 1.0 / (np.abs(x) ** beta + eps)
    return project_simplex(a)


def cos_u_init(n, alpha, beta_p):
    """Cosine U-shape."""
    x = np.linspace(0, 1, n)
    a = (1.0 + alpha * np.cos(np.pi * (2 * x - 1))) ** beta_p
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_initializations(n, rng, max_candidates=50):
    """Generate diverse initializations and return the best."""
    candidates = []

    for eps in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
        for gamma in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
            candidates.append(parametric_init(n, eps, gamma))

    for beta in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        candidates.append(power_law_init(n, beta))

    for alpha in [1.0, 2.0, 3.0, 5.0, 8.0]:
        for beta_p in [0.5, 1.0, 1.5, 2.0, 3.0]:
            candidates.append(cos_u_init(n, alpha, beta_p))

    for _ in range(5):
        a = rng.uniform(0.1, 1.0, n)
        ksize = max(3, n // 20)
        kernel = np.ones(ksize) / ksize
        a = np.convolve(a, kernel, mode='same')
        candidates.append(project_simplex(a))

    # Limit candidates
    if len(candidates) > max_candidates:
        # Keep a diverse subset
        idx = rng.choice(len(candidates), max_candidates, replace=False)
        candidates = [candidates[i] for i in idx]

    best_a = None
    best_obj = np.inf
    for c in candidates:
        obj = compute_obj(c)
        if obj < best_obj:
            best_obj = obj
            best_a = c

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Adam soft-max GD + peak-deflation QP + coordinate descent.

    """

    best_score = np.inf

    # Load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))

                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    # Phase 0: Parametric initialization
    remaining = deadline - time.monotonic()

        print(f"\n=== Phase 0: Parametric Init, time={remaining:.0f}s ===")
        t0 = time.monotonic()
        a_init, obj_init = diverse_initializations(n_current, rng)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Parametric init: obj={best_score:.8f} **")
        else:
            print(f"  Parametric init: obj={obj_init:.8f} (no improvement)")
        print(f"  Time: {time.monotonic()-t0:.1f}s")

    if best_seq is None:
        best_seq = parametric_init(n_current, 0.01, 1.0)
        best_score = compute_obj(best_seq)
        print(f"  Fallback init: obj={best_score:.8f}")

    # Phase 1: Adam GD (smooth)
    remaining = deadline - time.monotonic()

            best_seq = a1.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj1:.8f}")

    # Phase 2: Adam GD (sharper)

        if obj2 < best_score:

            best_seq = a2.copy()
            print(f"  Phase2: obj={best_score:.8f} **")
        else:
            print(f"  Phase2: obj={obj2:.8f}")

    # Phase 3: Peak deflation (QP + fast steps)
    remaining = deadline - time.monotonic()
    if remaining > 60:
