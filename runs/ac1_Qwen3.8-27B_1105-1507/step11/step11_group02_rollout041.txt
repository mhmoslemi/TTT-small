Optimizes step function coefficients using a hybrid of:
1. L-BFGS-B second-order optimization on log-sum-exp smoothed objective
2. Exact coordinate descent with golden section 1D minimization
3. LP-based direction finding via cvxpy

5. Diverse initialization and adaptive restarts
The key innovation over the previous approach is using L-BFGS-B as the primary

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def smoothed_obj_grad(a, tau):
    """
    Compute smoothed objective and gradient using log-sum-exp approximation.
    f(a) ≈ 2*n*logsumexp(tau*conv(a,a)) / (tau * (sum(a))^2)
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    conv = compute_conv(a)
    tc = tau * conv
    max_tc = float(np.max(tc))
    exp_tc = np.exp(tc - max_tc)
    sum_exp = float(np.sum(exp_tc))
    
    log_sum_exp = max_tc + np.log(sum_exp)
    f = 2.0 * n * log_sum_exp / (tau * s * s)
    
    # Softmax weights
    p = exp_tc / sum_exp
    
    # Gradient: df/da[i] = 4*n*(p cross-correlate a)[i]/s^2 - 4*n*log_sum_exp/(tau*s^3)
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p.astype(np.complex128), N)
    A = fft(a.astype(np.complex128), N)
    cross = ifft(P * np.conj(A)).real[:n]
    
    grad = 4.0 * n * cross / (s * s) - 4.0 * n * log_sum_exp / (tau * s * s * s)
    
    return f, grad


def lbgfs_run(a0, tau, time_budget=30.0, maxiter=100, maxfun=2000):
    """Run L-BFGS-B on smoothed objective."""
    start = time.monotonic()
    
    def fun_grad(x):
        f, g = smoothed_obj_grad(x, tau)
        return f, g
    
    bounds = [(0.0, None)] * len(a0)
    try:
        result = minimize(fun_grad, a0, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'maxfun': maxfun,
                                   'ftol': 1e-14, 'gtol': 1e-12,
                                   'maxls': 50})
        if time.monotonic() - start < time_budget:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return a0


def coordinate_descent(a, time_budget=60.0, seed=42):
    """Exact coordinate descent with golden section search."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    while time.monotonic() - start < time_budget:
        order = rng.permutation(n)
        S = float(np.sum(a))
        
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            
            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue
            
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            
            # Linear part: conv[k] changes by 2*(x-v)*a[k-i] for k != 2i
            k_quad = 2 * i
            mask = (ks != k_quad)
            ks_lin = ks[mask]
            coeff_lin = a[ks_lin - i]  # a[k-i] for linear terms
            c_lin = conv[ks_lin].copy()
            
            c_quad = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0
            
            def f_1d(x):
                if len(ks_lin) > 0:
                    vals = c_lin + 2.0 * coeff_lin * (x - v)
                    max_lin = float(np.max(vals))
                else:
                    max_lin = 0.0
                quad_val = c_quad + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom
            
            # Try several candidates
            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, v * 0.5, v * 0.7, v * 1.3, v * 1.5,
                          v * 2.0, 1.0 / n, 2.0 / n, v + 0.01 * scale, v - 0.01 * scale]
            candidates = [max(0.0, c) for c in candidates]
            candidates = sorted(set(candidates))
            
            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x
            
            # Golden section refinement
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.0
            if x_hi - x_lo < 1e-14:
                continue
            
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)
            
            for _ in range(40):
                if x_hi - x_lo < 1e-14 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)
            
            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-14:
                continue
            
            delta = x_opt - v
            a[i] = x_opt
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * coeff_lin
            S += delta
            
            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def lp_direction(a, time_budget=20.0):
    """
    Solve LP: min M s.t. (a_norm * g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns g (the optimal direction) or None.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    start = time.monotonic()
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def diverse_init(n, rng, mode=0):

    elif mode == 4:

    elif mode == 5:

    elif mode == 6:

        a = 0.5 + 0.3 * np.sin(2 * np.pi * rng.integers(1, 5) * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:

    return a

```python
"""
Optimizes step function coefficients using a hybrid of:
1. L-BFGS-B second-order optimization on log-sum-exp smoothed objective (primary)
2. Exact coordinate descent with golden section 1D minimization (refinement)
3. LP-based direction finding via cvxpy (escape机制)
4. Multi-length exploration with interpolation-based transfer
5. Momentum-based smoothed descent as backup
Key innovation: L-BFGS-B exploits curvature to navigate the non-convex landscape
more effectively than pure coordinate descent, especially for escaping shallow basins.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def smoothed_obj_grad(a, tau):
    """
    Compute smoothed objective and gradient using log-sum-exp.
    f(a) = 2*n*logsumexp(tau*conv(a,a)) / (tau * (sum(a))^2)
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return 1e10, np.zeros(n)

    conv = compute_conv(a)
    tc = tau * conv
    max_tc = float(np.max(tc))
    exp_tc = np.exp(tc - max_tc)
    sum_exp = float(np.sum(exp_tc))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(tc))] = 1.0
        log_sum_exp = max_tc
    else:
        log_sum_exp = max_tc + np.log(sum_exp)
        p = exp_tc / sum_exp

    f = 2.0 * n * log_sum_exp / (tau * s * s)

    # Gradient via cross-correlation
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p.astype(np.complex128), N)
    A = fft(a.astype(np.complex128), N)
    cross = ifft(P * np.conj(A)).real[:n]

    grad = 4.0 * n * cross / (s * s) - 4.0 * n * log_sum_exp / (tau * s * s * s)

    return f, grad


def lbgfs_run(a0, tau, time_budget=30.0, maxiter=100, maxfun=2000):
    """Run L-BFGS-B on smoothed objective with non-negativity bounds."""
    start = time.monotonic()
    a0 = np.maximum(a0, 0.0).astype(np.float64).copy()

    def fun_grad(x):
        if time.monotonic() - start > time_budget:
            return 1e10, np.zeros(len(x))
        return smoothed_obj_grad(x, tau)

    bounds = [(0.0, None)] * len(a0)
    try:
        result = minimize(fun_grad, a0, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'maxfun': maxfun,
                                   'ftol': 1e-14, 'gtol': 1e-12,
                                   'maxls': 50, 'iprint': -1})
        if result.x is not None:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return a0


def momentum_smoothed(a0, tau, time_budget=60.0, lr_init=0.5, decay=0.9995):
    """Momentum-based smoothed gradient descent with adaptive lr."""
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = compute_obj(a)
    lr = lr_init
    start = time.monotonic()
    momentum = 0.85

    while time.monotonic() - start < time_budget:
        f, g = smoothed_obj_grad(a, tau)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break

        # Adaptive lr based on gradient magnitude
        lr_eff = lr / (1.0 + g_norm * 0.1)
        v = momentum * v - lr_eff * g
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
            lr *= 0.5
            continue
        a_new = a_new / s_new
        a = a_new
        lr *= decay

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a


def coordinate_descent(a, time_budget=60.0, seed=42):
    """Exact coordinate descent with golden section 1D minimization."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)

            k_quad = 2 * i
            mask = (ks != k_quad)
            ks_lin = ks[mask]
            coeff_lin = a[ks_lin - i]
            c_lin = conv[ks_lin].copy()
            c_quad = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            # Evaluate 1D function
            def f_1d(x, v=v, S_other=S_other, n=n, c_lin=c_lin,
                     coeff_lin=coeff_lin, c_quad=c_quad, ks_lin=ks_lin):
                if len(ks_lin) > 0:
                    vals = c_lin + 2.0 * coeff_lin * (x - v)
                    max_lin = float(np.max(vals))
                else:
                    max_lin = 0.0
                quad_val = c_quad + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Candidate points
            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, v * 0.5, v * 0.7, v * 1.3, v * 1.5,
                          v * 2.0, 1.0 / n, 2.0 / n,
                          max(0.0, v + 0.05 * scale), max(0.0, v - 0.05 * scale)]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.0
            if x_hi - x_lo < 1e-14:
                x_opt = best_x
            else:
                c_pt = x_hi - gr * (x_hi - x_lo)
                d_pt = x_lo + gr * (x_hi - x_lo)
                fc = f_1d(c_pt)
                fd = f_1d(d_pt)

                for _ in range(40):
                    if x_hi - x_lo < 1e-14 * max(1.0, x_hi):
                        break
                    if fc < fd:
                        x_hi = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = x_hi - gr * (x_hi - x_lo)
                        fc = f_1d(c_pt)
                    else:
                        x_lo = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = x_lo + gr * (x_hi - x_lo)
                        fd = f_1d(d_pt)

                x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * coeff_lin
            S += delta

            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def lp_direction(a, time_budget=20.0):
    """Solve LP: min M s.t. (a_norm * g)[k] <= M, sum(g)=1, g>=0. Returns g or None."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    start = time.monotonic()
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0.0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        s = a.sum()
        a = a / s if s > 0 else np.ones(n) / n
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.3) ** 2) / 0.02) + np.exp(-((x - 0.7) ** 2) / 0.02)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = 1.0 / np.sqrt(1 - (2 * x - 1) ** 2 + 0.05)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        a[:n // 8] *= 0.4
        a[-n // 8:] *= 0.4
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        freq = rng.integers(2, 8)
        a = 0.6 + 0.4 * np.sin(2 * np.pi * freq * x + rng.uniform(0, 6.28))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 7:
        # Two-peak with random positions
        x = np.linspace(0, 1, n)
        c1, c2 = np.sort(rng.uniform(0.1, 0.9, 2))
        a = np.exp(-((x - c1) ** 2) / 0.01) + np.exp(-((x - c2) ** 2) / 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a
