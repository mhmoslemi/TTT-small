    """

        if np.isnan(x) or np.isinf(x):

    sequence = [float(x) for x in sequence]

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):

def smoothed_grad(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""

    shifted = tau * (conv - exact_max)

        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    P = fft(p, N)

def lp_solve(a, solver_list=None):
    """Solve min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1. Returns (g, M)."""

        A[j:j + n, j] = a

    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]

                sg = float(np.sum(g_val))

                    return g_val / sg, float(M_var.value)

            continue

def staircase_gd(a, time_budget, tau_levels=None, verbose=False, beta=0.9, lr_scale=2.0):
    """Nesterov accelerated gradient descent with staircase tau schedule."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [3, 8, 20, 50, 120, 300, 700, 1500, 3000]

    for tau in tau_levels:
        lr = lr_scale / np.sqrt(n)
        v = np.zeros(n)
        tau_iters = 0
        max_iters_per_level = 5000

        while time.monotonic() - start < time_budget:
            if tau_iters >= max_iters_per_level:
                break

            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new
            tau_iters += 1
            lr = max(1e-7, lr * 0.99999)

        if verbose:
            print(f"    tau={tau:5d}: obj={best_obj:.10f}, iters={tau_iters}")

    return best_a

def active_set_refine(a, time_budget, verbose=False, seed=42, max_iters=200):
    """
    Targeted optimization: specifically reduce the lags where convolution is maximal.
    Uses subgradient of the max + LP blending.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        # Find all near-max lags
        threshold = max_val * 0.999
        max_lags = np.where(conv >= threshold)[0]

        # Compute subgradient: average gradient over active lags
        subgrad = np.zeros(n)
        for lag in max_lags:
            for i in range(n):
                j = lag - i
                if 0 <= j < n:
                    subgrad[i] += 2.0 * a[j]
        subgrad /= len(max_lags)

        # Normalize and step
        gn = np.linalg.norm(subgrad)
        if gn < 1e-15:
            break
        subgrad = subgrad / gn

        # Try multiple step sizes
        improved = False
        for step in [0.0005, 0.001, 0.002, 0.005, 0.01]:
            a_try = a - step * subgrad
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            # Fallback: LP step
            if it % 3 == 0:
                g_lp, _ = lp_solve(a)
                if g_lp is not None:
                    for t in [0.05, 0.1, 0.2]:
                        a_try = (1 - t) * a + t * g_lp
                        a_try = np.maximum(a_try, 0)
                        a_try = a_try / a_try.sum()
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break
            if not improved:
                a = a + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                if it > 50 and it % 30 == 0:
                    # Check if we should give up
                    if compute_obj(a) > best_obj * 1.001:
                        a = best_a.copy()

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_structured_init(n, rng, mode=0):
    """Create structured initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        # Sinc-like (optimal for spectral concentration)
        x = np.linspace(-1, 1, n)
        a = np.maximum(np.sinc(x), 0.001)
        return a / a.sum()
    elif mode == 2:
        # Hamming window
        a = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
        return a / a.sum()
    elif mode == 3:
        # Blackman window
        x = np.arange(n) / (n - 1)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        return a / a.sum()
    elif mode == 4:
        # Gaussian
        x = np.linspace(-3, 3, n)
        a = np.exp(-x ** 2)
        return a / a.sum()
    elif mode == 5:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
        return a / a.sum()
    elif mode == 6:
        # Triangle
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * np.abs(x - 0.5), 0.01)
        return a / a.sum()
    elif mode == 7:
        # Flat top with exponential taper
        taper = 0.1
        a = np.ones(n)
        for i in range(n):
            d = min(i, n - 1 - i)
            if d < taper * n:
                a[i] = np.exp(-((taper * n - d) / (taper * n)) ** 2)
        return a / a.sum()
    elif mode == 8:
        # Multi-lobe
        x = np.arange(n) / n
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.1 * np.cos(2 * np.pi * 4 * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        a = rng.uniform(0.1, 1.0, n)
        return a / a.sum()
