"""
Optimizes non-negative step function coefficients to minimize 2*n*max(conv(a,a))/sum(a)^2.
Key insight: optimal sequences are non-palindromic, with mass skewed to reduce the
maximum self-convolution below the palindromic bound. Uses diverse skewed initializations,
peak-targeted multiplicative weights with exact gradient formulas, coordinate descent
polishing, simulated annealing restarts, and multi-n exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def gradient_lag(a, k):
    """
    Exact gradient of conv(a,a)[k] w.r.t. a.
    Formula: g[i] = 2*a[k-i] if max(0,k-n+1) <= i <= min(k,n-1), else 0.
    """
    n = len(a)
    g = np.zeros(n)
    i_min = max(0, k - n + 1)
    i_max = min(k, n - 1)
    if i_min <= i_max:
        # g[i] = 2 * a[k-i] for i in [i_min, i_max]
        # k-i ranges from k-i_min to k-i_max
        idx = k - np.arange(i_min, i_max + 1)
        g[i_min:i_max + 1] = 2.0 * a[idx]
    return g


def gradient_smoothed(a, conv, tau):
    """
    Smoothed gradient: 2 * sum_k p_k * a[k-i] where p is softmax over conv.
    Computed via FFT for efficiency.
    """
    n = len(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    if shifted.max() > 500:
        shifted = shifted - shifted.max()
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(2 * n - 1)
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    # g[i] = 2 * sum_k p[k] * a[k-i] for valid k-i
    # This is 2 * (p * a_reversed)[i + n - 1] truncated
    # Actually: sum_k p[k]*a[k-i] = (p convolve a_reversed)[n-1+i] ... let me think
    # sum_k p[k] * a[k-i] for k-i in [0,n-1], i in [0,n-1]
    # = cross-correlation of p and a, evaluated at lag i
    # Use FFT: correlate(p, a)
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1
    P = rfft(p, N_fft)
    A = rfft(a, N_fft)
    # cross-correlation: ifft(conj(P) * A) gives sum_k p[k]*a[k+i]... 
    # Actually ifft(conj(P)*A)[m] = sum_k p[k] * a[k+m] (circular)
    # We want sum_k p[k]*a[k-i] = sum_k p[k]*a[k-i]
    # = ifft(conj(A) * P)[i] (circular cross-correlation)
    corr = irfft(np.conj(A) * P, N_fft)
    # corr[i] = sum_k a[k]*p[k+i] = sum_k p[k]*a[k-i] (shifted)
    # For linear: corr[i] for i=0..n-1 gives sum_k p[k]*a[k-i] (with proper bounds)
    g = 2.0 * corr[:n]
    return g


def topk_gradient(a, conv, k_top=5):
    """Gradient using top-k peaks with uniform weights."""
    n = len(a)
    idx = np.argsort(conv)[-k_top:]
    g = np.zeros(n)
    for k in idx:
        g += gradient_lag(a, k)
    g /= k_top
    return g


def blended_gradient(a, conv, tau, k_top=3, alpha=0.3):
    """Blend smoothed gradient with top-k subgradient."""
    g_s = gradient_smoothed(a, conv, tau)
    g_t = topk_gradient(a, conv, k_top)
    return alpha * g_t + (1.0 - alpha) * g_s


def mirror_step(a, grad, eta):
    """Multiplicative weights: a_i <- a_i * exp(-eta * grad_i), normalized."""
    exponent = -eta * grad
    exponent = np.clip(exponent, -40, 40)
    a_new = a * np.exp(exponent)
    s = float(np.sum(a_new))
    if s < 1e-300:
        return a.copy()
    return a_new / s


def mirror_descent(a, time_budget, tau_init=50.0, tau_end=2000.0,
                   k_top=3, alpha=0.3, anneal_temp=0.01, verbose=False):
    """
    Peak-targeted mirror descent with adaptive temperature and simulated annealing.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    it = 0
    stall = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.3:
            break

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Temperature schedule
        tau = tau_init + (tau_end - tau_init) * progress ** 0.8

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        k_star = int(np.argmax(conv))

        # Choose gradient type
        phase = it % 10
        if phase < 3:
            # Exact peak gradient
            g = gradient_lag(a, k_star)
        elif phase < 5:
            # Top-k gradient
            g = topk_gradient(a, conv, k_top=k_top)
        elif phase < 8:
            # Smoothed gradient
            g = gradient_smoothed(a, conv, tau)
        else:
            # Blended
            g = blended_gradient(a, conv, tau, k_top=k_top, alpha=alpha)

        g_max = float(np.max(np.abs(g)))
        if g_max < 1e-15:
            break

        # Adaptive step size
        eta = 1.5 / (g_max + 1e-10)
        # Larger steps early, smaller later
        eta *= (1.2 - 0.5 * progress)

        a_new = mirror_step(a, g, eta)

        # Simulated annealing perturbation
        if it % 50 == 25 and progress < 0.85:
            temp = anneal_temp * (1.0 - progress) * np.sqrt(n)
            a_new = a_new + temp * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            # Try smaller step
            if stall > 5:
                a_half = mirror_step(a, g, eta * 0.3)
                obj_half = compute_obj(a_half)
                if obj_half < best_obj - 1e-14:
                    best_obj = obj_half
                    best_a = a_half.copy()
                    a = a_half
                    stall = 0
                elif stall > 30:
                    # Perturb to escape
                    strength = 0.03 + 0.02 * progress
                    a_pert = a * np.exp(strength * rng.standard_normal(n))
                    a_pert = np.maximum(a_pert, 0)
                    s_p = float(np.sum(a_pert))
                    if s_p > 1e-10:
                        a = a_pert / s_p
                    else:
                        a = best_a.copy()
                    stall = 0

        a = a_new
        it += 1

        if verbose and (it % 200 == 0 or it == 1):
            print(f"      MD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, "
                  f"k*={k_star}, stall={stall}, t={time.monotonic()-start:.0f}s")

    return best_a


def coordinate_descent(a, time_budget, tau=100.0, verbose=False):
    """
    Coordinate descent: iteratively update single coordinates to reduce max conv.
    For coordinate i, conv[k] depends on a[i] linearly (plus quadratic at k=2i).
    We find the optimal a[i] given all other coordinates fixed.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)
    order = np.arange(n)

    improved = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        np.random.default_rng(it_seed := int(time.monotonic() * 1000) % 100000)
        # Shuffle order periodically
        if improved % 50 == 0:
            order = rng.permutation(n)

        # Compute current convolution
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        k_star = int(np.argmax(conv))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            # Try perturbing a[i]
            # conv[k] = sum_j a[j]*a[k-j]
            # If we change a[i] from old_val to new_val, the change in conv[k] is:
            # (new_val - old_val) * (a[k-i] + a[k-i]) for k != 2i
            # (new_val^2 - old_val^2) for k = 2i
            # Actually, the change is:
            # delta * (2*a[k-i]) for k where k-i is valid and k != 2i
            # delta * a[2i-i] + (new_val - old_val)*old_val... 
            # Let's just try a few values and pick the best

            old_val = a[i]
            if old_val < 1e-10:
                # Try small positive values
                candidates = [1e-3, 0.01, 0.05]
            else:
                # Try scaling
                candidates = [old_val * (1 + d) for d in [-0.3, -0.15, -0.05, 0.05, 0.15, 0.3]]
                candidates = [max(0.0, c) for c in candidates]
                if len(candidates) == 0:
                    continue

            best_c = old_val
            best_local = None

            for c in candidates:
                if abs(c - old_val) < 1e-10:
                    continue
                a[i] = c
                # Quick check: recompute conv and check max
                conv_test = compute_conv(a)
                new_max = float(np.max(conv_test))
                if best_local is None or new_max < best_local:
                    best_local = new_max
                    best_c = c

            if best_local is not None and best_local < max_conv - 1e-10:
                a[i] = best_c
                s = float(np.sum(a))
                a = a / s
                improved += 1
                no_improve = 0
            else:
                a[i] = old_val
                no_improve += 1

        if improved > 0:
            obj_now = compute_obj(a)
            if obj_now < best_obj:
                best_obj = obj_now
                best_a = a.copy()

        if no_improve >= n * 3:
            # Try random restart of coordinates
            a = best_a.copy()
            no_improve = 0
            improved = 0
            # Add small noise
            a = a * (1 + 0.01 * rng.standard_normal(n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s

        if verbose and improved % 5 == 0:
            print(f"      CD: imp={improved}, obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")

    # Final recompute
    obj_final = compute_obj(a)
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = a.copy()

    return best_a


def fast_conv_check(a, k_list):
    """Compute convolution at specific lags only (faster than full conv for few lags)."""
    n = len(a)
    results = {}
    for k in k_list:
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        if i_min <= i_max:
            idx = k - np.arange(i_min, i_max + 1)
            results[k] = float(np.dot(a[i_min:i_max + 1], a[idx]))
        else:
            results[k] = 0.0
    return results


def initialize_ramp(n, skew=1.0):
    """Linear ramp initialization (non-palindromic)."""
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = np.power(x, skew)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def initialize_asymmetric_gauss(n, center=0.3, width=0.25):
    """Asymmetric Gaussian centered off-center."""
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = np.exp(-((x - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def initialize_sigmoid(n, sharpness=8.0, center=0.4):
    """Sigmoid-shaped initialization."""
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = 1.0 / (1.0 + np.exp(-sharpness * (x - center)))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def initialize_fourier(n, num_basis=5, rng=None):
    """Fourier-based initialization to create smooth non-palindromic shapes."""
    if rng is None:
        rng = np.random.default_rng(42)
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = np.ones(n)
    for k in range(1, num_basis + 1):
        freq = k * 2.0 * np.pi
        amp = rng.uniform(0.1, 0.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(freq * x + phase)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def lp_polish(a, time_budget, max_iters=50):
    """
    Simple LP-based direction finding: minimize max conv(a, g) over g >= 0, sum(g)=1.
    Then move towards g.
    Uses scipy.optimize for the LP.
    """
    from scipy.optimize import linprog

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start >= time_budget:
            break

        conv_aa = compute_conv(a)

        # Build A matrix: A[k,j] = a[k-j] if 0<=k-j<n else 0
        # We want to minimize M s.t. 2*A@g <= M + conv_aa, sum(g)=1, g>=0
        # This is: min M s.t. 2*A@g - M <= conv_aa, sum(g)=1, g>=0
        
        # For efficiency, only use constraints for top peaks
        top_k = np.argsort(conv_aa)[-min(10, 2*n-1):]
        
        # Number of constraints: len(top_k) + 1 (sum constraint)
        # Variables: n (for g) + 1 (for M)
        # c = [0,...,0, 1] (minimize M)
        m = len(top_k)
        c = np.zeros(n + 1)
        c[-1] = 1.0  # minimize M

        # Constraints: 2*A_subset@g - M <= conv_aa[top_k]
        # => 2*A_subset@g - M - conv_aa[top_k] <= 0
        A_ub = np.zeros((m, n + 1))
        b_ub = np.zeros(m)
        for idx, k in enumerate(top_k):
            i_min = max(0, k - n + 1)
            i_max = min(k, n - 1)
            if i_min <= i_max:
                idx_arr = k - np.arange(i_min, i_max + 1)
                A_ub[idx, i_min:i_max+1] = 2.0 * a[idx_arr]
            A_ub[idx, -1] = -1.0
            b_ub[idx] = conv_aa[k]

        # Equality: sum(g) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n + [(None, None)]  # g >= 0, M free

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status == 0:
                g = np.maximum(res.x[:n], 0)
                gs = float(np.sum(g))
                if gs > 1e-10:
                    g = g / gs
                    # Line search
                    best_t = 0.0
                    best_f = best_obj
                    for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
                        a_try = (1 - t) * a + t * g
                        a_try = np.maximum(a_try, 0)
                        st = float(np.sum(a_try))
                        if st > 1e-10:
                            f_try = compute_obj(a_try / st)
                            if f_try < best_f:
                                best_f = f_try
                                best_t = t
                    if best_t > 0:
                        a_new = (1 - best_t) * a + best_t * g
                        a_new = np.maximum(a_new, 0)
                        s_new = float(np.sum(a_new))
                        if s_new > 1e-10:
                            a_new = a_new / s_new
                            if best_f < best_obj - 1e-12:
                                best_obj = best_f
                                best_a = a_new.copy()
                                a = a_new
                            else:
                                a = a_new
                else:
                    break
            else:
                break
        except Exception:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: diverse non-palindromic initializations + peak-targeted mirror descent
    + coordinate descent + LP polish + multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    # === Phase 1: Diverse Initialization (200s) ===
    print(f"\n=== Phase 1: Diverse Initialization ===")
    p1_budget = min(200.0, (deadline - time.monotonic()) * 0.15)

    n_base = 800
    if best_seq is not None:
        n_base = len(best_seq)

    init_strategies = []

    # Ramp with different skews
    for skew in [0.5, 1.0, 1.5, 2.0, 3.0]:
        init_strategies.append(('ramp', skew))

    # Asymmetric Gaussians
    for center in [0.2, 0.3, 0.4, 0.6, 0.7, 0.8]:
        for width in [0.15, 0.25, 0.35]:
            init_strategies.append(('gauss', (center, width)))

    # Sigmoid
    for sharp in [4, 8, 12, 16]:
        for center in [0.3, 0.4, 0.6]:
            init_strategies.append(('sigmoid', (sharp, center)))

    # Fourier
    for fb in range(3):
        init_strategies.append(('fourier', fb))

    # Uniform (baseline)
    init_strategies.append(('uniform', 0))

    # If we have a parent, also try perturbed versions
    if best_seq is not None:
        for amp in [0.05, 0.1, 0.2]:
            init_strategies.append(('perturb', amp))

    # Budget per initialization
    per_init_budget = p1_budget / len(init_strategies)
    per_init_budget = max(per_init_budget, 2.0)

    candidates = []

    for name, params in init_strategies:
        elapsed = time.monotonic() - start_time
        if deadline - elapsed < 30:
            break

        if name == 'ramp':
            a_init = initialize_ramp(n_base, skew=params)
        elif name == 'gauss':
            a_init = initialize_asymmetric_gauss(n_base, center=params[0], width=params[1])
        elif name == 'sigmoid':
            a_init = initialize_sigmoid(n_base, sharpness=params[0], center=params[1])
        elif name == 'fourier':
            a_init = initialize_fourier(n_base, num_basis=3 + params, rng=rng)
        elif name == 'uniform':
            a_init = np.ones(n_base) / n_base
        elif name == 'perturb':
            a_init = best_seq * (1 + params * rng.standard_normal(n_base))
            a_init = np.maximum(a_init, 0)
            s = float(np.sum(a_init))
            if s > 1e-10:
                a_init = a_init / s
            else:
                a_init = np.ones(n_base) / n_base
        else:
            continue

        # Quick mirror descent
        alloc = min(per_init_budget, deadline - time.monotonic() - 15)
        if alloc > 3:
            a_opt = mirror_descent(a_init, time_budget=alloc,
                                   tau_init=30.0, tau_end=500.0,
                                   k_top=3, alpha=0.3, verbose=False)
        else:
            a_opt = a_init

        obj = compute_obj(a_opt)
        candidates.append((obj, a_opt, name))

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  {name}({params}): obj={obj:.8f} **")

    # Sort candidates and keep top few
    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1 done. Best so far: {best_score:.8f}")
    print(f"  Top 5: {[f'{c[2]}:{c[0]:.6f}' for c in candidates[:5]]}")

    # === Phase 2: Intensive Mirror Descent (400s) ===
    print(f"\n=== Phase 2: Intensive Mirror Descent ===")
    p2_budget = min(400.0, (deadline - time.monotonic()) * 0.3)

    # Optimize the best candidate
    a_curr = best_seq.copy()
    if p2_budget > 20:
        a_curr = mirror_descent(a_curr, time_budget=p2_budget * 0.6,
                                tau_init=50.0, tau_end=3000.0,
                                k_top=5, alpha=0.25, anneal_temp=0.015, verbose=True)
        obj = compute_obj(a_curr)
        if obj < best_score:
            best_score = obj
            best_seq = a_curr.copy()
            print(f"  MD main: obj={best_score:.8f} **")

    # Also try second-best candidate
    remaining = deadline - time.monotonic()
    if len(candidates) > 1 and remaining > 60:
        a_second = candidates[1][1].copy()
        alloc = min(remaining * 0.3, 100.0)
        a_second = mirror_descent(a_second, time_budget=alloc,
                                  tau_init=50.0, tau_end=2000.0,
                                  k_top=4, alpha=0.3, verbose=False)
        obj2 = compute_obj(a_second)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a_second.copy()
            print(f"  MD second: obj={best_score:.8f} **")

    # === Phase 3: Coordinate Descent (200s) ===
    print(f"\n=== Phase 3: Coordinate Descent ===")
    p3_budget = min(200.0, (deadline - time.monotonic()) * 0.2)
    if p3_budget > 20:
        a_cd = coordinate_descent(best_seq, time_budget=p3_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 4: LP Polish (150s) ===
    print(f"\n=== Phase 4: LP Polish ===")
    p4_budget = min(150.0, (deadline - time.monotonic()) * 0.15)
    if p4_budget > 20:
        a_lp = lp_polish(best_seq, time_budget=p4_budget, max_iters=100)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 5: Second MD Round with different params (150s) ===
    print(f"\n=== Phase 5: Second MD Round ===")
    p5_budget = min(150.0, (deadline - time.monotonic()) * 0.15)
    if p5_budget > 20:
        # Higher tau for finer optimization
        a_md2 = mirror_descent(best_seq, time_budget=p5_budget,
                               tau_init=200.0, tau_end=5000.0,
                               k_top=2, alpha=0.5, anneal_temp=0.005, verbose=False)
        obj_md2 = compute_obj(a_md2)
        if obj_md2 < best_score:
            best_score = obj_md2
            best_seq = a_md2.copy()
            print(f"  MD2: obj={best_score:.8f} **")

    # === Phase 6: Multi-n Exploration (200s) ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    p6_budget = (deadline - time.monotonic()) * 0.2
    n_best = len(best_seq)

    n_options = []
    for delta in [-200, -100, 100, 200, 300, 400, 500, 700, 1000]:
        n_try = n_best + delta
        if 500 <= n_try <= 3000:
            n_options.append(n_try)
    for n_abs in [600, 800, 1000, 1200, 1500, 2000, 2500]:
        if n_abs not in n_options:
            n_options.append(n_abs)
    n_options = sorted(set(n_options))[:6]
    print(f"  Trying n values: {n_options}")

    per_n_budget = p6_budget / len(n_options) if n_options else 0
    per_n_budget = max(per_n_budget, 15.0)

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30 or n_try == n_best:
            continue

        a_try = resample(best_seq, n_try)

        # Also try fresh initialization at this n
        a_fresh = initialize_ramp(n_try, skew=1.5)

        # Optimize both
        alloc_md = min(per_n_budget * 0.5, remaining * 0.4 - 5)
        if alloc_md > 5:
            a_try = mirror_descent(a_try, time_budget=alloc_md,
                                   tau_init=50.0, tau_end=1000.0, verbose=False)
        remaining = deadline - time.monotonic()
        alloc_md2 = min(per_n_budget * 0.3, remaining * 0.3 - 5)
        if alloc_md2 > 5:
            a_fresh = mirror_descent(a_fresh, time_budget=alloc_md2,
                                     tau_init=50.0, tau_end=800.0, verbose=False)

        # CD polish
        remaining = deadline - time.monotonic()
        alloc_cd = min(20.0, remaining * 0.2 - 3)
        if alloc_cd > 5:
            a_try = coordinate_descent(a_try, time_budget=alloc_cd)

        obj_try = compute_obj(a_try)
        obj_fresh = compute_obj(a_fresh)
        best_n_obj = min(obj_try, obj_fresh)
        best_n_seq = a_try if obj_try <= obj_fresh else a_fresh

        if best_n_obj < best_score:
            best_score = best_n_obj
            best_seq = best_n_seq.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={best_n_obj:.8f}")

    # === Phase 7: Final Intensive Polish (remaining time) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    print(f"  Time remaining: {remaining:.0f}s")

    if remaining > 30:
        # Final high-tau MD
        alloc = min(remaining * 0.4, 120.0)
        a_final = mirror_descent(best_seq, time_budget=alloc,
                                 tau_init=500.0, tau_end=8000.0,
                                 k_top=2, alpha=0.4, anneal_temp=0.003, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # Final CD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 80.0)
        a_cd_final = coordinate_descent(best_seq, time_budget=alloc, verbose=False)
        obj_cd_f = compute_obj(a_cd_final)
        if obj_cd_f < best_score:
            best_score = obj_cd_f
            best_seq = a_cd_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final tiny MD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 50.0)
        a_md_final = mirror_descent(best_seq, time_budget=alloc,
                                    tau_init=1000.0, tau_end=10000.0,
                                    k_top=1, alpha=0.6, verbose=False)
        obj_md_f = compute_obj(a_md_final)
        if obj_md_f < best_score:
            best_score = obj_md_f
            best_seq = a_md_final.copy()
            print(f"  Final MD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
