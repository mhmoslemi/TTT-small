import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_rows(a):
    n = len(a)
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            row[j] = a[k - j]
        rows.append(row)
    return np.array(rows)


def build_conv_matrix_sparse(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_lo = j
        k_hi = min(2 * n - 2, j + n - 1)
        A[k_lo:k_hi + 1, j] = a[n - 1 - (k_hi - j):-1 if n - 1 - (k_hi - j) < 0 else None] if False else \
            a[(n - 1 - (k_hi - j)):(n - 1 - (k_lo - j)) + 1][::-1]
    return A


def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(2 * n - 2, j + n - 1)
        A[lo:hi + 1, j] = a[n - 1 - (hi - j): n - 1 - (lo - j) + 1][::-1]
    return A


def solve_lp(a, time_limit=5.0):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None
    a_norm = a / s
    A = build_A(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > time_limit:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_g = float(np.sum(g_val))
                if s_g > 1e-10:
                    return g_val / s_g, float(M.value)
        except Exception:
            continue
    return None, None


def golden_section_search(a, g, f, tol=1e-8, max_iter=60):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.5
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(a, c)
    fd = f(a, d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(a, c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(a, d)
    t_best = (lo + hi) / 2.0
    return t_best, f(a, t_best)


def line_search_obj(a, g, t):
    a_try = (1.0 - t) * a + t * g
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 0.01:
        return 1e10
    return compute_obj(a_try / s)


def antisymmetric_perturb(a, rng, magnitude=0.05):
    n = len(a)
    a_s = a.copy()
    # Add anti-symmetric component: a[i] + eps * (i/n - 0.5)
    idx = np.arange(n, dtype=np.float64)
    asym = (idx / n - 0.5)
    # Make it zero-sum
    asym = asym - np.mean(asym)
    a_pert = a_s * (1.0 + magnitude * asym)
    a_pert = np.maximum(a_pert, 0)
    s = float(np.sum(a_pert))
    if s < 0.01:
        return a_s
    return a_pert / s


def cosine_perturb(a, rng, magnitude=0.03):
    n = len(a)
    freq = rng.integers(1, max(2, n // 60))
    phase = rng.uniform(0, 2 * np.pi)
    idx = np.arange(n, dtype=np.float64)
    pert = magnitude * np.cos(2 * np.pi * freq * idx / n + phase)
    a_pert = a * np.exp(pert)
    a_pert = np.maximum(a_pert, 0)
    s = float(np.sum(a_pert))
    if s < 0.01:
        return a
    return a_pert / s


def active_set_perturb(a, rng, magnitude=0.05):
    n = len(a)
    conv = compute_conv(a)
    max_val = np.max(conv)
    # Find coordinates that contribute most to the peak
    k_max = int(np.argmax(conv))
    contrib = np.zeros(n)
    for i in range(n):
        j1 = k_max - i
        j2 = i - k_max
        if 0 <= j1 < n:
            contrib[i] += a[j1]
        if 0 <= j2 < n:
            contrib[i] += a[j2]
    # Perturb coordinates with low contribution more
    norm_contrib = contrib / (np.max(contrib) + 1e-12)
    scale = 1.0 + magnitude * (1.0 - norm_contrib) * (rng.uniform(-1, 1, n))
    a_pert = a * scale
    a_pert = np.maximum(a_pert, 0)
    s = float(np.sum(a_pert))
    if s < 0.01:
        return a
    return a_pert / s


def parametric_init(n, mode, rng):
    x = np.linspace(0, 1, n)
    if mode == 0:
        # Offset Gaussian
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.05, 0.3)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    elif mode == 1:
        # Sigmoid
        steep = rng.uniform(3, 15)
        center = rng.uniform(0.3, 0.7)
        a = 1.0 / (1.0 + np.exp(steep * (x - center)))
    elif mode == 2:
        # Skewed exponential
        rate = rng.uniform(2, 10)
        center = rng.uniform(0.3, 0.7)
        a = np.exp(-rate * (x - center)) * (x - center > 0) + np.exp(-rate * 0.3 * (x - center)) * (x - center <= 0)
    elif mode == 3:
        # Power law from one end
        alpha = rng.uniform(1.5, 5.0)
        a = (1.0 - x) ** alpha + 0.01
    elif mode == 4:
        # Two asymmetric bumps
        c1 = rng.uniform(0.1, 0.3)
        c2 = rng.uniform(0.6, 0.9)
        w1 = rng.uniform(0.03, 0.1)
        w2 = rng.uniform(0.03, 0.1)
        h1 = rng.uniform(0.5, 1.0)
        h2 = rng.uniform(0.3, 0.8)
        a = h1 * np.exp(-((x - c1) ** 2) / (2 * w1 ** 2)) + h2 * np.exp(-((x - c2) ** 2) / (2 * w2 ** 2)) + 0.005
    elif mode == 5:
        # Stretched U-shape (asymmetric)
        alpha = rng.uniform(2, 8)
        beta = rng.uniform(2, 8)
        a = x ** alpha + (1 - x) ** beta + 0.01
    elif mode == 6:
        # Cosine-modulated
        freq = rng.integers(2, 6)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * freq * x + phase)
    else:
        # Random smooth
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(rng.integers(3, 11)) / 5.0
        a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 0.01:
        return np.ones(n) / n
    return a / s


def coordinate_local_search(a, time_budget=30.0):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 0.01:
        return a, compute_obj(a)
    a = a / s
    best_obj = compute_obj(a)
    best_a = a.copy()
    start = time.monotonic()
    conv = compute_conv(a)
    max_val = np.max(conv)

    step_sizes = [0.1, 0.05, 0.01, 0.005]
    
    while time.monotonic() - start < time_budget:
        improved = False
        for step in step_sizes:
            for i in range(0, n, max(1, n // 200)):
                if time.monotonic() - start > time_budget:
                    break
                # Try increasing
                a_new = a.copy()
                a_new[i] += step
                s_new = float(np.sum(a_new))
                if s_new < 0.01:
                    continue
                a_new = a_new / s_new
                obj = compute_obj(a_new)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a_new
                    a = a_new
                    conv = compute_conv(a)
                    improved = True
                    break
                # Try decreasing
                a_new = a.copy()
                if a[i] > step:
                    a_new[i] -= step
                    s_new = float(np.sum(a_new))
                    if s_new < 0.01:
                        continue
                    a_new = a_new / s_new
                    obj = compute_obj(a_new)
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_a = a_new
                        a = a_new
                        conv = compute_conv(a)
                        improved = True
                        break
            if improved:
                break
        if not improved:
            break
    return best_a, best_obj


def lp_polish(a, time_budget=100.0, max_iters=200, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(12345 + n)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        t0 = time.monotonic()
        g_lp, M_lp = solve_lp(a, time_limit=min(8.0, remaining * 0.5))
        if g_lp is None:
            break
        t_opt, obj_opt = golden_section_search(a, g_lp, line_search_obj, max_iter=40)
        if t_opt > 1e-6 and obj_opt < best_obj - 1e-14:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1
            # Try t=1 directly
            obj_direct = compute_obj(g_lp)
            if obj_direct < best_obj:
                best_obj = obj_direct
                best_a = g_lp.copy()
                a = g_lp.copy()
                stall = 0
        it += 1
        elapsed = time.monotonic() - start
        if verbose and it % 10 == 0:
            print(f"      LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, elapsed={elapsed:.1f}s")
        if stall >= 3:
            # Apply perturbation
            pert_type = it % 4
            if pert_type == 0:
                a = antisymmetric_perturb(a, rng, magnitude=0.03 * stall)
            elif pert_type == 1:
                a = cosine_perturb(a, rng, magnitude=0.02 * stall)
            elif pert_type == 2:
                a = active_set_perturb(a, rng, magnitude=0.04 * stall)
            else:
                # Blend with random parametric
                a_rand = parametric_init(n, rng.integers(0, 8), rng)
                a = 0.8 * best_a + 0.2 * a_rand
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 0.01:
                    a = a / s
                else:
                    a = best_a.copy()
            stall = 0
            if verbose:
                print(f"      Perturbed (type={pert_type}, stall was {stall+3})")
    return best_a, best_obj


def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s


def subgradient_polish(a, time_budget=30.0):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    a_avg = a.copy()
    moment = np.zeros(n)
    beta = 0.7
    lr0 = 0.03
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a_avg)
        # Use top-3 active constraints for better gradient
        sorted_idx = np.argsort(conv)[::-1]
        top_k = 3
        grad = np.zeros(n)
        for k_max in sorted_idx[:top_k]:
            for i in range(n):
                j1 = k_max - i
                j2 = i - k_max
                if 0 <= j1 < n:
                    grad[i] += a_avg[j1]
                if 0 <= j2 < n:
                    grad[i] += a_avg[j2]
        grad /= top_k
        lr_t = lr0 / (1.0 + 0.01 * it)
        moment = beta * moment - lr_t * grad
        a_new = a_avg + moment
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = best_a.copy()
            moment = np.zeros(n)
        else:
            a_new = a_new / s
        a_avg = (it / (it + 1.0)) * a_avg + (1.0 / (it + 1.0)) * a_new
        obj = compute_obj(a_avg)
        if obj < best_obj:
            best_obj = obj
            best_a = a_avg.copy()
            best_it = it
        it += 1
        if it - best_it > 5000:
            a_avg = best_a.copy()
            moment = np.zeros(n)
            lr0 = 0.01
            best_it = it
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        s = float(np.sum(parent_arr))
        if s > 0.01:
            best_seq = parent_arr / s
            best_score = compute_obj(best_seq)
            print(f"Parent loaded: n={parent_arr.size}, obj={best_score:.8f}")
        else:
            raise ValueError("Sum too small")
    except Exception:
        best_seq = parametric_init(800, 0, rng)
        best_score = compute_obj(best_seq)
        print(f"Default init: n={len(best_seq)}, obj={best_score:.8f}")

    n_best = len(best_seq)

    # Try alternative parametric inits at same n
    for mode in range(8):
        a_alt = parametric_init(n_best, mode, rng)
        obj_alt = compute_obj(a_alt)
        if obj_alt < best_score:
            best_score = obj_alt
            best_seq = a_alt.copy()
            print(f"  Param init mode={mode}: obj={best_score:.8f} **")

    # === Phase 1: Parametric family search (quick) ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 1: Parametric Search (n={n_best}) ===")
        for i in range(20):
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            mode = i % 8
            a_p = parametric_init(n_best, mode, rng)
            obj_p = compute_obj(a_p)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Param {i} (mode={mode}): obj={best_score:.8f} **")

    # === Phase 2: Intensive LP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 2: LP Polish (n={n_best}, budget={min(remaining*0.35, 400):.0f}s) ===")
        alloc = min(remaining * 0.35, 400.0)
        a_lp, obj_lp = lp_polish(best_seq, time_budget=alloc, max_iters=300, verbose=True)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase 2 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 done: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Coordinate local search ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 3: Coordinate Local Search ===")
        alloc = min(remaining * 0.08, 40.0)
        a_cl, obj_cl = coordinate_local_search(best_seq, time_budget=alloc)
        if obj_cl < best_score:
            best_score = obj_cl
            best_seq = a_cl.copy()
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj_cl:.8f} (no improvement)")

    # === Phase 4: n exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 4: n Exploration ===")
        n_cur = len(best_seq)
        n_options = []
        for offset in [-100, -50, -25, 25, 50, 100, 200, 300]:
            n_try = n_cur + offset
            if 300 <= n_try <= 2500:
                n_options.append(n_try)
        # Also try some fixed values
        for nv in [500, 600, 700, 900, 1000, 1200, 1500]:
            if nv not in n_options and abs(nv - n_cur) >= 50:
                n_options.append(nv)
        n_options = sorted(set(n_options), key=lambda x: abs(x - n_cur))[:8]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            a_try = interpolate_sequence(best_seq, n_try)
            if a_try is None:
                continue
            # Quick polish
            alloc = min(remaining * 0.25, 60.0)
            a_try2, obj_try = lp_polish(a_try, time_budget=alloc, max_iters=50, verbose=False)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try2.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts with LP ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 5: Diverse Restarts ===")
        n_cur = len(best_seq)
        for r in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            mode = (r * 3 + 1) % 8
            a_r = parametric_init(n_cur, mode, rng)
            blend = 0.4 + 0.1 * r
            a_r = (1.0 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            s = float(np.sum(a_r))
            if s > 0.01:
                a_r = a_r / s
            else:
                a_r = best_seq.copy()
            alloc = min(remaining * 0.25, 50.0)
            if alloc > 15:
                a_r2, obj_r = lp_polish(a_r, time_budget=alloc, max_iters=40, verbose=False)
            else:
                a_r2 = a_r
                obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r2.copy()
                print(f"  Restart {r} (mode={mode}, blend={blend:.1f}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Subgradient polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 6: Subgradient Polish ===")
        alloc = min(remaining * 0.15, 60.0)
        a_sg, obj_sg = subgradient_polish(best_seq, time_budget=alloc)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Phase 6: obj={best_score:.8f} **")
        else:
            print(f"  Phase 6: obj={obj_sg:.8f} (no improvement)")

    # === Phase 7: Final intensive LP ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 7: Final LP Polish ===")
        alloc = min(remaining - 10, 200.0)
        a_final, obj_final = lp_polish(best_seq, time_budget=alloc, max_iters=200, verbose=True)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Phase 7: obj={best_score:.8f} **")
        else:
            print(f"  Phase 7: obj={obj_final:.8f} (no improvement)")

    # === Phase 8: Final coordinate search + micro-LP ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 8: Final Squeeze ===")
        a_cf, obj_cf = coordinate_local_search(best_seq, time_budget=min(remaining * 0.3, 20.0))
        if obj_cf < best_score:
            best_score = obj_cf
            best_seq = a_cf.copy()
            print(f"  Coord: obj={best_score:.8f} **")
        remaining2 = deadline - time.monotonic()
        if remaining2 > 15:
            a_sf, obj_sf = lp_polish(best_seq, time_budget=min(remaining2 - 5, 40.0), max_iters=30, verbose=False)
            if obj_sf < best_score:
                best_score = obj_sf
                best_seq = a_sf.copy()
                print(f"  Squeeze: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
