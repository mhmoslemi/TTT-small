import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.ndimage import uniform_filter1d


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def gradient_from_weights(a, weights):
    """
    Compute gradient of sum_k weights[k]*conv(a,a)[k] with respect to a.
    Uses: grad[i] = 2 * sum_m a[m] * weights[m+i]
    Computed via cross-correlation using FFT.
    """
    n = len(a)
    b = a[::-1]
    N_fft = 1
    while N_fft < 3 * n - 2:
        N_fft <<= 1
    c = irfft(rfft(b, N_fft) * rfft(weights, N_fft), N_fft)
    return 2.0 * c[n - 1:2 * n - 1]


def lse_weights(conv, tau):
    """Compute softmax weights on convolution values with temperature tau."""
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
        return p
    return exp_vals / sum_exp

def spectral_init(n, seed=42):
    """
    Initialize using spectral approach: create a sequence whose power spectrum
    is as flat as possible (like a CAZAC sequence but real and non-negative).
    Use a windowed cosine as starting point.
    """
    rng = np.random.default_rng(seed)
    # Start with a raised cosine (Hanning window)
    x = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    # Add small random perturbation to break symmetry
    a = a * (1 + 0.1 * rng.standard_normal(n))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def symmetric_init(n, seed=42):
    """Initialize with a symmetric bump."""
    center = n / 2.0
    x = np.arange(n)
    sigma = n / 4.0
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    rng = np.random.default_rng(seed)
    a = a * (1 + 0.05 * rng.standard_normal(n))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        alloc_cd = min(remaining * 0.15, 25.0)
        if alloc_cd > 5:
            a_try = coordinate_descent(a_try, n_sweeps=2, time_budget=alloc_cd, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse Restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    restarts = []

    # Symmetric mirror
    a_r1 = (best_seq + best_seq[::-1]) / 2.0
    a_r1 = np.maximum(a_r1, 0)
    s_r = float(np.sum(a_r1))
    if s_r > 1e-10:
        a_r1 = a_r1 / s_r
    restarts.append(("mirror", a_r1))

    # Fourier perturbation
    pert = np.zeros(n_cur)
    for k in range(1, 8):
        freq = k * 2.0 * np.pi / n_cur
        amp = 0.03 * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n_cur) + phase)
    a_r2 = best_seq * np.exp(pert)
    a_r2 = np.maximum(a_r2, 0)
    s_r = float(np.sum(a_r2))
    if s_r > 1e-10:
        a_r2 = a_r2 / s_r
    restarts.append(("fourier", a_r2))

    # Smooth blend
    ks = max(5, n_cur // 10)
    if ks % 2 == 0:
        ks += 1
    a_smooth = uniform_filter1d(best_seq, size=ks)
    a_r3 = 0.6 * best_seq + 0.4 * a_smooth
    a_r3 = np.maximum(a_r3, 0)
    s_r = float(np.sum(a_r3))
    if s_r > 1e-10:
        a_r3 = a_r3 / s_r
    restarts.append(("smooth", a_r3))

    # Random multiplicative perturbation
    a_r4 = best_seq * np.exp(0.05 * rng.standard_normal(n_cur))
    a_r4 = np.maximum(a_r4, 0)
    s_r = float(np.sum(a_r4))
    if s_r > 1e-10:
        a_r4 = a_r4 / s_r
    restarts.append(("rand_perturb", a_r4))

    for name, a_r in restarts:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        alloc_fista = min(remaining * 0.25, 45.0)
        if alloc_fista > 10:
            a_r, obj_r = fista_optimize(a_r, time_budget=alloc_fista,
                                        tau_init=100.0, tau_end=3000.0, verbose=False)
            # Quick CD
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_r = coordinate_descent(a_r, n_sweeps=1,
                                         time_budget=min(remaining * 0.2, 15.0), verbose=False)
                obj_r = compute_obj(a_r)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart ({name}): obj={best_score:.8f} **")
        else:
            print(f"  Restart ({name}): obj={obj_r:.8f}")

    # === Phase 6: Final Intensive Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau FISTA
    if remaining > 30:
        alloc = min(remaining * 0.35, 120.0)
        a_final, obj_final = fista_optimize(best_seq, time_budget=alloc,
                                            tau_init=500.0, tau_end=10000.0,
                                            verbose=True)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final FISTA: obj={best_score:.8f} **")

    # Coordinate descent final
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 60.0)
        a_cd = coordinate_descent(best_seq, n_sweeps=5, time_budget=alloc, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Subgradient final
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 40.0)
        a_sg = _subgradient_polish(best_seq, time_budget=alloc)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # === Final CD sweep (always do this) ===
    remaining = deadline - time.monotonic()
    if remaining > 5:
        a_cd = coordinate_descent(best_seq, n_sweeps=2, time_budget=min(remaining, 30.0), verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
