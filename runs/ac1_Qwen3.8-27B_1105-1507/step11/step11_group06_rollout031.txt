Softmax-smoothed L-BFGS optimization with adaptive beta annealing,
LP refinement, and aggressive n exploration. Uses FFT-based gradients
for fast convergence, complemented by LP-based local refinement.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_softmax_obj_and_grad(z, beta):
    """Compute softmax-smoothed objective and its gradient w.r.t. z.
    z is the log-parameterization: a = softmax(z)."""
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a = a / a.sum()
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = conv - max_conv
    exp_b = np.exp(beta * shifted)
    sum_exp = float(np.sum(exp_b))
    weights = exp_b / sum_exp
    
    # Objective: 2n * (max_conv + log(sum_exp)) / beta
    obj = 2.0 * n * (max_conv + np.log(sum_exp)) / beta
    
    # Gradient of sum_k weights[k] * conv[k] w.r.t. a:
    # grad_a[i] = sum_k weights[k] * a[k-i] (where valid)
    # = convolve(weights, a[::-1], 'full')[n-1:n-1+n]
    size = 1
    while size < (2 * n - 1) + n - 1:
        size <<= 1
    W = rfft(weights, size)
    A_rev = rfft(a[::-1], size)
    conv_grad = irfft(W * A_rev, size)
    grad_a = conv_grad[n - 1:n - 1 + n]
    
    # Chain rule for softmax: d a[i] / d z[j] = a[i](delta_ij - a[j])
    # d obj / d z[j] = sum_i (d obj / d a[i]) * a[i] * (delta(i,j) - a[j])
    # = a[j] * (grad_a[j] * 2n/beta - sum_i grad_a[i] * a[i] * 2n/beta)
    # Wait, let me be more careful.
    # f(a) = 2n/beta * (max_conv + log(sum_exp))
    # But max_conv is not differentiable... I'm using the softmax approximation.
    # Actually, the softmax approximation IS: (1/beta) * log(sum(exp(beta * conv)))
    # So f(a) = 2n/beta * log(sum(exp(beta * conv)))
    # d f / d a[i] = 2n/beta * sum_k (exp(beta*conv[k])/sum(exp)) * d conv[k]/d a[i]
    # = 2n/beta * sum_k weights[k] * a[k-i]
    # = 2n/beta * grad_a[i]
    
    # Then d f / d z[j] = sum_i (d f / d a[i]) * (d a[i] / d z[j])
    # = sum_i (2n/beta * grad_a[i]) * a[i] * (delta(i,j) - a[j])
    # = (2n/beta) * (a[j] * grad_a[j] - a[j] * sum_i grad_a[i]*a[i])
    # = (2n/beta) * a[j] * (grad_a[j] - dot(grad_a, a))
    
    factor = 2.0 * n / beta
    grad_z = factor * a * (grad_a - np.dot(grad_a, a))
    
    return obj, grad_z


def softmax_lbfgs(a_init, beta_sequence, time_budget, verbose=False):
    """Run L-BFGS with a sequence of betas (annealing schedule)."""
    start = time.monotonic()
    n = len(a_init)
    best_a = np.maximum(a_init, 0).copy()
    best_obj = compute_obj(best_a)
    
    for beta in beta_sequence:
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        z0 = np.log(best_a + 1e-15)
        
        # Run L-BFGS
        result = minimize(
            lambda z: compute_softmax_obj_and_grad(z, beta)[0],
            z0,
            jac=lambda z: compute_softmax_obj_and_grad(z, beta)[1],
            method='L-BFGS-B',
            options={'maxiter': 200, 'gtol': 1e-10, 'maxfun': 1000,
                     'ftol': 1e-12}
        )
        
        if result.x is not None:
            z_opt = result.x
            a_opt = np.exp(z_opt - z_opt.max())
            a_opt = a_opt / a_opt.sum()
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
                if verbose:
                    print(f"    beta={beta}: obj={best_obj:.8f}")
    
    elapsed = time.monotonic() - start
    if verbose:
        print(f"  Softmax L-BFGS: obj={best_obj:.8f}, time={elapsed:.1f}s")
    return best_a, best_obj

For i != j, |i-j| = d: B[i,j] = alpha_d * 1/2 (only Q_d contributes)

So B is a **symmetric Toeplitz matrix** (or rather, banded with the (i,j) entry depending only on |i-j|)!

This means we can compute B @ v in O(n log n) using FFT!

`B @ v` where B[i,j] = alpha_0 if i=j, alpha_{|i-j|}/2 if i≠j.

Let me define: for d = 0, 1, ..., n-1, let `c[d] = alpha_d` for d=0, and `c[d] = alpha_d / 2` for d > 0.

Then `(B @ v)[i] = c[0] * v[i] + sum_{d=1}^{n-1} c[d] * (v[i-d] + v[i+d])` (with out-of-bounds = 0).

This is a convolution! Specifically:
`(B @ v)[i] = sum_d c[d] * (v[i-d] + v[i+d]) - c[0]*v[i]` (to avoid double-counting d=0)
`= sum_d c[d] * v[i-d] + sum_d c[d] * v[i+d] - c[0]*v[i]`

The first sum is a convolution of v with c (reversed): `(c_rev * v)[i]` where c_rev[d] = c[d].
The second sum is a correlation: `corr(c, v)[i]`.

Actually, since c is symmetric (c[d] depends on |d|), this is:
`(B @ v)[i] = sum_{d=-(n-1)}^{n-1} c[|d|] * v[i+d]`

Where c[|d|] = alpha_{|d|} for d=0, and alpha_{|d|}/2 for d≠0.

Wait, I need to be more careful. Let me define:
`B[i,j] = f(i-j)` where `f(0) = alpha_0` and `f(d) = alpha_{|d|} / 2` for d ≠ 0.

Then `(B @ v)[i] = sum_j f(i-j) * v[j] = sum_d f(d) * v[i-d]`

This is the convolution of f (extended to be zero outside [-n+1, n-1]) with v.

Using FFT: compute it in O(n log n).

So the MM algorithm becomes:
1. Compute conv(a,a)
2. Compute alpha = softmax(beta * (conv - max_conv)) / sum(...)
3. Form the Toeplitz symbol f (which depends on alpha)
4. Compute B @ a via FFT
5. Take a gradient step: a_new = a - lr * B @ a
6. Project onto simplex
7. Repeat

This is O(n log n) per iteration and is guaranteed to decrease the upper bound! Much more efficient than LP.

Let me also think about the step size. For the QP `min a^T B a s.t. a >= 0, sum(a) = 1`, a good step size can be found by line search or by using the eigenvalue information of B.

Actually, for the MM framework, we want to find the minimizer of the upper bound, not just take a gradient step. The minimizer of `a^T B a` on the simplex can be found by solving the KKT conditions, which for a Toeplitz matrix can be done efficiently.

But a simpler approach: use an over-relaxed projection step.

Let me also consider: the QP `min a^T B a s.t. a in simplex` has a unique solution (B is positive definite if alpha is a probability distribution with enough support). The solution satisfies:
`2 * B @ a* = lambda * 1 + mu` where mu >= 0 is the KKT multiplier for a* >= 0, and mu[i] = 0 if a*[i] > 0.

So for the support S = {i : a*[i] > 0}: the solution is `a*[S] = (B[S,S])^{-1} * (lambda/2) * 1[S]` where lambda is chosen so sum(a*) = 1.

This is: `a*[S] = (B[S,S])^{-1} 1 / (1^T (B[S,S])^{-1} 1) * (1/2)` ... hmm, let me redo.

`2 B @ a = lambda * 1 + mu`
On the support S: `2 B[S,S] @ a[S] = lambda * 1`
So `a[S] = lambda/2 * (B[S,S])^{-1} 1`
And `sum(a[S]) = 1`, so `lambda/2 * 1^T (B[S,S])^{-1} 1 = 1`
Thus `a[S] = (B[S,S])^{-1} 1 / (1^T (B[S,S])^{-1} 1)`

If B is Toeplitz, B[S,S] is not necessarily Toeplitz (if S is not contiguous). But if S = {0, ..., n-1} (all positive), then B[S,S] = B is Toeplitz, and we can use FFT to solve.

In practice, the support might be all of [0, n-1] (if the solution is fully positive), in which case we can solve efficiently.

"""
MM (majorize-minimize) algorithm with FFT-accelerated Toeplitz operations,
softmax-smoothed L-BFGS, LP refinement, and n exploration.
Key innovation: exploits the Toeplitz structure of the upper-bound Hessian
for O(n log n) per-iteration cost in the MM algorithm.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def toepolis_matvec(alpha, v):
    """Compute B @ v where B[i,j] = alpha[0] if i==j, alpha[|i-j|]/2 if i!=j.
    alpha has length n (alpha[d] for d=0,...,n-1).
    v has length n."""
    n = len(v)
    # B[i,j] = f(i-j) where f(0) = alpha[0], f(d) = alpha[|d|]/2 for d!=0
    # (B @ v)[i] = sum_d f(d) * v[i-d]
    # This is circular convolution if we extend v, but we want linear.
    
    # f has length 2n-1 (from -(n-1) to n-1)
    # f[d] for d in [0, n-1]: f[0] = alpha[0], f[d] = alpha[d]/2
    # f[d] for d in [-(n-1), -1]: f[d] = alpha[|d|]/2
    
    # Build f as a vector of length 2n-1 for convolution
    # f_vec[k] = f(k - (n-1)) for k = 0, ..., 2n-2
    # So f_vec[n-1] = f(0) = alpha[0]
    # f_vec[n-1+d] = f(d) = alpha[d]/2 for d=1,...,n-1
    # f_vec[n-1-d] = f(-d) = alpha[d]/2 for d=1,...,n-1
    
    f_vec = np.zeros(2 * n - 1)
    f_vec[n - 1] = alpha[0]
    f_vec[n:] = alpha[1:] / 2.0
    f_vec[:n - 1] = alpha[1:][::-1] / 2.0
    
    # Linear convolution of f_vec with v
    size = 1
    while size < (2 * n - 1) + n - 1:
        size <<= 1
    F = rfft(f_vec, size)
    V = rfft(v, size)
    result = irfft(F * V, size)
    # The result has length (2n-1) + n - 1 = 3n-2
    # We want (B @ v)[i] = (f * v)[i + (n-1)] for i = 0, ..., n-1
    # Wait, let me think about this more carefully.
    # (f * v)[k] = sum_j f[j] * v[k-j]
    # We want (B @ v)[i] = sum_d f(d) * v[i-d] where d ranges over valid values
    # = sum_j f_vec[j] * v[i - (j - (n-1))]
    # = sum_j f_vec[j] * v[i + n - 1 - j]
    # Let m = j - (n-1), so j = m + (n-1):
    # = sum_m f(m) * v[i - m]
    # This is the cross-correlation, or equivalently the convolution with f reversed.
    
    # Actually, (f_vec * v)[k] = sum_j f_vec[j] * v[k-j]
    # We want (B@v)[i] = sum_j f_vec[j] * v[i + (n-1) - j] = (f_vec * v)[i + n - 1]
    
    return result[n - 1:n - 1 + n]


def project_simplex(v):
    """Project v onto the simplex {x: x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def mm_step(a, beta, lr=None):
    """One MM step: compute upper bound, take gradient step, project."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = conv - max_conv
    exp_b = np.exp(beta * shifted)
    alpha_full = exp_b / np.sum(exp_b)
    
    # alpha_full has length 2n-1
    # We need alpha[d] for d = 0, ..., n-1
    # But note: conv is symmetric, so alpha_full[k] = alpha_full[2n-2-k]
    # The Toeplitz matrix B is formed from alpha[d] = alpha_full[d] for d = 0, ..., n-1
    # (since (Q_k) for k and 2n-2-k are the same matrix for k != 0)
    
    # Wait, I need to be more careful.
    # conv[k] = sum_i a[i]*a[k-i] for k = 0, ..., 2n-2
    # By symmetry: conv[k] = conv[2n-2-k]
    # Q_k for k = 0: identity
    # Q_k for k = 1, ..., n-1: (Q_k)[i, i+k] = (Q_k)[i+k, i] = 1/2
    # Q_k for k = n, ..., 2n-2: same as Q_{2n-2-k}
    
    # So B = sum_{k=0}^{2n-2} alpha_full[k] * Q_k
    # = alpha_full[0] * I + sum_{k=1}^{n-1} (alpha_full[k] + alpha_full[2n-2-k]) * Q_k
    
    # Since conv is symmetric, alpha_full[k] = alpha_full[2n-2-k] (they have the same conv value)
    # So B = alpha_full[0] * I + 2 * sum_{k=1}^{n-1} alpha_full[k] * Q_k
    
    # Thus: f(0) = alpha_full[0], f(d) = 2 * alpha_full[d] / 2 = alpha_full[d] for d = 1, ..., n-1
    # Wait: (B)[i,j] for |i-j| = d > 0 is (alpha_full[d] + alpha_full[2n-2-d]) / 2 = alpha_full[d] (by symmetry)
    # And (B)[i,i] = alpha_full[0]
    
    # So the Toeplitz symbol is: f[0] = alpha_full[0], f[d] = alpha_full[d] for d > 0
    
    alpha_sym = alpha_full[:n]  # f[0] = alpha_full[0], f[d] = alpha_full[d] for d=1,...,n-1
    
    # Compute B @ a
    Ba = toepolis_matvec(alpha_sym, a)
    
    if lr is None:
        # Adaptive step size
        # For the QP min a^T B a, a good lr is 1 / (2 * lambda_max(B))
        # Estimate lambda_max using power iteration or just use a heuristic
        lr = 1.0 / (2.0 * float(np.max(np.abs(Ba))) + 1e-10)
    
    # Gradient step + projection
    a_new = a - lr * Ba
    a_new = project_simplex(a_new)
    
    return a_new


def mm_optimize(a_init, beta=50, n_iters=500, verbose=False, time_budget=60):
    """Run MM optimization."""
    start = time.monotonic()
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        a = mm_step(a, beta)
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if verbose and (it % 50 == 0 or it == n_iters - 1):
            print(f"    MM it{it}: obj={best_obj:.8f}")
    
    # Post-process: try to improve with a few more steps at higher beta
    for beta_high in [100, 200, 500]:
        if time.monotonic() - start > time_budget:
            break
        for _ in range(100):
            if time.monotonic() - start > time_budget:
                break
            a = mm_step(a, beta_high)
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    elapsed = time.monotonic() - start
    if verbose:
        print(f"  MM: obj={best_obj:.8f}, time={elapsed:.1f}s")
    return best_a, best_obj


def lbfgs_obj(z, beta):
    """Objective for L-BFGS in z-parameterization."""
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a = a / a.sum()
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = conv - max_conv
    sum_exp = float(np.sum(np.exp(beta * shifted)))
    return 2.0 * n * (max_conv + np.log(sum_exp)) / beta


def lbfgs_grad(z, beta):
    """Gradient for L-BFGS in z-parameterization."""
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a = a / a.sum()
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = conv - max_conv
    exp_b = np.exp(beta * shifted)
    sum_exp = float(np.sum(exp_b))
    alpha_full = exp_b / sum_exp
    
    # grad_a[i] = sum_k alpha_full[k] * a[k-i] (where valid)
    # = toepolis_matvec(alpha_full[:n], a)  ... wait, need to verify
    
    # Actually, the full gradient is:
    # grad_a[i] = sum_{k=0}^{2n-2} alpha_full[k] * a[k-i] (where 0 <= k-i < n)
    # = sum_{k=0}^{2n-2} alpha_full[k] * a[k-i]
    
    # By the same logic as before, this equals:
    # sum_{d=-(n-1)}^{n-1} (sum_{k: k-i=d} alpha_full[k]) * a[d] ... no this isn't right
    
    # Let me just use the toepolis_matvec with the combined alpha:
    # alpha_combined[d] = alpha_full[d] + alpha_full[2n-2-d] for d = 0,...,n-1
    # But alpha_full[d] = alpha_full[2n-2-d] (by symmetry of conv), so:
    # alpha_combined[d] = 2 * alpha_full[d] for d > 0, and alpha_full[0] for d = 0
    
    # Hmm wait, the gradient formula is:
    # d/d a[i] of sum_k alpha_full[k] * conv[k]
    # = sum_k alpha_full[k] * a[k-i] (where 0 <= k-i < n)
    
    # Let m = k-i, so k = i+m:
    # = sum_m alpha_full[i+m] * a[m] (where 0 <= m < n and 0 <= i+m < 2n-1)
    
    # This is a cross-correlation, which I showed earlier equals:
    # convolve(alpha_full, a[::-1], 'full')[n-1:n-1+n]
    
    size = 1
    while size < (2 * n - 1) + n - 1:
        size <<= 1
    W = rfft(alpha_full, size)
    A_rev = rfft(a[::-1], size)
    result = irfft(W * A_rev, size)
    grad_a = result[n - 1:n - 1 + n]
    
    # Chain rule for softmax: d f / d z[j] = a[j] * (grad_a[j] * 2n/beta - a[j] * sum_i grad_a[i] * a[i] * 2n/beta)
    # = (2n/beta) * a[j] * (grad_a[j] - dot(grad_a, a))
    
    factor = 2.0 * n / beta
    grad_z = factor * a * (grad_a - np.dot(grad_a, a))
    
    return grad_z


def softmax_lbfgs(a_init, beta_schedule, time_budget, verbose=False):
    """Run L-BFGS with beta annealing."""
    start = time.monotonic()
    n = len(a_init)
    best_a = np.maximum(a_init, 0).copy()
    s = float(np.sum(best_a))
    if s < 1e-10:
        return a_init, np.inf
    best_a = best_a / s
    best_obj = compute_obj(best_a)
    
    for beta in beta_schedule:
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        z0 = np.log(best_a + 1e-15)
        
        result = minimize(
            lbfgs_obj, z0,
            jac=lbfgs_grad,
            args=(beta,),
            method='L-BFGS-B',
            options={'maxiter': 300, 'gtol': 1e-10, 'maxfun': 2000, 'ftol': 1e-14}
        )
        
        z_opt = result.x
        a_opt = np.exp(z_opt - z_opt.max())
        a_opt = a_opt / a_opt.sum()
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    L-BFGS beta={beta}: obj={best_obj:.8f}")
    
    elapsed = time.monotonic() - start
    if verbose:
        print(f"  L-BFGS total: obj={best_obj:.8f}, time={elapsed:.1f}s")
    return best_a, best_obj

def build_conv_matrix(a):
    """Build matrix A such that (A @ g)[k] = conv(a_norm, g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None
