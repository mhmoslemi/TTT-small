import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import math

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj_normalized(a):
    """Compute 2n*max(conv) assuming sum(a)=1."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def obj_full(a):
    """Compute the full objective 2n*max_b/sum_a^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-11:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def proj_simplex(v, target_sum=1.0):
    """Project v onto the simplex {x: sum(x)=target_sum, x>=0}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    # Find theta such that sum(max(v_i - theta, 0)) = target_sum
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(n, 0, -1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[0]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0)

def cross_correlation_fft(p, a):
    """Compute (p ⋆ a)[i] = sum_k p[k] * a[k-i] for i=0..n-1 using FFT."""
    n = len(a)
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    result = ifft(P * np.conj(A)).real
    return result[:n]

    Nesterov-accelerated gradient descent on logsumexp(tau * conv(a,a)).

    """

    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    
    # Nesterov parameters
    beta = 0.9  # momentum
    lr_init = 1.0 / np.sqrt(n)
    v_mom = np.zeros(n)
    it = 0
    patience = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        # Linearly interpolate tau
        tau = tau_init + (tau_end - tau_init) * progress

        y = a + beta * v_mom
        y = np.maximum(y, 0)

            s_y = 1.0

        sum_exp = float(np.sum(exp_vals))
