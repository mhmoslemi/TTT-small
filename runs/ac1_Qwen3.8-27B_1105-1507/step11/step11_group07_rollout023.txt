import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_soft_gradient(a, p):
    """Gradient of soft-max objective via FFT cross-correlation."""
    n = len(a)
    p_len = len(p)
    a_rev = a[::-1]
    size = 1
    while size < n + p_len - 1:
        size <<= 1
    A = rfft(a_rev, size)
    P = rfft(p, size)
    full = irfft(A * P, size)
    q = full[n - 1:n - 1 + n]
    return 2.0 * q


def peak_structure(a):
    """Analyze the peak structure of the convolution."""
    conv = compute_conv(a)
    c_max = float(np.max(conv))
    sorted_conv = np.sort(conv)[::-1]
    # Effective number of significant peaks
    threshold = c_max * 0.9
    n_sig = int(np.sum(conv >= threshold))
    # Gap ratio between top peak and second peak
    if len(sorted_conv) >= 2 and sorted_conv[0] > 0:
        gap = 1.0 - sorted_conv[1] / sorted_conv[0]
    else:
        gap = 0.0
    return n_sig, gap, c_max


def adaptive_soft_max_gd(a, time_budget=200.0, beta_start=10.0, beta_end=60.0,
                         lr_start=0.007, momentum=0.91, verbose=True, seed=42):
    """
    Soft-max gradient descent with adaptive beta based on peak structure.
    When peaks are well-separated, use higher beta to focus on the top peak.
    When peaks are similar (flat), use lower beta for smoother updates.

    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    it = 0
    stall = 0
    best_it = 0

        # Adaptive beta: based on peak structure

            # If few significant peaks and large gap, use higher beta
            # If many similar peaks, use lower beta

                beta_mult = 1.5
            elif n_sig <= 5:
                beta_mult = 1.2
            else:
                beta_mult = 0.8
            beta = (beta_start + (beta_end - beta_start) * progress) * beta_mult
            beta = np.clip(beta, 5.0, 100.0)
        else:
            beta = beta_start + (beta_end - beta_start) * progress

        # Decay learning rate

        conv = compute_conv(a)
        c_max = float(np.max(conv))
        # Numerical stability

        p = exp_vals / float(np.sum(exp_vals))

        grad = compute_soft_gradient(a, p)
        v = momentum * v + grad
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        start = j
        length = min(n, m - j)
        if length > 0:
            A[j:j + length, j] = a[:length]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    # Try solvers in order of expected performance
    solvers = [cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend_extended(a, g_lp, time_budget=3.0):
    """Extended golden section + grid search for optimal blend factor."""
    lo, hi = 0.0, 1.5

    def obj_at(t):
        if t < 0:
            t = 0
        a_try = (1 - t) * a + t * g_lp
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)

    # Fine grid search first
    grid_t = np.linspace(0, 1.2, 25)
    best_t = 0.0
    best_obj = obj_at(0)
    for t in grid_t:
        o = obj_at(t)
        if o < best_obj:
            best_obj = o
            best_t = t

    # Golden section around the best grid point
    center = best_t
    span = 0.15
    lo = max(0, center - span)
    hi = center + span
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi

    start = time.monotonic()
    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    for _ in range(15):
        if time.monotonic() - start >= time_budget:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)

    t_best = c if fc < fd else d
    obj_best = min(fc, fd)

    if obj_best < best_obj:
        best_obj = obj_best
        best_t = t_best

    return best_t, best_obj

def peak_targeted_step(a, n_steps=50, time_budget=30.0, verbose=False):
    """
    Direct subgradient steps targeting the argmax of the convolution.
    For each step, compute gradient of conv[k*] and move to reduce it.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    lr = 0.005
    rng = np.random.default_rng(123)
    stall = 0

    while time.monotonic() - start < time_budget and it < n_steps:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        c_max = float(conv[k_star])

        # Gradient of conv[k_star] with respect to a
        # conv[k] = sum_i a[i] * a[k-i]
        # d(conv[k])/da[j] = sum_i (delta_{i,j}*a[k-i] + delta_{k-i,j}*a[i])
        #                    = a[k-j] (if valid) + a[j'] (if k-j is valid source)
        grad = np.zeros(n)
        for j in range(n):
            # First term: a[k-j] if 0 <= k-j < n
            ki = k_star - j
            if 0 <= ki < n:
                grad[j] += a[ki]
            # Second term (from symmetry of the sum): a[j] * a[k-j] has derivative
            # with respect to a[i] = a[k-i] * delta... actually let's think more carefully
            # conv[k] = sum_i a[i]*a[k-i] where both i and k-i in [0,n-1]
            # d/dx[j] of this = sum_i [delta_{i,j}*a[k-i] + delta_{k-i,j}*a[i]]
            #                 = a[k-j] (if k-j in [0,n-1]) + a[k-j] (if j in [0,k] and k-j in [0,n-1])
            # Wait, the second term: delta_{k-i,j} means i = k-j, so we get a[k-j] * 1
            # So total gradient at position j = a[k-j] + (a[k-j] if k-j is valid as i)
            # Actually: first term contributes a[k-j] if k-j valid
            # Second term: i = k-j, contributes a[i] = a[k-j] if k-j valid (i in [0,n-1])
            # But wait, we need i in [0,n-1] AND k-i in [0,n-1]
            # For second term: k-i = j, so i = k-j. Need i in [0,n-1], i.e., k-j in [0,n-1]
            # And j in [0,n-1] (always true since j is our index)
            # So: grad[j] = a[k-j] + a[k-j] = 2*a[k-j] if k-j in [0,n-1], else a[k-j] if just one term valid
            # Actually let me redo: the sum is over i such that 0<=i<n and 0<=k-i<n
            # d/dx[j] sum_i x[i]*x[k-i] = sum_i [delta_{i,j}*x[k-i] + x[i]*delta_{k-i,j}]
            # = x[k-j] (from first part, if k-j in [0,n-1]) + x[k-j] (from second part, k-i=j => i=k-j, need i in [0,n-1])
            # Both conditions are the same: k-j in [0,n-1]
            # So grad[j] = 2*x[k-j] if k-j in [0,n-1], else 0
            pass
        # Recompute properly
        for j in range(n):
            ki = k_star - j
            g_val = 0.0
            if 0 <= ki < n:
                g_val = 2.0 * a[ki]
            grad[j] = g_val

        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-10:
            break
        grad = grad / g_norm

        # Move in negative gradient direction
        a_new = a - lr * grad
        a_new = project_simplex(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Slightly increase lr if improving
            lr = min(lr * 1.02, 0.02)
        else:
            stall += 1
            lr *= 0.9
            if stall > 30:
                a = project_simplex(best_a + rng.normal(0, 0.002, n))
                stall = 0

        it += 1

    return best_a, best_obj

    elif mode == 9:
        # Power law at edges
        a = (1 + (2 * x - 1) ** 2 ** 3) ** 0.5
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolated_resize(a, new_n):
    """Resize sequence via interpolation and re-project."""
    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_simplex(np.maximum(a_new, 0))


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Combined adaptive soft-max GD, aggressive LP polishing,
    peak-targeted steps, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Parent load error: {e}")

    if best_seq is None:
        best_seq = u_shaped_init(n_current, sharpness=3.0)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")

    # Try alternative initializations at same length
    for init_func, name in [(arcsine_init, "arcsine"), (cosine_init, "cosine"),
                            (lambda n: quadratic_init(n, 0.3, 0.4), "gauss_left"),
                            (lambda n: quadratic_init(n, 0.7, 0.4), "gauss_right")]:
        a_try = init_func(n_current)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"{name} init: obj={best_score:.8f} **")

    # === Phase 1: Adaptive Soft-Max GD ===
    remaining = deadline - time.monotonic()
    if remaining > 150:
        print(f"\n=== Phase 1: Adaptive SmGD, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.22, 280.0)
        a1, obj1 = adaptive_soft_max_gd(
            best_seq, time_budget=alloc,
            beta_start=8.0, beta_end=50.0,
            lr_start=0.008, momentum=0.91,
            verbose=True, seed=seed
        )
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1 SmGD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 SmGD: obj={obj1:.8f} (no improvement)")

    # === Phase 2: Peak-Targeted Steps ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 2: Peak-Targeted Steps, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.1, 80.0)
        a2, obj2 = peak_targeted_step(
            best_seq, n_steps=200, time_budget=alloc, verbose=True
        )
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 PeakTarget: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 PeakTarget: obj={obj2:.8f} (no improvement)")

    # === Phase 3: Aggressive LP Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 3: Aggressive LP Polish, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.25, 300.0)
        a3, obj3 = lp_polish_aggressive(
            best_seq, time_budget=alloc, max_iters=120, verbose=True
        )
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 LP: obj={obj3:.8f} (no improvement)")

    # === Phase 4: Second SmGD round (higher beta) ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 4: SmGD Round 2, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.15, 150.0)
        a4, obj4 = adaptive_soft_max_gd(
            best_seq, time_budget=alloc,
            beta_start=30.0, beta_end=80.0,
            lr_start=0.004, momentum=0.93,
            verbose=True, seed=seed + 50
        )
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4 SmGD2: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 SmGD2: obj={obj4:.8f} (no improvement)")

    # === Phase 5: More LP Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 5: LP Polish 2, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.25, 200.0)
        a5, obj5 = lp_polish_aggressive(
            best_seq, time_budget=alloc, max_iters=80, verbose=True
        )
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  Phase5 LP2: obj={best_score:.8f} **")
        else:
            print(f"  Phase5 LP2: obj={obj5:.8f} (no improvement)")

    # === Phase 6: Diverse Restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 6: Diverse Restarts, time={remaining:.0f}s ===")
        n_cur = len(best_seq)

        for r in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break

            mode = r % 10
            a_r = diverse_init(n_cur, rng, mode=mode)
            blend = 0.5 + 0.1 * r
            a_r = project_simplex((1 - blend) * a_r + blend * best_seq)

            # Quick GD
            alloc_sgd = min(remaining * 0.2, 60.0)
            if alloc_sgd > 15:
                a_r, obj_r = adaptive_soft_max_gd(
                    a_r, time_budget=alloc_sgd,
                    beta_start=15.0, beta_end=50.0,
                    lr_start=0.006, momentum=0.91,
                    verbose=False, seed=seed + 100 + r
                )

            # Quick LP
            remaining = deadline - time.monotonic()
            if remaining > 30:
                alloc_lp = min(remaining * 0.15, 60.0)
                a_r, obj_r = lp_polish_aggressive(
                    a_r, time_budget=alloc_lp, max_iters=40, verbose=False
                )

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Length Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 7: Length Exploration, time={remaining:.0f}s ===")
        n_best = len(best_seq)

        n_options = sorted(set([
            n_best - 100, n_best - 50, n_best + 50, n_best + 100,
            n_best + 200, n_best - 200,
            700, 750, 850, 900, 950, 1000
        ]))
        n_options = [n for n in n_options if 500 <= n <= 1200 and abs(n - n_best) >= 40]
        n_options = n_options[:4]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break

            a_try = interpolated_resize(best_seq, n_try)

            # Optimize at new length
            alloc_sgd = min(remaining * 0.3, 50.0)
            if alloc_sgd > 15:
                a_try, obj_try = adaptive_soft_max_gd(
                    a_try, time_budget=alloc_sgd,
                    beta_start=15.0, beta_end=50.0,
                    lr_start=0.007, momentum=0.91,
                    verbose=False, seed=seed + 200
                )

            remaining = deadline - time.monotonic()
            if remaining > 30:
                alloc_lp = min(remaining * 0.25, 50.0)
                a_try, obj_try = lp_polish_aggressive(
                    a_try, time_budget=alloc_lp, max_iters=40, verbose=False
                )

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Final Intensive Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 8: Final Polish, time={remaining:.0f}s ===")

        # Final LP
        alloc = min(remaining * 0.5, 100.0)
        if alloc > 10:
            a_f, obj_f = lp_polish_aggressive(
                best_seq, time_budget=alloc, max_iters=60, verbose=False
            )
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        # Peak targeted
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.4, 30.0)
            a_f2, obj_f2 = peak_targeted_step(
                best_seq, n_steps=100, time_budget=alloc, verbose=False
            )
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final PeakTarget: obj={best_score:.8f} **")

    # === Phase 9: Last-ditch micro-perturbations ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print(f"\n=== Phase 9: Micro-perturbations, time={remaining:.0f}s ===")
        for attempt in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            # Very small perturbation + quick LP
            noise = rng.normal(0, 0.001, len(best_seq))
            a_p = project_simplex(best_seq + noise)
            alloc = min(remaining * 0.4, 15.0)
            if alloc > 3:
                a_p, obj_p = lp_polish_aggressive(
                    a_p, time_budget=alloc, max_iters=15, verbose=False
                )
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Micro {attempt}: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
