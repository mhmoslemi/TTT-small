def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def power_mean_smooth(a, p=80.0):

    pm = (np.mean(shifted ** p)) ** (1.0 / p) * max_c
    return pm, conv

    Let me use: sum_l a[l]*p[l+i] = (a * p[::-1])[n-1-i+len(p)-1] ... this is getting confusing.
    
    Simplest: use the formula from the previous code.
    b = a[::-1], then (b * p)[i+n-1] = sum_j b[j]*p[i+n-1-j] = sum_j a[n-1-j]*p[i+n-1-j]
    Let m = n-1-j, j = n-1-m: = sum_m a[m]*p[i+n-1-n+1+m] = sum_m a[m]*p[i+m]
    Yes! So sum_k p[k]*2*a_{k-i} (for valid indices) = 2*(b*p)[i+n-1] where b=a[::-1].
    But this double-counts the diagonal: when k=2i, the term should be a_i not 2*a_i.
    The diagoonal correction: we need to subtract p[2i]*a_i for each i where 2i < len(p).
    
    Wait, in the sum sum_m a[m]*p[i+m], the term m=i gives a[i]*p[2i]. This is counted once.
    In the original gradient formula, for k=2i we have d(conv)_k/da_i = a_i (not 2*a_i).
    In our formula 2*sum_m a[m]*p[i+m], the m=i term gives 2*a[i]*p[2i].
    So we need to subtract a[i]*p[2i] for the correction.
    
    Final: grad[i] = 2*(b*p)[i+n-1] - a[i]*p[2i] (if 2i < len(p), else just 2*(b*p)[i+n-1])
    """
    n = len(a)
    conv_len = 2 * n - 1
    N_fft = 1
    while N_fft < n + conv_len - 1:
        N_fft <<= 1
    b = a[::-1]
    bp = irfft(rfft(b, N_fft) * rfft(p, N_fft), N_fft)
    grad = 2.0 * bp[n - 1: 2 * n - 1]
    # Diagonal correction: for each i, if 2i < conv_len, subtract a[i]*p[2i]
    diag_indices = np.arange(n)
    valid_diag = 2 * diag_indices < conv_len
    grad[valid_diag] -= a[valid_diag] * p[2 * diag_indices[valid_diag]]
    return grad


def lp_direction(a):
    """
    LP: max sum(g) s.t. conv(a_norm, g)_k <= M for all k, sum(g)=1, g>=0.
    Where M = max(conv(a_norm, a_norm)).
    Returns g (normalized to sum 1) or None.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv_aa = compute_conv(a_norm)
    M = float(np.max(conv_aa))

    # Build A: A[k,j] = a_norm[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(j, j + n):
            A[k, j] = a_norm[k - j]

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def lp_direction_min_M(a):
    """
    LP: min M s.t. 2*conv(a_norm, g)_k - conv(a_norm, a_norm)_k <= M for all k, sum(g)=1, g>=0.
    This linearizes the quadratic around a.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv_aa = compute_conv(a_norm)

    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(j, j + n):
            A[k, j] = a_norm[k - j]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [2.0 * (A @ g) - M <= conv_aa, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def golden_section_line_search(a, direction, obj_func, max_iter=30):
    """Golden section search for best t in [0, 1]."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj_func(c)
    fd = obj_func(d)
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj_func(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_func(d)
    t = (lo + hi) / 2.0
    return t, obj_func(t)


def momentum_gd_optimize(a, time_budget=300.0, p_power=80.0, verbose=False):
    """
    Momentum-based gradient descent on power-mean smoothed objective.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)

    momentum = 0.9
    v = np.zeros(n)
    lr = 0.1 / np.sqrt(n)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Adaptive power: start moderate, increase for sharper max approximation
        p_curr = p_power * (1.0 + 2.0 * progress)

        conv = compute_conv(a)
        max_c = float(np.max(conv))
        if max_c < 1e-300:
            break

        # Compute smoothed gradient
        c_scaled = conv / max_c
        weights = c_scaled ** (p_curr - 1)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            weights = np.zeros(2 * n - 1)
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights = weights / w_sum

        grad = gradient_from_weights(a, weights)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break

        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        # Backtracking line search if objective increases
        obj_new = compute_obj(a_new)
        obj_cur = compute_obj(a)
        if obj_new > obj_cur * 1.001:
            # Try smaller step
            for shrink in [0.5, 0.25, 0.1]:
                a_trial = a + shrink * v
                a_trial = np.maximum(a_trial, 0)
                s_trial = float(np.sum(a_trial))
                if s_trial > 1e-10:
                    a_trial = a_trial / s_trial
                    obj_trial = compute_obj(a_trial)
                    if obj_trial < obj_cur:
                        a_new = a_trial
                        obj_new = obj_trial
                        break
            else:
                a_new = a
                obj_new = obj_cur
                v = 0.5 * v

        a = a_new
        it += 1

        # Decay learning rate
        lr *= (1.0 - 0.5 * progress / max(time_budget, 1e-10))
        lr = max(lr, 1e-8 / np.sqrt(n))

        # Track best
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Perturbation to escape local minima
        if stall > 500 and it % 100 == 0:
            strength = 0.02 * (1.0 + progress)
            a_pert = a * np.exp(strength * rng.standard_normal(n))
            a_pert = np.maximum(a_pert, 0)
            s_pert = float(np.sum(a_pert))
            if s_pert > 1e-10:
                a = a_pert / s_pert
                v = np.zeros(n)
                stall = 0

        if verbose and it % 50 == 0:
            print(f"      MoGD it{it}: obj={best_obj:.8f}, p={p_curr:.0f}, "
                  f"lr={lr:.6f}, stall={stall}, t={time.monotonic()-start:.0f}s")

    return best_a


def lp_gd_combined_optimize(a, time_budget=400.0, verbose=False):
    """
    Combined LP-direction + momentum GD optimizer.
    Alternates between LP-based exact direction and gradient steps.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(456)

    it = 0
    stall = 0
    last_lp_time = 0.0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5.0:
            break
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Determine strategy: LP every ~5s, otherwise gradient
        use_lp = (it % 15 == 0 and remaining > 10 and (time.monotonic() - last_lp_time) > 3.0)

        if use_lp:
            last_lp_time = time.monotonic()
            g, M = lp_direction_min_M(a)
            if g is None:
                g, _ = lp_direction(a), None
            if g is not None:
                # Line search
                def obj_at_t(t):
                    a_t = np.maximum((1 - t) * a + t * g, 0)
                    s_t = float(np.sum(a_t))
                    if s_t < 1e-10:
                        return np.inf
                    return compute_obj(a_t / s_t)
                best_t, best_t_obj = golden_section_line_search(a, g, obj_at_t, max_iter=25)
                if best_t_obj < best_obj - 1e-12:
                    a_new = np.maximum((1 - best_t) * a + best_t * g, 0)
                    s_new = float(np.sum(a_new))
                    if s_new > 1e-10:
                        a = a_new / s_new
                        best_obj = best_t_obj
                        best_a = a.copy()
                        stall = 0
                        it += 1
                        continue
        # Gradient step
        p_curr = 60.0 * (1.0 + 3.0 * progress)
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        if max_c < 1e-300:
            break
        c_scaled = conv / max_c
        weights = c_scaled ** (p_curr - 1)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            weights = np.zeros(2 * n - 1)
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights = weights / w_sum

        grad = gradient_from_weights(a, weights)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break

        lr = 0.5 / (grad_norm * np.sqrt(n))
        lr = min(lr, 2.0 / np.sqrt(n))

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a + rng.standard_normal(n) * 0.01
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
            else:
                a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            # Try LP direction as escape
            if stall > 100 and remaining > 20:
                g_esc, _ = lp_direction_min_M(a)
                if g_esc is not None:
                    def obj_esc(t):
                        a_t = np.maximum((1 - t) * a + t * g_esc, 0)
                        s_t = float(np.sum(a_t))
                        if s_t < 1e-10:
                            return np.inf
                        return compute_obj(a_t / s_t)
                    t_esc, obj_esc = golden_section_line_search(a, g_esc, obj_esc, max_iter=20)
                    if obj_esc < best_obj - 1e-12:
                        a_new = np.maximum((1 - t_esc) * a + t_esc * g_esc, 0)
                        s_new = float(np.sum(a_new))
                        if s_new > 1e-10:
                            a = a_new / s_new
                            best_obj = obj_esc
                            best_a = a.copy()
                            stall = 0
                            it += 1
                            continue
            # Perturb
            if stall > 200:
                strength = 0.03
                a = np.maximum(a * np.exp(strength * rng.standard_normal(n)), 0)
                s_p = float(np.sum(a))
                if s_p > 1e-10:
                    a = a / s_p
                stall = 0

        a = a_new
        it += 1

        if verbose and it % 20 == 0:
            print(f"      LP+GD it{it}: obj={best_obj:.8f}, p={p_curr:.0f}, "
                  f"stall={stall}, t={time.monotonic()-start:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def make_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []

    # 1. Uniform
    inits.append((np.ones(n) / n, "uniform"))

    # 2. Raised cosine
    t = np.linspace(0, 2 * np.pi, n)
    rc = 0.5 * (1 + np.cos(t))
    rc = np.maximum(rc, 1e-6)
    inits.append((rc / np.sum(rc), "raised_cos"))

    # 3. Hamming window
    hw = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1)) if n > 1 else np.array([1.0])
    inits.append((hw / np.sum(hw), "hamming"))

    # 4. Gaussian window
    x = np.linspace(-3, 3, n)
    gw = np.exp(-x ** 2)
    inits.append((gw / np.sum(gw), "gaussian"))

    # 5. Triangular
    tri = 1 - 2 * np.abs(np.linspace(-0.5, 0.5, n))
    tri = np.maximum(tri, 1e-6)
    inits.append((tri / np.sum(tri), "triangular"))

    # 6. Periodic (sine-squared)
    for freq in [2, 3, 5]:
        period = n // freq
        if period < 4:
            continue
        sig = np.sinc(np.linspace(0, freq, n, endpoint=False))
        sig = np.abs(sig)
        sig = np.maximum(sig, 1e-6)
        inits.append((sig / np.sum(sig), f"sinc_{freq}"))

    # 7. Random (Dirichlet)
    for alpha in [0.5, 2.0, 10.0]:
        r = rng.dirichlet(np.full(n, alpha))
        inits.append((r, f"dirichlet_{alpha}"))

    return inits

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)

            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s

                best_seq = parent.copy()

            break
        inits = make_initializations(n_try, rng)

            a_second = top_results[1][1].copy()
            alloc = min(remaining * 0.4, 100.0)
            a_second = lp_gd_combined_optimize(a_second, time_budget=alloc, verbose=False)
            remaining = deadline - time.monotonic()
            alloc2 = min(remaining * 0.2, 50.0)
            if alloc2 > 10:
                a_second = momentum_gd_optimize(a_second, time_budget=alloc2, p_power=200.0, verbose=False)
            obj_second = compute_obj(a_second)
            if obj_second < best_score:
                best_score = obj_second
                best_seq = a_second.copy()
                print(f"  Phase6: obj={best_score:.8f} **")

    # Phase 7: Final intensive polish
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    if remaining > 20:
        alloc = min(remaining * 0.4, 100.0)
        a_fin = lp_gd_combined_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Final LP+GD: obj={best_score:.8f} **")

    # GD polish with very high power
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 80.0)
        a_fin2 = momentum_gd_optimize(best_seq, time_budget=alloc,
                                      p_power=500.0, verbose=True)
        obj_fin2 = compute_obj(a_fin2)
        if obj_fin2 < best_score:
            best_score = obj_fin2
            best_seq = a_fin2.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
