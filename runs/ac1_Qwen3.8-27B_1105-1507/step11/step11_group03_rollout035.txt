def compute_conv(a):
    n = len(a)

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:

def lp_solve_tight(a, solver_list=None):
    """Solve LP: min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1."""
    n = len(a)
    # Build constraint matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)

    return None, None

def smoothed_grad(a, tau):
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    shifted = tau * (conv - exact_max)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    # Gradient: d/dx_i sum_k p_k * conv(x,x)_k = 2 * sum_k p_k * x[k-i]
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            g[i] += 2.0 * p[k] * a[k - i]
    return g, exact_max

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses enhanced LP fixed-point with tight tolerances, fine line search,
FFT-accelerated smoothed gradients, multi-n exploration, and adaptive
active-set perturbation strategies.

import numpy as np
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution of a with itself via direct method."""
    n = len(a)
    if n == 0:
        return np.array([])
    return np.convolve(a, a, mode='full')


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad(a, tau):
    """FFT-accelerated gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    shifted = tau * (conv - exact_max)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n:
        N <<= 1
    P = np.fft.fft(p, N)
    A = np.fft.fft(a, N)
    cross = np.fft.ifft(P * np.conj(A)).real[:n]
    g = 2.0 * cross
    return g, exact_max


def lp_solve_tight(a, solver_list=None):
    """Solve LP: min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None


def line_search(a, g, best_obj, n_coarse=40, n_fine=30):
    """Two-stage line search for optimal blend parameter t."""
    best_t = 0.0
    best_local_obj = best_obj
    best_a_try = None
    for i in range(1, n_coarse + 1):
        t = i / n_coarse
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj_try = compute_obj(a_try)
        if obj_try < best_local_obj:
            best_local_obj = obj_try
            best_t = t
            best_a_try = a_try.copy()
    if best_t > 0:
        t_lo = max(0.0, best_t - 3.0 / n_coarse)
        t_hi = min(1.0, best_t + 3.0 / n_coarse)
        for i in range(n_fine):
            t = t_lo + (t_hi - t_lo) * i / (n_fine - 1)
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_t = t
                best_a_try = a_try.copy()
    return best_t, best_local_obj, best_a_try

def nesterov_gd(a, time_budget, tau_levels=None, verbose=False):
    """Nesterov accelerated gradient descent with staircase tau schedule."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [3, 10, 30, 100, 300, 1000, 3000, 10000]

    beta = 0.9
    for tau in tau_levels:
        lr = 1.5 / np.sqrt(n)
        v = np.zeros(n)
        iters = 0
        max_iters = 4000
        stall = 0

        while time.monotonic() - start < time_budget and iters < max_iters:
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            gn = float(np.linalg.norm(g))
            if gn < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            sn = float(np.sum(a_new))
            if sn < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / sn

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1

            a = a_new
            iters += 1
            if iters % 200 == 0:
                lr = max(1e-9, lr * 0.9)
            if stall > 500:
                break

        if verbose:
            print(f"    tau={tau:6d}: obj={best_obj:.12f}, iters={iters}")

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_structured_init(n, mode):
    """Create structured initializations for different modes."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.maximum(np.sinc(x), 0.001)
        return a / a.sum()
    elif mode == 2:
        a = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / max(1, n - 1))
        return a / a.sum()
    elif mode == 3:
        x = np.arange(n) / max(1, n - 1)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 4:
        x = np.linspace(-3, 3, n)
        a = np.exp(-x ** 2)
        return a / a.sum()
    elif mode == 5:
        # Asymmetric exponential
        x = np.arange(n)
        a = np.exp(-0.5 * x / n * 3)
        return a / a.sum()
    elif mode == 6:
        # Asymmetric: one end decays faster
        x = np.arange(n) / n
        a = (1.0 - x) ** 0.5
        return a / a.sum()
    else:
        x = np.arange(n) / n
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()


def active_set_targeted(a, time_budget, seed=42, max_iters=300, verbose=False):

    n = len(a)

    start = time.monotonic()
    rng = np.random.default_rng(seed)

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        threshold = max_val * 0.999
        max_lags = np.where(conv >= threshold)[0]

        # Compute subgradient from active lags
        subgrad = np.zeros(n)
        for lag in max_lags:
            lo = max(0, lag - n + 1)
            hi = min(lag, n - 1)
            for i in range(lo, hi + 1):
                subgrad[i] += 2.0 * a[lag - i]
        subgrad /= max(1, len(max_lags))

        gn = np.linalg.norm(subgrad)
        if gn < 1e-15:
            break
        subgrad = subgrad / gn

        improved = False
        # Try multiple step sizes with line search
        for step in [0.0003, 0.001, 0.002, 0.005, 0.01, 0.02]:
            a_try = a - step * subgrad

            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:

                        a_try = a_try / a_try.sum()
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break

            if not improved:

                    a = a + 0.006 * rng.standard_normal(n)

                    f = rng.integers(1, 6)
                    p = rng.uniform(0, 2 * np.pi)
                    a = a + 0.01 * np.sin(2 * np.pi * f * x / n + p)
                elif pert_type == 2:
                    # Reduce specific high-contribution elements

                        lo = max(0, lag - n + 1)
                        hi = min(lag, n - 1)

                sa = float(np.sum(a))

                else:

    return best_a

                                  verbose=False, seed=271828, max_iters=200)
        obj9 = compute_obj(a9)
        if obj9 < best_score:
            best_score = obj9
            best_seq = a9.copy()
            print(f"  Ultra active: obj={best_score:.12f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.12f}, time={elapsed:.1f}s ===")
    return result
