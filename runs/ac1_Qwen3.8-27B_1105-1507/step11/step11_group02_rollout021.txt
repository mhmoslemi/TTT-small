"""
Optimizes step function coefficients using:
1. FFT-based smoothed gradient descent with Nesterov momentum
2. Adaptive tau (smoothness) schedule
3. Top-k peaking strategy
4. Exact coordinate descent for final polishing
5. Multi-n exploration with resampling
6. Diverse restarts

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = fft(a, size)
    return ifft(fa * fa, size).real[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))

def compute_gradient_fft(a, tau, top_k=None):
    """Compute gradient of smoothed objective w.r.t. a (normalized to sum=1)."""
    n = len(a)
    conv = compute_conv(a)
    
    # Smoothed distribution over peaks
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    
    if top_k is not None and top_k < len(exp_vals):
        # Zero out all but top_k entries
        idx = np.argsort(exp_vals)[-top_k:]
        mask = np.zeros(len(exp_vals), dtype=bool)
        mask[idx] = True
        exp_vals = np.where(mask, exp_vals, 0)
    
    sum_exp = np.sum(exp_vals)
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    # Gradient: 4*n * (p ⋆ a)[j] = 4*n * ifft(fft(p,N) * conj(fft(a,N))).real[:n]
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    corr = ifft(P * np.conj(A), N).real[:n]
    
    grad = 4.0 * n * corr
    return grad, conv, p

def simplex_projection(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if s <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - s))[0][-1] if np.any(u * np.arange(1, n+1) > (cssv - s)) else 0
    theta = (cssv[rho] - s) / (rho + 1)
    proj = np.maximum(v - theta, 0)
    return proj

def smoothed_gradient_descent(a, time_budget=300.0, tau_init=20.0, tau_end=5000.0, 
                               lr_decay=0.999, verbose=False, seed=42, top_k_frac=None):
    """
    Nesterov-accelerated smoothed gradient descent with adaptive tau.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # Learning rate: start conservative, use backtracking if needed
    lr = 4.0 / (n * np.sqrt(n))  # small initial lr
    v = np.zeros(n)
    beta = 0.9
    
    start = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)
    
    top_k = int(n * 2) if top_k_frac is None else int(n * top_k_frac * 2)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        
        # Adaptive tau: start smooth, end sharp
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)
        
        # Occasionally use top-k focusing
        tk = top_k if progress > 0.5 else None
        
        try:
            grad, conv, p = compute_gradient_fft(a, tau, top_k=tk)
        except Exception:
            break
        
        # Nesterov momentum
        v = beta * v + grad
        v_norm = np.linalg.norm(v)
        if v_norm > 1e10:
            v = v / v_norm * 1e10
        
        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            # Scale lr to make step size reasonable
            lr_eff = lr * min(1.0, 1.0 / (grad_norm / (4.0 * n) + 1e-10))
            lr_eff = max(lr_eff, lr * 0.01)
        else:
            lr_eff = lr
        
        a_new = a - lr_eff * v
        # Project onto simplex
        a_new = simplex_projection(a_new, 1.0)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        lr *= lr_decay
        it += 1
        
        if verbose and it % 2000 == 0:
            print(f"    SGD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr_eff:.2e}")
    
    return best_a

2. Top-peak focusing for sharp convergence
3. Exact coordinate descent for final polishing
4. Multi-n exploration
5. Diverse restarts with annealed re-optimization
Key innovation: efficient O(N log N) gradient computation via FFT cross-correlation,
combined with multiplicative updates for natural non-negativity preservation.
"""

import numpy as np
from scipy.fft import fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = fft(a, size)
    return ifft(fa * fa, size).real[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def compute_gradient(a, tau, top_k=None):
    """
    Compute gradient of 2n * (1/tau) * log(sum exp(tau * conv[k])) w.r.t. a.
    Assumes a is normalized (sum = 1).
    Returns (gradient, conv, peak_distribution).
    """
    n = len(a)
    conv = compute_conv(a)
    
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    
    if top_k is not None and top_k < len(exp_vals):
        if top_k <= 0:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            idx = np.argpartition(exp_vals, -top_k)[-top_k:]
            mask = np.zeros(len(exp_vals), dtype=bool)
            mask[idx] = True
            exp_vals = np.where(mask, exp_vals, 0.0)
    
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    
    # Gradient = 4*n * (p cross-correlate a)
    # (p ⋆ a)[j] = sum_m p[m] * a[m-j] = ifft(fft(p,N) * conj(fft(a,N))).real[j]
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    corr = ifft(P * np.conj(A), N).real[:n]
    
    grad = 4.0 * n * corr
    return grad, conv, p


def multiplicative_gd(a, time_budget=300.0, tau_init=20.0, tau_end=5000.0,
                      lr_init=None, verbose=False, seed=42):
    """
    Multiplicative gradient descent with Nesterov momentum and adaptive tau.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    if lr_init is None:
        lr_init = 6.0 / np.sqrt(n)
    
    v = np.zeros(n)
    beta = 0.85
    start = time.monotonic()
    it = 0
    stagnation = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        
        # Adaptive tau
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)
        
        # Top-k focusing in later stages
        if progress < 0.3:
            top_k = None
        elif progress < 0.6:
            top_k = 4 * n
        else:
            top_k = 2 * n
        
        try:
            grad, conv, p = compute_gradient(a, tau, top_k=top_k)
        except Exception:
            break
        
        # Nesterov momentum
        v = beta * v + grad
        v_norm = float(np.linalg.norm(v))
        if v_norm > 1e15:
            v = v / v_norm * 1e15
        
        # Adaptive lr
        lr = lr_init * (1.0 - 0.6 * progress)
        
        # Multiplicative update
        log_update = -lr * v / max(v_norm, 1e-10)
        log_update = np.clip(log_update, -5.0, 5.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        a_new = a_new / s_new
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        a = a_new
        it += 1
        
        # Occasionally restart momentum
        if stagnation > 500:
            v = np.zeros(n)
            stagnation = 0
            # Small random perturbation
            a = a * np.exp(np.random.default_rng(it).normal(0, 0.01, n))
            a = np.maximum(a, 1e-15)
            a = a / float(np.sum(a))
        
        if verbose and it % 5000 == 0:
            print(f"    MGD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.4f}")
    
    return best_a


def coordinate_descent(a, time_budget=300.0, verbose=False, seed=42):
    """
    Exact coordinate descent with golden section search.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        S_cur = float(np.sum(a))
        
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            
            v = a[i]
            S_other = S_cur - v
            if S_other < 1e-14:
                continue
            
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            
            # Separate quadratic term (k = 2i) from linear terms
            mask_lin = (ks != 2 * i)
            ks_lin = ks[mask_lin]
            a_ki_lin = a[ks_lin - i]
            c_k_lin = conv[ks_lin].copy()
            
            c_2i = float(conv[2 * i]) if 0 <= 2 * i <= 2 * n - 2 else 0.0
            
            # 1D objective
            def f_1d(x):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom
            
            # Candidate points
            scale = max(v, 1.0 / n, 1e-6)
            candidates = sorted(set([
                0.0, min(v * 0.5, 1.0), min(v * 0.7, 1.0), v,
                min(v * 1.3, 3.0), min(v * 1.5, 3.0), min(v * 2.0, 4.0),
                1.0 / n, 2.0 / n, 3.0 / n
            ]))
            
            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                x = max(0.0, x)
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x
            
            # Golden section
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.5
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-8
            
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)
            
            for _ in range(40):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)
            
            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-14:
                continue
            
            # Update
            delta = x_opt - v
            a[i] = x_opt
            if 0 <= 2 * i <= 2 * n - 2:
                conv[2 * i] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            S_cur += delta
            
            if S_cur > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_cur * S_cur)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
        
        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2.0 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(2, 15, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Cosine window variant
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a += 0.1 * rng.uniform(0, 1, n)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")
    
    n_best = len(best_seq)
    
    # === Phase 1: Multiplicative Gradient Descent ===
    print(f"\n=== Phase 1: Multiplicative GD ===")

    if p1_budget > 30:
        a_mgd = multiplicative_gd(best_seq, time_budget=p1_budget,
                                   tau_init=10.0, tau_end=3000.0,

        obj_mgd = compute_obj(a_mgd)
