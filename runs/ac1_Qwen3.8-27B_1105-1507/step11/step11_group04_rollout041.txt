```python
"""
Optimizes step function coefficients using continuous coordinate descent
with in-place FFT convolution updates, LP alternating minimization,
and multi-restart exploration. Key improvement over previous: uses
golden-section 1D search for each coordinate instead of discrete candidates.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def solve_lp(a, warm_start=None):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass
    for solver in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using continuous coordinate descent
with O(n) in-place convolution updates and golden-section refinement per
coordinate, combined with LP alternating minimization and multi-restart
exploration. Key improvement: exact 1D optimization per coordinate instead
of discrete candidate evaluation.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None):
    """Solve LP: min M s.t. (a_norm*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass
    for solver in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def continuous_cd(a, time_budget, seed=42, verbose=False):
    """
    Continuous coordinate descent with golden-section 1D search per coordinate.
    Maintains sum(a)=1 normalization and updates convolution in-place.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    sweep = 0
    no_improve = 0
    max_no_improve = 3
    inv_n = 1.0 / n

    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            v_old = a[j]
            s_base = 1.0 - v_old

            if s_base < 1e-15:
                continue

            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)
            idx = np.arange(k_start, k_end + 1)
            other_idx = idx - j

            # Compute max_unaffected
            max_unaffected = 0.0
            if k_start > 0:
                mu = float(np.max(conv[:k_start]))
                if mu > max_unaffected:
                    max_unaffected = mu
            if k_end < 2 * n - 2:
                mu = float(np.max(conv[k_end + 1:]))
                if mu > max_unaffected:
                    max_unaffected = mu

            # Compute base convolution (with a[j] removed)
            base = conv[idx] - 2.0 * v_old * a[other_idx]
            diag_mask = (idx == 2 * j)
            if np.any(diag_mask):
                base[diag_mask] += v_old * v_old

            non_diag_mask = ~diag_mask
            other_non_diag = a[other_idx[non_diag_mask]] if np.any(non_diag_mask) else None
            base_non_diag = base[non_diag_mask] if np.any(non_diag_mask) else None
            diag_base = float(base[diag_mask][0]) if np.any(diag_mask) else float('-inf')

            # Define local evaluation function
            gr = (np.sqrt(5.0) - 1.0) / 2.0

            def eval_f(v, base_non_diag=base_non_diag, other_non_diag=other_non_diag,
                       diag_base=diag_base, max_unaffected=max_unaffected, s_base=s_base):
                if other_non_diag is not None:
                    nd_max = float(np.max(base_non_diag + 2.0 * v * other_non_diag))
                else:
                    nd_max = 0.0
                d_val = diag_base + v * v
                ma = max(max_unaffected, nd_max, d_val)
                d = s_base + v
                if d < 1e-15:
                    return 1e10
                return 2.0 * n * ma / (d * d)

            # Quick check: evaluate at a few points
            v_upper = max(5.0 * v_old, 10.0 * inv_n)
            if v_old > 0:
                v_upper = max(v_upper, 2.0 * v_old)
            v_upper = min(v_upper, 1.0)

            f_old = eval_f(v_old)
            f_zero = eval_f(0.0)

            # Quick candidates
            quick_vals = [0.0, v_old]
            if v_old > 0:
                quick_vals.extend([v_old * 0.3, v_old * 0.6, v_old * 1.5, v_old * 2.0])
            quick_vals.extend([2.0 * inv_n, 5.0 * inv_n])
            quick_vals = [v for v in quick_vals if 0 <= v <= v_upper]

            best_quick_v = v_old
            best_quick_f = f_old
            for vv in quick_vals:
                ff = eval_f(vv)
                if ff < best_quick_f:
                    best_quick_f = ff
                    best_quick_v = vv

            # If quick search found improvement, refine with golden section
            if best_quick_f < best_obj - 1e-14:
                # Determine search interval around best_quick_v
                if best_quick_v == 0.0:
                    lo, hi = 0.0, min(v_upper, 2.0 * 5.0 * inv_n)
                elif best_quick_v == v_upper:
                    lo, hi = max(0.0, v_upper * 0.5), v_upper
                else:
                    span = max(2.0 * v_old + 2.0 * inv_n, 4.0 * inv_n)
                    lo = max(0.0, best_quick_v - span * 0.5)
                    hi = min(v_upper, best_quick_v + span * 0.5)

                v1 = hi - gr * (hi - lo)
                v2 = lo + gr * (hi - lo)
                f1 = eval_f(v1)
                f2 = eval_f(v2)
                for _ in range(35):
                    if f1 < f2:
                        hi = v2
                        v2 = v1
                        f2 = f1
                        v1 = hi - gr * (hi - lo)
                        f1 = eval_f(v1)
                    else:
                        lo = v1
                        v1 = v2
                        f1 = f2
                        v2 = lo + gr * (hi - lo)
                        f2 = eval_f(v2)

                v_opt = (lo + hi) / 2.0
                f_opt = eval_f(v_opt)
                if f_opt > best_quick_f:
                    v_opt = best_quick_v
                    f_opt = best_quick_f
            else:
                continue

            # Apply update
            delta = v_opt - v_old
            a[j] = v_opt
            if np.any(non_diag_mask):
                conv[idx[non_diag_mask]] += 2.0 * delta * a[other_idx[non_diag_mask]]
            if np.any(diag_mask):
                conv[2 * j] += v_opt * v_opt - v_old * v_old

            new_s = s_base + v_opt
            if new_s > 1e-15:
                a /= new_s
                conv /= (new_s * new_s)

            new_obj = 2.0 * n * float(np.max(conv))
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_a = a.copy()
                sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"  CD sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")

    return best_a


def lp_alternate(a, time_budget, seed=42, verbose=False):
    """LP alternating minimization with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    iter_count = 0
    stall = 0

    while time.monotonic() - start < time_budget - 3:
        t0 = time.monotonic()
        g_lp, M_lp = solve_lp(a, warm_start=prev_g)
        if g_lp is None:
            break
        prev_g = g_lp
        iter_count += 1

        best_local_obj = best_obj
        best_local_a = a.copy()
        best_t = 0.0

        t_values = [1.0, 0.7, 0.5, 0.3, 0.15, 0.1, 0.05, 0.02, 0.01]
        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_t = t

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall = 0
            if verbose and iter_count % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP {iter_count}: obj={best_obj:.10f}, t={best_t:.3f}, elapsed={elapsed:.0f}s")
        else:
            stall += 1
            if stall >= 6:
                strength = 0.005 + 0.003 * min(stall, 10)
                pert = rng.standard_normal(n) * strength
                a = best_a + pert * best_a
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                prev_g = None
                stall = 0

    return best_a


def fourier_perturb(a, rng, strength=0.015):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 9):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_restart(n, rng, style='uniform'):
    """Generate a random initial sequence of length n."""
    if style == 'uniform':
        a = np.ones(n)
    elif style == 'hann':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))
    elif style == 'raised_cosine':
        a = np.sin(np.pi * np.arange(n) / (n - 1)) ** 2
    else:
        a = rng.random(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: continuous coordinate descent + LP alternating +
    multi-restart exploration + length variation.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 15.0

    # Initialize from parent
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0)
        if a.size == 0 or a.sum() < 0.01:
            raise ValueError("Invalid parent")
        a = a / a.sum()
        n0 = a.size
        print(f"Parent: n={n0}, obj={compute_obj_norm(a):.10f}")
    except Exception:
        a = ones_800()
        n0 = 800
        print(f"Default: n=800, obj={compute_obj_norm(a):.10f}")

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    rng = np.random.default_rng(seed)
    n = len(best_a)

    # === Phase 1: Continuous CD (45% time) ===
    print("\n--- Phase 1: Continuous CD ---")
    p1_budget = (budget_s - 30) * 0.45
    a1 = continuous_cd(best_a, time_budget=p1_budget, seed=seed, verbose=True)
    obj1 = compute_obj_norm(a1)
    if obj1 < best_obj:
        best_obj = obj1
        best_a = a1.copy()
        print(f"  P1 done: {best_obj:.10f} **")
    else:
        print(f"  P1: {obj1:.10f}")

    # === Phase 2: LP Alternating (20% time) ===
    print("\n--- Phase 2: LP Alternating ---")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p2_budget = min(remaining * 0.4, (budget_s - 30) * 0.20)
        a2 = lp_alternate(best_a, time_budget=p2_budget, seed=seed + 1000, verbose=True)
        obj2 = compute_obj_norm(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            print(f"  P2 done: {best_obj:.10f} **")
        else:
            print(f"  P2: {obj2:.10f}")
        # CD polish after LP
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a2b = continuous_cd(best_a, time_budget=min(remaining * 0.3, 60.0),
                                 seed=seed + 2000, verbose=False)
            obj2b = compute_obj_norm(a2b)
            if obj2b < best_obj:
                best_obj = obj2b
                best_a = a2b.copy()
                print(f"  P2b CD: {best_obj:.10f} **")

    # === Phase 3: Perturbation + CD (15% time) ===
    print("\n--- Phase 3: Perturbation + CD ---")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3_budget = min(remaining * 0.5, (budget_s - 30) * 0.15)
        for attempt in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            strength = 0.008 + 0.005 * attempt
            a_pert = fourier_perturb(best_a, rng, strength=strength)
            alloc = p3_budget / (4 - attempt)
            a3 = continuous_cd(a_pert, time_budget=alloc, seed=seed + 5000 + attempt * 100,
                                verbose=False)
            obj3 = compute_obj_norm(a3)
            if obj3 < best_obj:
                best_obj = obj3
                best_a = a3.copy()
                print(f"  P3 attempt {attempt}: {best_obj:.10f} **")
            else:
                print(f"  P3 attempt {attempt}: {obj3:.10f}")

    # === Phase 4: Length exploration (10% time) ===
    print("\n--- Phase 4: Length Exploration ---")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        n_best = len(best_a)
        n_options = []
        for delta in [-50, -25, 25, 50, 100, -100, 150]:
            n_try = n_best + delta
            if 300 <= n_try <= 2000:
                n_options.append(n_try)
        n_options = sorted(set(n_options))

        for n_try in n_options[:4]:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            x_old = np.linspace(0, 1, len(best_a))
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_a)
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                continue
            a_try = a_try / s

            alloc = min(remaining * 0.35, 100.0)
            if alloc > 25:
                a_try = continuous_cd(a_try, time_budget=alloc, seed=n_try * 7, verbose=False)

            obj_try = compute_obj_norm(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                print(f"  ** n={n_try}: {best_obj:.10f} **")
            else:
                print(f"  n={n_try}: {obj_try:.10f}")

    # === Phase 5: Different initializations (10% time) ===
    print("\n--- Phase 5: Alternative Initializations ---")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        styles = ['hann', 'raised_cosine', 'uniform']
        for style in styles:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            a_init = random_restart(n, rng, style=style)
            # Interpolate with best to create a blend
            a_blend = 0.3 * a_init + 0.7 * best_a
            a_blend = np.maximum(a_blend, 0)
            s = float(np.sum(a_blend))
            if s > 1e-10:
                a_blend = a_blend / s
            alloc = min(remaining * 0.4, 80.0)
            if alloc > 20:
                a_alt = continuous_cd(a_blend, time_budget=alloc, seed=seed + 9000,
                                       verbose=False)
                obj_alt = compute_obj_norm(a_alt)
                if obj_alt < best_obj:
                    best_obj = obj_alt
                    best_a = a_alt.copy()
                    print(f"  Alt ({style}): {best_obj:.10f} **")
                else:
                    print(f"  Alt ({style}): {obj_alt:.10f}")

    # === Phase 6: Final intensive CD polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("\n--- Phase 6: Final Polish ---")
        a_fin = continuous_cd(best_a, time_budget=min(remaining * 0.85, 90.0),
                               seed=seed + 42424, verbose=False)
        obj_fin = compute_obj_norm(a_fin)
        if obj_fin < best_obj:
            best_obj = obj_fin
            best_a = a_fin.copy()
            print(f"  Final: {best_obj:.10f} **")
        else:
            print(f"  Final: {obj_fin:.10f}")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        fallback = [1.0] * 100
        result = fallback
        best_obj = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
```
