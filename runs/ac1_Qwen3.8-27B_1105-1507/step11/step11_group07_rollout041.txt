"""
Improved max-convolution minimization using accelerated multiplicative weights
(mirror descent with entropy divergence), soft-max gradient descent with Nesterov
momentum, and LP-based polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv)/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def softmax_weights(conv, beta):
    """Compute softmax weights on convolution values."""
    c_max = float(np.max(conv))
    exp_vals = np.exp(beta * (conv - c_max))
    p = exp_vals / float(np.sum(exp_vals))
    return p

def cross_correlation(p, a):
    """Compute q[i] = sum_k p[k] * a[k-i] for i=0..n-1."""
    n = len(a)
    a_rev = a[::-1]
    size = 1
    while size < n + len(p) - 1:
        size <<= 1
    A = rfft(a_rev, size)
    P = rfft(p, size)
    full = irfft(A * P, size)
    return full[n - 1:n - 1 + n]

def compute_grad_soft(a, p):
    """Gradient of soft-max objective: 2 * cross_correlation(p, a)."""
    return 2.0 * cross_correlation(p, a)

def multiplicative_weights(a, time_budget=200.0, beta_start=5.0, beta_end=30.0,
                           eta_start=0.3, eta_end=0.05, verbose=True, seed=42):
    """
    Multiplicative weights (mirror descent with entropy) for soft-max objective.
    """
    n = len(a)
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    it = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * (progress ** 0.7)
        eta = eta_start * (eta_end / eta_start) ** progress
        
        conv = compute_conv(a)
        p = softmax_weights(conv, beta)
        grad = compute_grad_soft(a, p)
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            # Backtracking
            eta *= 0.7
            if stall > 50:
                # Reset and perturb
                a = best_a.copy()
                eta = eta_start * (1 - 0.5 * progress)
                pert = rng.normal(0, 0.001, n)
                a = a * np.exp(pert)
                a = a / float(np.sum(a))
                stall = 0
        
        a = a_new
        it += 1
        
        if verbose and it % 200 == 0:
            print(f"    MW it{it}: obj={best_obj:.8f}, beta={beta:.1f}, eta={eta:.4f}, t={elapsed:.1f}s")
    
    return best_a, best_obj

def nesterov_smax(a, time_budget=200.0, beta_start=10.0, beta_end=60.0,

        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * (progress ** 0.5)
        lr = lr_start * (1.0 - 0.4 * progress)
        mu = momentum_start + (momentum_end - momentum_start) * progress

        # y = a + (mu_t - 1) * (a - a_prev) ... simplified:
        # Use the standard Nesterov scheme:
        # t_k = (1 + sqrt(1 + 4*t_{k-1}^2)) / 2

        a_new = a - lr * v
        a_new = project_simplex(a_new)

                pert = rng.normal(0, 0.002, n)

                stall = 0

        it += 1
        
        if verbose and it % 300 == 0:
            print(f"    NSGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.5f}, t={elapsed:.1f}s")

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def line_search_blend(a, g_dir, time_budget=2.0):
    """Find optimal blend factor."""
    lo, hi = 0.0, 2.0
    best_t = 0.0
    best_obj = compute_obj(a)
    
    # Quick grid search
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        a_try = (1 - t) * a + t * g_dir

        if s < 0.01:
            continue
        obj = compute_obj(a_try / s)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    # Refine with golden section around best
    if best_t > 0:
        phi = (1 + np.sqrt(5)) / 2
        gr = 2 - phi
        span = 0.3
        lo = max(0, best_t - span)
        hi = best_t + span
        c = lo + gr * (hi - lo)
        d = lo + (1 - gr) * (hi - lo)
        
        def obj_at(t):
            a_try = (1 - t) * a + t * g_dir
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                return np.inf
            return compute_obj(a_try / s)
        
        fc = obj_at(c)
        fd = obj_at(d)
        
        start = time.monotonic()

            if time.monotonic() - start >= time_budget:
                break
            if fc < fd:

                d = c
                fd = fc
                c = lo + gr * (hi - lo)
                fc = obj_at(c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
                fd = obj_at(d)
        
        t_best = c if fc < fd else d
        obj_best = min(fc, fd)
        if obj_best < best_obj:

            break

Actually, let me think about coordinate descent more carefully. If I fix all coordinates except `a_i`, the objective becomes:
`max_k [(a*a)_k]` where `(a*a)_k` depends on `a_i` linearly (since `a_i` appears in `(a*a)_k` as `a_i * a_{k-i}` for each `k`).

Specifically, `(a*a)_k = a_i * a_{k-i} + (other terms not involving a_i)`.

So if I fix all `a_j` for `j ≠ i`, then `(a*a)_k` as a function of `a_i` is:
`(a*a)_k = a_i * a_{k-i} + c_k` (for `k` where `k-i` is in range)
`(a*a)_k = c_k` (for `k` where `k-i` is out of range)

And we want to minimize `max_k [(a*a)_k]` over `a_i >= 0` (with the normalization constraint `sum(a) = 1`).

This is a 1D problem! The optimal `a_i` is found by bisection: we want the largest `a_i` such that the max doesn't increase, or the smallest `a_i` that balances the active constraints.

This is exactly what the LP does in each iteration, but coordinate descent can be much faster since it's a 1D optimization at each step.

            if time.monotonic() - start >= time_budget:

            # (a*a)_k = a_i_old * a[k-i] + (terms not involving a_i)

```python
def coordinate_descent_fast(a, time_budget=60.0, verbose=False):
    """Fast coordinate descent exploiting the linear structure in each coordinate."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_obj = compute_obj(a)
    start = time.monotonic()
    sweep = 0
    
    while time.monotonic() - start < time_budget:
        for i in range(n):
            if time.monotonic() - start >= time_budget:
                break
            
            a_i = a[i]
            if a_i < 1e-10:
                continue
            
            # Base convolution: (a*a) with a[i] set to 0
            # (a*a)_k = base_k + delta_k
            # delta_k = 2*a_i*a[k-i] (if i != k-i, 0<=k-i<n) or a_i^2 (if k=2i)
            # More precisely:
            #   for k in [i, i+n-1] (clamped): delta_k = a_i * a[k-i] * mult
            #   where mult = 2 if k-i != i, 1 if k-i == i (i.e., k = 2i)
            
            # Compute base = conv(a with a[i]=0)
            a_base = a.copy()
            a_base[i] = 0.0
            base_conv = compute_conv(a_base)
            
            # Delta at each position k:
            # For k where 0 <= k-i < n:
            #   delta_k = a_i * a[k-i] + a_i * a[k-i] = 2*a_i*a[k-i] (if k-i != i)
            #   delta_k = a_i * a_i = a_i^2 (if k-i == i, i.e., k = 2i)
            # Actually, let me just compute it directly:
            # conv_full = base_conv + contribution from a[i]
            # contribution at k: a[i]*a[k-i] (from j=i) + a[k-i]*a[i] (from j=k-i)
            # = 2*a[i]*a[k-i] if i != k-i, a[i]^2 if i == k-i
            
            # For a new value t replacing a_i:
            # new_conv_k = base_conv_k + 2*t*a[k-i] (if k-i != i) or t^2 (if k = 2i)
            #           = base_conv_k + t*a[k-i] + t*a[k-i] (if k-i != i)
            #           = base_conv_k + t^2 (if k = 2i)
            
            # To find optimal t: minimize max_k [base_conv_k + f_k(t)]
            # where f_k(t) = 2*t*a[k-i] (for k in affected range, k != 2i)
            #                    t^2 (for k = 2i)
            #                    0 (for k not in affected range)
            
            # The affected range is k in [i, i+n-1] clamped to [0, 2n-2]
            k_min = max(0, i)
            k_max = min(2*n-2, i+n-1)
            
            # d[k] = coefficient of t in new_conv_k (for linear part)
            # For k in affected range:
            #   if k == 2*i: new_conv_k = base_conv_k + t^2
            #   else: new_conv_k = base_conv_k + 2*t*a[k-i]
            
            # For the linear part (k != 2*i): new_conv_k = base_conv_k + 2*t*a[k-i]
            # For k = 2*i: new_conv_k = base_conv_k + t^2
            
            # The max is: max(max_unaffected, max_linear(t), quadratic(t))
            # where max_unaffected = max of base_conv outside affected range
            # max_linear(t) = max over affected k (k != 2i) of [base_conv_k + 2*t*a[k-i]]
            # quadratic(t) = base_conv_{2i} + t^2 (if 2i is in range)
            
            # max_linear(t) = max over affected k (k != 2i) of [base_conv_k + 2*t*a[k-i]]
            # This is the max of linear functions of t, which is a convex piecewise linear function.
            # The optimal t satisfies: the largest of these equals the quadratic (if active)
            # or equals the max_unaffected.
            
            # For simplicity, use golden section on t in [0, 2*a_i]:
            
            # Precompute:
            affected_indices = np.arange(k_min, k_max + 1)
            # d[k] = 2*a[k-i] for k != 2*i, and we handle k=2*i separately
            d = np.zeros(k_max - k_min + 1)
            for idx, k in enumerate(affected_indices):
                ki = k - i  # this is 0 <= ki < n
                if k == 2 * i:
                    pass  # handle separately
                else:
                    d[idx] = 2.0 * a[ki]
            
            has_quad = (2 * i <= 2 * n - 2) and (0 <= 2 * i < 2 * n - 1)
            if has_quad:
                quad_base = base_conv[2 * i]
            
            # Unaffected max
            if k_min == 0 and k_max == 2 * n - 2:
                max_unaffected = 0.0
            else:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_min:k_max+1] = False
                if np.any(mask):
                    max_unaffected = float(np.max(base_conv[mask]))
                else:
                    max_unaffected = 0.0
            
            # Objective function for golden section:
            def obj_at_t(t):
                # Compute max over all k
                # Linear part: base_conv[k_min:k_max+1] + t * d (excluding quad position)
                vals = base_conv[k_min:k_max+1].copy()
                for idx in range(len(vals)):
                    k = affected_indices[idx]
                    if k == 2 * i:
                        vals[idx] = quad_base + t * t
                    else:
                        vals[idx] += t * d[idx]
                local_max = float(np.max(vals))
                return max(local_max, max_unaffected)
            
            # Golden section search for t in [0, 2*a_i]
            lo_t = 0.0
            hi_t = 2.0 * a_i
            if hi_t < 1e-10:
                hi_t = 0.01 * float(np.max(a))
            
            phi = (1 + np.sqrt(5)) / 2
            gr = 2 - phi
            c = lo_t + gr * (hi_t - lo_t)
            d_gt = lo_t + (1 - gr) * (hi_t - lo_t)
            fc = obj_at_t(c)
            fd = obj_at_t(d_gt)
            
            for _ in range(12):
                if fc < fd:
                    hi_t = d_gt
                    d_gt = c
                    fd = fc
                    c = lo_t + gr * (hi_t - lo_t)
                    fc = obj_at_t(c)
                else:
                    lo_t = c
                    c = d_gt
                    fc = fd
                    d_gt = lo_t + gr * (hi_t - lo_t)
                    fd = obj_at_t(d_gt)
            
            t_opt = c if fc < fd else d_gt
            a[i] = t_opt
        
        # Normalize
        s = float(np.sum(a))
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        
        obj_now = compute_obj(a)
        if obj_now < best_obj:
            best_obj = obj_now
        
        sweep += 1
        if verbose and sweep % 3 == 0:
            print(f"    CD sweep {sweep}: obj={obj_now:.8f}, t={time.monotonic()-start:.1f}s")
        
        # Early stopping
        if sweep > 10 and obj_now >= best_obj:
            break
    
    return a, best_obj
```

Wait, this is still slow because we're computing `base_conv` for each coordinate `i`, which requires an FFT of O(n log n). With n=800 coordinates, that's 800 FFTs per sweep. Each FFT of size ~2048 is fast (~10us), so 800 * 10us = 8ms per sweep. That's fine!

Actually, I realize there's an even faster way. Instead of computing the full base convolution for each `i`, I can:
1. Compute the full convolution `conv` once at the start of each sweep.
2. For coordinate `i`, the base convolution is `conv - contribution_of_i`.
3. The contribution of `i` is a known function of `a_i` and `a`.

Let me optimize:
