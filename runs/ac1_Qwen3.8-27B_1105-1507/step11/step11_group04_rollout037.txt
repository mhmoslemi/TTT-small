import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)

    n = len(a)

    return 2.0 * n * float(np.max(conv)) / (s * s)

            break
        a_try = a_try / s_try

        # Ensure non-negativity (should be automatic with exp, but clip for safety)
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            break
        a_try = a_try / s_try

        obj_try = compute_obj_norm(a_try)

        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()
            no_improve = 0
            # Adapt parameters: increase beta to sharpen focus
            beta = min(beta * 1.015, 300.0)
            # Slightly reduce gamma as we converge (less need for variance penalty)
            gamma = max(gamma * 0.995, 0.0)
            # Keep eta or slightly reduce
            eta = max(eta * 0.9995, 0.005)
        else:
            no_improve += 1
            # Reduce step size
            eta = eta * 0.92
            if no_improve >= max_no_improve:
                # Restart from best with perturbation
                a = best_a.copy()
                # Add structured perturbation
                strength = 0.005 + 0.003 * min(stall_count, 5)
                freqs = np.arange(n)
                for k in range(1, 5):
                    amp = strength * rng.uniform(0.5, 1.0) / k
                    phase = rng.uniform(0, 2 * np.pi)
                    a = a * np.exp(amp * np.cos(2 * np.pi * k * freqs / n + phase))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                # Reset parameters
                beta = 15.0
                eta = best_eta
                gamma = 0.15
                no_improve = 0
                stall_count += 1
                max_no_improve = min(max_no_improve + 10, 80)

        # Periodic verbose output
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD iter {iter_count}: obj={best_obj:.10f}, beta={beta:.1f}, "
                  f"eta={eta:.4f}, gamma={gamma:.3f}, elapsed={elapsed:.0f}s")

    if verbose:
        print(f"    MD done: {iter_count} iters, best={best_obj:.10f}")
    return best_a


def coordinate_descent_polish(a, time_budget, seed=42, verbose=False):
    """
    Exact coordinate descent: for each coordinate, try multiple candidate values
    and pick the one that minimizes the objective. Uses incremental convolution updates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    sweep = 0
    no_improve = 0
    max_no_improve = 4
    mean_val = 1.0 / n

    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            a_j = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)

            # Unaffected max
            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))

            # Candidate values
            if a_j < 1e-12:
                candidates = np.array([0.0, mean_val * 0.05, mean_val * 0.2, mean_val * 0.5, mean_val])
            else:
                candidates = np.array([
                    a_j * 0.3, a_j * 0.5, a_j * 0.7, a_j * 0.9,
                    a_j * 1.1, a_j * 1.3, a_j * 1.5, a_j * 2.0,
                    mean_val * 0.2, mean_val * 0.5, mean_val, mean_val * 2, mean_val * 5
                ])
                candidates = candidates[(candidates >= 0) & (candidates <= 0.5)]
                candidates = np.unique(candidates)

            best_trial_obj = best_obj
            best_trial_val = a_j

            for val in candidates:
                delta = val - a_j
                if abs(delta) < 1e-15:
                    continue

                S_new = 1.0 + delta
                if S_new <= 1e-10:
                    continue

                # Compute max of affected conv entries (before normalization)
                indices = np.arange(k_start, k_end + 1)
                affected = conv[k_start:k_end + 1].copy()
                mask = (indices == 2 * j)
                if np.any(~mask):
                    affected[~mask] = conv[k_start:k_end + 1][~mask] + 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    affected[mask] = conv[2 * j] + (val * val - a_j * a_j)

                max_conv_unscaled = max(max_unaffected, float(np.max(affected)))
                obj_trial = 2.0 * n * max_conv_unscaled / (S_new * S_new)

                if obj_trial < best_trial_obj - 1e-13:
                    best_trial_obj = obj_trial
                    best_trial_val = val

            if best_trial_val != a_j:
                delta = best_trial_val - a_j
                a[j] = best_trial_val
                S_new = 1.0 + delta

                indices = np.arange(k_start, k_end + 1)
                mask = (indices == 2 * j)
                if np.any(~mask):
                    conv[k_start:k_end + 1][~mask] += 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    conv[2 * j] += (best_trial_val * best_trial_val - a_j * a_j)

                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S

                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")

    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 8):
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(2 * np.pi * k * np.arange(n) / n + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses multiplicative gradient descent with
    adaptive parameters, coordinate descent polishing, and multi-restart
    exploration across different sequence lengths.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    # --- Initialization ---
    rng = np.random.default_rng(seed)

    # Try to load parent
    try:
        a_parent = np.asarray(height_sequence_1, dtype=np.float64)
        a_parent = np.maximum(a_parent, 0)
        if a_parent.size == 0 or a_parent.sum() < 0.01:
            raise ValueError("Invalid parent")
        a_parent = a_parent / a_parent.sum()
        n_parent = a_parent.size
        print(f"Parent: n={n_parent}, obj={compute_obj_norm(a_parent):.10f}")
    except Exception:
        a_parent = None
        n_parent = 800
        print(f"No valid parent, using n=800")

    best_a = None
    best_obj = np.inf

    # Define candidate lengths to try
    n_candidates = []
    if n_parent is not None:
        n_candidates.append(n_parent)
    n_candidates.extend([400, 500, 600, 700, 800, 900, 1000, 1200])
    n_candidates = sorted(set(n for n in n_candidates if 200 <= n <= 2000))

    print(f"\nCandidate lengths: {n_candidates}")
    print(f"Time budget: {budget_s}s")

    # --- Phase 1: Initial multiplicative descent on parent (or uniform) ---
    print("\n=== Phase 1: Initial Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        if a_parent is not None:
            a0 = a_parent.copy()
        else:
            a0 = np.ones(n_parent) / n_parent
        p1_budget = min(remaining * 0.35, 400.0)
        a1 = multiplicative_descent(a0, time_budget=p1_budget, seed=seed, verbose=True, gamma_var=0.1)
        obj1 = compute_obj_norm(a1)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()
            print(f"  P1: obj={best_obj:.10f}, n={len(best_a)} **")
        else:
            print(f"  P1: obj={obj1:.10f}, n={len(a1)}")

    # --- Phase 2: Coordinate descent polish ---
    print("\n=== Phase 2: CD Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60 and best_a is not None:
        p2_budget = min(remaining * 0.25, 250.0)
        a2 = coordinate_descent_polish(best_a, time_budget=p2_budget, seed=seed + 100, verbose=True)
        obj2 = compute_obj_norm(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            print(f"  P2: obj={best_obj:.10f} **")
        else:
            print(f"  P2: obj={obj2:.10f}")

    # --- Phase 3: Second multiplicative descent with different params ---
    print("\n=== Phase 3: Secondary Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 80 and best_a is not None:
        # Perturb and run again with different hyperparameters
        a3_init = fourier_perturb(best_a, rng, strength=0.008)
        p3_budget = min(remaining * 0.25, 300.0)
        a3 = multiplicative_descent(a3_init, time_budget=p3_budget, seed=seed + 5000,
                                    verbose=True, gamma_var=0.25)
        obj3 = compute_obj_norm(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            print(f"  P3: obj={best_obj:.10f} **")
        else:
            print(f"  P3: obj={obj3:.10f}")

    # --- Phase 4: CD polish again ---
    print("\n=== Phase 4: CD Polish 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 40 and best_a is not None:
        p4_budget = min(remaining * 0.2, 120.0)
        a4 = coordinate_descent_polish(best_a, time_budget=p4_budget, seed=seed + 9000, verbose=False)
        obj4 = compute_obj_norm(a4)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()
            print(f"  P4: obj={best_obj:.10f} **")
        else:
            print(f"  P4: obj={obj4:.10f}")

    # --- Phase 5: Length exploration ---
    print("\n=== Phase 5: Length Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a) if best_a is not None else 800

    # Try a few neighboring lengths
    n_options = []
    for delta in [-100, -50, -20, 20, 50, 100, 200]:
        n_try = n_best + delta
        if 200 <= n_try <= 2000 and n_try != n_best:
            n_options.append(n_try)
    # Also try some standard lengths
    for n_std in [400, 500, 600, 1000]:
        if n_std != n_best and 200 <= n_std <= 2000:
            if n_std not in n_options:
                n_options.append(n_std)
    n_options = sorted(n_options)[:6]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        # Interpolate best sequence to new length
        x_old = np.linspace(0, 1, len(best_a))
        x_new = np.linspace(0, 1, n_try)
        a_try = np.interp(x_new, x_old, best_a)
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # Run multiplicative descent on new length
        alloc = min(remaining * 0.3, 150.0)
        if alloc > 50:
            a_try = multiplicative_descent(a_try, time_budget=alloc * 0.6, seed=n_try * 7,
                                           verbose=False, gamma_var=0.15)
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_try = coordinate_descent_polish(a_try, time_budget=min(remaining * 0.25, 40.0),
                                                  seed=n_try * 13, verbose=False)

        obj_try = compute_obj_norm(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            print(f"  ** n={n_try}: obj={best_obj:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # --- Phase 6: Final intensive optimization on best ---
    print("\n=== Phase 6: Final Optimization ===")
    remaining = deadline - time.monotonic()
    if remaining > 30 and best_a is not None:
        # Run multiplicative descent one more time with low gamma
        p6_budget = min(remaining * 0.5, 150.0)
        a6 = multiplicative_descent(best_a, time_budget=p6_budget, seed=seed + 7777,
                                    verbose=False, gamma_var=0.05)
        obj6 = compute_obj_norm(a6)
        if obj6 < best_obj:
            best_obj = obj6
            best_a = a6.copy()
            print(f"  P6 MD: obj={best_obj:.10f} **")

        # Final CD
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_fin = coordinate_descent_polish(best_a, time_budget=min(remaining * 0.7, 80.0),
                                              seed=seed + 42424, verbose=False)
            obj_fin = compute_obj_norm(a_fin)
            if obj_fin < best_obj:
                best_obj = obj_fin
                best_a = a_fin.copy()
                print(f"  P6 CD: obj={best_obj:.10f} **")

    # --- Final validation and return ---
    if best_a is None:
        best_a = np.ones(100) / 100.0
        best_obj = 2.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
