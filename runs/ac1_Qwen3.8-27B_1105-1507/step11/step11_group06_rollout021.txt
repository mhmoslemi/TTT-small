def peak_shaving(a, time_budget=30.0):
    """Coordinate-level peak shaving: transfer mass from high-gradient to low-gradient coordinates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    alpha = 0.1
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Compute gradient
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            j2 = i - k_star
            if 0 <= j1 < n:
                grad[i] += a[j1]
            if 0 <= j2 < n:
                grad[i] += a[j2]
        
        # Normalize gradient
        g_max = grad.max()
        if g_max < 1e-15:
            break
        grad = grad / g_max
        
        # Transfer mass: decrease high-gradient, increase low-gradient
        # Use softmax-like weighting
        threshold = 0.5
        high_mask = grad > threshold
        low_mask = grad < (1 - threshold)
        
        if not np.any(high_mask) or not np.any(low_mask):
            alpha *= 0.9
            continue
            
        mass_out = alpha * a[high_mask].sum() * 0.1  # small fraction
        mass_out = min(mass_out, 0.1 * a[high_mask].sum())  # don't remove too much
        
        # Reduce high-gradient coordinates proportionally
        reduce_factor = mass_out / a[high_mask].sum() if a[high_mask].sum() > 0 else 0
        a[high_mask] -= a[high_mask] * reduce_factor
        a[high_mask] = np.maximum(a[high_mask], 0)
        
        # Distribute to low-gradient coordinates proportionally to (1 - grad)
        low_weights = (1 - grad[low_mask])
        low_weights = np.maximum(low_weights, 1e-10)
        a[low_mask] += mass_out * low_weights / low_weights.sum()
        
        # Renormalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        else:
            a = best_a.copy()
            alpha *= 0.95
    
    return best_a, best_obj

"""
Multi-peak Nesterov-accelerated subgradient with LP polish, IRLS quadratic
subproblem solving, peak shaving, and diverse n-exploration.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def compute_weighted_gradient(a, conv, K=3):
    """Compute gradient as weighted average of top-K peak gradients."""
    n = len(a)
    top_k = np.argsort(conv)[-K:][::-1]
    weights = conv[top_k].copy()
    w_sum = weights.sum()
    if w_sum < 1e-15:
        weights = np.ones(K) / K
    else:
        weights = weights / w_sum
    grad = np.zeros(n)
    for idx in range(K):
        k = int(top_k[idx])
        w = weights[idx]
        for i in range(n):
            j1 = k - i
            j2 = i - k
            if 0 <= j1 < n:
                grad[i] += w * a[j1]
            if 0 <= j2 < n:
                grad[i] += w * a[j2]
    return grad


def nesterov_subgradient(a, time_budget=100.0, K=3):
    """Nesterov-accelerated multi-peak subgradient descent."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    momentum = np.zeros(n)
    lr0 = 0.1
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        grad = compute_weighted_gradient(y, conv, K=K)

        # Nesterov step size: decays as 1/sqrt(it)
        t_k = it + 1
        lr_t = lr0 / np.sqrt(t_k)

        # Nesterov momentum schedule
        beta = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        beta_next = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
        momentum_factor = (beta - 1.0) / beta_next

        momentum = momentum_factor * momentum - lr_t * grad
        y_new = a + momentum
        y_new = np.maximum(y_new, 0)
        s_new = y_new.sum()
        if s_new < 1e-10:
            y_new = best_a.copy()
            momentum = np.zeros(n)
        else:
            y_new = y_new / s_new

        # Ergodic averaging
        a = (it / (it + 1.0)) * a + (1.0 / (it + 1.0)) * y_new

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            best_it = it

        it += 1
        if it - best_it > 2000:
            # Restart from best with reduced learning rate
            a = best_a.copy()
            y = best_a.copy()
            momentum = np.zeros(n)
            lr0 = max(0.01, lr0 * 0.5)
            best_it = it
            it = 0

    return best_a, best_obj

        return a, np.inf

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    alpha = 0.05
    no_improve = 0

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            j2 = i - k_star
            if 0 <= j1 < n:
                grad[i] += a[j1]
            if 0 <= j2 < n:
                grad[i] += a[j2]

        g_max = grad.max()
        if g_max < 1e-15:

        grad_norm = grad / g_max

        # Identify high and low gradient coordinates
        high_thresh = np.percentile(grad_norm, 75)
        low_thresh = np.percentile(grad_norm, 25)
        high_mask = grad_norm > high_thresh
        low_mask = grad_norm < low_thresh

        if not np.any(high_mask) or not np.any(low_mask):
            alpha *= 0.9
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Transfer small amount of mass
        mass = alpha * a[high_mask].sum() * 0.05
        reduce_amt = mass / a[high_mask].sum() * grad_norm[high_mask]
        a[high_mask] -= reduce_amt
        a[high_mask] = np.maximum(a[high_mask], 0)

        low_weights = np.maximum(1.0 - grad_norm[low_mask], 1e-10)
        a[low_mask] += mass * low_weights / low_weights.sum()

        a = np.maximum(a, 0)
        s_new = a.sum()
        if s_new < 1e-10:
            a = best_a.copy()
            break
        a = a / s_new

        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            a = best_a.copy()
            alpha *= 0.95
            no_improve += 1
            if no_improve > 100:
                break

    return best_a, best_obj


def lp_polish(a, time_budget=100.0, max_iters=80):
    """LP-based polishing with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng_restart = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        # Exhaustive line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in np.linspace(0, 1.2, 25):
            a_try = np.maximum((1 - t) * a + t * g_lp, 0)
            s_try = a_try.sum()
            if s_try < 0.01:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t

        # Fine search around best
        if 0.01 < best_t < 1.19:
            for t in np.linspace(max(0, best_t - 0.05), min(1.2, best_t + 0.05), 15):
                a_try = np.maximum((1 - t) * a + t * g_lp, 0)
                s_try = a_try.sum()
                if s_try < 0.01:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj_ls:
                    best_obj_ls = obj_try
                    best_t = t

        if best_t > 1e-6 and best_obj_ls < best_obj - 1e-14:
            a_new = np.maximum((1 - best_t) * a + best_t * g_lp, 0)
            a_new = a_new / a_new.sum()
            best_obj = best_obj_ls
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            # Try direct LP solution
            obj_direct = compute_obj(g_lp)
            if obj_direct < best_obj - 1e-14:
                best_obj = obj_direct
                best_a = g_lp.copy()
                a = g_lp.copy()
                stall = 0

        it += 1
        if stall >= 4:
            # Perturb to escape
            if stall <= 8:
                freq = rng_restart.integers(1, max(2, n // 100))
                phase = rng_restart.uniform(0, 2 * np.pi)
                idx = np.arange(n, dtype=np.float64)
                pert = 0.02 * stall * np.cos(2 * np.pi * freq * idx / n + phase)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                s_p = a.sum()
                if s_p > 0.01:
                    a = a / s_p
                else:
                    a = best_a.copy()
            else:
                break
            stall = 0

    return best_a, best_obj


def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n, dtype=np.float64)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n / 4) ** 2) / (2 * (n / 6) ** 2)) + \
            np.exp(-((idx - 3 * n / 4) ** 2) / (2 * (n / 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.02 + 5.0 * (2 * x - 1) ** 4)
    elif mode == 6:
        a = np.cos(np.pi * x) ** 2 + 0.01
    elif mode == 7:
        a = 1.0 / (0.08 + 3.0 * (2 * x - 1) ** 2)
    else:
        a = 1.0 / (0.1 + 2.0 * np.abs(2 * x - 1))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def u_shaped_init(n, sharpness=3.0):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s


def propose_candidate(seed=42, budget_s=1500):
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        s = float(np.sum(parent_arr))
        if s > 0.01:
            best_seq = parent_arr / s
            best_score = compute_obj(best_seq)
            print(f"Parent loaded: n={parent_arr.size}, obj={best_score:.8f}")
        else:
            raise ValueError("Sum too small")
    except Exception:
        best_seq = u_shaped_init(800, sharpness=3.0)
        best_score = compute_obj(best_seq)
        print(f"Default init: n={len(best_seq)}, obj={best_score:.8f}")

    n_best = len(best_seq)

    # Try alternative initializations at same n
    for mode in range(8):
        a_alt = diverse_init(n_best, rng, mode=mode)
        obj_alt = compute_obj(a_alt)
        if obj_alt < best_score:
            best_score = obj_alt
            best_seq = a_alt.copy()
            print(f"  Alt init mode={mode}: obj={best_score:.8f} **")

    # Phase 1: Nesterov subgradient (long, K=3)
    remaining = deadline - time.monotonic()
    if remaining > 80:
        alloc = min(remaining * 0.35, 400.0)
        print(f"\n=== Phase 1: Nesterov SG K=3 (budget={alloc:.0f}s) ===")
        t0 = time.monotonic()
        a_sg, obj_sg = nesterov_subgradient(best_seq, time_budget=alloc, K=3)
        print(f"  Time: {time.monotonic()-t0:.1f}s")
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_sg:.8f} (no improvement)")

    # Phase 2: IRLS quadratic
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(remaining * 0.15, 100.0)
        print(f"\n=== Phase 2: IRLS Quadratic (budget={alloc:.0f}s) ===")
        t0 = time.monotonic()
        a_irls, obj_irls = irls_quadratic(best_seq, time_budget=alloc, n_active=5)
        print(f"  Time: {time.monotonic()-t0:.1f}s")
        if obj_irls < best_score:
            best_score = obj_irls
            best_seq = a_irls.copy()
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj_irls:.8f} (no improvement)")

    # Phase 3: LP polish
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.25, 200.0)
        print(f"\n=== Phase 3: LP Polish (budget={alloc:.0f}s) ===")
        t0 = time.monotonic()
        a_lp, obj_lp = lp_polish(best_seq, time_budget=alloc, max_iters=100)
        print(f"  Time: {time.monotonic()-t0:.1f}s")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj_lp:.8f} (no improvement)")

    # Phase 4: Nesterov subgradient round 2 (K=5 for smoother)
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.2, 150.0)
        print(f"\n=== Phase 4: Nesterov SG K=5 (budget={alloc:.0f}s) ===")
        t0 = time.monotonic()
        a_sg2, obj_sg2 = nesterov_subgradient(best_seq, time_budget=alloc, K=5)
        print(f"  Time: {time.monotonic()-t0:.1f}s")
        if obj_sg2 < best_score:
            best_score = obj_sg2
            best_seq = a_sg2.copy()
            print(f"  Phase 4: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4: obj={obj_sg2:.8f} (no improvement)")

    # Phase 5: n exploration
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 5: n Exploration ===")
        n_cur = len(best_seq)
        n_options = sorted(set([
            n_cur - 100, n_cur - 50, n_cur + 50, n_cur + 100,
            n_cur + 200, n_cur + 300, n_cur + 500,
            600, 700, 900, 1000, 1100, 1300, 1500, 2000
        ]))
        n_options = [n for n in n_options if 400 <= n <= 2500 and abs(n - n_cur) >= 40]
        n_options.sort(key=lambda x: abs(x - n_cur))
        n_options = n_options[:8]
        print(f"  Candidates: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            a_try = interpolate_sequence(best_seq, n_try)
            if a_try is None:
                continue
            alloc_sg = min(remaining * 0.25, 50.0)
            a_try_sg, obj_try_sg = nesterov_subgradient(a_try, time_budget=alloc_sg, K=3)
            if obj_try_sg < best_score:
                best_score = obj_try_sg
                best_seq = a_try_sg.copy()
                print(f"  ** n={n_try} SG: obj={best_score:.8f} **")

            remaining2 = deadline - time.monotonic()
            if remaining2 > 30:
                alloc_lp = min(remaining2 * 0.3, 40.0)
                a_try_lp, obj_try_lp = lp_polish(a_try_sg, time_budget=alloc_lp, max_iters=30)
                if obj_try_lp < best_score:
                    best_score = obj_try_lp
                    best_seq = a_try_lp.copy()
                    print(f"  ** n={n_try} LP: obj={best_score:.8f} **")
                else:
                    print(f"  n={n_try}: obj={obj_try_lp:.8f}")

    # Phase 6: Final LP polish
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.4, 100.0)
        print(f"\n=== Phase 6: Final LP (budget={alloc:.0f}s) ===")
        a_final, obj_final = lp_polish(best_seq, time_budget=alloc, max_iters=50)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Phase 6: obj={best_score:.8f} **")
        else:
            print(f"  Phase 6: obj={obj_final:.8f}")

    # Phase 7: Peak shaving
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 40.0)
        print(f"\n=== Phase 7: Peak Shaving (budget={alloc:.0f}s) ===")
        a_pv, obj_pv = peak_shaving(best_seq, time_budget=alloc)
        if obj_pv < best_score:
            best_score = obj_pv
            best_seq = a_pv.copy()
            print(f"  Phase 7: obj={best_score:.8f} **")
        else:
            print(f"  Phase 7: obj={obj_pv:.8f}")

    # Phase 8: Final short Nesterov
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 50.0)
        print(f"\n=== Phase 8: Final SG (budget={alloc:.0f}s) ===")
        a_sq, obj_sq = nesterov_subgradient(best_seq, time_budget=alloc, K=5)
        if obj_sq < best_score:
            best_score = obj_sq
            best_seq = a_sq.copy()
            print(f"  Phase 8: obj={best_score:.8f} **")
        else:
            print(f"  Phase 8: obj={obj_sq:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
