"""
Hybrid optimizer for step function coefficients minimizing 2n*max(a*a)/(sum(a))^2.
Uses: (1) single-coordinate descent, (2) pairwise coordinate descent,
(3) simulated annealing basin hopping, (4) LP direction polish,
(5) multi-n exploration with diverse restarts.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective 2n*max(a*a)/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return float(2.0 * n * np.max(conv) / (s * s))


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full(len(a), 1.0 / len(a))
    return a / s


def cd_single(a, budget_s=100.0, verbose=False, seed=42):
    """Single-coordinate descent with golden section refinement."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0
    stall = 0

    while time.monotonic() - start < budget_s:
        sweep += 1
        order = rng.permutation(n)
        S = float(np.sum(a))
        improved = False

        for i in order:
            if time.monotonic() - start >= budget_s:
                break
            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = np.zeros(len(ks))
            valid = (ks - i >= 0) & (ks - i < n)
            a_ki[valid] = a[ks[valid] - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            scale = max(v, S / n, 1e-6)
            candidates = [0.0, v * 0.5, v, v * 1.5, v * 2.0, S / n, 2.0 * S / n]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            def f_1d(x):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x)
                if denom < 1e-15:
                    return 1e30
                return max_val / (denom * denom)

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            x_lo = max(0.0, best_x - scale * 0.5)

            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-10

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

                if x_hi - x_lo < 1e-14 * max(1.0, x_hi):

                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-15:
                continue

            delta = x_opt - v
            a[i] = x_opt
            if 0 <= k_quad <= 2 * n - 2:

            if len(ks_lin) > 0:

    stall = 0

                for _ in range(30):
                    if hi - lo < 1e-13:
                        break
                    if fc < fd:
                        hi = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = hi - gr * (hi - lo)
                        fc = float((conv + 2.0 * c_pt * b + c_pt ** 2 * d).max())
                    else:
                        lo = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = lo + gr * (hi - lo)
                        fd = float((conv + 2.0 * d_pt * b + d_pt ** 2 * d).max())
                delta_opt = (lo + hi) / 2.0

            new_ai = a[i] + delta_opt
            new_aj = a[j] - delta_opt
            if new_ai < 0:
                new_ai = 0.0
                new_aj = s_ij
            if new_aj < 0:
                new_aj = 0.0
                new_ai = s_ij

            if abs(new_ai - a[i]) < 1e-14:
                continue

            delta_i = new_ai - a[i]
            delta_j = new_aj - a[j]
            a[i] = new_ai
            a[j] = new_aj

            # Update conv
            for k in range(2 * n - 1):
                ki = k - i
                kj = k - j
                old_val = conv[k]
                new_val = old_val
                # Linear terms
                if 0 <= ki < n:
                    new_val += delta_i * a[ki]
                if 0 <= kj < n:
                    new_val += delta_j * a[kj]
                # Quadratic/cross terms
                if k == 2 * i:
                    new_val += new_ai ** 2 - old_ai_sq(a, old_val, i, j, delta_i, delta_j, ki, kj)
                # Actually let's just recompute affected entries
            # Simpler: recompute conv for affected k's
            # Affected k: those where i or j is involved
            # k ranges: [i, i+n-1] union [j, j+n-1] union {2i, 2j, i+j}
            k_affected = set()
            k_affected.update(range(max(0, i), min(2 * n - 1, i + n)))
            k_affected.update(range(max(0, j), min(2 * n - 1, j + n)))
            k_affected.add(min(2 * i, 2 * n - 2))
            k_affected.add(min(2 * j, 2 * n - 2))
            k_affected.add(min(i + j, 2 * n - 2))

            # Recompute these entries
            # Actually this is expensive. Let's just update using the formula:
            # Δ(a*a)[k] = δ_i*a[k-i] + δ_j*a[k-j] + δ_i*a[k-i] + δ_j*a[k-j] + δ_i*δ_j*(1_{k=i+j})*... 
            # Wait, the formula is:
            # Δ = δ_i*a_old[k-i] + δ_j*a_old[k-j] + δ_i*δ_j*1_{k=i+j} + δ_i^2*1_{k=2i} + δ_j^2*1_{k=2j}
            # But we also need to account for the fact that a[k-i] might be a[i] or a[j] themselves.
            # Let's use the old values:
            a_old_i = a[i] - delta_i  # this is new_ai - delta_i = old a[i]
            # Actually, let's just recompute affected entries directly
            ks = sorted(k_affected)
            for k in ks:
                val = 0.0
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for p in range(lo, hi + 1):
                    q = k - p
                    val += a[p] * a[q]
                conv[k] = val

            S = float(np.sum(a))
            improved = True
            obj_now = 2.0 * n * float(np.max(conv)) / (S * S)
            if obj_now < best_obj:
                best_obj = obj_now
                best_a = a.copy()

        # Periodic full recompute
        if sweep % 3 == 0:
            conv = compute_conv(a)
            S = float(np.sum(a))
            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    a = a / S

        if not improved:
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0

        if verbose and sweep % 5 == 0:
            print(f"      PairCD sweep {sweep}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

    return normalize(best_a)

    """Helper (unused, kept for interface)."""

def basin_hopping(a, budget_s=200.0, verbose=False, seed=42):
    """Simulated annealing with pairwise CD as local search."""
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    T_init = 0.05
    T_end = 0.001
    n_iter_max = 5000
    total_budget = budget_s

    it = 0
    while time.monotonic() - start < total_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / total_budget)
        T = T_init * (T_end / T_init) ** progress

        # Perturb: modify a random fraction of coordinates
        frac = 0.1 + 0.3 * progress  # More perturbation later
        n_perturb = max(5, int(n * frac))
        idx = rng.choice(n, size=n_perturb, replace=False)

        noise_scale = 0.1 * (1.0 + progress)
        a_pert = a.copy()
        a_pert[idx] += rng.normal(0, noise_scale, n_perturb)
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        # Local search: brief pairwise CD
        local_budget = 3.0
        a_local = cd_pairwise(a_pert, budget_s=local_budget, seed=seed + it, n_pairs_per_sweep=100)
        obj_local = 2.0 * n_local_obj(a_local)

        delta = obj_local - best_obj
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-10)):
            a = a_local
            if obj_local < best_obj:
                best_obj = obj_local
                best_a = a_local.copy()
                if verbose and it % 50 == 0:
                    print(f"      BH it{it}: obj={best_obj:.8f}, T={T:.6f}, t={elapsed:.1f}s")
        it += 1

    return best_a


def n_local_obj(a):
    n = len(a)
    conv = compute_conv(a)
    s = float(np.sum(a))
    return float(np.max(conv)) / (s * s) * 2.0 * n


def solve_lp_direction(a):
    """Solve LP to find best direction g: min M s.t. conv(a,g)<=M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))

    # Build convolution matrix A where (A@g)[k] = (a_norm * g)[k]
    # A is (2n-1) x n, A[k,j] = a_norm[k-j] if 0<=k-j<n else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r_start = j
        r_end = min(2 * n - 1, j + n)
        A[r_start:r_end, j] = a_norm[0:r_end - r_start]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:
                    g_val = g_val / g_sum
                    return g_val
        except Exception:
            continue
    return None


def lp_polish(a, budget_s=100.0, verbose=False, seed=42):
    """LP-guided direction with line search, alternating with CD."""
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < budget_s:
        remaining = budget_s - (time.monotonic() - start)
        if remaining < 10:
            break

        g = solve_lp_direction(a)
        if g is None:
            break

        # Line search on t
        best_t = 0.0
        best_t_obj = best_obj
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t > 0 and best_t_obj < best_obj - 1e-12:
            a = (1.0 - best_t) * a + best_t * g
            a = normalize(a)
            best_obj = best_t_obj
            best_a = a.copy()
            stall = 0
            if verbose:
                print(f"      LP t={best_t:.2f}: obj={best_obj:.8f}")
        else:
            stall += 1
            if stall >= 3:
                # Perturb and retry
                pert = rng.normal(0, 0.02, n)
                a = a * np.exp(pert)
                a = normalize(a)
                stall = 0
            elif stall >= 2:
                # Try CD for a bit
                a = cd_pairwise(a, budget_s=min(10, remaining * 0.2), seed=seed + stall)
                obj_cd = 2.0 * n * float(np.max(compute_conv(a)))
                if obj_cd < best_obj:
                    best_obj = obj_cd
                    best_a = a.copy()

    return normalize(best_a)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        a = np.zeros(n)
        w = n // 5
        a[:w] = np.exp(-((np.arange(w) - w // 2) ** 2) / (2 * (w // 4) ** 2))
        a[-w:] = np.exp(-((np.arange(w) - w // 2) ** 2) / (2 * (w // 4) ** 2))
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x * x + 1e-6)
    elif mode == 4:
        freqs = rng.integers(1, 10, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = np.maximum(a, 0)
    elif mode == 7:
        # Smoothed random with correlation
        a = np.zeros(n)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = 0.9 * a[i - 1] + rng.normal(0, 0.3)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n) / n
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main optimization: combines single CD, pairwise CD, basin hopping,
    LP polish, and multi-n exploration to minimize the step function objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = 2.0 * parent.size * float(np.max(compute_conv(parent)))
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(800) / 800.0
        best_score = 2.0
        print(f"Default init (n=800): obj=2.0")

    n_cur = len(best_seq)

    # === Phase 1: Single CD (quick convergence) ===
    print(f"\n=== Phase 1: Single CD (n={n_cur}) ===")
    remaining = deadline - time.monotonic()
    p1 = min(remaining * 0.20, 250.0)
    if p1 > 20:
        a1 = cd_single(best_seq, budget_s=p1, verbose=True, seed=seed)
        obj1 = 2.0 * a1.size * float(np.max(compute_conv(a1)))
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Single CD: {best_score:.8f} **")
        else:
            print(f"  Single CD: {obj1:.8f}")

    # === Phase 2: Pairwise CD (stronger local search) ===
    print(f"\n=== Phase 2: Pairwise CD ===")
    remaining = deadline - time.monotonic()
    p2 = min(remaining * 0.25, 350.0)
    if p2 > 20:
        a2 = cd_pairwise(best_seq, budget_s=p2, verbose=True, seed=seed + 100,
                         n_pairs_per_sweep=200)
        obj2 = 2.0 * a2.size * float(np.max(compute_conv(a2)))
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Pairwise CD: {best_score:.8f} **")
        else:
            print(f"  Pairwise CD: {obj2:.8f}")

    # === Phase 3: Basin Hopping (explore) ===
    print(f"\n=== Phase 3: Basin Hopping ===")
    remaining = deadline - time.monotonic()
    p3 = min(remaining * 0.25, 300.0)
    if p3 > 30:
        a3 = basin_hopping(best_seq, budget_s=p3, verbose=True, seed=seed + 200)
        obj3 = 2.0 * a3.size * float(np.max(compute_conv(a3)))
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Basin Hopping: {best_score:.8f} **")
        else:
            print(f"  Basin Hopping: {obj3:.8f}")
            # Try CD from BH result anyway
            if remaining > 30:
                a3b = cd_pairwise(a3, budget_s=min(30, remaining * 0.1),
                                  seed=seed + 300, n_pairs_per_sweep=150)
                obj3b = 2.0 * a3b.size * float(np.max(compute_conv(a3b)))
                if obj3b < best_score:
                    best_score = obj3b
                    best_seq = a3b.copy()
                    print(f"  BH+CD: {best_score:.8f} **")

    # === Phase 4: LP Polish ===
    print(f"\n=== Phase 4: LP Polish ===")
    remaining = deadline - time.monotonic()
    p4 = min(remaining * 0.15, 150.0)
    if p4 > 15:
        a4 = lp_polish(best_seq, budget_s=p4, verbose=True, seed=seed + 400)
        obj4 = 2.0 * a4.size * float(np.max(compute_conv(a4)))
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  LP Polish: {best_score:.8f} **")
        else:
            print(f"  LP Polish: {obj4:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_base = len(best_seq)

    n_candidates = sorted(set([
        max(200, n_base - 100), n_base + 50, n_base + 100,
        n_base + 200, n_base + 300, n_base + 500,
        400, 600, 1000, 1200, 1500
    ]))
    n_filtered = []
    for nc in n_candidates:
        if 300 <= nc <= 2500:
            if all(abs(nc - nf) > 80 for nf in n_filtered):
                n_filtered.append(nc)
    n_filtered = n_filtered[:5]
    print(f"  n candidates: {n_filtered}")

    for n_try in n_filtered:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick pairwise CD
        alloc = min(remaining * 0.12, 50.0)
        if alloc > 15:
            a_try = cd_pairwise(a_try, budget_s=alloc, seed=n_try, n_pairs_per_sweep=150)

        obj_try = 2.0 * a_try.size * float(np.max(compute_conv(a_try)))
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_base = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur2 = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = 1 + r
        a_r = diverse_init(n_cur2, rng, mode=mode)
        blend = 0.4 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = normalize(a_r)

        alloc = min(remaining * 0.2, 60.0)
        if alloc > 20:
            a_r = cd_pairwise(a_r, budget_s=alloc, seed=5000 + r * 100, n_pairs_per_sweep=200)
            obj_r = 2.0 * a_r.size * float(np.max(compute_conv(a_r)))
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): {best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode {mode}): {obj_r:.8f}")

    # === Phase 7: Final Polish (alternating CD + LP) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            # Pairwise CD
            cd_alloc = min(remaining * 0.4, 80.0)
            if cd_alloc > 10:
                a_f = cd_pairwise(best_seq, budget_s=cd_alloc, seed=8000 + rnd,
                                  n_pairs_per_sweep=250)
                obj_f = 2.0 * a_f.size * float(np.max(compute_conv(a_f)))
                if obj_f < best_score:
                    best_score = obj_f
                    best_seq = a_f.copy()
                    print(f"  Final PairCD {rnd}: {best_score:.8f} **")

            # Single CD
            remaining = deadline - time.monotonic()
            if remaining > 10:
                scd_alloc = min(remaining * 0.25, 40.0)
                if scd_alloc > 8:
                    a_f2 = cd_single(best_seq, budget_s=scd_alloc, seed=8500 + rnd)
                    obj_f2 = 2.0 * a_f2.size * float(np.max(compute_conv(a_f2)))
                    if obj_f2 < best_score:
                        best_score = obj_f2
                        best_seq = a_f2.copy()
                        print(f"  Final SingleCD {rnd}: {best_score:.8f} **")

            # LP
            remaining = deadline - time.monotonic()
            if remaining > 15:
                lp_alloc = min(remaining * 0.3, 40.0)
                if lp_alloc > 8:
                    a_f3 = lp_polish(best_seq, budget_s=lp_alloc, seed=9000 + rnd)
                    obj_f3 = 2.0 * a_f3.size * float(np.max(compute_conv(a_f3)))
                    if obj_f3 < best_score:
                        best_score = obj_f3
                        best_seq = a_f3.copy()
                        print(f"  Final LP {rnd}: {best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = 2.0

    best_seq = np.maximum(best_seq, 0.0)
    result = [float(min(1000.0, x)) for x in best_seq]

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
