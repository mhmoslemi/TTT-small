                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta
            any_improved = True

            S_new = float(np.sum(a))
            if S_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        # Stall detection
        if sweep % 10 == 0:
            if best_obj < prev_best - 1e-12:
                prev_best = best_obj
                stall = 0
            else:
                stall += 1
            if verbose and sweep % 40 == 0:
                elapsed = time.monotonic() - start
                print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s, stall={stall}")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:
                    g_val = g_val / g_sum
                    return g_val, None
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget=60.0, verbose=False, seed=42):
    """LP-based polishing: solve LP, blend, repeat with perturbations."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, _ = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Try multiple blend factors
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 3:
                # Perturb and retry
                pert = rng.normal(0, 0.03, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0

        if verbose:
            elapsed = time.monotonic() - start
            print(f"    LP: obj={best_obj:.8f}, t={elapsed:.1f}s, stall={stall}")

    return best_a


def mirror_descent(a, time_budget=100.0, verbose=False, seed=42):
    """
    Multiplicative weight mirror descent on the simplex.
    Uses softmax-smoothed gradient of max(conv) for stable updates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    tau = 20.0  # smoothing parameter (increases over time)
    eta = 2.0 / np.sqrt(n)
    it = 0
    patience = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = 20.0 + 480.0 * progress
        eta = max(0.5 / np.sqrt(n), 2.0 / np.sqrt(n) * (1.0 - progress))

        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Softmax weights over conv values
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] (where 0<=k-i<n)
        # Use FFT for efficiency
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_fft = fft(a, N)
        g_full = ifft(P * np.conj(A_fft)).real
        grad = 2.0 * g_full[:n]

        # Center the gradient (remove mean to preserve sum constraint)
        grad_centered = grad - float(np.mean(grad))

        # Multiplicative update
        log_mult = -eta * grad_centered
        # Clip to prevent overflow
        log_mult = np.clip(log_mult, -5.0, 5.0)
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1
            if patience > 30:
                # Reduce step, add noise
                eta *= 0.7
                if eta < 0.1 / np.sqrt(n):
                    pert = rng.normal(0, 0.02, n)
                    a = a * np.exp(pert)
                    a = np.maximum(a, 1e-15)
                    a = a / a.sum()
                    eta = 1.0 / np.sqrt(n)
                    patience = 0

        it += 1
        if verbose and it % 500 == 0:
            print(f"    Mirror it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.1f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-0.999, 0.999, n)
        a = 1.0 / np.sqrt(1 - x * x)
        a = a / a.sum()
    elif mode == 4:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 5:
        # Cosine bump
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 6:
        # Random frequency chirp (positive part)
        t = np.linspace(0, 1, n)
        f0 = rng.uniform(2, 8)
        f1 = rng.uniform(8, 20)
        a = np.cos(2 * np.pi * (f0 * t + (f1 - f0) * t ** 2 / 2))
        a = np.maximum(a, 0) ** 2
        a += 0.1
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def block_coordinate_descent(a, block_size=8, time_budget=30.0, seed=42):
    """
    Block coordinate descent: optimize groups of block_size coordinates
    simultaneously using a local CD approach within each block.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        # Random block partition
        order = rng.permutation(n)
        for block_start in range(0, n, block_size):
            if time.monotonic() - start >= time_budget:
                break
            block_end = min(block_start + block_size, n)
            block = order[block_start:block_end]

            # For each coordinate in block, do one CD update
            for i in block:
                v = a[i]
                S_other = float(np.sum(a)) - v
                if S_other < 1e-12:
                    continue

                k_min = max(0, i)
                k_max = min(2 * n - 2, i + n - 1)
                ks = np.arange(k_min, k_max + 1)
                a_ki = a[ks - i]
                mask = (ks != 2 * i)
                ks_lin = ks[mask]
                a_ki_lin = a_ki[mask]
                c_k_lin = conv[ks_lin].copy()
                k_quad = 2 * i
                c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

                scale = max(v, 1.0 / n, 1e-6)
                candidates = [0.0, v * 0.5, v, v * 1.5, 1.0 / n]
                best_x = v
                best_f = f_1d_cd(v, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
                for x in candidates:
                    x = max(0.0, float(x))
                    fx = f_1d_cd(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
                    if fx < best_f:
                        best_f = fx
                        best_x = x

                x_opt = max(0.0, best_x)
                if abs(x_opt - v) < 1e-14:
                    continue

                delta = x_opt - v
                a[i] = x_opt
                if 0 <= k_quad <= 2 * n - 2:
                    conv[k_quad] += x_opt * x_opt - v * v
                if len(ks_lin) > 0:
                    conv[ks_lin] += 2.0 * delta * a_ki_lin

        # Periodic recompute
        conv = compute_conv(a)
        s_now = float(np.sum(a))
        if s_now > 1e-10:
            obj = 2.0 * n * float(np.max(conv)) / (s_now * s_now)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    return best_a
