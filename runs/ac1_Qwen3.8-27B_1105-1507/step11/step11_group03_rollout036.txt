Wait, there's an issue. Computing `compute_conv(a)` inside the inner loop is O(n log n) each time, and we do it n times per sweep, so O(n^2 log n) per sweep. For n=800, that's 800 * 800 * 10 = 6.4 million operations per sweep just for the convolution. That's fine.

But actually, we don't need to recompute the full convolution after each coordinate update. We can update it incrementally:
- When we change `a_i` by `delta`, the convolution changes by `delta * (d_k shifted)` where `d_k = a_{k-i} + a_{i-k}`.
- So `c_new = c_old + delta * d` (element-wise).

This makes it O(n) per coordinate update instead of O(n log n). Total per sweep: O(n^2).

where a_{i+k} is the value at index i+k (0 if out of range) and a_{i-k} is the value at index i-k (0 if out of range).

c_k = sum_j a_j * a_{j+k}

The terms involving a_i are:
- When j = i: a_i * a_{i+k} (this term exists iff 0 <= i+k < n)
- When j+k = i, i.e., j = i-k: a_{i-k} * a_i (this term exists iff 0 <= i-k < n)

So if we change a_i → a_i + delta:
- The j=i term becomes (a_i + delta) * a_{i+k}, changing by delta * a_{i+k}
- The j=i-k term becomes a_{i-k} * (a_i + delta), changing by delta * a_{i-k}
- If both conditions hold (i+k in range AND i-k in range), both changes contribute.

Note: both conditions can hold simultaneously. For example, if i=1, k=0: i+k=1 (in range) and i-k=1 (in range). In this case, c_0 contains the term a_1 * a_1 = a_1^2. Changing a_1 by delta changes this to (a_1+delta)^2, a change of 2*a_1*delta + delta^2. The linear part is 2*a_1*delta, which matches delta * a_1 + delta * a_1 = 2*a_1*delta.

So the linear change in c_k is:
delta_k_effect = delta * (a_{i+k} * I(0<=i+k<n) + a_{i-k} * I(0<=i-k<n))

where I(condition) is 1 if condition holds, 0 otherwise.

- a_{i+k} is nonzero when 0 <= i+k <= n-1, i.e., -i <= k <= n-1-i. Since k >= 0: 0 <= k <= n-1-i.

  Since i < n: i-n+1 <= 0, so max(0, i-n+1) = 0. And min(i, 2n-2) = i (since i < n <= 2n-2 for n >= 2).
  So: 0 <= k <= i.

Therefore:
d[k] = a_{i+k} * I(0 <= k <= n-1-i) + a_{i-k} * I(0 <= k <= i)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses exact coordinate descent as the main workhorse, combined with LP fixed-point
iteration, smoothed gradient descent, and multi-n exploration.
Key innovation: exact 1D coordinate optimization using the structure of the
convolution's dependence on individual elements.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def cordescent_sweep(a, c, S, i, compute_d=False):
    """
    Perform one coordinate descent step for coordinate i.
    Returns (new_a_i, new_c, improved).
    """
    n = len(a)
    d = np.zeros(2 * n - 1)
    # d[0:n-i] += a[i:n]
    d[:n - i] += a[i:n]
    # d[0:i+1] += a[i::-1]
    d[:i + 1] += a[i::-1]

    # Find candidate optima: for each k with d_k > 0, delta_k = S - 2*c_k/d_k
    mask = d > 1e-18
    if np.any(mask):
        delta_k = S - 2.0 * c[mask] / d[mask]
        # Filter: delta >= -a[i] (non-negativity), and reasonable upper bound
        delta_k = delta_k[delta_k >= -a[i] - 1e-12]
        delta_k = np.clip(delta_k, -a[i], S * 5)
        delta_candidates = np.concatenate([delta_k, [-a[i], 0.0, -0.5 * a[i]]])
    else:
        delta_candidates = np.array([-a[i], 0.0])

    delta_candidates = np.unique(np.round(delta_candidates, 10))
    delta_candidates = delta_candidates[delta_candidates >= -a[i] - 1e-12]
    delta_candidates = np.clip(delta_candidates, -a[i], S * 5)

    if len(delta_candidates) == 0:
        return a[i], c, False

    # Evaluate F(delta) = max(c + delta*d) / (S + delta)^2
    # Vectorized
    conv_at_delta = c + np.outer(delta_candidates, d)
    max_at_delta = np.max(conv_at_delta, axis=1)
    denom = (S + delta_candidates) ** 2
    denom = np.maximum(denom, 1e-30)
    F_vals = max_at_delta / denom

    best_idx = np.argmin(F_vals)
    best_delta = delta_candidates[best_idx]
    new_val = a[i] + best_delta

    if new_val < 0:
        new_val = 0.0
    if new_val > 1000:
        new_val = 1000.0

    if abs(new_val - a[i]) > 1e-14:
        # Update c incrementally (exact for linear change)
        c = c + best_delta * d
        return new_val, c, True
    return a[i], c, False


def coordinate_descent(a, max_sweeps=500, time_budget=600, verbose=False, seed=42):
    """
    Exact coordinate descent for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0

    c = compute_conv(a)
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(c))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget - 1:
            break

        # Choose sweep direction (alternate for better convergence)
        if sweep % 2 == 0:
            indices = range(n)
        else:
            indices = range(n - 1, -1, -1)

        sweep_improved = 0
        for i in indices:
            new_val, c, improved = cordescent_sweep(a, c, S, i)
            if improved:
                a[i] = new_val
                S += new_val - (new_val - (new_val - new_val))  # track S
                S = float(np.sum(a))  # Just recompute (cheap for small n)
                sweep_improved += 1

        # Recompute S and c periodically for numerical stability
        S = float(np.sum(a))
        if sweep % 20 == 19:
            a = a / S
            S = 1.0
            c = compute_conv(a)

        obj = 2.0 * n * float(np.max(c)) / (S * S)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if verbose and sweep % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    CDS sweep {sweep}: obj={best_obj:.10f}, improved={sweep_improved}, time={elapsed:.1f}s")

        if stall > 100:
            # Perturb to escape local minimum
            a = best_a.copy()
            strength = min(0.05, 0.01 * (stall // 20))
            a = a + strength * rng.standard_normal(n)
            a = np.maximum(a, 0)
            S = float(np.sum(a))
            if S > 1e-10:
                a = a / S
                S = 1.0
            c = compute_conv(a)
            stall = 0

    return best_a, best_obj


def smoothed_grad(a, tau):
    """Compute gradient of smoothed max of conv(a,a) using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    shifted = tau * (conv - exact_max)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def nesterov_gd(a, time_budget, tau_levels=None, verbose=False, beta=0.9, lr_scale=3.0):
    """Nesterov accelerated gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [3, 8, 20, 50, 100, 200, 500, 1000, 2000]

    for tau in tau_levels:
        lr = lr_scale / np.sqrt(n)
        v = np.zeros(n)
        iters = 0
        max_iters = 5000

        while time.monotonic() - start < time_budget:
            if iters >= max_iters:
                break

            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new
            iters += 1
            lr = max(1e-8, lr * 0.99999)

        if verbose:
            print(f"    GD tau={tau:5d}: obj={best_obj:.10f}, iters={iters}")

    return best_a, best_obj


def lp_solve(a):
    """Solve min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1."""
    n = len(a)
    # Build constraint matrix A where (A @ g)_k = (a*g)_k
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(lo + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None


def lp_fixed_point(a, time_budget, verbose=False, seed=42, max_iters=300):
    """LP fixed-point iteration."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    t_current = 0.3

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, M_lp = lp_solve(a)
        if g_lp is None:
            if stall > 8:
                break
            stall += 1
            continue

        stall = 0
        t_candidates = [0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0,
                        max(0.02, 0.5 * t_current), min(1.0, 1.5 * t_current)]

        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t_current

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t

        if best_local_obj < best_obj - 1e-13:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a
            t_current = max(0.05, min(1.0, best_local_t))
        else:
            stall += 1
            # Perturb
            if stall % 5 == 0:
                a = best_a + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
            if stall > 50:
                break

        if verbose and it % 10 == 0:
            print(f"    LP it{it:3d}: obj={best_obj:.10f}, M={M_lp:.6f}, t={best_local_t:.2f}")

    return best_a, best_obj


def active_set_refine(a, time_budget, verbose=False, seed=42, max_iters=200):
    """Targeted active-set refinement to reduce peak lags."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        threshold = max_val * 0.9995
        max_lags = np.where(conv >= threshold)[0]

        # Subgradient: average over active lags
        subgrad = np.zeros(n)
        for lag in max_lags:
            for i in range(n):
                j = lag - i
                if 0 <= j < n:
                    subgrad[i] += 2.0 * a[j]
        subgrad /= len(max_lags)

        gn = np.linalg.norm(subgrad)
        if gn < 1e-15:
            break
        subgrad = subgrad / gn

        improved = False
        for step in [0.0002, 0.0005, 0.001, 0.002, 0.005]:
            a_try = a - step * subgrad
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            a = a + 0.005 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_obj


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

        alloc = min(remaining * 0.30, 80.0)
        a8, obj8 = lp_fixed_point(best_seq, time_budget=alloc, verbose=True,
                                   seed=4242, max_iters=100)
        if obj8 < best_score:
            best_score = obj8
            best_seq = a8.copy()
            print(f"  Phase8 LP: obj={best_score:.10f} **")

    # === Phase 9: Final CDS ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.40, 60.0)
        a9, obj9 = coordinate_descent(best_seq, max_sweeps=10000,
                                       time_budget=alloc, verbose=False, seed=7777)
        if obj9 < best_score:
            best_score = obj9
            best_seq = a9.copy()
            print(f"  Phase9 CDS: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    return result
