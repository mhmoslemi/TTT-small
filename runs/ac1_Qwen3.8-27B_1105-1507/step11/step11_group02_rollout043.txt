"""
Optimizes step function coefficients using smoothed natural gradient descent
with FFT-accelerated gradient computation, combined with coordinate descent
polishing and multi-length exploration.

Key innovations over previous approach:
- O(n log n) per-iteration cost via FFT-based gradient
- Hundreds of thousands of iterations possible in 1500s budget
- Adaptive tau annealing for the log-sum-exp smoothing
- Systematic multi-n exploration with resampling
- Final CD polish for precise convergence
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective: 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    return compute_obj_normalized(a_norm)


def compute_smoothed_gradient(a, tau):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*conv(a,a))) w.r.t. a.
    Returns (gradient, conv, probabilities).
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    conv = compute_conv(a)
    
    tau_conv = tau * conv
    tau_conv_max = float(np.max(tau_conv))
    p = np.exp(tau_conv - tau_conv_max)
    p_sum = float(np.sum(p))
    p = p / p_sum
    
    # Gradient: grad_a[i] = 2 * sum_j p[i+j] * a[j]
    # = 2 * (rev_p * a)[2n-2-i] where rev_p = p[::-1]
    rev_p = p[::-1]
    
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    
    R = rfft(rev_p, size)
    A = rfft(a, size)
    conv_grad = irfft(R * A, size)
    
    # indices: 2n-2, 2n-3, ..., n-2
    indices = 2 * n - 2 - np.arange(n)
    grad_a = 2.0 * conv_grad[indices]
    
    return grad_a, conv, p


def natural_gradient_descent(a, time_budget=100.0, tau_init=50.0, tau_max=50000.0,
                             lr_base=0.5, verbose=False, seed=42):
    """
    Natural gradient (multiplicative weight) descent on the simplex.
    Uses log-sum-exp smoothing of the max of convolution.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64).copy()
    a = a / float(np.sum(a))
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    tau = tau_init
    lr = lr_base
    it = 0
    stall = 0
    patience = 5000
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        
        # Adaptive tau: increase over time
        tau = min(tau_max, tau_init * (1 + progress * 10))
        
        # Adaptive learning rate: decrease over time
        lr = lr_base * (1.0 - 0.5 * progress)
        
        try:
            grad_a, conv, p = compute_smoothed_gradient(a, tau)
        except Exception:
            break
        
        # Compute the smoothed objective for tracking
        # (approximate: use the current max as proxy)
        cur_obj = 2.0 * n * float(np.max(conv))
        
        on_semi_grad = 2.0 * n * grad_a  # Scale gradient by 2n to match objective scale
        
        # Multiplicative update: a[i] *= exp(-lr * grad_i / scale)
        # Normalize the gradient so max update is meaningful
        grad_max = float(np.max(np.abs(on_semi_grad)))
        if grad_max < 1e-15:
            break
        
        # Use a normalized learning rate
        effective_lr = lr / (grad_max * 0.5)
        effective_lr = min(effective_lr, 1.0)
        
        # Center the gradient (remove mean to avoid uniform scaling)
        mean_g = float(np.dot(a, on_semi_grad))
        centered = on_semi_grad - mean_g
        
        update = np.exp(-effective_lr * centered)
        update = np.clip(update, 1e-10, 1e10)
        
        a_new = a * update
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(np.sum(a_new))
        
        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        it += 1
        
        # Occasionally re-normalize tau
        if it % 10000 == 0 and stall > patience:
            # Increase tau to sharpen the gradient
            tau = min(tau_max, tau * 2)
            stall = 0
        
        if verbose and it % 10000 == 0:
            print(f"      NGD it={it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.1f}s")
    
    return best_a


def coordinate_descent_fast(a, time_budget=100.0, verbose=False, seed=42):
    """
    Fast coordinate descent with exact 1D solve.
    For each coordinate, finds the exact minimizer of the 1D convex function
    using binary search on the subgradient.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    
    conv = compute_conv(a)
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Active set: lags within 5% of max
    max_c = float(np.max(conv))
    
    sweep = 0
    while time.monotonic() - start < time_budget:
        sweep += 1
        
        if sweep % 10 == 1:
            # Recompute conv and update active set
            conv = compute_conv(a)
            max_c = float(np.max(conv))
            obj = 2.0 * n * max_c
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            
            v = a[i]
            if v < 1e-15:
                continue
            
            # Lags affected by coordinate i: k in [i, i+n-1]
            k_min = i
            k_max = min(2 * n - 2, i + n - 1)
            
            # Get the relevant convolution values and a[k-i] values
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]  # a[k-i] for each k
            c_k = conv[ks].copy()
            
            # Find the 1D optimal: minimize max_k (c_k + 2*d*a_ki + d^2*(k==2i))
            # Binary search on subgradient
            d_lo = -v  # minimum d (makes a[i] = 0)
            d_hi = 2.0 / n  # reasonable upper bound
            
            def f_1d(d):
                """Evaluate g(d) = max_k (c_k + 2*d*a_ki + d^2*(k==2i))"""
                vals = c_k + 2.0 * d * a_ki
                mask_2i = (ks == 2 * i)
                if np.any(mask_2i):
                    vals[mask_2i] += d * d
                return float(np.max(vals))
            
            def grad_1d(d):
                """Subgradient of g at d."""
                vals = c_k + 2.0 * d * a_ki
                mask_2i = (ks == 2 * i)
                if np.any(mask_2i):
                    vals[mask_2i] += d * d
                idx = int(np.argmax(vals))
                if mask_2i[idx]:
                    return 2.0 * a_ki[idx] + 2.0 * d
                return 2.0 * a_ki[idx]
            
            # Binary search for where subgradient crosses zero
            g_lo = grad_1d(d_lo)
            if g_lo >= 0:
                d_opt = d_lo
            else:
                g_hi = grad_1d(d_hi)
                if g_hi <= 0:
                    d_opt = d_hi
                else:
                    for _ in range(40):
                        d_mid = (d_lo + d_hi) / 2.0
                        g_mid = grad_1d(d_mid)
                        if g_mid > 0:
                            d_hi = d_mid
                        else:
                            d_lo = d_mid
                    d_opt = (d_lo + d_hi) / 2.0
            
            x_opt = max(0.0, v + d_opt)
            if abs(x_opt - v) < 1e-15:
                continue
            
            # Update a and conv
            d = x_opt - v
            a[i] = x_opt
            S += d
            
            # Update conv in-place
            for idx, k in enumerate(ks):
                delta = 2.0 * d * a_ki[idx]
                if k == 2 * i:
                    delta += d * d
                conv[k] += delta
            
            # Quick objective check (every few coordinates)
            if sweep % 50 == 0 and i % 100 == 0:
                max_c = float(np.max(conv))
                obj = 2.0 * n * max_c
                if S > 1e-10:
                    obj = obj / (S * S)  # if not normalized
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()
        
    return best_a

"""
Optimizes step function coefficients using FFT-accelerated natural gradient
descent with adaptive log-sum-exp smoothing, combined with fast coordinate
descent polishing and systematic multi-length exploration.

Key innovations:
- O(n log n) gradient computation via FFT cross-correlation
- Natural gradient (multiplicative weight) updates on the simplex
- Adaptive tau annealing for log-sum-exp smoothing
- Exact 1D subgradient binary search in CD
- Systematic multi-n exploration with resampling and diverse restarts
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a))) / (s * s)


def compute_smoothed_gradient(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*c[k])) w.r.t. a using FFT.
    Returns (grad_a, conv, p) where p is the softmax weight over lags.
    """
    n = len(a)
    conv = compute_conv(a)

    tau_conv = tau * conv
    tau_max = float(np.max(tau_conv))
    p = np.exp(tau_conv - tau_max)
    p_sum = float(np.sum(p))
    p = p / p_sum

    # grad_a[i] = 2 * sum_j p[i+j] * a[j] = 2 * (rev_p * a)[2n-2-i]
    rev_p = p[::-1]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    R = rfft(rev_p, size)
    A = rfft(a, size)
    conv_grad = irfft(R * A, size)

    indices = 2 * n - 2 - np.arange(n)
    grad_a = 2.0 * conv_grad[indices]

    return grad_a, conv, p


def natural_gradient_descent(a, time_budget=100.0, tau_init=30.0, tau_end=30000.0,
                             lr_scale=0.8, momentum=0.9, verbose=False, seed=42):
    """
    Natural gradient (multiplicative weight) descent on the simplex with
    log-sum-exp smoothing and momentum.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64).copy()
    a = a / float(np.sum(a))

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Momentum buffer (in log-space)
    v = np.zeros(n)
    tau = tau_init
    it = 0
    stall = 0
    best_stall_obj = best_obj

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Adaptive tau: ramp up
        tau = tau_init + (tau_end - tau_init) * (progress ** 1.5)
        tau = max(tau, tau_init)

        try:
            grad_a, conv, p = compute_smoothed_gradient(a, tau)
        except Exception:
            break

        # Objective tracking
        cur_obj = 2.0 * n * float(np.max(conv))
        obj_scaled = 2.0 * n * grad_a

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
            best_stall_obj = best_obj
        else:
            stall += 1

        # Adaptive learning rate
        grad_inf = float(np.max(np.abs(obj_scaled)))
        if grad_inf < 1e-15:
            break
        lr = lr_scale / (grad_inf + 1e-15)
        lr *= (1.0 - 0.3 * progress)

        # Momentum update in log-space
        centered = obj_scaled - float(np.dot(a, obj_scaled))

        v = np.clip(v, -3.0, 3.0)

        a_new = a * np.exp(v)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(np.sum(a_new))

        a = a_new
        it += 1

        # Restart if stalled for too long

            tau = min(30000.0, tau * 3.0)
            lr_scale = max(0.1, lr_scale * 0.7)
            stall = 0

            a = np.maximum(a, 1e-15)
            a = a / float(np.sum(a))

        if verbose and it % 50000 == 0:
            print(f"        NGD it={it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.4f}, t={elapsed:.1f}s")

    return best_a


def cd_fast(a, time_budget=100.0, verbose=False, seed=42):
    """
    Fast coordinate descent using exact 1D subgradient binary search.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1

        if sweep % 5 == 1:
            conv = np.maximum(conv, 0)
            max_c = float(np.max(conv))
            obj = 2.0 * n * max_c
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            if v < 1e-15:
                # Try setting to a small positive value
                x_try = 1.0 / n
                if x_try > v:
                    d = x_try - v
                    k_min = i
                    k_max = min(2 * n - 2, i + n - 1)
                    for k in range(k_min, k_max + 1):
                        conv[k] += 2.0 * d * a[k - i]
                        if k == 2 * i:
                            conv[k] += d * d
                    a[i] = x_try
                    S += d
                continue

            k_min = i
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks]

            # Binary search on subgradient to find optimal d
            d_lo = -v + 1e-15
            d_hi = 3.0 / n

            # Check gradient at boundaries
            # g'(d) = 2*a[k*(d)-i] or 2*a[i]+2d if k*=2i
            def subgrad(d):
                vals = c_k + 2.0 * d * a_ki
                mask_2i = (ks == 2 * i)
                if np.any(mask_2i):
                    vals = vals.copy()
                    vals[mask_2i] += d * d
                else:
                    vals = vals
                idx = int(np.argmax(vals))
                if mask_2i[idx]:
                    return 2.0 * a_ki[idx] + 2.0 * d
                return 2.0 * a_ki[idx]

            g_lo = subgrad(d_lo)
            if g_lo >= 0:
                d_opt = d_lo
            else:
                g_hi = subgrad(d_hi)
                if g_hi <= 0:
                    d_opt = d_hi
                else:
                    for _ in range(35):
                        d_mid = 0.5 * (d_lo + d_hi)
                        g_mid = subgrad(d_mid)
                        if g_mid > 0:
                            d_hi = d_mid
                        else:
                            d_lo = d_mid
                    d_opt = 0.5 * (d_lo + d_hi)

            x_opt = v + d_opt
            if abs(x_opt - v) < 1e-14:
                continue

            d = x_opt - v
            a[i] = x_opt
            S += d

            # Update conv in-place
            delta_vals = 2.0 * d * a_ki
            mask_2i = (ks == 2 * i)
            if np.any(mask_2i):
                delta_vals = delta_vals.copy()
                delta_vals[mask_2i] += d * d
            conv[ks] += delta_vals

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations of length n."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:

    elif mode == 4:

        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 5:
        # Two-peak
        c1, c2 = n // 4, 3 * n // 4
        w = n // 8
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 6:
        # Random smooth
        freqs = rng.integers(1, 6, 3)
        phases = rng.uniform(0, 2 * np.pi, 3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01) / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)
