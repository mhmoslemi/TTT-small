def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_soft_gradient(a, p):
    """
    Compute gradient of soft-max objective: grad_i = 2 * sum_j a_j * p_{i+j}
    Uses FFT cross-correlation.
    """
    n = len(a)
    p_len = len(p)
    a_rev = a[::-1]
    size = 1
    while size < n + p_len - 1:
        size <<= 1
    A = rfft(a_rev, size)
    P = rfft(p, size)
    full = irfft(A * P, size)
    q = full[n - 1:n - 1 + n]
    return 2.0 * q

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, warm_start=None):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def line_search_blend(a, g_dir, max_iters=25):
    """Golden section search for optimal blend factor a_new = (1-t)*a + t*g_dir."""
    lo, hi = 0.0, 2.0
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    
    def obj_at(t):
        a_try = (1 - t) * a + t * g_dir
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)
    
    # First expand range if needed
    if obj_at(1.0) < obj_at(0.5):
        hi = 3.0
    
    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)
    
    for _ in range(max_iters):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)
    
    t_best = c if fc < fd else d
    obj_best = min(fc, fd)
    
    # Check some standard points
    for t_try in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        obj_try = obj_at(t_try)
        if obj_try < obj_best:
            obj_best = obj_try
            t_best = t_try
    
    return t_best, obj_best

def coordinate_descent(a, time_budget=60.0, verbose=False):
    """Direct coordinate descent: try to optimize each coordinate pair."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    
    # Precompute which coordinates interact
    # For coordinate i, changing a[i] affects conv[k] for k in range involving i
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Random subset of coordinates to update
        subset_size = min(50, n)
        idx = rng.choice(n, size=subset_size, replace=False)
        
        for i in idx:
            # Try small perturbations of a[i]
            current = a[i]
            best_delta = 0.0
            best_local = best_obj
            
            for delta_frac in [0.1, -0.1, 0.05, -0.05, 0.2, -0.2, 0.02, -0.02]:
                new_val = max(0, current * (1 + delta_frac))
                old_val = a[i]
                a[i] = new_val
                # Renormalize
                s = float(np.sum(a))
                if s < 0.01:
                    a[i] = old_val
                    continue
                obj_try = compute_obj(a / s)
                if obj_try < best_local:
                    best_local = obj_try
                    best_delta = new_val - old_val
                a[i] = old_val
            
            if best_delta != 0:
                a[i] += best_delta
                a = project_simplex(a)
                obj_now = compute_obj(a)
                if obj_now < best_obj:
                    best_obj = obj_now
                    best_a = a.copy()
        
        it += 1
        if verbose and it % 10 == 0:
            print(f"      CD round {it}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
    
    return best_a, best_obj

        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / np.cosh(4.0 * (2 * x - 1))
    elif mode == 6:
        # Two sharp peaks
        a = np.zeros(n)
        a[0] = 1.0
        a[-1] = 1.0
        a[1:min(3, n)] = 0.5
        a[max(0, n-4):n-1] = 0.5
        if n > 6:
            a[1:-1] += 0.03
    elif mode == 7:
        # Geometric decay from both ends
        r = 0.98
        a = np.maximum(r ** np.arange(n), r ** (n - 1 - np.arange(n)))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return project_simplex(np.maximum(a_new, 0))


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search with enhanced LP polishing, peak-targeted SGD,
    coordinate descent, and adaptive length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = u_shaped_init(n_current, sharpness=3.0)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")
    
    # Try several initializations at the parent length
    for init_mode in ['arcsine', 'u_sharp5', 'u_sharp8']:
        if init_mode == 'arcsine':
            a_try = arcsine_init(n_current)
        elif init_mode == 'u_sharp5':
            a_try = u_shaped_init(n_current, sharpness=5.0)
        elif init_mode == 'u_sharp8':
            a_try = u_shaped_init(n_current, sharpness=8.0)
        else:
            continue
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  Init {init_mode}: obj={best_score:.8f} **")
    
    print(f"Starting best: n={len(best_seq)}, score={best_score:.8f}")
    
    # === Phase 1: Peak-targeted SGD (quick refinement) ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 1: Peak-Targeted SGD, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.15, 120.0)
        a1, obj1 = peak_targeted_sgd(best_seq, time_budget=alloc, seed=seed, verbose=True)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1: obj={best_score:.8f} **")

            print(f"  Phase1: obj={obj1:.8f} (no improvement)")

    remaining = deadline - time.monotonic()

        print(f"\n=== Phase 2: Enhanced LP Polish, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.30, 250.0)
        a2, obj2 = lp_polish_enhanced(best_seq, time_budget=alloc, max_iters=100, verbose=True)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2: obj={best_score:.8f} **")
        else:
            print(f"  Phase2: obj={obj2:.8f} (no improvement)")
    
    # === Phase 3: Soft-max GD for smoothing ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 3: SoftMax GD, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.15, 100.0)
        a3, obj3 = soft_max_gd(best_seq, time_budget=alloc,
                               beta_start=20.0, beta_end=80.0,
                               lr_start=0.005, momentum=0.92,
                               seed=seed+10, verbose=True)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj3:.8f} (no improvement)")

        n_candidates = [700, 750, 850, 900, 950, 1000, 1100, 1200]

        alloc = min(remaining * 0.15, 60.0)
        a6, obj6 = coordinate_descent(best_seq, time_budget=alloc, verbose=True)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Phase6: obj={best_score:.8f} **")
        else:
            print(f"  Phase6: obj={obj6:.8f} (no improvement)")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 7: Final LP Polish, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.4, 80.0)
        a7, obj7 = lp_polish_enhanced(best_seq, time_budget=alloc, max_iters=40,
                                      verbose=True, seed=seed+500)
        if obj7 < best_score:

            best_seq = a7.copy()
            print(f"  Phase7: obj={best_score:.8f} **")
        else:
            print(f"  Phase7: obj={obj7:.8f} (no improvement)")
    
    # === Phase 8: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 8: Diverse Restarts, time={remaining:.0f}s ===")
        n_cur = len(best_seq)
        
        for r in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            
            mode = r % 8
            a_r = diverse_init(n_cur, rng, mode=mode)
            blend = 0.5 + 0.1 * r
            a_r = project_simplex((1 - blend) * a_r + blend * best_seq)
            
            alloc_sgd = min(remaining * 0.2, 40.0)
            if alloc_sgd > 10:
                a_r, obj_r = peak_targeted_sgd(a_r, time_budget=alloc_sgd, seed=seed+600+r)
            
            remaining = deadline - time.monotonic()
            if remaining > 20:
                alloc_lp = min(remaining * 0.3, 40.0)
                a_r, obj_r = lp_polish_enhanced(a_r, time_budget=alloc_lp,
                                                max_iters=20, verbose=False, seed=seed+700+r)
            
            if obj_r < best_score:

                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")
    
    # === Phase 9: Ultimate final polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 9: Ultimate Polish, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.6, 60.0)
        a9, obj9 = lp_polish_enhanced(best_seq, time_budget=alloc, max_iters=50,
                                      verbose=True, seed=seed+800)

            best_seq = a9.copy()
            print(f"  Phase9: obj={best_score:.8f} **")

    if best_seq is None:

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
