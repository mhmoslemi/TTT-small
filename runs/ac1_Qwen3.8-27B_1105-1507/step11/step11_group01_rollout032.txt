Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2
using LP-based fixed-point iteration, coordinate descent, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute the objective 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, top_k=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Optionally only use top_k constraints."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None
    a_norm = a / s
    A = build_conv_matrix(a_norm)
    conv_k = A @ a_norm  # = conv(a_norm, a_norm)
    
    if top_k is not None and top_k < 2 * n - 1:
        # Select top_k constraints
        idx = np.argsort(conv_k)[::-1][:top_k]
        idx = np.sort(idx)
        A = A[idx, :]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:
                    g_val = g_val / g_sum
                    return g_val
        except Exception:
            continue
    return None

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, top_k=None):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None
    a_norm = a / s
    A = build_conv_matrix(a_norm)
    conv_k = A @ a_norm

    if top_k is not None and top_k < 2 * n - 1 and top_k > 10:
        idx = np.argsort(conv_k)[::-1][:top_k]
        idx = np.sort(idx)
        A = A[idx, :]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:
                    return g_val / g_sum
        except Exception:
            continue
    return None


def eval_1d(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i):
    """Evaluate 1D objective for coordinate descent at value x."""
    if len(a_ki_lin) > 0:
        lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
        max_lin = float(np.max(lin_vals))
    else:
        max_lin = 0.0
    quad_val = c_2i + x * x - v * v
    max_val = max(max_lin, quad_val)
    denom = (S_other + x) * (S_other + x)
    if denom < 1e-30:
        return 1e30
    return 2.0 * n * max_val / denom


def coordinate_descent(a, time_budget=100.0, seed=42):
    """Exact coordinate descent with golden section search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        if sweep % 3 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    a = a / S_check

        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask].copy()

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            scale = max(v, S / n, 1e-6)
            candidates = [0.0, v * 0.5, v * 1.5, v * 2.0, S / n, 2.0 * S / n]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = eval_1d(v, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
            for x in candidates:
                if abs(x - v) < 1e-13:
                    continue
                fx = eval_1d(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            x_lo = max(0.0, best_x - scale * 0.5)
            x_hi = best_x + scale
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = eval_1d(c_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
            fd = eval_1d(d_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)

            for _ in range(35):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = eval_1d(c_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = eval_1d(d_pt, v, S_other, n, c_ki_lin, c_2i)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            S += delta

    return best_a


def structured_init(n, mode=0, seed=42):
    """Generate structured initializations."""
    rng = np.random.default_rng(seed)
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = np.ones(n)
        a[1::2] = 0.3
        return a / a.sum()
    elif mode == 2:
        a = np.zeros(n)
        w = max(1, n // 3)
        a[:w] = 1.0
        a[-w:] = 1.0
        return a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
        return a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.05)
        return a / a.sum()
    elif mode == 5:
        a = np.ones(n)
        period = max(2, n // 20)
        a[::period] = 2.0
        return a / a.sum()
    elif mode == 6:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 7:
        a = np.zeros(n)
        w = max(1, n // 4)
        a[:w] = 1.0
        a[n // 2 - w:n // 2 + w] = 1.5
        a[-w:] = 1.0
        return a / a.sum()
    elif mode == 8:
        a = np.ones(n)
        a[0] = 2.0
        a[-1] = 2.0
        return a / a.sum()
    elif mode == 9:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.7 * np.sin(np.pi * x) + 0.3 * np.sin(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def lp_iteration(a, time_budget=200.0, seed=42, use_top_k=False, top_k_ratio=0.3):
    """LP-based fixed-point iteration with adaptive step and perturbation."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    it = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        tk = int(top_k_ratio * (2 * n - 1)) if use_top_k else None
        g = solve_lp(a, top_k=tk)
        if g is None:
            break

        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 3:
                pert = rng.normal(0, 0.05, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0

        it += 1
        if it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP iter {it}: obj={best_obj:.8f}, stall={stall}, t={elapsed:.1f}s")

    return best_a


def subgradient_descent(a, time_budget=100.0, seed=42):
    """Subgradient descent with momentum and adaptive step."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    eta = 0.3 / np.sqrt(n)
    momentum = 0.6
    v_mom = np.zeros(n)
    it = 0
    patience = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        threshold = max_conv * (1.0 - 1e-8)
        argmax_set = np.where(conv >= threshold)[0]

        v_sub = np.zeros(n)
        for k in argmax_set:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                v_sub[lo:hi + 1] += a[k - hi:k - lo + 1]
        v_sub /= max(len(argmax_set), 1)

        grad = 2.0 * v_sub - (2.0 * max_conv / n) * np.ones(n)

        v_mom = momentum * v_mom - eta * grad
        a_new = a + v_mom
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            v_mom = np.zeros(n)
            continue
        a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1
            if patience > 40:
                eta *= 0.7
                if eta < 1e-5:
                    pert = rng.normal(0, 0.04, n)
                    a_pert = a * np.exp(pert)
                    a_pert = np.maximum(a_pert, 0)
                    a_pert = a_pert / a_pert.sum()
                    obj_pert = 2.0 * n * float(np.max(compute_conv(a_pert)))
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert.copy()
                        a = a_pert
                    else:
                        a = best_a.copy()
                    eta = 0.3 / np.sqrt(n)
                    v_mom = np.zeros(n)
                    patience = 0

        it += 1

    return best_a

        best_seq = np.ones(200)
    best_seq = np.maximum(best_seq, 0.0)
    result = [float(min(1000.0, max(0.0, x))) for x in best_seq]

    # Final validation
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = min(best_score, val)
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
