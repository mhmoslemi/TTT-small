The key insight is: **we want the autocorrelation to be as flat as possible**. 

A practical approach that's known to work well for this type of problem is **iterative reweighted least squares** or **the Gerchberg-Saxton algorithm** (also known as error reduction):

1. Start with `a` (non-negative, sum=1)
2. Compute `c = conv(a, a)`
3. In the "time domain" (autocorrelation domain): replace `c` with a flatter version `c_new`. For example, `c_new = (1-beta)*c + beta*flat` where `flat = ones/(2n-1)`.
4. In the "frequency domain": take FFT of `c_new` to get the new power spectrum `|A_new|^2`. But this might have negative values!
5. Clip: `|A_new|^2 = max(|A_new|^2, 0)`
6. `|A_new| = sqrt(|A_new|^2)`
7. Keep the phase from the previous `A`: `A_new = |A_new| * exp(i*angle(A))`
8. `a_new = IFFT(A_new)[:n]`
9. Clip to non-negative: `a_new = max(a_new, 0)`
10. Normalize: `a_new = a_new / sum(a_new)`
11. Repeat

This is essentially the **error reduction / Gerchberg-Saxton** algorithm adapted for our problem. The "constraints" are:
- Time domain: `a >= 0`, `sum(a) = 1`
- Autocorrelation domain: `c ≈ flat`

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a novel Gerchberg-Saxton spectral flattening approach combined with
adaptive mirror descent, LP-based refinement, and multi-n exploration.
The GS method iteratively projects between the non-negativity/sum constraint
and the "flat autocorrelation" constraint in the spectral domain.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    N = 1
    while N < 2 * n - 1:
        N *= 2
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def gradient_from_p(a, p):
    """
    Compute 2 * (a_reversed * p)[n-1:2n-1] for gradient w.r.t. a.
    p has length 2n-1 (distribution over convolution lags).
    """
    n = len(a)
    b = a[::-1]
    N_fft = 1
    while N_fft < 3 * n - 1:
        N_fft *= 2
    c = irfft(rfft(b, N_fft) * rfft(p, N_fft), N_fft)
    return 2.0 * c[n - 1:2 * n - 1]


def smoothed_gradient(a, conv, tau):
    """Compute gradient of LSE(tau * (conv - max_conv)) w.r.t. a."""
    n = len(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    if shifted.max() > 500:
        shifted = shifted - shifted.max()
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(2 * n - 1)
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    return gradient_from_p(a, p)


def topk_subgradient(a, conv, k=3):
    """Subgradient using only the top-k peaks (uniform weights)."""
    n = len(a)
    idx = np.argsort(conv)[-k:]
    p = np.zeros(2 * n - 1)
    p[idx] = 1.0 / k
    return gradient_from_p(a, p)


def blended_gradient(a, conv, tau, k_top=3, alpha=0.3):
    """Blend smoothed gradient with top-k subgradient."""
    g_smooth = smoothed_gradient(a, conv, tau)
    g_topk = topk_subgradient(a, conv, k_top)
    return alpha * g_topk + (1.0 - alpha) * g_smooth


def mirror_descent_step(a, grad, eta):
    """Multiplicative weights update."""
    exponent = -eta * grad
    exponent = np.clip(exponent, -50, 50)
    a_new = a * np.exp(exponent)
    s = float(np.sum(a_new))
    if s < 1e-300:
        return a.copy()
    return a_new / s

    """

    N = 1

    best_obj = compute_obj(a)
    start = time.monotonic()

            s = float(np.sum(a_work))
            if s > 1e-10:

        a_work = np.maximum(a_work, 0).copy().astype(np.float64)
        s = float(np.sum(a_work))

            c_new = np.maximum(c_new, 0)

            new_power = np.maximum(np.real(C_new), 0.0)

            temp = 0.008 * (1.0 - progress)
            a_new = a_new + temp * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            a_half = mirror_descent_step(a, g, eta * 0.3)
            obj_half = compute_obj(a_half)
            if obj_half < best_obj - 1e-13:
                best_obj = obj_half
                best_a = a_half.copy()
                best_it = it
                a = a_half
                stall = 0
            elif stall >= 30:
                strength = 0.03 * (1.0 + progress)
                a_pert = a * np.exp(strength * rng.standard_normal(n))
                a_pert = np.maximum(a_pert, 0)
                s_pert = float(np.sum(a_pert))
                if s_pert > 1e-10:
                    a = a_pert / s_pert
                else:
                    a = best_a.copy()
                stall = 0

        a = a_new
        it += 1

        if verbose and (it % 200 == 0 or it == 1):
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, "
                  f"eta={eta:.4f}, stall={stall}, time={time.monotonic()-start:.0f}s")

    return best_a


def lp_direction_improved(a):
    """
    LP: min M s.t. 2*conv(a,g)_k - conv(a,a)_k <= M, sum(g)=1, g>=0.
    First-order Taylor approximation of min max conv(g,g)_k.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv_aa = compute_conv(a_norm)

    A = np.zeros((2 * n - 1, n))
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A[valid] = a_norm[diff[valid]]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [2.0 * (A @ g) - M <= conv_aa, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, g, n_evals=30):
    """Line search for best t in [0,1]."""
    n = len(a)
    ts = np.linspace(0.01, 0.99, n_evals)
    objs = np.full(n_evals, np.inf)
    for i, t in enumerate(ts):
        a_try = np.maximum((1 - t) * a + t * g, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            objs[i] = compute_obj(a_try / s)

    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]

    if 0 < best_idx < n_evals - 1:
        t0, t1, t2 = ts[best_idx - 1], ts[best_idx], ts[best_idx + 1]
        f0, f1, f2 = objs[best_idx - 1], objs[best_idx], objs[best_idx + 1]
        denom = f0 - 2.0 * f1 + f2
        if abs(denom) > 1e-15:
            t_par = t1 - 0.5 * (t2 - t0) * (f0 - f1) / denom
            t_par = float(np.clip(t_par, 0.01, 0.99))
            a_par = np.maximum((1 - t_par) * a + t_par * g, 0)
            s_par = float(np.sum(a_par))
            if s_par > 1e-10:
                f_par = compute_obj(a_par / s_par)
                if f_par < best_obj:
                    best_t = t_par
                    best_obj = f_par

    return best_t, best_obj


def lp_driven_optimize(a, time_budget=150.0, verbose=True):
    """LP-driven optimization with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    no_improve = 0
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break

        g, M = lp_direction_improved(a)

        if g is None:
            conv = compute_conv(a)
            g = smoothed_gradient(a, conv, 200.0)
            g = np.maximum(g, 0)
            s_g = float(np.sum(g))
            if s_g < 1e-10:
                break
            g = g / s_g
            M = None

        best_t, obj_t = line_search(a, g, n_evals=25)
        it += 1

        if obj_t < best_obj - 1e-12:
            a_try = np.maximum((1 - best_t) * a + best_t * g, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                best_obj = obj_t
                best_a = a_try.copy()
                a = a_try
                no_improve = 0
                if verbose and (it % 5 == 0):
                    print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, "
                          f"time={time.monotonic()-start:.0f}s")
        else:
            no_improve += 1
            for t_small in [0.05, 0.02]:
                a_sm = np.maximum((1 - t_small) * a + t_small * g, 0)
                s_sm = float(np.sum(a_sm))
                if s_sm > 1e-10:
                    obj_sm = compute_obj(a_sm / s_sm)
                    if obj_sm < best_obj - 1e-12:
                        best_obj = obj_sm
                        best_a = (a_sm / s_sm).copy()
                        a = best_a
                        no_improve = 0
                        break

            if no_improve >= 6:
                a = best_a * np.exp(0.02 * rng.standard_normal(n))
                a = np.maximum(a, 0)
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new
                else:
                    a = best_a.copy()
                no_improve = 0
                if it > 100:
                    break

    return best_a


def nesterov_gd_optimize(a, time_budget=100.0, tau_init=50.0, tau_end=3000.0,
                         verbose=False):
    """Nesterov accelerated gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.5

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            s_y = 1.0
        else:
            y = y / s_y

        conv_y = compute_conv(y)
        g = smoothed_gradient(y, conv_y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm + 1e-10)
            adapt = min(max(adapt, 0.3 * base_lr), 2.0 * base_lr)
            lr = 0.8 * lr + 0.2 * adapt

        if prev_g is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-8), 4.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)

        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.3
            best_it = it

    if verbose:
        print(f"    NesterovGD: obj={best_obj:.8f}, iters={it}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def multi_method_optimize(a, time_budget=200.0, verbose=False):
    """Run GS + MD + LP + GD in sequence on a given starting point."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)

    # GS spectral flattening
    remaining = time_budget - (time.monotonic() - time.monotonic())
    gs_budget = min(remaining * 0.3, 60.0)
    if gs_budget > 10:
        a_gs = gs_spectral_flatten(best_a, time_budget=gs_budget,
                                   beta_init=0.3, beta_end=0.05,
                                   n_restarts=2, verbose=False)
        obj_gs = compute_obj(a_gs)
        if obj_gs < best_obj:
            best_obj = obj_gs
            best_a = a_gs.copy()
            if verbose:
                print(f"      GS: obj={best_obj:.8f}")

    # Mirror descent
    remaining = time_budget - 0
    md_budget = min(remaining * 0.35, 70.0)
    if md_budget > 10:
        a_md = mirror_descent_optimize(best_a, time_budget=md_budget,
                                       tau_init=30.0, tau_end=1000.0,
                                       verbose=False, blend_k=3, blend_alpha=0.25)
        obj_md = compute_obj(a_md)
        if obj_md < best_obj:
            best_obj = obj_md
            best_a = a_md.copy()
            if verbose:
                print(f"      MD: obj={best_obj:.8f}")

    # LP
    lp_budget = min(remaining * 0.2, 40.0)
    if lp_budget > 10:
        a_lp = lp_driven_optimize(best_a, time_budget=lp_budget, verbose=False)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            if verbose:
                print(f"      LP: obj={best_obj:.8f}")

    # GD polish
    gd_budget = min(remaining * 0.15, 30.0)
    if gd_budget > 5:
        a_gd = nesterov_gd_optimize(best_a, time_budget=gd_budget,
                                    tau_init=100.0, tau_end=2000.0, verbose=False)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_obj:
            best_obj = obj_gd
            best_a = a_gd.copy()
            if verbose:
                print(f"      GD: obj={best_obj:.8f}")

    return best_a

        obj_final3 = compute_obj(a_final3)
        if obj_final3 < best_score:
            best_score = obj_final3
            best_seq = a_final3.copy()
            print(f"  Final GD: obj={best_score:.8f} **")
        else:
            print(f"  Final GD: obj={obj_final3:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
