    n = len(a)

    for j in range(n):

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating minimization with momentum, exact golden-section coordinate
descent, diverse multi-restart perturbations, and adaptive sequence length search.

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None, timeout_hint=None):
    """
    Solve LP: min M s.t. (a_norm * g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_sum = float(np.sum(a))
    if a_sum < 1e-10:
        return None, None
    a_norm = a / a_sum
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass

    for solver in ['CLARABEL', 'SCS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

            a_aff = a[idx].copy()

            c_aff = conv[k_range].copy() - 2.0 * a_j * a_aff

            if k_end < 2 * n - 2:

            # Define 1D objective

                     has_diag=has_diag, max_unaff=max_unaff):

                if S < 1e-10:

def smoothed_gradient_step(a, tau, step_size):
    """One step of gradient descent on the log-sum-exp smoothed objective."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    exp_vals = np.exp(tau * (conv - max_conv))
    w = exp_vals / np.sum(exp_vals)

    # grad[j] = (2/tau) * sum_{m=0}^{n-1} w[j+m] * a[m]
    # Use convolution: grad[j] = (2/tau) * conv(a, w[::-1])[2*n-2-j]
    flipped_w = w[::-1]
    total_len = n + (2 * n - 1) - 1
    size = 1
    while size < total_len:
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(flipped_w, size)
    conv_aw = irfft(fa * fw, size)[:total_len]

    grad = (2.0 / tau) * conv_aw[2 * n - 2 - np.arange(n)]
    return a - step_size * grad


def smoothed_gradient_descent(a, time_budget, seed=42, verbose=False):
    """Projected gradient descent on log-sum-exp smoothed objective with annealing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    tau = 20.0
    step = 0.5 / (tau * 2.0 * n)
    prev_a = a.copy()
    prev_grad = None
    iters = 0

    while time.monotonic() - start < time_budget:
        iters += 1
        # Compute gradient
        conv = compute_conv(a)
        max_conv = np.max(conv)
        exp_vals = np.exp(tau * (conv - max_conv))
        w = exp_vals / np.sum(exp_vals)

        flipped_w = w[::-1]
        total_len = n + (2 * n - 1) - 1
        size = 1
        while size < total_len:
            size <<= 1
        fa = rfft(a, size)
        fw = rfft(flipped_w, size)
        conv_aw = irfft(fa * fw, size)[:total_len]
        grad = (2.0 / tau) * conv_aw[2 * n - 2 - np.arange(n)]

        # Barzilai-Borwein step size
        if prev_grad is not None:
            s_vec = a - prev_a
            y_vec = grad - prev_grad
            num = float(np.dot(s_vec, s_vec))
            den = float(np.dot(s_vec, y_vec))
            if abs(den) > 1e-15:
                step_bb = num / den
                if 1e-10 < step_bb < 10.0 / (tau * n):
                    step = 0.7 * step + 0.3 * step_bb

        prev_grad = grad.copy()
        prev_a = a.copy()

        # Gradient step with backtracking
        t = step
        a_try = a - t * grad
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try > 1e-10:
            a_try = a_try / s_try
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
            a = a_try
        else:
            a = a.copy()

        # Anneal tau
        tau *= 1.02

        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    SGD iter {iters}: tau={tau:.0f}, step={t:.2e}, obj={best_obj:.10f}, time={elapsed:.1f}s")

    return best_a


def resample_sequence(a, n_new):
    """Resample sequence to new length using linear interpolation."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new

    best_obj = compute_obj_norm(a)
    rng = np.random.default_rng(seed)

    # === Phase 1: LP Alternating (40% time) ===
    print("\n--- Phase 1: LP Alternating ---")
    p1_budget = budget_s * 0.40
    a1 = lp_alternate(best_a, time_budget=p1_budget, seed=seed, verbose=True)
    obj1 = compute_obj_norm(a1)
    if obj1 < best_obj:
        best_obj = obj1
        best_a = a1.copy()
        print(f"  P1: {best_obj:.10f} **")
    else:
        print(f"  P1: {obj1:.10f}")

    # === Phase 2: Golden Section CD (15% time) ===
    print("\n--- Phase 2: Golden Section CD ---")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = min(remaining * 0.35, 200.0)
        a2 = golden_section_cd(best_a, time_budget=p2_budget, seed=seed, verbose=True)
        obj2 = compute_obj_norm(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            print(f"  P2: {best_obj:.10f} **")
        else:
            print(f"  P2: {obj2:.10f}")

    # === Phase 3: Smoothed Gradient Descent (10% time) ===
    print("\n--- Phase 3: Smoothed Gradient Descent ---")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p3_budget = min(remaining * 0.25, 120.0)
        a3 = smoothed_gradient_descent(best_a, time_budget=p3_budget, seed=seed + 1000, verbose=False)
        obj3 = compute_obj_norm(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            print(f"  P3: {best_obj:.10f} **")
        else:
            print(f"  P3: {obj3:.10f}")
            # CD polish the SGD result anyway, it might help
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a3b = golden_section_cd(a3, time_budget=min(remaining * 0.3, 40.0), seed=seed+2000, verbose=False)
                obj3b = compute_obj_norm(a3b)
                if obj3b < best_obj:
                    best_obj = obj3b
                    best_a = a3b.copy()
                    print(f"  P3b: {best_obj:.10f} **")

    # === Phase 4: LP from perturbed states (20% time) ===
    print("\n--- Phase 4: LP Multi-Restart ---")
    remaining = deadline - time.monotonic()
    n_restarts = 3
    if remaining > 100:
        per_restart = remaining / (n_restarts + 1)
        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            # Diverse perturbation
            strategy = r % 4
            if strategy == 0:
                # Aggressive coordinate zeroing
                n_zero = max(1, int(len(best_a) * 0.15))
                a_r = best_a.copy()
                zero_idx = rng.choice(len(best_a), size=n_zero, replace=False)
                a_r[zero_idx] = 0
            elif strategy == 1:
                # Fourier with larger amplitude
                n = len(best_a)
                pert = np.zeros(n)
                for k in range(1, 10):
                    freq = k * 2.0 * np.pi / n
                    amp = 0.03 * rng.uniform(0.5, 1.5) / k
                    phase = rng.uniform(0, 2.0 * np.pi)
                    pert += amp * np.cos(freq * np.arange(n) + phase)
                a_r = best_a * np.exp(pert)
            elif strategy == 2:
                # Random reweighting
                a_r = best_a * rng.uniform(0.3, 2.0, size=len(best_a))
            else:
                # Anti-correlation: emphasize where conv is currently low
                conv = compute_conv(best_a)
                target = np.max(conv) - conv
                target = np.maximum(target, 0)
                a_r = best_a * (1.0 + 0.1 * target / max(np.max(target), 1e-10))

            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r < 1e-10:
                a_r = best_a.copy()
            else:
                a_r = a_r / s_r

            alloc = min(per_restart * 0.6, 100.0)
            a_r = lp_alternate(a_r, time_budget=alloc, seed=seed + 5000 + r * 1000, verbose=False)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_r = golden_section_cd(a_r, time_budget=min(remaining * 0.2, 30.0),
                                        seed=seed + 8000 + r * 1000, verbose=False)

            obj_r = compute_obj_norm(a_r)
            if obj_r < best_obj:
                best_obj = obj_r
                best_a = a_r.copy()
                print(f"  P4 restart {r}: {best_obj:.10f} **")
            else:
                print(f"  P4 restart {r}: {obj_r:.10f}")

    # === Phase 5: Length exploration (10% time) ===
    print("\n--- Phase 5: Length Exploration ---")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    n_options = sorted(set([
        n_best - 50, n_best + 50, n_best + 100, n_best + 200,
        n_best - 100, n_best + 300, n_best + 500,
        max(400, n_best - 200)
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2500 and n != n_best]

    for n_try in n_options[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_a, n_try)
        obj_before = compute_obj_norm(a_try)

        alloc = min(remaining * 0.3, 80.0)
        if alloc > 30:
            a_try = lp_alternate(a_try, time_budget=alloc * 0.6, seed=n_try * 7 + seed, verbose=False)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_try = golden_section_cd(a_try, time_budget=min(remaining * 0.25, 25.0),
                                          seed=n_try * 13 + seed, verbose=False)

        obj_try = compute_obj_norm(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            print(f"  ** n={n_try}: {best_obj:.10f} **")
        else:
            print(f"  n={n_try}: {obj_try:.10f} (before LP: {obj_before:.10f})")

    # === Phase 6: Final CD polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("\n--- Phase 6: Final Polish ---")
        a_fin = golden_section_cd(best_a, time_budget=min(remaining * 0.7, 50.0),
                                  seed=seed + 42424, verbose=False)
        obj_fin = compute_obj_norm(a_fin)
        if obj_fin < best_obj:
            best_obj = obj_fin
            best_a = a_fin.copy()
            print(f"  Final: {best_obj:.10f} **")
        else:
            print(f"  Final: {obj_fin:.10f}")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
