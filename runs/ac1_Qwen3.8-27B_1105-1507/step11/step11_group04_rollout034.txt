- **Mirror perturbations**: flip the sequence (this preserves the convolution since conv(a,a) = conv(flip(a), flip(a)))
- **Interpolation**: interpolate to a different length and re-optimize
- **Spectral perturbations**: add random frequency components
- **Anti-convolution perturbations**: perturb in the direction that specifically reduces the peak convolution entries

The last one is interesting. If we know which convolution entries are near the max, we can perturb in a direction that specifically reduces those entries while minimally affecting others.

Let me think about this more carefully. Suppose the active set of convolution entries (those at or near the max) is A = {k_1, ..., k_m}. We want to find a direction d such that:
- (a+d)*(a+d) at positions in A decreases
- (a+d)*(a+d) at other positions doesn't increase too much
- a + d ≥ 0, sum(a+d) = 1

The first-order change in conv at position k is: 2*(a*d)[k] + d[k-j]*d[j] ≈ 2*(a*d)[k] (for small d).

So we want: 2*(a*d)[k_i] < 0 for all i, and 2*(a*d)[k] ≤ ε for k not in A.

This is a linear feasibility problem! We can solve it as an LP:
d = arg min ||d||² (or some norm)
s.t. (a*d)[k_i] ≤ -δ for k_i ∈ A (decrease active entries)
     (a*d)[k] ≤ ε for k ∉ A (don't increase others too much)
     sum(d) = 0 (maintain normalization)
     a + d ≥ 0

This is a quadratic program! We can solve it with cvxpy.

This is a more principled perturbation than random Fourier perturbations.

"""
Optimizes step function coefficients using multiplicative weights (MWU),
analytical coordinate descent, and active-set LP perturbation.
New: MWU provides a fast first-order method with O(n log n) per iteration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv_full(a, b=None):
    """Full linear convolution. If b is None, computes a*a."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def mwu_optimize(a, time_budget, beta=None, eta=None, p=8, seed=42, verbose=True):
    """
    Multiplicative weights optimization.
    Gradients computed via FFT in O(n log n) per iteration.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    if beta is None:
        beta = 10.0
    if eta is None:
        eta = 1.0
    
    best_a = a.copy()
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    iter_count = 0
    patience = 0
    max_patience = 500
    
    # Adaptive beta and eta
    current_beta = beta
    current_eta = eta
    
    while time.monotonic() - start < time_budget:
        iter_count += 1
        
        # Compute soft-max weights
        conv_max = float(np.max(conv))
        if conv_max < 1e-15:
            break
        
        # Use power-based weights for numerical stability
        # w_k = conv_k^p / sum(conv^p)
        conv_clipped = np.maximum(conv, 0)
        wp = conv_clipped ** p
        w = wp / float(np.sum(wp))
        
        # Compute gradient: g[j] = 2 * sum_t w[j+t] * a[t]
        # = 2 * conv(a[::-1], w)[n-1 : 2n-1]
        flip_a = a[::-1]
        grad_full = compute_conv_full(flip_a, w)
        grad = 2.0 * grad_full[n-1:2*n-1]
        
        # Multiplicative update
        # Scale gradient to appropriate range
        grad_mean = float(np.mean(grad))
        grad_std = float(np.std(grad))
        
        # Normalize gradient for stable updates
        if grad_std > 1e-15:
            grad_normed = (grad - grad_mean) / grad_std
        else:
            grad_normed = grad - grad_mean
        
        # Update
        a = a * np.exp(-current_eta * grad_normed)
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            break
        a = a / s
        
        # Recompute convolution
        conv = compute_conv(a)
        obj = 2.0 * n * float(np.max(conv))
        
        if obj < best_obj * (1 - 1e-10):
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        # Adaptive learning rate
        if patience % 100 == 0 and patience > 0:
            current_eta *= 0.95
            if current_eta < 0.01:
                # Try increasing beta to focus more on the max
                current_beta = current_beta * 0.9  # wider soft-max
                current_eta = 0.5
        
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    MWU iter {iter_count}: obj={best_obj:.10f}, eta={current_eta:.4f}, patience={patience}, time={elapsed:.1f}s")
        
        if patience >= max_patience:
            break
    
    if verbose:
        print(f"    MWU done: {iter_count} iters, best={best_obj:.10f}")
    return best_a


def analytical_cd(a, time_budget, seed=42, verbose=True):
    """
    Analytical coordinate descent: for each coordinate, compute the exact
    optimal delta using the stationary point formula.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    sweep = 0
    no_improve = 0
    max_no_improve = 3
    
    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - start > time_budget:
                break
            
            a_j = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)
            
            # Unaffected max
            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))
            
            # Compute candidate deltas from stationary points
            candidates = [0.0]  # always include no change
            
            affected_range = np.arange(k_start, k_end + 1)
            for k in affected_range:
                if k == 2 * j:
                    # Stationary point for quadratic term
                    denom = 1.0 - a_j
                    if abs(denom) > 1e-15:
                        delta = (conv[k] - a_j) / denom
                        if -a_j * 0.99 < delta < a_j * 10:
                            candidates.append(delta)
                else:
                    # Stationary point for linear term
                    idx = k - j  # index into a
                    if 0 <= idx < n and a[idx] > 1e-15:
                        delta = 1.0 - conv[k] / a[idx]
                        if -a_j * 0.99 < delta < a_j * 10:
                            candidates.append(delta)
            
            # Also add some heuristic candidates
            candidates.append(0.0)
            if a_j > 1e-15:
                candidates.append(-a_j * 0.5)  # halve
                candidates.append(-a_j * 0.9)  # nearly zero
                candidates.append(a_j * 0.5)   # double
            candidates.append(1.0 / n)  # mean value target (minus current)
            
            candidates = np.array(candidates)
            candidates = candidates[candidates > -a_j * 0.999]  # maintain non-negativity
            candidates = np.unique(candidates)
            
            best_delta = 0.0
            best_trial_obj = best_obj
            
            for delta in candidates:
                if abs(delta) < 1e-15:
                    continue
                
                S_new = 1.0 + delta
                if S_new <= 1e-10:
                    continue
                
                # Compute max of affected conv entries at this delta
                val_new = a_j + delta
                max_affected = 0.0
                for k in affected_range:
                    if k == 2 * j:
                        new_val = conv[k] + 2 * delta * a_j + delta * delta
                    else:
                        idx = k - j
                        new_val = conv[k] + 2 * delta * a[idx]
                    if new_val > max_affected:
                        max_affected = new_val
                
                max_total = max(max_unaffected, max_affected)
                obj_trial = 2.0 * n * max_total / (S_new * S_new)
                
                if obj_trial < best_trial_obj - 1e-12:
                    best_trial_obj = obj_trial
                    best_delta = delta
            
            if best_delta != 0.0:
                # Apply
                val_new = a_j + best_delta
                a[j] = val_new
                S_new = 1.0 + best_delta
                
                # Update conv
                for k in affected_range:
                    if k == 2 * j:
                        conv[k] += 2 * best_delta * a_j + best_delta * best_delta
                    else:
                        idx = k - j
                        conv[k] += 2 * best_delta * a[idx]
                
                # Normalize
                a /= S_new
                conv /= (S_new * S_new)
                
                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True
        
        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")
    
    return best_a


def vectorized_cd(a, time_budget, seed=42, verbose=True):

        else:

def lp_alternate_v2(a, time_budget, seed=42, verbose=True):
    """
    Improved LP alternating with:
    - Better warm start strategy
    - More aggressive line search
    - Structured perturbations when stuck
    """
    try:
        import cvxpy as cp
    except ImportError:
        return a
    
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    iter_count = 0
    stall = 0
    
    # Build matrix once
    a_norm = a.copy()
    A_matrix = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_matrix[j:j + n, j] = a_norm
    
    while time.monotonic() - start < time_budget - 3:
        t0 = time.monotonic()
        
        # Rebuild A matrix if a changed significantly
        # (In practice, we rebuild every iteration since a changes)
        A_matrix = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_matrix[j:j + n, j] = a
        
        # Solve LP: min M s.t. A_matrix @ g <= M, sum(g) = 1, g >= 0
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A_matrix @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        if prev_g is not None:
            try:
                g.value = prev_g.copy()
            except Exception:
                pass
        
        g_val = None
        for solver in ['CLARABEL', 'SCS', 'ECOS']:
            try:
                prob.solve(solver=getattr(cp, solver))
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(np.asarray(g.value), 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        g_val = g_val / s_g
