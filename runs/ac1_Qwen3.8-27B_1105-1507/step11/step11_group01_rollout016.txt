"""
Hybrid optimizer: exact coordinate descent + LP-dual-guided line search + KKT flattening.
Focuses on squeezing maximum improvement from local optimization with multiple strategies.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns (g, M, dual_y)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    y = cp.Variable(2 * n - 1, nonneg=True)
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:
                    g_val = g_val / g_sum
                    dual_vals = None
                    for c in constraints:
                        if c.dual_value is not None:
                            if len(c.dual_value) == 2 * n - 1:
                                dual_vals = np.maximum(c.dual_value, 0)
                    return g_val, float(prob.value), dual_vals
        except Exception:
            continue
    return None, None, None


def kkt_flatten(a, active_set, eta):
    """KKT-based flattening step. Returns updated a."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    support = np.where(a > 1e-10)[0]
    if len(support) == 0:
        return a

    # Compute v[i] = (1/|K|) * sum_{k in active_set, 0<=k-i<n} a[k-i]
    # v[i] = (1/|K|) * sum_{j: i+j in K} a[j]
    K = set(int(k) for k in active_set)
    nK = len(K)
    if nK == 0:
        return a

    v = np.zeros(n)
    for i in support:
        total = 0.0
        for j in range(n):
            if (i + j) in K:
                total += a[j]
        v[i] = total / nK

    lam = np.mean(v[support])
    residual = v[support] - lam

    # Gradient direction: da[j] += eta * (1/nK) * sum_{i in support, i+j in K} residual[i]
    # Compute this efficiently
    r_ext = np.zeros(n)
    r_ext[support] = residual
    # grad[j] = (1/nK) * sum_{i in support} r_ext[i] * 1[i+j in K]
    # = (1/nK) * (r_ext * 1_K_shifted)[j]
    grad = np.zeros(n)
    for j in support:
        total = 0.0
        for i in support:
            if (i + j) in K:
                total += r_ext[i]
        grad[j] = total / nK

    a_new = a.copy()
    a_new[support] -= eta * grad
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        return a
    return a_new / s_new


def coordinate_descent(a, time_budget=100.0, verbose=False, seed=42):
    """Exact coordinate descent with golden section search on 1D subproblem."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    prev_best = best_obj
    stall = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        sweep += 1

        # Periodically recompute to avoid drift
        if sweep % 3 == 0:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                a = a / S_check
                conv = compute_conv(a)

        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k_lin if 'c_k_lin' in dir() else c_k[mask].copy()
            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            scale = max(v, S / n, 1e-6)
            candidates = [0.0, v * 0.25, v * 0.5, v * 0.75, v,
                          v * 1.25, v * 1.5, v * 2.0,
                          S / (2 * n), S / n, 2.0 * S / n]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            if len(a_ki_lin) > 0:
                max_lin_v = float(np.max(c_k_lin))
            else:
                max_lin_v = 0.0
            quad_v = c_2i
            best_f = max(max_lin_v, quad_v) / (S * S)

            for x in candidates:
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    continue
                f_val = 2.0 * n * max(max_lin, quad_val) / denom
                if f_val < best_f:
                    best_f = f_val
                    best_x = x

            x_lo = max(0.0, best_x - scale * 0.5)
            x_hi = best_x + scale
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            if len(a_ki_lin) > 0:
                lin_c = c_k_lin + 2.0 * a_ki_lin * (c_pt - v)
                fc = max(float(np.max(lin_c)), c_2i + c_pt * c_pt - v * v) / ((S_other + c_pt) ** 2)
                lin_d = c_k_lin + 2.0 * a_ki_lin * (d_pt - v)
                fd = max(float(np.max(lin_d)), c_2i + d_pt * d_pt - v * v) / ((S_other + d_pt) ** 2)
            else:
                fc = (c_2i + c_pt * c_pt - v * v) / ((S_other + c_pt) ** 2)
                fd = (c_2i + d_pt * d_pt - v * v) / ((S_other + d_pt) ** 2)

            for _ in range(50):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    if len(a_ki_lin) > 0:
                        lin_c = c_k_lin + 2.0 * a_ki_lin * (c_pt - v)
                        fc = max(float(np.max(lin_c)), c_2i + c_pt * c_pt - v * v) / ((S_other + c_pt) ** 2)
                    else:
                        fc = (c_2i + c_pt * c_pt - v * v) / ((S_other + c_pt) ** 2)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    if len(a_ki_lin) > 0:
                        lin_d = c_k_lin + 2.0 * a_ki_lin * (d_pt - v)
                        fd = max(float(np.max(lin_d)), c_2i + d_pt * d_pt - v * v) / ((S_other + d_pt) ** 2)
                    else:
                        fd = (c_2i + d_pt * d_pt - v * v) / ((S_other + d_pt) ** 2)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta

        if sweep % 5 == 0:
            obj_now = 2.0 * n * float(np.max(conv)) / (S * S)
            if obj_now < best_obj:
                best_obj = obj_now
                best_a = a.copy()
            if best_obj < prev_best - 1e-14:
                prev_best = best_obj
                stall = 0
            else:
                stall += 1
            if stall > 30:
                break

        if verbose and sweep % 10 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

def lp_line_search(a, g, time_budget=30.0):
    """Line search along direction g from a. Returns best (a_new, obj)."""
    n = len(a)
    conv_a = compute_conv(a)
    conv_ag = np.zeros(len(conv_a))

    # Compute conv(a, g) efficiently
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g, size)
    conv_ag = irfft(fa * fg, size)[:2 * n - 1]

    conv_g = compute_conv(g)

    # f(t) = max_k [(1-t)^2 * ca[k] + 2t(1-t) * cag[k] + t^2 * cg[k]]
    # Try a grid of t values
    best_t = 0.0
    best_obj = np.inf
    s_a = float(np.sum(a))
    s_g = float(np.sum(g))

    for t in np.linspace(0, 1, 50):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        conv_try = (1 - t) ** 2 * conv_a + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_g
        # Recompute exactly for accuracy
        conv_try = compute_conv(a_try)
        obj_try = 2.0 * n * float(np.max(conv_try))
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t

    # Refine around best_t
    t_lo = max(0.0, best_t - 0.05)
    t_hi = min(1.0, best_t + 0.05)
    for t in np.linspace(t_lo, t_hi, 40):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t

    a_new = (1 - best_t) * a + best_t * g
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        return a.copy(), best_obj
    return a_new / s_new, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x * x + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        a = rng.uniform(0.1, 1, n)
        # Add slow variation
        x = np.linspace(0, 2 * np.pi, n)
        a *= (1 + 0.3 * np.cos(x) + 0.2 * np.cos(2 * x))
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 6:
        # Symmetric bump
        c = n // 2
        a = np.exp(-((np.arange(n) - c) ** 2) / (2 * (n // 4) ** 2))
        a = a / a.sum()
    elif mode == 7:
        # Two peaks
        a = np.zeros(n)
        a[n//4] += 1
        a[3*n//4] += 1
        # Smooth
        kernel = np.exp(-np.arange(-n//8, n//8+1)**2 / (2*(n//16)**2))
        a = np.convolve(a, kernel, 'same')
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining exact CD, LP-dual-guided
    line search, KKT flattening, and systematic n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_default = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = 2.0 * parent.size * float(np.max(compute_conv(parent)))
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(n_default) / n_default
        best_score = 2.0 * n_default * float(np.max(compute_conv(best_seq)))
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(800) / 800.0
        best_score = 2.0

    n_best = len(best_seq)
    print(f"Starting optimization from n={n_best}, obj={best_score:.8f}")

    # === Phase 1: Initial CD refinement ===
    print(f"\n=== Phase 1: Initial CD (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 350.0)
    if p1_budget > 20:
        a_cd = coordinate_descent(best_seq, time_budget=p1_budget, verbose=True, seed=seed)
        obj_cd = 2.0 * a_cd.size * float(np.max(compute_conv(a_cd / a_cd.sum())))
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 2: LP + line search iterations ===
    print(f"\n=== Phase 2: LP + Line Search ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 300.0)
    stall_count = 0

    while time.monotonic() < start_time + p2_budget + 15 and time.monotonic() < deadline - 60:
        a_cur = best_seq / best_seq.sum()
        n = len(a_cur)

        g_lp, M_lp, dual_y = solve_lp_minpeak(a_cur)
        if g_lp is None:
            break

        a_new, obj_new = lp_line_search(a_cur, g_lp, time_budget=20.0)
        if obj_new < best_score - 1e-12:
            best_score = obj_new
            best_seq = a_new.copy()
            stall_count = 0
            print(f"  LP-LS: obj={best_score:.8f} **")
        else:
            stall_count += 1
            if stall_count >= 4:
                # Try perturbation
                pert = rng.normal(0, 0.03, n)
                a_pert = best_seq * np.exp(pert)
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                obj_pert = 2.0 * n * float(np.max(compute_conv(a_pert)))
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  Perturb: obj={best_score:.8f} **")
                stall_count = 0
                continue
            if stall_count >= 8:
                break

        # Follow up with short CD
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_cd2 = coordinate_descent(best_seq, time_budget=min(20, remaining * 0.05),
                                       verbose=False, seed=seed + int(time.monotonic() * 10) % 10000)
            obj_cd2 = 2.0 * a_cd2.size * float(np.max(compute_conv(a_cd2 / a_cd2.sum())))
            if obj_cd2 < best_score:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                print(f"  CD after LP: obj={best_score:.8f} **")

    # === Phase 3: KKT flattening ===
    print(f"\n=== Phase 3: KKT Flattening ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 120.0)
    if p3_budget > 10:
        a_kkt = best_seq / best_seq.sum()
        n = len(a_kkt)
        kkt_start = time.monotonic()
        kkt_stall = 0
        eta_kkt = 0.05

        while time.monotonic() - kkt_start < p3_budget:
            conv = compute_conv(a_kkt)
            M = float(np.max(conv))
            active_thresh = M * (1 - 1e-6)
            active_set = np.where(conv >= active_thresh)[0]

            if len(active_set) == 0:
                break

            a_new = kkt_flatten(a_kkt, active_set, eta_kkt)
            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_score - 1e-13:
                best_score = obj_new
                best_seq = a_new.copy()
                a_kkt = a_new
                kkt_stall = 0
                if int((time.monotonic() - kkt_start) / 5) != int((time.monotonic() - kkt_start - 0.1) / 5):
                    print(f"  KKT: obj={best_score:.8f}, eta={eta_kkt:.4f}")
            else:
                kkt_stall += 1
                eta_kkt *= 0.7
                if kkt_stall > 15:
                    # Try larger eta or break
                    eta_kkt = 0.02
                    kkt_stall = 0
                    if time.monotonic() - kkt_start > p3_budget * 0.8:
                        break

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_candidates = sorted(set([
        max(400, n_current - 100), n_current,
        n_current + 100, n_current + 200, n_current + 400,
        600, 700, 900, 1000, 1100, 1200
    ]))
    # Filter
    n_filtered = []
    for nc in n_candidates:
        if 400 <= nc <= 1500:
            if all(abs(nc - nf) > 50 for nf in n_filtered):
                n_filtered.append(nc)
    # Sort by proximity to current best n
    n_filtered.sort(key=lambda x: abs(x - n_current))
    n_filtered = n_filtered[:5]
    print(f"  n candidates: {n_filtered}")

    for n_try in n_filtered:
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break

        if n_try == n_current:
            continue

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        alloc = min(remaining * 0.12, 50.0)
        if alloc > 10:
            a_try = coordinate_descent(a_try, time_budget=alloc, verbose=False, seed=n_try + 7777)

        obj_try = 2.0 * a_try.size * float(np.max(compute_conv(a_try / a_try.sum())))
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts with CD ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = r + 1
        a_r = diverse_init(n_cur, rng, mode=mode)
        blend = 0.4 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * (best_seq / best_seq.sum())
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.15, 50.0)
        if alloc > 15:
            a_r = coordinate_descent(a_r, time_budget=alloc, verbose=False, seed=7000 + r * 111)
            obj_r = 2.0 * a_r.size * float(np.max(compute_conv(a_r / a_r.sum())))
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode {mode}): obj={obj_r:.8f}")

    # === Phase 6: Final intensive CD + LP polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Long CD
        cd_alloc = min(remaining * 0.45, 200.0)
        if cd_alloc > 15:
            a_f = coordinate_descent(best_seq, time_budget=cd_alloc, verbose=True, seed=99999)
            obj_f = 2.0 * a_f.size * float(np.max(compute_conv(a_f / a_f.sum())))
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

        # LP polish
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_cur = best_seq / best_seq.sum()
            g_lp, M_lp, _ = solve_lp_minpeak(a_cur)
            if g_lp is not None:
                a_new, obj_new = lp_line_search(a_cur, g_lp, time_budget=15.0)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    print(f"  Final LP: obj={best_score:.8f} **")
                # CD after LP
                remaining = deadline - time.monotonic()
                if remaining > 10:
                    a_f2 = coordinate_descent(best_seq, time_budget=min(remaining * 0.4, 30.0),
                                              verbose=False, seed=11111)
                    obj_f2 = 2.0 * a_f2.size * float(np.max(compute_conv(a_f2 / a_f2.sum())))
                    if obj_f2 < best_score:
                        best_score = obj_f2
                        best_seq = a_f2.copy()
                        print(f"  CD after Final LP: obj={best_score:.8f} **")

    # === Phase 7: Ultra-fine CD on best ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_final = coordinate_descent(best_seq, time_budget=min(remaining * 0.8, 60.0),
                                     verbose=False, seed=123456)
        obj_final = 2.0 * a_final.size * float(np.max(compute_conv(a_final / a_final.sum())))
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Ultra CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = 2.0

    best_seq = np.maximum(best_seq, 0.0)
    result = [float(min(1000.0, x)) for x in best_seq]

    val = evaluate_sequence(result)
    print(f"\n  evaluate_sequence gives: {val:.8f}")
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        print("  WARNING: invalid result, using fallback")
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
