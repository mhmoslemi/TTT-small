import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, minimize_scalar
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj_norm(a):
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def solve_lp(a, warm_start=None):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except:
            pass
    for solver in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except:
            continue
    return None, None

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of LP alternating minimization, analytically-guided coordinate
descent with exact 1D subproblem solving, and soft-max smoothed projected gradient
optimization. Multi-restart with diverse initializations explores the landscape.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, minimize_scalar
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass
    for solver in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a, time_budget, seed=42, verbose=True):
    """LP alternating minimization with adaptive step size and warm starts."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    iter_count = 0
    stall = 0

    while time.monotonic() - start < time_budget - 3:
        t0 = time.monotonic()
        g_lp, M_lp = solve_lp(a, warm_start=prev_g)
        if g_lp is None:
            break
        prev_g = g_lp
        iter_count += 1

        best_local_obj = best_obj
        best_local_a = a.copy()
        best_t = 0.0

        t_values = [1.0, 0.8, 0.6, 0.4, 0.25, 0.15, 0.1, 0.05, 0.03]
        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_t = t

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall = 0
            if verbose and iter_count % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP {iter_count}: obj={best_obj:.10f}, t={best_t:.3f}, elapsed={elapsed:.0f}s")
        else:
            stall += 1
            if stall >= 6:
                strength = 0.008 + 0.004 * min(stall, 15)
                pert = rng.standard_normal(n) * strength
                a = best_a + pert * best_a
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                prev_g = None
                stall = 0

        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - start
            remaining = time_budget - elapsed
            print(f"    LP iter {iter_count}: best={best_obj:.10f}, M_lp={M_lp:.6f}, rem={remaining:.0f}s")

    if verbose:
        print(f"    LP done: {iter_count} iters, best={best_obj:.10f}")
    return best_a

                res = minimize_scalar(obj_of_v, bounds=(lo, hi), method='bounded',
                                      options={'maxiter': 30, 'xatol': 1e-10})
                if res.success and res.fun < best_obj:
                    best_obj = res.fun
                    best_v = res.x
            except Exception:
                pass

    return best_v, best_obj


def cd_improved(a, time_budget, seed=42, verbose=True):
    """
    Improved coordinate descent with analytical 1D optimization.
    Maintains conv array and updates it incrementally.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    sweep = 0
    no_improve = 0
    max_no_improve = 8

    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            a_j = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)

            # Compute U (unaffected max)
            U = 0.0
            if k_start > 0:
                U = max(U, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                U = max(U, float(np.max(conv[k_end + 1:])))

            best_v, best_v_obj = find_best_v_for_coord(a, conv, j, U, n)

            if best_v != a_j and best_v_obj < best_obj - 1e-12:
                delta = best_v - a_j
                a[j] = best_v
                S_new = 1.0 + delta

                # Update conv in-place
                indices = np.arange(k_start, k_end + 1)
                special = 2 * j
                for idx in range(k_end - k_start + 1):
                    k = k_start + idx
                    if k == special:
                        conv[k] += (best_v * best_v - a_j * a_j)
                    else:
                        conv[k] += 2.0 * delta * a[k - j]

                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S

                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")

    if verbose:
        print(f"    CD done: {sweep} sweeps, best={best_obj:.10f}")
    return best_a


def softmax_obj_and_grad(a, tau):
    """
    Compute soft-max smoothed objective and gradient.
    f_tau(a) = 2*n * tau * log(sum exp(conv[k]/tau)) / (sum a)^2
    Returns (objective, gradient).
    """
    n = len(a)
    conv = compute_conv(a)
    d = float(np.sum(a))
    if d < 1e-10:
        return 1e10, np.zeros(n)

    # Soft-max: S = tau * log(sum exp(conv[k]/tau))
    max_conv = float(np.max(conv))
    exp_vals = np.exp((conv - max_conv) / tau)
    log_sum = max_conv / tau + np.log(np.sum(exp_vals))
    S = tau * log_sum  # This is the soft-max value = tau * log(sum(exp(conv/tau)))
    # Actually: tau * log(sum exp(conv/tau)) = tau * (max_conv/tau + log(sum exp((conv-max_conv)/tau)))
    # = max_conv + tau * log(sum exp((conv-max_conv)/tau))
    S = max_conv + tau * np.log(np.sum(exp_vals))

    obj = 2.0 * n * S / (d * d)

    # Gradient: dS/da[j] = sum_k (conv[k]/tau * exp(conv[k]/tau) / sum exp(conv[k]/tau)) * d(conv[k])/da[j]
    # = sum_k p_k * 2*a[k-j] where p_k = exp(conv[k]/tau) / sum exp(conv[k]/tau)
    p = exp_vals / np.sum(exp_vals)  # softmax weights

    # d(conv[k])/da[j] = 2*a[k-j] for k-j in [0,n-1]
    # So dS/da[j] = 2 * sum_k p_k * a[k-j] = 2 * (p * a)[treated as correlation]
    # This is 2 * conv(p, a)[j] where conv here is cross-correlation
    # Actually: sum_k p_k * a[k-j] for k from 0 to 2n-2
    # Let's compute: for fixed j, sum_k p[k] * a[k-j] where we need 0<=k-j<n and 0<=k<2n-1
    # This is the (2n-1 - 1 - j)th element of conv(p, a) reversed... let me just use FFT.
    # Actually sum_k p[k] * a[k-j] = (p * a reversed)[j] = cross-correlation
    # Let's compute it as: for each j, sum over k of p[k]*a[k-j]
    # This is the convolution of p and reversed a, evaluated at... 
    # Simpler: use the fact that d(conv[p,a][m])/da[j] ... 
    # Actually let me just compute it as: grad_S[j] = 2 * sum_{k} p[k] * a[k-j] where k-j in [0,n-1]
    # = 2 * (p cross-correlation a)[j]
    # The cross-correlation (p a)_j = sum_k p[k] a[k-j] = sum_i p[i+j] a[i]
    # This is the (n-1-j)th element of conv(p, a) if we convolve properly...
    # Let me just use FFT: 
    size = 1
    while size < 4 * n:
        size <<= 1
    fp = rfft(p, size)
    fa = rfft(a, size)
    # conv(p, a) has length 2n-1
    # (p*a)[m] = sum_i p[i]*a[m-i]
    # We want sum_k p[k]*a[k-j] = sum_i p[i+j]*a[i] (substituting k=i+j)
    # = (p*1)[shifted]... 
    # Actually: sum_i a[i] * p[i+j] = cross-correlation of a and p at lag j
    # = conv(a, p reversed)[j + n - 1] ... this is getting confusing
    # Let me just compute it directly using the formula:
    # dS/da[j] = 2 * sum_{m=0}^{2n-2} p[m] * d(conv[m])/da[j] / tau ... 
    # No wait, I already factored that in. Let me redo.
    
    # S = tau * log(sum_k exp(conv[k]/tau))
    # dS/d(conv[k]) = exp(conv[k]/tau) / sum_k exp(conv[k]/tau) = p[k]
    # d(conv[k])/da[j] = 2*a[k-j] if 0<=k-j<n, else 0
    # So dS/da[j] = sum_k p[k] * 2*a[k-j] * [0<=k-j<n]
    # = 2 * sum_{k=j}^{j+n-1} p[k] * a[k-j]
    # = 2 * sum_{i=0}^{n-1} p[j+i] * a[i]
    
    # This is: 2 * (p shifted by j, dot a) = 2 * sum_i a[i] * p[j+i]
    # Let's compute all at once using convolution:
    # sum_i a[i]*p[j+i] for j=0..n-1
    # This is the cross-correlation of a and p: corr[j] = sum_i a[i]*p[j+i]
    # = conv(a, p_reversed)[j + n - 1] ... 
    # Actually, let me define it as: 
    # For the full conv c = conv(a, p) of length 2n-1: c[m] = sum_i a[i]*p[m-i]
    # We want sum_i a[i]*p[j+i] = sum_i a[i]*p[j+i]
    # Let m-i = j+i => m = j+2i... no that's wrong.
    # c[m] = sum_{i:0<=i<n, 0<=m-i<n} a[i]*p[m-i]
    # We want: for each j in [0,n-1]: T[j] = sum_{i:0<=i<n, 0<=j+i<2n-1} a[i]*p[j+i]
    # T[j] = sum_i a[i]*p[j+i]
    # This is the same as sum_i p[j+i]*a[i]
    # If we define q = p reversed, i.e., q[k] = p[2n-2-k], then...
    # Actually the cleanest way: T[j] = (a * p)[j] where * denotes correlation
    # In numpy terms, this is np.correlate(p, a, mode='valid')... not quite.
    # Let me just use FFT:
    # T[j] = sum_i a[i]*p[j+i] for i=0..n-1 (and j+i < 2n-1)
    # = ifft(fft(a) * fft(p)[j:] ...) — no, this is getting too complicated.
    # Let me just compute it with a direct approach using the convolution matrix.
    
    # Actually, the simplest: dS/da[j] = 2 * sum_k p[k] * a[k-j] (with the indicator)
    # Let me compute this as: for the vector (p * a_full) where a_full is a padded to length 2n-1
    # No. Let me just do it the straightforward way:
    # grad_S[j] = 2 * sum_{i=0}^{min(n-1, 2n-2-j)} a[i] * p[j+i]
    # = 2 * dot(a[:L_j], p[j:j+L_j]) where L_j = min(n, 2n-1-j)
    
    # For efficiency, let's use the convolution:
    # If we compute c = conv(a, p) (length 2n-1), then c[m] = sum_i a[i]*p[m-i]
    # T[j] = sum_i a[i]*p[j+i]
    # Note c[m] = sum_i a[i]*p[m-i]. If we set m-i = j+i then m = j+2i... no.
    # Alternative: T[j] = sum_i p[j+i]*a[i]. Let's define r[k] = p[2n-2-k] (reversed p).
    # Then r[2n-2-k] = p[k], so p[j+i] = r[2n-2-j-i].
    # T[j] = sum_i a[i]*r[2n-2-j-i] = conv(a, r)[2n-2-j] (since conv(a,r)[m] = sum_i a[i]*r[m-i])
    # So T[j] = conv(a, p_reversed)[2n-2-j]
    
    p_rev = p[::-1]
    Ap = compute_conv_direct(a, p_rev)  # conv of a and p_rev
    # Actually I need a cross-convolution, not auto-convolution
    # conv(a, p_rev)[m] = sum_i a[i]*p_rev[m-i]
    # T[j] = conv(a, p_rev)[2n-2-j]
    
    # Let me compute cross-conv via FFT
    size2 = 1
    while size2 < 2 * n - 1:
        size2 <<= 1
    fa2 = rfft(a, size2)
    fp_rev = rfft(p_rev, size2)
    Ap_cross = irfft(fa2 * fp_rev, size2)[:2 * n - 1]
    
    # grad_S[j] = 2 * Ap_cross[2n-2-j]
    grad_S = 2.0 * Ap_cross[2 * n - 2 - np.arange(n)]
    
    # Now d(obj)/da[j] = 2n/(d^2) * grad_S[j] - 4n*S/d^3 (chain rule for 1/d^2)
    # obj = 2n*S/d^2
    # d(obj)/da[j] = 2n*(dS/da[j])/d^2 + 2n*S*(-2)/d^3 * (da_sum/da[j])
    # = 2n*grad_S[j]/d^2 - 4n*S/d^3
    grad_obj = 2.0 * n * grad_S / (d * d) - 4.0 * n * S / (d * d * d)
    
    return obj, grad_obj


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n]


def softmax_optimize(a, time_budget, seed=42, verbose=True):
    """
    Soft-max smoothed optimization with projected gradient descent.
    Anneals tau from large to small.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()

    # Anneal tau
    tau_schedule = [5.0, 3.0, 2.0, 1.5, 1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.12, 0.1]
    tau_idx = 0
    alpha = 0.01  # learning rate

    total_iters = 0
    for tau in tau_schedule:
        if time.monotonic() - start > time_budget - 5:
            break
        tau_time = (time_budget - 5) / len(tau_schedule)
        tau_start = time.monotonic()
        
        # Adaptive learning rate for this tau
        alpha = 0.5 / (n * tau + 1)
        
        stall = 0
        while time.monotonic() - tau_start < tau_time:
            t_iter_start = time.monotonic()
            obj_smooth, grad = softmax_obj_and_grad(a, tau)
            
            # Project gradient: remove mean component (to maintain sum=1 approximately)
            grad_proj = grad - np.mean(grad)
            
            # Take step
            a_new = a - alpha * grad_proj
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            
            # Check improvement using true objective
            true_obj = compute_obj_norm(a_new)
            if true_obj < best_obj:
                best_obj = true_obj
                best_a = a_new.copy()
                a = a_new.copy()
                stall = 0
            else:
                stall += 1
                if stall > 50:
                    break
                # Reduce learning rate
                alpha *= 0.95
            
            total_iters += 1
            if total_iters > 3000:
                break
        
        if verbose:
            elapsed = time.monotonic() - start
            print(f"    Softmax tau={tau:.2f}: best={best_obj:.10f}, elapsed={elapsed:.1f}s")

    if verbose:
        print(f"    Softmax done: {total_iters} iters, best={best_obj:.10f}")
    return best_a


def softmax_obj_and_grad(a, tau):
    """
    Compute soft-max smoothed objective and gradient (corrected version).
    Uses cross-convolution for efficient gradient computation.
    """
    n = len(a)
    conv = compute_conv(a)
    d = float(np.sum(a))
    if d < 1e-10:
        return 1e10, np.zeros(n)

    max_conv = float(np.max(conv))
    exp_vals = np.exp((conv - max_conv) / tau)
    sum_exp = np.sum(exp_vals)
    S = max_conv + tau * np.log(sum_exp)
    p = exp_vals / sum_exp

    obj = 2.0 * n * S / (d * d)

    # grad_S[j] = 2 * sum_i a[i] * p[j+i] for valid indices
    # = 2 * cross_conv(a, p_rev)[2n-2-j] where p_rev = p[::-1]
    p_rev = p[::-1]
    Ap_cross = cross_conv(a, p_rev)
    grad_S = 2.0 * Ap_cross[2 * n - 2 - np.arange(n)]

    grad_obj = 2.0 * n * grad_S / (d * d) - 4.0 * n * S / (d * d * d)

    return obj, grad_obj


def make_initialization(n, init_type, rng):
    """Create different types of initial sequences."""
    if init_type == 'uniform':
        a = np.ones(n) / n
    elif init_type == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
        a = a / a.sum()
    elif init_type == 'gaussian':
        x = np.linspace(-3, 3, n)
        a = np.exp(-x ** 2 / 2)
        a = a / a.sum()
    elif init_type == 'power_law':
        a = np.arange(1, n + 1) ** (-0.5)
        a = a / a.sum()
    elif init_type == 'bimodal':
        x = np.linspace(0, n - 1, n)
        a = np.exp(-((x - n * 0.2) ** 2) / (n * 0.1)) + np.exp(-((x - n * 0.8) ** 2) / (n * 0.1))
        a = a / a.sum()
    elif init_type == 'random_smooth':
        a = np.abs(rng.standard_normal(n)) + 0.1
        # Smooth it
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = a / a.sum()
    elif init_type == 'random_sparse':
        a = np.zeros(n)
        k = max(1, n // 4)
        idx = rng.choice(n, k, replace=False)
        a[idx] = 1.0
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return np.maximum(a, 0)


def fourier_perturb(a, rng, strength=0.015):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 10):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_fin2 = softmax_optimize(best_a, time_budget=min(remaining * 0.8, 30.0),
                                      seed=seed + 88888, verbose=False)
            obj_fin2 = compute_obj_norm(a_fin2)
            if obj_fin2 < best_obj:
                best_obj = obj_fin2
                best_a = a_fin2.copy()
                print(f"  Final Softmax: {best_obj:.10f} **")

    # Final validation and return
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
