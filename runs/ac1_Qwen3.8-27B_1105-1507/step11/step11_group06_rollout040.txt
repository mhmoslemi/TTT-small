"""
MWU-based minimax optimization for step function convolution minimization.
Uses multiplicative weight update on the dual (lag distribution) with
exact QP inner solves via cvxpy.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
import warnings
warnings.filterwarnings('ignore')


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_B_matrix(p):
    """Build the PSD matrix B where B[i,j] = p[i+j]."""
    n = len(p) // 2 + 1  # p has length 2n-1, so n = (len(p)+1)//2
    n = (len(p) + 1) // 2
    B = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            B[i, j] = p[i + j]
    return B


def solve_qp(p, n):
    """Solve the inner QP: minimize a^T B a subject to a >= 0, sum(a) = 1."""
    B = build_B_matrix(p)
    a_var = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(a_var) == 1]
    objective = cp.Minimize(cp.quad_form(a_var, cp.psd_wrap(B)))
    prob = cp.Problem(objective, constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if a_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.maximum(a_var.value, 0)
                s = float(np.sum(a_val))
                if s > 1e-10:
                    return a_val / s
        except Exception:
            continue
    return np.ones(n) / n


def mwu_optimize(n, time_budget=600.0, eta=None, verbose=True):
    """Run MWU optimization for the given sequence length n."""
    m = 2 * n - 1  # number of lags
    start = time.monotonic()
    
    # Initialize dual distribution
    p = np.ones(m) / m
    
    # Initialize primal
    a = u_shaped_init(n, sharpness=3.0)
    
    # Running average
    a_avg = a.copy()
    
    # Objective tracking
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    if eta is None:
        eta = 1.0  # will adapt
    
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        T0 = time.monotonic()
        
        # Solve inner QP
        a_new = solve_qp(p, n)
        if a_new is None:
            break
        
        # Compute convolution at a_new
        conv = compute_conv(a_new)
        
        # Update dual using MWU
        # exp(eta * conv) might overflow, so subtract max
        log_weights = eta * (conv - np.max(conv))
        p = p * np.exp(log_weights)
        p = p / np.sum(p)
        
        # Update running average
        a_avg = (it / (it + 1.0)) * a_avg + (1.0 / (it + 1.0)) * a_new
        
        it += 1
        
        # Evaluate running average
        obj_avg = compute_obj(a_avg)
        if obj_avg < best_obj - 1e-12:
            best_obj = obj_avg
            best_a = a_avg.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Also check a_new directly
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        
        # Adaptive eta: reduce if not improving
        if no_improve > 200 and eta > 0.1:
            eta *= 0.5
            no_improve = 0
            it = 0  # reset average
            a_avg = best_a.copy()
        
        # Periodic verbose
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    MWU it{it}: best_obj={best_obj:.8f}, eta={eta:.4f}, "
                  f"elapsed={elapsed:.1f}s")
        
        # Time per iteration check
        iter_time = time.monotonic() - T0
        if iter_time > 1.0:
            # QP is too slow, break
            if verbose:
                print(f"    MWU: iteration too slow ({iter_time:.2f}s), stopping")
            break
    
    return best_a, best_obj


def u_shaped_init(n, sharpness=3.0):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.02 + 5.0 * (2 * x - 1) ** 4)
    else:
        a = 1.0 / (0.1 + 2.0 * np.abs(2 * x - 1))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

"""
Hybrid MWU + LP optimization for step function convolution minimization.
Phase 1: Multiplicative Weight Update on dual with QP inner solves.
Phase 2: LP-based direction finding for polishing.
Phase 3: Multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
import warnings
warnings.filterwarnings('ignore')


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def u_shaped_init(n, sharpness=3.0):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.02 + 5.0 * (2 * x - 1) ** 4)
    else:

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def build_B_matrix(p):
    """Build B[i,j] = p[i+j] for QP inner solve."""
    n = (len(p) + 1) // 2
    i_idx, j_idx = np.meshgrid(np.arange(n), np.arange(n), indexing='ij')
    return p[i_idx + j_idx].copy()


def solve_qp_inner(p, n, solve_time_limit=2.0):
    """Solve min a^T B a s.t. a >= 0, sum(a) = 1."""
    B = build_B_matrix(p)
    # Add tiny regularization for numerical stability
    B = B + 1e-12 * np.eye(n)
    a_var = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(a_var) == 1]
    objective = cp.Minimize(cp.quad_form(a_var, cp.psd_wrap(B)))
    prob = cp.Problem(objective, constraints)
    
    t0 = time.monotonic()
    for solver in [cp.CLARABEL, cp.SCS]:
        if time.monotonic() - t0 > solve_time_limit:
            break
        try:
            prob.solve(solver=solver)
            if a_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.maximum(a_var.value, 0)
                s = float(np.sum(a_val))
                if s > 1e-10:
                    return a_val / s
        except Exception:
            continue
    return None

def build_cross_conv_matrix(a):
    """Build (2n-1) x n matrix A such that A @ g = cross_conv(a, g)."""

        A[j:j + r, j] = a[:r]

def solve_lp_direction(a):

def line_search(a, g, n_coarse=30, n_fine=20, t_max=1.5):

    best_obj = compute_obj(a)
    
    ts = np.linspace(0, t_max, n_coarse)
    for t in ts:
        a_try = np.maximum((1 - t) * a + t * g, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    if 0 < best_t < t_max:
        lo = max(0, best_t - 0.05)
        hi = min(t_max, best_t + 0.05)
        for t in np.linspace(lo, hi, n_fine):
            a_try = np.maximum((1 - t) * a + t * g, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t
    
    return best_t, best_obj

def subgradient_polish(a, time_budget=50.0):
    """Simple subgradient descent on the max-convolution objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.1
    it = 0
    momentum = np.zeros(n)
    beta = 0.8
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))
        
        # Subgradient of max(conv) at a: it's 2 * (B_kmax @ a)
        # B_kmax[i,j] = 1 if i+j = k_max
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_max - i
            if 0 <= j1 < n:
                grad[i] += 2 * a[j1]
        
        lr_t = lr / (1 + 0.01 * it)
        momentum = beta * momentum - lr_t * grad
        a_new = np.maximum(a + momentum, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
        else:
            a_new = a_new / s
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
        it += 1
    
    return best_a, best_obj


def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    if n_src == n_dst:
        return a_src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s

"""
Hybrid MWU + LP optimization for step function convolution minimization.
Uses Multiplicative Weight Update on the dual (lag distribution) with exact QP
inner solves for global optimization, followed by LP-based direction polishing
for final refinement. Also explores multiple sequence lengths.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
import warnings
warnings.filterwarnings('ignore')


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def u_shaped_init(n, sharpness=3.0):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05

             np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01)
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.02 + 5.0 * (2 * x - 1) ** 4)

        a = np.sin(np.pi * x) ** 2 + 0.01
    else:
        a = 1.0 / (0.1 + 2.0 * np.abs(2 * x - 1))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def build_B_matrix(p):
    n = (len(p) + 1) // 2
    i_idx, j_idx = np.meshgrid(np.arange(n), np.arange(n), indexing='ij')
    return p[i_idx + j_idx].copy()


def solve_qp_inner(p, n, time_limit=5.0):
    B = build_B_matrix(p)
    B = B + 1e-12 * np.eye(n)
    a_var = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(a_var) == 1]
    objective = cp.Minimize(cp.quad_form(a_var, cp.psd_wrap(B)))
    prob = cp.Problem(objective, constraints)
    t0 = time.monotonic()
    for solver in [cp.CLARABEL, cp.SCS]:
        if time.monotonic() - t0 > time_limit:
            break
        try:
            prob.solve(solver=solver)
            if a_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.maximum(a_var.value, 0)
                s = float(np.sum(a_val))
                if s > 1e-10:
                    return a_val / s
        except Exception:
            continue
    return None


def solve_qp_gradient(p, n, n_steps=100):
    """Fallback: projected gradient on the QP."""
    B = build_B_matrix(p)
    a = np.ones(n) / n
    # Estimate Lipschitz constant
    max_p = float(np.max(p))
    L = 2.0 * max_p * n + 1.0
    lr = 1.0 / L
    for _ in range(n_steps):
        grad = 2.0 * (B @ a)
        a = a - lr * grad
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    return a

            a = u_shaped_init(n)
        else:
            a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    avg_resets = 0
    consecutive_qp_fail = 0

    while time.monotonic() - start < time_budget:

        if a_qp is None:
            a_qp = solve_qp_gradient(p, n, n_steps=80)
            consecutive_qp_fail += 1
            if consecutive_qp_fail > 50:
                break
        else:
            consecutive_qp_fail = 0

        conv_qp = compute_conv(a_qp)
        log_w = eta * (conv_qp - np.max(conv_qp))
        p_new = p * np.exp(log_w)

        it += 1

                best_a = a_qp.copy()
                no_improve = 0
            else:
                no_improve += 1

        if verbose and it % 300 == 0:
            elapsed = time.monotonic() - start
            print(f"    MWU n={n} it{it}: best={best_obj:.8f}, eta={eta:.3f}, "

            if verbose:
                print(f"    MWU: avg reset #{avg_resets}, eta->{eta:.3f}")

    return best_a, best_obj, time.monotonic() - start


def build_cross_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_cross_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, g, n_coarse=30, n_fine=25, t_max=1.5):
    best_t = 0.0
    best_obj = compute_obj(a)
    ts = np.linspace(0, t_max, n_coarse)
    for t in ts:
        a_try = np.maximum((1.0 - t) * a + t * g, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    if 0.0 < best_t < t_max:
        lo = max(0.0, best_t - 0.05)
        hi = min(t_max, best_t + 0.05)
        for t in np.linspace(lo, hi, n_fine):
            a_try = np.maximum((1.0 - t) * a + t * g, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t
    return best_t, best_obj


def lp_polish(a, time_budget=200.0, max_iters=200, verbose=False):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(777)
    it = 0

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        T0 = time.monotonic()
        g, M_lp = solve_lp_direction(a)
        if g is None:
            break

        t_opt, obj_opt = line_search(a, g)
        if t_opt > 1e-8 and obj_opt < best_obj - 1e-14:
            a_new = np.maximum((1.0 - t_opt) * a + t_opt * g, 0)
            s = float(np.sum(a_new))
            if s > 0.01:
                a_new = a_new / s
                best_obj = obj_opt
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            obj_direct = compute_obj(g)
            if obj_direct < best_obj:
                best_obj = obj_direct
                best_a = g.copy()
                a = g.copy()
                stall = 0
            else:
                stall += 1

        it += 1
        if verbose and stall == 0 and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                  f"elapsed={elapsed:.1f}s")

        if stall >= 4:
            freq = rng.integers(1, max(2, n // 80))
            phase = rng.uniform(0, 2 * np.pi)
            idx = np.arange(n, dtype=np.float64)
            pert = 0.015 * stall * np.cos(2 * np.pi * freq * idx / n + phase)
            a = best_a * np.exp(pert)
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 0.01:
                a = a / s
            else:
                a = best_a.copy()
            stall = 0

    return best_a, best_obj


def subgradient_polish(a, time_budget=50.0, verbose=False):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.1
    it = 0
    momentum = np.zeros(n)
    beta = 0.8
    best_it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_max - i
            if 0 <= j1 < n:
                grad[i] += 2.0 * a[j1]

        lr_t = lr / (1.0 + 0.005 * it)
        momentum = beta * momentum - lr_t * grad
        a_new = np.maximum(a + momentum, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
        else:
            a_new = a_new / s

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            best_it = it
        a = a_new
        it += 1

        if it - best_it > 2000:
            a = best_a.copy()
            momentum = np.zeros(n)
            lr = max(0.02, lr * 0.5)
            best_it = it

    return best_a, best_obj


def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    if n_src == n_dst:
        return a_src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Hybrid MWU + LP optimization to minimize 2*n*max(self_convolution(a)).
    Phase 1: MWU with QP inner solves for global optimization.
    Phase 2: LP-based direction polishing.
    Phase 3: Subgradient refinement.
    Phase 4: Multi-n exploration.
    Phase 5: Final intensive polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 25.0
    rng = np.random.default_rng(seed)

    # Load parent
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        s = float(np.sum(parent_arr))
        if s > 0.01:
            best_seq = parent_arr / s
            best_score = compute_obj(best_seq)
            n_parent = len(best_seq)
            print(f"Parent: n={n_parent}, obj={best_score:.8f}")
        else:
            raise ValueError("sum too small")
    except Exception:
        best_seq = u_shaped_init(n_parent)
        best_score = compute_obj(best_seq)
        print(f"Default init: n={n_parent}, obj={best_score:.8f}")

    # Try alternative inits
    for mode in range(7):
        a_alt = diverse_init(n_parent, rng, mode=mode)
        obj_alt = compute_obj(a_alt)
        if obj_alt < best_score:
            best_score = obj_alt
            best_seq = a_alt.copy()
            print(f"  Better init mode={mode}: obj={best_score:.8f} **")

    a_u = u_shaped_init(n_parent, sharpness=5.0)
    obj_u = compute_obj(a_u)
    if obj_u < best_score:
        best_score = obj_u
        best_seq = a_u
        print(f"  U-shaped(5.0): obj={best_score:.8f} **")

    n_best = len(best_seq)
    print(f"\nStarting best: n={n_best}, obj={best_score:.8f}")

    # === Phase 1: MWU ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        alloc = min(remaining * 0.30, 350.0)
        print(f"\n=== Phase 1: MWU (n={n_best}, budget={alloc:.0f}s) ===")
        a_mwu, obj_mwu, mwu_time = mwu_optimize(
            n_best, initial_a=best_seq, time_budget=alloc, eta=1.0, verbose=True)
        if obj_mwu < best_score:
            best_score = obj_mwu
            best_seq = a_mwu.copy()
            print(f"  MWU improved: {best_score:.8f} **")
        else:
            print(f"  MWU: {obj_mwu:.8f} (no improvement)")

    # === Phase 2: LP Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(remaining * 0.25, 300.0)
        print(f"\n=== Phase 2: LP Polish (budget={alloc:.0f}s) ===")
        a_lp, obj_lp = lp_polish(best_seq, time_budget=alloc, max_iters=300, verbose=True)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP improved: {best_score:.8f} **")
        else:
            print(f"  LP: {obj_lp:.8f} (no improvement)")

    # === Phase 3: Subgradient ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.1, 80.0)
        print(f"\n=== Phase 3: Subgradient (budget={alloc:.0f}s) ===")
        a_sg, obj_sg = subgradient_polish(best_seq, time_budget=alloc)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  SG improved: {best_score:.8f} **")
        else:
            print(f"  SG: {obj_sg:.8f}")

    # === Phase 4: LP Polish Round 2 ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(remaining * 0.20, 200.0)
        print(f"\n=== Phase 4: LP Round 2 (budget={alloc:.0f}s) ===")
        a_lp2, obj_lp2 = lp_polish(best_seq, time_budget=alloc, max_iters=200, verbose=False)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  LP2 improved: {best_score:.8f} **")
        else:
            print(f"  LP2: {obj_lp2:.8f}")

    # === Phase 5: n Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        n_cur = len(best_seq)
        n_candidates = [500, 600, 700, 900, 1000, 1100, 1200, 1400, 1600,
                        n_cur - 100, n_cur + 100, n_cur + 200, n_cur + 400]
        n_options = sorted(set([n for n in n_candidates if 400 <= n <= 2000 and abs(n - n_cur) >= 50]))
        n_options.sort(key=lambda x: abs(x - n_cur))
        n_options = n_options[:5]
        print(f"\n=== Phase 5: n Exploration: {n_options} ===")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 80:
                break
            a_try = interpolate_sequence(best_seq, n_try)
            if a_try is None:
                continue
            alloc_mwu = min(remaining * 0.4, 120.0)
            a_m, obj_m, _ = mwu_optimize(n_try, initial_a=a_try,
                                         time_budget=alloc_mwu, eta=1.0, verbose=False)
            remaining2 = deadline - time.monotonic()
            if remaining2 > 40 and obj_m < best_score + 0.02:
                alloc_lp = min(remaining2 * 0.5, 80.0)
                a_p, obj_p = lp_polish(a_m, time_budget=alloc_lp, max_iters=50, verbose=False)
            else:
                a_p, obj_p = a_m, obj_m

            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  ** n={n_try}: {best_score:.8f} **")
            else:
                print(f"  n={n_try}: {obj_p:.8f}")

    # === Phase 6: Final MWU at best n ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        n_final = len(best_seq)
        alloc = min(remaining * 0.3, 150.0)
        print(f"\n=== Phase 6: Final MWU (n={n_final}, budget={alloc:.0f}s) ===")
        a_f, obj_f, _ = mwu_optimize(n_final, initial_a=best_seq,
                                     time_budget=alloc, eta=0.5, verbose=False)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final MWU: {best_score:.8f} **")

    # === Phase 7: Final LP ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining - 10, 150.0)
        print(f"\n=== Phase 7: Final LP (budget={alloc:.0f}s) ===")
        a_fin, obj_fin = lp_polish(best_seq, time_budget=alloc, max_iters=200, verbose=False)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Final LP: {best_score:.8f} **")

    # === Phase 8: Final squeeze ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining - 5, 40.0)
        if alloc > 10:
            a_sq, obj_sq = subgradient_polish(best_seq, time_budget=alloc)
            if obj_sq < best_score:
                best_score = obj_sq
                best_seq = a_sq.copy()
            rem3 = deadline - time.monotonic()
            if rem3 > 15:
                a_f2, obj_f2 = lp_polish(best_seq, time_budget=min(rem3 - 5, 30.0),
                                         max_iters=30, verbose=False)
                if obj_f2 < best_score:
                    best_score = obj_f2
                    best_seq = a_f2.copy()
                    print(f"  Squeeze: {best_score:.8f} **")

    # Final validation
    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(100) / 100

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = min(best_score, val)
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
