"""
Optimizes step function coefficients using:
1. FFT-based Nesterov accelerated gradient descent (fast coarse optimization)
2. Warm-started coordinate descent with tight brackets (fine optimization)
3. LP polishing via cvxpy (global refinement)
4. Adaptive multi-length exploration

import numpy as np
from scipy.fft import fft, ifft
import time
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = fft(a, size)
    return ifft(fa * fa).real[:2 * n - 1]

def obj_normalized(a):
    """Compute objective with sum(a)=1 normalization."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    return 2.0 * n * float(np.max(compute_conv(a_n)))

def fft_gradient(a, tau):
    """
    Compute gradient of smooth approximation of max(conv(a,a)).
    Uses log-sum-exp with temperature tau.
    Returns (gradient, conv, weights) where gradient is for the normalized a.
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    # Numerically stable log-sum-exp
    exp_vals = np.exp(tau * (conv - max_c))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    # gradient[j] = 2 * sum_k p[k] * a[k-j] (cross-correlation)
    # = 2 * ifft(FFT(p) * conj(FFT(a)))[:n]
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g = ifft(P * np.conj(A)).real[:n]
    return 2.0 * g, conv, p

def nesterov_fft_descent(a, time_budget, tau_init=20.0, tau_end=500.0, verbose=False):
    """Nesterov-accelerated FFT-based gradient descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = obj_normalized(a)
    start = time.monotonic()
    
    # Simplex projection
    def proj_simplex(v, target=1.0):
        n = len(v)
        if np.all(v <= 0):
            return np.ones(n) / n
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0][-1]
        theta = (cssv[rho] - target) / (rho + 1.0)
        return np.maximum(v - theta, 0)
    
    beta = 0.9
    v_mom = np.zeros(n)
    it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * (progress ** 0.7)
        
        # Nesterov lookahead
        y = a + beta * v_mom
        y = proj_simplex(y, 1.0)
        
        grad, conv, p = fft_gradient(y, tau)
        
        # Adaptive learning rate
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            lr = 1.0 / np.sqrt(n)
        else:
            lr = 1.0 / (np.sqrt(n) * g_norm)
        lr = min(lr, 5.0 / np.sqrt(n))
        
        v_mom = beta * v_mom - lr * grad
        a_new = y + v_mom
        a_new = proj_simplex(a_new, 1.0)
        
        obj_new = obj_normalized(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        # Backtracking if we diverged
        if obj_new > best_obj * 1.01:
            a = proj_simplex(a, 1.0)
            v_mom *= 0.5
        else:
            a = a_new
        
        it += 1
        if verbose and it % 5000 == 0:
            print(f"      Nesterov it={it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.1f}s")
    
    return best_a

            ks_list = np.array(ks_list, dtype=np.int64) if ks_list else np.array([], dtype=np.int64)
            a_ki_list = np.array(a_ki_list) if a_ki_list else np.array([])
            c_k_list = np.array(c_k_list) if c_k_list else np.array([])
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2*n-2 else 0.0
            
            def f_1d(x):
                if len(a_ki_list) > 0:
                    lin_vals = c_k_list + 2.0 * a_ki_list * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom
            
            # Determine bracket width based on sweep
            if sweep == 1:
                # First sweep: wider bracket
                scale = max(v, 1.0 / n) * 2.0
            else:
                # Later sweeps: very tight bracket (warm start)
                scale = max(v, 1.0 / n) * 0.1
            
            x_lo = max(0.0, v - scale)
            x_hi = v + scale
            
            # Quick evaluation at current point and a few candidates
            best_x = v
            best_f = f_1d(v)
            
            candidates = [x_lo, (x_lo + x_hi) / 2, x_hi, max(0, v * 0.5), max(0, v * 1.5), 1.0/n]
            for xc in candidates:
                xc = max(0.0, min(float(xc), x_hi * 2))
                fx = f_1d(xc)
                if fx < best_f:
                    best_f = fx
                    best_x = xc
            
            # Tighten bracket around best_x
            local_scale = max(x_hi - x_lo, 1e-8)
            x_lo = max(0.0, best_x - local_scale * 0.5)
            x_hi = best_x + local_scale * 0.5
            
            # Golden section (fewer iterations since bracket is tight)
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(n_gs):
                if x_hi - x_lo < 1e-14 * max(1.0, x_hi):
                    break

                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)
            
            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) > 1e-14:
                delta = x_opt - v
                a[i] = x_opt
                if 0 <= k_quad <= 2*n-2:
                    conv[k_quad] += x_opt*x_opt - v*v
                if len(ks_list) > 0:
                    conv[ks_list] += 2.0 * delta * a_ki_list
                S += delta
                improved += 1
                
                if S > 1e-10:
                    obj = 2.0 * n * float(np.max(conv)) / (S * S)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()
        
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"      CD sweep {sweep}: obj={best_obj:.8f}, improved={improved}, t={elapsed:.1f}s")
        
        if improved == 0 and sweep > 3:
            # Try a wider restart
            if sweep % 10 == 0:
                rng = np.random.default_rng(sweep * 31)
                a = a * np.exp(rng.normal(0, 0.01, n))
                a = np.maximum(a, 0)
                a = a / float(a.sum())
                conv = compute_conv(a)
                S = float(np.sum(a))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def lp_polish_fast(a, time_budget, verbose=False):
    """LP polishing: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = obj_normalized(a)
    start = time.monotonic()
    
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        solved = False
        for solver in [cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        g_val = g_val / s_g
                        solved = True
                        break
            except Exception:
                continue
        
        if not solved:
            break
        
        # Try multiple blending factors
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = obj_normalized(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            # Rebuild matrix for next iteration
            a = a / float(np.sum(a))
            A = build_conv_matrix(a)
        else:
            break
        
        if verbose:
            print(f"      LP: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
    
    return best_a

def penalty_method(a, time_budget, verbose=False):
    """
    Penalty method: minimize max(conv(a,a)) by iteratively solving:
    min ||g - a||^2 s.t. conv(a,g) <= M, sum(g)=1, g>=0
    with decreasing M.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = obj_normalized(a)
    current_M = float(np.max(compute_conv(a)))
    start = time.monotonic()
    
    for outer in range(20):
        if time.monotonic() - start >= time_budget:
            break
        
        # Try to achieve peak < current_M * (1 - 0.02)
        target_M = current_M * 0.98
        
        g = cp.Variable(n, nonneg=True)
        a_norm = a / float(np.sum(a))
        A = build_conv_matrix(a_norm)
        
        constraints = [A @ g <= target_M, cp.sum(g) == 1]
        obj = cp.Minimize(cp.sum_squares(g - a))
        prob = cp.Problem(obj, constraints)
        
        solved = False
        for solver in [cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    if g.value is not None:
                        g_val = np.maximum(g.value, 0)
                        s_g = float(np.sum(g_val))
                        if s_g > 1e-10:
                            g_val = g_val / s_g
                            solved = True
                            break
            except Exception:
                continue
        
        if not solved:
            # Relax target
            target_M = current_M * 1.01
            constraints = [A @ g <= target_M, cp.sum(g) == 1]
            obj = cp.Minimize(cp.sum_squares(g - a))
            prob = cp.Problem(obj, constraints)
            for solver in [cp.CLARABEL, cp.SCS]:
                try:
                    prob.solve(solver=solver)
                    if prob.status in ('optimal', 'optimal_inaccurate') and g.value is not None:
                        g_val = np.maximum(g.value, 0)
                        s_g = float(np.sum(g_val))
                        if s_g > 1e-10:
                            g_val = g_val / s_g
                            solved = True
                            break
                except Exception:
                    continue
        
        if not solved:
            break
        
        # Evaluate
        obj_new = obj_normalized(g_val)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = g_val.copy()
            a = g_val
            current_M = float(np.max(compute_conv(a)))
        else:
            # Can't improve, try with relaxed constraint
            a_try = 0.5 * a + 0.5 * g_val
            a_try = np.maximum(a_try, 0)
            a_try = a_try / float(a_try.sum())
            obj_try = obj_normalized(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                current_M = float(np.max(compute_conv(a)))
            else:
                break
        
        if verbose:
            elapsed = time.monotonic() - start
            print(f"      Penalty outer={outer}: obj={best_obj:.8f}, M={current_M:.6f}, t={elapsed:.1f}s")
    
    return best_a

def proj_simplex(v, target=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = target}."""
    n = len(v)
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cv = np.arange(1, n + 1)
    cond = u * cv > (cssv - target)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0)

For the specific case of non-negative sequences, there's a connection to the "Bohman sequence" which has a triangular PSD. The time-domain Boehman sequence of length n is:

Wait, that's the triangular window. Let me think about what autocorrelation it has.

The triangular window `w[k] = 1 - |2k/(n-1) - 1|` has an autocorrelation that's the convolution of two triangles, which is a "quartic" (or B-spline) shape. This doesn't minimize the peak.

Actually, for the problem of minimizing the peak of the autocorrelation of a non-negative sequence, I think the optimal solution might be related to the "flat-top" window or similar.

Let me try a different approach: **use the fact that the problem is equivalent to finding a probability distribution on {0,...,n-1} that minimizes the peak of its self-convolution**.

This is related to the "minimum peak-to-average power ratio" (PAPR) problem in communications. For real non-negative sequences, there are known results.

One approach that's worked well in practice: **iterative water-filling** or **iterative peak reduction**.

**New idea: Iterative peak reduction in the time domain.**

1. Start with uniform
2. Compute conv = a*a
3. Find k* = argmax(conv)
4. Reduce a[k*-i] for all i (to reduce the convolution peak at k*)
5. Redistribute the reduced mass to positions that don't contribute to the peak
6. Repeat

This is similar to the "TAM" (Toner-Alfert-Mathews) algorithm or "SP" (Schaub-Petersen) algorithm for PAPR reduction.

def peak_reduction(a, time_budget):
    n = len(a)
    a = np.maximum(a, 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_obj = obj_normalized(a)
    start = time.monotonic()
    
    alpha = 0.5  # reduction factor
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak = conv[k_star]
        
        # Positions that contribute to conv[k_star]
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        contributors = np.arange(i_min, i_max + 1)
        
        # Reduce these positions
        total_red = 0
        for i in contributors:
            red = alpha * a[i]
            a[i] -= red
            total_red += red
        
        # Redistribute to non-contributing positions
        non_contrib = np.setdiff1d(np.arange(n), contributors)
        if len(non_contrib) == 0:
            break
        non_contrib_weights = a[non_contrib]
        if np.sum(non_contrib_weights) < 1e-10:
            non_contrib_weights = np.ones(len(non_contrib))
        non_contrib_weights = non_contrib_weights / np.sum(non_contrib_weights)
        a[non_contrib] += total_red * non_contrib_weights
        
        a = np.maximum(a, 0)

"""
Optimizes step function coefficients using:
1. Warm-started coordinate descent with adaptive bracket tightening
2. Iterative peak reduction (novel approach targeting convolution peak directly)
3. LP-CD interleaved optimization for global+local refinement
4. FFT-based Nesterov accelerated gradient descent

import numpy as np
from scipy.fft import fft, ifft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns real array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = fft(a, size)
    return ifft(fa * fa).real[:2 * n - 1]


def obj_normalized(a):
    """Compute objective 2*n*max(conv)/sum(a)^2 assuming a>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a))) / (s * s)


def proj_simplex(v, target):
    """Project v onto simplex {x >= 0, sum(x) = target}."""
    n = len(v)
    if n == 0:
        return v
    if np.all(v <= 0):
        return np.full(n, target / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cv = np.arange(1, n + 1)
    cond = u * cv > (cssv - target)
    if not np.any(cond):
        return np.full(n, target / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def f_1d_eval(x, v, S_other, n, c_k_list, a_ki_list, c_2i):
    """Evaluate the 1D objective function for coordinate descent."""
    if len(a_ki_list) > 0:
        lin_vals = c_k_list + 2.0 * a_ki_list * (x - v)
        max_lin = float(np.max(lin_vals))
    else:
        max_lin = 0.0
    quad_val = c_2i + x * x - v * v
    max_val = max(max_lin, quad_val)
    denom = (S_other + x) * (S_other + x)
    if denom < 1e-30:
        return 1e30
    return 2.0 * n * max_val / denom


def fft_gradient(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g = ifft(P * np.conj(A)).real[:n]
    return 2.0 * g, conv, p


def nesterov_fft_descent(a, time_budget, tau_init, tau_end, verbose):
    """Nesterov-accelerated FFT-based gradient descent with simplex projection."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = obj_normalized(a)
    start = time.monotonic()
    beta = 0.85
    v_mom = np.zeros(n)
    it = 0
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * (progress ** 0.6)
        y = a + beta * v_mom
        y = proj_simplex(y, 1.0)
        grad, conv, p = fft_gradient(y, tau)
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            lr = 2.0 / np.sqrt(n)
        else:
            lr = min(1.0 / (np.sqrt(n) * g_norm), 4.0 / np.sqrt(n))
        v_mom = beta * v_mom - lr * grad
        a_new = y + v_mom
        a_new = proj_simplex(a_new, 1.0)
        obj_new = obj_normalized(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        if obj_new > best_obj * 1.005:
            v_mom *= 0.3
        else:
            a = a_new
        it += 1
        if verbose and it % 10000 == 0:
            print(f"      Nesterov it={it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.1f}s")
    return best_a


def coord_descent_fast(a, time_budget, verbose, seed):
    """
    Fast coordinate descent with warm-started brackets.
    Uses progressively tighter brackets after initial sweeps.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0
    order = np.arange(n)
    current_order_iter = 0
    while time.monotonic() - start < time_budget:
        sweep += 1
        current_order_iter += 1
        if current_order_iter % 7 == 0:
            rng = np.random.default_rng(sweep * 104729)
            order = rng.permutation(n)
        S = float(np.sum(a))
        bracket_scale_base = max(1.0 / n, 1e-8)
        if sweep <= 2:
            bracket_mult = 3.0
            n_gs = 40
        elif sweep <= 5:
            bracket_mult = 1.0
            n_gs = 30
        else:
            bracket_mult = max(0.1, 3.0 / sweep)
            n_gs = max(20, 35 - sweep)
        
        improved = 0
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            k_quad = 2 * i
            ks_list = []
            a_ki_list = []
            c_k_list = []
            for k in range(k_min, k_max + 1):
                if k == k_quad:
                    continue
                ks_list.append(k)
                a_ki_list.append(a[k - i])
                c_k_list.append(conv[k])
            ks_arr = np.array(ks_list, dtype=np.int64) if ks_list else np.array([], dtype=np.int64)
            aki_arr = np.array(a_ki_list) if a_ki_list else np.array([])
            ck_arr = np.array(c_k_list) if c_k_list else np.array([])
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0
            best_x = v
            best_f = f_1d_eval(v, v, S_other, n, ck_arr, aki_arr, c_2i)
            scale = max(v, bracket_scale_base) * bracket_mult
            x_lo = max(0.0, v - scale)
            x_hi = v + scale
            candidates = [x_lo, (x_lo + x_hi) * 0.33, (x_lo + x_hi) * 0.67, x_hi,
                          max(0.0, v * 0.5), max(0.0, v * 1.5), 1.0 / n, 2.0 / n]
            for xc in candidates:
                xc = max(0.0, float(xc))
                if xc > x_hi * 3:
                    xc = x_hi * 3
                fx = f_1d_eval(xc, v, S_other, n, ck_arr, aki_arr, c_2i)
                if fx < best_f:
                    best_f = fx
                    best_x = xc
            local_scale = max(scale * 0.5, 1e-10)
            x_lo = max(0.0, best_x - local_scale)
            x_hi = best_x + local_scale
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-10
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d_eval(c_pt, v, S_other, n, ck_arr, aki_arr, c_2i)
            fd = f_1d_eval(d_pt, v, S_other, n, ck_arr, aki_arr, c_2i)
            for _ in range(n_gs):
                if x_hi - x_lo < 1e-15 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d_eval(c_pt, v, S_other, n, ck_arr, aki_arr, c_2i)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d_eval(d_pt, v, S_other, n, ck_arr, aki_arr, c_2i)
            x_opt = max(0.0, (x_lo + x_hi) * 0.5)
            if abs(x_opt - v) > 1e-15:
                delta = x_opt - v
                a[i] = x_opt
                if 0 <= k_quad <= 2 * n - 2:
                    conv[k_quad] += x_opt * x_opt - v * v
                if len(ks_arr) > 0:
                    conv[ks_arr] += 2.0 * delta * aki_arr
                S += delta
                improved += 1
                if S > 1e-10:
                    obj_val = 2.0 * n * float(np.max(conv)) / (S * S)
                    if obj_val < best_obj:
                        best_obj = obj_val
                        best_a = a.copy()
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"      CD sweep {sweep}: obj={best_obj:.8f}, improved={improved}, t={elapsed:.1f}s")
        if improved == 0 and sweep > 5:
            if sweep % 15 == 0:
                rng2 = np.random.default_rng(sweep * 31 + 7)
                a = a * np.exp(rng2.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / float(a.sum())
                conv = compute_conv(a)
                S = float(np.sum(a))
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def lp_solve_round(a, time_budget, verbose):
    """One round of LP: min M s.t. conv(a,g)<=M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    start = time.monotonic()
    for solver in [cp.CLARABEL, cp.SCS, cp.PDLP]:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_g = float(np.sum(g_val))
                if s_g > 1e-10:
                    return g_val / s_g, float(M.value)
        except Exception:
            continue
    return None, None

def peak_reduction(a, time_budget, verbose, seed):
    """
    Iterative peak reduction: directly reduce elements contributing to
    the maximum convolution peak, redistribute to non-contributing positions.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    alpha_base = 0.3
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        alpha = alpha_base * (1.0 - 0.5 * progress)
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        contributors = np.arange(i_min, i_max + 1)
        total_red = 0.0
        red_amounts = np.zeros(n)
        for i in contributors:
            red = alpha * a[i]
            red_amounts[i] = red
            total_red += red
        if total_red < 1e-15:
            break
        non_contrib = np.setdiff1d(np.arange(n), contributors)
        if len(non_contrib) == 0:
            break
        weights = np.maximum(a[non_contrib], 1e-10)
        weights = weights / float(np.sum(weights))
        a[contributors] = np.maximum(a[contributors] - red_amounts[contributors], 0)
        a[non_contrib] += total_red * weights
        a = np.maximum(a, 0)
        s_a = float(np.sum(a))
        if s_a < 1e-10:
            break
        a = a / s_a
        if it % 50 == 0:
            obj_cur = obj_normalized(a)
            if obj_cur < best_obj:
                best_obj = obj_cur
                best_a = a.copy()
        it += 1
        if verbose and it % 5000 == 0:
            obj_cur = obj_normalized(a)
            print(f"      PeakRed it={it}: obj={obj_cur:.8f}, best={best_obj:.8f}, t={elapsed:.1f}s")
    obj_final = obj_normalized(a)
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = a.copy()
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode):
    """Generate diverse initializations for exploration."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
        return a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x * x + 1e-8)
        return a / a.sum()
    elif mode == 4:
        centers = [n * 0.25, n * 0.75]
        widths = [n * 0.1, n * 0.1]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w * w))
        return a / a.sum()
    else:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        return a / a.sum()
