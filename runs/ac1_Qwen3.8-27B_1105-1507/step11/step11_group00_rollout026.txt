**Alternative: Use the Karush-Kuhn-Tucker conditions**

At the optimum, the subgradient of the objective should be "balanced" in the sense that the directional derivative in any feasible direction is >= 0.

The objective is f(a) = 2n * max_k conv(a,a)[k] / (sum a)^2.

In the interior (all a[i] > 0, sum a = 1), the KKT conditions say:
- For each i with a[i] > 0: subgradient_i = lambda (Lagrange multiplier for sum constraint)
- The subgradient is a convex combination of gradients at the active peaks

If we can find a point where the subgradient is constant (all components equal), that's a stationary point.

This gives us a system of equations:
- sum_i a[i] = 1
- a[i] > 0 for all i (in the interior case)
- Sum of (weights at active peaks) * gradient_i = lambda for all i

The "weights at active peaks" are the dual variables from the max. If we fix which peaks are active, this becomes a system of nonlinear equations.

**Practical approach: Fixed-point iteration**

1. Start with a
2. Compute conv(a,a), find active set S (peaks close to max)
3. Compute subgradient g = (1/|S|) * sum_{k in S} grad_k
4. Update: a[i] <- a[i] * (lambda / g[i])^{eta} for some step size eta
   (This is the multiplicative weights update: move in the direction that equalizes g[i])
5. Repeat

The key insight is that at the optimum, g[i] should be the same for all i (in the interior). So the multiplicative weights update a[i] <- a[i] * (lambda/g[i]) pushes toward equalizing g.

def sparse_lp_direction(a, num_active=20):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for active k, sum(g)=1, g>=0.
    Returns direction g (normalized to sum=1) or None.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv_aa = compute_conv(a_norm)
    max_conv = float(np.max(conv_aa))
    
    # Get active constraints (top num_active peaks)
    active_full = np.argsort(conv_aa)[-num_active:]
    # Also always include the center
    center = n - 1
    if center not in active_full:
        active_full = np.append(active_full, center)
    
    active = sorted(set(active_full.tolist()))
    
    # Build constraint matrix: row k has A[k,j] = a_norm[k-j] if 0<=k-j<n
    K = len(active)
    A = np.zeros((K, n))
    for row, k in enumerate(active):
        for j in range(n):
            diff = k - j
            if 0 <= diff < n:
                A[row, j] = a_norm[diff]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def improved_lp_direction(a, num_active=20):
    """
    Improved LP: min M s.t. 2*conv(a,g)[k] - conv(a,a)[k] <= M for active k, sum(g)=1, g>=0.
    This linearizes the objective better.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv_aa = compute_conv(a_norm)
    
    active_full = np.argsort(conv_aa)[-num_active:]
    center = n - 1
    if center not in active_full:
        active_full = np.append(active_full, center)
    active = sorted(set(active_full.tolist()))
    
    K = len(active)
    A = np.zeros((K, n))
    for row, k in enumerate(active):
        for j in range(n):
            diff = k - j
            if 0 <= diff < n:
                A[row, j] = a_norm[diff]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    # 2*A@g - conv_aa[active] <= M
    rhs = conv_aa[active]
    constraints = [2.0 * (A @ g) - M <= rhs, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def line_search(a, direction, n_evals=30):
    """Line search for best t in [0, 1] to minimize obj((1-t)*a + t*direction)."""
    best_t = 0.0
    best_obj = compute_obj(a)
    ts = np.linspace(0.05, 0.95, n_evals)
    objs = []
    for t in ts:
        a_try = (1 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            objs.append(compute_obj(a_try / s))
        else:
            objs.append(np.inf)
    objs = np.array(objs)
    best_idx = int(np.argmin(objs))
    if objs[best_idx] < best_obj:
        best_t = ts[best_idx]
        best_obj = objs[best_idx]
    # Parabolic refinement
    if 0 < best_idx < n_evals - 1:
        f0, f1, f2 = objs[best_idx - 1], objs[best_idx], objs[best_idx + 1]
        t0, t1, t2 = ts[best_idx - 1], ts[best_idx], ts[best_idx + 1]
        denom = f0 - 2.0 * f1 + f2
        if abs(denom) > 1e-15:
            t_par = t1 - 0.5 * (t2 - t0) * (f0 - f1) / denom
            t_par = float(np.clip(t_par, 0.01, 0.99))
            a_par = np.maximum((1 - t_par) * a + t_par * direction, 0)
            s_par = float(np.sum(a_par))
            if s_par > 1e-10:
                f_par = compute_obj(a_par / s_par)
                if f_par < best_obj:
                    best_t = t_par
                    best_obj = f_par
    return best_t, best_obj


def geometric_init(n, r=1.002, noise=0.0):
    """Create a geometric progression initialization (breaks palindromic symmetry)."""
    j = np.arange(n, dtype=np.float64)
    a = r ** j
    if noise > 0:
        a += noise * np.random.default_rng(0).standard_normal(n)
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def linear_trend_init(n, slope=0.1, noise=0.0):
    """Create a sequence with a linear trend (breaks palindromic symmetry)."""
    j = np.arange(n, dtype=np.float64)
    a = 1.0 + slope * j / n
    if noise > 0:
        a += noise * np.random.default_rng(1).standard_normal(n)
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def mirror_descent_step_adaptive(a, grad, alpha=1.0):
    """
    Adaptive mirror descent step.
    eta is chosen based on the spread of the gradient.
    """
    n = len(a)
    g_mean = float(np.mean(grad))
    g_std = float(np.std(grad))
    if g_mean < 1e-15:
        return a.copy()
    # Adaptive step: larger when gradient is more spread out
    spread = g_std / (abs(g_mean) + 1e-10)
    eta = alpha * min(spread, 5.0) * 0.5
    # Clamp exponent for stability
    exponent = -eta * (grad - g_mean)
    exponent = np.clip(exponent, -20, 20)
    a_new = a * np.exp(exponent)
    s = float(np.sum(a_new))
    if s < 1e-300:
        return a.copy()
    return a_new / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_descent_sweep(a, n_sweeps=3):
    """
    Simple coordinate descent: for each coordinate, try a few multiplicative factors
    and keep the best.
    """
    n = len(a)
    a = a.copy()
    factors = [0.7, 0.85, 1.0, 1.15, 1.3]
    best_obj = compute_obj(a)
    
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            orig_val = a[i]
            best_local_obj = best_obj
            best_factor = 1.0
            for f in factors:
                if f == 1.0:
                    continue
                a[i] = orig_val * f
                s = float(np.sum(a))
                if s > 1e-10:
                    obj = compute_obj(a / s)
                    if obj < best_local_obj - 1e-12:
                        best_local_obj = obj
                        best_factor = f
            if best_factor != 1.0:
                a[i] = orig_val * best_factor
                a = a / float(np.sum(a))
                best_obj = best_local_obj
                improved = True
            else:
                a[i] = orig_val
        if not improved:
            break
    return a


def optimize_sequence(a, time_budget=200.0, verbose=False, seed=42):
    """
    Main optimization: adaptive mirror descent + sparse LP + coordinate descent.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    it = 0
    stall = 0
    lp_count = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1.0:
            break
        
        progress = min(1.0, (time.monotonic() - start) / max(time_budget, 1e-10))
        
        # Every 30 iterations, try LP direction
        if it % 30 == 15 and remaining > 5.0:
            use_improved = (lp_count % 2 == 1)
            if use_improved:
                g_dir = improved_lp_direction(a, num_active=min(30, 10 + it // 10))
            else:
                g_dir = sparse_lp_direction(a, num_active=min(30, 10 + it // 10))
            lp_count += 1
            
            if g_dir is not None:
                t, obj_t = line_search(a, g_dir, n_evals=25)
                if obj_t < best_obj - 1e-13:
                    a_try = np.maximum((1 - t) * a + t * g_dir, 0)
                    s_try = float(np.sum(a_try))
                    if s_try > 1e-10:
                        a_try = a_try / s_try
                        best_obj = obj_t
                        best_a = a_try.copy()
                        a = a_try
                        stall = 0
                        if verbose:
                            print(f"    LP#{lp_count} it{it}: obj={best_obj:.8f}, t={t:.3f}")
                        it += 1
                        continue
        
        # Mirror descent step
        conv = compute_conv(a)
        
        # Choose subgradient type based on progress
        if progress < 0.3:
            # Early: use top-5 for diversity
            grad = subgradient_topk(a, conv, k=5)
        elif progress < 0.7:
            # Middle: use top-10
            grad = subgradient_topk(a, conv, k=10)
        else:
            # Late: use smoothed for refinement
            tau = 100.0 + 500.0 * progress
            grad = subgradient_smoothed(a, conv, tau=tau)
        
        # Adaptive mirror descent step
        alpha = 1.0 * (1.0 + 0.5 * (1.0 - progress))
        a_new = mirror_descent_step_adaptive(a, grad, alpha=alpha)
        
        # Occasionally add perturbation to escape local minima
        if it % 50 == 25 and progress < 0.8:
            strength = 0.01 * (1.0 - progress)
            a_new = a_new * np.exp(strength * rng.standard_normal(n))
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            # Try smaller step
            a_small = mirror_descent_step_adaptive(a, grad, alpha=alpha * 0.3)
            obj_small = compute_obj(a_small)
            if obj_small < best_obj - 1e-14:
                best_obj = obj_small
                best_a = a_small.copy()
                a = a_small
                stall = 0
            elif stall >= 30:
                # Perturb to escape
                strength = 0.03
                a = best_a * np.exp(strength * rng.standard_normal(n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                stall = 0
        
        a = a_new
        it += 1
        
        if verbose and (it % 50 == 0):
            print(f"    MD it{it}: obj={best_obj:.8f}, stall={stall}, t={time.monotonic()-start:.0f}s")
    
    # Final coordinate descent polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5.0:
        a_cd = coordinate_descent_sweep(best_a, n_sweeps=min(5, int(remaining // 3)))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()
            if verbose:
                print(f"    CD polish: obj={best_obj:.8f}")
    
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: explores multiple initializations and sequence lengths,
    using adaptive mirror descent + sparse LP + coordinate descent.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Try to load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                parent_obj = compute_obj(parent)
                best_score = parent_obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default uniform init (n=800): obj={best_score:.8f}")
    
    n_current = len(best_seq)
    print(f"Starting optimization from n={n_current}, obj={best_score:.8f}")
    
    # === Phase 1: Optimize current sequence ===
    print(f"\n=== Phase 1: Optimize n={n_current} ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.3, 400.0)
    if p1_budget > 30:
        a_opt = optimize_sequence(best_seq, time_budget=p1_budget, verbose=True, seed=42)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase 1 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 result: obj={obj_opt:.8f} (no improvement)")
    
    # === Phase 2: Try anti-palindromic initializations ===
    print(f"\n=== Phase 2: Anti-palindromic inits ===")
    remaining = deadline - time.monotonic()
    
    inits = []
    # Geometric progressions with different ratios
    for r in [1.001, 1.003, 1.005, 0.997, 0.995]:
        inits.append(geometric_init(n_current, r=r))
    # Linear trends
    for slope in [0.2, 0.4, -0.2, -0.4]:
        inits.append(linear_trend_init(n_current, slope=slope))
    # Blend of best with these
    for init in inits[:4]:
        blend = 0.3 * best_seq + 0.7 * init
        blend = np.maximum(blend, 0)
        blend /= blend.sum()
        inits.append(blend)
    
    for i_a, a_init in enumerate(inits[:6]):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(remaining * 0.15, 100.0)
        if alloc < 15:
            break
        a_opt = optimize_sequence(a_init, time_budget=alloc, verbose=False, seed=100 + i_a)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Anti-pal init {i_a}: obj={best_score:.8f} **")
        else:
            print(f"  Anti-pal init {i_a}: obj={obj_opt:.8f}")
    
    # === Phase 3: Multi-n exploration ===
    print(f"\n=== Phase 3: Multi-n exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    n_options = []
    for delta in [-200, -100, 100, 200, 400]:
        n_try = n_best + delta
        if 400 <= n_try <= 2500:
            n_options.append(n_try)
    for n_abs in [500, 700, 900, 1100, 1300, 1500]:
        if n_abs not in n_options and 400 <= n_abs <= 2500:
            n_options.append(n_abs)
    n_options = sorted(set(n_options))[:5]
    print(f"  Trying n values: {n_options}")
    
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        
        a_try = resample_sequence(best_seq, n_try)
        # Add a small anti-palindromic perturbation
        j = np.arange(n_try, dtype=np.float64)
        a_try = a_try * (1.0 + 0.05 * (2 * j / (n_try - 1) - 1))
        a_try = np.maximum(a_try, 0)
        a_try /= a_try.sum()
        
        alloc = min(remaining * 0.2, 120.0)
        a_opt = optimize_sequence(a_try, time_budget=alloc, verbose=False, seed=200 + n_try % 100)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")
    
    # === Phase 4: Diverse restarts on best sequence ===
    print(f"\n=== Phase 4: Diverse restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    
    restarts = []
    # Mirror (palindromic) - will likely be worse but try
    a_m = (best_seq + best_seq[::-1]) / 2.0
    a_m = np.maximum(a_m, 0)
    a_m /= a_m.sum()
    restarts.append(("mirror", a_m))
    
    # Fourier perturbation
    pert = np.zeros(n_cur)
    for k in range(1, 8):
        freq = k * 2.0 * np.pi / n_cur
        amp = 0.03 / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n_cur) + phase)
    a_f = best_seq * np.exp(pert)
    a_f = np.maximum(a_f, 0)
    a_f /= a_f.sum()
    restarts.append(("fourier", a_f))
    
    # Random mix
    a_r = 0.7 * best_seq + 0.3 * (rng.uniform(0.5, 1.5, n_cur) / n_cur)
    a_r = np.maximum(a_r, 0)
    a_r /= a_r.sum()
    restarts.append(("random_mix", a_r))
    
    # Strong anti-palindromic
    j = np.arange(n_cur, dtype=np.float64)
    a_ap = best_seq * (1.0 + 0.15 * (2 * j / (n_cur - 1) - 1))
    a_ap = np.maximum(a_ap, 0)
    a_ap /= a_ap.sum()
    restarts.append(("anti_pal", a_ap))
    
    for name, a_r in restarts:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(remaining * 0.2, 100.0)
        if alloc < 15:
            break
        a_opt = optimize_sequence(a_r, time_budget=alloc, verbose=False, seed=300 + hash(name) % 100)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Restart ({name}): obj={best_score:.8f} **")
        else:
            print(f"  Restart ({name}): obj={obj_opt:.8f}")
    
    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final polish ===")
    remaining = deadline - time.monotonic()
    
    if remaining > 30:
        alloc = min(remaining * 0.4, 150.0)
        a_final = optimize_sequence(best_seq, time_budget=alloc, verbose=True, seed=500)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final opt: obj={best_score:.8f} **")
    
    # Final coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_cd = coordinate_descent_sweep(best_seq, n_sweeps=min(10, int(remaining // 2)))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
    
    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
