"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating minimization for initial convergence, then intensive
vectorized coordinate descent with adaptive candidate generation and
multi-restart exploration, combined with length variation to find global-ish optima.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return float(2.0 * n * np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_fast(a, warm_start=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass

    for solver_name in ['CLARABEL', 'ECOS', 'SCS']:
        try:
            prob.solve(solver=getattr(cp, solver_name))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0.0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a, time_budget, seed=42, verbose=True):
    """LP alternating minimization with adaptive step size."""
    n = len(a)

    best_obj = compute_obj_norm(a)

    prev_g = None
    iter_count = 0
    stall = 0

    while time.monotonic() - start < time_budget - 5:

        g_lp, M_lp = solve_lp_fast(a, warm_start=prev_g)
        if g_lp is None:

        # Adaptive line search
        best_local_obj = best_obj
        best_local_a = a.copy()

        t_values = [1.0, 0.8, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04]
        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0.0)

            stall = 0

                prev_g = None

                max_aff = float(np.max(affected))
                max_conv = max(max_unaff, max_aff)
                obj_trial = 2.0 * n * max_conv / (S_new * S_new)

                if obj_trial < best_trial_obj - 1e-14:
                    best_trial_obj = obj_trial
                    best_trial_val = val

            if best_trial_val != a_j:
                # Apply update
                delta = best_trial_val - a_j
                idx = np.arange(k_start, k_end + 1)
                conv[k_start:k_end + 1] += 2.0 * delta * a[idx - j]
                conv[2 * j] = conv[2 * j] + (best_trial_val * best_trial_val - a_j * a_j)
                # Wait, we already added 2*delta*a[idx-j] to conv[2j] above.
                # conv[2j] should be: old_conv[2j] + 2*delta*a[2j-j] + (val^2 - a_j^2) - 2*delta*a[j]
                # = old_conv[2j] + (val^2 - a_j^2)
                # But we added 2*delta*a[j] to it already, so:
                # conv[2j] = old_conv[2j] + 2*delta*a[j] + (val^2 - a_j^2) - 2*delta*a[j]
                # Hmm, let me redo this more carefully.

                # Actually, the convolution update when changing a[j] from a_j to val:
                # conv_new[k] = conv_old[k] + delta*(a[k-j] + a[k-j]) for k where both are valid
                # except at k=2j where it's delta^2 + 2*delta*a_j = val^2 - a_j^2... 
                # No: conv[k] = sum_i a_i * a_{k-i}
                # When a_j changes to val:
                # conv_new[k] = conv_old[k] + delta * (a_{k-j} + a_{k-j}) if k-j != j
                #             = conv_old[k] + delta * a_{k-j} + delta * a_{k-j} if k-j != j
                #             = conv_old[k] + 2*delta*a_{k-j} if k-j != j
                # conv_new[2j] = conv_old[2j] - a_j^2 + val^2 = conv_old[2j] + val^2 - a_j^2
                # But we also added delta * a_j to conv[2j] from the general formula
                # So: conv_new[2j] = conv_old[2j] + 2*delta*a_j + (val^2 - a_j^2 - 2*delta*a_j)
                # = conv_old[2j] + val^2 - a_j^2
                # So the correction is: after adding 2*delta*a[idx-j] everywhere,
                # we need to fix conv[2j] to be conv_old[2j] + val^2 - a_j^2
                # i.e., subtract 2*delta*a_j and add val^2 - a_j^2 to conv[2j]
                # Net change at 2j: val^2 - a_j^2 - 2*delta*a_j = val^2 - a_j^2 - 2*(val-a_j)*a_j
                # = val^2 - a_j^2 - 2*val*a_j + 2*a_j^2 = val^2 - 2*val*a_j + a_j^2 = (val-a_j)^2 = delta^2

                # So the correct update is:
                # conv[k] += 2*delta*a[k-j] for k != 2j (in range)
                # conv[2j] += delta^2

                # Let me redo the apply properly:
                # First undo what we did:
                conv[k_start:k_end + 1] -= 2.0 * delta * a[idx - j]
                conv[2 * j] -= (best_trial_val * best_trial_val - a_j * a_j) + 2.0 * delta * a[idx - j][2 * j - k_start]

                # This is getting messy. Let me just recompute properly.
                # The correct update:
                # For k in [j, j+n-1] (valid range for conv to be affected):
                #   conv_new[k] = conv_old[k] + 2*delta*a[k-j] - (delta^2 if k==2j else 0) + ...
                
                # Actually let me just redo from scratch with the correct formula:
                # conv[k] = sum_{i: valid} a[i]*a[k-i]
                # After changing a[j] to a[j]+delta:
                # conv_new[k] = conv_old[k] + delta*(a[k-j] if k-j != j else (a[j]+delta))
                #                                + delta*(a[j] if j != k-j else (a[j]+delta))
                # For k != 2j (i.e., k-j != j):
                #   conv_new[k] = conv_old[k] + delta*a[k-j] + delta*a[j]  ... wait no
                #   The terms involving a[j] are: a[j]*a[k-j] and a[k-j]*a[j] (if k-j is valid)
                #   New: (a[j]+delta)*a[k-j] + a[k-j]*(a[j]+delta) if both i=j and i=k-j are valid and j != k-j
                #   = old + delta*a[k-j] + delta*a[k-j] = old + 2*delta*a[k-j]
                # For k = 2j:
                #   The term a[j]*a[j] becomes (a[j]+delta)^2
                #   conv_new[2j] = conv_old[2j] + 2*delta*a[j] + delta^2 = conv_old[2j] + delta*(2*a[j]+delta)
                #   = conv_old[2j] + val^2 - a_j^2

                # So the correct update is:
                # conv[k] += 2*delta*a[k-j] for k in affected range, k != 2j
                # conv[2j] += val^2 - a_j^2

                # Let me redo:
                # First, the general update:
                conv[k_start:k_end + 1] += 2.0 * delta * a[idx - j]
                # Then correct the diagonal:
                # We added 2*delta*a[j] at position 2j, but we should have added val^2 - a_j^2
                # So correct: conv[2j] += (val^2 - a_j^2) - 2*delta*a[j] = delta^2
                conv[2 * j] += delta * delta

                a[j] = best_trial_val
                S_new = 1.0 + delta

                # Normalize
                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S

                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")

    return best_a


def cd_polish_v2(a, time_budget, seed=42, verbose=True):
    """
    Improved coordinate descent with correct convolution update.
    Uses the fact that changing a[j] by delta:
      - adds 2*delta*a[k-j] to conv[k] for k in [j, j+n-1], k != 2j
      - adds delta*(2*a[j]+delta) to conv[2j]  (i.e., val^2 - a_j^2)
    Then renormalize by 1/(1+delta).
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = float(2.0 * n * np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    sweep = 0
    no_improve = 0
    max_no_improve = 10
    mean_val = 1.0 / n

    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            a_j = a[j]
            k_start = j
            k_end = j + n - 1  # affected range is [j, j+n-1]

            # Unaffected max
            max_unaff = 0.0
            if k_start > 0:
                mu1 = float(np.max(conv[:k_start]))
                if mu1 > max_unaff:
                    max_unaff = mu1
            if k_end < 2 * n - 2:
                mu2 = float(np.max(conv[k_end + 1:]))
                if mu2 > max_unaff:
                    max_unaff = mu2

            # Current state
            affected_curr = conv[k_start:k_end + 1]
            curr_max = max(max_unaff, float(np.max(affected_curr)))
            current_obj = 2.0 * n * curr_max  # since sum=1

            # Generate candidates
            if a_j < 1e-12:
                candidates = np.array([mean_val * 0.05, mean_val * 0.1, mean_val * 0.2,
                                       mean_val * 0.5, mean_val, mean_val * 2,
                                       mean_val * 4])
            else:
                # Adaptive: small steps + base steps
                ratios = np.array([0.2, 0.3, 0.5, 0.7, 0.8, 1.2, 1.5, 2.0, 4.0])
                candidates = a_j * ratios
                base = mean_val * np.array([0.3, 0.5, 0.8, 1.0, 1.5, 3.0, 6.0])
                candidates = np.concatenate([candidates, base])
                candidates = candidates[(candidates >= 0) & (candidates < 0.4)]
                candidates = np.unique(candidates)

            best_trial_obj = current_obj
            best_trial_val = a_j

            # Precompute idx - j for the affected range
            idx_minus_j = np.arange(k_start - j, k_end - j + 1)  # = arange(0, n)
            a_slice = a[idx_minus_j]  # a[0..n-1] which is just a itself!
            
            # Actually idx_minus_j = arange(0, n) always!
            # So a[idx - j] = a[0:n] = a (the whole vector)
            # And 2j is at position 2j - j = j in the affected slice

            for val in candidates:
                val = float(val)
                delta = val - a_j
                if abs(delta) < 1e-15:
                    continue

                S_new = 1.0 + delta
                if S_new < 1e-10:
                    continue

                # Compute affected conv after update (before normalization)
                # new_aff = affected_curr + 2*delta*a (element-wise)
                # then new_aff[j] = affected_curr[j] + 2*delta*a_j + delta^2
                #   (because at position 2j in full conv = position j in affected slice)
                new_aff = affected_curr + 2.0 * delta * a
                new_aff[j] = affected_curr[j] + val * val - a_j * a_j

                max_new_aff = float(np.max(new_aff))
                max_conv_new = max(max_unaff, max_new_aff)
                obj_trial = 2.0 * n * max_conv_new / (S_new * S_new)

                if obj_trial < best_trial_obj - 1e-14:
                    best_trial_obj = obj_trial
                    best_trial_val = val

            if best_trial_val != a_j:
                # Apply the update
                delta = best_trial_val - a_j
                a[j] = best_trial_val
                S_new = 1.0 + delta

                # Update conv:
                # conv[k_start:k_end+1] += 2*delta*a (element-wise, since idx-j = 0..n-1)
                conv[k_start:k_end + 1] += 2.0 * delta * a
                # Fix the diagonal: we added 2*delta*a[j] at position 2j in conv
                # but should have added val^2 - a_j^2
                # Correction: conv[2j] += (val^2 - a_j^2) - 2*delta*a[j] = delta^2
                conv[2 * j] += delta * delta

                # Normalize
                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S

                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD v2 sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")

    return best_a


def fourier_perturb(a, rng, strength=0.015):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 10):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_perturb(a, rng, strength=0.02):
    """Simple random multiplicative perturbation."""
    n = len(a)
    pert = rng.standard_normal(n) * strength
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating + intensive CD polishing +
    multi-restart exploration + length variation.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    # Load or initialize
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0.0)
        if a.size == 0 or a.sum() < 0.01:
            raise ValueError("Invalid parent")
        a = a / a.sum()
        parent_obj = compute_obj_norm(a)
        print(f"Parent: n={a.size}, obj={parent_obj:.10f}, score={2*a.size*np.max(compute_conv(a))/(1.0)**2:.10f}")
    except Exception:
        a = np.ones(800) / 800.0
        parent_obj = compute_obj_norm(a)
        print(f"Default: n=800, obj={parent_obj:.10f}")

    best_a = a.copy()
    best_obj = parent_obj
    best_score = 2.0 * len(best_a) * float(np.max(compute_conv(best_a)))
    rng = np.random.default_rng(seed)

    # === Phase 1: LP Alternating (35% time) ===
    print("\n--- Phase 1: LP Alternating ---")
    p1_budget = budget_s * 0.35
    a1 = lp_alternate(best_a, time_budget=p1_budget, seed=seed, verbose=True)
    obj1 = compute_obj_norm(a1)
    if obj1 < best_obj:
        best_obj = obj1
        best_a = a1.copy()
        best_score = 2.0 * len(best_a) * float(np.max(compute_conv(best_a)))
        print(f"  P1: obj={best_obj:.10f}, score={best_score:.10f} **")

    # === Phase 2: Intensive CD (35% time) ===
    print("\n--- Phase 2: Intensive CD ---")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = min(remaining * 0.55, 450.0)
        a2 = cd_polish_v2(best_a, time_budget=p2_budget, seed=seed + 100, verbose=True)
        obj2 = compute_obj_norm(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            best_score = 2.0 * len(best_a) * float(np.max(compute_conv(best_a)))
            print(f"  P2: obj={best_obj:.10f}, score={best_score:.10f} **")
        else:
            print(f"  P2: obj={obj2:.10f}")

    # === Phase 3: LP from perturbed state (15% time) ===
    print("\n--- Phase 3: LP Restart ---")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        a_pert = fourier_perturb(best_a, rng, strength=0.01)
        p3_budget = min(remaining * 0.5, 220.0)
        a3 = lp_alternate(a_pert, time_budget=p3_budget, seed=seed + 5000, verbose=False)
        obj3 = compute_obj_norm(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            best_score = 2.0 * len(best_a) * float(np.max(compute_conv(best_a)))
            print(f"  P3: obj={best_obj:.10f}, score={best_score:.10f} **")
        else:
            print(f"  P3: obj={obj3:.10f}")
        # CD polish on the LP result
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a3p = cd_polish_v2(a3, time_budget=min(remaining * 0.3, 80.0), seed=seed + 6000, verbose=False)
            obj3p = compute_obj_norm(a3p)
            if obj3p < best_obj:
                best_obj = obj3p
                best_a = a3p.copy()
                best_score = 2.0 * len(best_a) * float(np.max(compute_conv(best_a)))
                print(f"  P3 polish: obj={best_obj:.10f} **")

    # === Phase 4: CD Polish again (10% time) ===
    print("\n--- Phase 4: CD Polish 2 ---")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p4_budget = min(remaining * 0.5, 200.0)
        a4 = cd_polish_v2(best_a, time_budget=p4_budget, seed=seed + 9000, verbose=True)
        obj4 = compute_obj_norm(a4)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()
            best_score = 2.0 * len(best_a) * float(np.max(compute_conv(best_a)))
            print(f"  P4: obj={best_obj:.10f}, score={best_score:.10f} **")
        else:
            print(f"  P4: obj={obj4:.10f}")

    # === Phase 5: Length exploration (remaining time) ===
    print("\n--- Phase 5: Length Exploration ---")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    n_options = sorted(set([
        max(400, n_best - 50), max(400, n_best - 20),
        n_best + 20, n_best + 50, n_best + 100,
        max(400, n_best - 100), n_best + 150
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and n != n_best]

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, n_try)
        a_try = np.interp(x_new, x_old, best_a)
        a_try = np.maximum(a_try, 0.0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s

        alloc = min(remaining * 0.35, 150.0)
        if alloc > 50:
            a_try = lp_alternate(a_try, time_budget=alloc * 0.5, seed=n_try * 7, verbose=False)
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_try = cd_polish_v2(a_try, time_budget=min(remaining * 0.3, 60.0),
                                     seed=n_try * 13, verbose=False)

        obj_try = compute_obj_norm(a_try)
        score_try = 2.0 * n_try * float(np.max(compute_conv(a_try)))
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            best_score = score_try
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_obj:.10f}, score={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}, score={score_try:.10f}")

    # === Phase 6: Final CD polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n--- Phase 6: Final Polish ---")
        a_fin = cd_polish_v2(best_a, time_budget=min(remaining * 0.85, 120.0),
                             seed=seed + 42424, verbose=False)
        obj_fin = compute_obj_norm(a_fin)
        if obj_fin < best_obj:
            best_obj = obj_fin
            best_a = a_fin.copy()
            best_score = 2.0 * len(best_a) * float(np.max(compute_conv(best_a)))
            print(f"  Final: obj={best_obj:.10f}, score={best_score:.10f} **")
        else:
            print(f"  Final: obj={obj_fin:.10f}")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    return result
