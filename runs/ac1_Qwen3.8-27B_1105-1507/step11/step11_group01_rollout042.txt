        try:
            res = minimize(
                smooth_obj_grad,
                w_init,
                args=(tau,),
                jac=True,
                method='L-BFGS-B',
                callback=cb,
                options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 40}
            )
        except Exception as e:
            if verbose:
                print(f"  L-BFGS tau={tau} error: {e}")
            continue

        w_new = res.x
        w_max = float(np.max(w_new))
        a_new = np.exp(w_new - w_max)
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()

        obj_new = obj_normalized(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        if verbose:
            elapsed = time.monotonic() - start
            print(f"  L-BFGS tau={tau}: obj={best_obj:.8f}, t={elapsed:.1f}s, iters={res.nit}")

    return best_a


def cd_1d_obj(x, v, S_other, n, c_lin, a_lin, c_2i):
    """1D objective for a single coordinate in CD."""
    if len(c_lin) > 0:
        lin = c_lin + 2.0 * a_lin * (x - v)
        ml = float(np.max(lin))
    else:
        ml = 0.0
    quad = c_2i + x * x - v * v
    mv = max(ml, quad)
    denom = (S_other + x) * (S_other + x)
    if denom < 1e-30:
        return 1e30
    return 2.0 * n * mv / denom


def coordinate_descent(a, time_budget=100.0, seed=42, verbose=False):
    """Exact coordinate descent with golden section on 1D subproblems."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    sweep = 0
    stall = 0
    prev_best = best_obj

    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-15:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_vals = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_lin = a_ki[mask]
            c_lin = c_vals[mask]

            c_2i = float(conv[2 * i]) if 0 <= 2 * i <= 2 * n - 2 else 0.0

            mid = S / n
            cands = [0.0, v * 0.25, v * 0.5, v * 0.75, v, v * 1.5, v * 2.0,
                     mid * 0.5, mid, mid * 2.0, mid * 4.0]
            cands = sorted(set(max(0.0, float(c)) for c in cands))

            bx = v
            bf = cd_1d_obj(v, v, S_other, n, c_lin, a_lin, c_2i)
            for c in cands:
                fc = cd_1d_obj(c, v, S_other, n, c_lin, a_lin, c_2i)
                if fc < bf:
                    bf = fc
                    bx = c

            scale = max(v, mid, 1e-8)
            lo = max(0.0, bx - scale)
            hi = bx + 2.0 * scale
            c1 = hi - gr * (hi - lo)
            d1 = lo + gr * (hi - lo)
            fc1 = cd_1d_obj(c1, v, S_other, n, c_lin, a_lin, c_2i)
            fd1 = cd_1d_obj(d1, v, S_other, n, c_lin, a_lin, c_2i)

            for _ in range(35):
                if hi - lo < 1e-16 * max(1.0, hi):
                    break
                if fc1 < fd1:
                    hi = d1
                    d1 = c1
                    fd1 = fc1
                    c1 = hi - gr * (hi - lo)
                    fc1 = cd_1d_obj(c1, v, S_other, n, c_lin, a_lin, c_2i)
                else:
                    lo = c1
                    c1 = d1
                    fc1 = fd1
                    d1 = lo + gr * (hi - lo)
                    fd1 = cd_1d_obj(d1, v, S_other, n, c_lin, a_lin, c_2i)

            xo = max(0.0, (lo + hi) / 2.0)

            if abs(xo - v) > 1e-15:
                delta = xo - v
                a[i] = xo
                if 0 <= 2 * i <= 2 * n - 2:
                    conv[2 * i] += xo * xo - v * v
                if len(ks_lin) > 0:
                    conv[ks_lin] += 2.0 * delta * a_lin
                S += delta

                s_now = S
                if s_now > 1e-10:
                    obj_now = 2.0 * n * float(np.max(conv)) / (s_now * s_now)
                    if obj_now < best_obj:
                        best_obj = obj_now
                        best_a = a.copy()

        if sweep % 8 == 0:
            conv = compute_conv(a)
            S = float(np.sum(a))
            if S > 1e-10:
                obj_now = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj_now < best_obj:
                    best_obj = obj_now
                    best_a = a.copy()
            if best_obj < prev_best - 1e-14:
                prev_best = best_obj
                stall = 0
            else:
                stall += 1
            if stall > 8 and sweep > 30:
                break

        if verbose and sweep % 20 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

    return best_a


def subgradient_bb(a, time_budget=100.0, seed=42, verbose=False):
    """KKT subgradient descent with Barzilai-Borwein adaptive steps and backtracking."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    x_prev = a.copy()
    g_prev = None
    alpha = 0.1 / n
    it = 0
    patience = 0
    bb_count = 0

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 2:
            break

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        tol = max_conv * 1e-8
        argmax_mask = conv >= max_conv - tol
        argmax_set = np.where(argmax_mask)[0]

        # Subgradient: g[i] = 2 * sum_{k in argmax} a[k-i] / |argmax|
        g = np.zeros(n)
        for k in argmax_set:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                g[lo:hi + 1] += a[k - hi:k - lo + 1]
        g /= max(len(argmax_set), 1)
        # Multiply by 2 (from derivative of conv wrt a)
        g *= 2.0

        # Center the gradient (since we're on the simplex, the direction should be perpendicular to 1)
        g_centered = g - np.mean(g)

        # BB step size
        if g_prev is not None:
            s_vec = a - x_prev
            y_vec = g - g_prev
            sy = float(s_vec @ y_vec)
            yy = float(y_vec @ y_vec)
            ss = float(s_vec @ s_vec)
            if yy > 1e-20 and sy > 1e-20:
                alpha_new = ss / sy
                alpha = 0.5 * alpha + 0.5 * alpha_new
                alpha = min(alpha, 1.0 / np.linalg.norm(g_centered + 1e-10))
                bb_count += 1
            elif yy > 1e-20:
                alpha_new = sy / yy if sy > 0 else alpha
                alpha = 0.7 * alpha + 0.3 * max(alpha_new, 1e-6)

        # Backtracking line search
        a_try = a - alpha * g_centered
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            a_try = a.copy()
        else:
            a_try = a_try / s_try

        obj_try = obj_normalized(a_try)
        c1 = 1e-4
        rhs = obj_normalized(a) + c1 * alpha * float(g_centered @ (a_try - a))
        backtrack = 0
        while obj_try > rhs and backtrack < 20 and alpha > 1e-10:
            alpha *= 0.5
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                a_try = a.copy()
                break
            a_try = a_try / s_try
            obj_try = obj_normalized(a_try)
            rhs = obj_normalized(a) + c1 * alpha * float(g_centered @ (a_try - a))
            backtrack += 1

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            patience = 0
        else:
            patience += 1
            if patience > 30:
                # Perturbation to escape
                pert = rng.normal(0, 0.02, n)
                a_pert = a * np.exp(pert)
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                obj_pert = obj_normalized(a_pert)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                a = a_pert
                alpha = 0.1 / n
                x_prev = a.copy()
                g_prev = None
                patience = 0
                it = 0

        x_prev = a.copy()
        a = a_try
        g_prev = g.copy()
        it += 1

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    Subgrad it{it}: obj={best_obj:.8f}, alpha={alpha:.6f}, t={elapsed:.1f}s")

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def asymmetric_init(n, mode, rng):
    """Generate diverse asymmetric initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        # Linear ramp
        a = np.linspace(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        # Exponential decay
        a = np.exp(-np.linspace(0, 3, n))
        a = a / a.sum()
    elif mode == 3:
        # Two bumps (asymmetric placement)
        a = np.zeros(n)
        c1 = n // 4
        c2 = 3 * n // 4
        w1 = n // 8
        w2 = n // 6
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w1 ** 2))
        a += 0.7 * np.exp(-((np.arange(n) - c2) ** 2)
