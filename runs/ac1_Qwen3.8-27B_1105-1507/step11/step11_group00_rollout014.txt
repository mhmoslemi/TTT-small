import numpy as np

from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

    """
    Compute gradient of sum_k p[k]*conv(a,a)[k] w.r.t. a.
    grad[i] = sum_k p[k] * d(conv[k])/da[i] = 2*sum_k p[k] * a[k-i] (valid k-i)
    = 2 * (flip(a) * p)[i+n-1]
    """
    n = len(a)
    b = a[::-1]
    N_fft = 1
    while N_fft < 3 * n - 1:
        N_fft <<= 1
    c = irfft(rfft(b, N_fft) * rfft(p, N_fft), N_fft)
    return 2.0 * c[n - 1:2 * n - 1]


def smoothed_gradient(a, conv, tau):
    """Compute gradient of LSE(tau, conv(a,a)) with respect to a."""
    n = len(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    if shifted.max() > 500:
        shifted = shifted - shifted.max()
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(2 * n - 1)
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    return gradient_from_p(a, p)


def topk_subgradient(a, conv, k=3):
    """Subgradient using only the top-k peaks (uniform weights)."""
    n = len(a)
    idx = np.argsort(conv)[-k:]
    p = np.zeros(2 * n - 1)
    p[idx] = 1.0 / k
    return gradient_from_p(a, p)


def weighted_topk_gradient(a, conv, k=5, temperature=10.0):
    """
    Gradient using top-k peaks with softmax weights (not uniform).
    Higher weights on peaks closer to the max.
    """
    n = len(a)
    top_idx = np.argsort(conv)[-k:]
    top_vals = conv[top_idx]
    max_val = top_vals.max()
    weights = np.exp(temperature * (top_vals - max_val))
    weights = weights / weights.sum()
    p = np.zeros(2 * n - 1)
    p[top_idx] = weights
    return gradient_from_p(a, p)


def mirror_descent_step(a, grad, eta):
    """Multiplicative weights update: a_i <- a_i * exp(-eta * grad_i), then normalize."""
    exponent = -eta * grad
    exponent = np.clip(exponent, -50, 50)
    a_new = a * np.exp(exponent)
    s = float(np.sum(a_new))
    if s < 1e-300:
        return a.copy()
    return a_new / s


def fast_lp_direction(a, k_constraints=30):
    """
    Fast LP: min M s.t. 2*conv(a,g)[k] - conv(a,a)[k] <= M for top-k lags,
    sum(g) = 1, g >= 0.
    Using only top-k constraints makes this much faster.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv_aa = compute_conv(a_norm)
    
    # Select top-k lags where conv is highest
    if k_constraints < 2 * n - 1:
        top_lags = np.argsort(conv_aa)[-k_constraints:]
    else:
        top_lags = np.arange(2 * n - 1)
    
    # Build matrix A such that (A @ g)[k] = conv(a,g)[k]
    A = np.zeros((len(top_lags), n))
    k_idx = top_lags[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A[valid] = a_norm[diff[valid]]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [2.0 * (A @ g) - M <= conv_aa[top_lags], cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def extended_lp_direction(a, k_constraints=50):
    """
    Extended LP: max sum(g) s.t. conv(a,g)[k] <= M for top-k lags, sum(g)=1, g>=0.
    Where M = max(conv(a,a)).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv_aa = compute_conv(a_norm)
    M_current = float(np.max(conv_aa))
    
    if k_constraints < 2 * n - 1:
        top_lags = np.argsort(conv_aa)[-k_constraints:]
    else:
        top_lags = np.arange(2 * n - 1)
    
    A = np.zeros((len(top_lags), n))
    k_idx = top_lags[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A[valid] = a_norm[diff[valid]]
    
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M_current, cp.sum(g) == 1]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def line_search(a, g, n_evals=30):
    """Line search for best t in [0,1] to minimize obj((1-t)*a + t*g)."""
    n = len(a)
    ts = np.linspace(0.01, 0.99, n_evals)
    objs = np.full(n_evals, np.inf)
    for i, t in enumerate(ts):
        a_try = np.maximum((1 - t) * a + t * g, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            objs[i] = compute_obj(a_try / s)
    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]
    
    if 0 < best_idx < n_evals - 1:
        t0, t1, t2 = ts[best_idx - 1], ts[best_idx], ts[best_idx + 1]
        f0, f1, f2 = objs[best_idx - 1], objs[best_idx], objs[best_idx + 1]
        denom = f0 - 2.0 * f1 + f2
        if abs(denom) > 1e-15:
            t_par = t1 - 0.5 * (t2 - t0) * (f0 - f1) / denom
            t_par = float(np.clip(t_par, 0.01, 0.99))
            a_par = np.maximum((1 - t_par) * a + t_par * g, 0)
            s_par = float(np.sum(a_par))
            if s_par > 1e-10:
                f_par = compute_obj(a_par / s_par)
                if f_par < best_obj:
                    best_t = t_par
                    best_obj = f_par
    return best_t, best_obj


def mirror_descent_optimize(a, time_budget=300.0, tau_init=20.0, tau_end=2000.0,
                            verbose=True, blend_k=4, blend_alpha=0.3, seed=777):
    """Mirror descent with adaptive temperature and blended gradients."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    stall = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.3:
            break

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        
        # Temperature: increase over time, with slight oscillation for exploration
        base_tau = tau_init + (tau_end - tau_init) * (progress ** 0.7)
        # Small oscillation to avoid getting stuck
        tau = base_tau * (1.0 + 0.1 * np.sin(0.3 * it))
        tau = max(tau, 5.0)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Adaptive k for top-k: more peaks early, fewer late
        adaptive_k = max(2, int(blend_k * (1.5 - progress)))
        
        # Choose gradient type based on iteration
        r = it % 10
        if r < 3:
            g = smoothed_gradient(a, conv, tau)
        elif r < 5:
            g = weighted_topk_gradient(a, conv, k=adaptive_k, temperature=tau * 0.5)
        elif r < 7:
            g_s = smoothed_gradient(a, conv, tau)
            g_t = topk_subgradient(a, conv, k=max(2, adaptive_k // 2))
            g = blend_alpha * g_t + (1.0 - blend_alpha) * g_s
        elif r < 9:
            g = topk_subgradient(a, conv, k=adaptive_k)
        else:
            # Use a "glow" gradient: smooth over a window around the peak
            peak_idx = int(np.argmax(conv))
            window = max(3, n // 20)
            p = np.zeros(2 * n - 1)
            lo = max(0, peak_idx - window)
            hi = min(2 * n - 2, peak_idx + window)
            p[lo:hi+1] = np.exp(-0.5 * ((np.arange(lo, hi+1) - peak_idx) / (window/2))**2)
            p = p / p.sum()
            g = gradient_from_p(a, p)

        g_max = float(np.max(np.abs(g)))
        if g_max < 1e-15:
            break
        
        # Adaptive step size
        eta = 1.5 / (g_max + 1e-10)
        # Scale: larger early, smaller late
        eta *= (1.2 - 0.5 * progress)
        # Additional scaling based on how "flat" the conv is
        conv_std = float(np.std(conv)) / max(max_conv, 1e-10)
        eta *= (1.0 + 0.5 * conv_std)

        a_new = mirror_descent_step(a, g, eta)

        # Occasional perturbation for exploration
        if it % 50 == 30 and progress < 0.85:
            temp = 0.008 * (1.0 - progress)
            a_new = a_new + temp * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            stall_count = 0
        else:
            stall += 1
            stall_count += 1
            # Try smaller step
            a_half = mirror_descent_step(a, g, eta * 0.3)
            obj_half = compute_obj(a_half)
            if obj_half < best_obj - 1e-14:
                best_obj = obj_half
                best_a = a_half.copy()
                a = a_half
                stall = 0
                stall_count = 0
            elif stall >= 30:
                # More aggressive perturbation
                strength = 0.03 * (1.0 + progress)
                a_pert = a * np.exp(strength * rng.standard_normal(n))
                a_pert = np.maximum(a_pert, 0)
                s_pert = float(np.sum(a_pert))
                if s_pert > 1e-10:
                    a = a_pert / s_pert
                else:
                    a = best_a.copy()
                stall = 0
                stall_count = 0

        a = a_new
        it += 1

        if verbose and (it % 50 == 0 or it == 1):
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, "
                  f"stall={stall}, time={time.monotonic()-start:.0f}s")

    return best_a


def lp_driven_optimize(a, time_budget=200.0, verbose=True, k_cons=30):
    """LP-driven optimization with reduced constraints."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        # Alternate between LP types
        if it % 3 == 0:
            g, M = fast_lp_direction(a, k_constraints=k_cons)
        elif it % 3 == 1:
            g = extended_lp_direction(a, k_constraints=k_cons)
            M = None
        else:
            # Gradient fallback
            conv = compute_conv(a)
            g = smoothed_gradient(a, conv, max(100.0, time_budget / 2))
            g = np.maximum(g, 0)
            s_g = float(np.sum(g))
            if s_g < 1e-10:
                break
            g = g / s_g
            M = None

        if g is None:
            conv = compute_conv(a)
            g = smoothed_gradient(a, conv, 200.0)
            g = np.maximum(g, 0)
            s_g = float(np.sum(g))
            if s_g < 1e-10:
                break
            g = g / s_g

        best_t, obj_t = line_search(a, g, n_evals=25)
        it += 1

        if obj_t < best_obj - 1e-13:
            a_try = np.maximum((1 - best_t) * a + best_t * g, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                best_obj = obj_t
                best_a = a_try.copy()
                a = a_try
                stall = 0
                if verbose and (it % 3 == 0):
                    print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, "
                          f"time={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            # Try smaller steps
            improved = False
            for t_small in [0.05, 0.02, 0.01]:
                a_sm = np.maximum((1 - t_small) * a + t_small * g, 0)
                s_sm = float(np.sum(a_sm))
                if s_sm > 1e-10:
                    obj_sm = compute_obj(a_sm / s_sm)
                    if obj_sm < best_obj - 1e-13:
                        best_obj = obj_sm
                        best_a = (a_sm / s_sm).copy()
                        a = best_a
                        stall = 0
                        improved = True
                        break
            if not improved and stall >= 5:
                # Perturb and restart
                conv_p = compute_conv(a)
                tau_p = 200.0
                max_p = float(np.max(conv_p))
                exp_p = np.exp(tau_p * (conv_p - max_p))
                p_p = exp_p / exp_p.sum()
                grad_p = gradient_from_p(a, p_p)
                lr_p = 0.02 / (float(np.linalg.norm(grad_p)) + 1e-10)
                a = np.maximum(a - lr_p * grad_p + 0.005 * rng.standard_normal(n), 0)
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new
                else:
                    a = best_a.copy()
                stall = 0
                if it > 100:
                    if verbose:
                        print(f"    LP: giving up after {it} iterations")
                    break

    return best_a


def coordinate_descent(a, time_budget=60.0, verbose=False):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate i, the objective as a function of a[i] is:
    2n * max_k (conv_k(a)) / (sum(a))^2
    
    where conv_k depends quadratically on a[i] for certain k values.
    We do a 1D minimization over a[i] >= 0.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)

    it = 0
    while time.monotonic() - start < time_budget:
        # Random order for coordinates
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            
            old_val = a[i]
            a[i] = 0.0
            base_sum = float(np.sum(a))
            base_conv_part = compute_conv(a)  # conv without the i-th coordinate's contribution
            
            # conv(a_new, a_new)[k] = base_conv_part[k] + 2*t*a[k-i] + t^2*a[?] 
            # Actually this is complex. Let's just do a simple 1D search.
            
            # Try several values
            candidates = [old_val]
            if old_val > 0:
                for frac in [0.3, 0.5, 0.7, 1.3, 1.5]:
                    candidates.append(old_val * frac)
            candidates.append(base_sum / n)  # average
            candidates = [max(0.0, c) for c in candidates]
            
            best_cand_val = old_val
            best_cand_obj = compute_obj(a) if old_val > 0 else np.inf
            
            for cv in candidates:
                a[i] = cv
                s_test = base_sum + cv
                if s_test < 1e-10:
                    continue
                a_test = a / s_test
                obj_test = compute_obj(a_test)
                if obj_test < best_cand_obj:
                    best_cand_obj = obj_test
                    best_cand_val = cv
            
            a[i] = best_cand_val
            s_final = float(np.sum(a))
            if s_final > 1e-10:
                a = a / s_final
                if compute_obj(a) < best_obj - 1e-14:
                    best_obj = compute_obj(a)
                    best_a = a.copy()
        
        a[i] = old_val  # restore (will be overwritten next iteration)
        it += 1
        if verbose and it % 3 == 0:
            print(f"    CD round {it}: obj={best_obj:.8f}")

    return best_a


def nesterov_gd_optimize(a, time_budget=100.0, tau_init=100.0, tau_end=3000.0,
                         verbose=False):
    """Nesterov accelerated gradient descent on smoothed max autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.85
    base_lr = 1.5 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.3

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        conv_y = compute_conv(y)
        g = smoothed_gradient(y, conv_y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm + 1e-10)
            adapt = min(max(adapt, 0.3 * base_lr), 2.0 * base_lr)
            lr = 0.8 * lr + 0.2 * adapt

        if prev_g is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-8), 4.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)

        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.3
            best_it = it

    if verbose:
        print(f"    NesterovGD: obj={best_obj:.8f}, iters={it}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_perturbation(a, rng, n_freqs=5, amp_scale=0.05):
    """Perturb in Fourier domain for diverse exploration."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, n_freqs + 1):
        freq = k * 2.0 * np.pi / n
        amp = amp_scale * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new

            print(f"  Phase1 MD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 MD: obj={obj_md:.8f} (no improvement)")

    # === Phase 2: Fast LP-Driven ===
    print(f"\n=== Phase 2: LP-Driven (reduced constraints) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.20, 250.0)
    if p2_budget > 20:
        a_lp = lp_driven_optimize(best_seq, time_budget=p2_budget,
                                  verbose=True, k_cons=25)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Nesterov GD Polish ===
    print(f"\n=== Phase 3: Nesterov GD ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.10, 100.0)
    if p3_budget > 10:
        a_gd = nesterov_gd_optimize(best_seq, time_budget=p3_budget,
                                    tau_init=100.0, tau_end=3000.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase3 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 GD: obj={obj_gd:.8f} (no improvement)")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Generate diverse n values to try
    n_options = []
    for delta in [-200, -100, -50, 50, 100, 200, 300, 500]:
        n_try = n_best + delta
        if 200 <= n_try <= 3000 and n_try not in n_options:
            n_options.append(n_try)
    # Also try some absolute values that might be optimal
    for n_abs in [400, 600, 800, 1000, 1200, 1500, 2000]:
        if n_abs not in n_options and abs(n_abs - n_best) > 100:
            n_options.append(n_abs)
    n_options = sorted(set(n_options))[:6]
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick MD
        alloc_md = min(remaining * 0.35, 70.0)
        if alloc_md > 15:
            a_try = mirror_descent_optimize(a_try, time_budget=alloc_md,
                                            tau_init=30.0, tau_end=1000.0,
                                            verbose=False, blend_k=3,
                                            blend_alpha=0.3, seed=100+n_try)

        # Quick LP
        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.25, 45.0)
        if alloc_lp > 10:
            a_try = lp_driven_optimize(a_try, time_budget=alloc_lp,
                                       verbose=False, k_cons=20)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse Restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    restart_strategies = [
        "fourier", "mirror_sym", "block_struct", "gaussian_smooth", "random_mix"
    ]
    for r, name in enumerate(restart_strategies):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if name == "fourier":
            a_r = fourier_perturbation(best_seq, rng, n_freqs=4, amp_scale=0.06)
        elif name == "mirror_sym":
            a_r = (best_seq + best_seq[::-1]) / 2.0
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r
            else:
                a_r = best_seq.copy()
        elif name == "block_struct":
            # Create block structure: alternating high/low segments
            n_blocks = max(3, n_cur // 100)
            block_size = n_cur // n_blocks
            a_r = best_seq.copy()
            for b in range(n_blocks):
                start = b * block_size
                end = min(start + block_size, n_cur)
                if b % 2 == 0:
                    a_r[start:end] = best_seq[start:end] * 1.2
                else:
                    a_r[start:end] = best_seq[start:end] * 0.8
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r
        elif name == "gaussian_smooth":
            # Gaussian smoothing
            kernel_size = max(5, n_cur // 50)
            kernel = np.exp(-0.5 * (np.linspace(-3, 3, kernel_size))**2)
            kernel = kernel / kernel.sum()
            from scipy.ndimage importConvolve1d as ndc
            try:
                from scipy.signal import convolve as sig_convolve
                a_smooth = sig_convolve(best_seq, kernel, mode='same')
            except Exception:
                a_smooth = np.convolve(best_seq, kernel, mode='same')
            a_r = 0.6 * best_seq + 0.4 * a_smooth
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r
        else:
            a_r = 0.7 * best_seq + 0.3 * (rng.uniform(0, 1, n_cur) / n_cur)
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r

        # Optimize
        alloc_md = min(remaining * 0.3, 50.0)
        if alloc_md > 10:
            a_r = mirror_descent_optimize(a_r, time_budget=alloc_md,
                                          tau_init=50.0, tau_end=800.0,
                                          verbose=False, blend_k=3,
                                          blend_alpha=0.3, seed=200+r)

        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.2, 35.0)
        if alloc_lp > 8:
            a_r = lp_driven_optimize(a_r, time_budget=alloc_lp,
                                     verbose=False, k_cons=20)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart ({name}): obj={best_score:.8f} **")
        else:
            print(f"  Restart ({name}): obj={obj_r:.8f}")

    # === Phase 6: Final Intensive Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau mirror descent
    if remaining > 40:
        alloc = min(remaining * 0.35, 120.0)
        a_final = mirror_descent_optimize(best_seq, time_budget=alloc,
                                          tau_init=500.0, tau_end=10000.0,
                                          verbose=True, blend_k=2,
                                          blend_alpha=0.5, seed=999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # LP polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 80.0)
        a_final2 = lp_driven_optimize(best_seq, time_budget=alloc,
                                      verbose=True, k_cons=40)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.35, 60.0)
        a_final3 = coordinate_descent(best_seq, time_budget=alloc, verbose=True)
        obj_final3 = compute_obj(a_final3)
        if obj_final3 < best_score:
            best_score = obj_final3
            best_seq = a_final3.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final quick GD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 40.0)
        a_final4 = nesterov_gd_optimize(best_seq, time_budget=alloc,
                                        tau_init=1000.0, tau_end=15000.0,
                                        verbose=True)
        obj_final4 = compute_obj(a_final4)
        if obj_final4 < best_score:
            best_score = obj_final4
            best_seq = a_final4.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
