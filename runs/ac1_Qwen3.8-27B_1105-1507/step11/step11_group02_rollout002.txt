import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a))) / (s * s)


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp_minpeak(a, A_cache=None, time_budget=20.0, g_init=None):

        A = A_cache

    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1

        if verbose and sweep % 3 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            # Identify affected convolution indices
            k_min = i
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            indices_in_a = ks - i  # These are the j such that a[j] * a[i] contributes

            c_cur = conv[ks].copy()
            a_vals = a[indices_in_a].copy()

            # The quadratic term is at k = 2*i
            k_quad = 2 * i
            has_quad = (0 <= k_quad <= 2 * n - 2) and (k_quad >= k_min) and (k_quad <= k_max)
            if has_quad:
                quad_idx = np.where(ks == k_quad)[0]
                if len(quad_idx) > 0:
                    quad_pos = quad_idx[0]
            else:
                quad_pos = -1

            def f_1d(x):
                """Evaluate objective when a[i] = x."""
                # Linear terms (excluding quadratic)
                if has_quad:
                    # Mask out the quadratic position
                    mask = np.ones(len(ks), dtype=bool)
                    mask[quad_pos] = False
                    lin_ks = ks[mask]
                    lin_a = a_vals[mask]
                    lin_c = c_cur[mask]
                    lin_vals = lin_c + 2.0 * lin_a * (x - v)
                    max_lin = float(np.max(lin_vals)) if len(lin_ks) > 0 else 0.0
                    # Quadratic term
                    quad_val = c_cur[quad_pos] - v * v + x * x
                    max_val = max(max_lin, quad_val)
                else:
                    lin_vals = c_cur + 2.0 * a_vals * (x - v)
                    max_val = float(np.max(lin_vals))

                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Get current value
            f_v = f_1d(v)

            # Try some candidate values
            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, v * 0.3, v * 0.7, v, v * 1.3, v * 1.7,
                          1.0 / n, 2.0 / n, 3.0 / n]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = f_v
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section around best_x
            x_lo = max(0.0, best_x - scale * 0.5)
            x_hi = best_x + scale * 1.5
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(40):
                if x_hi - x_lo < 1e-14 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-14:
                S = float(np.sum(a))
                continue

            # Update convolution incrementally
            delta = x_opt - v
            a[i] = x_opt

            # Update affected convolution entries
            if has_quad:
                conv[k_quad] += x_opt * x_opt - v * v
            # Linear updates
            for pos in range(len(ks)):
                if has_quad and pos == quad_pos:
                    continue
                conv[ks[pos]] += 2.0 * delta * a_vals[pos]

            S += delta

            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        # Renormalize periodically to avoid drift
        if sweep % 10 == 0:
            S = float(np.sum(a))
            if S > 1e-10:
                a = a / S
                conv = compute_conv(a)

    return best_a


def lp_iteration(a, time_budget=300.0, verbose=False, seed=42):
    """
    LP-based fixed-point iteration.
    Repeatedly solves the LP and blends the solution back.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    iteration = 0
    stall = 0

    # Build A matrix once (a changes, so we rebuild each iteration)
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break

        # Choose LP variant
        r = rng.random()
        if r < 0.5:
            g_lp, M_lp = solve_lp_minpeak(a, time_budget=min(remaining * 0.4, 15.0))
        elif r < 0.8:
            lam = 0.02 + 0.03 * stall
            g_lp, M_lp = solve_lp_proximal(a, lam=lam, time_budget=min(remaining * 0.4, 15.0))
        else:
            g_lp = solve_lp_weighted(a, time_budget=min(remaining * 0.4, 15.0))
            M_lp = None

        if g_lp is None:
            stall += 2
            if stall > 5:
                break
            continue

        # Line search for best blend
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
            iteration += 1
            if verbose and iteration % 5 == 0:
                print(f"    LP it {iteration}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 3:
                # Perturb to escape
                pert = rng.normal(0, 0.02, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / float(a.sum())
                stall = 0
            elif stall > 10:
                break

    return best_a


def spectral_perturb(a, rng, strength=0.3):
    """
    Perturb in frequency domain: modify the phase/amplitude of the FFT
    to create structured perturbations that preserve non-negativity.
    """
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    A = rfft(a, size)
    # Perturb amplitudes (preserve DC)
    for k in range(1, size // 2 + 1):
        scale = 1.0 + strength * rng.normal(0, 1)
        A[k] *= scale
    a_new = irfft(A, size)[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def block_cd(a, time_budget=60.0, block_size=10, seed=42):
    """
    Block coordinate descent: optimize blocks of coordinates jointly
    using a small-scale LP or direct search.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        # Try several random blocks
        n_blocks = max(1, int(remaining / 2))
        for _ in range(n_blocks):
            if time.monotonic() - start >= time_budget:
                break
            bs = rng.integers(max(2, n // 50), min(n // 3, n))
            i0 = rng.integers(0, n - bs)
            block = np.arange(i0, i0 + bs)

            # Optimize the block using small CD
            a_block = a[i0:i0+bs].copy()
            S_other = float(np.sum(a) - np.sum(a_block))

            # Simple: try scaling the block
            for factor in [0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0]:
                a_try = a.copy()
                a_try[i0:i0+bs] = a_block * factor
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Gaussian bump
        c = n // 2
        w = n // 6
        a = np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arc-shaped (like optimal for small n)
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x ** 2 + 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Two bumps
        c1, c2 = n // 4, 3 * n // 4
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * (n // 10) ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * (n // 10) ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Symmetric with center peak
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 7:
        # Random symmetric
        half = n // 2
        h = rng.uniform(0.5, 1.5, half)
        a = np.concatenate([h, h[::-1]])[:n]
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a, left=0, right=0)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines LP iteration, coordinate descent,
    spectral perturbations, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Coordinate Descent (quick initial polish) ===
    print(f"\n=== Phase 1: Initial CD ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.15, 150.0)
    if p1_budget > 20:
        a_cd = coordinate_descent(best_seq, time_budget=p1_budget,
                                  verbose=True, seed=seed)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 2: LP Iteration (main workhorse) ===
    print(f"\n=== Phase 2: LP Iteration ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 450.0)
    if p2_budget > 30:
        # Multiple rounds with different LP formulations
        for rnd in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(p2_budget / (4 - rnd), remaining * 0.6)
            if alloc < 20:
                break
            a_lp = lp_iteration(best_seq, time_budget=alloc, verbose=True,
                                seed=1000 + rnd * 100)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                n_best = len(best_seq)
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")

            # Quick CD between LP rounds
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_cd = coordinate_descent(best_seq, time_budget=min(remaining * 0.1, 40.0),
                                          verbose=False, seed=seed + 500 + rnd)
                obj_cd = compute_obj(a_cd)
                if obj_cd < best_score:
                    best_score = obj_cd
                    best_seq = a_cd.copy()
                    print(f"  CD after LP{rnd}: obj={best_score:.8f} **")

    # === Phase 3: Spectral Perturbation + CD ===
    print(f"\n=== Phase 3: Spectral Perturbation ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        for attempt in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            a_pert = spectral_perturb(best_seq, rng, strength=0.15)
            blend = 0.6
            a_pert = blend * a_pert + (1 - blend) * best_seq
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / float(a_pert.sum())

            alloc = min(remaining * 0.15, 60.0)
            if alloc > 10:
                a_pert = coordinate_descent(a_pert, time_budget=alloc,
                                            verbose=False, seed=seed + 3000 + attempt)
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  Spectral {attempt}: obj={best_score:.8f} **")
                else:
                    print(f"  Spectral {attempt}: obj={obj_pert:.8f}")

    # === Phase 4: Block CD ===
    print(f"\n=== Phase 4: Block CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.1, 80.0)
        a_bc = block_cd(best_seq, time_budget=alloc, block_size=10, seed=seed+7777)
        obj_bc = compute_obj(a_bc)
        if obj_bc < best_score:
            best_score = obj_bc
            best_seq = a_bc.copy()
            print(f"  Block CD: obj={best_score:.8f} **")
        else:
            print(f"  Block CD: obj={obj_bc:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 100),
        n_current + 50,
        n_current + 100,
        n_current + 200,
        1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_current) > 50]
    n_options = n_options[:5]

    print(f"  Trying n values: {n_options}, current n={n_current}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # CD
        alloc_cd = min(remaining * 0.12, 45.0)
        if alloc_cd > 10:
            a_try = coordinate_descent(a_try, time_budget=alloc_cd,
                                       verbose=False, seed=n_try)

        # Brief LP
        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.05, 30.0)
        if alloc_lp > 10:
            a_try = lp_iteration(a_try, time_budget=alloc_lp, verbose=False, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        mode = 1 + r
        a_r = diverse_init(n_final, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / float(a_r.sum())

        alloc = min(remaining * 0.12, 50.0)
        if alloc > 15:
            a_r = coordinate_descent(a_r, time_budget=alloc, verbose=False,
                                     seed=9000 + r * 111)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # CD
        alloc = min(remaining * 0.3, 100.0)
        a_fin = coordinate_descent(best_seq, time_budget=alloc,
                                   verbose=False, seed=seed + 99999)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_fin:.8f}")

    # One more LP if time
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 60.0)
        a_lp_fin = lp_iteration(best_seq, time_budget=alloc, verbose=False, seed=77777)
        obj_lp_fin = compute_obj(a_lp_fin)
        if obj_lp_fin < best_score:
            best_score = obj_lp_fin
            best_seq = a_lp_fin.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final CD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 40.0)
        a_cd_fin = coordinate_descent(best_seq, time_budget=alloc,
                                      verbose=False, seed=seed + 123456)
        obj_cd_fin = compute_obj(a_cd_fin)
        if obj_cd_fin < best_score:
            best_score = obj_cd_fin
            best_seq = a_cd_fin.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")

    # Validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
