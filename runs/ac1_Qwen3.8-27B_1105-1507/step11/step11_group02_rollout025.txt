from scipy.fft import rfft, irfft
import time

def conv_full(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    c = conv_full(a)
    return float(2.0 * n * np.max(c) / (s * s))

def subgradient(a):
    """Compute a subgradient of the objective at point a."""
    n = len(a)
    s = float(np.sum(a))
    c = conv_full(a)
    M = float(np.max(c))
    # Find active set (lags where c[k] is close to M)
    tol = 1e-10 * M
    active = np.where(c >= M - tol)[0]
    g = np.zeros(n)

        hi = min(n - 1, k)

Optimized step function coefficient search using:
1. Fast coordinate descent with incremental convolution updates
2. Mirror descent (multiplicative weights) with exact subgradients
3. Systematic n-length exploration
4. Peak-targeted perturbations
Key improvements: vectorized subgradient via FFT correlation, faster CD brackets,

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def conv_full(a):
    """Compute full linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    c = conv_full(a)
    return float(2.0 * n * np.max(c) / (s * s))

    # Full gradient of f = 2n*M/S^2:
    # df/da[i] = 2n/S^3 * (S*g_a[i] - 2*M)
    grad = (2.0 * n / (s ** 3)) * (s * g_a - 2.0 * M)
    return grad, M, active


def fast_cd(a, time_budget, seed=42, verbose=False):
    """
    Fast coordinate descent with incremental convolution updates.
    Each coordinate update only modifies O(n) entries of the convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    conv = conv_full(a).copy()
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        # Use a different random offset each sweep instead of full permutation
        offset = rng.integers(0, n)
        S = float(np.sum(a))

        for ii in range(n):
            if time.monotonic() - start >= time_budget:
                break
            i = (ii + offset) % n

            v = a[i]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)

            # Identify linear and quadratic parts
            k2i = 2 * i
            if 0 <= k2i <= 2 * n - 2:
                c2i = conv[k2i]
            else:
                c2i = 0.0

            # Linear entries: all k in [k_lo, k_hi] except k2i
            if k2i >= k_lo and k2i <= k_hi:
                ks_lin = np.concatenate([np.arange(k_lo, k2i), np.arange(k2i + 1, k_hi + 1)])
            else:
                ks_lin = np.arange(k_lo, k_hi + 1)

            if len(ks_lin) > 0:
                coeffs = 2.0 * a[ks_lin - i]
                lin_base = conv[ks_lin].copy()
            else:
                coeffs = np.array([])
                lin_base = np.array([])

            # Evaluate 1D function at several points
            def eval_1d(x):
                if len(ks_lin) > 0:
                    lin_max = float(np.max(lin_base + coeffs * (x - v)))
                else:
                    lin_max = 0.0
                quad_val = c2i + x * x - v * v
                peak = max(lin_max, quad_val)
                denom = (S + x - v) ** 2
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * peak / denom

            # Quick candidate evaluation
            scale = max(v, 1.0 / n, 1e-6)
            cands = [0.0, v * 0.25, v * 0.5, v * 0.75, v * 1.25, v * 1.5, v * 2.0, 1.0 / n]
            best_x = v
            best_f = eval_1d(v)
            for xc in cands:
                xc = max(0.0, float(xc))
                if abs(xc - v) < 1e-15:
                    continue
                fx = eval_1d(xc)
                if fx < best_f:
                    best_f = fx
                    best_x = xc

            # Golden section refinement
            half = max(best_x, scale * 0.5)
            x_lo = max(0.0, best_x - half)
            x_hi = best_x + half
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-8

            c1 = x_hi - gr * (x_hi - x_lo)
            c2 = x_lo + gr * (x_hi - x_lo)
            f1 = eval_1d(c1)
            f2 = eval_1d(c2)

            for _ in range(28):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if f1 < f2:
                    x_hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = x_hi - gr * (x_hi - x_lo)
                    f1 = eval_1d(c1)
                else:
                    x_lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = x_lo + gr * (x_hi - x_lo)
                    f2 = eval_1d(c2)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-14:
                continue

            # Update a and conv incrementally
            delta = x_opt - v
            a[i] = x_opt
            if 0 <= k2i <= 2 * n - 2:
                conv[k2i] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a[ks_lin - i]
            S += delta

            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        # Periodic full recomputation to prevent drift
        if sweep % 8 == 0:
            conv = conv_full(a).copy()
            s_new = float(np.sum(a))
            if s_new > 1e-10:
                a = a / s_new
                obj = 2.0 * n * float(np.max(conv))
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def mirror_descent(a, time_budget, eta_init=0.02, verbose=False, seed=42):
    """Mirror descent (multiplicative weights) with exact subgradients."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)

    # Momentum buffer
    grad_avg = np.zeros(n)
    momentum = 0.8

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        eta = eta_init * (1.0 - 0.6 * progress)

        grad, M, active = compute_subgradient(a)

        # Exponential moving average of gradient for stability
        grad_avg = momentum * grad_avg + (1 - momentum) * grad

        # Mirror descent: log-space update
        logits = np.log(np.maximum(a, 1e-30)) - eta * grad_avg
        max_logit = float(np.max(logits))
        logits -= max_logit
        weights = np.exp(logits)
        total_w = float(np.sum(weights))
        if total_w < 1e-30:
            break
        a_new = weights / total_w

        obj_new = compute_obj(a_new)
        a = a_new
        it += 1

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        # Occasionally try a small random perturbation to escape plateaus
        if it % 200 == 0 and progress > 0.3:
            pert = rng.normal(0, 0.01, n)
            a_pert = a * np.exp(pert)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / float(a_pert.sum())
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_obj:
                best_obj = obj_pert
                best_a = a_pert.copy()
                a = a_pert

        if verbose and it % 500 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, eta={eta:.6f}, |active|={len(active)}")

    return best_a


def peak_shifting(a, n_trials=50, seed=42):
    """
    Targeted perturbation: identify peak lags and push mass away from
    coordinates that contribute most to those peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(seed)

    for trial in range(n_trials):
        c = conv_full(a)
        M = float(np.max(c))
        active = np.where(c >= M - 1e-6 * M)[0]

        # For each active peak, compute which coordinates contribute most
        contribution = np.zeros(n)
        for k in active[:5]:  # limit to 5 peaks for efficiency
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            ki = k - idx
            mask = (ki >= 0) & (ki < n)
            contribution[idx[mask]] += a[idx[mask]] * a[ki[mask]]

        # Reduce high-contribution coordinates, boost low-contribution ones
        top_frac = max(1, n // 20)
        top_idx = np.argsort(contribution)[-top_frac:]
        bot_idx = np.argsort(contribution)[:top_frac]

        factor = np.exp(rng.uniform(0.05, 0.15))
        a_try = a.copy()
        a_try[top_idx] /= factor
        a_try[bot_idx] *= factor
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        # Binomial-like
        m = n - 1
        x = np.arange(m + 1)
        a = np.exp(-((x - m / 2) ** 2) / (2 * (m / 4) ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Raised cosine
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        a = a / a.sum()
    elif mode == 4:
        # Triangular
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)
        a = a / a.sum()
    elif mode == 5:
        # Two-hump
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.3) ** 2) + np.exp(-20 * (x - 0.7) ** 2)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines fast CD, mirror descent, peak shifting,
    and n-length exploration to minimize the step function objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={best_seq.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = None

    # Initialize if no parent
    if best_seq is None:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Fast Coordinate Descent ===
    print(f"\n=== Phase 1: Fast CD (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.32, 420.0)
    if p1_budget > 30:
        a_cd = fast_cd(best_seq, time_budget=p1_budget, verbose=True, seed=seed)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD improved: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 2: Mirror Descent ===
    print(f"\n=== Phase 2: Mirror Descent (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.18, 220.0)
    if p2_budget > 20:
        a_md = mirror_descent(best_seq, time_budget=p2_budget, eta_init=0.015,
                              verbose=True, seed=seed + 100)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  MD improved: obj={best_score:.8f} **")
        else:
            print(f"  MD: obj={obj_md:.8f} (no improvement)")

    # === Phase 3: Peak Shifting + CD ===
    print(f"\n=== Phase 3: Peak Shifting (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_ps = peak_shifting(best_seq, n_trials=30, seed=seed + 200)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  PS improved: obj={best_score:.8f} **")
        else:
            print(f"  PS: obj={obj_ps:.8f}")

        # Quick CD after peak shifting
        remaining = deadline - time.monotonic()
        if remaining > 40:
            a_cd2 = fast_cd(best_seq, time_budget=min(remaining * 0.4, 100.0),
                            seed=seed + 300)
            obj_cd2 = compute_obj(a_cd2)
            if obj_cd2 < best_score:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                print(f"  CD after PS: obj={best_score:.8f} **")
            else:
                print(f"  CD after PS: obj={obj_cd2:.8f}")

    # === Phase 4: Another CD round ===
    print(f"\n=== Phase 4: CD Round 2 (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.2, 200.0)
    if p4_budget > 30:
        a_cd3 = fast_cd(best_seq, time_budget=p4_budget, verbose=True, seed=seed + 400)
        obj_cd3 = compute_obj(a_cd3)
        if obj_cd3 < best_score:
            best_score = obj_cd3
            best_seq = a_cd3.copy()
            print(f"  CD2 improved: obj={best_score:.8f} **")
        else:
            print(f"  CD2: obj={obj_cd3:.8f}")

    # === Phase 5: n-length Exploration ===
    print(f"\n=== Phase 5: n-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_candidates = sorted(set([
        n_current - 100, n_current - 50,
        n_current + 50, n_current + 100, n_current + 150,
        n_current + 200, n_current + 300,
    ]))
    # Keep only reasonable values
    n_candidates = [n for n in n_candidates if 400 <= n <= 2000 and abs(n - n_current) >= 30]
    n_candidates = n_candidates[:5]
    print(f"  Testing n values: {n_candidates}")

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        # Initialize from current best via resampling
        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try > 1e-10:
            a_try = a_try / s_try

        # Brief CD
        alloc_cd = min(remaining * 0.35, 80.0)
        if alloc_cd > 20:
            a_try = fast_cd(a_try, time_budget=alloc_cd, seed=n_try)

        # Brief mirror descent
        remaining = deadline - time.monotonic()
        alloc_md = min(remaining * 0.15, 40.0)
        if alloc_md > 15:
            a_try = mirror_descent(a_try, time_budget=alloc_md, eta_init=0.01, seed=n_try + 500)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = 2 + r
        a_r = diverse_init(n_cur, rng, mode=mode)
        # Blend with current best
        blend = 0.3 + 0.2 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / float(a_r.sum())

        alloc = min(remaining * 0.25, 80.0)
        if alloc > 20:
            a_r = fast_cd(a_r, time_budget=alloc, seed=7000 + r * 100)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 150.0)
        a_final = fast_cd(best_seq, time_budget=alloc, verbose=True, seed=seed + 9999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_md_final = mirror_descent(best_seq, time_budget=min(remaining * 0.6, 60.0),
                                    eta_init=0.005, seed=seed + 8888)
        obj_md_final = compute_obj(a_md_final)
        if obj_md_final < best_score:
            best_score = obj_md_final
            best_seq = a_md_final.copy()
            print(f"  Final MD: obj={best_score:.8f} **")
        else:
            print(f"  Final MD: obj={obj_md_final:.8f}")

    # One more CD if time
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_cd_final = fast_cd(best_seq, time_budget=min(remaining * 0.7, 50.0),
                             seed=seed + 25000)
        obj_cd_final = compute_obj(a_cd_final)
        if obj_cd_final < best_score:
            best_score = obj_cd_final
            best_seq = a_cd_final.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = 2.0
        val = evaluate_sequence(result)
        if val < np.inf:
            best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
