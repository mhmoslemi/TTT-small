            c_quad = conv[2 * i] - v * v  # conv[2i] + x^2 - v^2 = x^2 + (conv[2i] - v^2)

                quad_val = c_quad + x * x

                    max_val = max(quad_val, float(np.max(lin_vals)))
                else:
                    max_val = quad_val

            # Initial bracket: try a few points

            best_x = v
            best_f = f_eval(v)
            for x in candidates:
                fx = f_eval(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x
            
            # Golden section around best_x
            scale = max(best_x, 1.0/n, 1e-6)
            x_lo = max(0.0, best_x - 0.8 * scale)
            x_hi = best_x + 1.2 * scale
            
            if x_hi - x_lo < 1e-12:
                x_opt = best_x
            else:
                c_pt = x_hi - gr * (x_hi - x_lo)
                d_pt = x_lo + gr * (x_hi - x_lo)
                fc = f_eval(c_pt)
                fd = f_eval(d_pt)

                        d_pt = c_pt

                    else:
                        x_lo = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = x_lo + gr * (x_hi - x_lo)
                        fd = f_eval(d_pt)
                x_opt = max(0.0, (x_lo + x_hi) / 2.0)

                delta = x_opt - v
                a[i] = x_opt
                # Update conv

            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

def multiplicative_grad_descent(a, time_budget, tau=100.0, lr_scale=5.0, seed=42):
    """Multiplicative gradient descent with smoothed max."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    lr = lr_scale / np.sqrt(n)
    it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau_eff = tau * (1 + 2 * progress)  # Increase sharpness over time
        lr_eff = lr * (1.0 - 0.6 * progress)
        
        grad, conv, p = smoothed_grad(a, tau_eff)
        
        # Multiplicative update
        a_new = a * np.exp(-lr_eff * grad)
        a_new = np.maximum(a_new, 1e-12)
        a_new = a_new / float(np.sum(a_new))
        
        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        it += 1
        
        if it % 5000 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}")
    
    return best_a


def lp_direction(a, time_budget=20.0):
    """Solve LP: min M s.t. conv(a,g)<=M, sum(g)=1, g>=0. Returns g."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    # Build convolution matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    start = time.monotonic()
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.PDLP]:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget, seed=42, verbose=False):
    """LP-based polishing with blending."""
    n = len(a)
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break
        
        g_lp, M_lp = lp_direction(a, time_budget=min(remaining * 0.3, 15.0))
        if g_lp is None:
            break
        
        best_t = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = np.maximum((1-t)*a + t*g_lp, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t:
                best_t = obj_try
                best_a_local = a_try.copy()
        
        if best_t < best_obj - 1e-12:
            best_obj = best_t
            best_a = best_a_local
            a = best_a
        else:
            break
    
    return best_a, best_obj


def targeted_balance(a, n_steps=100):
    """
    Targeted convolution balancing: identify peak positions and reduce
    coordinates that contribute most to them.
    """
    n = len(a)
    a = a / float(np.sum(a))
    conv = compute_conv(a)
    c_max = float(np.max(conv))
    
    for step in range(n_steps):
        # Find top positions
        c_max = float(np.max(conv))
        threshold = c_max * 0.95
        peak_positions = np.where(conv >= threshold)[0]
        
        if len(peak_positions) == 0:
            break
        
        # For each peak position k, find which a[i] contribute most
        # a[i] contributes to conv[k] if 0<=k-i<n: contribution is a[i]*a[k-i]
        # We want to reduce a[i] proportional to its total contribution to peaks
        
        reduction = np.zeros(n)
        for k in peak_positions:
            excess = conv[k] - c_max * 0.9  # how much to reduce
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            for i in range(i_start, i_end + 1):
                partner = k - i
                if 0 <= partner < n:
                    reduction[i] += a[partner] * excess
        
        # Apply multiplicative reduction
        if np.max(reduction) > 1e-15:
            factor = 0.1  # reduction strength
            a = a * np.exp(-factor * reduction / (np.max(reduction) + 1e-30))
            a = np.maximum(a, 1e-12)
            a = a / float(np.sum(a))
            conv = compute_conv(a)
    
    return a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        c1, c2 = n//4, 3*n//4
        w = n//10
        a = np.exp(-((np.arange(n)-c1)**2)/(2*w*w)) + np.exp(-((np.arange(n)-c2)**2)/(2*w*w))
        return a / a.sum()
    elif mode == 3:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2*np.pi*np.arange(n)/n))
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    elif mode == 4:
        # Power law
        x = np.arange(1, n+1)
        a = 1.0 / (x ** 0.3)
        return a / a.sum()
    elif mode == 5:
        # Alternating high/low
        a = np.ones(n)
        a[::2] = 1.5
        a[1::2] = 0.5
        return a / a.sum()
    else:
        # Smooth random
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(rng.uniform(0, 1, n), sigma=n//20)
        a = np.maximum(a, 1e-6)
        return a / a.sum()


def resample(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/sum(a)^2.
Multi-phase approach: multiplicative gradient descent, exact coordinate descent
with precomputed convolution updates, LP direction finding, targeted convolution
balancing, and multi-n exploration with diverse restarts.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def obj_of(a):
    """Objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def smoothed_grad(a, tau):
    """
    Smoothed gradient of max(conv(a,a)) using log-sum-exp.
    Returns (gradient, convolution, probability weights).
    """
    n = len(a)
    conv = compute_conv(a)
    m = float(np.max(conv))
    shifted = np.clip(tau * (conv - m), -50.0, 50.0)
    exp_vals = np.exp(shifted)
    p = exp_vals / float(np.sum(exp_vals))

    # grad[i] = 2 * sum_{j=0}^{n-1} a[j] * p[i+j]
    # Compute via FFT cross-correlation
    N = 1
    while N < 3 * n:
        N <<= 1
    a_rev = a[::-1]
    A_rev = fft(a_rev, N)
    P = fft(conv.astype(np.float64), N)
    cross = ifft(A_rev * P).real
    R = cross[n - 1:n - 1 + n]

    return 2.0 * R, conv, p


def cd_eval(x, c_quad, c_lin, m_lin):
    """Evaluate 1D objective for coordinate descent line search."""
    quad_val = c_quad + x * x
    if len(c_lin) > 0:
        lin_vals = c_lin + m_lin * x
        max_val = max(quad_val, float(np.max(lin_vals)))
    else:
        max_val = quad_val
    return max_val


def fast_cd(a, time_budget, seed=42, verbose=False):
    """
    Exact coordinate descent with precomputed convolution updates.
    For each coordinate i, solves a 1D problem: minimize max of
    (quadratic at 2i) and (lines from linear contributions).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a).copy()
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]

            if v < 1e-14:
                a[i] = 1e-9
                S += 1e-9
                continue

            k_start = i
            k_end = i + n - 1
            ks = np.arange(k_start, k_end + 1)

            mask = ks != 2 * i
            if mask.any():
                c_lin = conv[ks[mask]] - 2.0 * v * a[ks[mask] - i]
                m_lin = 2.0 * a[ks[mask] - i]
            else:
                c_lin = np.array([], dtype=np.float64)
                m_lin = np.array([], dtype=np.float64)

            c_quad = conv[2 * i] - v * v

            # Find best x via candidate evaluation + golden section
            candidates = [0.0, v * 0.4, v * 0.7, v, v * 1.3, v * 1.6, 3.0 / n]
            best_x = v
            best_f = 2.0 * n * cd_eval(v, c_quad, c_lin, m_lin)
            for x in candidates:
                x = max(0.0, float(x))
                fx = 2.0 * n * cd_eval(x, c_quad, c_lin, m_lin)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section refinement
            scale = max(best_x, 1.0 / n, 1e-7)
            x_lo = max(0.0, best_x - 0.8 * scale)
            x_hi = best_x + 1.5 * scale

            if x_hi - x_lo > 1e-12:
                c_pt = x_hi - gr * (x_hi - x_lo)
                d_pt = x_lo + gr * (x_hi - x_lo)
                fc = cd_eval(c_pt, c_quad, c_lin, m_lin)
                fd = cd_eval(d_pt, c_quad, c_lin, m_lin)

                for _ in range(40):
                    if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                        break
                    if fc < fd:
                        x_hi = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = x_hi - gr * (x_hi - x_lo)
                        fc = cd_eval(c_pt, c_quad, c_lin, m_lin)
                    else:
                        x_lo = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = x_lo + gr * (x_hi - x_lo)
                        fd = cd_eval(d_pt, c_quad, c_lin, m_lin)

                x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            else:
                x_opt = best_x

            if abs(x_opt - v) > 1e-14:
                delta = x_opt - v
                a[i] = x_opt
                if 0 <= 2 * i <= 2 * n - 2:
                    conv[2 * i] += x_opt * x_opt - v * v
                if mask.any():
                    conv[ks[mask]] += 2.0 * delta * a[ks[mask] - i]
                S += delta

                obj_new = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a.copy()

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def multiplicative_grad_descent(a, time_budget, tau_init=80.0, lr_scale=4.0):
    """Multiplicative (exponential) gradient descent with smoothed max."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    lr = lr_scale / np.sqrt(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau_eff = tau_init * (1.0 + 3.0 * progress)
        lr_eff = lr * (1.0 - 0.5 * progress)

        grad, conv, p = smoothed_grad(a, tau_eff)

        a_new = a * np.exp(-lr_eff * grad)
        a_new = np.maximum(a_new, 1e-14)
        a_new = a_new / float(np.sum(a_new))

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if it % 3000 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau_eff:.0f}")

    return best_a


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def lp_direction(a, time_budget=20.0):
    """Solve LP: min M s.t. conv(a,g)<=M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    start = time.monotonic()
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.PDLP]:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0.0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget, seed=42):
    """LP-based polishing with multiple blending factors."""
    n = len(a)
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        g_lp, M_lp = lp_direction(a, time_budget=min(remaining * 0.35, 18.0))
        if g_lp is None:
            break

        best_t = best_obj
        best_a_local = best_a.copy()
        for t in [0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = np.maximum((1.0 - t) * a + t * g_lp, 0.0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t:
                best_t = obj_try
                best_a_local = a_try.copy()

        if best_t < best_obj - 1e-13:
            best_obj = best_t
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break

def targeted_balance(a, n_steps=150, strength=0.15):
    """
    Targeted convolution balancing: iteratively reduce coordinates
    that contribute most to peak convolution positions.
    """
    n = len(a)
    a = a / float(np.sum(a))
    conv = compute_conv(a)

    for step in range(n_steps):
        c_max = float(np.max(conv))
        threshold = c_max * 0.92
        peak_positions = np.where(conv >= threshold)[0]

        if len(peak_positions) == 0:
            break

        reduction = np.zeros(n)
        for k in peak_positions:
            excess = max(0.0, conv[k] - c_max * 0.85)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                partner = k - i
                if 0 <= partner < n:
                    reduction[i] += a[partner] * excess

        if float(np.max(reduction)) > 1e-15:
            factor = strength / (float(np.max(reduction)) + 1e-30)
            a = a * np.exp(-factor * reduction)
            a = np.maximum(a, 1e-14)
            a = a / float(np.sum(a))
            conv = compute_conv(a)

    return a


def diverse_init(n, rng, mode=0):
    """Generate diverse initial sequences."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 12)
        x = np.arange(n)
        a = np.exp(-((x - c1) ** 2) / (2 * w * w)) + np.exp(-((x - c2) ** 2) / (2 * w * w))
        return a / a.sum()
    elif mode == 3:
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / n))
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    elif mode == 4:
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / (x ** 0.3)
        return a / a.sum()
    elif mode == 5:
        a = np.ones(n)
        a[::2] = 1.4
        a[1::2] = 0.6
        return a / a.sum()
    elif mode == 6:
        # Smooth random via multiple scales
        a = rng.uniform(0, 1, n)
        # Simple smoothing
        kernel = np.ones(7) / 7.0
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    else:
        return np.ones(n) / n


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)

            print(f"  CD2: obj={best_score:.8f} **")
        else:
            print(f"  CD2: obj={obj_cd2:.8f}")

    # === Phase 5: Targeted Balancing + CD ===
    print("\n=== Phase 5: Targeted Balancing ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        a_bal = targeted_balance(best_seq, n_steps=100, strength=0.12)
        obj_bal = obj_of(a_bal)
        if obj_bal < best_score:
            best_score = obj_bal
            best_seq = a_bal.copy()
            print(f"  Balance: obj={best_score:.8f} **")
            # Follow with CD
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_bcd = fast_cd(best_seq, min(remaining * 0.3, 50.0), seed=seed + 400)
                obj_bcd = obj_of(a_bcd)
                if obj_bcd < best_score:
                    best_score = obj_bcd
                    best_seq = a_bcd.copy()
                    print(f"  Balance+CD: obj={best_score:.8f} **")
        else:
            print(f"  Balance: obj={obj_bal:.8f} (no improvement)")

    # === Phase 6: Multi-n Exploration ===
    print("\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    n_options = [650, 700, 750, 850, 900, 950, 1000, 1050, 1100, 1150, 1200]
    n_options = [n for n in n_options if abs(n - n_best) > 20]
    n_options = n_options[:8]
    print(f"  Current n={n_best}, trying: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        a_try = resample_seq(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        alloc_cd = min(remaining * 0.18, 35.0)
        if alloc_cd > 10:
            a_try = fast_cd(a_try, alloc_cd, seed=n_try, verbose=False)

        obj_try = obj_of(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse Restarts ===
    print("\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        mode = r + 1
        a_r = diverse_init(n_cur, rng, mode=mode)
        blend = 0.35 + 0.1 * r
        a_r = (1.0 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0.0)
        s_r = float(np.sum(a_r))
        if s_r < 1e-10:
            continue
        a_r = a_r / s_r

        alloc = min(remaining * 0.18, 45.0)
        if alloc > 12:
            a_r = fast_cd(a_r, alloc * 0.6, seed=7000 + r * 133, verbose=False)
            remaining = deadline - time.monotonic()
            if remaining > 12:
                a_r = multiplicative_grad_descent(a_r, min(remaining * 0.4, 20.0),
                                                   tau_init=50.0, lr_scale=3.5)
            obj_r = obj_of(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Final Polish ===
    print("\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.35, 90.0)
        a_fin = fast_cd(best_seq, alloc, seed=seed + 9999, verbose=True)
        obj_fin = obj_of(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_fin:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_lp2, obj_lp2 = lp_polish(best_seq, min(remaining * 0.45, 35.0), seed=seed + 8888)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 12:
        a_cd3 = fast_cd(best_seq, min(remaining * 0.6, 25.0), seed=seed + 7777)
        obj_cd3 = obj_of(a_cd3)
        if obj_cd3 < best_score:
            best_score = obj_cd3
            best_seq = a_cd3.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = obj_of(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
