```python
def smooth_max_gradient(conv, tau, K=50):
    """Compute gradient of smooth max using top-K lags."""
    n = len(conv)
    # Get top K indices
    top_k_idx = np.argpartition(conv, -K)[-K:]
    top_k_vals = conv[top_k_idx]
    # Softmax weights over top K
    shifted = tau * (top_k_vals - top_k_vals.max())
    weights = np.exp(shifted)
    weights /= weights.sum()
    return top_k_idx, weights
```

```python
"""
Optimized step function coefficient search using:
1. Smooth-max gradient descent with softmax parameterization (fast, many iterations)
2. Direct projected subgradient with momentum (handles sharp features)
3. LP-based polish (alternating minimization via cvxpy)
4. Multi-n exploration and diverse restarts
Target: beat 1.5027 (previous best: 1.506312)

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def normalize_a(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix_rows(k_list, n, a):
    """Build rows of the convolution matrix for specific lags."""
    rows = []
    for k in k_list:
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            row[i] += a[k - i]
        rows.append(row)
    return np.array(rows)


def smooth_max_optimize(a_init, c_init, time_budget, tau_start=5.0, tau_end=500.0,
                        n_topk=50, lr_init=0.01, verbose=True):
    """
    Optimize using smooth-max objective with softmax parameterization.
    a_i = exp(c_i) / sum(exp(c_j))
    Objective: 2n * smooth_max_k (a*a)[k]
    """
    n = len(a_init)
    start = time.monotonic()
    c = c_init.copy().astype(np.float64)
    # Make c relative (subtract max for numerical stability)
    c -= c.max()
    
    best_a = normalize_a(a_init)
    best_obj = compute_obj(best_a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    patience = 2000
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        # Anneal tau
        progress = min(1.0, (time.monotonic() - start) / (time_budget * 0.7))
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Compute a from c
        c_shifted = c - c.max()
        exp_c = np.exp(c_shifted)
        a = exp_c / exp_c.sum()
        
        # Compute convolution
        conv = compute_conv(a)
        
        # Get top-K lags
        if n_topk < len(conv):
            top_idx = np.argpartition(conv, -n_topk)[-n_topk:]
        else:
            top_idx = np.arange(len(conv))
        top_vals = conv[top_idx]
        
        # Softmax weights over top-K
        shifted_tau = tau * (top_vals - top_vals.max())
        weights = np.exp(shifted_tau)
        weights /= weights.sum()
        
        # Compute gradient of smooth max w.r.t. a
        # d(smooth_max)/da[j] = sum_k w_k * (a[k-j] + a[j-k] if valid)
        grad_a = np.zeros(n)
        for idx_i, k in enumerate(top_idx):
            w = weights[idx_i]
            if w < 1e-10:
                continue
            # For lag k: (a*a)[k] = sum_i a[i]*a[k-i]
            # d/da[j] = a[k-j] (if valid) + a[j-k] (if valid)
            for j in range(n):
                j1 = k - j
                j2 = j - k
                if 0 <= j1 < n:
                    grad_a[j] += w * a[j1]
                if 0 <= j2 < n:
                    grad_a[j] += w * a[j2]
        
        # Transform gradient from a-space to c-space
        # da_i/dc_j = a_i*(delta_ij - a_j)
        # dc_j = a_j * (grad_a[j] - sum_i a_i * grad_a[i])
        mean_grad = float(np.dot(a, grad_a))
        grad_c = a * (grad_a - mean_grad)
        
        # Adam update
        lr = lr_init / (1.0 + 0.001 * it)
        m = beta1 * m + (1 - beta1) * grad_c
        v = beta2 * v + (1 - beta2) * grad_c ** 2
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        c = c - lr * m_hat / (np.sqrt(v_hat) + eps)
        c -= c.max()  # Renormalize for stability
        
        # Evaluate
        a_new = np.exp(c - c.max())
        a_new = a_new / a_new.sum()
        obj = 2.0 * n * float(np.max(compute_conv(a_new)))
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            best_it = it
        else:
            if it - best_it > patience:
                # Reset to best and reduce learning rate
                c = np.log(best_a + 1e-15)
                c -= c.max()
                m = np.zeros(n)
                v = np.zeros(n)
                lr_init *= 0.5
                patience = int(patience * 1.3)
                best_it = it
        
        it += 1
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    SM it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.5f}, t={elapsed:.1f}s")
    
    return best_a, best_obj, c


def smooth_max_optimize_fast(a_init, time_budget, tau_start=10.0, tau_end=200.0,

    """Faster version with vectorized gradient computation."""
    n = len(a_init)
    c = np.log(a_init + 1e-15).copy()
    c -= c.max()
    
    start = time.monotonic()
    best_a = normalize_a(a_init)
    best_obj = compute_obj(best_a)

            top_idx = np.arange(len(conv))
        top_vals = conv[top_idx]
        
        # Softmax weights
        max_tv = top_vals.max()
        shifted = tau * (top_vals - max_tv)
        weights = np.exp(shifted)

        v_hat = v / (1 - beta2 ** (it + 1))

        it += 1

def direct_subgradient(a_init, time_budget, lr0=0.03, momentum_beta=0.7, verbose=True):
    """Direct projected subgradient on a (non-softmax)."""
    n = len(a_init)
    a = normalize_a(a_init).copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    momentum = np.zeros(n)
    it = 0
    patience = 5000
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))
        
        # Subgradient of f_k at a
        grad = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1) + 1
        for j in range(lo, hi):
            grad[j] += a[k_max - j]
        lo2 = max(k_max, 0)
        hi2 = min(n, k_max + n)
        for j in range(lo2, hi2):
            grad[j] += a[j - k_max]
        
        lr_t = lr0 / (1.0 + 0.002 * it)
        momentum = momentum_beta * momentum - lr_t * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = best_a.copy()
            momentum[:] = 0
        else:
            a_new = a_new / s
        a = a_new
        
        obj = 2.0 * n * float(np.max(compute_conv(a)))
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            best_it = it
        else:
            if it - best_it > patience:
                a = best_a.copy()
                momentum[:] = 0
                lr0 *= 0.5
                patience = int(patience * 1.3)
                best_it = it
        
        it += 1
        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - start
            print(f"    SG it{it}: obj={best_obj:.8f}, lr={lr_t:.5f}, t={elapsed:.1f}s")
    
    return best_a, best_obj


def build_conv_matrix_full(n, a):
    """Build full convolution matrix A where (A@g)[k] = (a*g)[k]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # (a*g)[k] includes a[k-j]*g[j] for valid k-j
        # So A[k, j] = a[k-j] if 0 <= k-j < n, else 0
        lo = max(0, j)  # k - j >= 0 => k >= j
        hi = min(2 * n - 1, j + n)  # k - j < n => k < j + n
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_lp(a, max_time=30.0):
    """Solve LP: minimize M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix_full(n, a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS]:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > max_time:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def line_search(a, g, t_values=None):
    """Exhaustive line search along a -> (1-t)*a + t*g."""
    n = len(a)
    if t_values is None:
        t_values = np.concatenate([
            np.linspace(0, 0.5, 20),
            np.linspace(0.5, 1.0, 20),
            np.linspace(1.0, 1.2, 10)
        ])
    best_t = 0.0
    best_obj = compute_obj(a)
    for t in t_values:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    return best_t, best_obj


def lp_polish(a_init, time_budget, max_iters=50, verbose=False):
    """LP-based alternating minimization."""
    n = len(a_init)
    a = normalize_a(a_init).copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(12345 + it)
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        g_lp = solve_lp(a, max_time=min(remaining * 0.5, 40.0))
        if g_lp is None:
            break
        
        t_opt, obj_opt = line_search(a, g_lp)
        
        if t_opt > 1e-6 and obj_opt < best_obj - 1e-12:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            obj_new = compute_obj(a_new)
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Try direct
            obj_direct = compute_obj(g_lp)
            if obj_direct < best_obj:
                best_obj = obj_direct
                best_a = g_lp.copy()
                a = g_lp.copy()
                stall = 0
            else:
                stall += 1
        
        if stall >= 3:
            # Perturb
            if stall <= 5:
                freq = rng.integers(1, max(2, n // 100))
                phase = rng.uniform(0, 2 * np.pi)
                idx = np.arange(n, dtype=np.float64)
                pert = 0.02 * stall * np.cos(2 * np.pi * freq * idx / n + phase)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
            else:
                # More aggressive restart
                x = np.linspace(0, 1, n)
                mode = it % 4
                if mode == 0:
                    a_div = 1.0 / (0.05 + 3.0 * (2 * x - 1) ** 2)
                elif mode == 1:
                    a_div = np.exp(-((idx - n * 0.2) ** 2) / (2 * (n * 0.1) ** 2)) + \

                a = 0.5 * a_div + 0.5 * best_a
                a = a / a.sum()
            stall = 0
        
        it += 1
        if verbose and it % 3 == 0:
            elapsed = time.monotonic() - start
            print(f"      LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, t_elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def diverse_u_init(n, sharpness=3.0, asym=0.0, rng=None):
    """U-shaped initialization with optional asymmetry."""
    x = np.linspace(0, 1, n)
    center = 0.5 + asym
    a = 1.0 / (0.05 + sharpness * (x - center) ** 2)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def diverse_init(n, mode, rng=None):
    """Various initialization shapes."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n, dtype=np.float64)
    if mode == 0:
        a = 1.0 / (0.05 + 3.0 * (2 * x - 1) ** 2)
    elif mode == 1:
        a = 1.0 / (0.03 + 5.0 * (2 * x - 1) ** 4)
    elif mode == 2:
        a = np.exp(-((idx - n * 0.15) ** 2) / (2 * (n * 0.08) ** 2)) + \
            np.exp(-((idx - n * 0.85) ** 2) / (2 * (n * 0.08) ** 2)) + 0.005
    elif mode == 3:
        a = (2 * x - 1) ** 3 + 0.3
        a = np.maximum(a, 0.01)
    elif mode == 4:
        a = 1.0 / (0.1 + 2.0 * np.abs(2 * x - 1))
    elif mode == 5:
        a = np.exp(3.0 * (2 * x - 1) ** 2)
    elif mode == 6:
        a = 1.0 / (0.02 + 8.0 * (2 * x - 1) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_to_n(a_src, n_dst):
    """Interpolate a sequence to a different length."""
    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s

        a_sm2, obj_sm2 = smooth_max_optimize_fast(best_seq, time_budget=alloc,
                                                   tau_start=50.0, tau_end=500.0,
                                                   n_topk=30, lr_init=0.01, verbose=True)
        if obj_sm2 < best_score:
            best_score = obj_sm2
            best_seq = a_sm2.copy()
            print(f"  Phase 4 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4 done: obj={obj_sm2:.8f} (no improvement)")
    
    # === Phase 5: n Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 150:
        print(f"\n=== Phase 5: n Exploration (budget={remaining:.0f}s) ===")
        n_cur = len(best_seq)
        n_options = []
        for dn in [-200, -100, -50, 50, 100, 200, 300, 500]:
            n_try = n_cur + dn
            if 300 <= n_try <= 3000 and n_try != n_cur:
                n_options.append(n_try)
        # Add some fixed options
        for n_fixed in [500, 600, 700, 900, 1000, 1200, 1500, 1800, 2000]:
            if 300 <= n_fixed <= 3000 and abs(n_fixed - n_cur) >= 50:
                n_options.append(n_fixed)
        n_options = sorted(set(n_options), key=lambda x: abs(x - n_cur))
        n_options = n_options[:5]
        print(f"  Trying n: {n_options}")
        
        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 80:
                break
            a_try = interpolate_to_n(best_seq, n_try)
            if a_try is None:
                continue
            
            # Quick optimization: smooth-max + short LP
            alloc_sm = min(remaining * 0.2, 60.0)
            alloc_lp = min(remaining * 0.25, 100.0)
            
            if alloc_sm > 20:
                a_try2, obj_try2 = smooth_max_optimize_fast(a_try, time_budget=alloc_sm,
                                                             tau_start=20.0, tau_end=200.0,
                                                             n_topk=30, lr_init=0.02, verbose=False)
            else:
                a_try2, obj_try2 = a_try, compute_obj(a_try)
            
            if remaining - alloc_sm > 40 and alloc_lp > 20:
                a_try3, obj_try3 = lp_polish(a_try2, time_budget=alloc_lp, max_iters=20, verbose=False)
                if obj_try3 < obj_try2:
                    a_try2, obj_try2 = a_try3, obj_try3
            
            if obj_try2 < best_score:
                best_score = obj_try2
                best_seq = a_try2.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try2:.8f}")
    
    # === Phase 6: Diverse restarts at best n ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 6: Diverse Restarts (budget={remaining:.0f}s) ===")
        n_cur = len(best_seq)
        for r in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            mode = (r + 1) % 7
            a_r = diverse_init(n_cur, mode)
            blend = 0.3 + 0.1 * r
            a_r = (1.0 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
            
            alloc = min(remaining * 0.3, 80.0)
            if alloc > 20:
                a_r2, obj_r2 = smooth_max_optimize_fast(a_r, time_budget=alloc * 0.5,
                                                         tau_start=10.0, tau_end=200.0,
                                                         n_topk=30, lr_init=0.02, verbose=False)
                if remaining - alloc * 0.5 > 30:
                    a_r3, obj_r3 = lp_polish(a_r2, time_budget=alloc * 0.4, max_iters=10, verbose=False)
                    if obj_r3 < obj_r2:
                        a_r2, obj_r2 = a_r3, obj_r3
            else:
                a_r2, obj_r2 = a_r, compute_obj(a_r)
            
            if obj_r2 < best_score:
                best_score = obj_r2
                best_seq = a_r2.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r2:.8f}")
    
    # === Phase 7: Final LP polish + subgradient squeeze ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 7: Final Squeeze (budget={remaining:.0f}s) ===")
        # Short subgradient
        alloc_sg = min(remaining * 0.3, 30.0)
        if alloc_sg > 10:
            a_sg, obj_sg = direct_subgradient(best_seq, time_budget=alloc_sg, lr0=0.01, verbose=False)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  SG: obj={best_score:.8f} **")
        
        # Short LP
        remaining2 = deadline - time.monotonic()
        alloc_lp = min(remaining2 * 0.5, 60.0)
        if alloc_lp > 15:
            a_lp, obj_lp = lp_polish(best_seq, time_budget=alloc_lp, max_iters=20, verbose=True)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: obj={best_score:.8f} **")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    # Clamp to valid range
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
```
