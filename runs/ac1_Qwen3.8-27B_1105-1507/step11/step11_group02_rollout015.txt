def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    fa = rfft(a, fft_size)
    return irfft(fa * fa, fft_size)[:size]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def convolve_with_a(a, g):

        fft_size <<= 1
    fa = rfft(a, fft_size)
    fg = rfft(g, fft_size)
    return irfft(fa * fg, fft_size)[:size]

def lp_direction(a, time_budget=30.0, verbose=False):
    """
    Solve LP: min M s.t. conv(a, g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses explicit matrix construction (feasible for n up to ~2000).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    # Build the matrix A where (A @ g)[k] = conv(a_norm, g)[k]
    # (A @ g)[k] = sum_j a_norm[j] * g[k-j] for valid j
    # This is a Toeplitz-like matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] contributes to positions k = j, j+1, ..., j+n-1 with weights a_norm[0], a_norm[1], ...
        # (A @ g)[k] includes a_norm[k-j] * g[j] when 0 <= k-j < n
        # So A[k, j] = a_norm[k-j] when 0 <= k-j < n
        row_start = j
        row_end = min(2 * n - 1, j + n)
        col_slice = np.arange(row_start - j, row_end - j)
        A[row_start:row_end, j] = a_norm[col_slice]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    start = time.monotonic()
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_g = float(np.sum(g_val))
                if s_g > 1e-10:
                    return g_val / s_g
        except Exception:
            continue
    return None


def lp_polish_fast(a, time_budget=60.0, verbose=False, seed=42):
    """LP-based polishing with multiple blend factors."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp = lp_direction(a, time_budget=min(remaining * 0.4, 20.0))
        if g_lp is None:
            break

        # Try multiple blend factors
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))

            a = best_a
            stall = 0
        else:

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * (n // 8) ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution (1/sqrt(1-x^2))
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-8)
        a = a / a.sum()
    elif mode == 4:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 5:
        # Triangular
        a = 1.0 - np.abs(2 * np.linspace(0, 1, n) - 1)
        a += 0.1 * rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 6:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
        a = a / a.sum()
    elif mode == 7:
        # Raised cosine
        x = np.linspace(0, 1, n)
        beta = rng.uniform(0.5, 2)
        a = 0.5 * (1 + np.cos(2 * np.pi * beta * (x - 0.5)))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
