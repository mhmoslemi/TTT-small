def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = fft(a, size)
    return ifft(fa * fa).real[:2*n-1]

from scipy.fft import fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = fft(a, size)
    return ifft(fa * fa).real[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv)/sum^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def compute_obj_from_conv(a, conv):
    """Compute objective from precomputed convolution."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(conv)) / (s * s)


def compute_smoothed_gradient(a, tau):
    """
    Compute smoothed gradient of max(conv(a,a)) via log-sum-exp.
    Returns (gradient, conv, p) where p is the softmax distribution.
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    grad_full = ifft(P * np.conj(A)).real
    return 2.0 * grad_full[:n], conv, p


def gradient_descent(a, time_budget=600.0, tau_init=20.0, tau_end=300.0,
                     cd_interval=5000, verbose=False, seed=42):
    """
    FFT-based projected gradient descent with adaptive tau and backtracking.
    Periodically inserts CD sweeps to handle non-convexity.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    conv = compute_conv(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    cd_count = 0

    # Adaptive learning rate
    lr = 5.0 / n
    lr_min = 1.0 / (n * n * 10)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * (progress ** 0.7)

        # Compute gradient
        try:
            grad, conv, p = compute_smoothed_gradient(a, tau)
        except Exception:
            break

        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-14:
            # Do a CD sweep if stuck
            a = cd_polish(a, time_budget=5.0, seed=seed + cd_count * 100)
            cd_count += 1
            obj_now = compute_obj(a)
            if obj_now < best_obj:
                best_obj = obj_now
                best_a = a.copy()
            continue

        # Backtracking line search
        cur_obj_val = 2.0 * n * float(np.max(conv))
        accepted = False
        lr_try = lr
        for _ in range(15):
            a_new = a - lr_try * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                lr_try *= 0.5
                continue
            a_new = a_new / s_new

            conv_new = compute_conv(a_new)
            obj_new_val = 2.0 * n * float(np.max(conv_new))

            # Armijo condition (simplified)
            if obj_new_val < cur_obj_val + 1e-6:
                a = a_new
                conv = conv_new
                cur_obj_val = obj_new_val
                it += 1
                lr = min(lr * 1.02, 10.0 / n)
                if obj_new_val < best_obj:
                    best_obj = obj_new_val
                    best_a = a_new.copy()
                accepted = True
                break
            else:
                lr_try *= 0.5

        if not accepted:
            # Try smaller step or CD
            lr = max(lr * 0.8, lr_min)
            if lr < lr_min * 1.5:
                # Insert a CD sweep
                a = cd_polish(a, time_budget=3.0, seed=seed + cd_count * 200)
                cd_count += 1
                conv = compute_conv(a)
                cur_obj_val = 2.0 * n * float(np.max(conv))
                obj_now = compute_obj(a)
                if obj_now < best_obj:
                    best_obj = obj_now
                    best_a = a.copy()

        # Periodic CD polish
        if it > 0 and it % cd_interval == 0 and time.monotonic() - start < time_budget - 10:
            a = cd_polish(a, time_budget=5.0, seed=seed + cd_count * 300)
            cd_count += 1
            conv = compute_conv(a)
            cur_obj_val = 2.0 * n * float(np.max(conv))
            obj_now = compute_obj(a)
            if obj_now < best_obj:
                best_obj = obj_now
                best_a = a.copy()

        if verbose and it % 10000 == 0 and it > 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, lr={lr:.2e}, tau={tau:.0f}, t={elapsed:.1f}s")

    return best_a


def cd_polish(a, time_budget=30.0, seed=42, verbose=False):
    """
    Fast coordinate descent with gradient-informed ordering.
    Uses incremental convolution updates for speed.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1

        # Compute gradient for ordering (every few sweeps)
        if sweep % 3 == 1:
            try:
                grad, _, _ = compute_smoothed_gradient(a, 30.0)
                order = np.argsort(-np.abs(grad))
            except Exception:
                order = rng.permutation(n)
        else:
            order = rng.permutation(n)

        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12 and v < 1e-12:
                continue

            # Indices affected: conv[i : i+n] (linear part) plus conv[2*i] (quadratic)
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i].copy()
            c_k = conv[ks].copy()

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            # 1D function to minimize
            mask = (ks != k_quad)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            def f_1d(x):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Quick candidate evaluation
            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, v * 0.5, v * 0.75, v * 1.25, v * 1.5,
                          1.0 / n, 2.0 / n, scale * 3]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section refinement
            x_lo = max(0.0, best_x - scale * 0.8)
            x_hi = best_x + scale * 1.5
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-10

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(35):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            S += delta

            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def multiplicative_update(a, time_budget=30.0, tau=50.0, seed=42):
    """
    Multiplicative gradient descent (like quasi-Newton for positive variables).
    a_new = a * exp(-lr * grad)
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 2.0 / n

    while time.monotonic() - start < time_budget:
        try:
            grad, conv, p = compute_smoothed_gradient(a, tau)
        except Exception:
            break

        # Multiplicative update
        update = np.exp(-lr * grad)
        update = np.clip(update, 0.001, 1000.0)
        a_new = a * update
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(np.sum(a_new))

        conv_new = compute_conv(a_new)
        obj_new = 2.0 * n * float(np.max(conv_new))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            lr = min(lr * 1.03, 10.0 / n)
        else:
            lr *= 0.9

        if lr < 1.0 / (n * n):
            break

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        # U-shaped
        x = np.linspace(0, 1, n)
        a = (x - 0.5) ** 2 + 0.1
        a = a / a.sum()
    elif mode == 2:
        # V-shaped (sharper)
        x = np.linspace(0, 1, n)
        a = np.abs(x - 0.5) + 0.05
        a = a / a.sum()
    elif mode == 3:
        # Random
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 4:
        # Two peaks
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * (n // 10) ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Cosine modulation
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** (-0.5)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def lp_polish(a, time_budget=20.0, seed=42):
    """
    Simple LP-based direction: maximize sum(g) subject to conv(a,g) <= M, g>=0.
    Uses scipy.optimize.linprog for simplicity (no cvxpy dependency issues).
    Falls back to gradient step if LP fails.
    """
    try:
        from scipy.optimize import linprog
    except ImportError:
        return a

    n = len(a)
    a_norm = a / float(np.sum(a))
    M = float(np.max(compute_conv(a_norm)))

    if time_budget < 3:
        return a

    # Build constraint matrix: conv(a_norm, g)[k] <= M * sum(g) for all k
    # This is: sum_j A[k,j] * g[j] <= M * sum_j g[j]
    # => sum_j (A[k,j] - M) * g[j] <= 0
    # Where A[k,j] = a_norm[k-j] if 0<=k-j<=n-1 else 0
    # (this gives conv(a, g)[k] = sum_j A[k,j] * g[j])
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]

    # conv(a,g)[k] - M*sum(g) <= 0
    # sum_j A[k,j]*g[j] - M*sum_j g[j] <= 0
    # sum_j (A[k,j] - M)*g[j] <= 0
    A_ub = A - M
    b_ub = np.zeros(2 * n - 1)

    # Maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': time_budget - 1})
        if res.success and res.x is not None:
            g_val = np.maximum(res.x, 0)
            s_g = float(np.sum(g_val))
            if s_g > 1e-10:
                g_val = g_val / s_g
                return g_val
    except Exception:
        pass
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines fast gradient descent, CD polishing,
    multi-length exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
            else:
                raise ValueError("sum too small")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    n_best_len = len(best_seq)

    # === Phase 1: Quick CD to establish baseline ===
    print(f"\n=== Phase 1: Quick CD ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 60.0)
    if p1_budget > 10:
        a_cd = cd_polish(best_seq, time_budget=p1_budget, seed=seed, verbose=False)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 2: Main Gradient Descent ===
    print(f"\n=== Phase 2: Gradient Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.45, 600.0)
    if p2_budget > 30:
        # Run GD with different tau schedules
        for gi, (ti, te) in enumerate([(15, 200), (30, 400), (5, 100)]):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = p2_budget / (3 - gi)
            alloc = min(alloc, remaining * 0.5)
            if alloc < 15:
                break

            a_gd = gradient_descent(best_seq, time_budget=alloc,
                                    tau_init=ti, tau_end=te,
                                    cd_interval=3000, verbose=True,
                                    seed=seed + gi * 1000)
            obj_gd = compute_obj(a_gd)
            if obj_gd < best_score:
                best_score = obj_gd
                best_seq = a_gd.copy()
                print(f"  GD {gi} (tau {ti}->{te}): obj={best_score:.8f} **")
            else:
                print(f"  GD {gi} (tau {ti}->{te}): obj={obj_gd:.8f}")

    # === Phase 3: CD Polish ===
    print(f"\n=== Phase 3: CD Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 180.0)
    if p3_budget > 15:
        a_cd2 = cd_polish(best_seq, time_budget=p3_budget,
                          seed=seed + 7777, verbose=True)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd2:.8f}")

    # === Phase 4: Multiplicative Update ===
    print(f"\n=== Phase 4: Multiplicative Update ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.1, 60.0)
        a_mu = multiplicative_update(best_seq, time_budget=alloc, tau=50.0,
                                     seed=seed + 8888)
        obj_mu = compute_obj(a_mu)
        if obj_mu < best_score:
            best_score = obj_mu
            best_seq = a_mu.copy()
            print(f"  MU: obj={best_score:.8f} **")
        else:
            print(f"  MU: obj={obj_mu:.8f}")
            # Try from perturbed start
            pert = rng.normal(0, 0.02, len(best_seq))
            a_pert = best_seq * np.exp(pert)
            a_pert = np.maximum(a_pert, 0) / float(a_pert.sum())
            a_mu2 = multiplicative_update(a_pert, time_budget=alloc, tau=30.0,
                                          seed=seed + 9999)
            obj_mu2 = compute_obj(a_mu2)
            if obj_mu2 < best_score:
                best_score = obj_mu2
                best_seq = a_mu2.copy()
                print(f"  MU (pert): obj={best_score:.8f} **")

    # === Phase 5: Multi-length Exploration ===
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = []
    for delta in [-200, -100, -50, 50, 100, 200, 300]:
        candidate = n_current + delta
        if 300 <= candidate <= 3000:
            n_options.append(candidate)
    # Also try some fixed lengths
    for fixed in [500, 700, 1000, 1200, 1500]:
        if 300 <= fixed <= 3000 and fixed not in n_options:
            n_options.append(fixed)
    n_options = sorted(set(n_options))[:5]
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # Brief GD
        alloc_gd = min(remaining * 0.15, 40.0)
        if alloc_gd > 10:
            a_try = gradient_descent(a_try, time_budget=alloc_gd,
                                     tau_init=20.0, tau_end=200.0,
                                     cd_interval=2000, verbose=False,
                                     seed=n_try)

        # Brief CD
        remaining = deadline - time.monotonic()
        alloc_cd = min(remaining * 0.05, 20.0)
        if alloc_cd > 5:
            a_try = cd_polish(a_try, time_budget=alloc_cd, seed=n_try + 500)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = 1 + r
        a_r = diverse_init(n_cur, rng, mode=mode)
        blend = 0.3 + 0.15 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / float(a_r.sum())

        alloc_gd = min(remaining * 0.15, 40.0)
        if alloc_gd > 10:
            a_r = gradient_descent(a_r, time_budget=alloc_gd,
                                   tau_init=10.0, tau_end=150.0,
                                   cd_interval=2000, verbose=False,
                                   seed=5000 + r * 500)
            alloc_cd = min(remaining * 0.05, 15.0)
            if alloc_cd > 5:
                a_r = cd_polish(a_r, time_budget=alloc_cd, seed=6000 + r * 500)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode {mode}): obj={obj_r:.8f}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # CD final
    if remaining > 20:
        alloc = min(remaining * 0.3, 100.0)
        a_final = cd_polish(best_seq, time_budget=alloc,
                            seed=seed + 19999, verbose=False)
        obj_f = compute_obj(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_f:.8f}")

    # Gradient final
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 60.0)
        a_gf = gradient_descent(best_seq, time_budget=alloc,
                                tau_init=50.0, tau_end=500.0,
                                cd_interval=1000, verbose=False,
                                seed=seed + 29999)
        obj_gf = compute_obj(a_gf)
        if obj_gf < best_score:
            best_score = obj_gf
            best_seq = a_gf.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # LP final (if time allows)
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 25.0)
        a_lp = lp_polish(best_seq, time_budget=alloc, seed=seed + 39999)
        if a_lp is not None and a_lp is not best_seq:
            obj_lp = compute_obj(a_lp)
            # Blend
            for t in [0.3, 0.5, 0.7, 1.0]:
                a_blend = (1 - t) * best_seq + t * a_lp
                a_blend = np.maximum(a_blend, 0)
                s_b = float(np.sum(a_blend))
                if s_b > 1e-10:
                    a_blend = a_blend / s_b
                    obj_b = compute_obj(a_blend)
                    if obj_b < best_score:
                        best_score = obj_b
                        best_seq = a_blend.copy()
                        print(f"  Final LP blend t={t}: obj={best_score:.8f} **")
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

    # One more CD if time
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_last = cd_polish(best_seq, time_budget=min(remaining * 0.5, 30.0),
                           seed=seed + 49999)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
