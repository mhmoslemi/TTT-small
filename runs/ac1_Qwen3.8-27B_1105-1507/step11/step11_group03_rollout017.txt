But for the LINEAR convolution, the relationship is different. However, for a sequence that's mostly concentrated in the middle (with small boundaries), the linear and circular convolutions are similar in the "bulk" region.

1. Make the spectrum as flat as possible (minimize variance of |A(ω)|^2)
2. This makes the circular autocorrelation concentrated at lag 0
3. For the linear convolution, this means the "bulk" of the convolution (away from the edges) is small
4. The edge effects (first few and last few lags) need to be managed separately

For step 4, we can use the fact that the edge lags are:
- `conv[0] = a_0^2`
- `conv[1] = 2*a_0*a_1`
- `conv[2] = 2*a_0*a_2 + a_1^2`
- etc.

These are controlled by the boundary values. If the boundary values are small relative to the middle, the edge effects are small.

Actually, I realize that for a truly flat spectrum, we'd need something like a maximum-length sequence (m-sequence) or a Zadoff-Chu sequence, but those have negative values. For non-negative sequences, the best we can do is approximate a flat spectrum.

Let me now code up the improved algorithm.

Key improvements over the previous code:
1. **Spectral equalization as primary optimizer**: Instead of LP as the main workhorse, use iterative spectrum flattening.
2. **Better gradient computation**: Use the FFT-based gradient of the log-sum-exp smoothed max more efficiently.
3. **Adaptive tau scheduling**: More aggressive annealing.
4. **Multiple restarts with different spectral templates**.
5. **Boundary-aware optimization**: Specifically handle the edge effects of the linear convolution.

Let me write the code:

Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses spectral equalization, smoothed gradient descent with adaptive annealing,
and LP-based active-set refinement for final polish.
"""

import time

from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def spectral_equalize(a, time_budget, n_iter_max=2000, verbose=False):
    """
    Iteratively flatten the spectrum of a while maintaining non-negativity.
    This is analogous to the Gerchberg-Saxton / Fienup algorithm.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(n_iter_max):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute spectrum
        A = fft(a, N)
        
        # Equalize magnitudes (with some smoothing to avoid oscillations)
        mag = np.abs(A)
        target_mag = np.ones(N)
        
        # Apply partial equalization (relaxation)
        alpha = 0.3 + 0.5 * (it / min(it + 1, n_iter_max))  # increase from 0.3 to 0.8
        new_mag = (1 - alpha) * mag + alpha * target_mag
        
        # Preserve phase
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        
        # Ensure Hermitian symmetry for real output
        A_new[0] = np.abs(A_new[0])
        if N % 2 == 0:
            A_new[N//2] = np.abs(A_new[N//2])
        
        # Inverse FFT
        a_new = ifft(A_new).real[:n]
        
        # Project onto non-negative cone
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn < 1e-10:
            # Reset with random perturbation
            a_new = np.abs(rng.standard_normal(n)) + 0.1
            sn = float(np.sum(a_new))
        a_new = a_new / sn
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if verbose and it % 50 == 0:
                print(f"    SpectralEq it{it}: obj={best_obj:.10f}")
        else:
            # Try a smaller step
            a_blend = 0.7 * a + 0.3 * a_new
            a_blend = np.maximum(a_blend, 0)
            sb = float(np.sum(a_blend))
            if sb > 1e-10:
                a_blend = a_blend / sb
                obj_blend = compute_obj(a_blend)
                if obj_blend < best_obj - 1e-14:
                    best_obj = obj_blend
                    best_a = a_blend.copy()
                    a = a_blend
                else:
                    # Perturb
                    a = a + 0.01 * rng.standard_normal(n)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
    
    return best_a


def smoothed_gd(a, time_budget, verbose=False, seed=42):
    """
    Nesterov accelerated gradient descent on the smoothed objective
    tau * log(sum exp(conv_k / tau)) with adaptive tau annealing.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Tau schedule: start large (smooth), decrease to small (sharp)
    tau_start = max(n * 0.5, 50.0)
    tau_end = max(n * 0.005, 1.0)
    n_stages = 8
    
    for stage in range(n_stages):
        tau = tau_start * (tau_end / tau_start) ** (stage / max(n_stages - 1, 1))
        # Add some randomness to tau
        tau *= (1 + 0.1 * rng.standard_normal())
        tau = max(tau_end, tau)
        
        # Learning rate schedule
        lr = 5.0 / np.sqrt(n)
        beta = 0.9
        v = np.zeros(n)
        
        stage_iters = 0
        max_stage_iters = 3000
        stage_start = time.monotonic()
        stage_time_budget = time_budget * 0.8 / n_stages  # reserve 20% for later stages
        
        while stage_iters < max_stage_iters:
            if time.monotonic() - start > time_budget * 0.85:
                break
            if time.monotonic() - stage_start > stage_time_budget * 1.5:
                break
            
            # Nesterov momentum
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy
            
            # Compute gradient
            g, _ = smoothed_grad(y, tau)
            
            # Project gradient onto the tangent space of the simplex
            g = g - np.mean(g)  # remove component along ones
            # Also project out negative direction
            g = np.maximum(g, -abs(np.min(g)) * 0.5)  # soft projection
            
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break
            
            # Adaptive learning rate
            lr_eff = lr / (1 + stage_iters * 0.01)
            
            v = beta * v - lr_eff * g / g_norm
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                s_new = 1.0
            a_new = a_new / s_new
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
            
            a = a_new
            stage_iters += 1
        
        if verbose:
            print(f"    SmoothedGD stage{stage} tau={tau:.1f}: obj={best_obj:.10f}, iters={stage_iters}")
        
        # Occasionally add perturbation between stages
        if stage < n_stages - 1:
            a = a + 0.005 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a


def lp_fixed_point(a, time_budget, verbose=False, seed=42, max_iters=500):
    """
    LP fixed-point iteration. Solves min M s.t. conv(a, g) <= M, g >= 0, sum(g) = 1.
    Then blends: a_new = (1-t)*a + t*g_optimal.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Build the constraint matrix (sparse)
    # conv(a, g)[k] = sum_i a[i] * g[k-i] (valid indices)
    # For efficiency, we use the fact that this is a convolution.
    
    stall_count = 0
    t_current = 0.15
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Solve LP: min M s.t. conv(a, g) <= M for all k, g >= 0, sum(g) = 1
        # conv(a, g)[k] = sum_{i} a[i] * g[k-i] where 0<=i<n, 0<=k-i<n
        # This is a convolution, so we can build the matrix efficiently.
        
        # For the LP, we need the matrix A where A[k, j] = contribution of g[j] to conv[k]
        # conv[k] = sum_j a[k-j] * g[j] (where 0 <= k-j < n)
        # So A[k, j] = a[k-j] if 0 <= k-j < n, else 0
        
        # This is a Toeplitz-like matrix. For n=800, it's (2n-1) x n = 1599 x 800.
        # We can build it as a sum of shifted versions of a.
        
        A_mat = np.zeros((2 * n - 1, n))
        for j in range(n):
            # g[j] contributes a[k-j] to conv[k] for k = j, j+1, ..., j+n-1
            # a[k-j] for k-j = 0, 1, ..., n-1
            A_mat[j:j+n, j] = a
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [A_mat @ g_var <= M_var, cp.sum(g_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        solved = False
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g_var.value, 0)
                    sg = float(np.sum(g_val))
                    if sg > 1e-10:
                        g_val = g_val / sg
                        M_val = float(M_var.value)
                        solved = True
                        break
            except Exception:
                continue
        
        if not solved:
            stall_count += 2
            if stall_count > 10:
                break
            continue
        
        # Try multiple blending factors
        t_candidates = [0.01, 0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        
        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t_current
        
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t
        
        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a
            t_current = max(0.02, min(1.0, best_local_t * 0.9))
            stall_count = 0
            if verbose and it % 20 == 0:
                print(f"    LP it{it:3d}: obj={best_obj:.10f}, t={best_local_t:.3f}")
        else:
            stall_count += 1
            if stall_count % 5 == 0:
                # Perturb to escape local optima
                strategy = stall_count % 4
                if strategy == 1:
                    # Target max lags
                    conv = compute_conv(a)
                    max_val = float(np.max(conv))
                    max_lags = np.where(conv >= max_val * 0.999)[0]
                    for lag in max_lags[:2]:
                        # Subgradient direction to reduce conv[lag]
                        sg = np.zeros(n)
                        for j in range(n):
                            k_j = lag - j
                            if 0 <= k_j < n:
                                sg[j] = 2.0 * a[k_j]
                        gn = np.linalg.norm(sg)
                        if gn > 1e-15:
                            a = a - 0.002 * sg / gn
                elif strategy == 2:
                    a = a + 0.01 * rng.standard_normal(n)
                elif strategy == 3:
                    # Low-frequency perturbation
                    x = np.arange(n)
                    f = rng.integers(1, 5)
                    p = rng.uniform(0, 2 * np.pi)
                    a = a + 0.01 * np.sin(2 * np.pi * f * x / n + p)
                elif strategy == 4:
                    # High-frequency perturbation
                    x = np.arange(n)
                    f = rng.integers(n//10, n//3 + 1)
                    p = rng.uniform(0, 2 * np.pi)
                    a = a + 0.008 * np.sin(2 * np.pi * f * x / n + p)
                
                a = np.maximum(a, 0)
                sa = float(np.sum(a))
                if sa > 1e-10:
                    a = a / sa
                else:
                    a = best_a.copy()
            
            if stall_count > 50:
                a = best_a.copy() + 0.05 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_count = 0
                t_current = 0.1
    
    return best_a


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_init(n, mode, rng):
    """Create structured initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        # "Anti-triangular" - high at edges, low in middle (to counteract convolution peak)
        x = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * (2 * x - 1) ** 2
        return a / a.sum()
    elif mode == 2:
        # Cosine modulation
        x = np.arange(n) / n
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x)
        return np.maximum(a, 0.01) / np.maximum(a, 0.01).sum()
    elif mode == 3:
        # Two-peak
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.25) / 0.1) ** 2) + np.exp(-((x - 0.75) / 0.1) ** 2)
        return a / a.sum()
    elif mode == 4:
        # Random positive
        a = rng.uniform(0.1, 1.0, n)
        return a / a.sum()
    elif mode == 5:
        # Sinc-like (abs)
        x = np.linspace(-1, 1, n)
        a = np.abs(np.sinc(x)) + 0.05
        return a / a.sum()
    elif mode == 6:
        # Inverse of triangular (to make convolution more uniform)
        x = np.linspace(0, 1, n)
        tri = 1 - np.abs(2 * x - 1)  # triangular, peaks at 0.5
        a = 1.0 / (0.1 + tri)  # inverse: high at edges, low in middle
        return a / a.sum()
    elif mode == 7:
        # Periodic-like
        x = np.arange(n) / n
        a = 1.0 + 0.5 * np.cos(2 * np.pi * 1.0 * x) + 0.2 * np.cos(2 * np.pi * 2.0 * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n))
        return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines spectral equalization, smoothed GD, and LP fixed-point.
    Explores multiple n values and initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # --- Load parent ---
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        parent = None
    
    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.10f}")
    
    n_current = len(best_seq)
    
    # === Phase 1: Multi-init screening with spectral equalization ===
    print(f"\n=== Phase 1: Multi-Init + Spectral Equalization ===")
    remaining = deadline - time.monotonic()
    
    # Screen different initializations
    n_candidates = [n_current]
    if n_current > 600:
        n_candidates.append(600)
    if n_current < 1000:
        n_candidates.append(1000)
    n_candidates.append(900)
    n_candidates = sorted(set([n for n in n_candidates if 400 <= n <= 1500]))
    
    for n_test in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break
        
        for mode in [0, 1, 6, 4, 7]:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            
            a_init = make_init(n_test, mode, rng)
            # Quick spectral equalization
            alloc = min(remaining * 0.03, 15.0)
            if alloc < 5:
                break
            a_screen = spectral_equalize(a_init, time_budget=alloc, n_iter_max=500, verbose=False)
            obj_screen = compute_obj(a_screen)
            
            if obj_screen < best_score:
                best_score = obj_screen
                best_seq = a_screen.copy()
                n_current = n_test
                print(f"  n={n_test} mode={mode}: obj={best_score:.10f} **")
            else:
                print(f"  n={n_test} mode={mode}: obj={obj_screen:.10f}")
    
    # === Phase 2: Smoothed GD on best ===
    print(f"\n=== Phase 2: Smoothed Gradient Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 120:
        alloc = min(remaining * 0.2, 180.0)
        a2 = smoothed_gd(best_seq, time_budget=alloc, verbose=True, seed=42)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 GD: obj={best_score:.10f} **")
        else:
            print(f"  Phase2 GD: obj={obj2:.10f}")
    
    # === Phase 3: Spectral equalization (extended) ===
    print(f"\n=== Phase 3: Extended Spectral Equalization ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        alloc = min(remaining * 0.15, 150.0)
        a3 = spectral_equalize(best_seq, time_budget=alloc, n_iter_max=5000, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
