def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def obj_norm(a):
    """Normalized objective: 2n*max(conv)/sum^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    return 2.0 * n * float(np.max(compute_conv(a_n)))


def build_conv_matrix(a):
    """Matrix A (2n-1 x n) such that A@g = conv(a,g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a, time_budget=15.0):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum g = 1, g >= 0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    start = time.monotonic()
    for solver in [cp.CLARABEL, cp.SCS, cp.MOSEK]:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                s = float(np.sum(gv))
                if s > 1e-10:
                    return gv / s
        except Exception:
            continue
    return None


def fast_cd(a, time_budget, seed=42, verbose=False):
    """
    Fast coordinate descent with vectorized convolution updates.
    Uses prefix/suffix max for efficient 'unaffected max' computation.
    Verifies each move against true objective before accepting.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0
    conv = compute_conv(a).copy()

    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    stall = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    # Precompute affected ranges
    k_lo_arr = np.arange(n)
    k_hi_arr = np.minimum(np.arange(n) + n - 1, 2 * n - 2)

    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        # Rebuild prefix/suffix max at start of sweep
        prefix_max = np.maximum.accumulate(conv)
        suffix_max = np.maximum.accumulate(conv[::-1])[::-1]
        sweep_best = best_obj
        accepted = 0

        for i in order:
            v = a[i]
            S_other = S - v
            if S_other < 1e-14 and v < 1e-14:
                continue

            k_lo = int(k_lo_arr[i])
            k_hi = int(k_hi_arr[i])

            # Unaffected max from prefix/suffix
            unmax = 0.0
            if k_lo > 0:
                unmax = max(unmax, float(prefix_max[k_lo - 1]))
            if k_hi < 2 * n - 2:
                unmax = max(unmax, float(suffix_max[k_hi + 1]))

            # Precompute affected base and corresponding a values
            affected_base = conv[k_lo:k_hi + 1]  # view, not copy
            length = k_hi - k_lo + 1
            a_ks = a[k_lo - i: k_hi + 1 - i]

            # Compute current f at v (should match current obj contribution)
            # Quick grid to find good bracket
            scale = max(v, 1.0 / n, 1e-8)
            candidates = np.array([0.0, max(0.0, v - scale), max(0.0, v - 0.5 * scale),
                                   v, max(0.0, v + 0.5 * scale), max(0.0, v + scale),
                                   max(0.0, v + 2.0 * scale), 2.0 / n])
            candidates = np.unique(np.maximum(candidates, 0.0))

            best_x = v
            best_f = (2.0 * n * max(unmax, float(np.max(affected_base)))) / (S * S)

            for x in candidates:
                S_new = S_other + x
                if S_new < 1e-30:
                    continue
                aff = affected_base + 2.0 * (x - v) * a_ks
                if 2 * i >= k_lo and 2 * i <= k_hi:
                    aff = aff.copy()
                    aff[2 * i - k_lo] = conv[2 * i] + x * x - v * v
                new_max = max(unmax, float(np.max(aff)))
                f_val = 2.0 * n * new_max / (S_new * S_new)
                if f_val < best_f:
                    best_f = f_val
                    best_x = x

            # Golden section search
            lo = max(0.0, best_x - scale * 1.5)
            hi = best_x + scale * 2.5
            if hi - lo < 1e-14:
                hi = lo + 1e-10

            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)

            # Inline eval as a helper (avoiding lambda due to constraints)
            # We'll use a local function defined in the outer scope pattern
            fc = _cd_eval(c, v, S_other, n, unmax, conv, a_ks, k_lo, k_hi, 2 * i)
            fd = _cd_eval(d, v, S_other, n, unmax, conv, a_ks, k_lo, k_hi, 2 * i)

            for _ in range(22):
                if hi - lo < 1e-13 * max(1.0, hi):
                    break
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = _cd_eval(c, v, S_other, n, unmax, conv, a_ks, k_lo, k_hi, 2 * i)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = _cd_eval(d, v, S_other, n, unmax, conv, a_ks, k_lo, k_hi, 2 * i)

            x_opt = max(0.0, (lo + hi) / 2.0)

            if abs(x_opt - v) < 1e-14:
                continue

            # Verify: check true objective improvement
            f_opt = _cd_eval(x_opt, v, S_other, n, unmax, conv, a_ks, k_lo, k_hi, 2 * i)
            if f_opt < best_f - 1e-15:
                # Apply update

                a[i] = x_opt
                S += delta
                conv[k_lo:k_hi + 1] = conv[k_lo:k_hi + 1] + 2.0 * delta * a_ks
                if 2 * i >= k_lo and 2 * i <= k_hi:
                    conv[2 * i] += delta * delta
                accepted += 1

                # Check true objective (conservative: use np.max(conv))
                true_obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if true_obj < sweep_best - 1e-15:
                    sweep_best = true_obj
                    if true_obj < best_obj:
                        best_obj = true_obj
                        best_a = a.copy()

        # End of sweep
        if sweep_best < best_obj - 1e-14:
            best_obj = sweep_best
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.9f}, t={elapsed:.1f}s, acc={accepted}")

        if stall > 30:
            # Perturb to escape local minimum
            pert_scale = 0.005 + 0.002 * min(stall - 30, 20)
            a = best_a * np.exp(rng.normal(0, pert_scale, n))
            a = np.maximum(a, 0)
            S = float(np.sum(a))
            if S < 1e-10:
                a = best_a.copy()
                S = 1.0
            else:
                a = a / S
                S = 1.0
            conv = compute_conv(a)
            best_obj = 2.0 * n * float(np.max(conv))
            best_a = a.copy()
            stall = 0

    return best_a


def _cd_eval(x, v, S_other, n, unmax, conv, a_ks, k_lo, k_hi, k_quad):
    """Evaluate the 1D objective for changing a[i] from v to x."""
    S_new = S_other + x
    if S_new < 1e-30:
        return 1e30
    aff = conv[k_lo:k_hi + 1] + 2.0 * (x - v) * a_ks
    if k_quad >= k_lo and k_quad <= k_hi:
        aff = aff.copy()
        aff[k_quad - k_lo] = conv[k_quad] + x * x - v * v
    new_max = max(unmax, float(np.max(aff)))
    return 2.0 * n * new_max / (S_new * S_new)


def lp_polish(a, time_budget=40.0, seed=42):
    """LP-based Frank-Wolfe polishing with multiple blend factors."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp = solve_lp_direction(a, time_budget=min(remaining * 0.5, 12.0))
        if g_lp is None:
            break

        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = obj_norm(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 2:
                a = a * np.exp(rng.normal(0, 0.005, n))
                a = np.maximum(a, 0) / float(np.sum(a))
                stall = 0

    return best_a


def subgradient_polish(a, time_budget=30.0, seed=42):
    """Subgradient-based polishing using multiplicative updates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a

        grad = np.zeros(n)
        active_ks = np.where(active)[0]
        for k in active_ks:

            # d/db[i] = b[k-i] (if valid) + b[i-k] (if valid)

            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            # d(conv_k)/db[j] = b[k-j] for j in range
            # For our purposes, the gradient w.r.t. b[i] is:
            # sum over active k of (b[k-i] if valid + b[i-k] if valid)
            # Let's just compute it directly
            pass

        # Simpler: compute gradient via FFT
        # grad = 2 * ifft(fftconv) approximately
        # Actually, the subgradient of max_k (b*b)[k] at point b is:
        # (1/|A|) * sum_{k in A} d(b*b)[k]/db
        # where A = argmax set
        # d(b*b)[k]/db[i] = b[k-i] + b[i-k] (for valid indices)
        # This is the same as the "cross-correlation" of b with the indicator of active positions

        # Build the "active" convolution: sum_{k active} (b*b)[k] gradient
        # = convolve(b, indicator_active) + convolve(b, flip(indicator_active))
        # Actually simpler: the gradient of (b*b)[k] w.r.t. b is:
        # [b[k], b[k-1], ..., b[0], 0, ..., 0, 0, ..., b[k-n+1]] (length 2n-1 mapped back to n)
        # Let's use the direct formula:
        # grad[i] = sum_{k active} (b[(k-i) mod n] if cyclic) -> not cyclic
        # For linear conv: d(conv_k)/db_i = b_{k-i} (if 0<=k-i<n) + b_{i-k} (if 0<=i-k<n)

        # Efficiently: for each active k, add b shifted
        # Let's use FFT: the gradient is the "correlation" of b with the active set
        N = 1
        while N < 4 * n:
            N <<= 1
        active_full = np.zeros(N)
        active_full[:len(active)] = active
        B = fft(a, N)
        # The gradient of sum_k active[k] * (b*b)[k] w.r.t. b
        # = 2 * IFFT(FFT(b) * FFT(active))[:n]  (approximately, for the linear case)
        # Actually it's more nuanced. Let me just do it directly for small active sets.
        if len(active_ks) <= 100:
            for k in active_ks:
                lo = max(0, k - n + 1)
                hi = min(k, n - 1)
                # d/db[i] (b*b)[k] = b[k-i] for i in [k-n+1, k]
                # So for i in [lo, hi]: grad[i] += b[k-i]
                # And for i in [k-n+1, k]: also b[i-k] if i != k
                # Actually for k != 0: d/db_i = b[k-i] + b[i-k] (the two terms)
                # But if k-i == i-k, i.e., i = k/2, it's just b[k/2]
                # For simplicity, just add b[k-i] for valid i
                js = np.arange(lo, hi + 1)
                grad[js] += a[k - js]
                # Also the "other" term: b[i-k] for i in [k, k+n-1] intersect [0,n-1]
                lo2 = max(0, k)
                hi2 = min(n - 1, k + n - 1)
                if lo2 <= hi2:
                    js2 = np.arange(lo2, hi2 + 1)
                    grad[js2] += a[js2 - k]
        else:
            # Too many active, use uniform weighting
            grad = np.full(n, 1.0 / n)

        # Normalize gradient
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break

        # Adaptive learning rate
        lr = 2.0 / (np.sqrt(n) * (1.0 + 0.5 * (it / max(1, time_budget / 0.1))))
        lr = min(lr, 5.0 / g_norm)

        # Multiplicative update

        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(np.sum(a_new))

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:

            best_a = a_new.copy()
        a = a_new
        it += 1

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        x = np.arange(n)
        c1, c2 = n // 3, 2 * n // 3
        w = n // 10
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x ** 2 + 1e-8)
        return a / a.sum()
    elif mode == 4:
        # Sinc-squared centered
        x = np.arange(n) - n / 2.0
        a = np.sinc(x / (n * 0.3)) ** 2
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    elif mode == 5:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a += 0.1 * rng.uniform(0, 1, n)
        return a / a.sum()
    else:
        a = rng.chisquare(2, n)
        return a / a.sum()

        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = obj_norm(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.9f}")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = obj_norm(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.9f}")

    n_best = len(best_seq)

    # === Phase 1: Main CD run ===
    print(f"\n=== Phase 1: Fast CD (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 480.0)
    if p1_budget > 30:
        a_cd = fast_cd(best_seq, time_budget=p1_budget, verbose=True, seed=seed)
        obj_cd = obj_norm(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.9f} **")
        else:
            print(f"  CD: obj={obj_cd:.9f} (no improvement)")

    # === Phase 2: LP Polish ===
    print(f"\n=== Phase 2: LP Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 150.0)
    if p2_budget > 15:
        for rnd in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(p2_budget / (4 - rnd), remaining * 0.6)
            if alloc < 10:
                break
            a_lp = lp_polish(best_seq, time_budget=alloc, seed=5000 + rnd * 7)
            obj_lp = obj_norm(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP r{rnd}: obj={best_score:.9f} **")
            else:
                print(f"  LP r{rnd}: obj={obj_lp:.9f}")

    # === Phase 3: Post-LP CD ===
    print(f"\n=== Phase 3: Post-LP CD ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.2, 200.0)
    if p3_budget > 20:
        a_cd2 = fast_cd(best_seq, time_budget=p3_budget, seed=seed + 999)
        obj_cd2 = obj_norm(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  CD2: obj={best_score:.9f} **")
        else:
            print(f"  CD2: obj={obj_cd2:.9f}")

    # === Phase 4: Subgradient polish ===
    print(f"\n=== Phase 4: Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.08, 60.0)
        a_sg = subgradient_polish(best_seq, time_budget=alloc, seed=seed + 777)
        obj_sg = obj_norm(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  SG: obj={best_score:.9f} **")
        else:
            print(f"  SG: obj={obj_sg:.9f}")
        # Follow up with CD
        remaining = deadline - time.monotonic()
        if remaining > 30 and obj_sg < best_score + 1e-6:
            alloc2 = min(remaining * 0.1, 60.0)
            a_cd3 = fast_cd(best_seq, time_budget=alloc2, seed=seed + 1111)
            obj_cd3 = obj_norm(a_cd3)
            if obj_cd3 < best_score:
                best_score = obj_cd3
                best_seq = a_cd3.copy()
                print(f"  CD3: obj={best_score:.9f} **")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n ===")
    remaining = deadline - time.monotonic()
    n_options = []
    for dn in [-100, -50, 50, 100, 150, 200, 300]:
        nt = n_best + dn
        if 500 <= nt <= 2000:
            n_options.append(nt)
    # Also try some fixed values
    for fixed_n in [700, 900, 1000, 1200]:
        if fixed_n not in n_options and abs(fixed_n - n_best) > 50:
            n_options.append(fixed_n)
    n_options = sorted(set(n_options))[:5]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break

        a_try = resample_seq(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        alloc_cd = min(remaining * 0.15, 40.0)
        if alloc_cd > 15:
            a_try = fast_cd(a_try, time_budget=alloc_cd, seed=n_try)

        remaining = deadline - time.monotonic()
        if remaining > 25:
            alloc_lp = min(remaining * 0.05, 15.0)
            if alloc_lp > 8:
                a_try = lp_polish(a_try, time_budget=alloc_lp, seed=n_try + 3000)

        obj_try = obj_norm(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.9f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.9f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        mode = 1 + r
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / float(a_r.sum())

        alloc = min(remaining * 0.15, 50.0)
        if alloc > 20:
            a_r = fast_cd(a_r, time_budget=alloc, seed=9000 + r * 137)
            obj_r = obj_norm(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): obj={best_score:.9f} **")
            else:
                print(f"  Restart {r} (mode {mode}): obj={obj_r:.9f}")

    # === Phase 7: Final CD polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 100.0)
        a_final = fast_cd(best_seq, time_budget=alloc, seed=seed + 25000)
        obj_final = obj_norm(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.9f} **")
        else:
            print(f"  Final CD: obj={obj_final:.9f}")

    # Final LP if time
    remaining = deadline - time.monotonic()
    if remaining > 12:
        a_lp_f = lp_polish(best_seq, time_budget=min(remaining * 0.5, 30.0), seed=9999)
        obj_lp_f = obj_norm(a_lp_f)
        if obj_lp_f < best_score:
            best_score = obj_lp_f
            best_seq = a_lp_f.copy()
            print(f"  Final LP: obj={best_score:.9f} **")

    # Final CD if time
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_cd_f = fast_cd(best_seq, time_budget=min(remaining * 0.6, 40.0), seed=seed + 30000)
        obj_cd_f = obj_norm(a_cd_f)
        if obj_cd_f < best_score:
            best_score = obj_cd_f
            best_seq = a_cd_f.copy()
            print(f"  Final CD2: obj={best_score:.9f} **")

    # Validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = obj_norm(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.9f}, time={elapsed:.1f}s ===")

    return result
