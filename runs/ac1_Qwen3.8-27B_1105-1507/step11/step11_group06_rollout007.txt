parametric shape exploration, symmetry exploitation, and better time management.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_fast(a):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    # LP: min M s.t. A@g <= M, sum(g)=1, g>=0
    # Variables: g (n), M (1) -> n+1 variables
    # Constraints: A@g - M <= 0 (2n-1), sum(g) = 1
    # Objective: minimize M
    # Using scipy linprog:
    # c = [0,...,0, 1] (minimize M, last variable)
    # A_ub: [A, -I_{2n-1}] (2n-1 rows, n+1 cols)
    # b_ub: zeros(2n-1)
    # A_eq: [ones(1,n), 0]
    # b_eq: [1]
    # bounds: g_i in [0,inf], M in [0,inf]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([A, -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    if result.success and result.x is not None:
        g_val = np.maximum(result.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s
    return None


def solve_lp_cvxpy(a):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def solve_lp_combined(a, timeout=2.0):
    """Try fast scipy first, fall back to cvxpy."""
    t0 = time.monotonic()
    try:
        g = solve_lp_fast(a)
        if g is not None:
            return g
    except Exception:
        pass
    if time.monotonic() - t0 > timeout:
        return None
    try:
        g = solve_lp_cvxpy(a)
        return g
    except Exception:
        return None


def line_search(a, direction, n_coarse=50, n_fine=40, t_max=1.5):
    """Exhaustive line search for (1-t)*a + t*direction."""
    best_t = 0.0
    best_obj = compute_obj(a)
    ts = np.linspace(0.0, t_max, n_coarse)
    for t in ts:
        a_try = (1.0 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    if 0.0 < best_t < t_max:
        lo = max(0.0, best_t - 2.0 / n_coarse)
        hi = min(t_max, best_t + 2.0 / n_coarse)
        ts_fine = np.linspace(lo, hi, n_fine)
        for t in ts_fine:
            a_try = (1.0 - t) * a + t * direction
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t
    return best_t, best_obj


def nesterov_subgradient_polish(a, time_budget=30.0):
    """Nesterov-accelerated subgradient method on the exact objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a.copy(), compute_obj(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Nesterov variant
    x = a.copy()  # current point
    y = a.copy()  # extrapolated point
    beta = 0.7   # momentum
    lr0 = 0.3
    it = 0
    best_it = 0
    patience = 500

    while time.monotonic() - start < time_budget:
        # Compute subgradient at y (the extrapolated point)
        conv_y = compute_conv(y)
        k_max = int(np.argmax(conv_y))
        
        # Subgradient of max_k C_k at y: 2*shift(y, k_max)
        grad = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                grad[i] = 2.0 * y[j]
        
        # Step size
        lr_t = lr0 / np.sqrt(1.0 + it)
        
        # Update y
        x_new = y - lr_t * grad
        x_new = np.maximum(x_new, 0)
        s_new = float(np.sum(x_new))
        if s_new < 1e-10:
            x_new = best_a.copy()
        else:
            x_new = x_new / s_new
        
        # Nesterov extrapolation
        t_k = 2.0 / (2.0 + it + np.sqrt(4.0 + it * it))
        y = x_new + (t_k - 1.0) * (x_new - y)
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y > 1e-10:
            y = y / s_y
        else:
            y = x_new.copy()
        
        x = x_new
        it += 1
        
        # Check on averaged iterate (Hedge / multipolarity)
        if it % 5 == 0:
            obj_x = compute_obj(x)
            if obj_x < best_obj:
                best_obj = obj_x
                best_a = x.copy()
                best_it = it
        
        if it - best_it > patience:
            # Restart
            x = best_a.copy()
            y = best_a.copy()
            lr0 = max(0.05, lr0 * 0.5)
            best_it = it
            patience = max(200, patience // 2)
    
    obj_final = compute_obj(x)
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = x.copy()
    
    return best_a, best_obj


def mirror_descent_polish(a, time_budget=30.0, entropy_param=1.0):
    """Mirror descent with entropic regularization on the simplex."""
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr0 = 0.1
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))
        grad = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        lr_t = lr0 / np.sqrt(1.0 + it)
        # Mirror descent update (exponential family)
        a_new = a * np.exp(-lr_t * grad / entropy_param)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a = a_new / s_new
        it += 1
        
        if it % 3 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                best_it = it
        
        if it - best_it > 300:
            a = best_a.copy()
            lr0 = max(0.02, lr0 * 0.5)
            best_it = it
    
    return best_a, best_obj


def u_shaped_init(n, sharpness=3.0, eps=0.01):
    x = np.linspace(0, 1, n)
    a = 1.0 / (eps + sharpness * (2 * x - 1) ** 2)
    # Enforce symmetry
    a = 0.5 * (a + a[::-1])
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def cosine_init(n, power=1.0):
    x = np.linspace(0, np.pi, n)
    a = np.abs(np.cos(x)) ** power + 0.01
    a = 0.5 * (a + a[::-1])
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.02 + 5.0 * (2 * x - 1) ** 4)
    elif mode == 6:
        a = 1.0 / (0.005 + 10.0 * (2 * x - 1) ** 2)
    elif mode == 7:
        a = (1 + np.cos(2 * np.pi * x)) ** 2 + 0.01
    else:
        a = 1.0 / (0.1 + 2.0 * np.abs(2 * x - 1))
    a = 0.5 * (a + a[::-1])
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def fw_optimize(a, time_budget=100.0, max_iters=500, verbose=False):
    """Frank-Wolfe optimization with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        T0 = time.monotonic()
        g_fw = solve_lp_combined(a, timeout=min(2.0, remaining * 0.3))
        T1 = time.monotonic()
        
        if g_fw is None:
            break
        
        t_opt, obj_opt = line_search(a, g_fw, n_coarse=40, n_fine=30, t_max=1.5)
        
        if t_opt > 1e-6 and obj_opt < best_obj - 1e-12:
            a_new = (1.0 - t_opt) * a + t_opt * g_fw
            a_new = np.maximum(a_new, 0)
            # Enforce symmetry
            a_new = 0.5 * (a_new + a_new[::-1])
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            # Try direct replacement
            obj_direct = compute_obj(g_fw)
            if obj_direct < best_obj:
                best_obj = obj_direct
                best_a = g_fw.copy()
                a = g_fw.copy()
                stall = 0
            else:
                stall += 1
        
        it += 1
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    FW it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                  f"elapsed={elapsed:.1f}s, remaining={remaining:.1f}s")
        
        if stall >= 5:
            if stall <= 8:
                # Small perturbation
                freq = rng.integers(1, max(2, n // 100))
                phase = rng.uniform(0, 2 * np.pi)
                idx = np.arange(n, dtype=np.float64)
                pert = 0.02 * np.cos(2 * np.pi * freq * idx / n + phase)
                a = best_a * np.exp(pert)
                a = 0.5 * (a + a[::-1])
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 0.01:
                    a = a / s
                else:
                    a = best_a.copy()
            else:
                # Larger restart
                mode = rng.integers(0, 8)
                a_div = diverse_init(n, rng, mode=mode)
                a = 0.4 * a_div + 0.6 * best_a
                a = 0.5 * (a + a[::-1])
                a = np.maximum(a, 0)
                a = a / a.sum()
            stall = 0
    
    return best_a


def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = 0.5 * (a_dst + a_dst[::-1])
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s

            print(f"  Micro-polish: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure symmetry for final answer (often helps)
    best_seq = 0.5 * (best_seq + best_seq[::-1])
    best_seq = np.maximum(best_seq, 0)
    s = float(np.sum(best_seq))
    if s > 0.01:
        best_seq = best_seq / s

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
