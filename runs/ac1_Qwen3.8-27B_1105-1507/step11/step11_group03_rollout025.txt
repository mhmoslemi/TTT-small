import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def lp_solve(n, a, solver_list=None):
    """Solve min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1. Returns (g, M, active_lags)."""
    # conv(a,g)_k = sum_j a_{k-j} * g_j for j=0..n-1, where a is padded with zeros
    # Matrix form: A @ g <= M where A[k,j] = a_{k-j} if 0 <= k-j < n else 0
    A = np.zeros((2 * n - 1, n))
    a_ext = np.zeros(2 * n - 1)
    a_ext[:n] = a
    for j in range(n):
        A[j:j+n, j] = a_ext[j:j+n]
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    
    best_g = None
    best_M = np.inf

            continue

        return None, None, None

        return None, None, None

    # Find active lags
    conv_ag = A @ best_g
    active_lags = np.where(conv_ag >= best_M * 0.9999)[0]
    
    return best_g, best_M, active_lags

def line_search_step(a, g, n, max_t=2.0):
    """Find optimal t in [0, max_t] minimizing f(a + t*(g-a)).

    Since f is convex in t, use golden section search.
    """
    d = g - a
    C = compute_conv(a)
    L = compute_conv_cross(a, d)  # (a*d)_k = sum_j a_j * d_{k-j}
    Q = compute_conv(d)
    
    # f(t) = 2n * max_k (C_k + 2*t*L_k + t^2*Q_k)
    # We want to minimize over t in [0, max_t]
    
    # Quick check: is t=0 the minimum?
    f0 = float(np.max(C))
    # Check derivative at 0 (subgradient): if all active terms have non-negative slope, t=0 is optimal
    max_lags = np.where(C >= f0 * 0.9999)[0]
    slopes = 2.0 * L[max_lags] + 2.0 * 0.0 * Q[max_lags]  # d/dt at t=0 is 2*L_k
    if np.all(slopes >= -1e-12):
        return 0.0
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~ 0.618
    
    def f_of_t(t):
        vals = C + 2.0 * t * L + t * t * Q
        return 2.0 * n * float(np.max(vals))
    
    lo, hi = 0.0, max_t
    # First check if f(hi) < f(lo), if not, reduce hi
    f_lo = f_of_t(lo)
    f_hi = f_of_t(hi)
    if f_hi >= f_lo:
        # Minimum is at t=0 or very small
        hi = 0.5
        f_hi = f_of_t(hi)
        if f_hi >= f_lo:
            hi = 0.1
            f_hi = f_of_t(hi)
            if f_hi >= f_lo:
                return 0.0
    
    # Golden section
    c = hi - gr * (hi - lo)
    dpt = lo + gr * (hi - lo)
    fc = f_of_t(c)
    fd = f_of_t(dpt)
    
    for _ in range(50):  # 50 iterations gives precision ~ (0.618)^50 ~ 1e-11

            hi = dpt
            dpt = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f_of_t(c)
        else:
            lo = c
            c = dpt
            fc = fd
            dpt = lo + gr * (hi - lo)
            fd = f_of_t(dpt)
    
    return (lo + hi) / 2

def compute_conv_cross(a, b):

    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:2 * n - 1]

def smoothed_max_grad(a, tau=10.0):
    """Compute subgradient of smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    shifted = tau * (conv - max_val)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    # Gradient of sum_k p_k * (a*a)_k with respect to a
    # = 2 * sum_k p_k * (P_k a) where P_k is the anti-diagonal for lag k
    # = 2 * (p_conv * a) where p_conv is the circular-ish convolution
    # More precisely: grad_j = 2 * sum_k p_k * a_{k-j} (where a is 0 outside [0,n))
    # This is 2 * (p_pad * a_pad) evaluated at the right positions
    
    # Use FFT: the gradient at position j is 2 * sum_k p_k * a_{k-j}
    # = 2 * (p * a)_j where * is the cross-correlation
    # In FFT: ifft(fft(p) * conj(fft(a))) gives cross-correlation
    
    N = 1
    while N < 2 * n:
        N <<= 1
    P = np.fft.fft(p, N)
    A = np.fft.fft(a, N)
    cross = np.fft.ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    
    return g, max_val

def gradient_polish(a, time_budget, tau_levels=None, verbose=True):
    """Polish using gradient descent on smoothed max."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if tau_levels is None:
        tau_levels = [5, 15, 50, 150, 400, 1000, 3000]
    
    for tau in tau_levels:
        if time.monotonic() - start > time_budget:
            break
        
        lr = 5.0 / np.sqrt(n)
        beta = 0.9
        v = np.zeros(n)
        iters_this_tau = 0
        
        while time.monotonic() - start < time_budget and iters_this_tau < 3000:
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                break
            y = y / sy
            
            g, _ = smoothed_max_grad(y, tau)
            gn = float(np.linalg.norm(g))
            if gn < 1e-15:
                break
            
            # Line search (simple backtracking)
            step = lr
            best_try = a
            best_try_obj = compute_obj(a)
            for _ in range(10):
                a_try = y - step * g / gn
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    step *= 0.5
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_try_obj - 1e-15:
                    best_try = a_try.copy()
                    best_try_obj = obj_try
                step *= 0.7
            
            if best_try_obj < best_obj - 1e-14:
                best_obj = best_try_obj
                best_a = best_try.copy()
            
            v = beta * v + (best_try - y)
            a = best_try
            iters_this_tau += 1
            lr *= 0.9999
        
        if verbose:
            print(f"    GradPolish tau={tau:5d}: obj={best_obj:.10f}")
    
    return best_a

def resample(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP fixed-point with golden-section line search, multi-solver LP,
gradient polish on smoothed max, and multi-n exploration.
Key improvement over prior work: proper 1D line search along LP direction
instead of fixed step sizes, leading to faster convergence.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Compute (a*b)_k = sum_j a_j * b_{k-j} for sequences of same length n."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lp_solve(n, a, solver_list=None):
    """Solve min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1.
    Returns (g, M, active_lags) or (None, None, None) on failure."""
    A = np.zeros((2 * n - 1, n))
    a_pad = np.zeros(2 * n - 1)
    a_pad[:n] = a
    for j in range(n):
        A[j:j + n, j] = a_pad[j:j + n]

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]

    best_g = None
    best_M = np.inf
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                M_val = float(M_var.value) if M_var.value is not None else np.inf
                if M_val < best_M:
                    best_M = M_val
                    best_g = np.maximum(g_var.value, 0)
        except Exception:
            continue

    if best_g is None:
        return None, None, None

    sg = float(np.sum(best_g))
    if sg < 1e-10:
        return None, None, None
    best_g = best_g / sg

    conv_ag = A @ best_g
    threshold = best_M * 0.9999
    active_lags = np.where(conv_ag >= threshold)[0]

    return best_g, best_M, active_lags


def line_search_lp_step(a, g, n):
    """Find optimal t in [0,1] minimizing 2n*max(conv(a+t*(g-a), a+t*(g-a))).
    Uses golden section search on the convex function f(t)."""
    d = g - a
    C = compute_conv(a)
    L = 2.0 * compute_conv_cross(a, d)
    Q = compute_conv(d)

    def f_of_t(t):
        vals = C + t * L + t * t * Q
        return float(np.max(vals))

    f0 = f_of_t(0.0)
    f1 = f_of_t(1.0)

    # Check if t=0 is optimal: if f is non-decreasing at 0
    if f1 >= f0:
        # Try smaller steps to see if there's a local minimum
        f_half = f_of_t(0.5)
        if f_half >= f0:
            f_q = f_of_t(0.25)
            if f_q >= f0:
                return 0.0

    # Golden section search on [0, 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    dpt = lo + gr * (hi - lo)
    fc = f_of_t(c)
    fd = f_of_t(dpt)

    for _ in range(60):
        if fc < fd:
            hi = dpt
            dpt = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f_of_t(c)
        else:
            lo = c
            c = dpt
            fc = fd
            dpt = lo + gr * (hi - lo)
            fd = f_of_t(dpt)

    t_opt = (lo + hi) / 2.0
    # Verify the result is actually better
    f_opt = f_of_t(t_opt)
    if f_opt >= f0 - 1e-15:
        return 0.0
    return t_opt


def lp_fixed_point(a, time_budget, verbose=True, seed=42, max_iters=3000):
    """LP fixed-point with golden-section line search for step size."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g, M_lp, active_lags = lp_solve(n, a)
        if g is None:
            stall += 1
            if stall > 5:
                break
            continue

        # Line search for optimal step
        t_opt = line_search_lp_step(a, g, n)

        if t_opt < 1e-8:
            # LP direction gives no improvement; perturb
            stall += 1
            if stall <= 3:
                a = a + 0.003 * rng.standard_normal(n)
            elif stall <= 6:
                # Target active lags: reduce their contribution
                conv = compute_conv(a)
                max_val = float(np.max(conv))
                max_lags = np.where(conv >= max_val * 0.999)[0]
                for lag in max_lags[:2]:
                    grad = np.zeros(n)
                    for i in range(n):
                        j = lag - i
                        if 0 <= j < n:
                            grad[i] = a[j]
                    gn = np.linalg.norm(grad)
                    if gn > 1e-15:
                        a = a - 0.002 * grad / gn
                a = np.maximum(a, 0)
                a = a / a.sum()
            else:
                # Larger perturbation
                a = best_a.copy() + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
            continue

        # Apply the step
        a_new = (1.0 - t_opt) * a + t_opt * g
        a_new = np.maximum(a_new, 0)
        sa = float(np.sum(a_new))
        if sa < 1e-10:
            a_new = a.copy()
            continue
        a_new = a_new / sa

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 30 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it{it:4d}: obj={best_obj:.10f}, t={t_opt:.4f}, "
                      f"M_lp={M_lp:.6f}, elapsed={elapsed:.0f}s")
        else:
            # Try a smaller step
            improved = False
            for t_try in [0.5 * t_opt, 0.2 * t_opt, 0.05, 0.01]:
                if t_try < 1e-6:
                    break
                a_try = (1.0 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            if not improved:
                stall += 1
                if stall > 10:
                    a = best_a.copy() + 0.005 * rng.standard_normal(n)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    stall = 0

        if verbose and it % 100 == 0 and it > 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it:4d}: obj={best_obj:.10f}, elapsed={elapsed:.0f}s")

    return best_a


def smoothed_grad(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    shifted = tau * (conv - max_val)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    N = 1
    while N < 2 * n:
        N <<= 1
    P = np.fft.fft(p, N)
    A = np.fft.fft(a, N)
    cross = np.fft.ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, max_val


def gradient_polish(a, time_budget, tau_levels=None, verbose=True):
    """Polish using accelerated gradient descent on smoothed max."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [5, 15, 50, 150, 500, 1500, 5000]

    for tau in tau_levels:
        if time.monotonic() - start > time_budget:
            break

        lr = 3.0 / np.sqrt(n)
        beta = 0.9
        v = np.zeros(n)
        iters = 0
        max_iters = 5000

        while time.monotonic() - start < time_budget and iters < max_iters:
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                break
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            gn = float(np.linalg.norm(g))
            if gn < 1e-15:
                break

            # Backtracking line search
            step = lr
            a_try = y.copy()
            obj_try = compute_obj(a_try)
            for _ in range(15):
                a_cand = y - step * g / gn
                a_cand = np.maximum(a_cand, 0)
                sc = float(np.sum(a_cand))
                if sc < 1e-10:
                    step *= 0.5
                    continue
                a_cand = a_cand / sc
                obj_cand = compute_obj(a_cand)
                if obj_cand < obj_try - 1e-16:
                    a_try = a_cand
                    obj_try = obj_cand
                    break
                step *= 0.6

            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()

            v = beta * v + (a_try - y)
            a = a_try
            iters += 1
            lr = max(1e-8, lr * 0.99999)

        if verbose:
            print(f"    GradPolish tau={tau:5d}: obj={best_obj:.10f}, iters={iters}")

    return best_a


def make_init(n, mode=0, rng=None):
    """Create structured initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.maximum(np.sinc(x), 0.001)
        return a / a.sum()
    elif mode == 2:
        a = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
        return np.maximum(a, 0.001) / a.sum()
    elif mode == 3:
        x = np.arange(n) / (n - 1)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        return np.maximum(a, 0.001) / a.sum()
    elif mode == 4:
        x = np.linspace(-3, 3, n)
        a = np.exp(-x ** 2)
        return a / a.sum()
    elif mode == 5:
        center = n / 2.0
        x = np.abs(np.arange(n) - center) / center
        a = np.exp(-2.0 * x ** 2)
        return a / a.sum()
    elif mode == 6:
        x = np.arange(n) / n
        a = 0.5 * (1.0 + np.cos(2 * np.pi * x))
        return a / a.sum()
    elif mode == 7:
        x = np.arange(n) / n
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
        return np.maximum(a, 0.01) / a.sum()
    elif mode == 8:
        # Two-lobe
        center = n / 2.0
        x = (np.arange(n) - center) / center
        a = np.exp(-2.0 * x ** 2) + 0.3 * np.exp(-8.0 * (x - 0.3) ** 2) + 0.3 * np.exp(-8.0 * (x + 0.3) ** 2)
        return a / a.sum()
    else:
        a = rng.uniform(0.1, 1.0, n)
        return a / a.sum()


def resample(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s
