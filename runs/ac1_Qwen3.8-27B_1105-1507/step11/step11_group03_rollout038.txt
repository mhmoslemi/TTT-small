- excess[k] = max(0, conv[k] - target)

Then the "flattening gradient" pushes mass away from positions that contribute to the excess lags.

For lag k with excess, the contributing pairs (i, k-i) tell us which elements of a are "responsible." We can reduce those elements.

This gives a more targeted gradient than the smooth max.

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov accelerated gradient with log-sum-exp smoothing, multiplicative weights,
LP fixed-point iteration, and multi-n exploration.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_max_grad(a, tau):
    """Gradient of (1/tau)*log(sum exp(tau*c[m])) where c=conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    
    shifted = tau * (conv - max_val)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    # cross[j] = sum_m p[m] * a[m-j]
    N = 1
    while N < 4 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real[:n]
    
    grad = cross + a.copy()
    return grad, max_val, p


def nesterov_gd(a, time_budget, tau_start=10, tau_end=5000, max_iters=20000, verbose=False):
    """Nesterov accelerated gradient descent with adaptive step and tau."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive tau
        progress = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Nesterov step
        y = a + beta * v
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy < 1e-10:
            y = best_a.copy()
            v = np.zeros(n)
            sy = 1.0
        y = y / sy
        
        # Gradient
        grad, max_val, _ = smooth_max_grad(y, tau)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        lr_try = lr
        improved = False
        for _ in range(20):
            v_new = beta * v - lr_try * grad / gnorm
            a_try = y + v_new
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                lr_try *= 0.5
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                v = v_new
                a = a_try
                improved = True
                break
            lr_try *= 0.5
        
        if not improved:
            # Accept anyway to maintain momentum
            v = beta * v - lr * grad / gnorm
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            sn = float(np.sum(a_new))
            if sn > 1e-10:
                a = a_new / sn
            else:
                a = best_a.copy()
                v = np.zeros(n)
        
        # Update momentum
        t_k = (1 + np.sqrt(1 + 4 * beta**2)) / 2
        beta_new = (1 - t_k) / (1 + t_k)
        beta = max(0.5, min(0.95, beta_new))
        
        if verbose and it % 500 == 0:
            print(f"    Nesterov it{it}: obj={best_obj:.10f}, tau={tau:.1f}, lr={lr:.6f}")
    
    return best_a


def multiplicative_weights(a, time_budget, verbose=False, max_iters=5000):
    """Multiplicative weights update for non-negative optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = 1.0  # step size
    tau = 50
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        grad, max_val, _ = smooth_max_grad(a, tau)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        # Multiplicative update: a_new[j] = a[j] * exp(-eta * grad[j] / gnorm)
        log_update = -eta * grad / gnorm
        log_update = np.clip(log_update, -10, 10)  # prevent overflow
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-30)
        sn = float(np.sum(a_new))
        if sn < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / sn
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Increase tau gradually
            tau = min(5000, tau * 1.02)
        else:
            # Decrease step size
            eta *= 0.7
            if eta < 0.01:
                eta = 0.5
                tau = min(5000, tau * 1.5)
                # Perturb to escape
                rng = np.random.default_rng(it)
                a = a + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
        
        if verbose and it % 200 == 0:
            print(f"    MW it{it}: obj={best_obj:.10f}, eta={eta:.4f}, tau={tau:.1f}")

def lp_refine(a, time_budget, verbose=False, seed=42, max_iters=1000):
    """LP-based fixed point iteration."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Precompute the convolution matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    
    consecutive_fails = 0
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Update A with current a
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j+n, j] = a
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        solved = False
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g_var.value, 0)
                    sg = float(np.sum(g_val))
                    if sg > 1e-10:
                        g_val = g_val / sg
                        solved = True
                        break
            except Exception:
                continue
        
        if not solved:
            consecutive_fails += 1
            if consecutive_fails > 5:
                break
            continue
        consecutive_fails = 0
        
        # Try to move toward g
        best_local_obj = best_obj
        best_local_a = best_a.copy()
        
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
        
        if best_local_obj < best_obj - 1e-15:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 20:
                # Perturb
                a = best_a + 0.02 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
        
        if verbose and it % 50 == 0:
            print(f"    LP it{it}: obj={best_obj:.10f}")
    
    return best_a


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

def variance_
eduction(a, time_budget):

Let me also add a key innovation: **self-consistent field iteration.**

The idea: if we want conv(a,a) to be flat at value v = 1/(2n-1) (normalized), then at each point j, we want:
sum_m a[m] * a[m-j] = v (approximately)

This is like a self-consistent equation: a[j] should be such that its "response" to the rest of the sequence gives v at every lag.

One way to enforce this: iterate
a_new[j] = f(a[j], conv_target)

where f is chosen to reduce the deviation from the target.

Specifically, if conv[k] > target, we want to reduce the contributions at lag k. The contributions at lag k come from pairs (i, k-i). We can reduce a[i] and a[k-i] for the pairs that contribute most.

This is similar to the "water-filling" algorithm in communications.

Let me implement a simplified version:

```python
def flatten_iteration(a, max_iters=1000):
    """Iteratively flatten the convolution."""
    n = len(a)
    a = a / np.sum(a)
    target = 1.0 / (2*n - 1)
    
    for it in range(max_iters):
        conv = compute_conv(a)
        excess = conv - target
        max_excess = float(np.max(excess))
        if max_excess < 1e-8:
            break
        
        # For each lag with excess, reduce the contributing elements
        # Use the gradient: for lag k, grad[j] = a[k-j]
        # Weighted sum over lags with excess
        weights = np.maximum(excess, 0) ** 2  # quadratic penalty
        # Gradient of the quadratic penalty
        grad = np.zeros(n)
        N = 1
        while N < 4*n:
            N <<= 1
        W = fft(weights, N)
        A = fft(a, N)
        grad = ifft(W * np.conj(A)).real[:n]
        
        # Step
        step = 0.1
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        sn = np.sum(a_new)
        if sn > 1e-10:
            a = a_new / sn
    
    return a
```

This is basically gradient descent on the quadratic penalty of the excess. It should naturally flatten the convolution.

The temperature tau controls the smoothness of the approximation to max.

The key is to increase tau fast enough to make progress on the true objective, but slow enough to maintain progress on the smoothed objective.

A heuristic: increase tau when the smoothed objective decreases, and decrease tau when it doesn't (similar to simulated annealing).
