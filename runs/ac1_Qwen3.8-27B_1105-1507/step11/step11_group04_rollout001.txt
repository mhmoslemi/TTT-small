"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses improved LP alternating with backtracking line search, momentum,
gradient-based refinement via logsumexp smoothing, adaptive coordinate descent,
and multi-restart exploration with diverse perturbation strategies.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass

    for solver_name in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            solver = getattr(cp, solver_name, None)
            if solver is None:
                continue
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def compute_smooth_gradient(a, beta):
    """
    Compute gradient of logsumexp(beta * conv(a,a)) w.r.t. a.
    Uses FFT for efficiency.
    grad_j = 2 * sum_k p_k * a_{k-j} where p_k = softmax(beta*conv_k)
    """
    n = len(a)
    conv = compute_conv(a)
    # Softmax with numerical stability
    conv_clipped = np.clip(conv, -50.0 / beta, 50.0 / beta)
    exp_vals = np.exp(beta * conv_clipped)
    p = exp_vals / np.sum(exp_vals)
    
    # grad_j = 2 * sum_k p_k * a_{k-j}
    # = 2 * cross_correlation(p, a) at lag j
    # Compute via FFT: conv(p, flip(a))[n-1 : 2n-1]
    b = a[::-1].copy()
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fp = rfft(p, size)
    fb = rfft(b, size)
    corr = irfft(fp * fb, size)[:2 * n - 1]
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    beta = 5.0  # Smoothing parameter
    lr = 0.05
    momentum = 0.9
    v = np.zeros(n)
    iters = 0
    
    while time.monotonic() - start < time_budget:

        else:

        if verbose and iters % 50 == 0 and iters > 0:
            elapsed = time.monotonic() - start
            print(f"    GD {iters}: obj={best_obj:.10f}, lr={lr:.6f}, time={elapsed:.1f}s")
    
    return best_a


def coordinate_descent_polish(a, time_budget, seed=42, verbose=True):
    """
    Vectorized coordinate descent on normalized sequence (sum=1).
    Uses adaptive candidate values focused on reducing max convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    sweep = 0
    no_improve = 0
    max_no_improve = 4
    
    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - start > time_budget:
                break
            
            a_j = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)
            
            # Unaffected max
            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))
            
            # Current max for reference
            current_max = float(np.max(conv))
            
            # Adaptive candidates: more values around current, plus some exploration
            if a_j < 1e-12:
                candidates = np.array([0.0, 0.5 / n, 1.0 / n, 1.5 / n, 2.0 / n, 3.0 / n, 5.0 / n])
            else:
                # Fine-grained around current value
                fine = a_j * np.array([0.5, 0.7, 0.85, 0.95, 1.0, 1.05, 1.15, 1.3, 1.5, 2.0, 0.3])
                # Also try values that would reduce the max conv
                # If this coordinate contributes to max, try reducing it
                # If not, try increasing it to spread mass
                mean_val = 1.0 / n
                coarse = np.array([0.0, 0.3 * mean_val, 0.7 * mean_val, 1.0 * mean_val,
                                   1.5 * mean_val, 2.5 * mean_val, 4.0 * mean_val, 0.1 / n])
                candidates = np.concatenate([fine, coarse])
                candidates = candidates[(candidates >= 0) & (candidates <= 0.5)]
                candidates = np.unique(candidates)
            
            best_trial_obj = best_obj
            best_trial_val = a_j
            
            for val in candidates:
                delta = val - a_j
                if abs(delta) < 1e-15:
                    continue
                
                S_new = 1.0 + delta
                if S_new <= 1e-10:
                    continue
                
                indices = np.arange(k_start, k_end + 1)
                affected = conv[k_start:k_end + 1].copy()
                mask = (indices == 2 * j)
                if np.any(~mask):
                    affected[~mask] = conv[k_start:k_end + 1][~mask] + 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    affected[mask] = conv[2 * j] + (val * val - a_j * a_j)
                
                max_conv_unscaled = max(max_unaffected, float(np.max(affected)))
                obj_trial = 2.0 * n * max_conv_unscaled / (S_new * S_new)
                
                if obj_trial < best_trial_obj - 1e-13:
                    best_trial_obj = obj_trial
                    best_trial_val = val
            
            if best_trial_val != a_j:
                delta = best_trial_val - a_j
                a[j] = best_trial_val
                S_new = 1.0 + delta
                
                indices = np.arange(k_start, k_end + 1)
                mask = (indices == 2 * j)
                if np.any(~mask):
                    conv[k_start:k_end + 1][~mask] += 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    conv[2 * j] += (best_trial_val * best_trial_val - a_j * a_j)
                
                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S
                
                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True
        
        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")
    
    return best_a


def lp_alternate(a, time_budget, seed=42, verbose=True):
    """
    LP alternating minimization with backtracking line search,
    momentum, and diverse perturbation strategies.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    iter_count = 0
    stall = 0
    
    # Momentum: keep a running average of recent LP solutions
    momentum_g = None
    momentum_alpha = 0.3
    
    while time.monotonic() - start < time_budget - 2:
        t0 = time.monotonic()
        g_lp, M_lp = solve_lp(a, warm_start=prev_g)
        if g_lp is None:
            break
        prev_g = g_lp
        iter_count += 1
        
        # Update momentum
        if momentum_g is None:
            momentum_g = g_lp.copy()
        else:
            momentum_g = momentum_alpha * momentum_g + (1.0 - momentum_alpha) * g_lp
        
        # Try both g_lp and momentum_g
        candidates_g = [g_lp, momentum_g]
        
        best_local_obj = best_obj
        best_local_a = a.copy()
        
        for g_try in candidates_g:
            # Backtracking line search
            t = 1.0
            for _ in range(12):
                a_try = (1.0 - t) * a + t * g_try
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj_norm(a_try)
                
                if obj_try < best_local_obj - 1e-15:
                    best_local_obj = obj_try
                    best_local_a = a_try.copy()
                    break
                t *= 0.6
        
        if best_local_obj < best_obj - 1e-15:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall = 0
            if verbose and iter_count % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP {iter_count}: obj={best_obj:.10f}, M_lp={M_lp:.6f}, elapsed={elapsed:.0f}s")
        else:
            stall += 1
            if stall >= 5:
                # Diverse perturbation strategies
                strategy = rng.integers(0, 4)
                if strategy == 0:
                    # Fourier perturbation
                    strength = 0.008 + 0.004 * min(stall, 8)
                    pert = np.zeros(n)
                    for k in range(1, 9):
                        freq = k * 2.0 * np.pi / n
                        amp = strength * rng.uniform(0.5, 1.5) / k
                        phase = rng.uniform(0, 2.0 * np.pi)
                        pert += amp * np.cos(freq * np.arange(n) + phase)
                    a = best_a * np.exp(pert)
                elif strategy == 1:
                    # Gaussian perturbation
                    strength = 0.005 + 0.003 * min(stall, 10)
                    a = best_a * (1.0 + strength * rng.standard_normal(n))
                elif strategy == 2:
                    # Block swap
                    a = best_a.copy()
                    n_blocks = rng.integers(3, 8)
                    block_size = n // n_blocks
                    perm = rng.permutation(n_blocks)
                    new_a = np.zeros(n)
                    for i in range(n_blocks):
                        start_i = i * block_size
                        end_i = (i + 1) * block_size if i < n_blocks - 1 else n
                        start_p = perm[i] * block_size
                        end_p = (perm[i] + 1) * block_size if perm[i] < n_blocks - 1 else n
                        new_a[start_i:end_i] = a[start_p:end_p]
                    a = new_a
                else:
                    # Random coordinate emphasis
                    a = best_a.copy()
                    n_emph = n // 10
                    indices = rng.choice(n, size=n_emph, replace=False)
                    a[indices] *= rng.uniform(0.5, 2.0, size=n_emph)
                
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                prev_g = None
                momentum_g = None
                stall = 0
        
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - start
            remaining = time_budget - elapsed
            print(f"    LP iter {iter_count}: best={best_obj:.10f}, M_lp={M_lp:.6f}, rem={remaining:.0f}s")
    
    if verbose:
        print(f"    LP done: {iter_count} iters, best={best_obj:.10f}")
    return best_a


def fourier_perturb(a, rng, strength=0.015):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 10):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def try_length(a_best, n_try, time_budget, seed=42):
    """Try a different sequence length by interpolation and optimize."""
    n_best = len(a_best)
    x_old = np.linspace(0, 1, n_best)
    x_new = np.linspace(0, 1, n_try)
    a_try = np.interp(x_new, x_old, a_best)
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 1e-10:
        return a_best, compute_obj_norm(a_best)
    a_try = a_try / s
    
    # Short LP + CD
    remaining = time_budget
    if remaining > 30:
        a_try = lp_alternate(a_try, time_budget=remaining * 0.55, seed=seed, verbose=False)
        remaining2 = time_budget - (time.monotonic() - (time.monotonic() - remaining))
        # Just use a fraction
        a_try = coordinate_descent_polish(a_try, time_budget=remaining * 0.3, seed=seed + 7, verbose=False)
    
    obj_try = compute_obj_norm(a_try)
    return a_try, obj_try
