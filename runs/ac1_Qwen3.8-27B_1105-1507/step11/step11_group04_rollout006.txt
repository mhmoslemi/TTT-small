```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated LP alternation, smoothed gradient descent,
coordinate descent polishing, and multi-restart with length exploration.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass
    for solver in ['CLARABEL', 'ECOS', 'SCS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate_nesterov(a, time_budget, seed=42, verbose=True):
    """LP alternating with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    momentum = np.zeros(n)
    beta = 0.9
    iter_count = 0
    stall = 0
    t_schedule = [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02]
    
    while time.monotonic() - start < time_budget - 3:
        t0 = time.monotonic()
        g_lp, M_lp = solve_lp(a, warm_start=prev_g)
        if g_lp is None:
            break
        prev_g = g_lp
        iter_count += 1
        
        # Nesterov-style: use momentum in the direction
        direction = g_lp - a
        momentum = beta * momentum + direction
        beta = min(0.99, beta * 1.01)  # slowly increase momentum
        
        # Line search with momentum direction
        best_local_obj = best_obj
        best_local_a = a.copy()
        
        for t in t_schedule:
            a_try = a + t * momentum
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
        
        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall = 0
        else:
            # Fall back to direct LP direction
            best_fb_obj = best_obj
            best_fb_a = a.copy()
            for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
                a_try = (1 - t) * a + t * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj_norm(a_try)
                if obj_try < best_fb_obj - 1e-14:
                    best_fb_obj = obj_try
                    best_fb_a = a_try.copy()
            
            if best_fb_obj < best_obj - 1e-14:
                best_obj = best_fb_obj
                best_a = best_fb_a
                a = best_a.copy()
                stall = 0
            else:
                stall += 1
                if stall >= 5:
                    momentum *= 0.5
                    beta = 0.8
                    # Perturb
                    strength = 0.01 + 0.005 * min(stall, 10)
                    pert = rng.standard_normal(n) * strength
                    a = best_a * np.exp(pert)
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                    else:
                        a = best_a.copy()
                    prev_g = None
                    stall = 0
        
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP-Nest {iter_count}: obj={best_obj:.10f}, M_lp={M_lp:.6f}, elapsed={elapsed:.0f}s")
    
    if verbose:
        print(f"    LP-Nest done: {iter_count} iters, best={best_obj:.10f}")
    return best_a
```

```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated LP alternation, log-sum-exp smoothed mirror descent,
coordinate descent polishing, and systematic length exploration with diverse restarts.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass
    for solver in ['CLARABEL', 'ECOS', 'SCS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate_nesterov(a, time_budget, seed=42, verbose=True):
    """LP alternating minimization with Nesterov-style momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    momentum = np.zeros(n)
    beta = 0.85
    iter_count = 0
    stall = 0

    while time.monotonic() - start < time_budget - 3:
        g_lp, M_lp = solve_lp(a, warm_start=prev_g)
        if g_lp is None:
            break
        prev_g = g_lp
        iter_count += 1

        # Update momentum
        direction = g_lp - a
        momentum = beta * momentum + direction
        beta = min(0.98, beta + 0.005)

        # Try multiple step sizes with momentum direction
        best_local_obj = best_obj
        best_local_a = a.copy()
        t_values = [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.02]

        for t in t_values:
            a_try = a + t * momentum
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()

        # Also try direct LP direction

            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_norm(a_try)

def smoothed_mirror_descent(a, time_budget, seed=42, verbose=True):
    """Mirror descent with log-sum-exp smoothing of the max-convolution objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    tau = 0.08
    eta_base = 0.8
    iter_count = 0
    stall = 0

    while time.monotonic() - start < time_budget - 3:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Softmax weights
        shifted = (conv - max_conv) / tau
        exp_shifted = np.exp(shifted)
        p = exp_shifted / float(np.sum(exp_shifted))

        # Gradient: grad_j = (4n/tau^2) * sum_k p_k * a_{k-j}
        # = (4n/tau^2) * sum_m a_m * p_{j+m} = (4n/tau^2) * correlate(p, a, 'valid')[j]
        grad_raw = np.correlate(p, a, mode='valid')
        grad = (4.0 * n / (tau * tau)) * grad_raw

        # Scale learning rate
        grad_norm = float(np.max(np.abs(grad))) + 1e-10
        eta = eta_base / grad_norm

        # Mirror descent update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj_norm(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                # Perturb and reset
                pert = rng.standard_normal(n) * 0.01
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / float(np.sum(a))
                stall = 0
                tau = max(0.03, tau * 0.8)

        a = a_new
        iter_count += 1
        tau = max(0.02, tau * 0.997)

        if verbose and iter_count % 300 == 0:
            elapsed = time.monotonic() - start
            print(f"    SMD {iter_count}: obj={best_obj:.10f}, tau={tau:.4f}, t={elapsed:.0f}s")

    if verbose:
        print(f"    SMD done: {iter_count} iters, best={best_obj:.10f}")
    return best_a


def coordinate_descent_polish(a, time_budget, seed=42, verbose=True):
    """Coordinate descent with subgradient-guided candidate generation."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    sweep = 0
    no_improve = 0
    max_no_improve = 8
    mean_val = 1.0 / n

    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            a_j = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)

            # Max of unaffected entries
            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))

            # Generate candidates: mix of multiplicative and additive around current
            candidates = np.array([0.0, a_j * 0.3, a_j * 0.6, a_j * 0.85,
                                   a_j * 1.15, a_j * 1.4, a_j * 2.0,
                                   mean_val * 0.2, mean_val * 0.5, mean_val,
                                   mean_val * 2, mean_val * 4, mean_val * 8])
            candidates = candidates[(candidates >= 0) & (candidates <= 0.4)]
            # Add a few random perturbations
            candidates = np.concatenate([candidates,
                                         np.abs(a_j + rng.normal(0, a_j * 0.3 + mean_val * 0.1, 3))])
            candidates = np.unique(candidates[(candidates >= 0) & (candidates <= 0.4)])

            best_trial_obj = best_obj
            best_trial_val = a_j
            S_new_best = 1.0

            for val in candidates:
                delta = val - a_j
                if abs(delta) < 1e-15:
                    continue

                S_new = 1.0 + delta
                if S_new <= 1e-10:
                    continue

                # Compute max of affected conv entries
                indices = np.arange(k_start, k_end + 1)
                affected = conv[k_start:k_end + 1].copy()
                mask = (indices == 2 * j)
                if np.any(~mask):
                    affected[~mask] = conv[k_start:k_end + 1][~mask] + 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    affected[mask] = conv[2 * j] + (val * val - a_j * a_j)

                max_conv_unscaled = max(max_unaffected, float(np.max(affected)))
                obj_trial = 2.0 * n * max_conv_unscaled / (S_new * S_new)

                if obj_trial < best_trial_obj - 1e-13:
                    best_trial_obj = obj_trial
                    best_trial_val = val
                    S_new_best = S_new

            if best_trial_val != a_j:
                delta = best_trial_val - a_j
                a[j] = best_trial_val

                indices = np.arange(k_start, k_end + 1)
                mask = (indices == 2 * j)
                if np.any(~mask):
                    conv[k_start:k_end + 1][~mask] += 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    conv[2 * j] += (best_trial_val * best_trial_val - a_j * a_j)

                inv_S = 1.0 / S_new_best
                a *= inv_S
                conv *= inv_S * inv_S

                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, t={elapsed:.1f}s")

def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation for diverse restarts."""
    n = len(a)
    pert = np.zeros(n)
    num_modes = rng.integers(3, 9)
    for _ in range(num_modes):
        k = rng.integers(1, 12)
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def interpolate_sequence(a, n_new):
    """Interpolate sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(n_new) / n_new
    return a_new / s

    n_options = sorted(set(n_options))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = interpolate_sequence(best_a, n_try)
        alloc = min(remaining * 0.25, 80.0)

        if alloc > 20:
            # Short LP optimization
            a_try = lp_alternate_nesterov(a_try, time_budget=alloc * 0.6,
                                          seed=n_try * 7 + seed, verbose=False)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_try = coordinate_descent_polish(a_try, time_budget=min(remaining * 0.3, 30.0),
                                                  seed=n_try * 13 + seed, verbose=False)

        obj_try = compute_obj_norm(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            n_best = n_try
            best_a = a_try.copy()
            print(f"  ** n={n_try}: {best_obj:.10f} **")
        else:
            print(f"  n={n_try}: {obj_try:.10f}")

    # === Phase 6: Final polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n--- Phase 6: Final Polish ---")
        a_fin = coordinate_descent_polish(best_a, time_budget=min(remaining * 0.7, 60.0),
                                          seed=seed + 42424, verbose=False)
        obj_fin = compute_obj_norm(a_fin)
        if obj_fin < best_obj:
            best_obj = obj_fin
            best_a = a_fin.copy()
            print(f"  Final CD: {best_obj:.10f} **")

        # One more short LP pass
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_fin2 = lp_alternate_nesterov(best_a, time_budget=min(remaining * 0.6, 50.0),
                                           seed=seed + 99999, verbose=False)
            obj_fin2 = compute_obj_norm(a_fin2)
            if obj_fin2 < best_obj:
                best_obj = obj_fin2
                best_a = a_fin2.copy()
                print(f"  Final LP: {best_obj:.10f} **")

        # Final CD
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_fin3 = coordinate_descent_polish(best_a, time_budget=min(remaining * 0.7, 30.0),
                                               seed=seed + 77777, verbose=False)
            obj_fin3 = compute_obj_norm(a_fin3)
            if obj_fin3 < best_obj:
                best_obj = obj_fin3
                best_a = a_fin3.copy()
                print(f"  Final CD2: {best_obj:.10f} **")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    try:
        val = evaluate_sequence(result)
        if val < np.inf:
            best_obj = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
```
