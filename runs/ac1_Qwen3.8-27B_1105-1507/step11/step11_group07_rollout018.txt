def solve_lp_minpeak(a, top_k=None, warm_start=None, verbose=False):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    if top_k is not None and top_k < 2 * n - 1:
        A, _ = build_conv_matrix_reduced(a_norm, top_k)
    else:
        A = build_conv_matrix_full(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if warm_start is not None:
        g.value = warm_start
    
    solvers = [cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g_lp, budget=2.0):

def soft_max_gradient_descent(a, time_budget=200.0, beta_start=10.0, beta_end=60.0,
                              lr_start=0.01, momentum=0.9, verbose=False, seed=42):
    """Soft-max gradient descent with momentum on the simplex."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    it = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress
        lr = lr_start * (1.0 - 0.5 * progress)
        
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient via FFT cross-correlation
        a_rev = a[::-1]
        size = 1
        while size < n + len(p) - 1:
            size <<= 1
        A = rfft(a_rev, size)
        P = rfft(p, size)
        full = irfft(A * P, size)
        q = full[n - 1:n - 1 + n]
        grad = 2.0 * q
        
        v = momentum * v + grad
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall > 150:
                v = np.zeros(n)
                stall = 0
                pert = rng.normal(0, 0.003, n)
                a = project_simplex(best_a + pert)
        
        a = a_new
        it += 1
        
        if verbose and it % 300 == 0:
            print(f"    SmaxGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, t={elapsed:.1f}s")
        
        if it - best_it > 3000 and it > 500:
            a = best_a.copy()
            v = np.zeros(n)
            lr *= 0.7
    
    return best_a, best_obj

            current = a[i]
            # Try: current, current*0.7, current*1.3, current+0.005, current-0.005
            candidates = [current * f for f in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 2.0]]
            candidates += [current + d for d in [-0.01, -0.005, 0.005, 0.01]]
            candidates = [max(0, c) for c in candidates if c >= 0]
            
            for c in candidates:
                a[i] = c
                s = float(np.sum(a))
                if s < 0.01:
                    a[i] = current
                    continue
                o = compute_obj(a / s)
                if o < best_val:
                    best_val = o
                    best_local = c
            a[i] = best_local
        
        # Re-normalize
        a = project_simplex(a)
        o = compute_obj(a)
        if o < best_obj:
            best_obj = o
        else:
            # Restore
            pass
    
    return a, best_obj

def subsymmetric_init(n):
    """U-shaped symmetric initialization."""
    half = (n + 1) // 2

        a = np.concatenate([a_half, a_half[::-1]])

def diverse_init(n, rng, mode=0):

"""
Improved step function optimizer using:
1. Fast LP with progressively increasing active constraints
2. Adaptive soft-max gradient descent with FFT acceleration
3. Efficient coordinate descent for fine-tuning
4. Multi-length exploration
5. Aggressive iterative polishing with warm starts

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix_reduced(a, top_k):
    """Build reduced convolution matrix using only top_k lags."""
    n = len(a)
    conv_len = 2 * n - 1
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:min(top_k, conv_len)]
    top_idx = np.sort(top_idx)
    A = np.zeros((len(top_idx), n))
    for i, k in enumerate(top_idx):
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                A[i, m] = a[j]
    return A, top_idx


def build_conv_matrix_full(a):
    """Build full convolution matrix A (2n-1 x n) where A @ g = conv(a, g)."""
    n = len(a)
    conv_len = 2 * n - 1
    A = np.zeros((conv_len, n))
    for k in range(conv_len):
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                A[k, m] = a[j]
    return A


def solve_lp(a, top_k, warm_start=None):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    if top_k is not None and top_k < 2 * n - 1:
        A, _ = build_conv_matrix_reduced(a_norm, top_k)
    else:
        A = build_conv_matrix_full(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if warm_start is not None:
        g.value = warm_start
    
    solvers = [cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, g, budget=1.5):

        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        o = compute_obj(a_try / s)
        if o < best_obj:
            best_obj = o
            best_t = t
    
    # Fine around best

        lo2 = max(0.0, best_t - 0.15)

            if o < best_obj:
                best_obj = o
                best_t = t
    
    return best_t, best_obj


def soft_max_gd(a, time_budget, beta_start, beta_end, lr_start, momentum, seed, verbose=False):
    """Soft-max gradient descent with momentum on the simplex."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    it = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)
    lr = lr_start
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress
        lr = lr_start * (1.0 - 0.4 * progress)
        
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p = exp_vals / float(np.sum(exp_vals))
        
        a_rev = a[::-1]
        size = 1
        while size < n + len(p) - 1:
            size <<= 1
        A_f = rfft(a_rev, size)
        P_f = rfft(p, size)
        full = irfft(A_f * P_f, size)
        q = full[n - 1:n - 1 + n]
        grad = 2.0 * q
        
        v = momentum * v + grad
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall > 200:
                v = np.zeros(n)
                stall = 0
                a = project_simplex(best_a + rng.normal(0, 0.002, n))
        
        a = a_new
        it += 1
        
        if verbose and it % 400 == 0:
            print(f"    SmaxGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, t={elapsed:.1f}s")
        
        if it - best_it > 4000 and it > 1000:
            a = best_a.copy()
            v = np.zeros(n)
            lr *= 0.6
    
    return best_a, best_obj


def lp_polish_progressive(a, time_budget, max_iters, verbose=False):
    """LP polishing with progressive top_k (start small, grow when stuck)."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    warm_g = a.copy()
    top_k = 15
    rng = np.random.default_rng(555)
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        g_lp, M_lp = solve_lp(a, top_k, warm_start=warm_g)
        if g_lp is None:
            break
        
        warm_g = g_lp.copy()
        t_opt, obj_opt = line_search(a, g_lp, budget=min(1.0, remaining * 0.03))
        
        if t_opt > 0.005 and obj_opt < best_obj - 1e-12:
            a_new = np.maximum((1 - t_opt) * a + t_opt * g_lp, 0)
            a_new = project_simplex(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        it += 1
        if verbose and it % 15 == 0:
            print(f"    LP-prog it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, top_k={top_k}, t={time.monotonic()-start:.1f}s")
        
        if stall >= 3:
            if stall == 3 and top_k < 60:
                # Increase top_k to escape
                top_k = min(top_k + 15, 2 * n - 1)
                stall = 0
            elif stall <= 6:
                noise = rng.normal(0, 0.002 * stall, n)
                a = project_simplex(best_a + noise)
                warm_g = a.copy()
                stall = 0
            else:
                # Aggressive restart
                a = project_simplex(best_a * 0.7 + 0.3 * diverse_init(n, rng, mode=rng.integers(0, 7)))
                warm_g = a.copy()
                top_k = 15
                stall = 0
    
    return best_a, best_obj


def lp_polish_full(a, time_budget, max_iters, verbose=False):
    """Full LP polishing with all constraints."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    warm_g = a.copy()
    rng = np.random.default_rng(999)
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break
        
        g_lp, M_lp = solve_lp(a, None, warm_start=warm_g)
        if g_lp is None:
            break
        
        warm_g = g_lp.copy()
        t_opt, obj_opt = line_search(a, g_lp, budget=min(2.0, remaining * 0.05))
        
        if t_opt > 0.005 and obj_opt < best_obj - 1e-12:
            a_new = np.maximum((1 - t_opt) * a + t_opt * g_lp, 0)
            a_new = project_simplex(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        it += 1
        if verbose and it % 5 == 0:
            print(f"    LP-full it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, t={time.monotonic()-start:.1f}s")
        
        if stall >= 4:
            noise = rng.normal(0, 0.004, n)
            a = project_simplex(best_a + noise)
            warm_g = a.copy()
            stall = 0
    
    return best_a, best_obj


def coordinate_descent_fast(a, budget, n_sweeps, verbose=False):
    """Fast coordinate descent with limited candidates per coordinate."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(4242)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - start > budget:
            break
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.monotonic() - start > budget:
                break
            current = a[i]
            if current < 1e-10:
                continue
            best_local_obj = best_obj
            best_local_val = current
            # Fewer, smarter candidates
            for factor in [0.6, 0.8, 1.2, 1.5]:
                candidate = current * factor
                a[i] = candidate
                s = float(np.sum(a))
                if s < 0.01:
                    a[i] = current
                    continue
                o = compute_obj(a / s)
                if o < best_local_obj - 1e-10:
                    best_local_obj = o
                    best_local_val = candidate
            a[i] = best_local_val
            if best_local_val != current:
                improved = True
        
        a = project_simplex(a)
        o = compute_obj(a)
        if o < best_obj:
            best_obj = o
    
    return a, best_obj


def subsymmetric_init(n):
    """U-shaped symmetric initialization."""
    half = (n + 1) // 2
    x = np.linspace(0, 1, half)
    a_half = 1.0 / (0.02 + 4.0 * (2 * x - 1) ** 4)
    if n % 2 == 0:
        a = np.concatenate([a_half, a_half[::-1]])
    else:
        a = np.concatenate([a_half, a_half[-2::-1]])
    return project_simplex(a)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n, dtype=float)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.03
    elif mode == 1:
        a = (np.exp(-((idx - n // 4) ** 2) / (2 * (n // 8) ** 2)) +
             np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 8) ** 2)) + 0.01)
    elif mode == 2:
        a = 3 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.05 + 6.0 * (2 * x - 1) ** 6)
    elif mode == 6:
        a = 1.0 / np.cosh(5.0 * (2 * x - 1))
    else:
        a = (2 * x - 1) ** 2 * np.exp(-2 * (2 * x - 1) ** 2) + 0.1
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: progressive LP polishing + soft-max GD + coordinate descent + length exploration.
    Uses warm-started LPs with increasing constraint sets for efficiency.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = subsymmetric_init(n_current)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")
    
    # === Phase 1: Quick soft-max GD to potentially escape local minimum ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 1: SoftMax GD (global exploration) ===")
        alloc = min(remaining * 0.10, 100.0)
        a1, obj1 = soft_max_gd(
            best_seq, time_budget=alloc,
            beta_start=3.0, beta_end=20.0,
            lr_start=0.015, momentum=0.89,
            seed=seed, verbose=False
        )
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1 improved: obj={best_score:.8f}")
    
    # === Phase 2: Progressive LP polishing (main workhorse) ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 2: Progressive LP Polish ===")
        alloc = min(remaining * 0.35, 450.0)
        a2, obj2 = lp_polish_progressive(
            best_seq, time_budget=alloc,
            max_iters=500, verbose=True
        )
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 improved: obj={best_score:.8f}")
    
    # === Phase 3: Coordinate descent fine-tuning ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 3: Coordinate Descent ===")
        alloc = min(remaining * 0.08, 60.0)
        a3, obj3 = coordinate_descent_fast(
            best_seq, budget=alloc, n_sweeps=15, verbose=False
        )
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3 improved: obj={best_score:.8f}")
    
    # === Phase 4: Full LP polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: Full LP Polish ===")
        alloc = min(remaining * 0.25, 200.0)
        a4, obj4 = lp_polish_full(
            best_seq, time_budget=alloc,
            max_iters=40, verbose=True
        )
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4 improved: obj={best_score:.8f}")
    
    # === Phase 5: Another round of progressive LP ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 5: Progressive LP Round 2 ===")
        alloc = min(remaining * 0.25, 200.0)
        a5, obj5 = lp_polish_progressive(
            best_seq, time_budget=alloc,
            max_iters=300, verbose=True
        )
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  Phase5 improved: obj={best_score:.8f}")
    
    # === Phase 6: Coordinate descent again ===
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.08, 30.0)
        a6, obj6 = coordinate_descent_fast(best_seq, budget=alloc, n_sweeps=10)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Phase6 (CD): obj={best_score:.8f}")
    
    # === Phase 7: Length exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 70:
        print(f"\n=== Phase 7: Length Exploration ===")
        n_best = len(best_seq)
        
        n_options = sorted(set([
            n_best - 200, n_best - 100, n_best + 100, n_best + 200,
            n_best + 300, n_best - 300,
            600, 900, 1000, 1200
        ]))
        n_options = [n for n in n_options if 300 <= n <= 2500 and abs(n - n_best) >= 80]
        n_options = n_options[:4]
        print(f"  Trying n: {n_options}")
        
        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = project_simplex(np.maximum(a_try, 0))
            
            # Quick LP
            alloc_lp = min(remaining * 0.5, 50.0)
            if alloc_lp > 10:
                a_try, obj_try = lp_polish_progressive(
                    a_try, time_budget=alloc_lp,
                    max_iters=80, verbose=False
                )
            
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")
    
    # === Phase 8: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 8: Final Polish ===")
        alloc = min(remaining * 0.5, 80.0)
        a_f, obj_f = lp_polish_progressive(
            best_seq, time_budget=alloc,
            max_iters=300, verbose=True
        )
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
        
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_f2, obj_f2 = coordinate_descent_fast(
                best_seq, budget=min(remaining * 0.5, 20.0), n_sweeps=8
            )
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
