```python
def press_sorted_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(0, v[0])]) if v[0] > 0 else np.array([1.0])
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
"""
Optimizes step function coefficients using smoothed proximal gradient (FISTA),
exact coordinate descent, and LP-based direction finding.
Target: minimize 2*n*max(conv(a,a))/sum(a)^2 for a >= 0.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return float(2.0 * n * np.max(conv)) / (s * s)


def obj_normalized(a):
    """Compute objective assuming sum(a) = 1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return float(2.0 * n * np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        # All negative, return uniform
        return np.ones(n) / n
    rho_val = idx[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient(a, conv, tau):
    """Compute smoothed gradient of max(conv) using log-sum-exp."""
    n = len(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    # grad[i] = 2 * sum_j p[i+j] * a[j] = 2 * (p ⋆ a)[i]
    N = 1
    while N < (2 * n - 1) + n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    corr = ifft(P * np.conj(A)).real
    grad = 2.0 * corr[:n]
    return grad, p, max_conv


def fista_smoothed(a, time_budget=300.0, tau_init=50.0, tau_end=500.0, verbose=False):
    """FISTA with smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = obj_normalized(a)
    start = time.monotonic()
    
    lr = 1.0
    t = 1.0
    y = a.copy()
    it = 0
    max_iter = 100000
    
    while time.monotonic() - start < time_budget and it < max_iter:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress
        
        # Compute gradient at y
        try:
            conv_y = compute_conv(y)
            grad, p, max_c = smoothed_gradient(y, conv_y, tau)
        except Exception:
            break
        
        # Backtracking line search
        a_new = project_simplex(y - lr * grad)

            obj_new = obj_normalized(a_new)
            # Simple check: if objective improved, accept
            if obj_new < obj_normalized(y) + 1e-15:
                break
            lr *= 0.5
            a_new = project_simplex(y - lr * grad)
        
        # FISTA update
        t_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t * t))
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)

        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = a_new.copy()
        else:
            y = y / s_y
        
        a = a_new
        t = t_new
        it += 1
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        if verbose and it % 1000 == 0:
            print(f"    FISTA it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.4f}")

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a, time_budget=20.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns g."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    start = time.monotonic()
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def lp_line_search(a, g, time_budget=30.0):
    """Find best t in (0,1] such that (1-t)*a + t*g minimizes the objective."""
    n = len(a)
    best_t = 0.0
    best_obj = float('inf')
    
    # Coarse search
    for t in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj = obj_normalized(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    # Fine search around best_t
    if best_t > 0:
        lo = max(0.0, best_t - 0.05)
        hi = min(1.0, best_t + 0.05)
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        
        def eval_t(t):
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                return 1e30
            a_try = a_try / s
            return obj_normalized(a_try)
        
        fc = eval_t(c_pt)
        fd = eval_t(d_pt)
        for _ in range(40):
            if hi - lo < 1e-10:
                break
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                fc = eval_t(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                fd = eval_t(d_pt)
        
        t_opt = (lo + hi) / 2.0
        obj_t = eval_t(t_opt)
        if obj_t < best_obj:
            best_obj = obj_t
            best_t = t_opt
    
    if best_t == 0:
        return a.copy()
    
    a_new = (1.0 - best_t) * a + best_t * g
    a_new = np.maximum(a_new, 0)
    a_new = a_new / float(np.sum(a_new))
    return a_new


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
```
