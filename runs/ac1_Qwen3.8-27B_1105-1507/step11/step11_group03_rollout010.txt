"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lp_best_response(a, solver_list=None):
    """
    Solve: min M s.t. (a*g)_k <= M for all k, g >= 0, sum(g) = 1.
    Returns (g, M, dual_vars) where dual_vars are the shadow prices of the convolution constraints.
    """
    n = len(a)
    # Build constraint matrix: row k, col j has a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    g_val = g_val / sg
                    M_val = float(M_var.value)
                    # Get dual variables
                    dual_vals = None
                    try:
                        # The dual of (A @ g <= M) constraints
                        dual_vals = None  # cvxpy doesn't easily expose these
                    except:
                        pass
                    return g_val, M_val, dual_vals
        except Exception:
            continue
    return None, None, None


def lp_direction_finding(a, p, solver_list=None):
    """
    Find direction d that minimizes p · (a + d * 0 + d) * (a + d) approximately.
    Linearized: min (grad_p · d) s.t. sum(d) = 0, a[i] + d[i] >= 0, |d[i]| <= 1.
    
    Where grad_p[i] = 2 * sum_k p[k] * a[k-i] (cross-correlation of p with a).
    """
    n = len(a)
    
    # Compute gradient: grad[i] = 2 * sum_k p[k] * a[k-i]
    # This is 2 * cross_correlation(p, a) evaluated at index i
    # cross_corr[i] = sum_k p[k] * a[k-i] (where a[k-i] = 0 if out of range)
    grad = np.zeros(n)
    for i in range(n):
        # sum over k where 0 <= k-i < n, i.e., i <= k <= i+n-1
        k_start = max(0, i)
        k_end = min(2 * n - 1, i + n)
        if k_start < k_end:
            # a[k-i] for k in [k_start, k_end)
            # a indices: k - i for k in [k_start, k_end) = [k_start-i, k_end-i)
            # = [max(0,i)-i, min(2n-1,i+n)-i) = [0 or negative, n)
            # Actually: k ranges from i to i+n-1 (clamped to [0, 2n-2])
            valid_k = np.arange(max(0, i), min(2*n-1, i+n))
            valid_a_idx = valid_k - i  # these are in [0, n)
            grad[i] = 2.0 * np.dot(p[valid_k], a[valid_a_idx])
    
    # Solve: min grad · d s.t. sum(d) = 0, -a[i] <= d[i] <= 1
    d_var = cp.Variable(n)
    constraints = [
        cp.sum(d_var) == 0,
        d_var >= -a,
        d_var <= 1.0
    ]
    prob = cp.Problem(cp.Minimize(grad @ d_var), constraints)
    
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.SCS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if d_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.maximum(d_var.value, -a + 1e-10)
        except Exception:
            continue
    return None


def smoothed_grad_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    shifted = tau * (conv - max_val)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    # grad[i] = 2 * sum_k p[k] * a[k-i] (approximately)
    # Use FFT: cross-correlation of p and a
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real[:n]
    g = 2.0 * cross
    return g, float(np.max(conv))

def gd_polish(a, time_budget, tau_levels=None, verbose=False):
    """Smoothed gradient descent with FFT-based gradient computation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [5, 20, 80, 300, 1000, 3000]

    N = 1
    while N < 3 * n:
        N <<= 1

    for tau in tau_levels:
        lr = 1.0 / np.sqrt(n)
        v = np.zeros(n)
        beta = 0.9
        iters = 0
        max_iters = 3000

        while time.monotonic() - start < time_budget:
            if iters >= max_iters:
                break

            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            # Compute gradient
            conv = compute_conv(y)
            max_val = float(np.max(conv))
            shifted = tau * (conv - max_val)
            shifted = np.clip(shifted, -700, 700)
            exp_vals = np.exp(shifted)
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = exp_vals / sum_exp

            P = fft(p, N)
            A = fft(y, N)
            cross = ifft(P * np.conj(A)).real[:n]
            g = 2.0 * cross

            gn = float(np.linalg.norm(g))
            if gn < 1e-15:
                break

            v = beta * v - lr * g / gn
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new
            iters += 1
            lr = max(1e-7, lr * 0.99999)

        if verbose:
            print(f"    GD tau={tau:5d}: obj={best_obj:.10f}, iters={iters}")

    return best_a


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_init(n, mode, rng):
    """Create structured initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        # U-shaped (two peaks at edges)
        x = np.arange(n) / max(n - 1, 1)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * (x - 0.5))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 2:
        # Gaussian
        x = np.linspace(-3, 3, n)
        a = np.exp(-x ** 2)
        return a / a.sum()
    elif mode == 3:
        # Hamming
        x = np.arange(n) / max(n - 1, 1)
        a = 0.54 - 0.46 * np.cos(2 * np.pi * x)
        return a / a.sum()
    elif mode == 4:
        # Blackman
        x = np.arange(n) / max(n - 1, 1)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        return a / a.sum()
    elif mode == 5:
        # Multi-peak
        x = np.arange(n) / n
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x) + 0.1 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 6:
        # Random
        a = rng.uniform(0.1, 1.0, n)
        return a / a.sum()
    elif mode == 7:
        # Flat-top with Gaussian taper
        x = np.arange(n) / n
        taper = 0.15
        a = np.ones(n)
        for i in range(n):
            d = min(i, n - 1 - i) / n
            if d < taper:
                a[i] = np.exp(-((taper - d) / taper) ** 2)
        return a / a.sum()
    elif mode == 8:
        # Sinc-like (clipped)
        x = np.linspace(-1, 1, n)
        a = np.sinc(x)
        a = np.maximum(a, 0.005)
        return a / a.sum()
    else:
        a = rng.uniform(0.5, 1.5, n)
        return a / a.sum()
