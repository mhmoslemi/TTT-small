"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses exact coordinate descent with proper unaffected-max tracking, LP-based
polishing with adaptive line search, subgradient escape, and systematic n-exploration.
Key improvements: exact CD (tracks global max correctly), binary-search LP step size,
wider n range, and iterative CD-LP-alternation for deeper convergence.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def obj_normalized(a):
    """Objective for sum-normalized sequence: 2n * max(conv)."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def obj_general(a):
    """Objective for arbitrary non-negative sequence."""
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    a_n = a / S
    return obj_normalized(a_n)


def coordinate_descent(a, time_budget=300.0, seed=42, verbose=False):
    """
    Exact coordinate descent with golden section search.
    Properly tracks the max over ALL convolution positions (both affected and unaffected).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0
    prev_best = best_obj
    stall = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in range(n):
            idx = order[i]
            if time.monotonic() - start >= time_budget:
                break

            v = a[idx]
            S_other = S - v
            if S_other < 1e-14 and v < 1e-14:
                continue

            # Affected conv positions: k in [idx, min(2n-2, idx+n-1)]
            k_lo = idx
            k_hi = min(2 * n - 2, idx + n - 1)
            quad_k = 2 * idx

            # Pre-compute data for the 1D subproblem
            ks_all = np.arange(k_lo, k_hi + 1)
            mask_non_quad = (ks_all != quad_k)
            ks_lin = ks_all[mask_non_quad]
            c_lin = conv[ks_lin].copy()
            # lin_coeff[k] = 2 * a[k - idx]
            lin_coeff = 2.0 * a[ks_lin - idx]
            quad_in_range = (k_lo <= quad_k <= k_hi)
            c_quad = float(conv[quad_k]) if quad_in_range else 0.0

            # Unaffected max
            if k_lo > 0:
                unaffected_left = conv[:k_lo]
            else:
                unaffected_left = np.array([])
            if k_hi < 2 * n - 2:
                unaffected_right = conv[k_hi + 1:]
            else:
                unaffected_right = np.array([])

            if len(unaffected_left) > 0 or len(unaffected_right) > 0:
                parts = [p for p in [unaffected_left, unaffected_right] if len(p) > 0]
                unaffected_max = float(np.max(np.concatenate(parts)))
            else:
                unaffected_max = 0.0

            # Define 1D objective
            def f1d(x, v=v, S_other=S_other, n=n, c_lin=c_lin,
                    lin_coeff=lin_coeff, quad_in_range=quad_in_range,
                    c_quad=c_quad, unaffected_max=unaffected_max):
                S_new = S_other + x
                if S_new < 1e-30:
                    return 1e30
                if len(c_lin) > 0:
                    vals = c_lin + lin_coeff * (x - v)
                    max_lin = float(np.max(vals))
                else:
                    max_lin = 0.0
                if quad_in_range:
                    val_q = c_quad + x * x - v * v
                else:
                    val_q = 0.0
                max_aff = max(max_lin, val_q)
                max_all = max(max_aff, unaffected_max)
                return 2.0 * n * max_all / (S_new * S_new)

            # Golden section search
            scale = max(v, 1.0 / n, 1e-5)
            x_lo = 0.0
            x_hi = v + 3.0 * scale
            if x_hi < 1e-10:
                x_hi = 1e-5

            # Quick candidate check to narrow the bracket
            candidates = [0.0, v * 0.5, v, v * 1.5, 1.0 / (2 * n), 1.0 / n, 2.0 / n, 4.0 / n]
            best_fx = f1d(v)
            best_xc = v
            for xc in candidates:
                fx = f1d(xc)
                if fx < best_fx:
                    best_fx = fx
                    best_xc = xc

            # Refine bracket around best candidate
            span = max(scale, abs(best_xc) * 0.5)
            x_lo = max(0.0, best_xc - span)
            x_hi = best_xc + 2.0 * span

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f1d(c_pt)
            fd = f1d(d_pt)

            for _ in range(45):
                if x_hi - x_lo < 1e-13 * max(1.0, abs(x_hi)):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f1d(d_pt)

                delta = x_opt - v
                a[idx] = x_opt
                # Update conv in-place
                for k in range(k_lo, k_hi + 1):
                    if k == quad_k:
                        conv[k] += x_opt * x_opt - v * v
                    else:
                        j = k - idx
                        if 0 <= j < n:
                            conv[k] += 2.0 * delta * a[j]
                S += delta

        # Periodic verification and best tracking
        if sweep % 3 == 0:
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                a_n = a / S_check
                obj_check = 2.0 * n * float(np.max(compute_conv(a_n)))
                if obj_check < best_obj - 1e-14:
                    best_obj = obj_check
                    best_a = a_n.copy()
                    a = a_n.copy()
                    conv = compute_conv(a)
                    S = 1.0
                    stall = 0
                else:
                    # Re-sync
                    a_n = a / S_check
                    a = a_n
                    conv = compute_conv(a)
                    S = 1.0
                    stall += 1

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g) for fixed a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = min(2 * n - 1, j + n)
        length = end - start
        if length > 0:
            A[start:end, j] = a[:length]
    return A


def solve_lp(a, verbose=False):
    """Solve LP: min M s.t. A@b <= M, sum(b)=1, b>=0, where A=build_conv_matrix(a)."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return None
    a_n = a / S
    A = build_conv_matrix(a_n)

    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ b <= M, cp.sum(b) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.SCP]:
        try:
            prob.solve(solver=solver)
            if b.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(b.value, 0)
                s_b = float(np.sum(b_val))
                if s_b > 1e-10:
                    b_val = b_val / s_b
                    return b_val
        except Exception:
            continue
    return None


def lp_polish(a, time_budget=120.0, seed=42, verbose=False):
    """LP-based polishing with binary search step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        g = solve_lp(a)
        if g is None:
            break

        # Binary search for best step size t
        best_t = 0.0
        best_t_obj = best_obj
        lo, hi = 0.0, 1.0
        for _ in range(8):
            mid = (lo + hi) / 2.0
            a_try = (1.0 - mid) * a + mid * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                hi = mid
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = mid
            if obj_try < 2.0 * n * float(np.max(compute_conv(a))):
                lo = mid
            else:
                hi = mid

        if best_t_obj < best_obj - 1e-13 and best_t > 0:
            a_try = (1.0 - best_t) * a + best_t * g
            a_try = np.maximum(a_try, 0)
            a = a_try / a_try.sum()
            best_obj = best_t_obj
            best_a = a.copy()
            stall = 0
        else:
            # Try smaller fixed steps
            improved = False
            for t in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:
                a_try = (1.0 - t) * a + t * g
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
                if obj_try < best_obj - 1e-13:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                    break
            if not improved:
                stall += 1
                if stall >= 3:
                    # Perturb and reset
                    pert = rng.normal(0, 0.03, n)
                    a = a * np.exp(pert)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    stall = 0

        if verbose:
            elapsed = time.monotonic() - start
            print(f"    LP: obj={best_obj:.8f}, t={elapsed:.1f}s, stall={stall}")

    return best_a


def subgradient_escape(a, time_budget=60.0, seed=42):
    """Subgradient-based escape from local minima using smoothed max."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    for tau in [10, 30, 50, 100, 200, 400]:
        while time.monotonic() - start < time_budget:
            elapsed = time.monotonic() - start
            if elapsed / time_budget > 0.8:
                break

            conv = compute_conv(a)
            max_c = float(np.max(conv))
            exp_vals = np.exp(tau * (conv - max_c))
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = exp_vals / sum_exp

            # Gradient of smoothed objective w.r.t. a
            # d/d(a[i]) sum_k p[k] * conv[k]
            # = sum_k p[k] * (a[k-i] + a[i]) for valid indices (with quadratic correction)
            # Use FFT: gradient = 2 * (a * p_conv) truncated
            size = 1
            while size < 3 * n:
                size <<= 1
            P = rfft(p, size)
            A_fft = rfft(a, size)
            grad_full = irfft(P * A_fft, size)
            grad = 2.0 * grad_full[:n]

            g_norm = float(np.linalg.norm(grad))
            if g_norm < 1e-15:
                break

            lr = 50.0 / (n * g_norm + 1e-10)
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                break
            a_new = a_new / s_new

            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations for given length n."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Smooth bump
        a = np.exp(-5.0 * (x - 0.5) ** 2)
        a = a / a.sum()
    elif mode == 3:
        # Two bumps
        a = np.exp(-10.0 * (x - 0.25) ** 2) + np.exp(-10.0 * (x - 0.75) ** 2)
        a = a / a.sum()
    elif mode == 4:
        # Cosine window
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        a = (x + 0.01) ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        # Anti-symmetric-ish (front-heavy)
        a = np.exp(-x * rng.uniform(1, 5))
        a = a / a.sum()
    elif mode == 7:
        # Random smooth
        freqs = rng.uniform(0.5, 3, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

                best_seq = a_f.copy()
                print(f"  Final CD {final_round}: obj={best_score:.8f} **")

        # LP
        remaining = deadline - time.monotonic()
        if remaining > 10:
            lp_alloc = min(remaining * 0.25, 40.0)
            if lp_alloc > 8:
                a_f2 = lp_polish(best_seq, time_budget=lp_alloc,
                                 seed=9500 + final_round * 11, verbose=False)
                obj_f2 = obj_normalized(a_f2)
                if obj_f2 < best_score:
                    best_score = obj_f2
                    best_seq = a_f2.copy()
                    print(f"  Final LP {final_round}: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(200) / 200.0
        best_score = 2.0

    best_seq = np.maximum(best_seq, 0.0)
    result = [float(min(1000.0, max(0.0, x))) for x in best_seq]

    # Validate
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
