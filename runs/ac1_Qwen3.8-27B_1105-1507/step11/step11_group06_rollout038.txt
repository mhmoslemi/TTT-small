"""
LP-based optimization with improved initialization, adaptive line search,
softmax-parameterized gradient refinement, and aggressive n exploration.
Key improvements: scipy minimize_scalar for line search, softmax GD as
alternative refinement, better U-shape parameter sweep, and n sweeping.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build the linear operator that maps g -> a * g (convolution)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # (a * g)[k] = sum_j a[k-j] * g[j]
        # For a fixed j, a[k-j] is nonzero when 0 <= k-j < n, i.e., j <= k < j+n
        # So A[k, j] = a[k-j] for k in [j, j+n-1] (clipped to [0, 2n-2])
        k_start = j
        k_end = min(2 * n - 1, j + n)
        for k in range(k_start, k_end):
            A[k, j] = a[k - j]
    return A


def solve_lp_direction(a):
    """Solve LP to find best direction g given current a."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    best_M = None
    best_g = None
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    if best_M is None or M_val < best_M:
                        best_M = M_val
                        best_g = g_val
        except Exception:
            continue
    return best_g, best_M


def line_search(a, g, current_obj):
    """Find optimal t in [0, 1.5] to minimize obj((1-t)*a + t*g)."""
    n = len(a)
    
    def obj_at_t(t):
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return 1e10
        a_try = a_try / s
        return compute_obj(a_try)
    
    # Coarse grid first
    ts = np.linspace(0, 1.5, 30)
    best_t = 0.0
    best_val = current_obj
    for t in ts:
        v = obj_at_t(t)
        if v < best_val:
            best_val = v
            best_t = t
    
    # Fine search around best
    if 0 < best_t < 1.5:
        lo = max(0, best_t - 0.1)
        hi = min(1.5, best_t + 0.1)
        res = minimize_scalar(obj_at_t, bounds=(lo, hi), method='bounded',
                             options={'xatol': 1e-6})
        if res.success and res.fun < best_val:
            best_val = res.fun
            best_t = res.x
    
    return best_t, best_val


def softmax_gradient_step(a, lr, momentum_buf, beta):
    """
    One step of momentum gradient descent in the softmax parameterization.
    a is the current sequence (sum=1, all positive).
    Returns updated a.
    """
    n = len(a)
    conv = compute_conv(a)
    k_max = int(np.argmax(conv))
    
    # Gradient of max_k (a*a)[k] w.r.t. a:
    # grad[i] = a[k_max - i] + a[i - k_max] (with 0 for out of bounds)
    grad = np.zeros(n)
    for i in range(n):
        j1 = k_max - i
        j2 = i - k_max
        if 0 <= j1 < n:
            grad[i] += a[j1]
        if 0 <= j2 < n:
            grad[i] += a[j2]
    
    # Project gradient onto the tangent space of the simplex:
    # Subtract the mean to ensure sum of updates is 0 (preserves sum=1)
    grad_centered = grad - np.mean(grad)
    
    # Momentum
    momentum_buf = beta * momentum_buf - lr * grad_centered
    
    a_new = a + momentum_buf
    a_new = np.maximum(a_new, 1e-10)  # Keep positive
    a_new = a_new / a_new.sum()
    
    return a_new, momentum_buf


def softmax_momentum_descent(a, time_budget=30.0):
    """
    Momentum gradient descent in softmax parameterization for the smoothed problem.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    momentum_buf = np.zeros(n)
    beta = 0.9
    lr = 0.01
    it = 0
    
    while time.monotonic() - start < time_budget:
        a, momentum_buf = softmax_gradient_step(a, lr, momentum_buf, beta)
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        it += 1
        # Adaptive lr
        if it % 100 == 0:
            lr = max(0.001, lr * 0.95)
    
    return best_a, best_obj


def u_shaped_init(n, sharpness):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.01 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n, dtype=np.float64)
    if mode == 0:
        a = 1.0 / (0.01 + 3.0 * (2 * x - 1) ** 2)
    elif mode == 1:
        a = np.exp(-3 * (2 * x - 1) ** 2) + 0.3 * np.exp(-((2 * x - 1) ** 2) / 0.05)
    elif mode == 2:
        a = (2 * x - 1) ** 4 + 0.1
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.005 + 10.0 * (2 * x - 1) ** 4)
    elif mode == 6:
        # Two peaks
        a = np.exp(-((idx - n * 0.15) ** 2) / (n * 0.05)) + \
            np.exp(-((idx - n * 0.85) ** 2) / (n * 0.05)) + 0.05
    else:
        a = 1.0 / (0.1 + 2.0 * np.abs(2 * x - 1))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def lp_polish(a, time_budget=100.0, max_iters=200, verbose=False, seed=42):
    """
    LP-based iterative refinement.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        T0 = time.monotonic()
        g_lp, M_lp = solve_lp_direction(a)
        lp_time = time.monotonic() - T0
        
        if g_lp is None:
            break
        
        t_opt, obj_opt = line_search(a, g_lp, best_obj)
        
        improved = False
        if t_opt > 1e-6 and obj_opt < best_obj - 1e-12:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    stall = 0
                    improved = True
        
        if not improved:
            # Try direct replacement
            obj_direct = compute_obj(g_lp)
            if obj_direct < best_obj:
                best_obj = obj_direct
                best_a = g_lp.copy()
                a = g_lp.copy()
                stall = 0
                improved = True
        
        if not improved:
            stall += 1
        
        it += 1
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                  f"lp_time={lp_time:.2f}s, elapsed={elapsed:.1f}s, stall={stall}")
        
        if stall >= 3:
            if stall <= 8:
                # Perturb
                freq = rng.integers(1, max(2, n // 100))
                phase = rng.uniform(0, 2 * np.pi)
                pert = 0.02 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 0.01:
                    a = a / s
                else:
                    a = best_a.copy()
            else:
                # Restart with diverse init
                mode = it % 7
                a_div = diverse_init(n, rng, mode=mode)
                blend = 0.5 if stall < 15 else 0.7
                a = (1.0 - blend) * a_div + blend * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()
            stall = 0
    
    return best_a


def coordinate_descent_polish(a, time_budget=30.0, verbose=False):
    """
    Coordinate descent: for each coordinate, do a 1D optimization.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        improved_any = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            old_val = a[i]
            # Try a range of values for coordinate i
            # The optimal value is typically near old_val
            candidates = [0.0, old_val * 0.5, old_val * 0.75, old_val,
                         old_val * 1.25, old_val * 1.5, old_val * 2.0]
            best_cand = old_val
            best_cand_obj = best_obj
            for cv in candidates:
                if cv < 0:
                    continue
                a_test = a.copy()
                a_test[i] = cv
                s = float(np.sum(a_test))
                if s < 0.01:
                    continue
                a_test = a_test / s
                obj = compute_obj(a_test)
                if obj < best_cand_obj:
                    best_cand_obj = obj
                    best_cand = cv
            if best_cand != old_val:
                a[i] = best_cand
                s = float(np.sum(a))
                if s > 0.01:
                    a = a / s
                    best_obj = compute_obj(a)
                    improved_any = True
        
        if not improved_any:
            break
    
    return a, best_obj


def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s
