from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def simplex_project(v):
    """Project v onto {x : x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    a = np.maximum(a, 0).astype(np.float64).copy()

        return a

    best_obj = compute_obj(a)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1 / np.sqrt(n)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    while time.monotonic() - start < time_budget:
        # Compute convolution
        conv = compute_conv(a)
        max_conv = float(np.max(conv))

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def compute_gradient(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) via log-sum-exp."""
    n = len(a)

    max_conv = float(np.max(conv))

    shifted = tau * (conv - max_conv)

    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real

"""
Optimizes step function coefficients using a hybrid Adam + CD + peak-directed approach.
Key differences from previous SOTA:
- Adam optimizer (vs Nesterov) for better convergence on non-smooth landscape
- Adaptive smoothing parameter tau that increases over time
- Cycle method: interleaves gradient steps with CD sweeps
- Peak-directed coordinate scaling (new)
- No LP solvers (faster, more reliable)
- Aggressive multi-length exploration
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2*n*max(conv)/sum^2 (scale-invariant)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def compute_gradient_smoothed(a, tau):
    """Compute smoothed gradient via log-sum-exp. Returns (grad, conv, p, max_conv)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    grad = 2.0 * g_full[:n]
    return grad, conv, p, max_conv


def eval_cd_1d(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i):
    """Evaluate 1D objective for a single coordinate in CD."""
    if len(a_ki_lin) > 0:
        lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
        max_lin = float(np.max(lin_vals))
    else:
        max_lin = 0.0
    quad_val = c_2i + x * x - v * v
    max_val = max(max_lin, quad_val)
    denom = (S_other + x) * (S_other + x)
    if denom < 1e-30:
        return 1e30
    return 2.0 * n * max_val / denom


def golden_section_search(f, x_lo, x_hi, max_iter=35):
    """Golden section minimization on [x_lo, x_hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if x_hi - x_lo < 1e-14:
        return (x_lo + x_hi) / 2.0
    
    c_pt = x_hi - gr * (x_hi - x_lo)
    d_pt = x_lo + gr * (x_hi - x_lo)
    fc = f(c_pt)
    fd = f(d_pt)
    
    for _ in range(max_iter):
        if x_hi - x_lo < 1e-13 * max(1.0, abs(x_hi)):
            break

            d_pt = c_pt
            fd = fc

            fc = f(c_pt)
        else:
            x_lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = x_lo + gr * (x_hi - x_lo)
            fd = f(d_pt)
    
    return max(0.0, (x_lo + x_hi) / 2.0)
