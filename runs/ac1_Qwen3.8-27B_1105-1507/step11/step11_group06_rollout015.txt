Uses Nesterov-accelerated multiplicative weight updates on a smoothed objective,
followed by exact LP refinement and multi-n exploration.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def convolve_fft(a):
    n = len(a)
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = convolve_fft(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_smooth_gradient(w, a):
    """Compute grad_j = sum_k w[k] * a[k-j] using FFT (cross-correlation)."""

    A_rev = rfft(a_rev, size)

    grad = e[n - 1:2 * n - 1]

    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]

def solve_lp(a, verbose=False):
    """Solve the LP: min M s.t. A@g <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS, cp.SCS]:

            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:

    return None, None

        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    if 0.0 < best_t < t_max:
        lo = max(0.0, best_t - 0.03)
        hi = min(t_max, best_t + 0.03)
        ts_fine = np.linspace(lo, hi, n_fine)
        for t in ts_fine:
            a_try = (1.0 - t) * a + t * direction
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t
    return best_t, best_obj


def lp_polish(a, time_budget=100.0, max_iters=80, verbose=False):
    """Iterative LP-based polishing with line search and perturbations."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(12345 + it)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        T0 = time.monotonic()
        g_lp, M_lp = solve_lp(a)
        lp_time = time.monotonic() - T0

        if g_lp is None:
            break

        t_opt, obj_opt = exhaustive_line_search(a, g_lp, t_max=1.5, n_coarse=40, n_fine=30)
        if t_opt > 1e-6 and obj_opt < best_obj - 1e-14:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            obj_direct = compute_obj(g_lp)
            if obj_direct < best_obj:
                best_obj = obj_direct
                best_a = g_lp.copy()
                a = g_lp.copy()
                stall = 0

        it += 1
        if verbose and it % 3 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                  f"lp_time={lp_time:.2f}s, elapsed={elapsed:.1f}s")

        if stall >= 3:
            if stall <= 5:
                freq = rng.integers(1, max(2, n // 100))
                phase = rng.uniform(0, 2 * np.pi)
                idx = np.arange(n, dtype=np.float64)
                pert = 0.02 * np.cos(2 * np.pi * freq * idx / n + phase)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 0.01:
                    a = a / s
                else:
                    a = best_a.copy()
            else:
                # Larger restart
                a = best_a.copy()
                idx = rng.integers(0, n, size=max(1, n // 10))
                a[idx] *= rng.uniform(0.3, 1.5, size=len(idx))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 0.01:
                    a = a / s
                else:
                    a = best_a.copy()
            stall = 0

    return best_a, best_obj


def diversify_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.03
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * (n // 5) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 5) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.03
    elif mode == 3:
        a = rng.uniform(0.05, 1.0, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.01 + 8.0 * (2 * x - 1) ** 4)
    elif mode == 6:
        # Two-sided peaks with valley
        mid = n // 2
        a = np.exp(-2.0 * ((x - 0.3) ** 2) / (0.1 ** 2)) + \
            np.exp(-2.0 * ((x - 0.7) ** 2) / (0.1 ** 2)) + 0.05
    elif mode == 7:
        # Broad U-shape
        a = 0.5 + 1.5 * (2 * x - 1) ** 2
    else:
        a = 1.0 / (0.05 + 3.0 * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    if n_src < 2:
        return None
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s


def multiplicative_weight_update(a, n_iter=200, time_budget=60.0, eta=0.1, verbose=False):
    """
    Multiplicative weight / exponential tilt method.
    At each step, identify the peak of the convolution and reduce its contribution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(n_iter):
        if time.monotonic() - start > time_budget:
            break

        conv = convolve_fft(a)
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]

        # Multiplicative weight: reduce a[j] proportionally to its contribution to k_max
        # Contribution of a[j] to conv[k_max] is a[k_max - j] (if valid)
        contrib = np.zeros(n)
        for j in range(n):
            jm = k_max - j
            if 0 <= jm < n:
                contrib[j] = a[jm]
            jm2 = j - k_max
            if 0 <= jm2 < n:
                contrib[j] = a[jm2]

        # Update: a_j *= exp(-eta * contrib_j / max_val)
        # This reduces elements that contribute most to the peak
        scale = eta * contrib / (max_val + 1e-10)
        a = a * np.exp(-scale)
        a = np.maximum(a, 1e-10)
        s = float(np.sum(a))
        a = a / s

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        if verbose and it % 50 == 0:
            print(f"    MW it={it}: obj={best_obj:.8f}, eta_used={eta:.4f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main optimization pipeline:
    1. Load parent / diverse initialization
    2. Smooth gradient descent (log-sum-exp)
    3. Multiplicative weight refinement
    4. LP polish
    5. n-exploration with LP
    6. Final combined polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    # === Load parent ===
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        s = float(np.sum(parent_arr))
        if s > 0.01:
            best_seq = parent_arr / s
            best_score = compute_obj(best_seq)
            print(f"Parent loaded: n={parent_arr.size}, obj={best_score:.8f}")
        else:
            raise ValueError("Sum too small")
    except Exception:
        best_seq = diversify_init(800, rng, mode=0)
        best_score = compute_obj(best_seq)
        print(f"Default init: n={len(best_seq)}, obj={best_score:.8f}")

    # Try alternative initializations at same n
    n_best = len(best_seq)
    for mode in range(8):
        a_alt = diversify_init(n_best, rng, mode=mode)
        obj_alt = compute_obj(a_alt)
        if obj_alt < best_score:
            best_score = obj_alt
            best_seq = a_alt.copy()
            print(f"  Alt init mode={mode}: obj={best_score:.8f} **")

    # === Phase 1: Smooth gradient descent ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 1: Smooth Gradient (n={n_best}, budget={min(remaining * 0.35, 300):.0f}s) ===")
        alloc = min(remaining * 0.35, 300.0)
        a_sgd, obj_sgd = smooth_gradient_descent(best_seq, time_budget=alloc,
                                                  beta_start=5.0, beta_end=800.0,
                                                  n_warmup=80, verbose=True)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase 1 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 done: obj={obj_sgd:.8f} (no improvement)")

    # === Phase 2: Multiplicative weight refinement ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 2: Multiplicative Weight ===")
        alloc = min(remaining * 0.1, 60.0)
        # Try different eta values
        for eta in [0.05, 0.1, 0.2, 0.4]:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            a_mw, obj_mw = multiplicative_weight_update(best_seq, n_iter=500,
                                                        time_budget=alloc / 4, eta=eta)
            if obj_mw < best_score:
                best_score = obj_mw
                best_seq = a_mw.copy()
                print(f"  MW (eta={eta}): obj={best_score:.8f} **")

    # === Phase 3: LP Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 3: LP Polish (budget={min(remaining * 0.25, 200):.0f}s) ===")
        alloc = min(remaining * 0.25, 200.0)
        a_lp, obj_lp = lp_polish(best_seq, time_budget=alloc, max_iters=100, verbose=True)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase 3 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3 done: obj={obj_lp:.8f} (no improvement)")

    # === Phase 4: Another smooth gradient pass (with higher beta) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 4: Fine-tune SGD ===")
        alloc = min(remaining * 0.15, 80.0)
        a_sgd2, obj_sgd2 = smooth_gradient_descent(best_seq, time_budget=alloc,
                                                    beta_start=50.0, beta_end=2000.0,
                                                    n_warmup=30, verbose=False)
        if obj_sgd2 < best_score:
            best_score = obj_sgd2
            best_seq = a_sgd2.copy()
            print(f"  Phase 4: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4: obj={obj_sgd2:.8f} (no improvement)")

    # === Phase 5: n Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 5: n Exploration ===")
        n_cur = len(best_seq)
        n_options = sorted(set([
            n_cur - 100, n_cur - 50, n_cur - 200,
            n_cur + 50, n_cur + 100, n_cur + 200, n_cur + 400,
            500, 600, 700, 900, 1000, 1100, 1200, 1400, 1600, 2000
        ]))
        n_options = [n for n in n_options if 200 <= n <= 3000 and abs(n - n_cur) >= 20]
        n_options.sort(key=lambda x: abs(x - n_cur))
        n_options = n_options[:8]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            a_try = interpolate_sequence(best_seq, n_try)
            if a_try is None:
                continue
            # Quick smooth gradient
            alloc_sg = min(remaining * 0.2, 40.0)
            if alloc_sg > 10:
                a_try, _ = smooth_gradient_descent(a_try, time_budget=alloc_sg,
                                                    beta_start=10.0, beta_end=300.0,
                                                    n_warmup=30)
            # Quick LP
            remaining = deadline - time.monotonic()
            if remaining > 20:
                alloc_lp = min(remaining * 0.3, 50.0)
                a_try, obj_try = lp_polish(a_try, time_budget=alloc_lp, max_iters=40)
            else:
                obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 6: Diverse Restarts ===")
        n_cur = len(best_seq)
        for r in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            mode = (r + 3) % 8
            a_r = diversify_init(n_cur, rng, mode=mode)
            # Blend with current best
            blend = 0.4
            a_r = (1.0 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
            alloc = min(remaining * 0.25, 50.0)
            if alloc > 15:
                a_r, obj_r = smooth_gradient_descent(a_r, time_budget=alloc * 0.4,
                                                      beta_start=10, beta_end=200, n_warmup=20)
                remaining = deadline - time.monotonic()
                if remaining > 15:
                    a_r, obj_r = lp_polish(a_r, time_budget=remaining * 0.3, max_iters=20)
            else:
                obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive LP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 7: Final Polish ===")
        alloc = min(remaining - 5, 100.0)
        if alloc > 10:
            a_final, obj_final = lp_polish(best_seq, time_budget=alloc, max_iters=60, verbose=True)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final: obj={best_score:.8f} **")
            else:
                print(f"  Final: obj={obj_final:.8f}")

    # === Phase 8: Last squeeze ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print(f"\n=== Phase 8: Last Squeeze ===")
        a_sq, obj_sq = smooth_gradient_descent(best_seq, time_budget=min(remaining * 0.4, 30.0),
                                                beta_start=100.0, beta_end=3000.0, n_warmup=20)
        if obj_sq < best_score:
            best_score = obj_sq
            best_seq = a_sq.copy()
            print(f"  Squeeze: obj={best_score:.8f} **")
        remaining2 = deadline - time.monotonic()
        if remaining2 > 10 and best_seq is not None:
            a_sq2, obj_sq2 = lp_polish(best_seq, time_budget=min(remaining2 - 3, 30.0), max_iters=15)
            if obj_sq2 < best_score:
                best_score = obj_sq2
                best_seq = a_sq2.copy()
                print(f"  Squeeze2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
