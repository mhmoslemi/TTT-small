"""
Optimizes step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses variance-based gradient descent, progressive log-sum-exp smoothing,
Barzilai-Borwein step sizes, frequency-domain equalization, and multi-n exploration.
"""

import numpy as np
import time
from scipy.fft import fft, ifft, rfft, irfft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def gradient_sliding_dot(weight, a):
    """
    Compute g[i] = sum_{j=0}^{n-1} weight[i+j] * a[j] for i=0,...,n-1.
    Uses FFT for efficiency.
    weight: length 2n-1
    a: length n
    Returns: length n
    """
    n = len(a)
    a_flip = a[::-1]
    # Need N >= (2n-1) + (n-1) = 3n-2
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    W = fft(weight, N)
    A = fft(a_flip, N)
    result = ifft(W * A, N).real
    # g[i] = result[i + n - 1]
    return result[n-1 : n-1+n]

def grad_variance(a):
    """Gradient of sum((c[k]-mu)^2) where c=conv(a,a), mu=mean(c)."""
    n = len(a)
    c = compute_conv(a)
    s = float(np.sum(a))
    mu = s * s / (2 * n - 1)
    d = c - mu
    g = 4.0 * gradient_sliding_dot(d, a)
    return g

def grad_smoothed_max(a, tau):
    """Gradient of (1/tau)*log(sum(exp(tau*c[k]))) where c=conv(a,a)."""
    n = len(a)
    c = compute_conv(a)
    c_max = float(np.max(c))
    shifted = tau * (c - c_max)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    p = exp_vals / float(np.sum(exp_vals))
    g = 2.0 * gradient_sliding_dot(p, a)
    return g, c_max

def project_simplex(a):
    """Project a onto the simplex {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s

def bb_step_size(g_new, g_old, a_new, a_old, step_prev):
    """Barzilai-Borwein step size."""
    s_vec = a_new - a_old
    y_vec = g_new - g_old
    sy = float(np.dot(s_vec, y_vec))
    ss = float(np.dot(s_vec, s_vec))
    yy = float(np.dot(y_vec, y_vec))
    if ss < 1e-30:
        return step_prev
    if yy < 1e-30:
        return step_prev
    if sy <= 0:
        return step_prev * 0.5
    # BB1: <s,s>/<s,y>, BB2: <s,y>/<y,y>
    bb1 = ss / sy
    bb2 = sy / yy
    # Use the one that's closer to the previous step size
    if abs(bb1 - step_prev) < abs(bb2 - step_prev):
        return max(1e-10, min(10.0, bb1))
    else:
        return max(1e-10, min(10.0, bb2))

def gd_variance(a, time_budget, max_iters=500, verbose=False):
    """Gradient descent minimizing variance of convolution."""
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    step = 1.0 / n
    g_prev = None
    a_prev = None
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        g = grad_variance(a)
        # Project gradient onto tangent space of simplex
        g = g - np.mean(g)
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Try a few step sizes
        best_local = best_obj
        best_local_a = best_a.copy()
        for factor in [0.5, 1.0, 2.0, 0.1]:
            s_try = step * factor
            a_try = a - s_try * g / g_norm * 0.1
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_local:
                best_local = obj_try
                best_local_a = a_try.copy()
        
        if best_local < best_obj:
            a_new = best_local_a
            best_obj = best_local
            best_a = a_new.copy()
            # Update BB step size
            if g_prev is not None and a_prev is not None:
                step = bb_step_size(g, g_prev, a_new, a_prev, step)
            g_prev = g.copy()
            a_prev = a.copy()
            a = a_new
        else:
            step *= 0.7
            a = a + 0.001 * np.random.standard_normal(n)
            a = project_simplex(a)
        
        if verbose and it % 50 == 0:
            print(f"      VarGD it{it}: obj={best_obj:.10f}, step={step:.6f}")
    
    return best_a

def gd_smoothed_max(a, time_budget, tau_schedule=None, max_total_iters=2000, verbose=False):
    """Gradient descent with progressive log-sum-exp smoothing."""
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if tau_schedule is None:

    total_iters = 0

        if time.monotonic() - start > time_budget:
            break
        
        step = 0.1 / np.sqrt(n)
        g_prev = None
        a_prev = None
        iters_this_tau = max(50, 5000 // len(tau_schedule))
        
        for it in range(iters_this_tau):
            if time.monotonic() - start > time_budget:
                break
            if total_iters >= max_total_iters:

            g, c_max = grad_smoothed_max(a, tau)
            g = g - np.mean(g)

            a_try = a - step * g / g_norm * 0.05
            a_try = project_simplex(a_try)

            if obj_try < best_obj:

def freq_equalize(a, time_budget, verbose=False):

    Uses iterative amplitude modification (like Gerchberg-Saxton / error reduction).
    """

    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        # Get current auto-spectrum
        A = fft(a, N)
        auto_spec = A * np.conj(A)  # = |A|^2, this is the DFT of the circular autocorrelation
        
        # Get the linear autocorrelation
        c = compute_conv(a)
        c_full = np.zeros(N)
        c_full[:len(c)] = c
        
        # Target: make the autocorrelation flat
        target = np.full(N, float(np.sum(c)) / N)
        
        # Error correction: adjust the spectrum
        error = c_full - target
        E = fft(error, N)
        
        # Modify the phase of A to reduce the error
        # New A should have |A|^2 closer to target spectrum
        target_spec = np.maximum(np.real(fft(target, N)), 0)
        A_new_mag = np.sqrt(np.maximum(auto_spec.astype(complex) - np.real(E) * 0.1, 1e-20)).astype(np.float64) if False else None
        
        # Simpler approach: just dampen the frequencies that contribute most to the peak
        # Find the peak lag and reduce its contribution
        peak_lag = int(np.argmax(c))
        # The contribution to c[peak_lag] from frequency k is |A[k]|^2 * exp(2*pi*i*k*peak_lag/N)
        # To reduce c[peak_lag], we want to reduce |A[k]|^2 * cos(2*pi*k*peak_lag/N) for positive cos terms
        
        phases = 2 * np.pi * np.arange(N) * peak_lag / N
        contrib = np.real(A * np.conj(A) * np.exp(1j * phases))  # = |A|^2 * cos(phase)
        
        # Dampen frequencies that positively contribute to the peak
        damp_mask = contrib > np.median(contrib)
        A_modified = A.copy()
        A_modified[damp_mask] *= 0.95
        A_modified[~damp_mask] *= 1.02  # slightly boost others to maintain energy
        
        a_new = ifft(A_modified, N).real[:n]
        a_new = project_simplex(a_new)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

                best_a = a_new2.copy()
                a = a_new2
            else:

        if verbose and it % 20 == 0:
            print(f"      FreqEq it{it}: obj={best_obj:.10f}")
    
    return best_a

def local_search(a, time_budget, verbose=False):

        j = (peak_lag - i + n) % n

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = project_simplex(a_new)
    return a_new
