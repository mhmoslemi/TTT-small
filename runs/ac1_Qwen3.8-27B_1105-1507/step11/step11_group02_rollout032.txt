import numpy as np
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2. Assumes a is normalized (sum=1)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    if s != 1.0:
        a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def multiplicative_peak_reduction(a, time_budget=300.0, tau=100.0, eta_init=0.5, verbose=False):
    """
    Iterative multiplicative update to reduce the peak of conv(a,a).
    a[i] *= exp(-eta * grad[i]) / normalization
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stalled = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Adaptive tau: increase over time for sharper peak targeting
        tau_eff = tau * (1.0 + 2.0 * progress)
        # Adaptive eta: decrease when stalled
        eta = eta_init * (1.0 - 0.6 * progress) / (1.0 + 0.1 * stalled)

        try:
            grad, conv, p = compute_peakedness_gradient(a, tau_eff)
        except Exception:
            break

        # Ensure gradient is non-negative (it should be since a >= 0 and p >= 0)
        grad = np.maximum(grad, 0)
        
        # Normalize gradient for stability
        g_max = float(np.max(grad))
        if g_max < 1e-15:
            break
        grad_norm = grad / g_max

        # Multiplicative update
        a_new = a * np.exp(-eta * grad_norm)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(np.sum(a_new))

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stalled = 0
        else:
            a = a_new  # Still move forward (might help escape saddle points)
            stalled += 1
            if stalled > 50:
                # Reset with current best
                a = best_a.copy()
                stalled = 0

        it += 1
        if verbose and it % 500 == 0:
            print(f"    MPR it{it}: obj={best_obj:.8f}, tau={tau_eff:.1f}, eta={eta:.4f}")

    return best_a


def coordinate_descent(a, time_budget=300.0, verbose=False, seed=42):
    """
    Exact coordinate descent with golden section search.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        sweep += 1

        # Periodic full recompute
        if sweep % 4 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            # Define 1D objective
            def f_1d(x):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Candidate points
            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, v * 0.3, v * 0.7, v, v * 1.3, v * 1.7,
                          v + scale, v - scale if v > scale else 0.0, 1.0 / n]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates if c >= 0))

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section
            x_lo = max(0.0, best_x - scale * 1.5)
            x_hi = best_x + scale * 2.0
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(40):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            S += delta

            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, time_budget=20.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    start = time.monotonic()
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget=60.0, seed=42):
    """LP-based polishing with blending."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp(a, time_budget=min(remaining * 0.4, 15.0))
        if g_lp is None:
            break

        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 3:
                pert = rng.normal(0, 0.02, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / float(a.sum())
                stall = 0

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-8)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 8, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Concentrated in middle (like a triangular window)
        x = np.ones(n) - np.abs(2 * np.linspace(0, 1, n) - 1)
        a = x ** 2
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def peakedness_shaping(a, time_budget=100.0, verbose=False):
    """
    Alternative approach: directly minimize peak by iteratively
    reducing elements that contribute most to the peak.
    Uses a "responsibility" Based update.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        if max_conv < 1e-15:
            break

        # Compute "responsibility": how much each element contributes to near-peak positions
        # Threshold: positions where conv[k] > 0.95 * max_conv
        threshold = 0.95 * max_conv
        peak_mask = (conv >= threshold).astype(np.float64)

        # Responsibility of a[i] = sum_k peak_mask[k] * a[k-i] (for valid k-i)
        # This is a correlation-like computation
        N = 1
        while N < 3 * n:
            N <<= 1
        A = fft(a, N)
        PM = fft(peak_mask, N)
        resp = ifft(A * np.conj(PM)).real[:n]
        resp = np.maximum(resp, 0)

        if float(np.max(resp)) < 1e-15:
            break

        # Adaptive learning rate
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        eta = 0.3 * (1.0 - 0.5 * progress)

        # Update: reduce responsible elements
        a_new = a * np.exp(-eta * resp / max(float(np.max(resp)), 1e-15))
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(np.sum(a_new))

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 100 == 0:
            print(f"    PS it{it}: obj={best_obj:.8f}")

    return best_a

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def compute_peakedness_gradient(a, tau=100.0):
    """
    Compute gradient of smoothed max(conv(a,a)) using log-sum-exp.
    Returns (gradient, conv, peak_weights).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp

    # grad[i] = 2 * sum_k p[k] * a[k-i] for valid k-i
    # = 2 * cross_correlation(a, p) at lag i
    N = 1
    while N < 4 * n:
        N <<= 1
    A = fft(a, N)
    P = fft(p, N)
    corr = ifft(A * np.conj(P)).real[:n]
    grad = 2.0 * corr

    return grad, conv, p

def eval_1d(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i):
    """Evaluate 1D objective for coordinate descent at a single coordinate."""
    if len(a_ki_lin) > 0:

    max_val = max(max_lin, quad_val)
    denom = (S_other + x) * (S_other + x)
    if denom < 1e-30:
        return 1e30
    return 2.0 * n * max_val / denom

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, time_budget=20.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    start = time.monotonic()
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def peakedness_shaping(a, time_budget=100.0, verbose=False):
    """
    Responsibility-based peak reduction: iteratively reduce elements
    that contribute most to the convolution peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        if max_conv < 1e-15:
            break

        # Focus on positions near the peak
        threshold = 0.90 * max_conv
        peak_weights = np.maximum(conv - threshold, 0) / max_conv
        peak_weights = peak_weights / max(float(np.sum(peak_weights)), 1e-15)

        # Responsibility via correlation
        N = 1
        while N < 4 * n:
            N <<= 1
        A = fft(a, N)
        PW = fft(peak_weights, N)
        resp = ifft(A * np.conj(PW)).real[:n]
        resp = np.maximum(resp, 0)

        r_max = float(np.max(resp))
        if r_max < 1e-15:
            break

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        eta = 0.4 * (1.0 - 0.4 * progress)

        a_new = a * np.exp(-eta * resp / r_max)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(np.sum(a_new))

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 200 == 0:
            print(f"    PS it{it}: obj={best_obj:.8f}")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x * x + 1e-8)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2.0 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.ones(n) - np.abs(2.0 * np.linspace(0, 1, n) - 1)
        a = x ** 2
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        alloc = min(remaining * 0.12, 50.0)
        if alloc > 15:
            a_try = coordinate_descent(a_try, time_budget=alloc,
                                       verbose=False, seed=n_try)
        obj_try = compute_obj(a_try)
        print(f"  n={n_try}: obj={obj_try:.8f}")
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try

    # === Phase 8: Diverse restart ===
    print(f"\n=== Phase 8: Diverse Restart ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    for r in range(2):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        a_r = diverse_init(n_cur, rng, mode=1 + r)
        blend = 0.3 + 0.2 * r
        a_r = (1.0 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / float(a_r.sum())
        alloc = min(remaining * 0.12, 50.0)
        if alloc > 15:
            a_r = coordinate_descent(a_r, time_budget=alloc,
                                     verbose=False, seed=8000 + r * 111)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 9: Final Polish ===
    print(f"\n=== Phase 9: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 100.0)
        a_final = coordinate_descent(best_seq, time_budget=alloc,
                                     verbose=False, seed=99999)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_lp_final = lp_polish(best_seq, time_budget=min(remaining * 0.5, 40.0),
                               seed=77777)
        obj_lp_final = compute_obj(a_lp_final)
        if obj_lp_final < best_score:
            best_score = obj_lp_final
            best_seq = a_lp_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final MPR burst
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_mpr_f = multiplicative_peak_reduction(best_seq, time_budget=min(remaining * 0.4, 30.0),
                                                tau=200.0, eta_init=0.2)
        obj_mpr_f = compute_obj(a_mpr_f)
        if obj_mpr_f < best_score:
            best_score = obj_mpr_f
            best_seq = a_mpr_f.copy()
            print(f"  Final MPR: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
