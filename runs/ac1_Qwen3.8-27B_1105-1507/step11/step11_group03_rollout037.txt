"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a novel KKT-based direct solver combined with LP fixed-point iteration,
adaptive active-set tracking, and multi-n exploration with gradient polish.

Key innovation: Instead of iterating a <- (1-t)*a + t*g (LP direction),
we directly solve the KKT linear system that characterizes the fixed point.
This converges in O(1) iterations per active-set update rather than O(1/t).
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lp_direction(a, solver_list=None):
    """Solve min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1. Returns (g, M)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None


def kkt_direct_solve(n, K_active):
    """
    Directly solve the KKT linear system for a given active set K_active.
    The KKT conditions at the optimum (dense support) are:
        sum_{k in K} a[k-j] = C  for all j in [0, n)
        sum(a) = 1
    This gives an (n+1) x (n+1) linear system.
    """
    K_set = set(K_active)
    # Build R matrix: R[j, i] = 1 if (i+j) in K_set and 0<=i<n
    R = np.zeros((n, n))
    for j in range(n):
        for k in K_set:
            i = k - j
            if 0 <= i < n:
                R[j, i] = 1.0

    # System: R * a - C * 1 = 0  (n equations)
    #         sum(a) = 1          (1 equation)
    # Unknowns: a[0..n-1], C     (n+1 unknowns)
    A_mat = np.zeros((n + 1, n + 1))
    b_vec = np.zeros(n + 1)
    A_mat[:n, :n] = R
    A_mat[:n, n] = -1.0
    A_mat[n, :n] = 1.0
    b_vec[n] = 1.0

    try:
        x = np.linalg.solve(A_mat, b_vec)
    except np.linalg.LinAlgError:
        try:
            x = np.linalg.lstsq(A_mat, b_vec, rcond=None)[0]
        except Exception:
            return None, None

    a_sol = x[:n]
    C_sol = x[n]
    return a_sol, C_sol


def kkt_optimize(a, time_budget, verbose=False, max_iters=100):
    """
    KKT-based optimization: iteratively determine the active set and solve
    the resulting linear system directly. Much faster than LP fixed-point
    when near the optimum.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    prev_K = None
    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))

        # Determine active set with adaptive tolerance
        improved_this_iter = False
        for tol_frac in [0.9999, 0.999, 0.998, 0.995, 0.99, 0.98, 0.95]:
            threshold = max_val * tol_frac
            K_active = list(np.where(conv >= threshold)[0])

            if len(K_active) < 2:
                K_active = [int(np.argmax(conv))]
            if len(K_active) > 50:
                # Too many active lags - use top 50
                idx = np.argsort(conv)[::-1][:50]
                K_active = list(idx)

            # Solve KKT system
            a_sol, C_sol = kkt_direct_solve(n, K_active)
            if a_sol is None:
                continue

            # Check: C should be positive for a valid solution
            if C_sol <= 0:
                continue

            # Clip negative values and renormalize
            a_new = np.maximum(a_sol, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new

            # Evaluate
            obj_new = compute_obj(a_new)

            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                improved_this_iter = True

                # Check KKT satisfaction
                conv_new = compute_conv(a_new)
                max_new = float(np.max(conv_new))
                K_set = set(K_active)
                # For positions where a > 0, check R[j,:]*a ≈ C
                violations = []
                for j in range(n):
                    if a_new[j] > 1e-6:
                        val = 0.0
                        for k in K_set:
                            i = k - j
                            if 0 <= i < n:
                                val += a_new[i]
                        violations.append(abs(val - C_sol))
                max_viol = max(violations) if violations else np.inf

                if verbose and (it % 5 == 0 or max_viol < 1e-6):
                    print(f"    KKT it{it:3d}: obj={best_obj:.10f}, |K|={len(K_active)}, "
                          f"tol={tol_frac:.4f}, viol={max_viol:.2e}")

                if max_viol < 1e-8:
                    if verbose:
                        print(f"    KKT CONVERGED at it={it}, obj={best_obj:.10f}")
                    return best_a

                break

        if not improved_this_iter:
            # No improvement - try expanding the active set
            conv = compute_conv(a)
            max_val = float(np.max(conv))
            # Use a wider tolerance
            K_wide = list(np.where(conv >= max_val * 0.90)[0])
            if len(K_wide) > 5:
                a_sol, C_sol = kkt_direct_solve(n, K_wide)
                if a_sol is not None and C_sol > 0:
                    a_new = np.maximum(a_sol, 0)
                    s_new = float(np.sum(a_new))
                    if s_new > 1e-10:
                        a_new = a_new / s_new
                        obj_new = compute_obj(a_new)
                        if obj_new < best_obj - 1e-12:
                            best_obj = obj_new
                            best_a = a_new.copy()
                            a = a_new
                        else:
                            # Blend
                            a = 0.5 * a + 0.5 * a_new
                            a = np.maximum(a, 0)
                            a = a / a.sum()

            if it > 30:
                break

    return best_a


def lp_fixed_point(a, time_budget, verbose=False, seed=42, max_iters=300):
    """
    LP fixed-point iteration with bisection step-size selection.
    Improved over parent: uses bisection to find the largest improving t.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall_count = 0
    consecutive_lp_fails = 0

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, M_lp = lp_direction(a)
        if g_lp is None:
            consecutive_lp_fails += 1
            if consecutive_lp_fails > 8:
                break
            continue
        consecutive_lp_fails = 0

        # Bisection on t to find best step
        t_lo, t_hi = 0.0, 1.0
        best_t = 0.1
        best_local_obj = best_obj

        # First try a range of t values
        t_candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_t = t

        # Refine around best_t with bisection
        if best_t > 0:
            for _ in range(5):
                t_mid = best_t * 1.3
                if t_mid > 1.0:
                    t_mid = min(1.0, best_t + 0.05)
                a_try = (1.0 - t_mid) * a + t_mid * g_lp
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st > 1e-10:
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_local_obj - 1e-14:
                        best_local_obj = obj_try
                        best_t = t_mid

        if best_local_obj < best_obj - 1e-14:
            a_try = (1.0 - best_t) * a + best_t * g_lp
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            best_obj = best_local_obj
            best_a = a_try.copy()
            a = a_try
            stall_count = 0
            if verbose and it % 10 == 0:
                print(f"    LP it{it:3d}: obj={best_obj:.10f}, t={best_t:.3f}")
        else:
            stall_count += 1
            # Perturbation to escape local optima
            pert = 0.01 * min(1.0, stall_count / 10.0)
            a = a + pert * rng.standard_normal(n)
            a = np.maximum(a, 0)
            sa = float(np.sum(a))
            if sa > 1e-10:
                a = a / sa
            else:
                a = best_a.copy()

            if stall_count > 50:
                a = best_a.copy()
                a = a + 0.05 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_count = 0

    return best_a


def smoothed_grad(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    shifted = tau * (conv - exact_max)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def gradient_polish(a, time_budget, tau_levels=None, verbose=False, beta=0.9):
    """Nesterov accelerated gradient descent for final polish."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [5, 20, 80, 300, 1000, 3000]

    for tau in tau_levels:
        lr = 1.0 / np.sqrt(n)
        v = np.zeros(n)
        tau_iters = 0

        while time.monotonic() - start < time_budget and tau_iters < 3000:
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new
            tau_iters += 1
            lr = max(1e-8, lr * 0.99999)

        if verbose:
            print(f"    GD tau={tau:5d}: obj={best_obj:.10f}, iters={tau_iters}")

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_structured_init(n, mode=0):
    """Create structured initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.maximum(np.sinc(x), 0.001)
        return a / a.sum()
    elif mode == 2:
        a = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / max(1, n - 1))
        return a / a.sum()
    elif mode == 3:
        x = np.arange(n) / max(1, n - 1)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        return a / a.sum()
    elif mode == 4:
        x = np.linspace(-3, 3, n)
        a = np.exp(-x ** 2)
        return a / a.sum()
    elif mode == 5:
        # U-shaped (high at edges)
        x = np.arange(n) / n
        d = np.minimum(x, 1 - x) * 2
        a = 1.0 + 0.5 * np.exp(-d ** 2 / 0.3)
        return a / a.sum()
    elif mode == 6:
        # Bimodal
        x = np.arange(n) / n
        a = np.exp(-((x - 0.2) * 8) ** 2) + np.exp(-((x - 0.8) * 8) ** 2) + 0.01
        return a / a.sum()
    elif mode == 7:
        # Inverse cosine (anti-Hann)
        a = 0.46 + 0.54 * np.cos(2 * np.pi * np.arange(n) / max(1, n - 1))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 8:
        # Power-law taper
        x = np.arange(n) / n
        d = np.minimum(x, 1 - x) * 2
        a = 1.0 + 0.3 * d ** 0.5
        return a / a.sum()
    else:
        return np.ones(n) / n

                for blend in [0.3, 0.5, 0.7]:
                    a_blend = (1 - blend) * best_seq + blend * a_try
                    a_blend = np.maximum(a_blend, 0)
                    a_blend = a_blend / a_blend.sum()
                    obj_blend = compute_obj(a_blend)
                    if obj_blend < best_score:
                        best_score = obj_blend
                        best_seq = a_blend.copy()
                        print(f"  Init mode {mode}, blend={blend}: obj={best_score:.10f} **")
                        break
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Init mode {mode}: obj={best_score:.10f} **")
            else:
                print(f"  Init mode {mode}: obj={obj_try:.10f}")

    # === Phase 7: Final intensive LP + KKT ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    if remaining > 30:
        alloc = min(remaining * 0.30, 100.0)
        a_f1 = lp_fixed_point(best_seq, time_budget=alloc, verbose=True, seed=7777)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  Final LP: obj={best_score:.10f} **")

    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.30, 80.0)
        a_f2 = kkt_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final KKT: obj={best_score:.10f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.35, 60.0)
        a_f3 = gradient_polish(best_seq, time_budget=alloc,
                               tau_levels=[50, 200, 1000, 3000, 8000], verbose=False)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Final GD: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    return result
