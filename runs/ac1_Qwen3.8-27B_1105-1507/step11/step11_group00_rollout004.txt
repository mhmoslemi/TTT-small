    S = sum(a)

        for i in range(n):
            # Try decreasing a[i]
            # The peak is at k* = argmax c
            # Decreasing a[i] by δ reduces c[k] by δ*(a[k-i]+a[i]) for k≠2i
            # and by 2δ*a[i]+δ^2 for k=2i

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def subgradient_at_peaks(a, conv, top_k=1, weights=None):
    """
    Compute subgradient of max_k conv(a,a)[k] with respect to a.
    Uses top_k peaks with given weights (default uniform).
    Returns gradient vector of length n.
    """
    n = len(a)
    if weights is None:
        # Find top_k peaks
        if top_k >= len(conv):
            idx = np.arange(len(conv))
            w = np.ones(len(conv)) / len(conv)
        else:
            idx = np.argsort(conv)[-top_k:]
            w = np.ones(top_k) / top_k
    else:
        idx = np.argsort(conv)[-len(weights):]
        w = np.array(weights)

        # If k == 2i: d = 2*a[i]
        # If 0 <= k-i < n and k != 2i: d = a[k-i] + a[i]

                if k == 2 * i:
                    g[i] += wk * 2.0 * a[i]
                else:
                    g[i] += wk * (a[k - i] + a[i])

Multi-phase: fast mirror descent + coordinate descent + multi-n + restarts.

from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    c = compute_conv(a_n)
    return 2.0 * n * float(np.max(c))


def peak_subgrad(a, k):
    """d(conv(a,a)[k])/d(a[i]) for all i."""
    n = len(a)
    g = np.zeros(n)
    i_lo = max(0, k - n + 1)
    i_hi = min(k, n - 1)
    if i_lo <= i_hi:
        idx = np.arange(i_lo, i_hi + 1)
        g[idx] = a[k - idx] + a[idx]
        if k % 2 == 0 and i_lo <= k // 2 <= i_hi:
            g[k // 2] = 2.0 * a[k // 2]
    return g
