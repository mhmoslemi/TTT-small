import numpy as np

from scipy.optimize import minimize
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):

    A = build_conv_matrix(a_norm)

                s = float(np.sum(g_val))

                if s > 0.01:
                    a_try = a_try / s
                    obj_par = compute_obj(a_try)
                    if obj_par < best_obj:
                        best_obj = obj_par
                        best_t = t_par

    # Fine local search
    if best_t > 0:
        margin = 0.03
        lo = max(0.0, best_t - margin)
        hi = min(t_max, best_t + margin)
        ts_fine = np.linspace(lo, hi, 25)
        for t in ts_fine:
            a_try = (1.0 - t) * a + t * g_dir
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t

    return best_t, best_obj


def lp_optimize(a, time_budget=300.0, max_iters=200, verbose=False):
    """LP-based optimization with over-relaxation and smart stagnation handling."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        T0 = time.monotonic()
        g_lp, M_lp = solve_lp(a)
        lp_time = time.monotonic() - T0

        if g_lp is None:
            break

        t_opt, obj_opt = line_search_extended(a, g_lp, best_obj)

        if t_opt > 1e-6 and obj_opt < best_obj - 1e-14:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            # Try direct replacement
            obj_direct = compute_obj(g_lp)
            if obj_direct < best_obj:
                best_obj = obj_direct
                best_a = g_lp.copy()
                a = g_lp.copy()
                stall = 0
            else:
                stall += 1

        it += 1
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                  f"lp={lp_time:.2f}s, elapsed={elapsed:.0f}s")

        if stall >= 3:
            if stall <= 5:
                # Structured perturbation: shift mass between pairs
                n_shift = max(2, n // 100)
                for _ in range(n_shift):
                    i = rng.integers(0, n - 1)
                    j = rng.integers(0, n)
                    if i == j:
                        j = (j + 1) % n
                    amt = 0.02 * a[i]
                    a[i] -= amt
                    a[j] += amt
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 0.01:
                    a = a / s
                else:
                    a = best_a.copy()
            else:
                # Larger perturbation
                mode = it % 4
                a_div = diverse_init(n, rng, mode=mode)
                blend = min(0.5, 0.1 * stall)
                a = (1.0 - blend) * best_a + blend * a_div
                a = np.maximum(a, 0)
                a = a / a.sum()
            stall = 0

        if best_obj < 1.502:
            break

    return best_a, best_obj


def nlp_optimize(a, time_budget=60.0):
    """Direct NLP optimization using scipy minimize with log-parameterization."""
    n = len(a)
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Use log parameterization: a_i = exp(x_i) / sum(exp(x_j))
    x0 = np.log(a + 1e-15)

    def obj_func(x):
        xm = x - np.max(x)
        exp_x = np.exp(xm)
        s = exp_x.sum()
        if s < 1e-30:
            return 1e10
        a_val = exp_x / s
        conv = compute_conv(a_val)
        return 2.0 * n * np.max(conv)

    def grad_func(x):
        # Subgradient of max(conv) via scoreboard
        xm = x - np.max(x)
        exp_x = np.exp(xm)
        s = exp_x.sum()
        if s < 1e-30:
            return np.zeros(n)
        a_val = exp_x / s
        conv = compute_conv(a_val)
        k_max = int(np.argmax(conv))
        # Gradient of (a*a)(k_max) w.r.t. a
        grad_a = np.zeros(n)
        for i in range(n):
            j1 = k_max - i
            if 0 <= j1 < n:
                grad_a[i] += a_val[j1]
            j2 = i - k_max
            if 0 <= j2 < n:
                grad_a[i] += a_val[j2]
        # Chain rule through softmax
        # d(a_i)/dx_j = a_i * (delta_ij - a_j)
        # d(obj)/dx_j = sum_i d(obj)/da_i * d(a_i)/dx_j
        #             = sum_i grad_a[i] * a_i * (delta_ij - a_j)
        #             = grad_a[j]*a_j - a_j * sum_i grad_a[i]*a_i
        dot = np.dot(grad_a, a_val)
        grad_x = a_val * (grad_a - dot) * 2.0 * n
        return grad_x

    # Try different methods
    methods_configs = [
        ('L-BFGS-B', {'maxiter': 300, 'maxfev': 2000, 'gtol': 1e-8}),
        ('Powell', {'maxiter': 200, 'maxfev': 3000}),
    ]

    for method, opts in methods_configs:
        if time.monotonic() - start > time_budget - 5:
            break
        x_start = x0.copy()
        try:
            result = minimize(
                obj_func, x_start, jac=grad_func,
                method=method, options=opts
            )
            x_opt = result.x
            xm = x_opt - np.max(x_opt)
            a_new = np.exp(xm)
            a_new = a_new / a_new.sum()
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                print(f"    NLP ({method}): obj={best_obj:.8f}")
        except Exception as e:
            print(f"    NLP ({method}) failed: {e}")

        # One restart with perturbation
        if time.monotonic() - start > time_budget - 10:
            break
        x_start = x0 + 0.05 * np.random.default_rng(7).standard_normal(n)
        try:
            result = minimize(
                obj_func, x_start, jac=grad_func,
                method=method, options=opts
            )
            x_opt = result.x
            xm = x_opt - np.max(x_opt)
            a_new = np.exp(xm)
            a_new = a_new / a_new.sum()
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                print(f"    NLP ({method} restart): obj={best_obj:.8f}")
        except Exception:
            pass

    return best_a, best_obj


def subgradient_refine(a, time_budget=40.0):
    """Subgradient descent with averaging and momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    a_avg = a.copy()
    momentum = np.zeros(n)
    beta = 0.75
    lr0 = 0.03
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a_avg)
        k_max = int(np.argmax(conv))
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_max - i
            if 0 <= j1 < n:
                grad[i] += a_avg[j1]
            j2 = i - k_max
            if 0 <= j2 < n:
                grad[i] += a_avg[j2]
        lr_t = lr0 / (1.0 + 0.003 * it)
        momentum = beta * momentum - lr_t * grad
        a_new = a_avg + momentum
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
        else:
            a_new = a_new / s
        a_avg = (it / (it + 1.0)) * a_avg + (1.0 / (it + 1.0)) * a_new
        obj = compute_obj(a_avg)
        if obj < best_obj:
            best_obj = obj
            best_a = a_avg.copy()
            best_it = it
        it += 1
        if it - best_it > 2000:
            a_avg = best_a.copy()
            momentum = np.zeros(n)
            lr0 = 0.01
            beta = 0.5
            best_it = it

    return best_a, best_obj


def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 3) ** 2) / (2 * (n // 8) ** 2)) + \
            np.exp(-((idx - 2 * n // 3) ** 2) / (2 * (n // 8) ** 2)) + 0.01
    elif mode == 2:
        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 3:
        a = rng.uniform(0.05, 1.0, n)
        kernel = np.ones(3) / 3.0
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = 1.0 / (0.03 + 8.0 * (x - 0.5) ** 2)
    elif mode == 5:
        # Skewed: heavy on left side
        a = np.exp(-3.0 * x) + 0.01
    elif mode == 6:
        # Very skewed
        a = np.exp(-5.0 * x) + 0.005
    else:
        a = np.ones(n) * 0.1 + rng.uniform(0, 0.5, n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s


def propose_candidate(seed=42, budget_s=1500):
    start_time = time.monotonic()

    # Load parent

        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        s = float(np.sum(parent_arr))
        if s > 0.01:
            best_seq = parent_arr / s
            best_score = compute_obj(best_seq)
            print(f"Parent loaded: n={parent_arr.size}, obj={best_score:.8f}")
        else:
            raise ValueError("Sum too small")
    except Exception:
        best_seq = diverse_init(800, rng, mode=0)
        best_score = compute_obj(best_seq)
        print(f"Default init: n={len(best_seq)}, obj={best_score:.8f}")

    n_best = len(best_seq)

    # Try diverse initializations at same n
    for mode in range(7):
        a_alt = diverse_init(n_best, rng, mode=mode)
        obj_alt = compute_obj(a_alt)
        if obj_alt < best_score:
            best_score = obj_alt
            best_seq = a_alt.copy()
            print(f"  Alt init mode={mode}: obj={best_score:.8f} **")

    # === Phase 1: LP optimization ===
    remaining = deadline - time.monotonic()

        print(f"\n=== Phase 1: LP Optimization (n={n_best}, budget={min(remaining, 400):.0f}s) ===")
        alloc = min(remaining * 0.35, 400.0)
        a_lp, obj_lp = lp_optimize(best_seq, time_budget=alloc, max_iters=300, verbose=True)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_lp:.8f} (no improvement)")

    remaining = deadline - time.monotonic()

        print(f"\n=== Phase 2: NLP Optimization ===")

        a_nlp, obj_nlp = nlp_optimize(best_seq, time_budget=alloc)

            best_score = obj_nlp

        else:
            print(f"  Phase 2: obj={obj_nlp:.8f} (no improvement)")

    # === Phase 3: Subgradient refinement ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 3: Subgradient Refinement ===")
        alloc = min(remaining * 0.05, 40.0)
        a_sg, obj_sg = subgradient_refine(best_seq, time_budget=alloc)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj_sg:.8f} (no improvement)")

    # === Phase 4: LP polish round 2 ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 4: LP Round 2 ===")
        alloc = min(remaining * 0.15, 120.0)
        a_lp2, obj_lp2 = lp_optimize(best_seq, time_budget=alloc, max_iters=100, verbose=True)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Phase 4: obj={best_score:.8f} **")

    # === Phase 5: n Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 5: n Exploration ===")
        n_cur = len(best_seq)
        n_options = sorted(set([
            n_cur - 100, n_cur - 50, n_cur + 50, n_cur + 100,
            n_cur + 200, n_cur + 400, n_cur + 600,
            400, 500, 600, 700, 900, 1000, 1100, 1200, 1400, 1600, 2000
        ]))
        n_options = [n for n in n_options if 300 <= n <= 2500 and abs(n - n_cur) >= 40]
        n_options.sort(key=lambda x: abs(x - n_cur))
        n_options = n_options[:8]
        print(f"  Candidate n values: {n_options}")

        for n_try in n_options:

                break

            else:

    # === Phase 7: Final intensive polish ===
    remaining = deadline - time.monotonic()

        print(f"\n=== Phase 7: Final Polish ===")
        alloc = min(remaining - 10, 100.0)

        a_sg2, obj_sg2 = subgradient_refine(best_seq, time_budget=min(remaining * 0.4, 20.0))
        if obj_sg2 < best_score:
            best_score = obj_sg2
            best_seq = a_sg2.copy()
            print(f"  SG: obj={best_score:.8f} **")
        remaining2 = deadline - time.monotonic()
        if remaining2 > 10:
            a_sq, obj_sq = lp_optimize(best_seq, time_budget=min(remaining2 - 5, 30.0), max_iters=20, verbose=False)
            if obj_sq < best_score:

                best_seq = a_sq
                print(f"  Squeeze: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:

        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
