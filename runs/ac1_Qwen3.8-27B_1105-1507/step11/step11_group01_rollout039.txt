def compute_conv(a):

    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n - 1]

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def objective_normalized(a):
    """Compute 2*n*max(conv) assuming a is normalized to sum=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:
                    g_val = g_val / g_sum
                    conv_check = A @ g_val
                    M_actual = float(np.max(conv_check))
                    return g_val, M_actual
        except Exception:
            continue
    return None, None

def subgradient_descent(a, time_budget=150.0, seed=42, verbose=False):
    """
    Nesterov-accelerated subgradient descent using the exact subgradient
    of max_k conv_k(a). Active set averaging for stability.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    beta = 0.8  # momentum
    v_mom = np.zeros(n)
    it = 0
    patience = 0
    lr_base = 2.0 / np.sqrt(n)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = elapsed / max(time_budget, 1e-10)
        lr = lr_base / (1.0 + 0.1 * it)

        # Nesterov lookahead
        y = a + beta * v_mom
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            s_y = 1.0
        else:
            y = y / s_y

        # Compute conv and active set
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        threshold = max_conv * (1.0 - 1e-8)
        active_mask = conv >= threshold
        active_k = np.where(active_mask)[0]

        if len(active_k) == 0:
            active_k = np.array([np.argmax(conv)])

        # Compute subgradient: average gradient over active lags
        grad = np.zeros(n)
        for k in active_k:
            # d(conv_k)/d(a[i]) = 2 * a[k-i] if 0 <= k-i < n
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1]
        grad /= len(active_k)

        # Center the gradient (remove mean to maintain sum=1 approximately)
        grad -= np.mean(grad)

        # Update with momentum
        v_mom = beta * v_mom - lr * grad
        a_new = y + v_mom
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = y.copy()
            v_mom = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1
            if patience > 30:
                # Add perturbation to escape
                pert_scale = 0.02 * (1.0 + progress)
                pert = rng.normal(0, pert_scale, n)
                a_try = a * np.exp(pert)
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                    else:
                        a = best_a.copy()
                    v_mom = np.zeros(n)
                    patience = 0

        it += 1
        if verbose and it % 500 == 0:
            print(f"    SubGrad it{it}: obj={best_obj:.10f}, lr={lr:.6f}, t={elapsed:.1f}s")

    return best_a


def peak_deflation(a, time_budget=60.0, seed=42):
    """
    Targeted peak deflation: identify the peak lag(s) and reduce
    the products a[j]*a[k-j] for high-contributing pairs by
    redistributing mass to lower-impact positions.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    for iteration in range(50):
        if time.monotonic() - start >= time_budget:
            break

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        peak_k = int(np.argmax(conv))

        # Find which pairs contribute most to conv[peak_k]
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        if lo > hi:
            break

        pairs = np.arange(lo, hi + 1)
        contributions = a[pairs] * a[peak_k - pairs]
        total_contrib = float(np.sum(contributions))
        if total_contrib < 1e-10:
            break

        # Identify top contributors (not the diagonal)
        diag_mask = pairs != (peak_k - pairs)
        off_diag_contribs = contributions[diag_mask]
        off_diag_pairs = pairs[diag_mask]

        if len(off_diag_contribs) == 0:
            # Peak is at the diagonal; hard to reduce without changing n
            break

        # Find positions to reduce: high a[j] where a[k-j] is also high
        # and positions to increase: low a[j] where a[k-j] is also low
        top_frac = 0.1
        n_top = max(1, int(len(off_diag_pairs) * top_frac))
        top_idx = np.argsort(-off_diag_contribs)[:n_top]
        top_positions = off_diag_pairs[top_idx]

        # Find "safe" positions to move mass to
        safe_mask = np.ones(n, dtype=bool)
        safe_mask[top_positions] = False
        safe_positions = np.where(safe_mask)[0]

        if len(safe_positions) == 0:
            break

        # Score safe positions by how much they'd increase other conv values
        # Prefer positions where the "anti-contribution" is low
        safe_scores = np.zeros(len(safe_positions))
        for idx, pos in enumerate(safe_positions):
            # What would conv[peak_k] become if we added mass here?
            pair_pos = peak_k - pos
            if 0 <= pair_pos < n:
                safe_scores[idx] = a[pair_pos]
            else:
                safe_scores[idx] = 0.0

        # Move mass from top positions to lowest-scoring safe positions
        n_move = max(1, min(len(top_positions), len(safe_positions) // 2))
        bottom_idx = np.argsort(safe_scores)[:n_move]
        bottom_positions = safe_positions[bottom_idx]

        # Determine how much to move
        move_amount = 0.01 * s / n  # small fraction
        total_move = 0.0
        for pos in top_positions[:n_move]:
            move = min(move_amount, a[pos] * 0.3)
            a[pos] -= move
            total_move += move
            # Distribute to safe positions
            for bpos in bottom_positions:
                fraction = 1.0 / len(bottom_positions)
                a[bpos] += move * fraction

        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        obj = 2.0 * n * float(np.max(compute_conv(a)))
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation on [0,1]."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse structured initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        # Exponential decay
        a = np.exp(-np.linspace(0, 3, n))
        a = a / a.sum()
    elif mode == 2:
        # Power law
        a = np.linspace(1, 0.1, n) ** 2
        a = a / a.sum()
    elif mode == 3:
        # Random
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 4:
        # Two-bump asymmetric
        a = np.zeros(n)
        center1 = n // 5
        width1 = n // 10
        center2 = 4 * n // 5
        width2 = n // 8
        a += np.exp(-((np.arange(n) - center1) ** 2) / (2 * width1 ** 2))
        a += 0.5 * np.exp(-((np.arange(n) - center2) ** 2) / (2 * width2 ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Sine-modulated decay
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * x) * (1 + 0.3 * np.sin(2 * np.pi * 3 * x))
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 6:
        # Cosine window (asymmetric)
        x = np.linspace(0, 1, n)
        a = (1 - x) * np.cos(np.pi * x / 2) ** 2
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 7:
        # Log-sine
        x = np.linspace(0.01, 1, n)
        a = np.sin(np.pi * x) * x ** 0.5
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a
