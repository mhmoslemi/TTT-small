"""
Optimizes step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses high-precision LP fixed-point with momentum, backtracking line search,
active-set Newton polish, and multi-n exploration.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lp_solve(a):
    """Solve min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1.
    Returns (g, M, dual_vars) or (None, None, None)."""
    n = len(a)
    # Build constraint matrix: (a*g)[k] = sum_j a[max(0,k-n+1):min(k,n-1)+1] * g[...]
    # More efficiently: (a*g)[k] for k=0..2n-2
    # Row k: has a[j] at column j for j where 0 <= k-j < n, i.e., k-n+1 <= j <= k
    # So column j appears in rows j, j+1, ..., j+n-1 with values a[j], a[j-1], ..., a[j-n+1]... 
    # Wait, let me think again.
    # (a*g)[k] = sum_j a[k-j] * g[j] for valid j (0 <= j < n, 0 <= k-j < n)
    # So for fixed k, the terms are: g[j] * a[k-j] for j = max(0, k-n+1) to min(k, n-1)
    
    # Build the matrix A where A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # A has shape (2n-1, n)
    # For efficiency, use the Toeplitz-like structure
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] contributes to (a*g)[k] for k = j, j+1, ..., j+n-1
        # with coefficient a[k-j] = a[0], a[1], ..., a[n-1]
        A[j:j+n, j] = a
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    g_val = g_val / sg
                    return g_val, float(M_var.value), None
        except Exception:
            continue
    return None, None, None


def lp_fixed_point_v2(a, time_budget, verbose=False, seed=42, max_iters=5000, momentum=0.5):
    """Improved LP fixed-point with momentum and backtracking line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Momentum state
    v = np.zeros(n)
    beta = momentum  # momentum coefficient
    
    stall_count = 0
    no_improve_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp, _ = lp_solve(a)
        if g_lp is None:
            break
        
        # Backtracking line search
        t = 1.0
        current_obj = compute_obj(a)
        
        # First try t=1 (full LP step)
        a_try = g_lp  # t=1: a = g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st > 1e-10:
            a_try = a_try / st
        obj_try = compute_obj(a_try)
        
        if obj_try < current_obj - 1e-15:
            best_obj = obj_try
            best_a = a_try.copy()
            # Update momentum
            v = beta * v + (a_try - a)
            a = a_try
            stall_count = 0
            no_improve_count = 0
        else:
            # Backtrack
            improved = False
            for t in [0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
                a_try = (1.0 - t) * a + t * g_lp
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < current_obj - 1e-15:
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                    # Update momentum
                    v = beta * v + (a_try - a)
                    a = a_try
                    improved = True
                    stall_count = 0
                    no_improve_count = 0
                    if verbose and it % 50 == 0:
                        print(f"  LP it{it:4d}: obj={best_obj:.10f}, t={t:.3f}")
                    break
            
            if not improved:
                # Apply momentum step
                a_mom = a + 0.1 * v
                a_mom = np.maximum(a_mom, 0)
                sm = float(np.sum(a_mom))
                if sm > 1e-10:
                    a_mom = a_mom / sm
                    obj_mom = compute_obj(a_mom)
                    if obj_mom < current_obj:
                        a = a_mom
                        no_improve_count += 1
                    else:
                        no_improve_count += 1
                        v = 0.5 * v  # damp momentum
                else:
                    no_improve_count += 1
                
                if no_improve_count > 20:
                    # Perturb
                    a = best_a.copy()
                    pert_strength = min(0.05, 0.01 * (no_improve_count // 10))
                    a = a + pert_strength * rng.standard_normal(n)
                    a = np.maximum(a, 0)
                    sa = float(np.sum(a))
                    if sa > 1e-10:
                        a = a / sa
                    v = np.zeros(n)
                    no_improve_count = 0
                    stall_count = 0
    
    return best_a

def smoothed_grad(a, tau):
    """Smoothed gradient of max(conv(a,a)) using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    shifted = tau * (conv - exact_max)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    # Gradient of sum_k p_k * (a*a)[k] with respect to a[j]
    # = sum_k p_k * d/d(a[j]) (a*a)[k]
    grad = np.zeros(n)
    for k in range(len(p)):
        if p[k] < 1e-10:
            continue
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            if 2 * j == k:
                grad[j] += p[k] * a[j]
            else:
                grad[j] += p[k] * 2.0 * a[k - j]
    
    return grad


def nesterov_gd(a, time_budget, verbose=False, tau_levels=None, beta=0.9, lr_init=None):
    """Nesterov accelerated gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if tau_levels is None:
        tau_levels = [5, 20, 60, 200, 600, 1500]
    if lr_init is None:
        lr_init = 2.0 / np.sqrt(n)
    
    for tau in tau_levels:
        if time.monotonic() - start > time_budget:
            break
        lr = lr_init
        v = np.zeros(n)
        
        for it in range(5000):
            if time.monotonic() - start > time_budget:
                break
            
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy
            
            g = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break
            
            v = beta * v - lr * g / g_norm
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
            
            a = a_new
            lr = max(1e-8, lr * (1 - it * 0.0001))
        
        if verbose:
            print(f"    tau={tau:5d}: obj={best_obj:.10f}")
    
    return best_a


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

    elif mode == 2:

        return np.maximum(a, 0.01) / np.maximum(a, 0.01).sum()
    elif mode == 4:

        a = np.exp(-x ** 2)
        return a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
        return a / a.sum()
    elif mode == 6:
        a = rng.uniform(0.1, 1.0, n)
        return a / a.sum()
    elif mode == 7:

        a = np.exp(-40*(x-0.3)**2) + np.exp(-40*(x-0.7)**2)

    elif mode == 8:

        x = np.arange(n) / n
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n
