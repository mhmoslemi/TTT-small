"""
Full-constraint LP-based optimization for step function coefficients.
Key innovation: Use the FULL LP (all 2n-1 convolution constraints) solved with PDLP,
alternating with projected gradient steps for escape. Adaptive constraint selection
and multi-direction combination for robust local minimization.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n


def build_conv_rows(a, indices):
    """Build rows of convolution matrix for given peak indices.
    Row k: (a*g)[k] = sum_m A[k,m] * g[m] where A[k,m] = a[k-m] if valid."""
    n = len(a)
    A = np.zeros((len(indices), n))
    for idx, k in enumerate(indices):
        k = int(k)
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        for m in range(m_lo, m_hi + 1):
            A[idx, m] = a[k - m]
    return A


def solve_lp(a, indices, solver_name):
    """Solve LP: min M s.t. A_indices @ g <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_conv_rows(a, indices)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [solver_name]
    if solver_name == 'PDLP':
        solvers = [cp.PDLP, cp.MOSEK, cp.CLARABEL, cp.ECOS]
    elif solver_name == 'MOSEK':
        solvers = [cp.MOSEK, cp.PDLP, cp.CLARABEL]
    else:
        solvers = [solver_name, cp.CLARABEL, cp.ECOS, cp.PDLP]
    
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def find_best_blend(a, g, obj_a):
    """Find best blend factor t in (0, 1.5) for a_new = (1-t)*a + t*g."""
    best_t = 0.0
    best_obj = obj_a
    
    # Coarse search
    for t in np.linspace(0.05, 1.5, 20):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
    
    # Fine search around best_t
    if best_t > 0:
        for t in np.linspace(best_t - 0.05, best_t + 0.05, 11):
            if t < 0.01 or t > 2.0:
                continue
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_t = t
    
    return best_t, best_obj


def lse_gradient(a, T):
    """Gradient of 2n * LSE_T(conv(a,a)) for normalized a."""
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)
    exp_vals = np.exp(T * (conv - cmax))
    w = exp_vals / exp_vals.sum()
    
    # Gradient of sum_k w[k] * conv[k] where conv[k] = sum_j a[j]*a[k-j]
    # d/d a[i] = sum_k w[k] * (a[k-i] + a[i-k]) (where indices are valid)
    grad = np.zeros(n)
    
    # Use FFT-based approach for efficiency
    # sum_k w[k] * a[k-i] for valid k-i: this is (w * a)[i] (cross-correlation)
    # sum_k w[k] * a[i-k] for valid i-k: this is (w * a)[i] (same thing by symmetry... no)
    
    # Actually: conv[k] = sum_j a[j] * a[k-j]
    # d conv[k] / d a[i] = a[k-i] + a[i-k] (if both indices valid)
    # grad[i] = 2n * sum_k w[k] * (a[k-i] + a[i-k])
    
    # Term 1: sum_k w[k] * a[k-i] where 0 <= k-i < n → i <= k < i+n
    # Term 2: sum_k w[k] * a[i-k] where 0 <= i-k < n → i-n < k <= i
    
    # Both terms are weighted cross-correlations
    # Term 1: (w reversed cross-correlated with a) - actually let me just do it directly
    
    # For efficiency, use the fact that:
    # sum_k w[k] * a[k-i] = sum_j w[j+i] * a[j] (where j = k-i, so k = j+i)
    # = sum_j a[j] * w[j+i]
    
    # This is a "sliding dot product" - can be done with FFT
    # Let's construct an extended w: w_ext = [0, w, 0] (length 3n-1)
    # Then term1_i = sum_j a[j] * w[j+i] = cross-correlation
    
    # Actually for n=800, direct computation is fine (O(n^2) per gradient)
    # But let's use FFT for speed
    
    # Term 1: for each i, sum_{k=max(i,0)}^{min(2n-2, i+n-1)} w[k] * a[k-i]
    # This is the cross-correlation of w and a, evaluated at lag i
    # cross_corr_w_a[i] = sum_k w[k] * a[k-i]
    
    # Using FFT: cross_corr = ifft(fft(w, 2n-1) * conj(fft(a, 2n-1)))
    # But we need linear cross-correlation, not circular
    
    size = 1
    while size < 4 * n - 1:
        size <<= 1
    
    W = rfft(w, size)
    A = rfft(a, size)
    # Cross-correlation: ifft(W * conj(A)) gives sum_k w[k] * a[k+i] (or similar)
    # Let me be more careful
    # ifft(FFT(w) * conj(FFT(a))) [n] = sum_k w[k] * a[(k-n)%size + n] ... circular
    # For linear: pad to size >= 2n-1
    
    # Actually, let me just use the direct approach for n=800
    # It's O(n * avg_bandwidth) ≈ O(n^2/2) ≈ 320K operations - very fast
    
    for i in range(n):
        # Term 1: sum_k w[k] * a[k-i] where max(0,i) <= k <= min(2n-2, i+n-1)
        lo1 = max(0, i)
        hi1 = min(2 * n - 2, i + n - 1)
        if lo1 <= hi1:
            grad[i] += np.dot(w[lo1:hi1 + 1], a[lo1 - i:hi1 - i + 1])
        # Term 2: sum_k w[k] * a[i-k] where max(0, i-n+1) <= k <= min(2n-2, i)
        lo2 = max(0, i - n + 1)
        hi2 = min(2 * n - 2, i)
        if lo2 <= hi2:
            grad[i] += np.dot(w[lo2:hi2 + 1], a[i - hi2:i - lo2 + 1][::-1])
    
    return 2.0 * n * grad


def lse_objective(a, T):
    """Smoothed LSE objective."""
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)
    exp_vals = np.exp(T * (conv - cmax))
    lse = cmax + np.log(np.sum(exp_vals)) / T
    return 2.0 * n * lse


def projected_gradient_step(a, T, lr, time_budget):
    """A few steps of projected LSE gradient descent."""
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    while time.monotonic() - start < time_budget:
        grad = lse_gradient(a, T)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        
        # Nesterov-like: use momentum
        a_try = project_simplex(a - lr * grad / gn)
        obj_try = compute_obj(a_try)
        
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            lr *= 1.05
        else:
            lr *= 0.7
            if lr < 1e-8:
                break
    
    return best_a, best_obj


def coordinate_descent(a, time_budget, max_sweeps=5):
    """Coordinate descent: for each coordinate, try to optimize it."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start >= time_budget:
            break
        sweep_improved = False
        
        for i in range(n):
            if time.monotonic() - start >= time_budget:
                break
            
            ai_old = a[i]
            s_rest = float(np.sum(a)) - ai_old
            
            # Optimal value of a[i] (given others fixed):
            # We want to minimize max_k (a_new * a_new)[k] / s_new^2
            # This is complex, so use ternary search on the scaling of a[i]
            
            # Try a few values
            candidates = [0.0, ai_old * 0.5, ai_old * 0.75, ai_old * 1.25, ai_old * 1.5, ai_old * 2.0]
            best_val = ai_old
            best_local = best_obj
            
            for cv in candidates:
                a[i] = cv
                s_new = float(np.sum(a))
                if s_new < 0.01:
                    a[i] = ai_old
                    continue
                a_try = a / s_new
                obj_try = compute_obj(a_try)
                if obj_try < best_local:
                    best_local = obj_try
                    best_val = cv
            
            a[i] = best_val
            s_new = float(np.sum(a))
            if s_new > 0.01:
                a = a / s_new
                if best_local < best_obj:
                    best_obj = best_local
                    best_a = a.copy()
                    sweep_improved = True
        
        if not sweep_improved:
            break
    
    return best_a, best_obj


def u_shaped_init(n, sharpness):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s if s > 1e-10 else np.ones(n) / n


def arcsine_init(n):
    x = np.linspace(0.005, 0.995, n)
    a = 1.0 / np.sqrt(np.maximum(1e-10, 1 - (2 * x - 1) ** 2))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s if s > 1e-10 else np.ones(n) / n


def power_init(n, alpha):
    x = np.linspace(0, 1, n)
    d = np.abs(x - 0.5)
    a = np.maximum(d, 1.0 / (2 * n)) ** (-alpha) + 0.01
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s if s > 1e-10 else np.ones(n) / n


def diverse_init(n, mode):
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n / 4.0) ** 2) / (2 * (n / 6.0) ** 2)) + \
            np.exp(-((idx - 3 * n / 4.0) ** 2) / (2 * (n / 6.0) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 4:
        a = (1 + np.cos(np.pi * (2 * x - 1))) ** 0.5 + 0.01
    elif mode == 5:
        a = 1.0 / np.cosh(3 * (2 * x - 1))
    elif mode == 6:
        a = (1 + np.sin(np.pi * x)) ** 2
    elif mode == 7:
        # Double peak
        a = np.exp(-20 * (x - 0.2) ** 2) + np.exp(-20 * (x - 0.8) ** 2) + 0.01
    elif mode == 8:
        # W-shaped
        a = np.abs(np.sin(2 * np.pi * x)) ** 0.5 + 0.05
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s if s > 1e-10 else np.ones(n) / n

"""
Hybrid optimization for step function coefficients combining:
1. IAA-style frequency domain flattening (fast, O(n log n) per iteration)
2. Full-constraint LP with sparse matrices (precise, all 2n-1 convolution constraints)
3. Adaptive truncated LP for intermediate precision
4. LSE-smoothed projected gradient for escape from local minima
5. Coordinate descent for final polishing
Key innovation: frequency-domain IAA method provides fast convergence to a good basin,
then full LP with sparse structure provides precise local optimization.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy import sparse
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n


def build_sparse_conv_rows(a, indices):
    """Build sparse rows of convolution matrix for given peak indices."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(indices):
        k = int(k)
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        for m in range(m_lo, m_hi + 1):
            rows.append(idx)
            cols.append(m)
            vals.append(a[k - m])
    return sparse.csr_matrix((vals, (rows, cols)), shape=(len(indices), n))


def solve_lp(a, indices, solver_pref='PDLP'):
    """Solve LP: min M s.t. A_indices @ g <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_sparse_conv_rows(a, indices)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solver_order = [cp.PDLP, cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    if solver_pref == 'MOSEK':
        solver_order = [cp.MOSEK, cp.PDLP, cp.CLARABEL]
    elif solver_pref == 'CLARABEL':
        solver_order = [cp.CLARABEL, cp.PDLP, cp.MOSEK]

    for solver in solver_order:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def find_best_blend(a, g, obj_a):
    """Find best blend factor t for a_new = (1-t)*a + t*g."""
    best_t = 0.0
    best_obj = obj_a

    for t in np.linspace(0.05, 1.5, 25):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t

    if best_t > 0:
        lo = max(0.01, best_t - 0.05)
        hi = min(2.0, best_t + 0.05)
        for t in np.linspace(lo, hi, 11):
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_t = t

    return best_t, best_obj


def iaa_flatten(a, time_budget, alpha_start=0.3, alpha_end=0.9, verbose=False):
    """
    Iterative Amplitude Alignment: compress |FFT(a)| dynamic range
    to flatten the auto-convolution. Fast O(n log n) per iteration.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    alpha = alpha_start

    size = 1
    while size < 4 * n - 1:
        size <<= 1

    while time.monotonic() - start < time_budget:
        # Compute full-size FFT
        A = fft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Compute target magnitude: interpolate between current and flat
        mag_mean = np.mean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
        # Soft compression: mag_target = (1-alpha)*mag + alpha*mag_mean
        mag_target = (1 - alpha) * mag + alpha * mag_mean
        mag_target = np.maximum(mag_target, 1e-10)

        # Reconstruct with new magnitudes, same phases
        A_new = mag_target * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, size)[:n])

        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 0.01:
            alpha = max(0.05, alpha * 0.8)
            it += 1
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Slowly increase alpha (more aggressive compression)
            alpha = min(alpha * 1.005, alpha_end)
        else:
            alpha = max(0.05, alpha * 0.95)
            if alpha < 0.06:
                break

        it += 1
        if verbose and it % 200 == 0:
            print(f"      IAA it{it}: obj={best_obj:.8f}, alpha={alpha:.4f}, "
                  f"t={time.monotonic()-start:.1f}s")

    return best_a


def lse_gradient(a, T):
    """Gradient of 2n * LSE_T(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)
    # Clip for numerical stability
    exp_vals = np.exp(np.clip(T * (conv - cmax), -50, 50))
    w = exp_vals / (exp_vals.sum() + 1e-30)

    grad = np.zeros(n)
    for i in range(n):
        lo1 = max(0, i)
        hi1 = min(2 * n - 2, i + n - 1)
        if lo1 <= hi1:
            grad[i] += np.dot(w[lo1:hi1 + 1], a[lo1 - i:hi1 - i + 1])
        lo2 = max(0, i - n + 1)
        hi2 = min(2 * n - 2, i)
        if lo2 <= hi2:
            grad[i] += np.dot(w[lo2:hi2 + 1], a[i - hi2:i - lo2 + 1][::-1])
    return 2.0 * n * grad


def projected_gradient_steps(a, T, lr_init, time_budget, verbose=False):
    """Projected LSE gradient descent with adaptive learning rate."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    it = 0
    momentum = np.zeros(n)
    beta = 0.5

    while time.monotonic() - start < time_budget:
        grad = lse_gradient(a, T)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break

        momentum = beta * momentum - lr * grad / gn
        a_try = project_simplex(a + momentum)
        obj_try = compute_obj(a_try)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            lr *= 1.02
        else:
            lr *= 0.7
            momentum *= 0.5
            if lr < 1e-9:
                break

        it += 1
        if verbose and it % 500 == 0:
            print(f"      PG it{it}: obj={best_obj:.8f}, lr={lr:.2e}, T={T:.1f}, "
                  f"t={time.monotonic()-start:.1f}s")

    return best_a, best_obj


def coordinate_descent(a, time_budget, max_sweeps=5, verbose=False):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start >= time_budget:
            break
        sweep_improved = False

        for i in range(n):
            if time.monotonic() - start >= time_budget:
                break

            ai_old = a[i]

            candidates = [0.0, ai_old * 0.3, ai_old * 0.6, ai_old * 1.4, ai_old * 2.0]
            best_val = ai_old
            best_local = best_obj

            for cv in candidates:
                a[i] = cv
                s_new = float(np.sum(a))
                if s_new < 0.01:
                    a[i] = ai_old
                    continue
                a_try = a / s_new
                obj_try = compute_obj(a_try)
                if obj_try < best_local:
                    best_local = obj_try
                    best_val = cv

            a[i] = best_val
            s_new = float(np.sum(a))
            if s_new > 0.01:
                a = a / s_new
                if best_local < best_obj:
                    best_obj = best_local
                    best_a = a.copy()
                    sweep_improved = True

        if verbose:
            print(f"      CD sweep {sweep}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
        if not sweep_improved:
            break

    return best_a, best_obj

def lp_iteration(a, top_m, solver_pref='PDLP'):
    """Single LP iteration: solve LP, find best blend, return updated a and obj."""
    n = len(a)
    conv_a = compute_conv(a)

    if top_m >= 2 * n - 1:
        indices = list(range(2 * n - 1))
    else:
        top_indices = np.argsort(conv_a)[::-1][:top_m]
        indices = top_indices

    g_lp = solve_lp(a, indices, solver_pref)
    if g_lp is None:
        return a, compute_obj(a), False

    obj_a = compute_obj(a)
    t_opt, obj_opt = find_best_blend(a, g_lp, obj_a)

    if t_opt > 0 and obj_opt < obj_a - 1e-14:
        a_new = (1 - t_opt) * a + t_opt * g_lp
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 0.01:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < obj_a:
                return a_new, obj_new, True

    return a, obj_a, False

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # === Initialization ===
    n_current = 800

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.all(np.isfinite(parent_arr)):
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        print("Could not load parent, using defaults")

    # Try diverse initializations
    inits = [
        ("u_sharp3", u_shaped_init(n_current, 3.0)),
        ("u_sharp5", u_shaped_init(n_current, 5.0)),
        ("u_sharp10", u_shaped_init(n_current, 10.0)),
        ("arcsine", arcsine_init(n_current)),
        ("power1.5", power_init(n_current, 1.5)),
        ("power2.0", power_init(n_current, 2.0)),
        ("power3.0", power_init(n_current, 3.0)),
        ("div0", diverse_init(n_current, 0)),
        ("div3", diverse_init(n_current, 3)),
        ("div7", diverse_init(n_current, 7)),
        ("div8", diverse_init(n_current, 8)),
        ("div9", diverse_init(n_current, 9)),
    ]

    for name, a_init in inits:
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Init '{name}': obj={obj_init:.8f} **")
        else:
            print(f"  Init '{name}': obj={obj_init:.8f}")

    if best_seq is None:
        best_seq = u_shaped_init(n_current, 3.0)
        best_score = compute_obj(best_seq)
        print(f"  Default init: obj={best_score:.8f}")

    n_best = len(best_seq)
    print(f"\nBest initial: n={n_best}, obj={best_score:.8f}")

    # === Phase 1: IAA Frequency Flattening (fast) ===
    remaining = deadline - time.monotonic()
    phase1_budget = min(120.0, remaining * 0.12)

    if phase1_budget > 20:
        print(f"\n=== Phase 1: IAA Frequency Flattening (budget={phase1_budget:.0f}s) ===")
        a_iaa = iaa_flatten(best_seq, time_budget=phase1_budget,
                            alpha_start=0.2, alpha_end=0.85, verbose=True)
        obj_iaa = compute_obj(a_iaa)
        if obj_iaa < best_score:
            best_score = obj_iaa
            best_seq = a_iaa.copy()
            print(f"  Phase 1 IAA: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 IAA: obj={obj_iaa:.8f} (no improvement)")

    # === Phase 2: Truncated LP (top-30 to top-100) ===
    remaining = deadline - time.monotonic()
    phase2_budget = min(350.0, remaining * 0.3)

    if phase2_budget > 30:
        print(f"\n=== Phase 2: Truncated LP (budget={phase2_budget:.0f}s) ===")
        a_work = best_seq.copy()
        obj_work = compute_obj(a_work)
        start_p2 = time.monotonic()
        it = 0
        stall = 0
        n_w = len(a_work)

        while time.monotonic() - start_p2 < phase2_budget and it < 200:
            rem = phase2_budget - (time.monotonic() - start_p2)
            if rem < 5:
                break

            frac = min(1.0, it / 80.0)
            top_m = int(round(30 + frac * 70))  # 30 to 100

            a_work, obj_work, improved = lp_iteration(a_work, top_m, 'PDLP')

            if improved:
                stall = 0
            else:
                stall += 1

            if obj_work < best_score:
                best_score = obj_work
                best_seq = a_work.copy()

            it += 1

            # Escape strategy
            if stall >= 3 and it % 4 == 0:
                a_esc, obj_esc = projected_gradient_steps(
                    a_work, T=80.0, lr_init=0.003, time_budget=2.0)
                if obj_esc < obj_work:
                    a_work = a_esc
                    obj_work = obj_esc
                    stall = 0

            if stall >= 8:
                noise = rng.normal(0, 0.002 * stall, n_w)
                a_pert = a_work + noise
                a_pert = np.maximum(a_pert, 0)
                s_p = a_pert.sum()
                if s_p > 0.01:
                    a_pert = a_pert / s_p
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < obj_work + 0.01:
                        a_work = a_pert
                        obj_work = obj_pert
                stall = 0

            if it % 25 == 0:
                print(f"    P2 it{it}: obj={obj_work:.8f}, best={best_score:.8f}, "
                      f"top_m={top_m}, t={time.monotonic()-start_p2:.1f}s")

        print(f"  Phase 2 done: obj={obj_work:.8f}, best={best_score:.8f}, "
              f"iters={it}, time={time.monotonic()-start_p2:.1f}s")

    # === Phase 3: Full LP (all 2n-1 constraints) ===
    remaining = deadline - time.monotonic()
    phase3_budget = min(450.0, remaining * 0.35)

    if phase3_budget > 30:
        print(f"\n=== Phase 3: Full LP (all {2*n_best-1} constraints, budget={phase3_budget:.0f}s) ===")
        a_work = best_seq.copy()
        obj_work = compute_obj(a_work)
        start_p3 = time.monotonic()
        it = 0
        stall = 0
        all_indices = list(range(2 * len(a_work) - 1))

        while time.monotonic() - start_p3 < phase3_budget and it < 200:
            rem = phase3_budget - (time.monotonic() - start_p3)
            if rem < 15:
                break

            t0 = time.monotonic()
            a_work, obj_work, improved = lp_iteration(a_work, len(all_indices), 'PDLP')
            lp_time = time.monotonic() - t0

            if improved:
                stall = 0
            else:
                stall += 1

            if obj_work < best_score:
                best_score = obj_work
                best_seq = a_work.copy()

            it += 1

            # Escape
            if stall >= 2 and it % 3 == 0:
                a_esc, obj_esc = projected_gradient_steps(
                    a_work, T=150.0, lr_init=0.002, time_budget=2.0)
                if obj_esc < obj_work:
                    a_work = a_esc
                    obj_work = obj_esc
                    stall = 0

            if stall >= 6:
                noise = rng.normal(0, 0.001 * stall, len(a_work))
                a_pert = a_work + noise
                a_pert = np.maximum(a_pert, 0)
                s_p = a_pert.sum()
                if s_p > 0.01:
                    a_pert = a_pert / s_p
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < obj_work + 0.005:
                        a_work = a_pert
                        obj_work = obj_pert
                stall = 0

            if it % 10 == 0:
                print(f"    P3 it{it}: obj={obj_work:.8f}, best={best_score:.8f}, "
                      f"lp_t={lp_time:.2f}s, t={time.monotonic()-start_p3:.1f}s")

        print(f"  Phase 3 done: obj={obj_work:.8f}, best={best_score:.8f}, "
              f"iters={it}, time={time.monotonic()-start_p3:.1f}s")

    # === Phase 4: IAA + LP alternating (fine-tuning) ===
    remaining = deadline - time.monotonic()
    phase4_budget = min(200.0, remaining * 0.18)

    if phase4_budget > 30:
        print(f"\n=== Phase 4: IAA + LP Alternation (budget={phase4_budget:.0f}s) ===")
        a_work = best_seq.copy()
        obj_work = compute_obj(a_work)
        start_p4 = time.monotonic()
        it = 0

        while time.monotonic() - start_p4 < phase4_budget and it < 40:
            rem = phase4_budget - (time.monotonic() - start_p4)
            if rem < 10:
                break

            # IAA step
            a_iaa = iaa_flatten(a_work, time_budget=8.0,
                                alpha_start=0.5, alpha_end=0.9)
            obj_iaa = compute_obj(a_iaa)
            if obj_iaa < obj_work:
                a_work = a_iaa
                obj_work = obj_iaa

            # LP step
            a_work, obj_work, _ = lp_iteration(a_work, 60, 'PDLP')

            if obj_work < best_score:
                best_score = obj_work
                best_seq = a_work.copy()

            it += 1
            if it % 5 == 0:
                print(f"    P4 it{it}: obj={obj_work:.8f}, best={best_score:.8f}, "
                      f"t={time.monotonic()-start_p4:.1f}s")

        print(f"  Phase 4 done: obj={obj_work:.8f}, best={best_score:.8f}, "
              f"iters={it}, time={time.monotonic()-start_p4:.1f}s")

    # === Phase 5: LSE Projected Gradient (high T for escape) ===
    remaining = deadline - time.monotonic()
    phase5_budget = min(80.0, remaining * 0.1)

    if phase5_budget > 15:
        print(f"\n=== Phase 5: LSE Projected Gradient (budget={phase5_budget:.0f}s) ===")
        a_pg, obj_pg = projected_gradient_steps(
            best_seq, T=100.0, lr_init=0.005, time_budget=phase5_budget, verbose=True)
        if obj_pg < best_score:
            best_score = obj_pg
            best_seq = a_pg.copy()
            print(f"  Phase 5 PG: obj={best_score:.8f} **")
        else:
            print(f"  Phase 5 PG: obj={obj_pg:.8f}")

    # === Phase 6: Coordinate Descent ===
    remaining = deadline - time.monotonic()
    phase6_budget = min(150.0, remaining * 0.2)

    if phase6_budget > 20:
        print(f"\n=== Phase 6: Coordinate Descent (budget={phase6_budget:.0f}s) ===")
        a_cd, obj_cd = coordinate_descent(best_seq, time_budget=phase6_budget,
                                          max_sweeps=8, verbose=True)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase 6 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase 6 CD: obj={obj_cd:.8f}")

    # === Phase 7: Final LP polish ===
    remaining = deadline - time.monotonic()
    phase7_budget = min(80.0, remaining * 0.15)

    if phase7_budget > 20:
        print(f"\n=== Phase 7: Final LP Polish (budget={phase7_budget:.0f}s) ===")
        a_work = best_seq.copy()
        obj_work = compute_obj(a_work)
        start_p7 = time.monotonic()
        it = 0

        while time.monotonic() - start_p7 < phase7_budget and it < 30:
            rem = phase7_budget - (time.monotonic() - start_p7)
            if rem < 10:
                break
            a_work, obj_work, _ = lp_iteration(a_work, 50, 'PDLP')
            if obj_work < best_score:
                best_score = obj_work
                best_seq = a_work.copy()
            it += 1

        print(f"  Phase 7 done: obj={obj_work:.8f}, best={best_score:.8f}, "
              f"iters={it}, time={time.monotonic()-start_p7:.1f}s")

    # === Phase 8: n exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 8: n Exploration ===")
        n_cur = len(best_seq)
        n_options = [n_cur - 100, n_cur - 50, n_cur + 50, n_cur + 100, n_cur + 200]
        n_options = [n for n in n_options if 400 <= n <= 2000]
        n_options = sorted(set(n_options))[:4]

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            x_old = np.linspace(0, 1, n_cur)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = np.maximum(a_try, 0)
            s_t = a_try.sum()
            if s_t < 0.01:
                continue
            a_try = a_try / s_t

            # Quick optimization: 2 IAA + 2 LP
            a_try = iaa_flatten(a_try, time_budget=10.0, alpha_start=0.3, alpha_end=0.7)
            for _ in range(3):
                a_try, _, _ = lp_iteration(a_try, 40, 'PDLP')

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_cur = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Final: One more CD pass ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_final, obj_final = coordinate_descent(
            best_seq, time_budget=min(remaining - 5, 30.0), max_sweeps=2)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"\n  Last CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
