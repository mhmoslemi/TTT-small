"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a hybrid of LP alternating minimization, smoothed gradient descent (log-sum-exp),
and adaptive coordinate descent with multi-length exploration.
Key improvements over previous: (1) smoothed gradient descent phase using FFT-based
gradients of the log-sum-exp approximation, (2) faster LP with aggressive warm starts,
(3) analytic 1D subproblem solving in coordinate descent, (4) broader length exploration.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv_fft(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv_fft(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_fast(a, warm_start=None, max_iter=200):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
            M.value = np.max(A @ warm_start)
        except Exception:
            pass

    for solver in ['CLARABEL', 'SCS']:
        try:
            prob.solve(solver=getattr(cp, solver), max_iter=max_iter)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient_descent(a, time_budget, beta_init=30.0, lr_init=0.05, seed=42, verbose=True):
    """
    Smoothed gradient descent using log-sum-exp approximation of max.
    Minimize 2*n*max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    beta = beta_init
    lr = lr_init
    v = np.zeros(n)  # momentum
    step = 0
    patience = 0
    patience_limit = 300

    size_fft = 1
    while size_fft < 2 * n - 1:
        size_fft <<= 1
    size_fft = max(size_fft, 2 * n)

    while time.monotonic() - start < time_budget:
        conv = compute_conv_fft(a)

        # Compute softmax weights (smooth approximation of argmax)
        max_conv = float(np.max(conv))
        shifted = beta * (conv - max_conv)
        exp_shifted = np.exp(shifted)
        sum_exp = float(np.sum(exp_shifted))
        if sum_exp < 1e-300:
            weights = np.zeros(len(conv))
            weights[np.argmax(conv)] = 1.0
        else:
            weights = exp_shifted / sum_exp

        # Gradient: grad_a[j] = 2 * sum_k weights[k] * a[k-j] (with bounds)
        # = 2 * cross_correlation(weights, a)[j]
        a_rev = a[::-1]
        W = rfft(weights, size_fft)
        A_rev = rfft(a_rev, size_fft)
        cross_corr_full = irfft(W * A_rev, size_fft)
        grad = 2.0 * cross_corr_full[n - 1:n - 1 + n]

        # Momentum update
        v = 0.9 * v - lr * grad
        a_trial = a + v

        # Project onto {x >= 0, sum(x) = 1}
        a_trial = np.maximum(a_trial, 0)
        s_trial = float(np.sum(a_trial))
        if s_trial < 1e-10:
            a_trial = a.copy()
        else:
            a_trial = a_trial / s_trial

        obj_trial = compute_obj_norm(a_trial)

        if obj_trial < best_obj - 1e-12:
            best_obj = obj_trial
            best_a = a_trial.copy()
            patience = 0
        else:
            patience += 1

        a = a_trial

        # Adapt learning rate
        if patience > patience_limit:
            lr *= 0.92
            v *= 0.5
            if lr < 1e-7:
                break
        elif patience < patience_limit // 3:
            lr = min(lr * 1.005, 0.5)

        # Adapt beta (increase for sharper approximation)
        if step % 50 == 0:
            beta = min(beta * 1.3, 300.0)

        step += 1
        if verbose and step % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    SGD step {step}: obj={best_obj:.10f}, beta={beta:.1f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")

    return best_a


def coordinate_descent_polish(a, time_budget, seed=42, verbose=True):
    """
    Coordinate descent on normalized sequence (sum=1) with analytic 1D optimization.
    For each coordinate j, the objective as a function of a[j]=v is:
    f(v) = 2n * max(m_ua, max_k g_k(v)) / (1+delta)^2
    where delta = v - a[j] (pre-normalization), and we find the optimal v.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv_fft(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    sweep = 0
    no_improve = 0
    max_no_improve = 4

    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            a_j = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)

            # Unaffected max (precompute)
            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))

            # Affected entries: conv[k] for k in [k_start, k_end]
            # conv[k] = old_conv[k] + 2*delta*a[k-j] (for k != 2j)
            # conv[2j] = old_conv[2j] + delta*(2*a_j + delta) = old_conv[2j] + (v^2 - a_j^2)
            # After normalization by (1+delta)^2:
            # f(v) = 2n * max(m_ua/(1+delta)^2, max_k affected_k(v)/(1+delta)^2)
            #     = 2n * max(m_ua, max_k affected_k(v)) / (1+delta)^2
            # where affected_k(v) = conv_old[k] + 2*(v-a_j)*a[k-j] for k != 2j
            #                      conv_old[2j] + v^2 - a_j^2 for k = 2j

            # To find optimal v, we try a set of candidate values
            # including the breakpoints where the max changes identity
            mean_val = 1.0 / n
            candidates = np.array([
                0.0, mean_val * 0.2, mean_val * 0.5, mean_val, mean_val * 1.5,
                mean_val * 2, mean_val * 3, mean_val * 5, mean_val * 8, mean_val * 12
            ])
            if a_j > 1e-12:
                candidates = np.concatenate([candidates, [
                    a_j * 0.3, a_j * 0.6, a_j * 0.8, a_j * 1.2, a_j * 1.5, a_j * 2.0
                ]])
            candidates = candidates[(candidates >= 0) & (candidates <= 0.5)]
            candidates = np.unique(candidates)

            best_trial_obj = best_obj
            best_trial_val = a_j

            # Also try the point where the quadratic term equals the max linear term
            # conv[2j] new = conv[2j] + v^2 - a_j^2
            # This equals max_unaffected when v^2 = max_unaffected - conv[2j] + a_j^2
            m_ua = max_unaffected
            if m_ua > conv[2 * j] + a_j * a_j:
                v_cross = np.sqrt(m_ua - conv[2 * j])
                if v_cross > 0 and v_cross < 0.5:
                    candidates = np.concatenate([candidates, [v_cross]])

            for val in candidates:
                val = float(val)
                delta = val - a_j
                if abs(delta) < 1e-15:
                    continue

                S_new = 1.0 + delta
                if S_new <= 1e-10:
                    continue

                # Compute max of affected conv entries at new value
                if val < 1e-15:
                    # All affected entries become conv_old[k] - 2*a_j*a[k-j] (for k!=2j)
                    # and conv[2j] - a_j^2
                    idx = np.arange(k_start, k_end + 1)
                    affected = conv[k_start:k_end + 1].copy()
                    mask = (idx == 2 * j)
                    if np.any(~mask):
                        affected[~mask] = conv[k_start:k_end + 1][~mask] - 2.0 * a_j * a[idx[~mask] - j]
                    if np.any(mask):
                        affected[mask] = conv[2 * j] - a_j * a_j
                    max_affected = float(np.max(affected)) if len(affected) > 0 else 0.0
                else:
                    idx = np.arange(k_start, k_end + 1)
                    mask = (idx == 2 * j)
                    if np.any(~mask):
                        affected_linear = conv[k_start:k_end + 1][~mask] + 2.0 * delta * a[idx[~mask] - j]
                        max_affected = float(np.max(affected_linear))
                    else:
                        max_affected = conv[2 * j] + (val * val - a_j * a_j)
                    if np.any(mask):
                        quad_val = conv[2 * j] + (val * val - a_j * a_j)
                        max_affected = max(max_affected, quad_val)

                max_conv_new = max(max_unaffected, max_affected)
                obj_trial = 2.0 * n * max_conv_new / (S_new * S_new)

                if obj_trial < best_trial_obj - 1e-14:
                    best_trial_obj = obj_trial
                    best_trial_val = val

            if best_trial_val != a_j:
                delta = best_trial_val - a_j
                a[j] = best_trial_val
                S_new = 1.0 + delta

                # Update conv in-place
                idx = np.arange(k_start, k_end + 1)
                mask = (idx == 2 * j)
                if np.any(~mask):
                    conv[k_start:k_end + 1][~mask] += 2.0 * delta * a[idx[~mask] - j]
                if np.any(mask):
                    conv[2 * j] += (best_trial_val * best_trial_val - a_j * a_j)

                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S

                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")

    return best_a


def lp_alternate(a, time_budget, seed=42, verbose=True):
    """LP alternating minimization with adaptive step size and warm starts."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    iter_count = 0
    stall = 0

    while time.monotonic() - start < time_budget - 5:
        t0 = time.monotonic()
        g_lp, M_lp = solve_lp_fast(a, warm_start=prev_g)
        if g_lp is None:
            break
        prev_g = g_lp
        iter_count += 1

        # Adaptive line search
        best_local_obj = best_obj
        best_local_a = a.copy()

        t_values = [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]
        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall = 0
            if verbose and iter_count % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP {iter_count}: obj={best_obj:.10f}, elapsed={elapsed:.0f}s")
        else:
            stall += 1
            if stall >= 5:
                # Perturb to escape
                strength = 0.008 + 0.004 * min(stall, 8)
                pert = rng.standard_normal(n)
                a = best_a + pert * np.maximum(best_a, 1e-6)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                prev_g = None
                stall = 0

    if verbose:
        print(f"    LP done: {iter_count} iters, best={best_obj:.10f}")
    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    n_modes = min(10, n // 10)
    for k in range(1, n_modes + 1):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / (k ** 0.7)
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_initialization(n, seed=0):
    """Generate a diverse random initial sequence."""
    rng = np.random.default_rng(seed)
    strategy = seed % 5
    if strategy == 0:
        # Uniform with noise
        a = np.ones(n) + 0.3 * rng.standard_normal(n)
    elif strategy == 1:
        # Bimodal
        a = np.ones(n)
        a[n // 4:3 * n // 4] *= 0.5
    elif strategy == 2:
        # Periodic pattern
        a = 1.0 + 0.3 * np.cos(4 * np.pi * np.arange(n) / n)
    elif strategy == 3:
        # Random sparse
        a = rng.random(n)
        a[a < np.median(a)] *= 0.3
    else:
        # Smooth random
        a = np.cumsum(rng.standard_normal(n))
        a = a - a.min() + 0.1
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample_sequence(a, n_new):
    """Resample sequence to new length using linear interpolation."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(n_new) / n_new
    return a_new / s

        if 300 <= n_try <= 2000:
            n_options.append(n_try)
    n_options = sorted(set(n_options))
    n_options = [n for n in n_options if n != n_best][:8]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        a_try = resample_sequence(best_a, n_try)

        # Quick LP + CD
        alloc_lp = min(remaining * 0.25, 60.0)
        alloc_cd = min(remaining * 0.15, 30.0)
        if alloc_lp > 10:
            a_try = lp_alternate(a_try, time_budget=alloc_lp, seed=n_try * 3, verbose=False)
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_try = coordinate_descent_polish(a_try, time_budget=min(alloc_cd, remaining * 0.3),
                                              seed=n_try * 7, verbose=False)

        obj_try = compute_obj_norm(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            print(f"  ** n={n_try}: {best_obj:.10f} **")
        else:
            print(f"  n={n_try}: {obj_try:.10f}")

    # === Phase 7: Try different initializations ===
    print("\n--- Phase 7: Diverse Restarts ---")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        for restart_i in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            n_r = int(len(best_a) * rng.uniform(0.8, 1.2))
            n_r = max(400, min(1500, n_r))
            a_r = random_initialization(n_r, seed=seed + 5000 + restart_i * 100)
            alloc = min(remaining * 0.2, 120.0)
            if alloc < 15:
                break
            # Quick LP
            a_r = lp_alternate(a_r, time_budget=alloc * 0.5, seed=seed + 6000 + restart_i, verbose=False)
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_r = coordinate_descent_polish(a_r, time_budget=min(alloc * 0.3, remaining * 0.3),
                                                seed=seed + 7000 + restart_i, verbose=False)
            obj_r = compute_obj_norm(a_r)
            if obj_r < best_obj - 1e-12:
                best_obj = obj_r
                best_a = a_r.copy()
                print(f"  ** restart {restart_i} (n={n_r}): {best_obj:.10f} **")
            else:
                print(f"  restart {restart_i} (n={n_r}): {obj_r:.10f}")

    # === Phase 8: Final CD polish ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print("\n--- Phase 8: Final Polish ---")
        a_fin = coordinate_descent_polish(best_a, time_budget=min(remaining * 0.7, 50.0),
                                          seed=seed + 42424, verbose=False)
        obj_fin = compute_obj_norm(a_fin)
        if obj_fin < best_obj - 1e-12:
            best_obj = obj_fin
            best_a = a_fin.copy()
            print(f"  Final: {best_obj:.10f} **")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        final_score = val
    else:
        result = [1.0] * 100
        final_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={final_score:.10f}, time={elapsed:.1f}s ===")
    return result
