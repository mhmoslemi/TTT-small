def mirror_descent_smooth(a, time_budget, tau=20.0, eta_base=1.0, verbose=False):
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Precompute FFT sizes
    N1 = 1
    while N1 < 2*n-1:
        N1 <<= 1
    N2 = 1
    while N2 < 3*n-2:
        N2 <<= 1
    
    a_rev = a[::-1].copy()  # for gradient computation
    
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        # Compute autocorrelation
        fa = rfft(a, N1)
        c = irfft(fa * fa, N1)[:2*n-1]
        
        # Compute softmax weights
        c_max = c.max()
        exp_c = np.exp(tau * (c - c_max))
        s_weights = exp_c / exp_c.sum()
        
        # Compute gradient via FFT
        # R[j] = sum_m s_weights[j+m] * a[m]
        # = (s_weights * a_rev)[j + n - 1] where * is full convolution
        fs = rfft(s_weights, N2)
        fa_rev = rfft(a_rev, N2)
        R = irfft(fs * fa_rev, N2)
        R = R[n-1:n-1+n]  # extract valid part
        grad = 2.0 * R
        
        # Clip gradient to avoid numerical issues
        grad = np.maximum(grad, 0)
        
        # Exponentiated gradient update
        max_grad = grad.max()
        if max_grad < 1e-10:
            break
        eta = eta_base / max_grad
        log_update = -eta * grad
        # For numerical stability, subtract max
        log_update -= log_update.max()
        update = np.exp(log_update)
        a_new = a * update
        a_new = np.maximum(a_new, 1e-15)
        Z = a_new.sum()
        a_new = a_new / Z
        
        # Compute exact objective periodically
        if it % 10 == 0:
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > 100:
                # Reset to best and perturb
                a = best_a * (1 + 0.01 * np.random.randn(n))
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                no_improve = 0
        
        a = a_new
        it += 1
    
    return best_a, best_obj

"""
Hybrid optimization combining smooth mirror descent (exponentiated gradient with
log-sum-exp smoothing and annealing), LP-based directional polishing with diversified
perturbations, adaptive n exploration, and final squeeze. Uses FFT-accelerated
gradient computation for the smooth phase.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def get_fft_size(min_size):
    size = 1
    while size < min_size:
        size <<= 1
    return size


def compute_autocorr(a):
    n = len(a)
    size = get_fft_size(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    c = compute_autocorr(a_norm)
    return 2.0 * n * float(np.max(c))


def mirror_descent_smooth(a, time_budget, tau_start=5.0, tau_end=80.0,
                          eta_scale=1.0, verbose=False):
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N1 = get_fft_size(2 * n - 1)
    N2 = get_fft_size(3 * n - 2)

    it = 0
    no_improve = 0
    eval_count = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / max(time_budget, 1.0))
        tau = tau_start + (tau_end - tau_start) * progress
        # Anneal eta: start aggressive, become conservative
        eta_adj = eta_scale * (1.0 - 0.5 * progress)

        fa = rfft(a, N1)
        c = irfft(fa * fa, N1)[:2 * n - 1]

        c_max = c.max()
        shifted = tau * (c - c_max)
        shifted = np.minimum(shifted, 500.0)
        exp_c = np.exp(shifted)
        s_weights = exp_c / exp_c.sum()

        a_rev = a[::-1]
        fs = rfft(s_weights, N2)
        fa_rev = rfft(a_rev, N2)
        R_full = irfft(fs * fa_rev, N2)
        R = R_full[n - 1:n - 1 + n]
        grad = 2.0 * np.maximum(R, 0.0)

        max_grad = grad.max()
        if max_grad < 1e-12:
            break
        eta = eta_adj / max_grad
        log_update = -eta * grad
        log_update = log_update - log_update.max()
        update = np.exp(log_update)
        a_new = a * update
        a_new = np.maximum(a_new, 1e-15)
        Z = a_new.sum()
        if Z < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / Z

        if it % 5 == 0:
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > 200:
                a = best_a * (1.0 + 0.02 * rng.standard_normal(n))
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                no_improve = 0

        a = a_new
        it += 1

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD it={it}, tau={tau:.1f}, best_obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a, perturbation=None):
    n = len(a)
    a_use = a.copy()
    if perturbation is not None:
        a_use = a_use * (1.0 + perturbation)
        a_use = np.maximum(a_use, 0)
        s = a_use.sum()
        if s < 1e-10:
            a_use = a.copy()
        else:
            a_use = a_use / s
    A = build_conv_matrix(a_use)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def line_search(a, direction, n_coarse=50, n_fine=30, t_max=1.5):
    best_t = 0.0
    best_obj = compute_obj(a)
    ts = np.linspace(0.0, t_max, n_coarse)
    for t in ts:
        a_try = (1.0 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    if 0.0 < best_t < t_max:
        lo = max(0.0, best_t - 0.05)
        hi = min(t_max, best_t + 0.05)
        ts_fine = np.linspace(lo, hi, n_fine)
        for t in ts_fine:
            a_try = (1.0 - t) * a + t * direction
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t
    return best_t, best_obj


def lp_polish_diverse(a, time_budget, max_iters=200, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Solve LP with possible perturbation
        if stall > 3:
            freq = rng.integers(1, max(2, n // 60))
            phase = rng.uniform(0, 2 * np.pi)
            idx = np.arange(n, dtype=np.float64)
            pert = 0.02 * (1 + stall * 0.5) * np.cos(2 * np.pi * freq * idx / n + phase)
            perturbation = pert
        else:
            perturbation = None

        g_lp = solve_lp_direction(a, perturbation)
        if g_lp is None:
            break

        t_opt, obj_opt = line_search(a, g_lp, n_coarse=40, n_fine=25)
        if t_opt > 1e-6 and obj_opt < best_obj - 1e-14:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                best_obj = compute_obj(a_new)
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            # Try direct replacement
            obj_direct = compute_obj(g_lp)
            if obj_direct < best_obj - 1e-14:
                best_obj = obj_direct
                best_a = g_lp.copy()
                a = g_lp.copy()
                stall = 0
            else:
                stall += 1
                if stall > 6:
                    # More aggressive perturbation: blend with diverse init
                    mode = it % 5
                    a_div = diverse_init(n, rng, mode=mode)
                    a = 0.4 * a_div + 0.6 * best_a
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    stall = 0

        it += 1
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"      LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, elapsed={elapsed:.1f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n, dtype=np.float64)
    if mode == 0:
        a = (2 * x - 1) ** 2 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n * 0.15) ** 2) / (2 * (n * 0.08) ** 2)) + \
            np.exp(-((idx - n * 0.85) ** 2) / (2 * (n * 0.08) ** 2)) + 0.01
    elif mode == 2:
        a = np.abs(2 * x - 1) ** 1.5 + 0.03
    elif mode == 3:
        a = 1.0 / (0.1 + 3.0 * (2 * x - 1) ** 2)
    elif mode == 4:
        a = (2 * x - 1) ** 4 + 0.02
    elif mode == 5:
        a = np.cos(np.pi * (2 * x - 1)) ** 2 + 0.01
    else:
        a = np.exp(3.0 * (2 * x - 1) ** 2)
    a = np.maximum(a, 1e-6)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def u_shaped_init(n, sharpness=3.0):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s


def combined_optimize(a, time_budget, verbose=False):
    """Alternate between mirror descent and LP polishing."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    cycles = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break

        # Mirror descent phase
        md_budget = min(remaining * 0.4, 80.0)
        if md_budget > 10:
            a_md, obj_md = mirror_descent_smooth(best_a, md_budget,
                                                  tau_start=10.0, tau_end=60.0,
                                                  eta_scale=1.0)
            if obj_md < best_obj:
                best_obj = obj_md
                best_a = a_md.copy()

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        # LP phase
        lp_budget = min(remaining * 0.6, 150.0)
        if lp_budget > 8:
            a_lp = lp_polish_diverse(best_a, lp_budget, max_iters=100, verbose=verbose)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()

        cycles += 1

    return best_a, best_obj
