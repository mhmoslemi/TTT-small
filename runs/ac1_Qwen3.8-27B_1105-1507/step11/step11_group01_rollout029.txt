import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def obj_val(a):
    """Compute 2n*max(conv)/(sum^2)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def cd_optimize(a, time_budget, seed=42):

    n = len(a)

    if s < 1e-10:
        return a

    conv = compute_conv(a)

                xc = max(0.0, xc)
                fx = f1(xc)
                if fx < best_f:
                    best_f = fx
                    best_x = xc
            
            # Golden section

            lo = max(0.0, best_x - 0.4*scale)

            if hi <= lo:

def solve_lp(a):
    """Solve LP: min M s.t. conv_a(g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)

        return None

        k_start = j
        k_end = min(j + n - 1, 2*n - 2)
        A[k_start:k_end+1, j] = a_n[k_end - j + 1 - n + 1 : k_end + 1]

    A = np.zeros((2*n-1, n))

            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a_n[idx]

def _resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def obj_normalized(a):
    """Compute 2n*max(conv) assuming sum(a)=1."""
    n = len(a)

def obj_val(a):

def subgradient_direction(a, conv=None):
    """Compute subgradient of max(conv(a,a)) w.r.t. a. Used for ordering."""
    n = len(a)
    if conv is None:
        conv = compute_conv(a)
    max_conv = float(np.max(conv))
    tol = max_conv * 1e-8
    active = np.where(conv >= max_conv - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1]
    grad /= max(len(active), 1)
    return grad

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a[:n]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns g or None."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    return g_val
        except Exception:
            continue
    return None

def smoothed_gd(a, time_budget, seed=42, tau_init=50.0, tau_end=500.0):
    """Nesterov-accelerated smoothed gradient descent for escape."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))

    a = a / s

    lr = 3.0 / np.sqrt(n)
    v_mom = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 5:
            break
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress

        y = a + beta * v_mom
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy < 1e-10:
            y = best_a.copy()
            sy = 1.0
        else:
            y = y / sy

        # Smoothed gradient via softmax approximation of max
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient: d/dy [sum_k p_k * conv_k] = 2 * (P^T * F(y))[:n] where P is FFT of p
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        Y = fft(y, N)
        g_full = ifft(np.conj(P) * Y).real
        grad = 2.0 * g_full[:n]

        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-15:
            lr_eff = min(lr, 2.0 / (g_norm + 1e-10))

            lr_eff = lr

        a = a_new

def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x * x + 1e-6))
        return a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        return a / a.sum()
    elif mode == 4:
        a = np.zeros(n)
        w = n // 4
        for start in [0, n - w]:
            a[start:start + w] = np.exp(-((np.arange(w) - w / 2) ** 2) / (2 * (w / 4) ** 2))
        return a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        return a / a.sum()
    else:
        a = rng.exponential(1, n)
        return a / a.sum()
