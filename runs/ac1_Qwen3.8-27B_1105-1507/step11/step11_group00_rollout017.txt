Key innovation: exact 1D optimization per coordinate exploits the structure of

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    n = len(a)
    if n == 0:
        return np.inf

        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

    """Compute gradient of LSE(tau, conv(a,a)) with respect to a."""
    n = len(a)

            if obj_opt < best_obj - tol:
                # Update a and conv incrementally
                dx = x_opt - x
                a[i] = x_opt
                # Update conv
                conv[k_start:k_end + 1] += 2.0 * a_offsets * dx
                if np.any(is_center):
                    conv[2 * i] += dx * dx
                best_obj = obj_opt
                best_a = a.copy()
                pass_improvement += (best_obj + (best_obj - obj_opt))  # just track
                best_obj = obj_opt
                n_changed += 1
            
            # Periodic conv recomputation to avoid drift
            if (idx + 1) % 200 == 0:
                conv = compute_conv(a)
        
        pass_count += 1
        
        if verbose and (pass_count % 50 == 1 or pass_count == 1):
            elapsed = time.monotonic() - start_time
            print(f"    CD pass {pass_count}: obj={best_obj:.10f}, "
                  f"changed={n_changed}/{n}, time={elapsed:.1f}s")
        
        if n_changed == 0:
            stagnant_passes += 1
            if stagnant_passes >= 3:
                if verbose:
                    print(f"    CD converged at pass {pass_count}: obj={best_obj:.10f}")
                break
        else:
            stagnant_passes = 0
    
    return best_a


def gradient_perturb(a, tau=200.0, lr=0.05, rng=None):
    """Apply a small gradient perturbation to escape local minima."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)
    g = smoothed_gradient(a, conv, tau)
    a_new = a - lr * g / (np.linalg.norm(g) + 1e-10)
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new > 1e-10:
        a_new = a_new / s_new
    else:
        a_new = a
    return a_new


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: coordinate descent + gradient perturbation + multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 15.0
    rng = np.random.default_rng(seed)
    
    best_seq = None
    best_score = np.inf
    
    # Initialize from parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")
    
    n_current = len(best_seq)
    print(f"Starting optimization with n={n_current}, score={best_score:.8f}")
    
    # === Phase 1: Intensive Coordinate Descent ===
    print(f"\n=== Phase 1: Coordinate Descent (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 450.0)
    if p1_budget > 30:
        a_cd = coordinate_descent(best_seq, time_budget=p1_budget,
                                  start_time=time.monotonic(), verbose=True,
                                  rng=rng)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase1 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 CD: obj={obj_cd:.8f} (no improvement)")
    
    # === Phase 2: Gradient Perturbation + CD ===
    print(f"\n=== Phase 2: Gradient Perturb + CD ===")
    remaining = deadline - time.monotonic()
    n_perturb = 3
    for p in range(n_perturb):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        
        tau_p = 100.0 + p * 200.0
        lr_p = 0.03 / (1.0 + p * 0.5)
        a_pert = gradient_perturb(best_seq, tau=tau_p, lr=lr_p, rng=rng)
        
        # Additional random perturbation
        a_pert = a_pert * np.exp(0.02 * rng.standard_normal(len(a_pert)))
        a_pert = np.maximum(a_pert, 0)
        s_p = float(np.sum(a_pert))
        if s_p > 1e-10:
            a_pert = a_pert / s_p
        
        # CD from perturbed point
        alloc = min(remaining * 0.25, 120.0)
        if alloc > 15:
            a_try = coordinate_descent(a_pert, time_budget=alloc,
                                       start_time=time.monotonic(), verbose=False,
                                       rng=rng)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Perturb {p}: obj={best_score:.8f} **")
            else:
                print(f"  Perturb {p}: obj={obj_try:.8f}")
    
    # === Phase 3: Multi-n Exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best_cur = len(best_seq)
    
    n_options = []
    for delta in [-200, -100, -50, 50, 100, 200, 400, 600]:
        n_try = n_best_cur + delta
        if 200 <= n_try <= 4000:
            n_options.append(n_try)
    for n_abs in [300, 500, 700, 1000, 1500, 2000]:
        if n_abs not in n_options:
            n_options.append(n_abs)
    n_options = sorted(set(n_options))
    
    # Sort by proximity to current best n (likely to be competitive)
    n_options.sort(key=lambda x: abs(x - n_best_cur))
    n_options = n_options[:6]
    print(f"  Trying n values: {n_options}")
    
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        
        a_try = resample_sequence(best_seq, n_try)
        
        # CD
        alloc_cd = min(remaining * 0.4, 150.0)
        if alloc_cd > 20:
            a_try = coordinate_descent(a_try, time_budget=alloc_cd,
                                       start_time=time.monotonic(), verbose=False,
                                       rng=rng)
        
        # Quick gradient perturb + short CD
        remaining = deadline - time.monotonic()
        if remaining > 40:
            a_try2 = gradient_perturb(a_try, tau=300.0, lr=0.02, rng=rng)
            alloc2 = min(remaining * 0.3, 60.0)
            if alloc2 > 10:
                a_try2 = coordinate_descent(a_try2, time_budget=alloc2,
                                            start_time=time.monotonic(), verbose=False,
                                            rng=rng)
                obj2 = compute_obj(a_try2)
                if obj2 < compute_obj(a_try):
                    a_try = a_try2
        
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")
    
    # === Phase 4: Final Intensive CD ===
    print(f"\n=== Phase 4: Final CD Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 200.0)
        a_final = coordinate_descent(best_seq, time_budget=alloc,
                                     start_time=time.monotonic(), verbose=True,
                                     rng=rng)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")
    
    # === Phase 5: Sparsification attempts ===
    print(f"\n=== Phase 5: Sparsification ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_cur = len(best_seq)
        # Try zeroing small elements
        a_spar = best_seq.copy()
        s_cur = float(np.sum(a_spar))
        a_norm = a_spar / s_cur
        # Sort by value, try zeroing smallest
        sorted_idx = np.argsort(a_norm)
        improved = False
        for frac in [0.05, 0.1, 0.2]:
            n_zero = int(n_cur * frac)
            a_test = a_norm.copy()
            a_test[sorted_idx[:n_zero]] = 0
            s_test = float(np.sum(a_test))
            if s_test > 0.01:
                a_test = a_test / s_test
                obj_test = compute_obj(a_test)
                if obj_test < best_score:
                    best_score = obj_test
                    best_seq = a_test.copy()
                    improved = True
                    print(f"  Sparsify {frac:.0%}: obj={best_score:.8f} **")
                    # Continue CD from here
                    remaining = deadline - time.monotonic()
                    if remaining > 20:
                        a_test = coordinate_descent(a_test, time_budget=min(remaining*0.5, 60.0),
                                                    start_time=time.monotonic(), verbose=False, rng=rng)
                        obj_test2 = compute_obj(a_test)
                        if obj_test2 < best_score:
                            best_score = obj_test2
                            best_seq = a_test.copy()
                            print(f"  After CD: obj={best_score:.8f} **")
                    break
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
