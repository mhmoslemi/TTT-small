import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_conv(a_norm):
    """Compute 2n*max(conv) for already-normalized a (sum=1)."""
    n = len(a_norm)
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n


def build_full_conv_matrix(a):
    """Build A matrix where A[k, m] = a[k-m] if 0 <= k-m < n, else 0.
    Shape: (2n-1, n). The constraint A @ g <= M means (a*g)[k] <= M for all k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            A[k, m_start:m_end + 1] = a[k - m_end:k - m_start + 1][::-1]
    return A


def solve_full_lp(a, solver_list=None):
    """Solve the full LP: min M s.t. (a*g)[k] <= M (all k), sum(g)=1, g>=0.
    Returns normalized g or None."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_full_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.ECOS, cp.SCS]

    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def solve_truncated_lp(a, top_m, solver_list=None):

    n = len(a)
    a_norm = a / float(np.sum(a))

    A = np.zeros((len(top_indices), n))

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

def lp_blend_line_search(a, g_lp, n_points=30):
    """Golden-section-like line search for best blend between a and g_lp."""
    best_t = 0.0
    best_obj = compute_obj_conv(a)

    # Coarse grid
    ts = np.linspace(0.05, 1.5, n_points)
    for t in ts:

        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj_try = compute_obj_conv(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t

    # Refine around best_t
    if best_t > 0:
        for _ in range(3):
            step = max(0.01, 1.5 / n_points)
            for dt in [-step, -step/2, step/2, step]:
                t_try = best_t + dt
                if t_try < 0 or t_try > 2.0:

                a_try = (1 - t_try) * a + t_try * g_lp
                a_try = np.maximum(a_try, 0)

                obj_try = compute_obj_conv(a_try)

def lse_objective(a, T):
    """Smoothed log-sum-exp objective for normalized a."""
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)
    exp_vals = np.exp(T * (conv - cmax))
    lse = cmax + np.log(np.sum(exp_vals)) / T
    return 2.0 * n * lse


def lse_gradient(a, T):
    """Gradient of 2n * LSE_T(conv(a,a)) for normalized a."""
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)
    exp_vals = np.exp(T * (conv - cmax))
    w = exp_vals / exp_vals.sum()

    grad = np.zeros(n)
    for i in range(n):
        lo1 = max(0, i)
        hi1 = min(2 * n - 2, i + n - 1)
        if lo1 <= hi1:
            grad[i] += np.dot(w[lo1:hi1 + 1], a[lo1 - i:hi1 - i + 1])
        lo2 = max(0, i - n + 1)
        hi2 = min(2 * n - 2, i)
        if lo2 <= hi2:
            grad[i] += np.dot(w[lo2:hi2 + 1], a[i - hi2:i - lo2 + 1][::-1])
    return 2.0 * n * grad


def lse_gradient_descent(a, time_budget, T_start=5.0, T_end=60.0, verbose=False):
    """LSE smoothed gradient descent with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_conv(a)
    start = time.monotonic()
    lr = 0.05 / np.sqrt(n)
    it = 0
    best_it = 0
    patience = 500

    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** frac

        grad = lse_gradient(a, T)
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-14:
            break

        a_old = a.copy()
        obj_old = lse_objective(a, T)
        t = lr
        improved = False
        for _ in range(8):
            a_try = a_old - t * grad
            a_try = project_simplex(a_try)
            obj_try = lse_objective(a_try, T)
            if obj_try < obj_old - 1e-13:
                a = a_try
                improved = True
                break
            t *= 0.5

        if not improved:
            break

        obj_true = compute_obj_conv(a)

            best_obj = obj_true
            best_a = a.copy()
            best_it = it

            lr *= 0.5

        if verbose and it % 200 == 0:
            print(f"      LSE it{it}: obj={best_obj:.8f}, T={T:.1f}, t={time.monotonic()-start:.1f}s")

    return best_a


def mirror_descent(a, time_budget, verbose=False):
    """Mirror descent with entropy mirror (multiplicative updates)."""
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / float(np.sum(a))

    best_a = a.copy()
    best_obj = compute_obj_conv(a)
    start = time.monotonic()
    eta = 1.0 / np.sqrt(n)
    it = 0
    best_it = 0
    patience = 300

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        cmax = np.max(conv)
        # Subgradient at the max peak
        k_max = int(np.argmax(conv))

        # Gradient of (a*a)[k_max] w.r.t. a
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_max - i
            j2 = i - k_max
            if 0 <= j1 < n:
                grad[i] += a[j1]
            if 0 <= j2 < n:
                grad[i] += a[j2]
        grad *= 2.0  # because (a*a)[k] has each cross-term twice

        # Mirror descent update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 0.01:
            eta *= 0.5
            continue
        a_new = a_new / s

        obj_new = compute_obj_conv(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            eta *= 1.05
        else:
            eta *= 0.8

        a = a_new
        it += 1
        if it - best_it > patience:
            a = best_a.copy()
            eta = 0.5 / np.sqrt(n)
            patience = max(50, patience // 2)
            best_it = it

        if verbose and it % 500 == 0:
            print(f"      MD it{it}: obj={best_obj:.8f}, eta={eta:.6f}, t={time.monotonic()-start:.1f}s")

    return best_a


def coordinate_descent(a, time_budget, max_sweeps=5, verbose=False):

    n = len(a)

    for sweep in range(max_sweeps):

            break
        sweep_improved = False

        for i in range(n):
            if time.monotonic() - start >= time_budget:
                break

            ai_old = a[i]

                candidates = [0.0, 0.01, 0.02]

                candidates = [0.0, ai_old * 0.5, ai_old * 0.75, ai_old * 1.25, ai_old * 1.5,

def u_shaped_init(n, sharpness):
    """U-shaped initialization: mass concentrated at edges."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.01 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def arcsine_init(n):
    """Arcsine distribution."""
    x = np.linspace(0.001, 0.999, n)
    a = 1.0 / np.sqrt(np.maximum(1e-10, 1 - (2 * x - 1) ** 2))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def power_init(n, alpha):
    """Power-law edge concentration."""
    x = np.linspace(0, 1, n)
    d = np.abs(x - 0.5)
    a = np.maximum(d, 1.0 / (2 * n)) ** (-alpha) + 0.01
    a = np.maximum(a, 0)
    s = float(np.sum(a))

        return np.ones(n) / n
    return a / s


def beta_init(n, alpha, beta):
    """Beta distribution initialization (discretized)."""
    x = (np.arange(n) + 0.5) / n
    a = x ** (alpha - 1) * (1 - x) ** (beta - 1)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_sequence(a, new_n):

    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)

    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 0.01:
        return np.ones(new_n) / new_n
    return a_new / s

        n_candidates.remove(n_parent)
    n_candidates.insert(0, n_parent)

    n_results = {}  # n -> (best_a, best_obj)

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Create initialization
        if parent is not None:
            a_init = resample_sequence(parent, n_try)
        else:
            a_init = u_shaped_init(n_try, 3.0)

        # Quick LP polish (short budget)
        t_quick = min(15.0, remaining * 0.15)
        a_quick = full_lp_polish(a_init, time_budget=t_quick, use_full=(n_try <= 800),
                                 max_iters=30, verbose=False)
        obj_quick = compute_obj_conv(a_quick)

        n_results[n_try] = (a_quick, obj_quick)
        print(f"  n={n_try}: obj={obj_quick:.8f} ({t_quick:.1f}s)")

        if obj_quick < best_score:
            best_score = obj_quick
            best_seq = a_quick.copy()
            print(f"    ** New best: {best_score:.8f}")

    # Also try parametric inits at the current best n
    n_best = len(best_seq)
    print(f"\n  Trying parametric inits at n={n_best}:")
    parametric_inits = [
        ("beta(0.5,0.5)", beta_init(n_best, 0.5, 0.5)),
        ("beta(0.3,0.3)", beta_init(n_best, 0.3, 0.3)),
        ("beta(0.7,0.7)", beta_init(n_best, 0.7, 0.7)),
        ("beta(2,2)", beta_init(n_best, 2.0, 2.0)),
        ("power1.5", power_init(n_best, 1.5)),
        ("power2.5", power_init(n_best, 2.5)),
        ("u_sharp10", u_shaped_init(n_best, 10.0)),
        ("u_sharp0.5", u_shaped_init(n_best, 0.5)),
    ]

    for name, a_init in parametric_inits:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        obj_init = compute_obj_conv(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"    {name}: obj={obj_init:.8f} **")
        else:
            print(f"    {name}: obj={obj_init:.8f}")

    # === Phase 2: Intensive optimization on top n values ===
    print("\n=== Phase 2: Intensive Optimization ===")

    # Sort n results by objective
    sorted_ns = sorted(n_results.keys(), key=lambda x: n_results[x][1])
    top_ns = sorted_ns[:4]  # Top 4 n values

    # Also include the current best n
    if n_best not in top_ns:
        top_ns.append(n_best)

    print(f"  Top n values: {top_ns}")

    for n_try in top_ns:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        # Use the best initialization for this n
        if n_try in n_results:
            a_start = n_results[n_try][0].copy()
        else:
            a_start = resample_sequence(best_seq, n_try)

        # Run intensive optimization
        t_alloc = min(remaining * 0.4, 200.0)
        print(f"\n  Intensive optimization at n={n_try} ({t_alloc:.0f}s):")
        a_opt, obj_opt = optimize_at_n(a_start, n_try, time_budget=t_alloc, verbose=True)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"    ** New global best: n={n_try}, obj={best_score:.8f}")
        else:
            print(f"    n={n_try}: obj={obj_opt:.8f}")

    # === Phase 3: Try slightly different n around the best ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 3: Fine n Tuning (around n={len(best_seq)}) ===")
        n_cur = len(best_seq)
        n_fine = [n_cur - 50, n_cur - 25, n_cur + 25, n_cur + 50]
        n_fine = [n for n in n_fine if 100 <= n <= 2500]

        for n_try in n_fine:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            a_try = resample_sequence(best_seq, n_try)
            t_alloc = min(remaining * 0.3, 60.0)
            a_opt, obj_opt = optimize_at_n(a_try, n_try, time_budget=t_alloc, verbose=False)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f}")
            else:
                print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 4: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 4: Final Polish (remaining {remaining:.0f}s) ===")
        n_final = len(best_seq)

        # LP polish
        t_lp = min(remaining * 0.4, 120.0)
        if t_lp > 10:
            a_f1 = full_lp_polish(best_seq, time_budget=t_lp, use_full=(n_final <= 1000),
                                  max_iters=500, verbose=True)
            obj_f1 = compute_obj_conv(a_f1)
            if obj_f1 < best_score:
                best_score = obj_f1
                best_seq = a_f1.copy()
                print(f"  Final LP: obj={best_score:.8f} **")
            else:
                print(f"  Final LP: obj={obj_f1:.8f}")

        # LSE GD
        remaining = deadline - time.monotonic()
        t_lse = min(remaining * 0.25, 60.0)
        if t_lse > 5:
            a_f2 = lse_gradient_descent(best_seq, time_budget=t_lse,
                                        T_start=10.0, T_end=80.0, verbose=False)
            obj_f2 = compute_obj_conv(a_f2)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final LSE: obj={best_score:.8f} **")
            else:
                print(f"  Final LSE: obj={obj_f2:.8f}")

        # Coordinate descent
        remaining = deadline - time.monotonic()
        t_cd = min(remaining * 0.3, 60.0)
        if t_cd > 5:
            a_f3 = coordinate_descent(best_seq, time_budget=t_cd, max_sweeps=5, verbose=True)
            obj_f3 = compute_obj_conv(a_f3)
            if obj_f3 < best_score:
                best_score = obj_f3
                best_seq = a_f3.copy()
                print(f"  Final CD: obj={best_score:.8f} **")
            else:
                print(f"  Final CD: obj={obj_f3:.8f}")

        # One more LP
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_f4 = full_lp_polish(best_seq, time_budget=remaining - 5,
                                  use_full=(n_final <= 1000), max_iters=100, verbose=False)
            obj_f4 = compute_obj_conv(a_f4)
            if obj_f4 < best_score:
                best_score = obj_f4
                best_seq = a_f4.copy()
                print(f"  Last LP: obj={best_score:.8f} **")

    # === Phase 5: Diverse restarts if time permits ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 5: Diverse Restarts ===")
        n_cur = len(best_seq)
        for r in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            # Create diverse init
            mode = r
            x = np.linspace(0, 1, n_cur)
            if mode == 0:
                a_r = np.exp(-3 * (2 * x - 1) ** 4) + 0.01
            elif mode == 1:
                a_r = 1.0 / (1.0 + 5.0 * np.abs(2 * x - 1) ** 3)
            elif mode == 2:
                a_r = (1 + 0.5 * np.sin(2 * np.pi * x)) ** 2
            else:
                a_r = np.exp(-2 * (2 * x - 1) ** 2) * np.exp(-2 * (2 * x - 1) ** 6) + 0.01

            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            a_r = a_r / s_r

            # Blend with best
            blend = 0.3
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

            t_r = min(remaining * 0.25, 30.0)
            a_r = full_lp_polish(a_r, time_budget=t_r, use_full=(n_cur <= 1000),
                                 max_iters=50, verbose=False)
            obj_r = compute_obj_conv(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # Final validation and output
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_conv(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
