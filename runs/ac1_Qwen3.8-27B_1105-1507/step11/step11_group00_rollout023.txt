import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def subgradient_at_peak(a, conv):
    """Exact subgradient at the peak index of conv. Returns (gradient, peak_idx)."""
    n = len(a)
    m = int(np.argmax(conv))
    grad = np.zeros(n)
    lo = max(0, m - (n - 1))
    hi = min(n, m + 1)
    js = np.arange(lo, hi)
    grad[js] = 2.0 * a[m - js]
    return grad, m


def topk_subgradient(a, conv, k=3):
    """Subgradient using top-k peaks with uniform weights."""
    n = len(a)
    idx = np.argsort(conv)[-k:]
    grad = np.zeros(n)
    for m in idx:
        lo = max(0, m - (n - 1))
        hi = min(n, m + 1)
        js = np.arange(lo, hi)
        grad[js] += (2.0 / k) * a[m - js]
    return grad


def lse_gradient(a, conv, tau):
    """Gradient of LSE(tau, conv) w.r.t. a. Uses FFT for efficiency."""
    n = len(a)
    shifted = tau * (conv - conv.max())
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()

    a_rev = a[::-1]
    N_fft = 1
    while N_fft < 3 * n:
        N_fft <<= 1
    c = irfft(rfft(p, N_fft) * rfft(a_rev, N_fft), N_fft)
    return 2.0 * c[n - 1:2 * n - 1]


def mirror_step(a, grad, eta):
    """Multiplicative weights update: a_new = a * exp(-eta * grad) / sum(...)."""
    exponent = -eta * grad
    exponent = np.clip(exponent, -50, 50)
    a_new = a * np.exp(exponent)
    s = float(np.sum(a_new))
    if s < 1e-300:
        return a.copy()
    return a_new / s


def mirror_step_bisection(a, grad, eta_max, eta_min=1e-8, best_obj=None):

    if best_obj is None:

    best_a = a.copy()
    best_f = best_obj

    # Try a range of step sizes
    etas = np.logspace(np.log10(eta_min), np.log10(eta_max), 20)

        f_try = compute_obj(a_try)
        if f_try < best_f:

            best_a = a_try.copy()

def lp_direction(a, n_constraints=None, verbose=False):
    """
    Solve LP: min M s.t. 2*conv(a,g)_m - conv(a,a)_m <= M for selected m,
    sum(g) = 1, g >= 0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv_aa = compute_conv(a_norm)

    if n_constraints is None:
        n_constraints = min(2 * n - 1, max(50, n))

    # Select top constraints (where conv_aa is largest, as these are most likely binding)
    all_idx = np.argsort(conv_aa)[::-1][:n_constraints]
    all_idx = np.sort(all_idx)

    # Build constraint matrix
    A_sel = np.zeros((len(all_idx), n))
    for i, m in enumerate(all_idx):
        lo = max(0, m - (n - 1))
        hi = min(n, m + 1)
        js = np.arange(lo, hi)
        A_sel[i, js] = 2.0 * a_norm[m - js]

    b_sel = conv_aa[all_idx]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_sel @ g - M <= b_sel, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, g, n_evals=30):
    """Line search for best t in [0,1] to minimize obj((1-t)*a + t*g)."""
    ts = np.linspace(0.005, 0.995, n_evals)
    objs = np.full(n_evals, np.inf)
    for i, t in enumerate(ts):
        a_try = np.maximum((1 - t) * a + t * g, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            objs[i] = compute_obj(a_try / s)

    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]

    # Parabolic refinement
    if 0 < best_idx < n_evals - 1:
        t0, t1, t2 = ts[best_idx - 1], ts[best_idx], ts[best_idx + 1]
        f0, f1, f2 = objs[best_idx - 1], objs[best_idx], objs[best_idx + 1]
        denom = f0 - 2.0 * f1 + f2
        if abs(denom) > 1e-15:
            t_par = t1 - 0.5 * (t2 - t0) * (f0 - f1) / denom
            t_par = float(np.clip(t_par, 0.005, 0.995))

            s_par = float(np.sum(a_par))
            if s_par > 1e-10:
                f_par = compute_obj(a_par / s_par)
                if f_par < best_obj:
                    best_t = t_par
                    best_obj = f_par

    return best_t, best_obj

                break

            if it > 20:

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            eta_base = min(eta_base * 1.05, 5.0)  # Slightly increase step
        else:
            stall += 1
            # Try smaller step
            for frac in [0.5, 0.2, 0.1]:
                a_half = mirror_step(a, g, eta * frac)
                obj_half = compute_obj(a_half)
                if obj_half < best_obj - 1e-13:
                    best_obj = obj_half
                    best_a = a_half.copy()
                    a = a_half
                    stall = 0
                    eta_base = eta * frac
                    break
            else:
                if stall >= 15:
                    # Perturb to escape
                    strength = 0.03 * (1.0 + progress)
                    a_pert = a * np.exp(strength * rng.standard_normal(n))
                    a_pert = np.maximum(a_pert, 0)
                    s_pert = float(np.sum(a_pert))
                    if s_pert > 1e-10:
                        a = a_pert / s_pert
                    eta_base = max(eta_base * 0.5, 0.01)
                    stall = 0

        a = a_new
        it += 1

        if verbose and (it % 50 == 0 or it == 1):
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, "
                  f"eta={eta:.4f}, stall={stall}, time={time.monotonic()-start:.0f}s")

    return best_a


def nesterov_gd_optimize(a, time_budget=150.0, tau_init=50.0, tau_end=3000.0,
                         verbose=False):
    """Nesterov accelerated gradient descent on LSE-smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a.copy()
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.8
    base_lr = 1.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * (progress ** 1.3)

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            v = np.zeros(n)
        else:
            y = y / s_y

        conv_y = compute_conv(y)
        g = lse_gradient(y, conv_y, tau)

        # Adaptive learning rate
        g_norm = float(np.linalg.norm(g))
        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm + 1e-10)
            adapt = min(max(adapt, 0.2 * base_lr), 3.0 * base_lr)
            lr = 0.9 * lr + 0.1 * adapt

        # Barzilai-Borwein
        if prev_g is not None and it % 10 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / max(sy, 1e-15)
                bb = min(max(bb, 1e-10), 5.0 / np.sqrt(n))
                lr = 0.8 * lr + 0.2 * bb

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1

        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it

    if verbose:
        print(f"    NesterovGD: obj={best_obj:.8f}, iters={it}")
    return best_a


def lp_driven_optimize(a, time_budget=200.0, verbose=True):
    """LP-driven optimization with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a.copy()
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    no_improve = 0
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time.monotonic() and time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        # Solve LP with reduced constraints
        n_cons = min(2 * n - 1, max(100, n // 2))
        g, M = lp_direction(a, n_constraints=n_cons)

        if g is None:
            # Fallback to gradient direction
            conv = compute_conv(a)
            g = lse_gradient(a, conv, 200.0)
            g = np.maximum(g, 0)
            s_g = float(np.sum(g))
            if s_g < 1e-10:
                break
            g = g / s_g

        # Line search
        best_t, obj_t = line_search(a, g, n_evals=25)

        if obj_t < best_obj - 1e-11:
            a_try = np.maximum((1 - best_t) * a + best_t * g, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                best_obj = obj_t
                best_a = a_try.copy()
                a = a_try
                no_improve = 0
                if verbose and it % 3 == 0:
                    print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, "
                          f"M={M:.6f}, time={time.monotonic()-start:.0f}s")
        else:
            no_improve += 1
            if no_improve >= 5:
                # Perturb and reset
                a_pert = a * np.exp(0.02 * rng.standard_normal(n))
                a_pert = np.maximum(a_pert, 0)
                s_pert = float(np.sum(a_pert))
                if s_pert > 1e-10:
                    a = a_pert / s_pert
                no_improve = 0
                if it > 50:
                    if verbose:
                        print(f"    LP: stopping after {it} iterations")
                    break

        it += 1

    return best_a


def propose_candidate(height_sequence_1, time_budget=1500):
    """
    Main search function: multi-phase optimization combining mirror descent,
    LP-guided steps, Nesterov GD, coordinate polishing, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(time_budget) - 25.0

    best_seq = None
    best_score = np.inf

    # Initialize from parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Mirror Descent ===
    print(f"\n=== Phase 1: Mirror Descent (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.28, 350.0)
    if p1_budget > 30:
        a_md = mirror_descent_optimize(best_seq, time_budget=p1_budget,
                                       tau_init=20.0, tau_end=2000.0,
                                       verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Phase1 MD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 MD: obj={obj_md:.8f} (no improvement)")

    # === Phase 2: LP-Driven ===
    print(f"\n=== Phase 2: LP-Driven ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 250.0)
    if p2_budget > 20:
        a_lp = lp_driven_optimize(best_seq, time_budget=p2_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Nesterov GD ===
    print(f"\n=== Phase 3: Nesterov GD ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 150.0)
    if p3_budget > 15:
        a_gd = nesterov_gd_optimize(best_seq, time_budget=p3_budget,
                                    tau_init=100.0, tau_end=3000.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase3 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 GD: obj={obj_gd:.8f} (no improvement)")

    # === Phase 4: Coordinate Polish ===
    print(f"\n=== Phase 4: Coordinate Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_pol, obj_pol = coordinate_polish(best_seq, max_iters=300, verbose=True)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  Phase4 Polish: obj={best_score:.8f} **")

    # === Phase 5: Second MD round (higher tau) ===
    print(f"\n=== Phase 5: Second MD (higher tau) ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.15, 150.0)
    if p5_budget > 20:
        a_md2 = mirror_descent_optimize(best_seq, time_budget=p5_budget,
                                        tau_init=300.0, tau_end=8000.0,
                                        verbose=True)
        obj_md2 = compute_obj(a_md2)
        if obj_md2 < best_score:
            best_score = obj_md2
            best_seq = a_md2.copy()
            print(f"  Phase5 MD2: obj={best_score:.8f} **")
        else:
            print(f"  Phase5 MD2: obj={obj_md2:.8f} (no improvement)")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = []
    for delta in [-200, -100, -50, 50, 100, 200, 300, 500]:
        n_try = n_best + delta
        if 400 <= n_try <= 2500 and n_try not in n_options:
            n_options.append(n_try)
    for n_abs in [600, 900, 1200, 1500, 1800, 2000]:
        if n_abs not in n_options and 400 <= n_abs <= 2500:
            n_options.append(n_abs)
    n_options = sorted(set(n_options))

    # Sort by proximity to current best n (try nearby first)
    n_options.sort(key=lambda x: abs(x - n_best))
    n_options = n_options[:6]
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            a_try = np.ones(n_try) / n_try
        else:
            a_try = a_try / s_try

        # MD
        alloc_md = min(remaining * 0.35, 70.0)
        if alloc_md > 15:
            a_try = mirror_descent_optimize(a_try, time_budget=alloc_md,
                                            tau_init=50.0, tau_end=1000.0, verbose=False)

        # LP
        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.25, 50.0)
        if alloc_lp > 15:
            a_try = lp_driven_optimize(a_try, time_budget=alloc_lp, verbose=False)

        # Coordinate polish
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_try, _ = coordinate_polish(a_try, max_iters=100, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse Restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    rng = np.random.default_rng(99)

    restart_types = ["sym", "fourier", "smooth", "random"]
    for name in restart_types:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        if name == "sym":
            a_r = (best_seq + best_seq[::-1]) / 2.0
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r
            else:
                a_r = best_seq.copy()
        elif name == "fourier":
            pert = np.zeros(n_cur)
            for k in range(1, 8):
                freq = k * 2.0 * np.pi / n_cur
                amp = 0.05 * rng.uniform(0.5, 1.5) / k
                phase = rng.uniform(0, 2.0 * np.pi)
                pert += amp * np.cos(freq * np.arange(n_cur) + phase)
            a_r = best_seq * np.exp(pert)
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r
        elif name == "smooth":
            ks = max(7, n_cur // 10)
            if ks % 2 == 0:
                ks += 1
            a_smooth = uniform_filter1d(a_r if 'a_r' in dir() else best_seq, size=ks)
            a_r = 0.6 * best_seq + 0.4 * a_smooth
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r
        else:
            a_r = 0.7 * best_seq + 0.3 * (rng.uniform(0, 1, n_cur) / n_cur)
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r

        # Optimize
        alloc_md = min(remaining * 0.3, 50.0)
        if alloc_md > 12:
            a_r = mirror_descent_optimize(a_r, time_budget=alloc_md,
                                          tau_init=100.0, tau_end=2000.0, verbose=False)

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_r, _ = coordinate_polish(a_r, max_iters=80, verbose=False)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart ({name}): obj={best_score:.8f} **")
        else:
            print(f"  Restart ({name}): obj={obj_r:.8f}")

    # === Phase 8: Final Intensive Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau MD
    if remaining > 40:
        alloc = min(remaining * 0.4, 120.0)
        a_final = mirror_descent_optimize(best_seq, time_budget=alloc,
                                          tau_init=1000.0, tau_end=15000.0,
                                          verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.35, 80.0)
        a_final2 = lp_driven_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final coordinate polish (long)
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final3, obj_final3 = coordinate_polish(best_seq, max_iters=500, verbose=True)
        if obj_final3 < best_score:
            best_score = obj_final3
            best_seq = a_final3.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
