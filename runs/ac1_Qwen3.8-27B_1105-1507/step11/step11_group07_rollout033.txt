"""
Hybrid optimization for step function peak minimization.
Combines: (1) Adam-based soft-max gradient descent, 
(2) Nesterov-accelerated soft-max GD,
(3) LP polishing with diverse directions,
(4) Greedy coordinate descent,

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_soft_weights(conv, beta):
    """Compute softmax weights over convolution values."""
    c_max = float(np.max(conv))
    exp_vals = np.exp(beta * (conv - c_max))
    total = float(np.sum(exp_vals))
    if total < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_vals / total


def compute_soft_gradient(a, p):
    """Gradient of soft-max objective via cross-correlation."""
    n = len(a)
    a_rev = a[::-1]
    size = 1
    while size < n + len(p) - 1:
        size <<= 1
    A = rfft(a_rev, size)
    P = rfft(p, size)
    full = irfft(A * P, size)
    q = full[n - 1:n - 1 + n]
    return 2.0 * q


def adam_softmax_gd(a, time_budget, beta_start, beta_end, lr=0.005,
                    beta1=0.9, beta2=0.999, eps=1e-8, verbose=True, seed=42):
    """Adam-based soft-max gradient descent on simplex."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress
        
        conv = compute_conv(a)
        p = compute_soft_weights(conv, beta)
        grad = compute_soft_gradient(a, p)
        
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Adam step with decaying lr
        lr_t = lr * (1.0 - 0.5 * progress)
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = t
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Reset Adam state and perturb
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                pert = rng.normal(0, 0.003, n)
                a = project_simplex(best_a + pert)
                stall = 0
        
        a = a_new
        
        if verbose and t % 2000 == 0:
            print(f"    Adam it{t}: obj={best_obj:.8f}, beta={beta:.1f}, t={elapsed:.1f}s")
        
        if t - best_it > 10000 and t > 5000:
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr *= 0.8
    
    return best_a, best_obj


def nesterov_softmax_gd(a, time_budget, beta_start, beta_end, lr=0.004,
                        momentum=0.93, verbose=True, seed=42):
    """Nesterov-accelerated soft-max gradient descent."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    t = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress
        lr_t = lr * (1.0 - 0.4 * progress)

        conv = compute_conv(y)
        p = compute_soft_weights(conv, beta)
        grad = compute_soft_gradient(y, p)
        
        v = momentum * v - lr_t * grad
        a_new = project_simplex(a + v)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = t
            stall = 0
        else:
            stall += 1
            if stall > 300:
                v = np.zeros(n)
                pert = rng.normal(0, 0.002 * (1 + stall/100), n)
                a = project_simplex(best_a + pert)
                stall = 0
        
        a = a_new
        t += 1
        
        if verbose and t % 2000 == 0:
            print(f"    Nesterov it{t}: obj={best_obj:.8f}, beta={beta:.1f}, t={elapsed:.1f}s")
        
        if t - best_it > 8000 and t > 3000:
            a = best_a.copy()
            v = np.zeros(n)
            lr *= 0.85
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, max_time=5.0):
    """Solve LP: min M s.t. conv(a_norm, g)_k <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    start = time.monotonic()
    solvers = [cp.CLARABEL, cp.SCS, cp.ECOS, cp.CVXOPT]
    for solver in solvers:
        if time.monotonic() - start > max_time:
            break
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def greedy_coordinate_optimize(a, time_budget, verbose=True, seed=42):

    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # First pass: try zeroing out small coordinates
    obj_cache = compute_conv(a)
    current_max = float(np.max(obj_cache))
    
    # Sort coordinates by value (smallest first)
    order = np.argsort(a)[:n // 2]  # Try zeroing the smallest half
    
    for i in order:
        if time.monotonic() - start >= time_budget:
            break
        if a[i] < 0.01 / n:  # Only try very small coordinates
            a_trial = a.copy()
            a_trial[i] = 0
            s = float(np.sum(a_trial))
            if s > 0.01:
                a_trial = a_trial / s
                conv_t = compute_conv(a_trial)
                obj_t = 2.0 * n * float(np.max(conv_t))
                if obj_t < best_obj - 1e-10:
                    best_obj = obj_t
                    best_a = a_trial.copy()
                    a = a_trial.copy()
    
    # Second phase: pairwise mass transfer
    # For random pairs, try moving mass from i to j
    max_rounds = 50
    for rnd in range(max_rounds):

            break
        improved = False
        for _ in range(n):
            if time.monotonic() - start >= time_budget:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if a[i] < 1e-10:
                continue
            # Try moving fraction of a[i] to a[j]
            for frac in [0.1, 0.3, 0.5, 0.8]:
                a_trial = a.copy()
                a_trial[i] -= frac * a[i]
                a_trial[j] += frac * a[i]
                if a_trial[i] < 0:
                    a_trial[i] = 0
                s = float(np.sum(a_trial))
                a_trial = a_trial / s
                conv_t = compute_conv(a_trial)
                obj_t = 2.0 * n * float(np.max(conv_t))
                if obj_t < best_obj - 1e-12:
                    best_obj = obj_t
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break
        
        if not improved and rnd > 5:
            if verbose:
                print(f"    Greedy coord: converged at round {rnd}, obj={best_obj:.8f}")
            break
    
    if verbose:
        print(f"    Greedy coord: final obj={best_obj:.8f}, time={time.monotonic()-start:.1f}s")
    
    return best_a, best_obj


def lp_polish_diverse(a, time_budget, verbose=False, seed=42):
    """
    LP polishing with diverse directions.
    In addition to the standard LP direction, also tries:
    - Random feasible directions
    - Anti-LP direction (away from worst peaks)
    - Multiple blend factors
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(seed)
    
    conv = compute_conv(a)
    peak_pos = np.argmax(conv)
    
    while time.monotonic() - start < time_budget and it < 200:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break
        
        # Standard LP direction
        g_lp, M_lp = solve_lp_minpeak(a, max_time=min(5.0, remaining * 0.1))
        
        directions = []
        if g_lp is not None:
            directions.append((g_lp, 'lp'))
        
        # Random direction on simplex
        g_rand = rng.dirichlet(np.ones(n) * 0.5)
        directions.append((g_rand, 'rand'))
        
        # Peak-avoiding direction: reduce mass at positions contributing to peak
        conv = compute_conv(a)
        peak_idx = np.where(conv > 0.9 * np.max(conv))[0]
        g_anti = a.copy()
        for k in peak_idx:
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                g_anti[i] *= 0.95
        g_anti = project_simplex(g_anti)
        directions.append((g_anti, 'anti'))
        
        best_trial_obj = best_obj
        best_trial_a = a.copy()
        best_trial_t = 0
        best_trial_dir = ''
        
        for g_dir, dir_name in directions:
            # Try multiple blend factors
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                a_try = (1 - t) * a + t * g_dir
                a_try = project_simplex(np.maximum(a_try, 0))
                obj_try = compute_obj(a_try)
                if obj_try < best_trial_obj:
                    best_trial_obj = obj_try
                    best_trial_a = a_try.copy()
                    best_trial_t = t
                    best_trial_dir = dir_name
        
        if best_trial_obj < best_obj - 1e-12:
            best_obj = best_trial_obj
            best_a = best_trial_a.copy()
            a = best_trial_a.copy()
            stall = 0
        else:
            stall += 1
        
        it += 1
        
        if verbose and it % 5 == 0:
            print(f"    LP-div it{it}: obj={best_obj:.8f}, stall={stall}, "
                  f"t={best_trial_t:.1f}, dir={best_trial_dir}, "
                  f"elapsed={time.monotonic()-start:.1f}s")
        
        if stall >= 15:
            # Aggressive restart
            noise_scale = 0.01 + 0.01 * (stall - 15)
            noise = rng.normal(0, noise_scale, n)
            a = project_simplex(best_a + noise)
            stall = 0
    
    return best_a, best_obj


def subgradient_method(a, time_budget, verbose=True, seed=42):
    """
    True subgradient method for the max-convolution problem.
    Uses the actual subgradient (from active peaks) rather than soft-max approximation.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Step size schedule: c / sqrt(t)
    c = 0.05
    t = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        # Decreasing step size
        step = c / np.sqrt(1 + t) * (1.0 - 0.5 * progress)
        
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        
        # Find active peaks (within 5% of max)
        threshold = max_val * 0.95
        active = np.where(conv >= threshold)[0]
        
        # Compute subgradient: average over active peaks
        grad = np.zeros(n)
        for k in active:
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]
        grad /= len(active)
        grad *= 2.0  # factor of 2 for the symmetric convolution
        
        # Step
        a_new = project_simplex(a - step * grad)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        t += 1
        
        if verbose and t % 1000 == 0:
            print(f"    Subgrad it{t}: obj={best_obj:.8f}, step={step:.6f}, "
                  f"active={len(active)}, t={elapsed:.1f}s")
        
        if t - stall > 5000 and stall > 2000:
            # Reset with perturbation
            a = project_simplex(best_a + rng.normal(0, 0.002, n))
            c = max(0.01, c * 0.7)
            stall = 0
    
    return best_a, best_obj


def u_shaped_init(n, sharpness=3.0):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    return project_simplex(a)


def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        idx = np.arange(n)
        a = (np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) +
             np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01)
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / np.sqrt(0.1 + (2 * x - 1) ** 2)
    elif mode == 6:
        a = 1.0 / np.cosh(2.0 * (2 * x - 1))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

"""
Hybrid optimization for step function peak minimization.
Combines: (1) Adam soft-max gradient descent, (2) Nesterov-accelerated soft-max GD,
(3) FFT-based subgradient method on active peaks, (4) diverse LP polishing,
(5) greedy coordinate descent, (6) length exploration.
Key differences from previous: Adam optimizer, Nesterov momentum, correct subgradient
computation via FFT, diverse LP directions, and coordinate-level fine-tuning.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_soft_weights(conv, beta):
    """Compute softmax weights over convolution values."""
    c_max = float(np.max(conv))
    exp_vals = np.exp(beta * (conv - c_max))
    total = float(np.sum(exp_vals))
    if total < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_vals / total


def compute_soft_gradient(a, p):
    """Gradient of soft-max objective via cross-correlation (approximate)."""
    n = len(a)
    a_rev = a[::-1]
    size = 1
    while size < n + len(p) - 1:
        size <<= 1
    A = rfft(a_rev, size)
    P = rfft(p, size)
    full = irfft(A * P, size)
    q = full[n - 1:n - 1 + n]
    return 2.0 * q


def compute_subgradient_fft(a, active_bool):
    """
    Compute exact subgradient of max(conv(a,a)) at the active set.
    Uses FFT for O(n log n) computation.
    active_bool: boolean array of length 2n-1.
    """
    n = len(a)
    active = active_bool.astype(np.float64)
    count = float(np.sum(active))
    if count < 1:
        count = 1.0

    # q[m] = sum_i a[i] * active[m+i]  (via cross-correlation)
    a_rev = a[::-1]
    size = 1
    while size < n + len(active) - 1:
        size <<= 1
    A = rfft(a_rev, size)
    P = rfft(active, size)
    full = irfft(A * P, size)
    q = full[n - 1:n - 1 + n]

    # Correction for diagonal terms: a[m] * 1_{2m in active} / count
    correction = np.zeros(n)
    for m in range(n):
        k = 2 * m
        if k < len(active) and active_bool[k]:
            correction[m] = a[m] / count

    grad = q / count + correction
    return grad


def adam_softmax_gd(a, time_budget, beta_start, beta_end, lr=0.005,
                    beta1=0.9, beta2=0.999, eps=1e-8, verbose=True, seed=42):
    """Adam-based soft-max gradient descent on simplex."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress

        conv = compute_conv(a)
        p = compute_soft_weights(conv, beta)
        grad = compute_soft_gradient(a, p)

        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        lr_t = lr * (1.0 - 0.5 * progress)
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = t
            stall = 0
        else:
            stall += 1
            if stall > 200:
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                pert = rng.normal(0, 0.003, n)
                a = project_simplex(best_a + pert)
                stall = 0

        a = a_new

        if verbose and t % 2000 == 0:
            print(f"    Adam it{t}: obj={best_obj:.8f}, beta={beta:.1f}, t={elapsed:.1f}s")

        if t - best_it > 10000 and t > 5000:
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr *= 0.8

    return best_a, best_obj


def nesterov_softmax_gd(a, time_budget, beta_start, beta_end, lr=0.004,
                        momentum=0.93, verbose=True, seed=42):

    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress
        lr_t = lr * (1.0 - 0.4 * progress)

        # Nesterov lookahead
        y = project_simplex(a + momentum * v)
        conv = compute_conv(y)
        p = compute_soft_weights(conv, beta)
        grad = compute_soft_gradient(y, p)

        v = momentum * v - lr_t * grad
        a_new = project_simplex(a + v)

        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = t
            stall = 0
        else:
            stall += 1
            if stall > 300:
                v = np.zeros(n)
                scale = 0.002 * (1.0 + 0.1 * (stall // 100))
                pert = rng.normal(0, scale, n)
                a = project_simplex(best_a + pert)
                stall = 0

        a = a_new
        t += 1

        if verbose and t % 2000 == 0:
            print(f"    Nesterov it{t}: obj={best_obj:.8f}, beta={beta:.1f}, t={elapsed:.1f}s")

        if t - best_it > 8000 and t > 3000:
            a = best_a.copy()
            v = np.zeros(n)
            lr *= 0.85

    return best_a, best_obj


def subgradient_method(a, time_budget, tol=0.05, verbose=True, seed=42):
    """
    Exact subgradient method for max-convolution minimization.
    Uses FFT-based subgradient computation and active set selection.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    c_init = 0.08
    t = 0
    stall = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        # Adaptive step size
        step = c_init / np.sqrt(1 + t) * (1.0 - 0.3 * progress)

        conv = compute_conv(a)
        max_val = float(np.max(conv))

        # Active set: positions within (1-tol) of max
        threshold = max_val * (1.0 - tol)
        active_bool = conv >= threshold

        # Compute subgradient via FFT
        grad = compute_subgradient_fft(a, active_bool)

        # Step
        a_new = project_simplex(a - step * grad)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = t
            stall = 0
        else:
            stall += 1

        a = a_new
        t += 1

        if verbose and t % 500 == 0:
            print(f"    Subgrad it{t}: obj={best_obj:.8f}, step={step:.6f}, "
                  f"active={int(np.sum(active_bool))}, t={elapsed:.1f}s")

        # Restart logic
        if stall > 2000 and t > 1000:
            a = project_simplex(best_a + rng.normal(0, 0.002, n))
            c_init = max(0.01, c_init * 0.7)
            t = 0
            stall = 0

        if t - best_it > 5000 and t > 2000:
            a = best_a.copy()

    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, max_time=5.0):
    """Solve LP: min M s.t. conv(a_norm, g)_k <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    start = time.monotonic()
    solvers = [cp.CLARABEL, cp.SCS, cp.ECOS, cp.CVXOPT]
    for solver in solvers:
        if time.monotonic() - start > max_time:
            break
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None
