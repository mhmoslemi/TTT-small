def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_obj_unnormalized(a):
    """Compute objective without assuming normalization."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def gradient_from_p(a, p):
    """Compute gradient: 2 * (a[::-1] * p)[n-1:2n-1]."""
    n = len(a)
    b = a[::-1]
    N_fft = 1
    while N_fft < 3 * n - 1:
        N_fft <<= 1
    c = irfft(rfft(b, N_fft) * rfft(p, N_fft), N_fft)
    return 2.0 * c[n - 1:2 * n - 1]


def smoothed_gradient(a, conv, tau):
    """Smoothed gradient using softmax over convolution peaks."""
    n = len(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    shifted = shifted - shifted.max()
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(2 * n - 1)
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    return gradient_from_p(a, p)


def subgradient_at_max(a, conv):
    """Subgradient at the maximizing lag only."""
    n = len(a)
    k_max = int(np.argmax(conv))
    p = np.zeros(2 * n - 1)
    p[k_max] = 1.0
    return gradient_from_p(a, p)


def mirror_descent_step(a, grad, eta):
    """Multiplicative weights update."""
    exponent = np.clip(-eta * grad, -50, 50)
    a_new = a * np.exp(exponent)
    s = float(np.sum(a_new))
    if s < 1e-300:
        return a.copy()
    return a_new / s


def coordinate_descent_1d(a, i, conv, a_i):
    """
    Find optimal value of a[i] given all other coordinates fixed.
    a is normalized (sum = 1). Returns optimal x and new objective.
    """
    n = len(a)
    
    def obj_of_x(x):
        # Compute (a'*a')[k] for all k
        # h_k(x) as derived above
        new_conv = conv.copy()
        # Update for k = 2i
        if 2 * i < 2 * n - 1:
            new_conv[2 * i] = conv[2 * i] - a_i * a_i + x * x
        # Update for other k where 0 <= k-i < n
        k_valid = np.arange(2 * n - 1)
        diff = k_valid - i
        valid_mask = (diff >= 0) & (diff < n)
        valid_mask[2 * i] = False  # Already handled
        for k_idx in np.where(valid_mask)[0]:
            new_conv[k_idx] = conv[k_idx] + 2.0 * (x - a_i) * a[k_idx - i]
        
        s_new = 1.0 + x - a_i
        if s_new < 0.01:
            return np.inf
        return 2.0 * n * float(np.max(new_conv)) / (s_new * s_new)
    
    # Search range
    upper = max(5.0 * max(a_i, np.mean(a)), 0.1)
    x0 = a_i
    
    result = minimize_scalar(obj_of_x, bounds=[0, upper], method='bounded',
                             options={'xatol': 1e-10})
    if result.success:
        return result.x, result.fun
    return x0, obj_of_x(x0)


def coordinate_descent_pass(a, n_passes=1, verbose=False, time_budget=None):
    """Perform coordinate descent: sweep over all coordinates."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))

        return a
    a = a / s
    
    best_obj = compute_obj(a)
    start = time.monotonic() if time_budget else None
    rng = np.random.default_rng(12345)
    
    for sweep in range(n_passes):
        if time_budget and (time.monotonic() - start) > time_budget:
            break
        order = list(range(n))
        rng.shuffle(order)
        
        conv = compute_conv(a)
        
        for i in order:
            if time_budget and (time.monotonic() - start) > time_budget:
                break
            x_new, obj_new = coordinate_descent_1d(a, i, conv, a[i])
            if x_new >= 0 and obj_new < best_obj + 1e-12:
                # Update a and conv
                a[i] = x_new
                # Recompute conv (expensive but safe)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    conv = compute_conv(a)
                    obj = compute_obj(a)
                    if obj < best_obj:
                        best_obj = obj
            if verbose and i % 200 == 0:
                elapsed = time.monotonic() - start
                print(f"      CD sweep {sweep}, coord {i}/{n}: obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return a, best_obj

def nesterov_gd_optimize(a, time_budget=100.0, verbose=False):
    """Nesterov accelerated gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 3.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    tau = 100.0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = 100.0 + 2000.0 * progress ** 1.5

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        conv_y = compute_conv(y)
        g = smoothed_gradient(y, conv_y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            lr = base_lr * (0.5 + 0.5 / (g_norm * np.sqrt(n) + 1e-10))
            lr = min(max(lr, base_lr * 0.1), base_lr * 2.0)

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1

        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.3
            best_it = it

    if verbose:
        print(f"    NesterovGD: obj={best_obj:.8f}, iters={it}")
    return best_a


def lp_direction(a, use_improved=True):
    """
    Solve LP for direction finding.
    Standard: min M s.t. conv(a,g)_k <= M, sum(g)=1, g>=0
    Improved: min M s.t. 2*conv(a,g)_k - conv(a,a)_k <= M
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv_aa = compute_conv(a_norm)

    # Build matrix A: (A @ g)[k] = conv(a, g)[k]
    A = np.zeros((2 * n - 1, n))
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A[valid] = a_norm[diff[valid]]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    if use_improved:
        constraints = [2.0 * (A @ g) - M <= conv_aa, cp.sum(g) == 1]
    else:
        constraints = [A @ g <= M, cp.sum(g) == 1]
    
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def lp_driven_optimize(a, time_budget=150.0, verbose=False):
    """LP-driven optimization with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - start < time_budget:
        if (time.monotonic() - start) > time_budget - 2.0:
            break

        g = lp_direction(a, use_improved=(it % 2 == 0))
        if g is None:
            no_improve += 1
            if no_improve >= 3:
                break
            # Fallback to gradient
            conv = compute_conv(a)
            g = np.maximum(subgradient_at_max(a, conv), 0)
            s_g = float(np.sum(g))
            if s_g < 1e-10:
                break
            g = g / s_g

        # Line search
        best_t = 0
        best_t_obj = best_obj
        for t in [0.1, 0.3, 0.5, 0.7, 0.9]:
            a_try = np.maximum((1 - t) * a + t * g, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                obj_try = compute_obj(a_try / s_try)
                if obj_try < best_t_obj:
                    best_t_obj = obj_try
                    best_t = t

        if best_t > 0:
            a_new = np.maximum((1 - best_t) * a + best_t * g, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                best_obj = best_t_obj
                best_a = a_new.copy()
                a = a_new
                no_improve = 0
        else:
            no_improve += 1

        it += 1
        if verbose and it % 3 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={best:.3f}, time={time.monotonic()-start:.0f}s")

        if no_improve >= 8:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
