"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses stochastic subgradient methods, mirror descent, and sparse LP polishing.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def subgradient_at_lag(a, lag):
    """Compute gradient of (conv(a,a))[lag] with respect to a.
    (conv(a,a))[lag] = sum_i a[i] * a[lag-i]
    d/d a[j] = 2 * a[lag-j] (for valid j)
    """
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        k = lag - j
        if 0 <= k < n:
            grad[j] = 2.0 * a[k]
    return grad


def stochastic_subgrad_descent(a, time_budget, lr_init=0.01, lr_decay=0.999, 
                                top_k=5, verbose=False, seed=42):
    """
    Stochastic subgradient descent: at each step, sample from top-k max lags,
    compute subgradient, take a step.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    lr = lr_init
    patience = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        
        # Find top-k lags
        if top_k >= len(conv):
            lags = np.arange(len(conv))
        else:
            # Use partition to find top-k
            top_indices = np.argpartition(conv, -top_k)[-top_k:]
            lags = top_indices
        
        # Randomly pick one from top-k
        lag = lags[rng.integers(len(lags))]
        
        # Compute subgradient
        grad = subgradient_at_lag(a, lag)
        
        # Step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
            if patience > 100:
                lr *= 0.9
                patience = 0
        
        a = a_new
        lr *= lr_decay
        
        if verbose and (best_obj < 1.51 or patience % 500 == 0):
            elapsed = time.monotonic() - start
            print(f"    SSGD: obj={best_obj:.10f}, lr={lr:.6f}, patience={patience}, t={elapsed:.1f}s")
    
    return best_a


def mirror_descent(a, time_budget, eta_init=0.05, eta_decay=0.9995,
                   top_k=3, verbose=False, seed=42):
    """
    Mirror descent with exponential weighting (multiplicative updates).
    a_i <- a_i * exp(-eta * g_i) / Z
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    eta = eta_init
    no_improve = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        
        # Find top-k
        if top_k >= len(conv):
            lags = np.arange(len(conv))
        else:
            top_indices = np.argpartition(conv, -top_k)[-top_k:]
            lags = top_indices
        
        # Average subgradient over top-k lags
        grad = np.zeros(n)
        for lag in lags:
            grad += subgradient_at_lag(a, lag)
        grad /= len(lags)
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10 or not np.isfinite(s_new):
            eta *= 0.5
            continue
        a_new = a_new / s_new
        a_new = np.maximum(a_new, 0)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                eta *= 0.95
                no_improve = 0
        
        a = a_new
        eta *= eta_decay
        
        if verbose and (int((time.monotonic() - start) * 2) % 10 == 0):
            elapsed = time.monotonic() - start
            print(f"    Mirror: obj={best_obj:.10f}, eta={eta:.6f}, no_improve={no_improve}, t={elapsed:.1f}s")
    
    return best_a


def smoothed_grad_fft(a, tau):
    """Compute smoothed gradient using FFT-based approach."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    shifted = tau * (conv - exact_max)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    
    # Gradient: g_j = 2 * sum_k p_k * a[k-j] (for valid k-j)
    # This is 2 * (p * a) (cross-correlation)
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    # cross-correlation: sum_k p_k * a_{k-j}
    # = ifft(fft(p) * conj(fft(a)))
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max

    if tau_levels is None:

    beta = 0.9
    for tau in tau_levels:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        budget_level = min(remaining, time_budget * 0.15)
        lr = 2.0 / np.sqrt(n)

def lp_solve(a):
    """Solve min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1."""
    n = len(a)
    # Build the constraint matrix efficiently
    # (a*g)_k = sum_j a_j * g_{k-j} for valid indices
    # This is a Toeplitz-like matrix
    
    # For efficiency, build the matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget, max_iters=50, verbose=False, seed=42):
    """LP-based polishing: alternate between LP step and perturbation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)

        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, M_lp = lp_solve(a)
        if g_lp is None:
            break

        # Try multiple blend factors
        best_local = best_obj
        best_local_a = best_a.copy()
        for t in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_local - 1e-14:
                best_local = obj_try
                best_local_a = a_try.copy()

        if best_local < best_obj - 1e-14:
            best_obj = best_local
            best_a = best_local_a.copy()
            a = best_a
            if verbose:
                print(f"    LP it{it}: obj={best_obj:.10f}")
        else:

            a = a + 0.01 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            sa = float(np.sum(a))
            if sa > 1e-10:
                a = a / sa
            else:
                a = best_a.copy()

    return best_a


def coordinate_descent(a, time_budget, verbose=False, seed=42):
    """
    Cyclic coordinate descent: optimize one coordinate at a time.
    For a fixed a_{-i}, the objective is 2n * max_k (sum_j a_j a_{k-j}) / (sum a)^2.
    This is complex, but we can use the subgradient approach per coordinate.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Random order
    order = rng.permutation(n)
    sweep = 0
    
    while time.monotonic() - start < time_budget:
        for idx in order:
            i = idx
            # Compute current max convolution
            conv = compute_conv(a)
            max_val = float(np.max(conv))
            # The gradient of max_conv with respect to a[i]:
            # For the max lag k*, d/d a[i] of (a*a)[k*] = 2*a[k*-i]
            # But we need to be more careful since changing a[i] changes all lags
            
            # Simple approach: try increasing and decreasing a[i]
            best_i_val = a[i]
            best_i_obj = best_obj  # Use global best for comparison
            
            # Try a few values
            for delta in [-0.01, -0.005, -0.001, 0.001, 0.005, 0.01]:
                a_try = a.copy()
                a_try[i] = max(0, a[i] + delta)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_i_obj:
                    best_i_obj = obj_try
                    best_i_val = a_try[i]
            
            if best_i_val != a[i]:
                a[i] = best_i_val
                if best_i_obj < best_obj - 1e-14:
                    best_obj = best_i_obj
                    best_a = a.copy()
        
        sweep += 1
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, t={elapsed:.1f}s")
        
        # Shuffle order
        order = rng.permutation(n)
    
    return best_a


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a hybrid approach: fast batch subgradient descent, periodic LP steps,
mirror descent (multiplicative updates), adaptive perturbation, multi-n exploration,
and intensive LP polishing.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def subgradient_at_lag(a, lag):
    """Gradient of (conv(a,a))[lag] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        k = lag - j
        if 0 <= k < n:
            grad[j] = 2.0 * a[k]
    return grad


def batch_subgradient(a, eps=0.002):
    """Average subgradient over near-max lags. Returns (grad, max_val, active_count)."""
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    threshold = max_val * (1.0 - eps)
    mask = conv >= threshold
    lags = np.where(mask)[0]
    if len(lags) == 0:
        lags = np.array([np.argmax(conv)])
    grad = np.zeros(n)
    for lag in lags:
        grad += subgradient_at_lag(a, lag)
    grad /= len(lags)
    return grad, max_val, len(lags)


def smoothed_grad_fft(a, tau):
    """FFT-based smoothed gradient using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    shifted = tau * (conv - exact_max)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def lp_solve(a):
    """Solve min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1. Returns (g, M)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None

            y = y / sy

            g, _ = smoothed_grad_fft(y, tau)
            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new
            iters += 1
            lr *= 0.99995

        if verbose:
            elapsed = time.monotonic() - start
            print(f"    NGD tau={tau}: obj={best_obj:.10f}, iters={iters}, t={elapsed:.1f}s")

    return best_a


def lp_polish(a, time_budget, max_iters=100, verbose=False, seed=42):
    """Intensive LP-based polishing."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, M_lp = lp_solve(a)
        if g_lp is None:
            break

        best_local = best_obj
        best_local_a = best_a.copy()
        for t in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_local - 1e-14:
                best_local = obj_try
                best_local_a = a_try.copy()

        if best_local < best_obj - 1e-14:
            best_obj = best_local
            best_a = best_local_a.copy()
            a = best_a
            stall = 0
            if verbose and it % 5 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP polish it{it}: obj={best_obj:.10f}, t={elapsed:.1f}s")
        else:
            stall += 1
            if stall > 3:
                a = best_a + 0.005 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                sa = float(np.sum(a))
                if sa > 1e-10:
                    a = a / sa
                else:
                    a = best_a.copy()
                stall = 0

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation. Returns sum=1."""
    n = len(a)
    if n == new_n:
        s = float(np.sum(a))
        return a / s if s > 1e-10 else np.ones(new_n) / new_n
    if n == 1:
        return np.ones(new_n) / new_n
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_structured_init(n, rng, mode=0):
    """Create structured initializations (sum=1)."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.maximum(np.sinc(x), 0.001)
        return a / a.sum()
    elif mode == 2:
        a = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / max(1, n - 1))
        return a / a.sum()
    elif mode == 3:
        x = np.arange(n) / max(1, n - 1)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 4:
        x = np.linspace(-3, 3, n)
        a = np.exp(-x ** 2)
        return a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
        return a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * np.abs(x - 0.5), 0.01)
        return a / a.sum()
    elif mode == 7:
        taper = 0.15
        a = np.ones(n)
        for i in range(n):
            d = min(i, n - 1 - i)
            if d < taper * n:
                a[i] = np.exp(-((taper * n - d) / (taper * n)) ** 2)
        return a / a.sum()
    elif mode == 8:
        x = np.arange(n) / n
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.1 * np.cos(2 * np.pi * 4 * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        a = rng.uniform(0.1, 1.0, n)
        return a / a.sum()


def iterative_amplitude_mod(a, time_budget, verbose=False, seed=42):
    """
    Iterative Amplitude Modification (IAM) in the frequency domain.
    Transforms to frequency, shapes the spectrum, transforms back,
    and clips to non-negativity.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    N = 1
    while N < 4 * n:
        N <<= 1

    for it in range(200):
        if time.monotonic() - start >= time_budget:
            break

        # Compute current convolution (autocorrelation in freq domain)
        A = fft(a, N)
        power = np.abs(A) ** 2

        # Target: flatten the power spectrum (reduce peaks)
        # Use a soft-clipping approach
        max_power = float(np.max(power))
        threshold = max_power * 0.7
        power_clipped = np.minimum(power, threshold + (power - threshold) * 0.3)

        # Reconstruct with modified amplitude
        phase = np.angle(A)
        A_new = np.sqrt(power_clipped) * np.exp(1j * phase)
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn < 1e-10:
            continue
        a_new = a_new / sn

        # Blend
        a_try = 0.6 * a + 0.4 * a_new
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            continue
        a_try = a_try / st

        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            if verbose and it % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    IAM it{it}: obj={best_obj:.10f}, t={elapsed:.1f}s")
        else:
            # Smaller blend
            a_try2 = 0.85 * a + 0.15 * a_new
            a_try2 = np.maximum(a_try2, 0)
            a_try2 = a_try2 / a_try2.sum()
            obj_try2 = compute_obj(a_try2)
            if obj_try2 < best_obj - 1e-13:
                best_obj = obj_try2
                best_a = a_try2.copy()
                a = a_try2
            else:
                a = a + 0.005 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines hybrid optimization, Nesterov GD,
    IAM spectral shaping, LP polishing, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # --- Load parent ---
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        parent = None

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.10f}")

    n_current = len(best_seq)

    # === Phase 1: Quick screening of structured inits ===
    print(f"\n=== Phase 1: Initialization Screening (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 120:
        alloc_each = min(12.0, remaining * 0.04)
        for mode in [1, 2, 3, 4, 5, 7, 8]:
            remaining = deadline - time.monotonic()

                break
