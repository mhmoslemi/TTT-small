"""
Optimizes step function coefficients using:
1. Multiplicative weight descent with exact active-set subgradients
2. Block coordinate descent (solves small convex subproblems)
3. FFT-based fast convolution
4. Structured initializations and multi-length exploration

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def normalize(a):
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s


def gradient_of_conv(a, p):
    """
    Compute h[i] = sum_j p[i+j]*a[j] (cross-correlation of p and a).
    This is the 'gradient' of the weighted sum sum_k p[k]*(a*a)[k] w.r.t. a[i],
    divided by 2.
    """
    n = len(a)
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    Fa = rfft(a, N)
    Fp = rfft(p, N)
    corr = irfft(np.conj(Fa) * Fp, N)[:n]
    return corr


def multiplicative_descent(a, time_budget=100.0, eta_start=0.5, eta_end=0.05,
                           active_frac=0.05, verbose=False, seed=42):
    """
    Multiplicative weight descent with exact active-set subgradient.
    a should be normalized (sum=1) and non-negative.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = normalize(a)
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        eta = eta_start * (eta_end / eta_start) ** progress
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        curr_obj = 2.0 * n * max_conv
        
        # Active set
        threshold = (1.0 - active_frac) * max_conv
        active = conv >= threshold
        n_active = int(np.sum(active))
        if n_active == 0:
            active[np.argmax(conv)] = True
            n_active = 1
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / n_active
        
        # Compute gradient h[i] = sum_j p[i+j]*a[j]
        h = gradient_of_conv(a, p)
        
        # Multiplicative update
        a_new = a * np.exp(-eta * h)
        a_new = np.maximum(a_new, 1e-15)
        a_new = normalize(a_new)
        
        obj_new = compute_obj_normalized(a_new)

            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:

                a_new = normalize(a_new)
                stall = 0
        
        a = a_new
        it += 1
        
        if verbose and it % 1000 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, eta={eta:.4f}, n_active={n_active}")

def coordinate_descent(a, time_budget=100.0, verbose=False, seed=42):
    """
    Fast coordinate descent with golden section search.
    a should be normalized (sum=1) and non-negative.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = normalize(a)
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    conv = compute_conv(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    sweep = 0
    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        S = 1.0  # sum(a) should be 1
        
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            
            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue
            
            # Affected convolution indices
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            
            # For each affected k, (a*a)[k] changes linearly or quadratically
            # (a*a)[k] = old + 2*(x-v)*a[k-i] + x^2 - v^2 (if k = 2i)
            
            mask_quad = (ks == 2 * i)
            ks_lin = ks[~mask_quad]
            
            def f_1d(x):
                # Compute max of affected conv values
                if len(ks_lin) > 0:
                    partners = a[ks_lin - i]
                    max_lin = float(np.max(conv[ks_lin] + 2.0 * (x - v) * partners))
                else:
                    max_lin = 0.0
                if mask_quad.any():
                    k_q = int(ks[mask_quad][0])
                    max_quad = float(conv[k_q] + x * x - v * v)
                    max_val = max(max_lin, max_quad)
                else:
                    max_val = max_lin
                # Also consider the current global max at unaffected positions
                # (approximation: use the pre-computed max at unaffected)
                # For speed, we only consider affected + a correction
                return 2.0 * n * max_val  # sum=1 so objective = 2n*max(conv)
            
            # Golden section search
            scale = max(v, 1.0 / n, 1e-6)
            x_lo = max(0.0, v - 0.5 * scale)
            x_hi = v + 1.5 * scale
            
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)
            
            for _ in range(40):
                if x_hi - x_lo < 1e-13:
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def full_objective(a):
    """Compute the actual evaluate_sequence objective."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    return float(2 * n * np.max(conv) / (s * s))


def normalize(a):
    """Normalize to sum=1, ensuring non-negativity."""
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s


def cross_correlation(p, a):
    """
    Compute corr[i] = sum_j p[i+j]*a[j] for i=0,...,n-1.
    This equals the 'gradient' of sum_k p[k]*(a*a)[k] w.r.t. a[i], divided by 2.
    """
    n = len(a)
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    Fa = rfft(a, N)
    Fp = rfft(p, N)
    corr = irfft(np.conj(Fa) * Fp, N)[:n]
    return np.maximum(corr, 0)  # p and a are non-negative, so corr should be >= 0


def multiplicative_descent(a, time_budget=100.0, eta_start=0.3, eta_end=0.01,
                           active_frac=0.03, verbose=False, seed=42):
    """
    Multiplicative weight descent using exact subgradient of max(conv(a,a)).
    Key innovation: uses the exact active set (not smoothed approximation).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = normalize(a)

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    stall = 0
    best_stall = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        eta = eta_start * (eta_end / eta_start) ** progress

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        curr_obj = 2.0 * n * max_conv

        # Find active set: positions where conv is close to max
        threshold = (1.0 - active_frac) * max_conv
        active_mask = conv >= threshold
        n_active = int(np.sum(active_mask))
        if n_active == 0:
            active_mask[np.argmax(conv)] = True
            n_active = 1

        p = np.zeros(2 * n - 1)
        p[active_mask] = 1.0 / n_active

        # Compute gradient: h[i] = sum_j p[i+j]*a[j]
        h = cross_correlation(p, a)

        # Adaptive eta: scale by the magnitude of h
        h_max = float(np.max(h))
        if h_max > 1e-15:
            eta_eff = eta / h_max

        # Multiplicative update
        a_new = a * np.exp(-eta_eff * h)
        a_new = np.maximum(a_new, 1e-15)
        a_new = normalize(a_new)

        obj_new = compute_obj_normalized(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                # Perturb to escape local structure
                scale = 0.005 + 0.002 * (stall / 30.0)
                pert = rng.normal(0, scale, n)
                a_new = a_new * np.exp(pert)
                a_new = normalize(a_new)
                obj_new = compute_obj_normalized(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                stall = 0

        a = a_new
        it += 1

        if verbose and it % 2000 == 0:
            print(f"    MWD it{it}: obj={best_obj:.8f}, eta_eff={eta_eff:.4f}, n_active={n_active}")

    return best_a


def smoothed_multiplicative_descent(a, time_budget=100.0, tau_start=30.0, tau_end=500.0,
                                     eta_start=0.5, eta_end=0.05, verbose=False, seed=42):
    """
    Multiplicative descent using log-sum-exp smoothed gradient.
    Complements the exact MWD by providing smoother updates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = normalize(a)

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    it = 0
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_start + (tau_end - tau_start) * progress
        eta = eta_start * (eta_end / eta_start) ** progress

        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Log-sum-exp smoothing
        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -700, 0)
        exp_vals = np.exp(shifted)
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        h = cross_correlation(p, a)
        h_max = float(np.max(h))
        if h_max < 1e-15:
            break
        eta_eff = eta / h_max

        a_new = a * np.exp(-eta_eff * h)
        a_new = np.maximum(a_new, 1e-15)
        a_new = normalize(a_new)

        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 2000 == 0:
            print(f"    SMD it{it}: obj={best_obj:.8f}, tau={tau:.0f}")

    return best_a


def coordinate_descent(a, time_budget=100.0, verbose=False, seed=42, renorm_interval=20):
    """
    Coordinate descent with exact 1D subproblem using background max.
    Key: tracks the max over non-affected positions for exact 1D objective.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = normalize(a)

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    conv = compute_conv(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    sweep = 0
    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        S = float(np.sum(a))
        S_orig = S

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)

            mask_quad = (ks == 2 * i)
            ks_lin = ks[~mask_quad]

            # Background max: max of conv at positions NOT affected by coordinate i
            # Affected positions are ks (from i to i+n-1, clipped)
            if len(ks) < 2 * n - 1:
                bg_mask = np.ones(2 * n - 1, dtype=bool)
                bg_mask[ks] = False
                bg_max = float(np.max(conv[bg_mask])) if np.any(bg_mask) else 0.0
            else:
                bg_max = 0.0

            def f_1d(x, v=v, S_other=S_other, ks_lin=ks_lin, mask_quad=mask_quad,
                     ks=ks, bg_max=bg_max, a_local=a, conv_local=conv):
                if len(ks_lin) > 0:
                    partners = a_local[ks_lin - i]
                    vals_lin = conv_local[ks_lin] + 2.0 * (x - v) * partners
                    max_aff = float(np.max(vals_lin))
                else:
                    max_aff = 0.0
                if mask_quad.any():
                    k_q = int(ks[mask_quad][0])
                    max_aff = max(max_aff, float(conv_local[k_q] + x * x - v * v))
                max_val = max(bg_max, max_aff)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Golden section with adaptive bracket
            scale = max(v, 1.0 / n, 1e-6)
            x_lo = max(0.0, v - 0.3 * scale)
            x_hi = v + 0.7 * scale
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-10

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(35):
                if x_hi - x_lo < 1e-14 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = max(0.0, x_opt)
            S += delta

            # Update conv in place
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a[ks_lin - i]
            if mask_quad.any():
                k_q = int(ks[mask_quad][0])
                conv[k_q] += x_opt * x_opt - v * v

            # Check objective periodically
            if i % renorm_interval == 0:
                if S > 1e-10:
                    obj = 2.0 * n * float(np.max(conv)) / (S * S)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()

        # End of sweep: renormalize
        S = float(np.sum(a))
        if S > 1e-10 and abs(S - S_orig) > 0.005:
            a = a / S
            conv = compute_conv(a)
            obj = compute_obj_normalized(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def build_conv_matrix(b):
    """Build matrix A (2n-1 x n) such that A @ g = conv(b, g)."""
    n = len(b)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = b[:r]
    return A


def lp_direction(a, time_budget=20.0, lambda_reg=0.0, verbose=False):
    """
    Solve LP: min M + lambda*||g-a||_1 s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns g (normalized) or None.
    """
    n = len(a)
    if n > 3000:
        return None

    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

    if lambda_reg > 0:
        objective = cp.Minimize(M + lambda_reg * cp.norm1(g - a))
    else:
        objective = cp.Minimize(M)

    prob = cp.Problem(objective, constraints)

    start = time.monotonic()
    solvers = [cp.CLARABEL, cp.SCS, cp.MOSEK, cp.PDLP]
    for solver in solvers:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def lp_polish(a, time_budget=60.0, verbose=False, seed=42):
    """LP-based polishing with blending and restarts."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = normalize(a)

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        # Alternate between pure LP and proximity LP
        lam = 0.0 if stall == 0 else min(0.1 * stall, 1.0)
        g_lp = lp_direction(a, time_budget=min(remaining * 0.4, 15.0), lambda_reg=lam)

        if g_lp is None:
            break

        # Try multiple blend factors
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 2:
                pert = rng.normal(0, 0.01, n)
                a = a * np.exp(pert)
                a = normalize(a)
                stall = 0

    return best_a


def nesterov_descent(a, time_budget=80.0, tau_init=30.0, tau_end=800.0, verbose=False):
    """Nesterov-accelerated smoothed gradient descent with multiplicative updates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = normalize(a)

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    beta = 0.9
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress
        lr_eff = lr * (1.0 - 0.5 * progress)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = a.copy()
        else:
            y = y / s_y

        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -700, 0)
        exp_vals = np.exp(shifted)
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        h = cross_correlation(p, y)
        h_max = float(np.max(h))
        if h_max < 1e-15:
            break

        v = beta * v - (lr_eff / h_max) * h
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 3000 == 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.0f}")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 3:
        # Arcsine-like (heavy at edges)
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-8)
        return a / a.sum()
    elif mode == 4:
        # Cosine modulation
        a = 1.0 + 0.5 * np.cos(2 * np.pi * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 5:
        # Power law decay from center
        center = n / 2.0
        d = np.abs(np.arange(n) - center)
        a = 1.0 / (1.0 + d / (n * 0.3))
        return a / a.sum()
    elif mode == 6:
        # Periodic with random harmonics
        freqs = rng.integers(1, 8, size=3)
        phss = rng.uniform(0, 2 * np.pi, size=3)
        x = np.arange(n) / n
        a = np.ones(n)
        for f, p in zip(freqs, phss):
            a += 0.3 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 7:
        # "Anti-symmetric" around center: high on left, low on right
        a = np.ones(n)
        a[:n//2] *= 1.5
        a[n//2:] *= 0.5
        return a / a.sum()
    elif mode == 8:
        # Random smooth (via random low-freq DFT)
        size = 1
        while size < n:
            size <<= 1
        spec = rng.normal(0, 1, size) + 1j * rng.normal(0, 1, size)
        spec[:size//10] = 0  # low-pass
        spec[size - size//10:] = 0
        sig = np.abs(ifft(spec).real)[:n]
        sig = np.maximum(sig, 0.001)
        return sig / sig.sum()
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
