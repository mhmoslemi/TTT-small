Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2
using softmax gradient descent with FFT-based gradients, periodic LP refinement,
coordinate descent polishing, and multi-restart exploration.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = compute_conv(a)

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    # Algorithm from Wang & Carreira-Perpinan (2013)
    u = np.sort(v)[::-1]
    csv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (csv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (csv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """
    Compute gradient of smooth surrogate (1/alpha)*log(sum(exp(alpha*conv(a,a))))
    Returns (grad, conv, smooth_value, max_conv).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_conv = np.exp(alpha * (conv - max_conv))
    sum_exp = float(np.sum(exp_conv))
    weights = exp_conv / sum_exp
    smooth_val = max_conv + np.log(sum_exp) / alpha
    
    # grad[j] = sum_m weights[j+m] * a[m] for valid indices
    # Use FFT cross-correlation
    N = 1
    while N < 3 * n:
        N <<= 1
    W = fft(weights, N)
    A = fft(a, N)
    grad = ifft(np.conj(W) * A).real[:n]
    
    return grad, conv, smooth_val, max_conv


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_refine(a, warm_start=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value).flatten(), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def gradient_descent(a, time_budget, alpha=50.0, verbose=True, seed=42):

            best_trial_obj = best_obj
            best_trial_val = a_j
            
            for val in candidates:
                if abs(val - a_j) < 1e-15:
                    continue
                delta = val - a_j
                S_new = 1.0 + delta
                if S_new <= 1e-10:
                    continue
                
                indices = np.arange(k_start, k_end + 1)
                affected = conv[k_start:k_end + 1].copy()
                mask = (indices == 2 * j)
                if np.any(~mask):
                    affected[~mask] = conv[k_start:k_end + 1][~mask] + 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    affected[mask] = conv[2 * j] + (val * val - a_j * a_j)
                
                max_conv_unscaled = max(max_unaffected, float(np.max(affected)))
                obj_trial = 2.0 * n * max_conv_unscaled / (S_new * S_new)
                
                if obj_trial < best_trial_obj - 1e-13:
                    best_trial_obj = obj_trial
                    best_trial_val = val
            
            if best_trial_val != a_j:
                delta = best_trial_val - a_j
                a[j] = best_trial_val
                S_new = 1.0 + delta
                inv_S = 1.0 / S_new
                
                indices = np.arange(k_start, k_end + 1)
                mask = (indices == 2 * j)
                if np.any(~mask):
                    conv[k_start:k_end + 1][~mask] += 2.0 * delta * a[indices[~mask] - j] * inv_S
                if np.any(mask):
                    conv[2 * j] += (best_trial_val * best_trial_val - a_j * a_j) * inv_S * inv_S
                
                a *= inv_S
                conv *= inv_S * inv_S
                
                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True
        
        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
    
    return best_a


def lp_alternate(a, time_budget, seed=42, verbose=False):
    """LP alternating minimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    iter_count = 0
    stall = 0
    
    while time.monotonic() - start < time_budget - 2:
        g_lp, M_lp = solve_lp_refine(a)
        if g_lp is None:
            break
        iter_count += 1
        
        best_local_obj = best_obj
        best_local_a = a.copy()
        
        for t in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
            a_try = np.maximum((1.0 - t) * a + t * g_lp, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
        
        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 5:
                pert = rng.standard_normal(n) * 0.01
                a = np.maximum(best_a * (1 + pert), 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                stall = 0
    
    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 10):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / np.sqrt(k)
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def make_u_shaped(n, edge_weight=3.0):
    """Create a U-shaped initial guess."""
    x = np.linspace(0, 1, n)
    a = 1.0 + edge_weight * (2 * x - 1)**2
    a = np.maximum(a, 0.01)
    a /= a.sum()
    return a


def make_w_shaped(n, depth=2.0, num_peaks=3):
    """Create a W-shaped initial guess."""
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    for i in range(num_peaks):
        center = (i + 0.5) / num_peaks
        a += depth * np.exp(-((x - center) ** 2) / (0.02))
    a = np.maximum(a, 0.01)
    a /= a.sum()
    return a
