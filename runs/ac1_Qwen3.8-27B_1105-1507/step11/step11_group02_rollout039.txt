"""
Optimizes step function coefficients using spectral flattening combined with

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = next_fast_len(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def compute_obj_normalized(a):
    """Assume sum(a)=1, return 2*n*max(conv)."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def spectral_flatten(a, time_budget=100.0, alpha=0.5, num_iters=None, verbose=False):
    """
    Iterative spectral flattening:
    1. Compute power spectrum of zero-padded a
    2. Compress the power spectrum (reduce dynamic range)
    3. Reconstruct time domain, clip to non-negative, normalize
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    N = next_fast_len(2 * n - 1)
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(irfft(rfft(a, N) ** 2, N)[:2*n-1]))
    
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < time_budget:
        if num_iters and it >= num_iters:
            break
        
        # Compute FFT
        A = rfft(a, N)
        P = A.real ** 2 + A.imag ** 2  # Power spectrum
        
        # Find max power
        P_max = float(np.max(P))
        if P_max < 1e-30:
            break
        
        # Compress power spectrum: P_new = P_app^alpha where P_app is normalized
        # This reduces the dynamic range
        P_norm = P / P_max
        P_compressed = P_norm ** alpha  # alpha < 1 compresses
        
        # New amplitude
        new_amp = np.sqrt(P_compressed)
        
        # Keep phase
        phase = np.angle(A)
        A_new = new_amp * np.exp(1j * phase)
        
        # Inverse FFT
        a_new = irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            it += 1
            continue
        a_new = a_new / s_new
        
        # Evaluate
        conv_new = irfft(rfft(a_new, N) ** 2, N)[:2*n-1]
        obj_new = 2.0 * n * float(np.max(conv_new))
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        it += 1
        
        if verbose and it % 100 == 0:
            print(f"    Spectral it{it}: obj={best_obj:.8f}, alpha={alpha}")
    
    return best_a

def coordinate_descent_fast(a, time_budget=300.0, verbose=False, seed=42):
    """
    Fast coordinate descent. For each coordinate, the 1D objective is:
    f(x) = 2*n*max(c_2i + x^2 - v^2, max_k(c_k + 2*a_ki*(x-v)) ) / (S+v-x)^2
    where c_2i is the quadratic term, and the rest are linear in x.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0
    
    N = next_fast_len(2 * n - 1)
    conv = irfft(rfft(a, N) ** 2, N)[:2*n-1]
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0
    
    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            
            v = a[i]
            S_other = S - v
            
            # The convolution at position 2i: conv[2i] += (x-v)^2... no
            # conv[2i] = sum_j a[j]*a[2i-j]. When we change a[i] from v to x:
            # The term a[i]*a[2i-i] = a[i]*a[i] = v^2 becomes x^2
            # So conv[2i] changes by x^2 - v^2
            # For other positions k: conv[k] changes by 2*a[k-i]*(x-v) for k != 2i
            # (because a[i]*a[k-i] and a[k-i]*a[i] both change, each by (x-v)*a[k-i])
            
            k_min = 0
            k_max = 2 * n - 2
            # Indices of conv affected: k where k-i is in [0, n-1], i.e., k in [i, i+n-1]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            
            if k_lo > k_hi:
                continue
            
            affected_ks = np.arange(k_lo, k_hi + 1)
            a_indices = affected_ks - i  # These are the a[j] values that pair with a[i]
            
            # conv[k] for k in affected range
            c_vals = conv[affected_ks].copy()
            a_vals = a[a_indices].copy()
            
            # The quadratic term (at k=2i if 2i is in range)
            quad_idx = 2 * i
            if k_lo <= quad_idx <= k_hi:
                quad_pos = quad_idx - k_lo  # position in affected_ks
            else:
                quad_pos = -1
            
            def eval_1d(x):
                # Linear changes
                delta = x - v
                lin_vals = c_vals + 2.0 * a_vals * delta
                if quad_pos >= 0:
                    # Replace the quadratic change
                    lin_vals[quad_pos] = c_vals[quad_pos] + (x*x - v*v)
                max_val = float(np.max(lin_vals))
                denom = (S_other + x) ** 2
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom
            
            # Candidate points
            candidates = [0.0, v, S / (2.0 * n * max(1.0, best_obj) + S)]
            if v > 0:
                candidates.extend([v * 0.3, v * 0.5, v * 0.7, v * 1.3, v * 1.5, v * 2.0])
            candidates.append(1.0 / (2.0 * n))
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))
            
            best_x = v
            best_f = eval_1d(v)
            for x in candidates:
                fx = eval_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x
            
            # Golden section refinement
            scale = max(best_x, 1.0 / n, 1e-6)
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.0
            if x_hi - x_lo < 1e-12:
                x_hi = max(x_lo + 1e-8, 1e-10)
            
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = eval_1d(c_pt)
            fd = eval_1d(d_pt)
            
            for _ in range(50):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = eval_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = eval_1d(d_pt)
            
            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-13:
                continue
            
            delta = x_opt - v
            a[i] = x_opt
            
            # Update conv incrementally
            conv[affected_ks] += 2.0 * a_vals * delta
            if quad_pos >= 0:
                conv[quad_idx] += x_opt * x_opt - v * v - 2.0 * a_vals[quad_pos] * delta
                # Wait, I need to be more careful.
                # conv[2i] originally has v^2 as a contribution.
                # After changing to x, it has x^2.
                # The linear update adds 2*a_vals[quad_pos]*delta = 2*v*delta
                # But the actual change is x^2 - v^2 = (x-v)(x+v) = delta*(x+v)
                # So I need to correct: conv[2i] += delta*(x+v) - 2*v*delta = delta*(x+v-2v) = delta*(x-v) = delta^2
                # Hmm, this is getting confusing. Let me just recompute.
                # Original conv[2i] includes v^2 (from a[i]*a[i])
                # New conv[2i] should include x^2
                # The linear formula gives: conv[2i] + 2*a[i]*delta = conv[2i] + 2*v*delta
                # But the correct value is: conv[2i] + x^2 - v^2 = conv[2i] + (x-v)(x+v)
                # Difference: (x-v)(x+v) - 2*v*(x-v) = (x-v)(x+v-2v) = (x-v)^2 = delta^2
                # So: conv[2i] = conv[2i] + 2*v*delta + delta^2 = conv[2i] + (x^2-v^2)
                # Which is correct! Let me just use the direct formula.
                conv[quad_idx] = conv[quad_idx] - 2.0 * a_vals[quad_pos] * delta + (x_opt*x_opt - v*v)
            
            S += delta
            
            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
        
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_a


def dual_descent_idea(a, time_budget=150.0, verbose=False):
    """
    Alternative approach: minimize max_k conv(a,a)[k] subject to sum(a)=1, a>=0.
    Use a max-flow / water-filling style approach.
    
    Idea: The optimal solution satisfies KKT conditions. At the optimal point:
    - For any i where a[i] > 0: sum_k lambda[k] * d conv[k]/d a[i] = mu
      where lambda[k] >= 0 are dual variables for conv[k] <= M, and mu is the multiplier for sum(a)=1
    - At the optimum, lambda[k] > 0 only for "active" k where conv[k] = M
    
    We can iterate:
    1. Find the active set (k where conv[k] is close to max)
    2. Set a[i] proportional to 1 / (sum_{k in active} d conv[k]/d a[i]) for active coordinates
    3. Update
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    N = next_fast_len(2 * n - 1)
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(irfft(rfft(a, N) ** 2, N)[:2*n-1]))
    
    start = time.monotonic()
    it = 0
    tol = 0.05  # tolerance for "active" (fraction of max)
    
    while time.monotonic() - start < time_budget:
        conv = irfft(rfft(a, N) ** 2, N)[:2*n-1]
        max_conv = float(np.max(conv))
        
        # Active set: indices where conv[k] >= (1-tol)*max_conv
        active_mask = conv >= (1.0 - tol) * max_conv
        n_active = int(np.sum(active_mask))
        
        if n_active == 0:
            break
        
        # For each coordinate i, compute: grad_i = sum_{k in active} d conv[k]/d a[i]
        # d conv[k]/d a[i] = 2*a[k-i] (for k != 2i) or a[k-i] (for k = 2i)
        # Approximate as 2 * (cross-correlation of active_mask分摊 and a)[i]
        
        # Compute: g[i] = sum_{k in active} a[k-i] (approximately, ignoring the factor of 2 and the k=2i case)
        # This is: g[i] = sum_k active_mask[k] * a[k-i]
        # = cross-correlation of active_mask (length 2n-1) with a (length n)
        
        M = next_fast_len(3 * n)
        A_f = fft(a, M)
        P_f = fft(active_mask.astype(np.float64), M)
        g = ifft(np.conj(A_f) * P_f).real[:n]
        
        # a_new[i] proportional to 1/g[i] (water-filling / inverse gradient)
        # Clip to avoid division by zero
        g_safe = np.maximum(g, 1e-10)
        a_prop = 1.0 / g_safe
        
        # Blend with current
        blend = 0.3
        a_new = (1.0 - blend) * a + blend * a_prop
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            it += 1
            continue
        a_new = a_new / s_new
        
        conv_new = irfft(rfft(a_new, N) ** 2, N)[:2*n-1]
        obj_new = 2.0 * n * float(np.max(conv_new))
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        it += 1
        
        # Gradually tighten the active set
        tol = max(0.01, tol * 0.95)
        
        if verbose and it % 100 == 0:
            print(f"    Dual it{it}: obj={best_obj:.8f}, active={n_active}, tol={tol:.4f}")
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        centers = [0.3, 0.7]
        widths = [0.1, 0.1]
        x = np.linspace(0, 1, n)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / np.sqrt(1 - x**2)
        a = a / a.sum()
    elif mode == 4:
        # Random Fourier features
        n_freqs = 5
        freqs = rng.integers(1, 20, size=n_freqs)
        phases = rng.uniform(0, 2 * np.pi, size=n_freqs)
        x = np.linspace(0, 1, n)
        a = np.ones(n) * 0.5
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Bimodal with random parameters
        c1, c2 = sorted(rng.uniform(0.1, 0.9, 2))
        w1, w2 = rng.uniform(0.05, 0.2, 2)
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - c1) ** 2) / (2 * w1 ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w2 ** 2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

Wait, I have a bug in the coordinate_descent_fast function. The update of conv is incorrect. Let me fix it.

When we change a[i] from v to x:
- For k = 2i: conv[k] changes by x^2 - v^2 (the a[i]*a[i] term)
- For k != 2i, k in [i, i+n-1]: conv[k] changes by 2*a[k-i]*(x-v)

So the update should be:

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft, next_fast_len
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = next_fast_len(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def spectral_flatten(a, time_budget=100.0, verbose=False):
    """
    Iterative spectral flattening with adaptive alpha.
    Compresses the power spectrum to reduce autocorrelation peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    N = next_fast_len(2 * n - 1)
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(irfft(rfft(a, N) ** 2, N)[:2 * n - 1]))

    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        alpha = 0.9 - 0.7 * progress

        A = rfft(a, N)
        P = A.real ** 2 + A.imag ** 2
        P_max = float(np.max(P))
        if P_max < 1e-30:
            break

        P_norm = P / P_max
        P_compressed = P_norm ** alpha
        new_amp = np.sqrt(P_compressed)
        phase = np.angle(A)
        A_new = new_amp * np.exp(1j * phase)

        a_new = irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            it += 1
            continue
        a_new = a_new / s_new

        conv_new = irfft(rfft(a_new, N) ** 2, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(conv_new))

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 200 == 0:
            print(f"    Spectral it{it}: obj={best_obj:.8f}, alpha={alpha:.3f}")

    return best_a


def coordinate_descent(a, time_budget=300.0, verbose=False, seed=42):
    """
    Correct coordinate descent with:
    - Proper tracking of max over ALL conv positions (affected + unaffected)
    - Golden section for 1D subproblem
    - Incremental conv updates
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0

    N = next_fast_len(2 * n - 1)
    conv = irfft(rfft(a, N) ** 2, N)[:2 * n - 1]
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < -1e-10:
                continue

            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            if k_lo > k_hi:
                continue

            affected_ks = np.arange(k_lo, k_hi + 1)
            a_indices = affected_ks - i
            c_vals = conv[affected_ks].copy()
            a_vals = a[a_indices].copy()

            # Max of unaffected conv positions
            if k_lo > 0:
                max_unaff_left = float(np.max(conv[:k_lo]))
            else:
                max_unaff_left = -1e30
            if k_hi < 2 * n - 2:
                max_unaff_right = float(np.max(conv[k_hi + 1:]))
            else:
                max_unaff_right = -1e30
            max_unaffected = max(max_unaff_left, max_unaff_right)

            quad_idx = 2 * i
            quad_pos = (quad_idx - k_lo) if (k_lo <= quad_idx <= k_hi) else -1

            def eval_1d(x, c_vals=c_vals, a_vals=a_vals, quad_pos=quad_pos,
                        max_unaffected=max_unaffected, S_other=S_other, v=v, n=n):
                delta = x - v
                lin_vals = c_vals + 2.0 * a_vals * delta
                if quad_pos >= 0:
                    lin_vals[quad_pos] = c_vals[quad_pos] + (x * x - v * v)
                max_val = max(float(np.max(lin_vals)), max_unaffected)
                denom = (S_other + x) ** 2
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Collect candidates
            candidates = [0.0, v]
            if v > 0:
                candidates.extend([v * 0.3, v * 0.5, v * 0.7, v * 1.3, v * 1.7])
            candidates.append(1.0 / n)
            candidates.append(2.0 / n)
            # Quadratic critical point
            if S_other > 1e-15:
                x_quad = (float(conv[quad_idx]) - v * v) / S_other if quad_pos >= 0 else 0.0
                if x_quad > 0:
                    candidates.append(x_quad)
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = eval_1d(v)
            for x in candidates:
                fx = eval_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section refinement
            scale = max(best_x, 1.0 / n, 1e-6)
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.0
            if x_hi - x_lo < 1e-12:
                x_hi = max(x_lo + 1e-8, 1e-10)

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = eval_1d(c_pt)
            fd = eval_1d(d_pt)

            for _ in range(50):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = eval_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = eval_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-13:
                continue

            delta = x_opt - v
            a[i] = x_opt

            # Update conv incrementally
            conv[affected_ks] += 2.0 * a_vals * delta
            if quad_pos >= 0:
                conv[quad_idx] += delta * delta

            S += delta

            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 3 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def momentum_descent(a, time_budget=150.0, verbose=False):
    """Momentum-based gradient descent on smoothed max autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    N = next_fast_len(2 * n - 1)
    M = next_fast_len(3 * n)
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(irfft(rfft(a, N) ** 2, N)[:2 * n - 1]))

    start = time.monotonic()
    beta = 0.85
    v_mom = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        lr = 0.15 * (1.0 - 0.6 * progress)
        tau = 30.0 + 200.0 * progress

        # Compute smoothed gradient
        A = rfft(a, N)
        conv = irfft(A * A, N)[:2 * n - 1]
        max_conv = float(np.max(conv))

        shifted = np.clip(tau * (conv - max_conv), -50, 50)
        exp_shifted = np.exp(shifted)
        sum_exp = float(np.sum(exp_shifted))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_shifted / sum_exp

        # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] (cross-correlation)
        A_f = fft(a, M)
        P_f = fft(p, M)
        q_full = ifft(np.conj(A_f) * P_f).real[:n]
        grad = 2.0 * q_full

        v_mom = beta * v_mom - lr * grad
        a_new = a + v_mom
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
            v_mom = np.zeros(n)
        else:
            a_new = a_new / s_new

        conv_new = irfft(rfft(a_new, N) ** 2, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(conv_new))

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 500 == 0:
            print(f"    Momentum it{it}: obj={best_obj:.8f}, lr={lr:.4f}")

    return best_a


def dual_descent(a, time_budget=100.0, verbose=False):
    """
    Water-filling / dual-descent heuristic.
    At optimum, for active coordinates: sum_{k in active} a[k-i] should be constant.
    We iterate: a[i] ~ 1 / (sum_{k active} a[k-i])
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    N = next_fast_len(2 * n - 1)
    M = next_fast_len(3 * n)
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(irfft(rfft(a, N) ** 2, N)[:2 * n - 1]))

    start = time.monotonic()
    it = 0
    tol = 0.08

    while time.monotonic() - start < time_budget:
        conv = irfft(rfft(a, N) ** 2, N)[:2 * n - 1]
        max_conv = float(np.max(conv))
        if max_conv < 1e-15:
            break

        active_mask = (conv >= (1.0 - tol) * max_conv).astype(np.float64)
        n_active = int(np.sum(active_mask))
        if n_active == 0:
            break

        A_f = fft(a, M)
        P_f = fft(active_mask, M)
        g = ifft(np.conj(A_f) * P_f).real[:n]

        g_safe = np.maximum(g, 1e-10)
        a_prop = 1.0 / g_safe

        blend = 0.25
        a_new = (1.0 - blend) * a + blend * a_prop
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            it += 1
            continue
        a_new = a_new / s_new

        conv_new = irfft(rfft(a_new, N) ** 2, N)[:2 * n - 1]
        obj_new = 2.0 * n * float(np.max(conv_new))

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1
        tol = max(0.02, tol * 0.97)

        if verbose and it % 200 == 0:
            print(f"    Dual it{it}: obj={best_obj:.8f}, active={n_active}, tol={tol:.4f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        c1, c2 = sorted(rng.uniform(0.2, 0.8, 2))
        w1, w2 = rng.uniform(0.05, 0.15, 2)
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - c1) ** 2) / (2 * w1 ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w2 ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / np.sqrt(1 - x ** 2)
        a = a / a.sum()
    elif mode == 4:
        n_f = 5
        freqs = rng.integers(2, 20, size=n_f)
        phases = rng.uniform(0, 2 * np.pi, size=n_f)
        x = np.linspace(0, 1, n)
        a = np.ones(n) * 0.5
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Triangular dips
        x = np.linspace(0, 1, n)
        a = 1.0 - 0.5 * np.abs(2 * x - 1)
        a += 0.1 * rng.uniform(0, 1, n)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining spectral flattening,
    coordinate descent, momentum descent, dual descent, multi-n exploration,
    and diverse restarts to minimize 2*n*max(conv(a,a))/sum(a)^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
            else:
                raise ValueError("sum too small")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    # === Phase 1: Spectral Flattening ===
    print(f"\n=== Phase 1: Spectral Flattening ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.12, 100.0)
    if p1_budget > 15:
        a_spec = spectral_flatten(best_seq, time_budget=p1_budget, verbose=True)
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Spectral: obj={best_score:.8f} **")
        else:
            print(f"  Spectral: obj={obj_spec:.8f}")

    # === Phase 2: Coordinate Descent (main) ===
    print(f"\n=== Phase 2: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.28, 350.0)
    if p2_budget > 30:
        a_cd = coordinate_descent(best_seq, time_budget=p2_budget,
                                  verbose=True, seed=seed)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 3: Momentum Descent ===
    print(f"\n=== Phase 3: Momentum Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.12, 100.0)
        a_mom = momentum_descent(best_seq, time_budget=alloc, verbose=True)
        obj_mom = compute_obj(a_mom)
        if obj_mom < best_score:
            best_score = obj_mom
            best_seq = a_mom.copy()
            print(f"  Momentum: obj={best_score:.8f} **")
        else:
            print(f"  Momentum: obj={obj_mom:.8f}")

    # === Phase 4: Dual Descent ===
    print(f"\n=== Phase 4: Dual Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.1, 80.0)
        a_dual = dual_descent(best_seq, time_budget=alloc, verbose=True)
        obj_dual = compute_obj(a_dual)
        if obj_dual < best_score:
            best_score = obj_dual
            best_seq = a_dual.copy()
            print(f"  Dual: obj={best_score:.8f} **")
        else:
            print(f"  Dual: obj={obj_dual:.8f}")

    # === Phase 5: CD + Spectral alternation ===
    print(f"\n=== Phase 5: CD+Spectral Alternation ===")
    remaining = deadline - time.monotonic()
    for rnd in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc_s = min(remaining * 0.1, 30.0)
        alloc_c = min(remaining * 0.15, 60.0)

        if alloc_s > 10:
            a_s = spectral_flatten(best_seq, time_budget=alloc_s, verbose=False)
            obj_s = compute_obj(a_s)
            if obj_s < best_score:
                best_score = obj_s
                best_seq = a_s.copy()

        remaining = deadline - time.monotonic()
        if alloc_c > 15:
            a_c = coordinate_descent(best_seq, time_budget=alloc_c,
                                     verbose=False, seed=seed + rnd * 137)
            obj_c = compute_obj(a_c)
            if obj_c < best_score:
                best_score = obj_c
                best_seq = a_c.copy()
                print(f"  Alt {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  Alt {rnd}: obj={obj_c:.8f}")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = []
    for delta in [-100, -50, 50, 100, 150, 200, 300, -200]:
        nt = n_best + delta
        if 300 <= nt <= 2500:
            n_options.append(nt)
    for nf in [500, 600, 700, 900, 1000, 1200, 1500]:
        if abs(nf - n_best) > 80 and nf not in n_options:
            n_options.append(nf)
    n_options = sorted(set(n_options))[:10]
    print(f"  Trying n: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        if remaining > 60:
            a_try = spectral_flatten(a_try, time_budget=min(remaining * 0.1, 20.0), verbose=False)

        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_try = coordinate_descent(a_try, time_budget=min(remaining * 0.2, 40.0),
                                       verbose=False, seed=n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse Restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = 1 + r
        a_r = diverse_init(n_cur, rng, mode=mode)
        blend = 0.3 + 0.15 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / float(a_r.sum())

        alloc = min(remaining * 0.15, 60.0)
        if alloc > 15:
            a_r = coordinate_descent(a_r, time_budget=alloc, verbose=False,
                                     seed=9000 + r * 211)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 8: Final Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc_s = min(remaining * 0.15, 30.0)
        if alloc_s > 10:
            a_fs = spectral_flatten(best_seq, time_budget=alloc_s, verbose=False)
            obj_fs = compute_obj(a_fs)
            if obj_fs < best_score:
                best_score = obj_fs
                best_seq = a_fs.copy()
                print(f"  Final Spectral: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        alloc_c = min(remaining * 0.35, 80.0)
        if alloc_c > 10:
            a_fc = coordinate_descent(best_seq, time_budget=alloc_c,
                                      verbose=False, seed=seed + 31415)
            obj_fc = compute_obj(a_fc)
            if obj_fc < best_score:
                best_score = obj_fc
                best_seq = a_fc.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_fm = momentum_descent(best_seq, time_budget=min(remaining * 0.3, 30.0),
                                    verbose=False)
            obj_fm = compute_obj(a_fm)
            if obj_fm < best_score:
                best_score = obj_fm
                best_seq = a_fm.copy()
                print(f"  Final Momentum: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
