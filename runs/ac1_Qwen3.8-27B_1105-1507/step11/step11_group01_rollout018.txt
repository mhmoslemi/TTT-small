"""
Optimizes step function coefficients using DCA (LP-based block coordinate descent),
accelerated mirror descent, and active-set coordinate descent.
Key improvements: (1) LP as primary driver with adaptive step sizes,
(2) active-set CD focusing on peak-affecting coordinates,
(3) systematic multi-n exploration with warm starts,
(4) mirror descent for fast convergence.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def obj_normalized(a):
    """Objective for normalized a (sum=1): 2*n*max(conv)."""
    n = len(a)
    if n == 0:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a)))


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a = a / s  # normalize
    
    # A @ g = conv(a, g), where A is (2n-1) x n
    # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    # But we only need the matrix-vector product, so let's build it
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        lo = i
        hi = min(2 * n - 1, i + n)
        A[lo:hi, i] = a[:hi - lo]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    return g_val, None
        except Exception:
            continue
    return None, None

                    x_lo = c_pt
                    c_pt = d_pt

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = obj_normalized(a)
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 5:
            break
        
        g_lp, _ = solve_lp(a)
        if g_lp is None:
            break
        
        # Try multiple step sizes
        best_t_obj = best_obj
        best_a_try = best_a.copy()
        
        # Adaptive step sizes
        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        # Also try multiplicative blend
        for alpha in alphas:
            a_try = (1.0 - alpha) * a + alpha * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:

def mirror_descent(a, time_budget=100.0, verbose=False):
    """
    Mirror descent (exponential weights) on the simplex.
    Fast per iteration, good for initial convergence.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = obj_normalized(a)
    start = time.monotonic()
    
    eta = 1.0 / np.sqrt(n)
    it = 0
    patience = 0
    best_eta = eta
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        peak_lags = np.where(conv >= max_conv * (1.0 - 1e-10))[0]
        
        # Subgradient
        grad = np.zeros(n)
        for k in peak_lags:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += a[k - hi:k - lo + 1]
        grad /= max(len(peak_lags), 1)
        grad *= 2.0  # factor of 2 from autocorrelation
        
        # Mirror descent update: a_new = a * exp(-eta * grad) / Z
        log_a = np.log(np.maximum(a, 1e-300))
        log_new = log_a - eta * grad
        log_new -= np.max(log_new)  # numerical stability
        new_unnorm = np.exp(log_new)
        Z = float(np.sum(new_unnorm))
        if Z < 1e-300:
            break
        a_new = new_unnorm / Z
        
        obj_new = obj_normalized(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1
            if patience > 50:
                eta *= 0.9
                best_eta = eta
                patience = 0
                if eta < 1e-6:
                    break
        
        it += 1
        if verbose and it % 500 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, eta={eta:.6f}, t={time.monotonic()-start:.1f}s")
    
    return best_a


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients using DCA (LP block coordinate descent),
active-set coordinate descent, mirror descent, and systematic multi-n exploration.
Key ideas: (1) LP as primary driver with adaptive blending, (2) active-set CD
focusing on peak-affecting coordinates, (3) fast mirror descent for initial
convergence, (4) multi-length exploration with warm starts.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def obj_normalized(a):
    """Objective for normalized a (sum=1): 2*n*max(conv)."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        lo = i
        hi = min(2 * n - 1, i + n)
        A[lo:hi, i] = a[:hi - lo]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns g (normalized to sum=1) or None."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None
    a = a / s
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs
        except Exception:
            continue
    return None

            c_k_lin = c_k[mask].copy()
            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0
            scale = max(v, 1.0 / n, 1e-6)
            x_lo = 0.0
            x_hi = v + 5.0 * scale
            if x_hi < 1e-10:
                x_hi = 1e-6
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d_eval(c_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
            fd = f_1d_eval(d_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
            for _ in range(45):
                if x_hi - x_lo < 1e-14 * max(1.0, abs(x_hi)):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d_eval(c_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d_eval(d_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-15:
                continue
            delta = x_opt - v
            a[i] = x_opt
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            S += delta
        if sweep % 3 == 0:
            conv = compute_conv(a)
            S = float(np.sum(a))
            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a / S
                    a = best_a.copy()
                    conv = compute_conv(a)
        if verbose and sweep % 10 == 0:
            obj_now = 2.0 * n * float(np.max(conv)) / (float(np.sum(a)) ** 2)
            print(f"    CD sweep {sweep}: obj={obj_now:.8f}, best={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
    return best_a


def full_cd(a, time_budget=100.0, verbose=False, seed=99999):
    """Full coordinate descent (all coordinates each sweep)."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0
    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        S = float(np.sum(a))
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()
            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask].copy()
            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0
            scale = max(v, 1.0 / n, 1e-6)
            x_lo = 0.0
            x_hi = v + 5.0 * scale
            if x_hi < 1e-10:
                x_hi = 1e-6
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d_eval(c_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
            fd = f_1d_eval(d_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
            for _ in range(40):
                if x_hi - x_lo < 1e-14 * max(1.0, abs(x_hi)):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d_eval(c_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d_eval(d_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-15:
                continue
            delta = x_opt - v
            a[i] = x_opt
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            S += delta
        if sweep % 5 == 1:
            conv = compute_conv(a)
            S = float(np.sum(a))
            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a / S
                    a = best_a.copy()
                    conv = compute_conv(a)
        if verbose and sweep % 5 == 0:
            obj_now = 2.0 * n * float(np.max(conv)) / (float(np.sum(a)) ** 2)
            print(f"    FullCD sweep {sweep}: obj={obj_now:.8f}, best={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
    return best_a


def dca_lp(a, time_budget=200.0, verbose=False):
    """DCA with LP subproblem iterations."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    best_a = a.copy()
    best_obj = obj_normalized(a)
    start = time.monotonic()
    it = 0
    stall = 0
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if time_budget - elapsed < 5:
            break
        g_lp = solve_lp(a)
        if g_lp is None:
            break
        best_t_obj = best_obj
        best_try = best_a.copy()
        alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:
            a_try = (1.0 - alpha) * a + alpha * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = obj_normalized(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_try = a_try.copy()
        # Multiplicative blend
        eps = 1e-12
        for beta in [0.2, 0.3, 0.5]:
            a_mult = np.maximum(a, eps) ** (1.0 - beta) * np.maximum(g_lp, eps) ** beta
            sm = float(np.sum(a_mult))
            if sm > 1e-10:
                a_mult = a_mult / sm
                obj_mult = obj_normalized(a_mult)
                if obj_mult < best_t_obj:
                    best_t_obj = obj_mult
                    best_try = a_mult.copy()
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_try.copy()
            a = best_a.copy()
            stall = 0
            if verbose:
                print(f"    DCA it{it}: obj={best_obj:.8f}, t={elapsed:.1f}s")
        else:
            stall += 1
            if stall >= 4:
                if verbose:
                    print(f"    DCA stalled at it{it}: obj={best_obj:.8f}")
                break
        it += 1
    return best_a


def mirror_descent(a, time_budget=100.0, verbose=False):
    """Mirror descent (exponential weights) on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    best_a = a.copy()
    best_obj = obj_normalized(a)
    start = time.monotonic()
    eta = 2.0 / np.sqrt(n)
    it = 0
    patience = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        peak_lags = np.where(conv >= max_conv * (1.0 - 1e-10))[0]
        grad = np.zeros(n)
        for k in peak_lags:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += a[k - hi:k - lo + 1]
        grad /= max(len(peak_lags), 1)
        grad *= 2.0
        log_a = np.log(np.maximum(a, 1e-300))
        log_new = log_a - eta * grad
        log_new -= np.max(log_new)
        new_unnorm = np.exp(log_new)
        Z = float(np.sum(new_unnorm))
        if Z < 1e-300:
            break
        a_new = new_unnorm / Z
        obj_new = obj_normalized(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1
            if patience > 30:
                eta *= 0.85
                patience = 0
                if eta < 1e-7:
                    break
        it += 1
        if verbose and it % 500 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, eta={eta:.6f}, t={time.monotonic()-start:.1f}s")
    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, mode, rng):
    """Generate diverse initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x * x + 1e-6)
        return a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    elif mode == 4:
        c = n / 2.0
        w = n / 4.0
        a = np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        return a / a.sum()
    elif mode == 5:
        w = n // 5
        a = np.zeros(n)
        a[:w] = np.exp(-((np.arange(w) - w // 2) ** 2) / (2 * max(1, w // 4) ** 2))
        a[-w:] = np.exp(-((np.arange(w) - w // 2) ** 2) / (2 * max(1, w // 4) ** 2))
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    else:
        return np.ones(n) / n
