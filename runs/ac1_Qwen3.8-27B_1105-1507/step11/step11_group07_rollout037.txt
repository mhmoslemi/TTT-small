import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective assuming sum(a) = 1."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    rho_hat = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_hat] - 1) / (rho_hat + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_step(a, M, lr, momentum, v):
    """
    Take one subgradient step.
    M: number of top peaks to use
    lr: learning rate
    momentum: momentum coefficient
    v: velocity (for momentum)
    """
    n = len(a)
    conv = compute_conv(a)
    # Find top M peaks
    top_idx = np.argsort(conv)[::-1][:M]
    # Compute average subgradient
    grad = np.zeros(n)
    for k in top_idx:
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += 2.0 * a[idx]
    grad /= M
    # Momentum update
    v = momentum * v + grad
    a_new = project_simplex(a - lr * v)
    return a_new, v, conv

"""
Subgradient optimization with momentum for step function convolution minimization.
Key innovations over previous approach:
1. Direct subgradient method with top-M peak averaging (not soft-max approximation)
2. FFT-accelerated subgradient computation via cross-correlation
3. Adaptive learning rate with cyclical restarts
4. Symmetric optimization to reduce effective dimension
5. Multi-scale strategy: optimize at multiple sequence lengths
6. Aggressive LP polishing with blending
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Assumes sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_obj_fast(a):
    """Compute objective assuming sum(a) is already 1."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, float(v[0]))])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho_hat = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_hat] - 1) / (rho_hat + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_at_peak(a, k):
    """Compute subgradient of (a*a)_k with respect to a.
    grad[j] = 2*a[k-j] if 0<=k-j<n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    j_min = max(0, k - n + 1)
    j_max = min(n - 1, k)
    if j_min <= j_max:
        indices = np.arange(j_min, j_max + 1)
        src = k - indices
        grad[indices] = 2.0 * a[src]
    return grad


def subgradient_topM(a, M):
    """Compute average subgradient over top-M peaks of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    if M >= len(conv):
        top_idx = np.arange(len(conv))
    else:
        top_idx = np.argpartition(conv, -M)[::-1][:M]
    top_idx = top_idx[np.argsort(top_idx)]  # sort for consistency
    grad = np.zeros(n)
    for k in top_idx:
        grad += subgradient_at_peak(a, k)
    grad /= M
    return grad, conv


def beta_init(n, alpha=0.25):
    """Beta(alpha+1, alpha+1) distribution on {0,...,n-1}."""
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.power(x, -alpha) * np.power(n - x, -alpha)
    a = np.maximum(a, 1e-15)
    return project_simplex(a)


def arcsine_init(n):
    """Arcsine-like initialization (clipped to avoid singularity)."""
    x = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(np.maximum(x * (n - x), 1.0))
    return project_simplex(a)


def u_shaped_init(n, sharpness=2.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.1 + sharpness * (2 * x - 1) ** 2)
    return project_simplex(a)


def interpolate_seq(seq, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(seq)
    if n_old == n_new:
        return seq.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, seq)
    return project_simplex(np.maximum(a_new, 0))


def subgradient_optimizer(a, time_budget, lr_start=0.02, momentum=0.85,
                          M=5, cycle_len=5000, verbose=True, seed=42):
    """
    Subgradient method with momentum for minimizing max(conv(a,a)) subject to sum(a)=1, a>=0.
    Uses top-M peak averaging for the subgradient direction.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    v = np.zeros(n)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(seed)
    lr = lr_start

    while time.monotonic() - start < time_budget:
        # Adaptive learning rate: decay over time
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        lr_eff = lr_start * (1.0 + 2.0 * (1.0 - progress)) ** (-0.7)

        # Compute subgradient
        grad, conv = subgradient_topM(a, M)

        # Momentum update
        v = momentum * v + grad
        a_new = project_simplex(a - lr_eff * v)

        # Evaluate
        obj_new = compute_obj_fast(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            # If stalled, try smaller step or perturbation
            if stall % 500 == 0:
                # Perturb and try
                noise_scale = 0.002 * min(1.0, stall / 5000.0)
                a_try = project_simplex(best_a + rng.normal(0, noise_scale, n))
                obj_try = compute_obj_fast(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                v = v * 0.5

            if stall > 5000:
                # Reset and try different M
                a = best_a.copy()
                v = np.zeros(n)
                M = max(1, min(20, M + rng.integers(-2, 3)))
                stall = 0

        it += 1
        if verbose and it % 5000 == 0:
            print(f"      SubGD it{it}: obj={best_obj:.8f}, lr={lr_eff:.5f}, "
                  f"M={M}, stall={stall}, t={elapsed:.1f}s")

    return best_a, best_obj


def soft_max_gd(a, time_budget, beta_start=10.0, beta_end=60.0,
                lr_start=0.008, momentum=0.9, verbose=True, seed=42):
    """
    Soft-max gradient descent with momentum.
    Minimizes (1/beta)*log(sum exp(beta*conv_k)) as a smooth approximation of max(conv).
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    v = np.zeros(n)
    start = time.monotonic()
    it = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress ** 0.5
        lr = lr_start * (1.0 - 0.2 * progress)

        conv = compute_conv(a)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p = exp_vals / float(np.sum(exp_vals))

        # Gradient: grad_i = 2 * sum_k p_k * (a*a)_k gradient = 2 * cross_corr(a, p)
        # (a*p_conv)_i = sum_k p_k * a_{k-i} (cross-correlation)
        a_rev = a[::-1]
        size = 1
        while size < n + len(p) - 1:
            size <<= 1
        A = rfft(a_rev, size)
        P = rfft(p, size)
        full = irfft(A * P, size)
        q = full[n - 1:n - 1 + n]
        grad = 2.0 * q

        v = momentum * v + grad
        a_new = project_simplex(a - lr * v)
        obj_new = compute_obj_fast(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            best_it = it
            stall = 0
        else:
            stall += 1
            a = a_new
            if stall > 2000:
                v = np.zeros(n)
                pert = rng.normal(0, 0.003, n)
                a = project_simplex(best_a + pert)
                stall = 0

        it += 1
        if verbose and it % 5000 == 0:
            print(f"      SoftMaxGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, "
                  f"t={elapsed:.1f}s")

    return best_a, best_obj


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp_minpeak(a, time_limit=5.0):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver, time_limit=str(int(time_limit)))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g_dir, n_trials=15):
    """Find optimal blend: minimize obj((1-t)*a + t*g_dir) over t in [0,2]."""
    best_obj = np.inf
    best_t = 0.0
    for t in np.concatenate([np.linspace(0, 1, 10), np.linspace(1, 2, 5)]):
        a_try = (1 - t) * a + t * g_dir
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj = compute_obj_fast(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    return best_t, best_obj


def lp_polish(a, time_budget, max_iters=30, verbose=True, seed=42):
    """LP-based polishing: iteratively find best direction and blend."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        lp_time = min(4.0, remaining * 0.3)
        g_lp, M_lp = solve_lp_minpeak(a, time_limit=lp_time)
        if g_lp is None:
            break

        t_opt, obj_opt = line_search_blend(a, g_lp)

        if t_opt > 0.01 and obj_opt < best_obj - 1e-12:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = project_simplex(np.maximum(a_new, 0))
            obj_new = compute_obj_fast(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

            else:

        else:
            stall += 1

        it += 1

            print(f"      LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                  f"elapsed={time.monotonic()-start:.1f}s")

            # Restart with perturbation
            noise = rng.normal(0, 0.005, n)

            stall = 0

    return best_a, best_obj


def paired_swap_optimizer(a, time_budget, verbose=True, seed=42):
    """

    Try to move mass between pairs of coordinates to reduce max-convolution.
    """

    stall = 0

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # Try moving mass from i to j
        for delta in [0.01, 0.005, 0.002, 0.001]:
            if a[i] < delta:
                continue
            a_try = a.copy()
            a_try[i] -= delta
            a_try[j] += delta
            obj_try = compute_obj_fast(a_try)
            if obj_try < best_obj - 1e-14:

                best_a = a_try.copy()
                a = a_try

                it += 1
                break
            # Also try moving from j to i
            if a[j] >= delta:
                a_try2 = a.copy()
                a_try2[j] -= delta
                a_try2[i] += delta
                obj_try2 = compute_obj_fast(a_try2)
                if obj_try2 < best_obj - 1e-14:
                    best_obj = obj_try2
                    best_a = a_try2.copy()
                    a = a_try2
                    stall = 0

        if stall > 5000:

            stall = 0

        if verbose and it % 10000 == 0:

                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Intensive final polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 6: Final Polish, budget={remaining:.0f}s ===")
        # Subgradient with fine-tuning
        alloc1 = min(remaining * 0.3, 100.0)
        if alloc1 > 15:
            a_f, obj_f = subgradient_optimizer(
                best_seq, time_budget=alloc1,
                lr_start=0.005, momentum=0.92, M=3,
                verbose=False, seed=seed + 400
            )
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final SubGD: obj={best_score:.8f} **")

        # LP polish
        remaining = deadline - time.monotonic()
        if remaining > 25:
            alloc2 = min(remaining * 0.4, 80.0)
            if alloc2 > 10:
                a_f2, obj_f2 = lp_polish(best_seq, time_budget=alloc2,
                                         max_iters=20, verbose=True, seed=seed + 500)
                if obj_f2 < best_score:
                    best_score = obj_f2
                    best_seq = a_f2.copy()
                    print(f"  Final LP: obj={best_score:.8f} **")

        # One more round of paired swaps
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc3 = min(remaining * 0.4, 50.0)
            if alloc3 > 10:
                a_f3, obj_f3 = paired_swap_optimizer(
                    best_seq, time_budget=alloc3, verbose=False, seed=seed + 600)
                if obj_f3 < best_score:
                    best_score = obj_f3
                    best_seq = a_f3.copy()
                    print(f"  Final PairSwap: obj={best_score:.8f} **")

    # === Phase 7: Last-ditch ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        # Try a few random perturbations + quick LP
        for attempt in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            noise = rng.normal(0, 0.005, len(best_seq))
            a_p = project_simplex(best_seq + noise)
            alloc = min(remaining * 0.4, 15.0)
            if alloc > 5:
                a_p, obj_p = lp_polish(a_p, time_budget=alloc, max_iters=8,
                                       verbose=False, seed=seed + 700 + attempt)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Last-ditch {attempt}: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj_fast(best_seq)

    # Ensure output is valid
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    # Make sure sum is at least 0.01
    s = sum(result)
    if s < 0.01:
        result = [0.01 / len(result)] * len(result)

    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
