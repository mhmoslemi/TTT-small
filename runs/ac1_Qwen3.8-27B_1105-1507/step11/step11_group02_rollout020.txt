import time

def compute_conv(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    """Compute 2n*max(conv)/sum^2."""

    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * np.max(conv)

def mirror_pair_update(a, conv, n, num_pairs=20, step_scale=0.05):
    """
    Mirror-pair update: for each peak position k*, find the pair (j, k*-j)
    with the largest product a[j]*a[k*-j] and reduce both slightly.
    """
    a = a.copy()
    # Find top positions in convolution
    top_ks = np.argsort(conv)[::-1][:num_pairs]

    for k_star in top_ks:
        k_star = int(k_star)
        # Find j that maximizes a[j] * a[k_star - j] for valid j
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        if j_lo >= j_hi:
            continue
        js = np.arange(j_lo, j_hi + 1)
        products = a[js] * a[k_star - js]
        j_max = js[np.argmax(products)]
        j_mirror = k_star - j_max

        # Reduce both (the larger one more)
        val1 = a[j_max]
        val2 = a[j_mirror]
        reduction = step_scale * min(val1, val2)
        a[j_max] -= reduction * 0.5
        a[j_mirror] -= reduction * 0.5

    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    return a

    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    momentum = 0.7
    v_mom = np.zeros(n)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress
        lr_eff = lr * (1.0 - 0.6 * progress)

        # Compute smoothed gradient
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = tau * (conv - max_conv)
        exp_vals = np.exp(shifted)
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            weights = np.zeros_like(conv)
            weights[np.argmax(conv)] = 1.0
        else:
            weights = exp_vals / sum_exp

        # Gradient: grad[j] = 2*t * sum_k weights[k] * a[k-j] - 2*t*L/S
        # where L = logsumexp(t*conv) = log(sum_exp) + t*max_conv
        L = np.log(sum_exp) + tau * max_conv

        # Compute c[j] = sum_k weights[k] * a[k-j] via FFT
        N = 1
        while N < len(weights) + n:
            N <<= 1
        W = rfft(weights, N)
        A_flip = rfft(a[::-1], N)
        conv_wb = irfft(W * A_flip, N)
        c = conv_wb[n - 1:n - 1 + n]

        grad = 2.0 * tau * c - 2.0 * tau * L

        # Momentum update
        v_mom = momentum * v_mom - lr_eff * grad
        a_new = a + v_mom
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

    return best_a


def peak_shaving_iteration(a, n_shave=10, strength=0.02):
    """
    Direct peak shaving: identify the top positions in the convolution and
    reduce the coordinates that contribute most to them.
    """
    n = len(a)
    conv = compute_conv(a)
    a_new = a.copy()

    # Get top positions
    threshold = np.percentile(conv, 100.0 - 100.0 * n_shave / (2 * n - 1))
    hot_positions = np.where(conv > threshold)[0]

    for k in hot_positions:
        k = int(k)
        # Contributions to conv[k]: a[j] * a[k-j] for valid j
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo > j_hi:
            continue
        js = np.arange(j_lo, j_hi + 1)
        contribs = a[js] * a[k - js]
        # Reduce the top contributing pairs
        top_j = js[np.argsort(contribs)[::-1][:3]]
        for j in top_j:
            jm = k - j
            if 0 <= jm < n:
                a_new[j] *= (1.0 - strength)
                if jm != j:
                    a_new[jm] *= (1.0 - strength * 0.5)

    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def diverse_init(n, rng, mode=0):
    """Generate diverse initial patterns."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        # U-shaped (edge-concentrated)
        x = np.linspace(-1, 1, n)
        a = 1.0 + 0.5 * x * x
        return a / a.sum()
    elif mode == 3:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 4:
        # Sinusoidal modulation
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(4 * np.pi * x) + 0.1 * np.sin(8 * np.pi * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 5:
        # Gaussian mixture
        centers = rng.uniform(0.1, 0.9, size=3)
        widths = rng.uniform(0.05, 0.2, size=3)
        x = np.linspace(0, 1, n)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-0.5 * ((x - c) / w) ** 2)
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif mode == 6:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        return a / a.sum()
    elif mode == 7:
        # Alternating high-low
        a = np.ones(n)
        a[::2] *= rng.uniform(1.5, 3.0)
        return a / a.sum()
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def propose_candidate():
    """
    Main search: combines fast CD, peak shaving, LSE gradient, and multi-start.
    Strategy: aggressive CD with peak-aware ordering, interspersed with
    structured perturbations (mirror pairs, peak shaving) to escape local minima.
    """
    start_time = time.monotonic()
    deadline = start_time + 1500.0 - 15.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = obj_norm(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = obj_norm(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    rng = np.random.default_rng(42)
    n_current = len(best_seq)

    # === Phase 1: Intensive Coordinate Descent ===
    print(f"\n=== Phase 1: CD (main) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 450.0)
    if p1_budget > 30:
        a_cd = coord_descent(best_seq, time_budget=p1_budget, seed=42, verbose=True)
        obj_cd = obj_norm(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD1: obj={best_score:.8f} **")
        else:
            print(f"  CD1: obj={obj_cd:.8f} (no improvement)")

    # === Phase 2: Peak Shaving + CD cycles ===
    print(f"\n=== Phase 2: Peak Shaving Cycles ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 200.0)
    if p2_budget > 20:
        cycle_budget = p2_budget / 4.0
        for cycle in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc_shave = min(cycle_budget * 0.3, remaining * 0.1)
            alloc_cd = min(cycle_budget * 0.7, remaining * 0.4)
            if alloc_cd < 10:
                break

            # Peak shaving
            a_shaved = peak_shaving_iteration(best_seq, n_shave=15, strength=0.015)
            obj_shaved = obj_norm(a_shaved)
            if obj_shaved > best_score * 1.05:
                # Shaving made it too bad, try mirror pair instead
                a_shaved = mirror_pair_update(best_seq, compute_conv(best_seq),
                                              n_current, num_pairs=30, step_scale=0.03)
                obj_shaved = obj_norm(a_shaved)

            # CD from shaved state
            a_out = coord_descent(a_shaved, time_budget=alloc_cd,
                                  seed=100 + cycle * 7, verbose=False)
            obj_out = obj_norm(a_out)
            if obj_out < best_score:
                best_score = obj_out
                best_seq = a_out.copy()
                print(f"  Cycle {cycle}: obj={best_score:.8f} **")
            else:
                print(f"  Cycle {cycle}: obj={obj_out:.8f}")

    # === Phase 3: LSE Gradient Descent ===
    print(f"\n=== Phase 3: LSE Gradient Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 80.0)
    if p3_budget > 20:
        a_lse = logsumexp_gradient_descent(best_seq, time_budget=p3_budget,
                                           tau_init=20.0, tau_end=150.0, seed=777)
        obj_lse = obj_norm(a_lse)
        if obj_lse < best_score:
            best_score = obj_lse
            best_seq = a_lse.copy()
            print(f"  LSE: obj={best_score:.8f} **")
        else:
            print(f"  LSE: obj={obj_lse:.8f}")
            # Try from perturbed start
            pert = rng.normal(0, 0.02, n_current)
            a_pert = best_seq * np.exp(pert)
            a_pert = np.maximum(a_pert, 0) / float(np.sum(a_pert))
            a_lse2 = logsumexp_gradient_descent(a_pert, time_budget=min(p3_budget, remaining * 0.3),
                                                tau_init=30.0, tau_end=200.0, seed=888)
            obj_lse2 = obj_norm(a_lse2)
            if obj_lse2 < best_score:
                best_score = obj_lse2
                best_seq = a_lse2.copy()
                print(f"  LSE (perturbed): obj={best_score:.8f} **")
            else:
                print(f"  LSE (perturbed): obj={obj_lse2:.8f}")

    # === Phase 4: CD after LSE ===
    print(f"\n=== Phase 4: Post-LSE CD ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.12, 100.0)
    if p4_budget > 15:
        a_cd4 = coord_descent(best_seq, time_budget=p4_budget, seed=2000, verbose=False)
        obj_cd4 = obj_norm(a_cd4)
        if obj_cd4 < best_score:
            best_score = obj_cd4
            best_seq = a_cd4.copy()
            print(f"  CD4: obj={best_score:.8f} **")
        else:
            print(f"  CD4: obj={obj_cd4:.8f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(200, n_best - 100), max(200, n_best - 50),
        n_best + 50, n_best + 100, n_best + 150, n_best + 200,
        600, 700, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000 and abs(n - n_best) > 30]
    n_options = n_options[:5]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        alloc_cd = min(remaining * 0.15, 40.0)
        if alloc_cd > 10:
            a_try = coord_descent(a_try, time_budget=alloc_cd, seed=n_try, verbose=False)

        obj_try = obj_norm(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = (r * 2 + 1) % 8
        a_r = diverse_init(n_cur, rng, mode=mode)
        blend = 0.5 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0.0)
        a_r = a_r / float(a_r.sum())

        alloc = min(remaining * 0.15, 50.0)
        if alloc > 15:
            a_r = coord_descent(a_r, time_budget=alloc, seed=5000 + r * 100, verbose=False)
            obj_r = obj_norm(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode {mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive CD ===
    print(f"\n=== Phase 7: Final CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 150.0)
        a_final = coord_descent(best_seq, time_budget=alloc, seed=9999, verbose=True)
        obj_final = obj_norm(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    # === Phase 8: Last peak shaving + CD ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_ps = peak_shaving_iteration(best_seq, n_shave=8, strength=0.01)
        obj_ps = obj_norm(a_ps)
        if obj_ps < best_score * 1.01:  # Not too bad
            alloc = min(remaining * 0.4, 60.0)
            a_ps_cd = coord_descent(a_ps, time_budget=alloc, seed=12345, verbose=False)
            obj_ps_cd = obj_norm(a_ps_cd)
            if obj_ps_cd < best_score:
                best_score = obj_ps_cd
                best_seq = a_ps_cd.copy()
                print(f"  Final PS+CD: obj={best_score:.8f} **")
            else:
                print(f"  Final PS+CD: obj={obj_ps_cd:.8f}")
        else:
            print(f"  Final PS: obj={obj_ps:.8f} (skipped, too degraded)")

    # Safety: ensure valid output
    if best_seq is None:
        best_seq = np.ones(200)
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
