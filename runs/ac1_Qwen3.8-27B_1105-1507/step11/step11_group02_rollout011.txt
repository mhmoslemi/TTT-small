def compute_conv(a):

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute normalized objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)

    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))

def momentum_smoothed_descent(a, time_budget=200.0, seed=42, verbose=True):
    """
    Nesterov momentum smoothed gradient descent on the log-sum-exp approximation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    beta = 0.85
    lr = 5.0 / np.sqrt(n)
    v_mom = np.zeros(n)
    tau = 30.0
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau_curr = 20.0 + 80.0 * progress
        lr_curr = lr * (1.0 - 0.6 * progress)

        # Nesterov lookahead
        y = a + beta * v_mom
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        try:
            grad, conv, p = smoothed_gradient(y, tau_curr)
        except Exception:
            break

        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-15:
            grad = grad / g_norm * min(g_norm, 1.0)

        v_mom = beta * v_mom - lr_curr * grad
        a_new = y + v_mom
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v_mom = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 500 == 0:
            print(f"    MGD it{it}: obj={best_obj:.8f}, tau={tau_curr:.1f}")

    return best_a


def spectral_perturb(a, rng, strength=0.02):
    """
    Apply a small random perturbation in the Fourier domain.
    This can help escape local minima by modifying global structure.
    """
    n = len(a)
    N = 1
    while N < 2 * n:
        N <<= 1
    F = fft(a, N)

    # Random phase perturbation on a few frequency bins
    n_bands = rng.integers(2, max(3, n // 20))
    bins = rng.integers(1, N // 2, size=n_bands)
    for b in bins:
        phase_shift = rng.normal(0, strength)
        amp_change = 1.0 + rng.normal(0, strength * 0.5)
        F[b] *= amp_change * np.exp(1j * phase_shift)
        F[N - b] = np.conj(F[b])

    a_new = ifft(F).real[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 3:
        # Raised cosine with offset
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.3)))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Multi-peak
        n_peaks = rng.integers(3, 7)
        centers = rng.uniform(0, 1, n_peaks)
        widths = rng.uniform(0.02, 0.1, n_peaks)
        x = np.linspace(0, 1, n)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining fast CD, momentum smoothed descent, spectral perturbations,
    and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    n_base = len(best_seq)

    # === Phase 1: Initial Fast CD ===
    print(f"\n=== Phase 1: Initial Fast CD ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 450.0)
    if p1_budget > 30:
        a_cd = fast_cd(best_seq, time_budget=p1_budget, seed=seed, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD1: obj={best_score:.8f} **")
        else:
            print(f"  CD1: obj={obj_cd:.8f} (no improvement)")

    # === Phase 2: Momentum Smoothed Descent ===
    print(f"\n=== Phase 2: Momentum Smoothed Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.15, 150.0)
        a_mg = momentum_smoothed_descent(best_seq, time_budget=alloc, seed=seed + 100)
        obj_mg = compute_obj(a_mg)
        if obj_mg < best_score:
            best_score = obj_mg
            best_seq = a_mg.copy()
            print(f"  MGD: obj={best_score:.8f} **")
            # Follow up with CD
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_cd2 = fast_cd(best_seq, time_budget=min(remaining * 0.25, 120.0),
                                 seed=seed + 200, verbose=True)
                obj_cd2 = compute_obj(a_cd2)
                if obj_cd2 < best_score:
                    best_score = obj_cd2
                    best_seq = a_cd2.copy()
                    print(f"  CD after MGD: obj={best_score:.8f} **")
        else:
            print(f"  MGD: obj={obj_mg:.8f}")

    # === Phase 3: Spectral Perturbation + CD ===
    print(f"\n=== Phase 3: Spectral Perturbation + CD ===")
    remaining = deadline - time.monotonic()
    n_pert = 4
    if remaining > 60:
        per_pert_budget = min(remaining / (n_pert + 1), 100.0)
        for p in range(n_pert):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            strength = 0.01 + 0.005 * p
            a_pert = spectral_perturb(best_seq, rng, strength=strength)
            # Also try a small blend with a diverse init
            if p > 0:
                a_div = diverse_init(n_base, rng, mode=p)
                blend = 0.1 * p
                a_pert = (1 - blend) * a_pert + blend * a_div
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / float(a_pert.sum())

            alloc_cd = min(per_pert_budget * 0.7, remaining * 0.4)
            if alloc_cd > 15:
                a_opt = fast_cd(a_pert, time_budget=alloc_cd, seed=seed + 500 + p * 100)
                obj_opt = compute_obj(a_opt)
                if obj_opt < best_score:
                    best_score = obj_opt
                    best_seq = a_opt.copy()
                    print(f"  Perturb {p}: obj={best_score:.8f} **")
                else:
                    print(f"  Perturb {p}: obj={obj_opt:.8f}")

    # === Phase 4: Extended CD with more sweeps ===
    print(f"\n=== Phase 4: Extended CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 200.0)
        a_cd3 = fast_cd(best_seq, time_budget=alloc, seed=seed + 900, verbose=True)
        obj_cd3 = compute_obj(a_cd3)
        if obj_cd3 < best_score:
            best_score = obj_cd3
            best_seq = a_cd3.copy()
            print(f"  CD3: obj={best_score:.8f} **")
        else:
            print(f"  CD3: obj={obj_cd3:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        n_best - 50, n_best - 100,
        n_best + 50, n_best + 100, n_best + 150, n_best + 200,
        n_best + 300,
        600, 700, 900, 1000, 1100, 1200, 1400
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_best) > 20]
    n_options = n_options[:5]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        alloc = min(remaining * 0.2, 60.0)
        if alloc > 15:
            a_try = fast_cd(a_try, time_budget=alloc, seed=n_try, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 150.0)
        a_final = fast_cd(best_seq, time_budget=alloc, seed=seed + 19999, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    # One more momentum pass if time
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.4, 80.0)
        a_mg2 = momentum_smoothed_descent(best_seq, time_budget=alloc, seed=seed + 20000)
        obj_mg2 = compute_obj(a_mg2)
        if obj_mg2 < best_score:
            best_score = obj_mg2
            best_seq = a_mg2.copy()
            print(f"  Final MGD: obj={best_score:.8f} **")
            # Quick CD after
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_f = fast_cd(best_seq, time_budget=min(remaining * 0.6, 40.0), seed=seed + 21000)
                obj_f = compute_obj(a_f)
                if obj_f < best_score:
                    best_score = obj_f
                    best_seq = a_f.copy()
                    print(f"  Final CD2: obj={best_score:.8f} **")
        else:
            print(f"  Final MGD: obj={obj_mg2:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with the actual evaluation function
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
