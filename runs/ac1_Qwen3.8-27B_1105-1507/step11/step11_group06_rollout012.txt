import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

# --- Core computation functions ---
def _fft_size(m):
    size = 1
    while size < m:
        size <<= 1
    return size

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = _fft_size(2*n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    an = a / s
    c = compute_conv(an)
    return 2.0 * n * float(np.max(c))

def smooth_gradient(a, eps):
    """Gradient of smoothed objective on the simplex (assuming sum(a)=1)."""
    n = len(a)
    c = compute_conv(a)
    max_c = np.max(c)
    exp_c = np.exp((c - max_c) / eps)
    p = exp_c / np.sum(exp_c)
    # corr_j = sum_k p_k * a_{k-j}
    size = _fft_size(len(p) + n)
    fp = rfft(p, size)
    fa = rfft(a, size)
    corr = irfft(np.conj(fp) * fa, size)
    grad = 2.0 * corr[:n]  # factor of 2 from derivative of c_k
    # Note: the full gradient of 2n * smoothed_max is 2n * 2 * corr = 4n * corr
    # But we can absorb the constant into the learning rate
    return grad, p

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = (cssv[cond][-1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)

def simplex_descent(a, time_budget, verbose=True):
    """Adam-based smoothed gradient descent on the probability simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start_t = time.monotonic()
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1, beta2 = 0.9, 0.999
    t = 0
    base_lr = 0.5
    total_steps = 30000  # for cosine schedule
    no_improve = 0
    max_no_improve = 2000
    
    while time.monotonic() - start_t < time_budget:
        # Compute smoothed gradient
        conv = compute_conv(a)
        range_c = conv.max() - conv.min()
        eps = max(0.0005, 0.005 * range_c)
        
        grad, p = smooth_gradient(a, eps)
        
        # Cosine learning rate schedule
        t += 1
        progress = min(1.0, t / total_steps)
        lr = base_lr * 0.5 * (1 + np.cos(np.pi * progress))
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Project onto simplex
        a_new = project_simplex(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if no_improve > max_no_improve:
            # Reset and reduce learning rate
            m = np.zeros(n)
            v = np.zeros(n)
            base_lr *= 0.5
            no_improve = 0
            a = best_a.copy()
        
        if verbose and t % 5000 == 0:
            print(f"      SD t={t}: obj={best_obj:.8f}, lr={lr:.6f}, eps={eps:.6f}")
    
    return best_a, best_obj

def build_conv_matrix(a):
    n = len(a)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # (A @ g)[k] = (a * g)[k]
    A = np.zeros((2*n - 1, n))
    for j in range(n):
        lo = max(0, j)
        hi = min(2*n - 1, j + n)
        idx = hi - lo
        A[lo:hi, j] = a[:idx]
    return A

def solve_lp(a):
    """Solve the LP: min M s.t. (a*g)_k <= M for all k, g >= 0, sum(g) = 1."""
    n = len(a)
    an = a / a.sum()
    A = build_conv_matrix(an)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = g_val.sum()
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def lp_line_search(a, g_dir, n_coarse=50, n_fine=30):
    """Line search along (1-t)*a + t*g_dir."""
    best_t = 0.0
    best_obj = compute_obj(a)
    ts = np.linspace(0, 1.2, n_coarse)
    for t in ts:
        a_try = (1 - t) * a + t * g_dir
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 0.01:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    if 0 < best_t < 1.2:
        lo = max(0, best_t - 0.03)
        hi = min(1.2, best_t + 0.03)
        for t in np.linspace(lo, hi, n_fine):
            a_try = (1 - t) * a + t * g_dir
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 0.01:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t
    return best_t, best_obj

def lp_polish(a, time_budget, max_iters=50, verbose=False):
    """LP-based polishing."""
    n = len(a)
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start_t = time.monotonic()
    it = 0
    stall = 0
    
    while time.monotonic() - start_t < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start_t)
        if remaining < 5:
            break
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break
        t_opt, obj_opt = lp_line_search(a, g_lp)
        if t_opt > 1e-6 and obj_opt < best_obj - 1e-15:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            best_obj = compute_obj(a_new)
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            obj_dir = compute_obj(g_lp)
            if obj_dir < best_obj:
                best_obj = obj_dir
                best_a = g_lp.copy()
                a = g_lp.copy()
                stall = 0
        it += 1
        if verbose and it % 5 == 0:
            print(f"      LP it{it}: obj={best_obj:.8f}")
        if stall > 5:
            if stall == 6:
                # Perturb
                rng = np.random.default_rng(it)
                freq = rng.integers(1, max(2, n // 100))
                phase = rng.uniform(0, 2 * np.pi)
                idx = np.arange(n, dtype=np.float64)
                pert = 0.02 * np.cos(2 * np.pi * freq * idx / n + phase)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
            else:
                break
            stall = 0
    
    return best_a, best_obj

def resample(a, n_new):
    """Resample sequence to new length via interpolation."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 0.01:
        return None
    return a_new / s

def u_shaped(n, sharp=3.0):

    a = np.maximum(a, 0)

    else:

    a = np.maximum(a, 0)
    return a / a.sum()
