"""
Optimizes step function coefficients using peak-aware coordinate descent,
exponentiated gradient (mirror descent), LP polishing, and adaptive n search.
Key new elements: (1) peak-aware CD ordering for faster convergence,
(2) exponentiated gradient method for escaping local minima,
(3) spectral-based direction selection, (4) more aggressive n exploration
with better initialization via resampling + deconvolution.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute the objective 2*n*max(conv)/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return float(2.0 * n * np.max(conv) / (s * s))


def compute_obj_normalized(a):
    """Assume a is normalized (sum=1), return 2*n*max(conv)."""
    n = len(a)
    conv = compute_conv(a)
    return float(2.0 * n * np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:
                    g_val = g_val / g_sum
                    conv_check = A @ g_val
                    M_actual = float(np.max(conv_check))
                    return g_val, M_actual
        except Exception:
            continue
    return None, None


def peak_aware_order(a, conv, n):
    """Order coordinates by their contribution to the current peak."""
    peak_idx = int(np.argmax(conv))
    # Contribution of coordinate i to peak: a[i] * a[peak_idx - i] if valid
    contrib = np.zeros(n)
    for i in range(n):
        j = peak_idx - i
        if 0 <= j < n:
            contrib[i] = a[i] * a[j]
    # Sort by contribution descending (process most important first)
    order = np.argsort(-contrib)
    return order


def coordinate_descent(a, time_budget=300.0, verbose=False, seed=42):
    """
    Peak-aware exact coordinate descent with adaptive ordering.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = compute_obj_normalized(a)
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    prev_best = best_obj
    stall = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        # Periodic recomputation to avoid drift
        if sweep % 3 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    a = a / S_check
                    conv = compute_conv(a)

        # Peak-aware ordering with some randomization
        peak_idx = int(np.argmax(conv))
        contrib = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        # Mix: 70% peak-aware, 30% random
        for i in range(n):
            if rng.random() < 0.3:
                contrib[i] *= rng.uniform(0.5, 2.0)
        order = np.argsort(-contrib)

        S = float(np.sum(a))
        sweep_improved = False

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k_lin.copy()

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, v * 0.25, v * 0.5, v * 0.75, v,
                          v * 1.25, v * 1.5, v * 2.0, v * 3.0,
                          1.0 / n, 2.0 / n, 3.0 / n]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            # Define 1D objective
            def f_1d(x):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section refinement
            x_lo = max(0.0, best_x - scale * 0.5)
            x_hi = best_x + scale
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(35):
                if x_hi - x_lo < 1e-14 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-15:
                continue

            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta
            sweep_improved = True

        # Track convergence
        if sweep % 5 == 0 and sweep > 5:
            obj_now = 2.0 * n * float(np.max(conv)) / (S * S) if S > 1e-10 else 1e10
            if best_obj < prev_best - 1e-12:
                prev_best = best_obj
                stall = 0
            else:
                stall += 1
                if stall > 6:
                    # Restart ordering or break
                    break

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s, stall={stall}")

    return best_a


def exponentiated_gradient(a, time_budget=120.0, verbose=False, seed=42):
    """
    Exponentiated gradient (mirror descent) method for min-max problem.
    Uses multiplicative updates guided by the subgradient of max autocorrelation.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64).copy()
    a = a / float(np.sum(a))

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    eta = 2.0 / np.sqrt(n)
    best_eta = eta
    patience = 0
    it = 0
    momentum = 0.3
    v_mom = np.zeros(n)

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start < 2.0:
            pass
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        tol = max_conv * 1e-8
        argmax_set = np.where(conv >= max_conv - tol)[0]

        # Compute subgradient
        g_sub = np.zeros(n)
        for k in argmax_set:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                g_sub[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1]
        g_sub /= max(len(argmax_set), 1)

        # Center the gradient (for invariance to constant shifts in gradient)
        g_centered = g_sub - np.mean(g_sub)

        # Momentum
        v_mom = momentum * v_mom - eta * g_centered

        # Exponentiated update
        log_a = np.log(np.maximum(a, 1e-30)) + v_mom
        # Clip to avoid overflow
        log_a = np.clip(log_a, -50, 50)
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v_mom = np.zeros(n)
            eta *= 0.5
        else:
            a_new = a_new / s_new

        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            best_eta = eta
        else:
            patience += 1
            if patience > 30:
                eta *= 0.7
                v_mom = np.zeros(n)
                if eta < 0.01:
                    # Perturb and reset
                    pert = rng.normal(0, 0.03, n)
                    a_pert = a * np.exp(pert)
                    a_pert = np.maximum(a_pert, 0)
                    a_pert = a_pert / a_pert.sum()
                    obj_pert = compute_obj_normalized(a_pert)
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert.copy()
                    a = best_a.copy()
                    eta = 2.0 / np.sqrt(n)
                    patience = 0

        it += 1
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    EG it{it}: obj={best_obj:.8f}, eta={eta:.4f}, t={elapsed:.1f}s")

    return best_a


def spectral_descent(a, time_budget=100.0, verbose=False, seed=42):
    """
    Spectral descent: uses FFT structure to compute a direction that
    reduces the autocorrelation peak. Uses the "anti-peak" direction
    computed in the frequency domain.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / float(np.sum(a))

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    lr = 1.0 / np.sqrt(n)
    momentum = 0.5
    v_mom = np.zeros(n)
    it = 0
    patience = 0

    # Size for FFT
    N = 1
    while N < 3 * n:
        N <<= 1

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        peak_idx = int(np.argmax(conv))

        # Compute the "anti-peak" direction: gradient of (a*a)[peak_idx]
        # grad[i] = 2*a[peak_idx - i] if 0 <= peak_idx - i < n
        grad = np.zeros(n)
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        if lo <= hi:
            grad[lo:hi + 1] = 2.0 * a[peak_idx - hi:peak_idx - lo + 1]

        # Also add a small component from the second peak to avoid
        # over-focusing on one lag
        sorted_conv = np.sort(conv)[::-1]
        if len(sorted_conv) > 1 and sorted_conv[1] > 0.9 * max_conv:
            second_peaks = np.where(conv > 0.95 * max_conv)[0]
            for k in second_peaks:
                if k != peak_idx:
                    lo2 = max(0, k - n + 1)
                    hi2 = min(n - 1, k)
                    if lo2 <= hi2:
                        grad[lo2:hi2 + 1] += 0.3 * a[k - hi2:k - lo2 + 1]

        # Momentum
        v_mom = momentum * v_mom - lr * grad
        a_new = a + v_mom
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v_mom = np.zeros(n)
            lr *= 0.5
        else:
            a_new = a_new / s_new

        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1
            if patience > 20:
                lr *= 0.6
                v_mom = np.zeros(n)
                if lr < 1e-4:
                    # Perturbation escape
                    pert = rng.normal(0, 0.02, n)
                    a_pert = a * np.exp(pert)
                    a_pert = np.maximum(a_pert, 0)
                    a_pert = a_pert / a_pert.sum()
                    obj_pert = compute_obj_normalized(a_pert)
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert.copy()
                        a = a_pert
                    else:
                        a = best_a.copy()
                    lr = 1.0 / np.sqrt(n)
                    v_mom = np.zeros(n)
                    patience = 0

        it += 1
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    Spectral it{it}: obj={best_obj:.8f}, lr={lr:.6f}, t={elapsed:.1f}s")

    return best_a


def lp_polish(a, time_budget=60.0, verbose=False, seed=42):
    """LP-based polishing with adaptive blending and multiple LP solves."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    lp_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            # Try with slightly perturbed a
            a_pert = a + rng.normal(0, 0.005, n)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            g_lp, M_lp = solve_lp_minpeak(a_pert)
            if g_lp is None:
                break

        lp_count += 1
        best_t_obj = best_obj
        best_a_local = best_a.copy()

        # Try multiple blend factors
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 3:
                pert = rng.normal(0, 0.02, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / float(np.sum(a))
                stall = 0

    if verbose:
        print(f"    LP polish: {lp_count} LPs, obj={best_obj:.8f}")

    return best_a


def block_coordinate_descent(a, time_budget=100.0, block_size=3, verbose=False, seed=42):
    """
    Block coordinate descent: optimize groups of `block_size` consecutive
    coordinates simultaneously using a small grid/gradient search.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start < 1.0:
            pass
        sweep += 1
        conv = compute_conv(a)
        s = float(np.sum(a))

        # Process blocks
        block_starts = list(range(0, n, block_size))
        rng.shuffle(block_starts)

        for bs in block_starts:
            if time.monotonic() - start >= time_budget:
                break
            be = min(bs + block_size, n)
            block = a[bs:be].copy()
            block_sum = float(np.sum(block))
            if block_sum < 1e-12:
                continue

            # Try a few perturbation patterns for the block
            best_block = block.copy()
            best_block_obj = best_obj

            # Candidate patterns
            patterns = []
            # Uniform shift
            patterns.append(np.ones(be - bs) / (be - bs))
            # Perturbed current
            for _ in range(3):
                p = block + rng.normal(0, 0.1 * max(block.mean(), 1e-4), be - bs)
                p = np.maximum(p, 0)
                ps = float(np.sum(p))
                if ps > 1e-12:
                    patterns.append(p / ps)
            # Gaussian bump
            x = np.linspace(0, 1, be - bs)
            patterns.append(np.exp(-4 * (x - 0.5) ** 2))
            patterns[-1] = patterns[-1] / float(np.sum(patterns[-1]))
            # Linear ramp
            for direction in [1, -1]:
                p = np.linspace(0.1, 1.0, be - bs)
                if direction == -1:
                    p = p[::-1]
                patterns.append(p / float(np.sum(p)))

            for p in patterns:
                # Scale to maintain total sum
                scale = block_sum / max(float(np.sum(p)), 1e-12)
                new_block = p * scale

                # Evaluate
                a_trial = a.copy()
                a_trial[bs:be] = new_block
                obj_trial = 2.0 * n * float(np.max(compute_conv(a_trial))) / (s * s)
                if obj_trial < best_block_obj:
                    best_block_obj = obj_trial
                    best_block = new_block.copy()

            # Apply best block
            if best_block_obj < best_obj - 1e-14:
                a[bs:be] = best_block
                conv = compute_conv(a)
                s = float(np.sum(a))
                obj_now = 2.0 * n * float(np.max(conv)) / (s * s)
                if obj_now < best_obj:
                    best_obj = obj_now
                    best_a = a.copy()

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    BCD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        # Two bumps
        a = np.zeros(n)
        w = n // 4
        a[:w] = np.exp(-((np.arange(w) - w // 2) ** 2) / (2 * (w // 5) ** 2))
        a[n - w:] = np.exp(-((np.arange(w) - w // 2) ** 2) / (2 * (w // 5) ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Triangular
        x = np.linspace(0, 1, n)
        a = 1 - np.abs(2 * x - 1)
        a = a / a.sum()
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * (2 * x - 1)))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        # Alternating high-low
        a = np.ones(n)
        a[::2] = rng.uniform(0.5, 1.5)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def penalty_method(a, time_budget=60.0, verbose=False, seed=42):
    """
    Penalty method: minimize sum_k max(0, c_k - M)^2 + lambda*(sum(a)-1)^2
    using projected gradient descent. This directly targets reducing
    the autocorrelation peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / float(np.sum(a))

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Target M: start from current peak, slowly decrease
    conv = compute_conv(a)
    M_target = float(np.max(conv)) * 0.95
    lambda_ = 10.0
    lr = 2.0 / np.sqrt(n)
    it = 0

    # N for FFT-based gradient
    N = 1
    while N < 3 * n:
        N <<= 1

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak = float(np.max(conv))

        # Update M_target: slowly decrease
        M_target = max(peak * 0.98, peak - 0.01 / (it + 1))

        # Compute penalty gradient
        # w_k = max(0, c_k - M_target)
        w = np.maximum(conv - M_target, 0)
        w_sum = float(np.sum(w))

        if w_sum > 1e-12:
            # Gradient of sum_k w_k^2 w.r.t. a[i] = 2 * sum_k w_k * a[k-i] (if valid)
            # = 2 * (w_reversed * a)[n-1-i] ... use FFT
            w_ext = np.zeros(N)
            w_ext[:len(w)] = w
            a_ext = np.zeros(N)
            a_ext[:n] = a

            # c[i] = sum_j w[i+j] * a[j] for valid indices
            # This is the cross-correlation, computed via FFT
            # c = ifft(conj(FFT(w_ext)) * FFT(a_ext))[:n]
            W_fft = fft(w_ext)
            A_fft = fft(a_ext)
            c = ifft(np.conj(W_fft) * A_fft).real[:n]

            grad = 2.0 * c
        else:
            grad = np.zeros(n)

        # Add sum constraint gradient
        s = float(np.sum(a))
        grad += 2.0 * lambda_ * (s - 1.0)

        # Projected gradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
            lr *= 0.5
        else:
            a_new = a_new / s_new

        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            for shrink in [0.5, 0.1]:
                a_try = a - lr * shrink * grad
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    obj_try = compute_obj_normalized(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        break

        # Adaptive learning rate
        if it % 50 == 0:
            lr = max(lr * 0.95, 0.01 / n)

        it += 1
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    Penalty it{it}: obj={best_obj:.8f}, M_t={M_target:.6f}, t={elapsed:.1f}s")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search: peak-aware CD, exponentiated gradient, spectral descent,
    block CD, penalty method, LP polishing, and adaptive n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj_normalized(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj_normalized(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(800) / 800.0
        best_score = 2.0

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")

    # === Phase 1: Peak-aware Coordinate Descent ===
    print(f"\n=== Phase 1: Peak-Aware CD (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 420.0)
    if p1_budget > 30:
        a_cd = coordinate_descent(best_seq, time_budget=p1_budget, verbose=True, seed=seed)
        obj_cd = compute_obj_normalized(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 2: Exponentiated Gradient ===
    print(f"\n=== Phase 2: Exponentiated Gradient ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 180.0)
    if p2_budget > 20:
        a_eg = exponentiated_gradient(best_seq, time_budget=p2_budget,
                                      verbose=True, seed=seed + 1)
        obj_eg = compute_obj_normalized(a_eg)
        if obj_eg < best_score:
            best_score = obj_eg
            best_seq = a_eg.copy()
            print(f"  EG: obj={best_score:.8f} **")
        else:
            print(f"  EG: obj={obj_eg:.8f}")
            # Follow up with CD
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_cd2 = coordinate_descent(a_eg, time_budget=min(30, remaining * 0.05),
                                           verbose=False, seed=seed + 2)
                obj_cd2 = compute_obj_normalized(a_cd2)
                if obj_cd2 < best_score:
                    best_score = obj_cd2
                    best_seq = a_cd2.copy()
                    print(f"  CD after EG: obj={best_score:.8f} **")

    # === Phase 3: Spectral Descent ===
    print(f"\n=== Phase 3: Spectral Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.10, 120.0)
    if p3_budget > 15:
        a_spec = spectral_descent(best_seq, time_budget=p3_budget,
                                  verbose=False, seed=seed + 3)
        obj_spec = compute_obj_normalized(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Spectral: obj={best_score:.8f} **")
        else:
            print(f"  Spectral: obj={obj_spec:.8f}")

    # === Phase 4: Penalty Method ===
    print(f"\n=== Phase 4: Penalty Method ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.10, 120.0)
    if p4_budget > 15:
        a_pen = penalty_method(best_seq, time_budget=p4_budget,
                               verbose=False, seed=seed + 4)
        obj_pen = compute_obj_normalized(a_pen)
        if obj_pen < best_score:
            best_score = obj_pen
            best_seq = a_pen.copy()
            print(f"  Penalty: obj={best_score:.8f} **")
        else:
            print(f"  Penalty: obj={obj_pen:.8f}")

    # === Phase 5: LP Polishing ===
    print(f"\n=== Phase 5: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.20, 250.0)
    if p5_budget > 20:
        for rnd in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(p5_budget / (4 - rnd), remaining * 0.5)
            if alloc < 10:
                break
            a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True,
                             seed=1000 + rnd * 13)
            obj_lp = compute_obj_normalized(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
            # Follow-up CD
            remaining = deadline - time.monotonic()
            if remaining > 20 and rnd < 3:
                a_cd3 = coordinate_descent(best_seq, time_budget=min(20, remaining * 0.06),
                                           verbose=False, seed=2000 + rnd)
                obj_cd3 = compute_obj_normalized(a_cd3)
                if obj_cd3 < best_score:
                    best_score = obj_cd3
                    best_seq = a_cd3.copy()
                    print(f"  CD after LP{rnd}: obj={best_score:.8f} **")

    # === Phase 6: n Exploration ===
    print(f"\n=== Phase 6: n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    # Smart n candidates: around current, plus spread
    n_candidates_base = [n_current - 100, n_current - 200,
                         n_current + 100, n_current + 200, n_current + 400,
                         n_current + 600, n_current + 1000,
                         500, 600, 700, 900, 1000, 1200, 1500, 2000, 2500, 3000]
    n_filtered = []
    for nc in sorted(set(n_candidates_base)):
        if 300 <= nc <= 4000:
            if all(abs(nc - nf) > 80 for nf in n_filtered):
                n_filtered.append(nc)
    n_filtered = n_filtered[:7]
    print(f"  n candidates: {n_filtered}")

    n_best_found = n_current
    n_best_score = best_score

    for n_try in n_filtered:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick optimization: CD + brief EG
        alloc_cd = min(remaining * 0.12, 40.0)
        alloc_eg = min(remaining * 0.05, 15.0)
        if alloc_cd > 10:
            a_try = coordinate_descent(a_try, time_budget=alloc_cd,
                                       verbose=False, seed=n_try)
        obj_try = compute_obj_normalized(a_try)
        if obj_try < n_best_score - 1e-6:
            # Promising! Do more optimization
            if remaining - time.monotonic() > alloc_eg + 10:
                a_try = exponentiated_gradient(a_try, time_budget=alloc_eg,
                                               verbose=False, seed=n_try + 100)
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best_found = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try} (after EG): obj={obj_try:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # If we found a better n, go back and polish at the new n
    if n_best_found != n_current:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_final = coordinate_descent(best_seq, time_budget=min(remaining * 0.2, 60.0),
                                         verbose=False, seed=5000)
            obj_final = compute_obj_normalized(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Polish at n={len(best_seq)}: obj={best_score:.8f} **")

    # === Phase 7: Diverse Restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = 1 + r
        a_r = diverse_init(n_cur, rng, mode=mode)
        blend = 0.4 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / float(np.sum(a_r))

        alloc = min(remaining * 0.15, 50.0)
        if alloc > 15:
            a_r = coordinate_descent(a_r, time_budget=alloc, verbose=False,
                                     seed=7000 + r * 100)
            obj_r = compute_obj_normalized(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode {mode}): obj={obj_r:.8f}")

    # === Phase 8: Final Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        for fr in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            # CD
            cd_alloc = min(remaining * 0.3, 60.0)
            if cd_alloc > 8:
                a_f = coordinate_descent(best_seq, time_budget=cd_alloc,
                                         verbose=False, seed=9000 + fr)
                obj_f = compute_obj_normalized(a_f)
                if obj_f < best_score:
                    best_score = obj_f
                    best_seq = a_f.copy()
                    print(f"  Final CD {fr}: obj={best_score:.8f} **")

            # LP
            remaining = deadline - time.monotonic()
            if remaining > 8:
                lp_alloc = min(remaining * 0.25, 40.0)
                if lp_alloc > 6:
                    a_f2 = lp_polish(best_seq, time_budget=lp_alloc,
                                     verbose=False, seed=9500 + fr)
                    obj_f2 = compute_obj_normalized(a_f2)
                    if obj_f2 < best_score:
                        best_score = obj_f2
                        best_seq = a_f2.copy()
                        print(f"  Final LP {fr}: obj={best_score:.8f} **")

            # Quick EG
            remaining = deadline - time.monotonic()
            if remaining > 10:
                eg_alloc = min(remaining * 0.2, 20.0)
                if eg_alloc > 5:
                    a_f3 = exponentiated_gradient(best_seq, time_budget=eg_alloc,
                                                  verbose=False, seed=9700 + fr)
                    obj_f3 = compute_obj_normalized(a_f3)
                    if obj_f3 < best_score:
                        best_score = obj_f3
                        best_seq = a_f3.copy()
                        print(f"  Final EG {fr}: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = 2.0

    # Ensure valid output
    best_seq = np.maximum(best_seq, 0.0)
    result = [float(min(1000.0, x)) for x in best_seq]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
