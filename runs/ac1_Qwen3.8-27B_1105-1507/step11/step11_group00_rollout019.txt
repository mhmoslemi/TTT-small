def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)

def autocorr_gradient(a, k):
    """Gradient of r(k) = sum_j a[j]*a[j+k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    if k < n:
        g[:n-k] += a[k:]
        g[k:] += a[:n-k]
    return g

def topk_gradient(a, conv, k=3, temperature=10.0):
    """Weighted combination of top-k lag gradients."""
    n = len(a)

        g += w * autocorr_gradient(a, idx)
    return 2.0 * g  # Factor of 2 for the quadratic form

    """Optimize a[i] while keeping other coordinates fixed."""
    n = len(a)
    t = a[i]
    a[i] = 0
    
    # Compute r(k) for all k with a[i] = 0
    r_0 = compute_conv(a)  # length 2n-1, but we only need k=0..n-1
    
    # Compute c_k = coefficient of t in r(k)
    c = np.zeros(n)
    if i < n - 0:  # i+k < n for k >= 0
        # a[i+k] contributes to r(k) for k = 0, 1, ..., n-1-i
        for k in range(n - i):
            c[k] += a[i + k]
    if i >= 0:
        # a[i-k] contributes to r(k) for k = 0, 1, ..., i
        for k in range(i + 1):
            c[k] += a[i - k]
    
    # r(k|t) = r_0[k] + t * c[k] for k = 0, ..., n-1
    # We want to minimize max_k (r_0[k] + t * c[k]) subject to t >= 0, t <= 1 - sum(a[-i])
    
    # The function f(t) = max_k (r_0[k] + t*c[k]) is convex and piecewise linear.
    # Its minimum is at a "kink" where two lines cross, or at a boundary.
    
    # Since f is convex, we can use binary search on the subgradient.
    # The subgradient of f at t is: c[k*] where k* = argmax(r_0[k] + t*c[k]).
    # At the optimum, 0 is in the subgradient, i.e., there exist k1, k2 such that
    # c[k1] < 0 < c[k2] and the lines k1 and k2 cross at the optimum.
    
    # Simpler: just do golden section search or bisection on the derivative.
    
    t_min = 0.0
    t_max = 1.0 - (float(np.sum(a)) - t)  # = 1 - sum(a[j] for j != i)
    t_max = max(0.0, t_max)
    
    # Bisection on the subgradient
    for _ in range(50):
        t_mid = (t_min + t_max) / 2
        r_mid = r_0[:n] + t_mid * c
        k_star = int(np.argmax(r_mid))
        subgrad = c[k_star]
        if subgrad > 0:
            t_max = t_mid
        elif subgrad < 0:
            t_min = t_mid
        else:
            break
    
    t_opt = (t_min + t_max) / 2
    a[i] = t_opt
    return a
