def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def lp_solve_sparse(a):
    """Solve min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1 using sparse matrix."""
    n = len(a)
    # Build sparse constraint matrix
    # A[k,j] = a[j] if 0 <= k-j < n, else 0
    # For each column j, nonzeros are at rows j, j+1, ..., j+n-1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):
            if k < 2 * n - 1:
                rows.append(k)
                cols.append(j)
                vals.append(a[j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None


def line_search(a, direction, best_obj, max_alpha=1.0, steps=12):
    """Find best step size alpha in (0, max_alpha] along direction."""
    n = len(a)
    best_alpha = 0
    best_local = best_obj
    
    alphas = np.geomspace(0.01, max_alpha, steps)
    for alpha in alphas:
        a_try = a + alpha * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj_try = compute_obj(a_try)
        if obj_try < best_local:
            best_local = obj_try
            best_alpha = alpha
    
    return best_alpha, best_local

def subgradient_descent(a, time_budget, verbose=False, seed=42):
    """
    Direct subgradient descent on the normalized objective with Polyak step.
    Uses smoothing for better gradients when stuck.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:

    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Polyak step size: step = (f(a) - f*) / ||g||^2
    # We don't know f*, so use (f(a) - lower_bound) as surrogate
    # Lower bound estimate: 2 (from uniform sequence analysis) minus some margin
    lower_bound = max(1.0, best_obj * 0.9)
    
    step_scale = 1.0
    no_improve = 0
    
    for it in range(5000):
        remaining = time.monotonic() and time_budget - (time.monotonic() - start)
        if remaining < 2:
            break
        
        # Compute subgradient of max(conv) 
        subgrad, max_val, active_lags = compute_subgradient(a)
        
        # Also compute gradient of the full objective 2n*max_conv/(sum)^2
        # Since we work with normalized a (sum=1), obj = 2n*max_conv
        # grad_obj = 2n * subgrad / (but we need to project onto sum(a)=1)
        
        gn = float(np.linalg.norm(subgrad))
        if gn < 1e-15:
            break
        
        # Polyak-type step
        gap = max(0.01, best_obj - lower_bound)
        step = step_scale * gap / (gn * gn * n)
        step = min(step, 0.1)
        
        a_try = a - step * subgrad
        # Project: a >= 0, sum(a) = 1
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            a_try = best_a.copy()
        else:
            a_try = a_try / st
        
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            no_improve = 0
            step_scale = min(2.0, step_scale * 1.05)
            if verbose and it % 100 == 0:
                print(f"    SGD it{it:4d}: obj={best_obj:.10f}, step={step:.6f}")
        else:
            no_improve += 1
            step_scale = max(0.1, step_scale * 0.95)
            if no_improve > 50:

                a = a + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
            if no_improve > 200:

                break

    return best_a

            # Smoothed gradient
            conv = compute_conv(y)
            exact_max = float(np.max(conv))
            shifted = tau * (conv - exact_max)
            shifted = np.clip(shifted, -500, 500)
            exp_vals = np.exp(shifted)
            sum_exp = float(np.sum(exp_vals))
            if sum_exp < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = exp_vals / sum_exp

            # Gradient via FFT cross-correlation
            N_fft = 1
            while N_fft < 4 * n:
                N_fft <<= 1
            P = fft(p, N_fft)
            Y = fft(y, N_fft)
            cross = ifft(P * np.conj(Y)).real
            g = 2.0 * cross[:n]

            gn = float(np.linalg.norm(g))
            if gn < 1e-15:
                break

            v = beta * v - lr * g / gn
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new
            iters += 1
            lr *= 0.99995

        if verbose:
            print(f"    tau={tau:5d}: obj={best_obj:.10f}, iters={iters}")

    return best_a


def deflation_iteration(a, time_budget, verbose=False, seed=42, max_iters=500):
    """
    Targeted deflation: iteratively reduce the peak of the convolution
    by adjusting specific elements that contribute most to the peak.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        # Find top K lags
        K = min(5, 2*n-1)
        top_k_indices = np.argsort(conv)[-K:]
        
        # Build contribution matrix for top lags
        # For each element a[i], compute how much it contributes to the top lags
        contrib = np.zeros(n)
        for lag in top_k_indices:
            for i in range(n):
                j = lag - i
                if 0 <= j < n:
                    contrib[i] += a[i] * a[j] * (conv[lag] / max_val)
        
        # Reduce elements with high contribution, boost elements with low
        contrib_norm = contrib / (np.max(contrib) + 1e-10)
        
        # Try multiple step sizes
        improved = False
        for step in [0.002, 0.005, 0.01, 0.02, 0.03]:
            a_try = a - step * contrib_norm
            # Also boost the least contributing elements slightly
            a_try += step * 0.1 * (1 - contrib_norm)
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        
        if not improved:
            # Fallback: mix with LP solution
            if it % 10 == 0:
                g_lp, _ = lp_solve_sparse(a)
                if g_lp is not None:
                    for t in [0.05, 0.1, 0.2, 0.3]:
                        a_try = (1-t)*a + t*g_lp
                        a_try = np.maximum(a_try, 0)
                        a_try = a_try / a_try.sum()
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break
            if not improved:
                # Random restart from best
                if it > 100 and it % 50 == 0:
                    a = best_a + 0.02 * rng.standard_normal(n)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                else:
                    a = a + 0.005 * rng.standard_normal(n)
                    a = np.maximum(a, 0)
                    a = a / a.sum()

        if verbose and it % 50 == 0:
            print(f"    Deflation it{it:4d}: obj={best_obj:.10f}")

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_init(n, mode=0):
    """Create structured initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.maximum(np.sinc(x * 1.5), 0.01)
        return a / a.sum()
    elif mode == 2:
        x = np.arange(n) / (n - 1) if n > 1 else np.array([0.5])
        a = 0.54 - 0.46 * np.cos(2 * np.pi * x)
        return a / a.sum()
    elif mode == 3:
        x = np.arange(n) / (n - 1) if n > 1 else np.array([0.5])
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        return a / a.sum()
    elif mode == 4:
        x = np.linspace(-2.5, 2.5, n)
        a = np.exp(-0.8 * x ** 2)
        return a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
        return a / a.sum()
    elif mode == 6:
        # Slightly perturbed uniform
        rng = np.random.default_rng(123)
        a = 1.0 + 0.1 * rng.standard_normal(n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 7:
        # Two bumps
        x = np.arange(n) / n
        a = np.exp(-20*(x-0.25)**2) + np.exp(-20*(x-0.75)**2) + 0.05
        return a / a.sum()
    else:
        rng = np.random.default_rng(42)
        a = rng.uniform(0, 1, n)
        return a / a.sum()

            s = float(np.sum(parent_arr))
