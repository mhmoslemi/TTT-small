```python
import numpy as np
from scipy.fft import rfft, irfft
import time
import math

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def obj_normalized(a):
    """Objective for a normalized (sum=1) sequence."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def simplex_project(v):
    """Project onto probability simplex {x : x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_descent(a, time_budget, seed=42):
    """Subgradient descent with simplex projection."""
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    eta0 = 2.0 / (np.sqrt(n) * 10.0)
    t = 0
    patience = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        S = np.where(conv >= max_c * (1 - 1e-12))[0]
        
        # Compute subgradient
        subgrad = np.zeros(n)
        for k in S:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                subgrad[lo:hi+1] += 2.0 * a[k-hi:k-lo+1]
        subgrad = subgrad / len(S)
        
        # Step size
        eta = eta0 / np.sqrt(t + 1)
        
        # Update
        a_new = simplex_project(a - eta * 2.0 * n * subgrad)
        obj_new = obj_normalized(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1
            if patience > 50:
                # Reduce step size
                eta0 *= 0.5
                if eta0 < 1e-6:
                    # Add random perturbation
                    a_pert = a * np.exp(rng.normal(0, 0.1, n))
                    a = simplex_project(a_pert)
                    eta0 = 2.0 / (np.sqrt(n) * 5.0)
                    patience = 0
        
        t += 1
    
    return best_a
```

```python
import numpy as np
```

```python
def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def obj_norm(a):
    """Compute 2n*max(conv) for a normalized (sum=1) sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_prefix_suffix_max(conv):
    """Compute prefix and suffix max arrays for fast range-max queries."""
    L = len(conv)
    prefix = np.full(L + 1, -np.inf)
    for i in range(L):
        prefix[i + 1] = max(prefix[i], conv[i])
    suffix = np.full(L + 1, -np.inf)
    for i in range(L - 1, -1, -1):
        suffix[i] = max(suffix[i + 1], conv[i])
    return prefix, suffix

                a_ki = a[ks_lin - i]
                c_lin = conv[ks_lin].copy()
            else:
                a_ki = np.array([])
                c_lin = np.array([])
                ks_lin = np.array([], dtype=int)

            c_2i = float(conv[quad_k]) if 0 <= quad_k < conv_len else 0.0

            # Bracketing
            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, v * 0.25, v * 0.5, v * 0.75, v,
                          v * 1.25, v * 1.5, v * 2.0, v * 3.0,
                          1.0 / (2 * n), 2.0 / n]
            best_x = v
            best_f = f1d_eval(v, v, s_other, a_ki, c_lin, c_2i, unaffected_max, n)
            for c in candidates:
                c = max(0.0, float(c))
                fc = f1d_eval(c, v, s_other, a_ki, c_lin, c_2i, unaffected_max, n)
                if fc < best_f:
                    best_f = fc
                    best_x = c

            # Golden section
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.0
            if x_hi - x_lo < 1e-14:
                x_lo = 0.0
                x_hi = max(scale, 1e-8)

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f1d_eval(c_pt, v, s_other, a_ki, c_lin, c_2i, unaffected_max, n)
            fd = f1d_eval(d_pt, v, s_other, a_ki, c_lin, c_2i, unaffected_max, n)

            for _ in range(35):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f1d_eval(c_pt, v, s_other, a_ki, c_lin, c_2i, unaffected_max, n)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f1d_eval(d_pt, v, s_other, a_ki, c_lin, c_2i, unaffected_max, n)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) > 1e-14:
                delta = x_opt - v
                a[i] = x_opt
                if 0 <= quad_k < conv_len:
                    conv[quad_k] += x_opt * x_opt - v * v
                if len(ks_lin) > 0:
                    conv[ks_lin] += 2.0 * delta * a_ki
                any_improved = True
                # Update prefix/suffix (expensive but ensures correctness)
                # Only do this periodically to save time
                pass

        # Periodic renormalization and recomputation
        if sweep % 15 == 0:
            s_now = float(np.sum(a))
            if s_now > 1e-10:
                a = a / s_now
                conv = compute_conv(a)
                obj = 2.0 * n * float(np.max(conv))
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                prefix, suffix = compute_prefix_suffix_max(conv)

        # Check for stall
        if sweep % 15 == 0:
            if best_obj < prev_best - 1e-13:
                prev_best = best_obj
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 6:
                    break

    return best_a


def mirror_descent(a, time_budget, seed=42):
    """Mirror descent (entropy-regularized) for polishing."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    eta = 0.3
    temp = 5.0
    patience = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        if max_c < 1e-15:
            break

        # Soft weights over convolution entries
        weights = np.exp((conv - max_c) / (temp * max_c + 1e-15))
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[np.argmax(conv)] = 1.0
        else:
            weights /= w_sum

        # Compute gradient: grad[i] = 2 * sum_k weights[k] * a[k-i] * 1[0<=k-i<n]
        # This is 2 * cross_correlation(a, weights)[i]
        # Use FFT: cross_corr[i] = sum_k a[k] * weights[k-i] (with bounds)
        # = conv(a, weights_rev)[n-1+i] ... let me just do it directly for n<=3000
        grad = np.zeros(n)
        active = np.where(weights > 1e-12)[0]
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += weights[k] * a[k - hi:k - lo + 1]
        grad *= 2.0

        g_mean = float(np.mean(grad))
        if g_mean < 1e-15:
            break

        # Multiplicative update (mirror descent with neg-entropy)
        log_ratio = np.log(np.maximum(grad, 1e-15)) - math.log(g_mean)
        a_new = a * np.exp(-eta * log_ratio)
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            patience += 1
            if patience > 30:
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = a / a.sum()
                eta = 0.3
                patience = 0
            continue
        a_new = a_new / s_new

        obj_new = obj_norm(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1
            if patience % 20 == 0:
                eta *= 0.7
                temp *= 1.2
            if patience > 80:
                a = a * np.exp(rng.normal(0, 0.03, n))
                a = a / a.sum()
                eta = 0.3
                patience = 0

    return best_a


def make_initial(n, mode, rng):
    """Generate diverse initial sequences."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = a / a.sum()
    elif mode == 3:
        x = np.arange(n)
        c = n / 2
        sigma = n / 4
        a = np.exp(-(x - c) ** 2 / (2 * sigma ** 2))
        a = a / a.sum()
    elif mode == 4:
        # Two bumps
        w = n // 4
        a = np.zeros(n)
        a[:w] = np.exp(-((np.arange(w) - w / 2) ** 2) / (2 * (w / 5) ** 2))
        a[n - w:] = np.exp(-((np.arange(w) - w / 2) ** 2) / (2 * (w / 5) ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        # Symmetric random
        half = n // 2
        a_half = rng.uniform(0.1, 1.0, half)
        if n % 2 == 0:
            a = np.concatenate([a_half, a_half[::-1]])
        else:
            a = np.concatenate([a_half, [max(a_half[-1], a_half[0])], a_half[::-1]])
        a = a / a.sum()
    elif mode == 7:
        # Raised cosine variant
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def symmetrize(a):
    """Average with reverse to encourage symmetry."""
    return 0.5 * (a + a[::-1])
```
