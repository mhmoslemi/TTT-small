import numpy as np
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_conv(a_norm):
    """Compute 2n*max(conv) for already-normalized a (sum=1)."""
    n = len(a_norm)
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n

    lr = 0.05 / n  # smaller default

    patience = 500
    
    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** frac
        
        grad = lse_gradient_conv(a, T)
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-12:
            break
        
        a_old = a.copy()
        obj_old = lse_objective(a, T)
        t = lr

        for _ in range(8):

def multi_peak_subgrad(a, top_k):
    """Multi-peak subgradient."""
    n = len(a)
    conv = compute_conv(a)
    k_peaks = np.argsort(conv)[::-1][:top_k]
    peak_values = conv[k_peaks]
    weights = peak_values ** 2 / (np.sum(peak_values ** 2) + 1e-15)
    
    grad = np.zeros(n)
    for ki, w in zip(k_peaks, weights):
        ki = int(ki)
        lo1 = max(0, ki - n + 1)
        hi1 = min(n - 1, ki)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += w * a[ki - hi1:ki - lo1 + 1][::-1]
        lo2 = max(0, ki)
        hi2 = min(n - 1, ki + n - 1)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] += w * a[lo2 - ki:hi2 - ki + 1]
    return grad


def multi_peak_subgrad_descent(a, top_k, time_budget):
    """Multi-peak subgradient descent with heavy ball momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_conv(a)
    start = time.monotonic()
    lr = 0.1 / n
    momentum = np.zeros(n)
    beta = 0.8
    it = 0
    best_it = 0
    patience = 2000
    
    while time.monotonic() - start < time_budget:
        grad = multi_peak_subgrad(a, top_k)
        
        a_old = a.copy()
        obj_old = compute_obj_conv(a)
        t = lr
        improved = False
        for _ in range(6):
            momentum = beta * momentum - t * grad
            a_try = project_simplex(a_old + momentum)
            obj_try = compute_obj_conv(a_try)
            if obj_try <= obj_old:
                a = a_try
                improved = True
                break
            t *= 0.5
            momentum = beta * momentum - t * grad
        
        if not improved:
            momentum *= 0.3
            lr *= 0.95
        
        obj_true = compute_obj_conv(a)
        if obj_true < best_obj:
            best_obj = obj_true
            best_a = a.copy()
            best_it = it
        else:
            it += 1
            if it - best_it > patience:
                a = best_a.copy()
                momentum = np.zeros(n)
                lr = 0.05 / n
                patience = max(500, patience // 2)
                best_it = it
    
    return best_a


def coordinate_descent(a, time_budget, max_sweeps=None):
    """Enhanced coordinate descent with more candidates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_conv(a)
    start = time.monotonic()
    sweep = 0
    
    while time.monotonic() - start < time_budget:
        if max_sweeps is not None and sweep >= max_sweeps:
            break
        sweep_improved = False
        
        for i in range(n):
            if time.monotonic() - start >= time_budget:
                break
            
            ai_old = a[i]
            
            # More diverse candidates
            candidates = [0.0, ai_old * 0.5, ai_old * 0.75, ai_old * 1.25, ai_old * 1.5, ai_old * 2.0]
            # Add neighbors
            if i > 0:
                candidates.append(a[i-1] * 0.8)
            if i < n-1:
                candidates.append(a[i+1] * 0.8)
            
            best_val = ai_old
            best_local = best_obj
            
            for cv in candidates:
                a[i] = max(0.0, cv)
                s_new = float(np.sum(a))
                if s_new < 0.01:
                    a[i] = ai_old
                    continue
                a_try = a / s_new
                obj_try = compute_obj_conv(a_try)
                if obj_try < best_local:
                    best_local = obj_try
                    best_val = max(0.0, cv)
            
            if best_val != ai_old:
                a[i] = best_val
                s_new = float(np.sum(a))
                if s_new > 0.01:
                    a = a / s_new
                    sweep_improved = True
            else:
                a[i] = ai_old
        
        obj_now = compute_obj_conv(a)
        if obj_now < best_obj:
            best_obj = obj_now
            best_a = a.copy()
        
        if not sweep_improved:
            break
        sweep += 1
    
    return best_a


def parametric_optimize(a_init, time_budget, num_basis=5):
    """
    Parametric optimization using Chebyshev-like basis functions.
    Parametrize a[i] = max(0, w(t)) where t = 2*i/(n-1) - 1 and
    w(t) = sum_k c_k * (1 + t^2)^k (positive by construction)
    Actually, use: w(t) = exp(sum_k c_k * T_k(t)) where T_k are Chebyshev polynomials.
    This guarantees positivity.
    """
    n = len(a_init)
    t = 2.0 * np.arange(n) / (n - 1) - 1.0
    
    # Build Chebyshev basis
    basis = np.zeros((num_basis, n))
    basis[0, :] = 1.0
    basis[1, :] = t
    for k in range(2, num_basis):
        basis[k, :] = 2 * t * basis[k-1, :] - basis[k-2, :]
    
    # Initial coefficients: fit to a_init
    # a_init[i] = exp(sum_k c_k * basis[k, i])
    # So c should satisfy: log(a_init) ≈ basis @ c
    log_a = np.log(np.maximum(a_init, 1e-15))
    c_init, _, _, _ = np.linalg.lstsq(basis.T, log_a, rcond=None)
    
    def parametric_obj(c):
        a = np.exp(basis @ c)
        # Ensure non-negative (always true for exp)
        s = float(np.sum(a))
        if s < 0.01:
            return 1e10
        a = a / s
        return compute_obj_conv(a)
    
    def parametric_obj_smooth(c, T=20.0):
        a = np.exp(basis @ c)
        s = float(np.sum(a))
        if s < 0.01:
            return 1e10
        a = a / s
        return lse_objective(a, T)
    
    # Use Nelder-Mead
    from scipy.optimize import minimize
    
    best_c = c_init.copy()
    best_val = parametric_obj(best_c)
    
    # Try different T values
    for T in [20.0, 50.0, 100.0]:
        result = minimize(parametric_obj_smooth, best_c, args=(T,),
                         method='Nelder-Mead',
                         options={'maxiter': 500, 'xatol': 1e-6, 'fatol': 1e-8,
                                  'adaptive': True})
        if result.fun < best_val:
            best_c = result.x
            best_val = parametric_obj(best_c)
    
    # Also try with direct objective
    result = minimize(parametric_obj, best_c, method='Nelder-Mead',
                     options={'maxiter': 1000, 'xatol': 1e-8, 'fatol': 1e-10,
                              'adaptive': True})
    if result.fun < best_val:
        best_c = result.x
        best_val = result.fun
    
    a_result = np.exp(basis @ best_c)
    a_result = a_result / a_result.sum()
    return a_result


def u_shaped_init(n, sharpness):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s


def arcsine_init(n):
    x = np.linspace(0.005, 0.995, n)
    a = 1.0 / np.sqrt(np.maximum(1e-10, 1 - (2 * x - 1) ** 2))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s


def checkers_init(n, freq):
    """Checkerboard-like pattern."""
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * (2*x - 1))
    a = np.maximum(a, 0.01)
    s = float(np.sum(a))
    return a / s


def lp_polish_loop(a, time_budget, use_full_lp=True, max_iters=50):
    """Iterative LP polishing."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj_conv(a)
    start = time.monotonic()
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        if use_full_lp:
            g_lp = full_lp_solve(best_a)
        else:
            g_lp = full_lp_solve(best_a)  # Same for now
        
        if g_lp is None:
            break
        
        # Adaptive blending
        a_norm = best_a / best_a.sum()
        g_norm = g_lp / g_lp.sum()
        
        # Find best blend
        best_t = 0.0
        best_blend_obj = best_obj
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a_norm + t * g_norm
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 0.01:
                a_try = a_try / s_try
                obj_try = compute_obj_conv(a_try)
                if obj_try < best_blend_obj:
                    best_blend_obj = obj_try
                    best_t = t
        
        if best_t > 0 and best_blend_obj < best_obj - 1e-14:
            a_new = (1 - best_t) * a_norm + best_t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj_conv(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1
        
        it += 1
        if it % 5 == 0:
            print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, "
                  f"elapsed={time.monotonic()-start:.1f}s")
        
        if stall >= 5:
            # Perturb
            rng_local = np.random.default_rng(it * 104729)
            noise = rng_local.normal(0, 0.003 * min(stall, 5), n)
            a_pert = best_a + noise
            a_pert = np.maximum(a_pert, 0)
            s_p = float(np.sum(a_pert))
            if s_p > 0.01:
                best_a = a_pert / s_p
                best_obj = compute_obj_conv(best_a)
            stall = 0
    
    return best_a

**Key mathematical insight**: The problem is equivalent to finding a probability vector a that minimizes the peak of its autocorrelation. This is related to:
- Flat-top filter design
- Minimum variance beamforming
- Out-of-band emissions minimization

For the continuous version, the optimal solution is known: it's the sinc function (truncated) or related functions. For the discrete version with the non-negativity constraint, the optimal shape is U-shaped (concentrated at edges).

**Another idea: Use the fact that the LP is a "dual" problem.**

The LP we're solving is:
min M
s.t. (a*g)[k] <= M for all k
     sum(g) = 1
     g >= 0

The dual is:
max sum_k y_k
s.t. sum_k y_k * (a*g)[k] coefficient of g[m] >= 1 for all m
     y_k >= 0

The dual has 2n-1 variables (y_k) and n constraints. For n=800, that's 1599 variables and 800 constraints. This might be faster to solve!

Actually, one more important consideration: the `full_lp_solve` function builds the matrix A in a nested Python loop, which is slow for n=800 (1.3M iterations in Python). Let me vectorize it.

```python
def build_conv_matrix(a_norm):
    """Build matrix A where A[k, m] = a_norm[k-m] if 0 <= k-m < n, else 0."""
    n = len(a_norm)
    num_k = 2 * n - 1
    # k ranges from 0 to 2n-2, m ranges from 0 to n-1
    # A[k, m] = a_norm[k-m] if 0 <= k-m < n
    ks = np.arange(num_k)
    ms = np.arange(n)
    # k-m ranges from -（n-1) to 2n-2
    diffs = ks[:, None] - ms[None, :]  # shape (2n-1, n)
    mask = (diffs >= 0) & (diffs < n)
    A = np.zeros((num_k, n))
    A[mask] = a_norm[diffs[mask]]
    return A
```

This is vectorized and fast.
