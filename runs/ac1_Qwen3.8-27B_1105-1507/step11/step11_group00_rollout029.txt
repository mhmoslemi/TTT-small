"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Improved mirror descent with high-to-low temperature schedule,
peak-focused sparse LP, spectral reshaping, and theoretical initialization
based on the 1/sqrt(k) heuristic. Combines fast iterative methods with
occasional LP-based direction finding for escaping local minima.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def gradient_from_p(a, p):
    """Compute 2 * sum_j a[j] * p[i+j] for i=0..n-1. p has length 2n-1."""
    n = len(a)
    b = a[::-1]
    N_fft = 1
    while N_fft < 3 * n - 1:
        N_fft <<= 1
    c = irfft(rfft(b, N_fft) * rfft(p, N_fft), N_fft)
    return 2.0 * c[n - 1:2 * n - 1]


def smoothed_gradient(a, conv, tau):

        max_conv = 0.0

    sum_exp = float(np.sum(exp_vals))

def mirror_descent_step(a, grad, eta):
    """Multiplicative weights: a_i <- a_i * exp(-eta * grad_i), normalize."""
    # Center the gradient for stability
    grad_centered = grad - np.mean(grad)
    exponent = -eta * grad_centered
    exponent = np.clip(exponent, -40, 40)
    a_new = a * np.exp(exponent)
    s = float(np.sum(a_new))
    if s < 1e-300:
        return a.copy()
    return a_new / s


def backtracking_line_search(f, a, direction, max_iter=20):
    """Backtracking line search: find t in (0,1] such that f(a + t*dir) < f(a)."""
    f0 = f(a)
    t = 1.0
    c1 = 1e-4
    g_dir = np.dot(np.zeros(len(a)), direction)  # placeholder
    for i in range(max_iter):
        a_try = np.maximum(a + t * direction, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            t *= 0.5
            continue
        a_try = a_try / s
        f_try = f(a_try)
        if f_try <= f0 - c1 * t * abs(g_dir) or f_try < f0:
            return t, a_try
        t *= 0.5
    return t, a


def lp_direction_sparse(a, top_k=20, time_limit=5.0):
    """
    Solve LP with only top_k peak constraints for speed.
    min M s.t. 2*conv(a,g)_k - conv(a,a)_k <= M for k in top_peaks
              sum(g) = 1, g >= 0
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv_aa = compute_conv(a_norm)

    # Select top-k constraints
    k_idx = np.argsort(conv_aa)[-top_k:]

    # Build sparse constraint matrix for selected rows
    # A[k, j] = a_norm[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    rhs = []
    for k in k_idx:
        rhs.append(conv_aa[k])
        for j in range(n):
            diff = k - j
            if 0 <= diff < n:
                rows.append(k)
                cols.append(j)
                vals.append(a_norm[diff])

    # Convert to sparse
    row_map = {k: i for i, k in enumerate(k_idx)}
    rows_mapped = [row_map[r] for r in rows]

    A = cp.SparseMatrix((vals, (rows_mapped, cols)), shape=(len(k_idx), n))

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
   constraints = [2.0 * (A @ g) - M <= np.array(rhs), cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    start = time.monotonic()
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, time_limit=time_limit)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
        if time.monotonic() - start > time_limit:
            break
    return None, None


def spectral_reshape(a, n_peaks=5, strength=0.3):
    """
    Modify the spectrum to reduce autocorrelation peaks.
    In frequency domain: conv(a,a) corresponds to (DFT(a))^2.
    We try to make (DFT(a))^2 more flat by adjusting DFT(a).
    """
    n = len(a)
    # Use full FFT for the convolution (zero-padded)
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    A = fft(a, size)
    A_sq = A * A  # This is the FFT of conv(a,a)

    # Inverse to get time-domain convolution
    conv = np.real(ifft(A_sq))[:2 * n - 1]

    # Find peak lags
    peak_idx = np.argsort(conv)[-n_peaks:]
    peak_val = conv[peak_idx].max()
    mean_val = conv.mean()

    # Create a "correction" in frequency domain
    # We want to reduce the contribution at peak lags
    # The FFT of an impulse at lag k is exp(-i*2*pi*k*f/size)
    correction = np.ones(size, dtype=complex)
    for k in peak_idx:
        # Reduce amplitude at frequencies that constructively interfere at lag k
        freqs = np.arange(size)
        phase = np.exp(-1j * 2 * np.pi * k * freqs / size)
        correction -= strength * phase * (peak_val - mean_val) / (peak_val + 1e-10)

    A_new = A * correction
    a_new = np.real(ifft(A_new, size))[:n]
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def hilbert_init(n):
    """Initialize with a[k] proportional to 1/sqrt(k+1) (theoretical heuristic)."""
    k = np.arange(n) + 1.0
    a = 1.0 / np.sqrt(k)
    # Also try the mirror symmetric version
    a2 = 1.0 / np.sqrt(np.minimum(k, n - k + 1) + 0.5)
    # Blend
    a = 0.5 * a + 0.5 * a2
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s


def triangle_init(n):
    """Triangular initialization - peaks in the middle."""
    mid = n / 2.0
    a = 1.0 - np.abs(np.arange(n) - mid) / mid
    a = np.maximum(a, 0.01)
    s = float(np.sum(a))
    return a / s


def edge_heavy_init(n):
    """Initialization heavy at the edges (reduces conv at lag 0 and 2n-2)."""
    k = np.arange(n)
    center = n / 2.0
    a = np.exp(-0.5 * ((k - center) / center) ** 2)
    # Add edge boost
    edge = np.minimum(k, n - 1 - k)
    a += 0.5 * np.exp(-edge / (n * 0.1))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s


def mirror_descent_optimize(a, time_budget=300.0, verbose=True, seed=42):
    """
    Mirror descent with HIGH-to-LOW temperature schedule.
    High tau = smooth gradient (good for overall shape)
    Low tau = sharp gradient (good for fine-tuning peaks)
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    stall = 0
    tau_start = float(n)  # Start smooth
    tau_end = 1.0  # End sharp

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.1:
            break

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Temperature: geometric from tau_start to tau_end
        tau = tau_start * (tau_end / tau_start) ** progress

        conv = compute_conv(a)

        # Choose gradient type
        r = rng.random()
        if r < 0.2:
            # Top-k subgradient for diversity
            k_top = max(2, min(10, n // 50))
            g = topk_gradient(a, conv, k=k_top)
        elif r < 0.4:
            # Blended
            g_smooth = smoothed_gradient(a, conv, tau)
            k_top = max(2, min(5, n // 80))
            g_top = topk_gradient(a, conv, k=k_top)
            blend = 0.3 * (1.0 - progress)  # More top-k early
            g = blend * g_top + (1 - blend) * g_smooth
        else:
            g = smoothed_gradient(a, conv, tau)

        # Adaptive step size
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break

        # Step size: balance between progress and stability
        base_eta = 2.0 / (g_norm + 1e-10)
        # Smaller steps as we progress (finer tuning)
        eta = base_eta * (1.5 - progress)

        # Mirror descent step
        a_new = mirror_descent_step(a, g, eta)

        # Simulated annealing perturbation (occasional)
        if it % 50 == 25 and progress < 0.7:
            temp = 0.03 * (1.0 - progress)
            a_new = a_new * np.exp(temp * rng.standard_normal(n))
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            # Try smaller step
            if stall > 5:
                a_half = mirror_descent_step(a, g, eta * 0.3)
                obj_half = compute_obj(a_half)
                if obj_half < best_obj - 1e-12:
                    best_obj = obj_half
                    best_a = a_half.copy()
                    a = a_half
                    stall = 0
                elif stall > 30:
                    # Perturb to escape
                    strength = 0.05 * (1.0 + progress)
                    a = a * np.exp(strength * rng.standard_normal(n))
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                    else:
                        a = best_a.copy()
                    stall = 0

        a = a_new
        it += 1

        if verbose and (it % 200 == 0 or it == 1):
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, "
                  f"eta={eta:.5f}, stall={stall}, t={time.monotonic()-start:.0f}s")

    return best_a


def nesterov_gd_optimize(a, time_budget=100.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    base_lr = 1.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        # High tau for near-exact max
        tau = 50.0 + 200.0 * progress

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        conv_y = compute_conv(y)
        g = smoothed_gradient(y, conv_y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm + 1e-10)
            lr = 0.9 * lr + 0.1 * min(max(adapt, 0.1 * base_lr), 2.0 * base_lr)

        # Barzilai-Borwein
        if prev_g is not None and it % 10 == 5:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-8), 4.0 / np.sqrt(n))
                lr = 0.8 * lr + 0.2 * bb

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-12, lr * 0.99999)

        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.3
            best_it = it

    if verbose:
        print(f"    NesterovGD: obj={best_obj:.8f}, iters={it}")
    return best_a


def lp_driven_optimize(a, time_budget=150.0, verbose=True):
    """LP-driven optimization with sparse (top-k) constraints."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        remaining = time.monotonic() and (time_budget - (time.monotonic() - start))
        if remaining < 3.0:
            break

        # Adaptive top-k: start with fewer, increase as we converge
        progress = (time.monotonic() - start) / max(time_budget, 1e-10)
        k_constraints = max(10, int(20 + 30 * progress))

        g, M = lp_direction_sparse(a, top_k=min(k_constraints, 2 * n - 1),
                                   time_limit=3.0)

        if g is None:
            # Fallback: gradient direction
            conv = compute_conv(a)
            g = smoothed_gradient(a, conv, 100.0)
            g = np.maximum(g, 0)
            s_g = float(np.sum(g))
            if s_g < 1e-10:
                break
            g = g / s_g

        # Line search
        best_t = 0.01
        best_f = best_obj
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.15, 0.08, 0.35]:
            a_try = np.maximum((1 - t) * a + t * g, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                f_try = compute_obj(a_try / s_try)
                if f_try < best_f:
                    best_f = f_try
                    best_t = t

        it += 1
        if best_f < best_obj - 1e-12:
            a_try = np.maximum((1 - best_t) * a + best_t * g, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                best_obj = best_f
                best_a = a_try.copy()
                a = a_try
                no_improve = 0
                if verbose and (it % 3 == 0):
                    print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, "
                          f"k={k_constraints}, time={time.monotonic()-start:.0f}s")
        else:
            no_improve += 1
            if no_improve >= 6:
                # Perturb
                conv_p = compute_conv(a)
                g_p = smoothed_gradient(a, conv_p, 50.0)
                lr_p = 0.05 / (float(np.linalg.norm(g_p)) + 1e-10)
                a = np.maximum(a - lr_p * g_p + 0.005 * rng.standard_normal(n), 0)
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new
                else:
                    a = best_a.copy()
                no_improve = 0
                if it > 100:
                    if verbose:
                        print(f"    LP: stopping after {it} iterations")
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def peak_clipping_iterate(a, time_budget=60.0, verbose=False):
    """
    Iterative peak clipping: identify the worst lag, then solve a small
    QP to reduce it while maintaining constraints.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        peak_idx = np.argmax(conv)

        # For the peak lag k, conv[k] = sum_j a[j]*a[k-j]
        # We want to reduce this by adjusting a
        # Use a simple approach: reduce the elements that contribute most
        k = peak_idx
        contributions = np.zeros(n)
        for j in range(n):
            ki = k - j
            if 0 <= ki < n:
                contributions[j] = a[j] * a[ki]

        # Find top contributors
        n_reduce = max(3, n // 20)
        top_contrib = np.argsort(contributions)[-n_reduce:]

        # Reduce these elements proportionally
        reduce_amount = 0.02
        a_new = a.copy()
        for j in top_contrib:
            ki = k - j
            if 0 <= ki < n:
                # Reduce the larger of a[j] and a[ki]
                if a[j] >= a[ki]:
                    a_new[j] *= (1 - reduce_amount)
                else:
                    a_new[ki] *= (1 - reduce_amount)

        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller reduction
            for rm in [0.01, 0.005, 0.002]:
                a_try = a.copy()
                for j in top_contrib:
                    ki = k - j
                    if 0 <= ki < n:
                        if a_try[j] >= a_try[ki]:
                            a_try[j] *= (1 - rm)
                        else:
                            a_try[ki] *= (1 - rm)
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    f_try = compute_obj(a_try)
                    if f_try < best_obj - 1e-13:
                        best_obj = f_try
                        best_a = a_try.copy()
                        a = a_try
                        break

        if verbose:
            print(f"    PeakClip: obj={best_obj:.8f}, peak_lag={k}, "
                  f"time={time.monotonic()-start:.0f}s")

    return best_a


def propose_candidate():
    """
    Main search function: multi-phase optimization combining mirror descent,
    LP directions, spectral reshaping, and peak clipping.
    """
    start_time = time.monotonic()
    deadline = start_time + 1500.0 - 30.0  # Safety margin

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(2024)

    # === Initialization ===
    # Try multiple initializations and keep the best
    inits = []

    # From parent if available
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                inits.append(("parent", parent_arr / s))
    except Exception:
        pass

    n_base = 800
    inits.append(("hilbert", hilbert_init(n_base)))
    inits.append(("triangle", triangle_init(n_base)))
    inits.append(("edge_heavy", edge_heavy_init(n_base)))
    inits.append(("uniform", np.ones(n_base) / n_base))

    # Evaluate all inits
    for name, a_init in inits:
        obj = compute_obj(a_init)
        if obj < best_score:
            best_score = obj
            best_seq = a_init.copy()
        print(f"Init '{name}' (n={len(a_init)}): obj={obj:.8f}")

    print(f"\nBest init: n={len(best_seq)}, obj={best_score:.8f}")

    # === Phase 1: Mirror Descent (primary, longest budget) ===
    print(f"\n=== Phase 1: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 450.0)
    if p1_budget > 30:
        a_md = mirror_descent_optimize(best_seq, time_budget=p1_budget,
                                       verbose=True, seed=42)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj_md:.8f} (no improvement)")

    # === Phase 2: LP-Driven with sparse constraints ===
    print(f"\n=== Phase 2: LP-Driven ===")

    if p2_budget > 20:
        a_lp = lp_driven_optimize(best_seq, time_budget=p2_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2: obj={best_score:.8f} **")
        else:
            print(f"  Phase2: obj={obj_lp:.8f}")

    # === Phase 3: Peak Clipping ===
    print(f"\n=== Phase 3: Peak Clipping ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 100.0)
    if p3_budget > 10:
        a_pc = peak_clipping_iterate(best_seq, time_budget=p3_budget, verbose=False)
        obj_pc = compute_obj(a_pc)
        if obj_pc < best_score:
            best_score = obj_pc
            best_seq = a_pc.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj_pc:.8f}")

    # === Phase 4: Nesterov GD Polish ===
    print(f"\n=== Phase 4: Nesterov GD ===")

    p4_budget = min(remaining * 0.1, 100.0)
    if p4_budget > 10:
        a_gd = nesterov_gd_optimize(best_seq, time_budget=p4_budget, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd

            print(f"  Phase4: obj={best_score:.8f} **")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = []
    for delta in [-100, -50, 50, 100, 200, 300, 500]:

    for n_try in n_options:

        a_try = resample_sequence(best_seq, n_try)

        if alloc > 5:
            a_try = nesterov_gd_optimize(a_try, time_budget=alloc, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse Restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    restarts = []

    # Mirror-symmetric
    a_r = (best_seq + best_seq[::-1]) / 2.0
    a_r = np.maximum(a_r, 0)
    s_r = float(np.sum(a_r))
    if s_r > 1e-10:
        restarts.append(("sym", a_r / s_r))

    # Fourier perturbation
    pert = np.zeros(n_cur)
    for k in range(1, 8):
        freq = k * 2.0 * np.pi / n_cur
        amp = 0.05 * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n_cur) + phase)
    a_r = best_seq * np.exp(pert)
    a_r = np.maximum(a_r, 0)
    s_r = float(np.sum(a_r))
    if s_r > 1e-10:
        restarts.append(("fourier", a_r / s_r))

    # Smooth blend
    ks = max(7, n_cur // 15)
    if ks % 2 == 0:
        ks += 1
    a_smooth = uniform_filter1d(best_seq, size=ks, mode='nearest')
    a_r = 0.6 * best_seq + 0.4 * a_smooth
    a_r = np.maximum(a_r, 0)
    s_r = float(np.sum(a_r))
    if s_r > 1e-10:
        restarts.append(("smooth", a_r / s_r))

            print(f"  Restart ({name}): obj={best_score:.8f} **")
        else:
            print(f"  Restart ({name}): obj={obj_r:.8f}")

    # === Phase 8: Final Intensive Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau MD
    if remaining > 25:
        alloc = min(remaining * 0.35, 100.0)

                                          verbose=True, seed=999)

        if obj_final < best_score:

            best_seq = a_final.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.35, 80.0)
        a_final2 = lp_driven_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # GD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 50.0)

        obj_final3 = compute_obj(a_final3)

    if remaining > 10:
        alloc = min(remaining * 0.5, 40.0)
        a_fc = peak_clipping_iterate(best_seq, time_budget=alloc, verbose=False)
        obj_fc = compute_obj(a_fc)
        if obj_fc < best_score:
            best_score = obj_fc
            best_seq = a_fc.copy()
            print(f"  Final PeakClip: obj={best_score:.8f} **")

    # === Final Validation ===
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    # Ensure all values are valid
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
