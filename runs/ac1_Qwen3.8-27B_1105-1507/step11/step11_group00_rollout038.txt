def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lse_obj_and_grad(x, tau, n):
    """
    LSE-smoothed objective and gradient.
    x is the sequence (not necessarily normalized).
    Returns (objective_value, gradient).
    """
    s = float(np.sum(x))
    if s < 1e-10:
        return 1e10, np.zeros(n)
    a = x / s
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    shifted -= shifted.max()
    exp_v = np.exp(shifted)
    sum_exp = float(np.sum(exp_v))
    lse_val = max_c + np.log(sum_exp) / tau
    obj = 2.0 * n * lse_val
    
    # Probability distribution over lags
    p = exp_v / sum_exp
    
    # Gradient of conv(a,a)[k] w.r.t. a[i]:
    # = 2 * a[k-i] if 0 <= k-i < n, else 0
    # So grad[i] = 2 * sum_k p[k] * a[k-i] * 1{0<=k-i<n}
    # This is 2 * (cross-correlation of a and p) at lag i
    
    b = a[::-1]
    N_fft = 1
    while N_fft < 3 * n - 1:
        N_fft <<= 1
    c = irfft(rfft(b, N_fft) * rfft(p, N_fft), N_fft)
    grad_raw = 2.0 * c[n - 1:2 * n - 1]
    
    # Chain rule through normalization: a = x/s
    # d obj / d x[i] = d obj / d a[j] * d a[j] / d x[i]
    # d a[j] / d x[i] = (δ_{j,i} * s - x[j]) / s^2 = (δ_{j,i} - a[j]) / s
    # So grad_x[i] = sum_j grad_a[j] * (δ_{j,i} - a[j]) / s
    # = (grad_raw[i] - sum_j grad_a[j] * a[j]) / s
    # = (grad_raw[i] - dot(grad_raw, a)) / s
    
    dot_grad_a = float(np.dot(grad_raw, a))
    grad_x = (grad_raw - dot_grad_a) / s
    
    return obj, grad_x


def spectral_init(n, n_iter=200, seed=42):
    """
    Initialize via iterative projection between time and frequency domains.
    Goal: flat magnitude spectrum with non-negative time-domain values.
    """
    rng = np.random.default_rng(seed)
    a = rng.uniform(0.5, 1.5, n)
    a /= np.sum(a)
    
    for it in range(n_iter):
        # Time domain: enforce non-negativity and normalization
        # (a is already non-negative from clipping)
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
            continue
        a = a / s
        
        # Frequency domain: flatten the magnitude
        fa = rfft(a, 2 * n)
        mag = np.abs(fa)
        # Target: flat magnitude (exclude DC)
        target_mag = np.ones_like(mag)
        # Smooth transition to avoid Gibbs
        target_mag[0] = 0  # DC doesn't matter (normalization)
        # Blend with a slightly decaying target
        freqs = np.arange(len(mag))
        decay = np.exp(-freqs / (2 * n))
        target_mag = 0.7 + 0.3 * decay  # slightly decaying
        
        # Set new phase from current, magnitude from target
        phase = np.angle(fa)
        fa_new = target_mag * np.exp(1j * phase)
        a = irfft(fa_new, 2 * n)[:n]
        a = np.maximum(a, 0)
    
    a = a / np.sum(a)
    return a


def lse_lbfgs_optimize(a0, time_budget=300.0, tau_start=50.0, tau_end=2000.0, verbose=True):
    """
    Optimize using L-BFGS-B on the LSE-smoothed objective with adaptive temperature.
    """
    n = len(a0)
    a = a0.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # We'll do multiple rounds with increasing temperature
    tau = tau_start
    round_idx = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        # Create the objective function for this temperature
        current_tau = tau
        
        def obj_func(x):
            val, _ = lse_obj_and_grad(x, current_tau, n)
            return val
        
        def grad_func(x):
            _, g = lse_obj_and_grad(x, current_tau, n)
            return g
        
        # Run L-BFGS-B
        result = minimize(
            obj_func, a, jac=grad_func, method='L-BFGS-B',
            bounds=[(0, None)] * n,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-8, 'maxfun': 2000}
        )
        
        if result.success or result.fun < best_obj:
            a_new = np.maximum(result.x, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
                true_obj = compute_obj(a_new)
                if true_obj < best_obj:
                    best_obj = true_obj
                    best_a = a_new.copy()
                a = a_new
            round_idx += 1
            if verbose:
                print(f"    LSE-LBFGS round {round_idx}: tau={tau:.0f}, "
                      f"obj={best_obj:.8f}, time={time.monotonic()-start:.0f}s")
        
        # Increase temperature
        tau *= 2.0
        if tau > tau_end:
            tau = tau_end
    
    return best_a


def subgradient_optimize(a0, time_budget=200.0, verbose=True):
    """
    Direct subgradient method on the active set (max peaks).
    Uses adaptive step size and periodic smoothing.
    """
    n = len(a0)
    a = a0.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    
    it = 0
    stall = 0
    eta = 0.5  # step size
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break
        
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        
        # Determine active set: lags within tolerance of max
        tol = 1e-6 * max_c + 1e-10
        active = np.where(conv >= max_c - tol)[0]
        
        # Also include nearby peaks for robustness
        if len(active) < 3:
            idx = np.argsort(conv)[-5:]
            active = idx
        
        # Compute subgradient
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / len(active)
        
        b = a[::-1]
        N_fft = 1
        while N_fft < 3 * n - 1:
            N_fft <<= 1
        c = irfft(rfft(b, N_fft) * rfft(p, N_fft), N_fft)
        grad = 2.0 * c[n - 1:2 * n - 1]
        
        # Project onto tangent space of simplex (sum = 0)
        grad = grad - np.mean(grad)
        
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break
        
        # Adaptive step size (Polyak-like)
        obj_current = 2.0 * n * max_c  # current objective (normalized)
        lb = 1.0  # lower bound estimate (conservative)
        eta_polyak = (obj_current - lb) / (grad_norm ** 2 + 1e-10)
        eta = 0.7 * eta + 0.3 * min(eta_polyak, 1.0)
        
        # Take step
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            # Reduce step size
            eta *= 0.8
            if stall > 50:
                # Perturb
                a_new = a * np.exp(0.01 * rng.standard_normal(n))
                a_new = np.maximum(a_new, 0)
                s_new = np.sum(a_new)
                if s_new > 1e-10:
                    a = a_new / s_new
                else:
                    a = best_a.copy()
                stall = 0
                eta = 0.3
        
        a = a_new
        it += 1
        
        if verbose and (it % 100 == 0 or it == 1):
            print(f"    Subgrad it{it}: obj={best_obj:.8f}, eta={eta:.4f}, "
                  f"stall={stall}, time={time.monotonic()-start:.0f}s")
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    from scipy.interpolate import interp1d
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
    a_new = f(x_new)
    return np.maximum(a_new, 0)

F(x) = f(a) where a = x/s, s = sum(x), and f(a) = 2n * max_k conv(a,a)[k].

df/da_j = 2 * sum_k p[k] * a[k-j] (for valid k-j, i.e., 0<=k-j<n)

da_j/dx_i = (δ_{j,i} * s - x_j * 1) / s^2 = (δ_{j,i} - a_j) / s

So dF/dx_i = sum_j (df/da_j) * (da_j/dx_i)
= sum_j grad_a[j] * (δ_{j,i} - a_j) / s
= (grad_a[i] - sum_j grad_a[j] * a_j) / s
= (grad_a[i] - dot(grad_a, a)) / s

This is what I had before. Good.

But wait, there's a subtlety: since F is scale-invariant (F(cx) = F(x) for c>0), we have:
sum_i x_i * dF/dx_i = 0 (Euler's theorem for homogeneous functions)

Let me verify: sum_i x_i * (grad_a[i] - dot(grad_a, a)) / s
= (sum_i x_i * grad_a[i] - sum_i x_i * dot(grad_a, a)) / s
= (s * dot(a, grad_a) - s * dot(a, grad_a)) / s  [since x_i = s * a_i]
= 0 ✓

Good, the gradient is correct and the function is indeed scale-invariant.

Now, for L-BFGS-B, since the function is scale-invariant, the optimizer might have trouble because changing the overall scale doesn't change the objective. This creates a "valley" in the objective landscape along the radial direction. L-BFGS might waste iterations trying to figure out the scale.

To fix this, I should either:
1. Fix the scale (e.g., constrain sum(x) = 1) and optimize on the simplex
2. Or use a different parameterization

Since L-BFGS-B doesn't directly support equality constraints, option 1 would require a penalty or projection approach.

Alternative: Use the softmax parameterization. Let x = softmax(z) for z ∈ R^n. Then:
- x is automatically on the simplex (positive, sums to 1)
- The objective becomes G(z) = F(softmax(z))
- G is smooth and we can use L-BFGS without bounds

The gradient is:
dG/dz_i = sum_j (dF/dx_j) * (dx_j/dz_i)
= sum_j grad_F[j] * x_j * (δ_{j,i} - x_j)
= x_i * (grad_F[i] - sum_j grad_F[j] * x_j)

This is the "natural gradient" or "Fisher information metric" adjustment.

The issue with L-BFGS-B is that it maintains an internal model of the function, and if we normalize after each step, the function value changes in a way that confuses the optimizer. So we should either:

def lbfgs_softmax_optimize(a0, time_budget=300.0, tau_start=50.0, tau_end=5000.0, verbose=True):
    """
    L-BFGS optimization in softmax parameterization.
    a = softmax(z), optimize z.
    """
    n = len(a0)
    a0_norm = a0 / np.sum(a0)
    # Initialize z from a0 (inverse softmax)
    z = np.log(a0_norm + 1e-30)
    
    best_a = a0_norm.copy()
    best_obj = compute_obj(a0_norm)
    start = time.monotonic()
    
    tau = tau_start
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        current_tau = tau
        
        def obj_z(zz):
            # Compute a = softmax(zz)
            zz_shifted = zz - zz.max()
            a = np.exp(zz_shifted)
            s = a.sum()
            a = a / s
            # Compute LSE objective
            conv = compute_conv(a)
            max_c = np.max(conv)
            shifted = current_tau * (conv - max_c)
            shifted -= shifted.max()
            exp_v = np.exp(shifted)
            lse = max_c + np.log(exp_v.sum()) / current_tau
            return 2.0 * n * lse
        
        def grad_z(zz):
            zz_shifted = zz - zz.max()
            exp_z = np.exp(zz_shifted)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses softmax-parameterized L-BFGS with adaptive LSE temperature, active-set
subgradient with precise line search, peak-shaving, and multi-start exploration.
Key innovations: second-order optimization via L-BFGS in natural parameterization,
direct peak-attacking heuristics, and spectral initialization.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lse_obj_val(a, tau):
    """Compute LSE-smoothed objective value for normalized a (sum=1)."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    shifted -= shifted.max()
    exp_v = np.exp(shifted)
    lse = max_c + np.log(max(float(np.sum(exp_v)), 1e-300)) / tau
    return 2.0 * n * lse


def lse_prob(conv, tau):
    """Compute probability distribution p over lags from LSE."""
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    shifted -= shifted.max()
    exp_v = np.exp(shifted)
    s = float(np.sum(exp_v))
    if s < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
        return p
    return exp_v / s


def gradient_from_p(a, p):
    """Compute 2 * cross-correlation of a and p, giving gradient w.r.t. a."""
    n = len(a)
    b = a[::-1]
    N_fft = 1
    while N_fft < 3 * n - 1:
        N_fft <<= 1
    c = irfft(rfft(b, N_fft) * rfft(p, N_fft), N_fft)
    return 2.0 * c[n - 1:2 * n - 1]


def lbfgs_softmax_optimize(a0, time_budget=200.0, tau_start=50.0, tau_end=5000.0,
                           n_rounds=8, verbose=True):
    """
    L-BFGS optimization using softmax parameterization with adaptive LSE temperature.
    a = softmax(z), optimizing z directly avoids bound constraints.
    """
    n = len(a0)
    a0_nm = a0 / np.sum(a0)
    z = np.log(a0_nm + 1e-30)

    best_a = a0_nm.copy()
    best_obj = compute_obj(a0_nm)
    start = time.monotonic()
    tau = tau_start

    for rnd in range(n_rounds):
        elapsed = time.monotonic() - start
        if elapsed >= time_budget:
            break
        remaining = time_budget - elapsed
        if remaining < 8:
            break

        current_tau = min(tau, tau_end)

        def obj_z(zz):
            zz_s = zz - zz.max()
            a_tmp = np.exp(zz_s)
            s_tmp = a_tmp.sum()
            if s_tmp < 1e-300:
                return 1e10
            a_tmp = a_tmp / s_tmp
            return lse_obj_val(a_tmp, current_tau)

        def grad_z(zz):
            zz_s = zz - zz.max()
            exp_z = np.exp(zz_s)
            a_tmp = exp_z / exp_z.sum()

            conv = compute_conv(a_tmp)
            p = lse_prob(conv, current_tau)
            grad_a = gradient_from_p(a_tmp, p)

            # Natural gradient: d/dz [f(softmax(z))] = a * (grad_a - <grad_a, a>)
            grad_z_val = a_tmp * (grad_a - float(np.dot(grad_a, a_tmp)))
            return 2.0 * n * grad_z_val

        # Adaptive maxiter based on remaining time
        maxiter = min(400, max(100, int(remaining * 15)))
        maxfun = min(800, max(200, int(remaining * 30)))

        result = minimize(
            obj_z, z, jac=grad_z, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-9, 'maxfun': maxfun}
        )

        if result.fun < 1e9:
            zz_new = result.x
            zz_s = zz_new - zz_new.max()
            a_new = np.exp(zz_s)
            a_new = a_new / a_new.sum()
            true_obj = compute_obj(a_new)
            if true_obj < best_obj:
                best_obj = true_obj
                best_a = a_new.copy()
            z = zz_new

        if verbose:
            print(f"    LBFGS-softmax r{rnd}: tau={current_tau:.0f}, "
                  f"obj={best_obj:.8f}, time={time.monotonic()-start:.0f}s")

        tau *= 1.6

    return best_a


def subgradient_line_search(a0, time_budget=200.0, verbose=True, seed=42):
    """
    Active-set subgradient method with precise golden-section line search.
    """
    n = len(a0)
    a = a0.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    stall = 0
    base_eta = 0.4

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        conv = compute_conv(a)
        max_c = float(np.max(conv))

        # Determine active set with adaptive tolerance
        tol_rel = max(1e-7, 1e-5 / (it + 10))
        tol = tol_rel * max_c
        active = np.where(conv >= max_c - tol)[0]
        if len(active) < 2:
            active = np.argsort(conv)[-4:]

        # Gradient from active set
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / len(active)
        grad = gradient_from_p(a, p)

        # Project to tangent space of simplex
        grad = grad - np.mean(grad)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-14:
            break

        # Direction: -grad (we want to decrease the objective)
        d = -grad / grad_norm

        # Golden section line search
        a_base = a.copy()
        s_base = 1.0
        obj_base = 2.0 * n * max_c

        # Find bracket
        t_hi = 1.0
        f_hi = 1e10
        for _ in range(10):
            a_try = np.maximum(a_base + t_hi * d * grad_norm * 0.1, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                f_hi = compute_obj(a_try / s_try)
            else:
                f_hi = 1e10
            if f_hi < obj_base:
                break
            t_hi *= 0.5
        if f_hi >= obj_base:
            t_hi = 1e-4
            a_try = np.maximum(a_base + t_hi * d * grad_norm * 0.1, 0)
            s_try = float(np.sum(a_try))
            f_hi = compute_obj(a_try / s_try) if s_try > 1e-10 else 1e10

        if f_hi >= obj_base:
            stall += 1
            if stall > 30:
                # Perturb to escape
                a = a * np.exp(0.005 * rng.standard_normal(n))
                a = np.maximum(a, 0)
                s_new = np.sum(a)
                a = a / s_new if s_new > 1e-10 else best_a.copy()
                stall = 0
            it += 1
            continue

        # Golden section in [0, t_hi]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        t_lo = 0.0
        t_mid_hi = t_hi * (1.0 - gr)
        t_mid_lo = t_hi * gr
        f_mid_hi = 1e10
        f_mid_lo = 1e10

        for _ in range(40):
            for t_mid, f_mid_holder in [(t_mid_hi, None), (t_mid_lo, None)]:
                pass
            # Evaluate at both points
            a_mh = np.maximum(a_base + t_mid_hi * d * grad_norm * 0.1, 0)
            s_mh = float(np.sum(a_mh))
            f_mid_hi = compute_obj(a_mh / s_mh) if s_mh > 1e-10 else 1e10

            a_ml = np.maximum(a_base + t_mid_lo * d * grad_norm * 0.1, 0)
            s_ml = float(np.sum(a_ml))
            f_mid_lo = compute_obj(a_ml / s_ml) if s_ml > 1e-10 else 1e10

            if f_mid_hi < f_mid_lo:
                t_hi = t_mid_lo
                t_mid_lo = t_mid_hi
                f_mid_lo = f_mid_hi
                t_mid_hi = (1.0 - gr) * t_lo + gr * t_hi
            else:
                t_lo = t_mid_hi
                t_mid_hi = t_mid_lo
                f_mid_hi = f_mid_lo
                t_mid_lo = gr * t_lo + (1.0 - gr) * t_hi

            if abs(t_hi - t_lo) < 1e-8 * t_hi + 1e-10:
                break

        t_best = (t_lo + t_hi) / 2.0
        a_new = np.maximum(a_base + t_best * d * grad_norm * 0.1, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            a = a_new
        else:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                stall = 0
            a = a_new

        it += 1
        if verbose and (it % 50 == 0 or it == 1):
            print(f"    SubgradLS it{it}: obj={best_obj:.8f}, stall={stall}, "
                  f"time={time.monotonic()-start:.0f}s")

    return best_a


def peak_shave(a0, time_budget=100.0, verbose=False):
    """
    Directly attack the peak of the autocorrelation by reducing contributing terms.
    """
    n = len(a0)
    a = a0.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    it = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1:
            break

        conv = compute_conv(a)
        max_idx = int(np.argmax(conv))
        max_val = float(conv[max_idx])

        # Find which pairs contribute to this peak
        # conv[max_idx] = sum_j a[j]*a[max_idx-j] for valid j
        contributing = []
        for j in range(n):
            other = max_idx - j
            if 0 <= other < n and other != j:
                contributing.append((j, other, a[j] * a[other]))
            elif other == j and 0 <= other < n:
                contributing.append((j, j, a[j] * a[j]))

        # Sort by contribution (largest first)
        contributing.sort(key=lambda x: -x[2])

        # Reduce the top contributors and redistribute
        top_k = min(10, len(contributing))
        if top_k == 0:
            break

        total_contrib = sum(c[2] for c in contributing)
        if total_contrib < 1e-15:
            break

        # Identify the indices to reduce
        reduce_indices = set()
        for j, other, _ in contributing[:top_k]:
            reduce_indices.add(j)
            if other != j:
                reduce_indices.add(other)

        reduce_indices = list(reduce_indices)
        if not reduce_indices:
            break

        # Reduce these and add to others
        reduce_amount = 0.05 * max(a[i] for i in reduce_indices)
        total_reduce = 0.0
        for i in reduce_indices:
            actual_reduce = min(reduce_amount, a[i] * 0.5)
            a[i] -= actual_reduce
            total_reduce += actual_reduce

        # Redistribute to random other indices
        other_indices = [i for i in range(n) if i not in reduce_indices]
        if other_indices:
            # Add to indices that are "opposite" in convolution sense
            n_redistribute = min(5, len(other_indices))
            for _ in range(n_redistribute):
                idx = int(np.random.choice(other_indices))
                a[idx] += total_reduce / n_redistribute

        a = np.maximum(a, 0)
        s_new = float(np.sum(a))
        if s_new > 1e-10:
            a = a / s_new

        obj_new = compute_obj(a)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a.copy()
        else:
            # Revert
            a = best_a.copy()

        it += 1
        if verbose and it % 20 == 0:
            print(f"    PeakShave it{it}: obj={best_obj:.8f}, time={time.monotonic()-start:.0f}s")

    return best_a


def spectral_init(n, n_iter=300, seed=42):
    """
    Initialize via iterative spectral shaping (Gerchberg-Saxton style).
    Aims for flat magnitude spectrum with non-negative time-domain values.
    """
    rng = np.random.default_rng(seed)
    a = rng.uniform(0.3, 1.0, n)
    a /= np.sum(a)

    for it in range(n_iter):
        # Enforce non-negativity (time domain constraint)
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
            continue
        a = a / s

        # Shape frequency domain
        fa = rfft(a, 2 * n)
        mag = np.abs(fa)
        phase = np.angle(fa)

        # Target: flat with slight rolloff
        freqs = np.arange(len(mag))
        target = 1.0 / (1.0 + freqs.astype(float) / (n * 0.5))
        # Blend: 70% target, 30% current (for stability)
        new_mag = 0.7 * target + 0.3 * mag
        new_mag[0] = 0.0  # Remove DC (normalization handles it)

        fa_new = new_mag * np.exp(1j * phase)
        a = irfft(fa_new, 2 * n)[:n]

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    from scipy.interpolate import CubicSpline
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    try:
        cs = CubicSpline(x_old, a, bc_type='natural')
        a_new = cs(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def mirror_descent_refine(a0, time_budget=100.0, tau=200.0, verbose=False):
    """
    Quick mirror descent (multiplicative weights) refinement.
    """
    n = len(a0)
    a = a0.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99)

    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 1:
            break

        conv = compute_conv(a)
        max_c = float(np.max(conv))

        # LSE gradient
        p = lse_prob(conv, tau)
        grad = gradient_from_p(a, p)
        grad = grad - np.mean(grad)
        g_max = float(np.max(np.abs(grad)))
        if g_max < 1e-15:
            break

        eta = 1.5 / (g_max + 1e-10)
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                a = best_a * np.exp(0.01 * rng.standard_normal(n))
                a = np.maximum(a, 0)
                s_p = np.sum(a)
                a = a / s_p if s_p > 1e-10 else best_a.copy()
                stall = 0

        a = a_new
        it += 1
        if verbose and it % 100 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, time={time.monotonic()-start:.0f}s")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining L-BFGS (softmax),
    subgradient with line search, peak shaving, spectral init, and multi-n.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Initialize from parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Try spectral initialization ===
    print(f"\n=== Phase 1: Spectral Init (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_spec = spectral_init(n_current, n_iter=400, seed=42)
        obj_spec = compute_obj(a_spec)
        print(f"  Spectral init: obj={obj_spec:.8f}")

        # Optimize spectral init
        alloc = min(remaining * 0.2, 80.0)
        if alloc > 20:
            a_spec_opt = lbfgs_softmax_optimize(a_spec, time_budget=alloc,
                                                 tau_start=50.0, tau_end=3000.0,
                                                 n_rounds=5, verbose=False)
            obj_spec_opt = compute_obj(a_spec_opt)
            print(f"  Spectral + LBFGS: obj={obj_spec_opt:.8f}")

            if obj_spec_opt < best_score:
                best_score = obj_spec_opt
                best_seq = a_spec_opt.copy()
                print(f"  ** New best from spectral: {best_score:.8f} **")

    # === Phase 2: L-BFGS (softmax) on best ===
    print(f"\n=== Phase 2: L-BFGS Softmax (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.3, 350.0)
    if p2_budget > 40:
        a_lbfgs = lbfgs_softmax_optimize(best_seq, time_budget=p2_budget,
                                          tau_start=30.0, tau_end=5000.0,
                                          n_rounds=12, verbose=True)
        obj_lbfgs = compute_obj(a_lbfgs)
        if obj_lbfgs < best_score:
            best_score = obj_lbfgs
            best_seq = a_lbfgs.copy()
            print(f"  ** Phase2 LBFGS: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LBFGS: obj={obj_lbfgs:.8f} (no improvement)")

    # === Phase 3: Subgradient with line search ===
    print(f"\n=== Phase 3: Subgradient + Line Search ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.25, 250.0)
    if p3_budget > 30:
        a_sub = subgradient_line_search(best_seq, time_budget=p3_budget,
                                         verbose=True, seed=123)
        obj_sub = compute_obj(a_sub)
        if obj_sub < best_score:
            best_score = obj_sub
            best_seq = a_sub.copy()
            print(f"  ** Phase3 Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 Subgrad: obj={obj_sub:.8f} (no improvement)")

    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.1, 80.0)
    if p4_budget > 10:
        a_peak = peak_shave(best_seq, time_budget=p4_budget, verbose=False)
        obj_peak = compute_obj(a_peak)
        if obj_peak < best_score:
            best_score = obj_peak
            best_seq = a_peak.copy()
            print(f"  ** Phase4 PeakShave: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 PeakShave: obj={obj_peak:.8f} (no improvement)")

    # === Phase 5: Mirror descent refinement ===
    print(f"\n=== Phase 5: MD Refinement ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.15, 120.0)
    if p5_budget > 20:
        a_md = mirror_descent_refine(best_seq, time_budget=p5_budget,
                                      tau=300.0, verbose=False)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  ** Phase5 MD: obj={best_score:.8f} **")
        else:
            print(f"  Phase5 MD: obj={obj_md:.8f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = []
    for delta in [-100, -50, 50, 100, 200, 300, 400]:
        n_try = n_best + delta
        if 500 <= n_try <= 2200 and n_try not in n_options:
            n_options.append(n_try)
    n_options = sorted(set(n_options))[:4]
    print(f"  Trying n: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)

        # LBFGS
        alloc1 = min(remaining * 0.35, 80.0)
        if alloc1 > 20:
            a_try = lbfgs_softmax_optimize(a_try, time_budget=alloc1,
                                            tau_start=100.0, tau_end=3000.0,
                                            n_rounds=5, verbose=False)

        # Subgradient
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.2, 40.0)
        if alloc2 > 15:
            a_try = subgradient_line_search(a_try, time_budget=alloc2, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    restart_strategies = [
        "palindrome", "fourier_perturb", "smoothing", "random_blend"
    ]

    for r, name in enumerate(restart_strategies):
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break

        if name == "palindrome":
            a_r = 0.5 * (best_seq + best_seq[::-1])
        elif name == "fourier_perturb":
            pert = np.zeros(n_cur)
            for k in range(1, 10):
                freq = k * 2.0 * np.pi / n_cur
                amp = 0.025 * rng.uniform(0.3, 1.5) / k
                phase = rng.uniform(0, 2.0 * np.pi)
                pert += amp * np.cos(freq * np.arange(n_cur) + phase)
            a_r = best_seq * np.exp(pert)
        elif name == "smoothing":
            from scipy.ndimage import uniform_filter1d
            ks = max(3, (n_cur // 8) | 1)
            a_smooth = uniform_filter1d(best_seq, size=ks, mode='nearest')
            a_r = 0.75 * best_seq + 0.25 * a_smooth
        else:
            noise = rng.uniform(0, 0.5, n_cur) / n_cur
            a_r = 0.75 * best_seq + 0.25 * noise

        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r < 1e-10:
            a_r = best_seq.copy()
        else:
            a_r = a_r / s_r

        # Optimize
        alloc_md = min(remaining * 0.35, 60.0)
        if alloc_md > 15:
            a_r = lbfgs_softmax_optimize(a_r, time_budget=alloc_md,
                                          tau_start=100.0, tau_end=2000.0,
                                          n_rounds=4, verbose=False)

        remaining = deadline - time.monotonic()
        alloc_sub = min(remaining * 0.2, 30.0)
        if alloc_sub > 10:
            a_r = subgradient_line_search(a_r, time_budget=alloc_sub, verbose=False)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart ({name}): obj={best_score:.8f} **")
        else:
            print(f"  Restart ({name}): obj={obj_r:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        alloc = min(remaining * 0.35, 100.0)
        a_final = lbfgs_softmax_optimize(best_seq, time_budget=alloc,
                                          tau_start=500.0, tau_end=10000.0,
                                          n_rounds=6, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LBFGS: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.35, 80.0)
        a_final2 = subgradient_line_search(best_seq, time_budget=alloc,
                                            verbose=True, seed=777)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final Subgrad: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 60.0)
        a_final3 = mirror_descent_refine(best_seq, time_budget=alloc,
                                         tau=500.0, verbose=False)
        obj_final3 = compute_obj(a_final3)
        if obj_final3 < best_score:
            best_score = obj_final3
            best_seq = a_final3.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
