import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def compute_obj_normalized(a):
    """Assumes a is normalized to sum=1."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-15:
                continue

            delta = x_opt - v
            a[i] = x_opt
            # Update conv
            if idx_quad >= 0:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            S += delta

            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    improved += 1

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, improved={improved}, t={elapsed:.1f}s")

        # Early exit if no improvement in a full sweep
        if improved == 0 and sweep > 5:
            # Try a few more sweeps with different ordering
            pass

    return best_a


def subgradient_deflation(a, time_budget=60.0, n_iter=None, verbose=False, seed=42):
    """
    Subgradient-based peak deflation.
    For the active lag k*, the subgradient is 2*a[k*-i].
    We do multiplicative updates: a[i] *= exp(-eta * a[k*-i]) to reduce the peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    if n_iter is None:
        n_iter = int(time_budget * 100)  # estimate iterations

    for it in range(n_iter):
        if time.monotonic() - start >= time_budget:
            break

        conv = compute_conv(a)
        max_b = float(np.max(conv))
        obj = 2.0 * n * max_b

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Find active lags (within 5% of max)
        threshold = 0.95 * max_b
        active_ks = np.where(conv >= threshold)[0]

        # Compute subgradient: sum over active lags of 2*a[k-i]
        # For efficiency, only use top few active lags
        top_ks = active_ks[np.argsort(conv[active_ks])[::-1][:min(5, len(active_ks))]]

        # Compute the "deflation" direction
        # For lag k, the gradient wrt a[i] is 2*a[k-i] (if k-i in [0,n-1])
        # We want to reduce a[i] where a[k-i] is large (for active k)
        grad = np.zeros(n)
        for k in top_ks:
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    grad[i] += a[ki]

        # Multiplicative update with adaptive step size
        eta = 0.5 / (float(np.max(grad)) + 1e-10)
        eta = min(eta, 0.3 / n)  # cap step size

        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(np.sum(a_new))

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

        if verbose and it % 500 == 0:
            print(f"    SG it {it}: obj={best_obj:.10f}")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_active(a, n_active=None, time_budget=30.0, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Uses only the most active constraints for speed.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A_full = build_conv_matrix(a_norm)

    # Determine which constraints to use
    conv_current = compute_conv(a_norm)
    max_b = float(np.max(conv_current))
    if n_active is not None:
        # Use only the top n_active constraints
        top_indices = np.argsort(conv_current)[::-1][:n_active]
        A = A_full[top_indices]
    else:
        A = A_full

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    start = time.monotonic()
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS]
    for solver in solvers:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish_fast(a, time_budget=60.0, verbose=False, seed=42):
    """
    LP-based polishing with active set and multiple blending factors.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        # Use active set for speed
        conv = compute_conv(a)
        max_b = float(np.max(conv))
        threshold = 0.9 * max_b
        n_active = int(np.sum(conv >= threshold))
        n_active = max(n_active, 50)
        n_active = min(n_active, 2 * n - 1)

        alloc = min(remaining * 0.4, 25.0)
        g_lp, M_lp = solve_lp_active(a, n_active=n_active,
                                      time_budget=alloc, verbose=verbose)
        if g_lp is None:
            break

        # Try multiple blending factors
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 3:
                # Perturb to escape
                pert = rng.normal(0, 0.02 * stall, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / float(a.sum())
                stall = 0

        if verbose:
            print(f"    LP: obj={best_obj:.10f}, stall={stall}")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Asymmetric: more mass on one side
        x = np.arange(n)
        a = np.exp(-x / (n * rng.uniform(0.3, 0.8)))
        a = a / a.sum()
    elif mode == 3:
        # Asymmetric: more mass on the other side
        x = np.arange(n)
        a = np.exp(-x / (n * rng.uniform(0.3, 0.8)))
        a = a[::-1] / a.sum()
    elif mode == 4:
        # Two bumps
        c1, c2 = n * rng.uniform(0.2, 0.4), n * rng.uniform(0.6, 0.8)
        w = n * rng.uniform(0.05, 0.15)
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * (w * 1.5) ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Sine-modulated
        x = np.linspace(0, 2 * np.pi, n)
        a = 1.0 + 0.5 * np.sin(x * rng.integers(1, 5))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Power law (asymmetric)
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(1, 5)
        a = a / a.sum()
    elif mode == 7:
        # Inverse power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-5, -1)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def iterative_refinement(a, time_budget=100.0, seed=42, verbose=False):
    """
    Alternating CD and LP with adaptive time allocation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    cycle = 0
    while time.monotonic() - start < time_budget:
        remaining = time.monotonic() and (time_budget - (time.monotonic() - start))
        if remaining < 20:
            break

        cycle += 1
        # Alternate: CD (60%) then LP (30%) then SG (10%)
        if cycle % 3 == 1:
            alloc = min(remaining * 0.4, 80.0)
            if alloc > 15:
                a_new = coordinate_descent_precise(a, time_budget=alloc,
                                                    verbose=False, seed=seed + cycle * 100)
                obj_new = 2.0 * len(a_new) * float(np.max(compute_conv(a_new)))
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
        elif cycle % 3 == 2:
            alloc = min(remaining * 0.3, 40.0)
            if alloc > 10:
                a_new = lp_polish_fast(a, time_budget=alloc,
                                        verbose=False, seed=seed + cycle * 200)
                obj_new = 2.0 * len(a_new) * float(np.max(compute_conv(a_new)))
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
        else:
            alloc = min(remaining * 0.15, 30.0)
            if alloc > 5:
                a_new = subgradient_deflation(a, time_budget=alloc,
                                               verbose=False, seed=seed + cycle * 300)
                obj_new = 2.0 * len(a_new) * float(np.max(compute_conv(a_new)))
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                # Follow up with short CD
                remaining2 = time_budget - (time.monotonic() - start)
                if remaining2 > 15:
                    a_new = coordinate_descent_precise(a, time_budget=min(remaining2 * 0.3, 25.0),
                                                        verbose=False, seed=seed + cycle * 400)
                    obj_new = 2.0 * len(a_new) * float(np.max(compute_conv(a_new)))
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: iterative refinement combining CD, LP, subgradient,
    multi-n exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj_normalized(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj_normalized(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.10f}")

    n_current = len(best_seq)

    # === Phase 1: Long CD (main workhorse) ===
    print(f"\n=== Phase 1: Primary CD ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 500.0)
    if p1_budget > 30:
        a_cd = coordinate_descent_precise(best_seq, time_budget=p1_budget,
                                           verbose=True, seed=seed)
        obj_cd = compute_obj_normalized(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.10f} **")
        else:
            print(f"  CD: obj={obj_cd:.10f} (no improvement)")

    # === Phase 2: Iterative refinement (CD + LP + SG) ===
    print(f"\n=== Phase 2: Iterative Refinement ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 350.0)
    if p2_budget > 30:
        a_ref = iterative_refinement(best_seq, time_budget=p2_budget,
                                      seed=seed + 1000, verbose=True)
        obj_ref = compute_obj_normalized(a_ref)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"  Refinement: obj={best_score:.10f} **")
        else:
            print(f"  Refinement: obj={obj_ref:.10f}")

    # === Phase 3: Dedicated LP polish ===
    print(f"\n=== Phase 3: LP Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 200.0)
    if p3_budget > 20:
        for rnd in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(p3_budget / (4 - rnd), remaining * 0.4)
            if alloc < 12:
                break
            a_lp = lp_polish_fast(best_seq, time_budget=alloc,
                                   verbose=False, seed=2000 + rnd * 137)
            obj_lp = compute_obj_normalized(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.10f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.10f}")
            # Short CD after each LP
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_cd = coordinate_descent_precise(best_seq, time_budget=min(remaining * 0.15, 30.0),
                                                   verbose=False, seed=3000 + rnd * 71)
                obj_cd = compute_obj_normalized(a_cd)
                if obj_cd < best_score:
                    best_score = obj_cd
                    best_seq = a_cd.copy()

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        n_best - 20, n_best + 20,
        n_best - 50, n_best + 50,
        n_best + 100, n_best - 100,
        750, 850, 900, 950, 1000, 1100,
        600, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500 and abs(n - n_best) > 10]
    n_options = n_options[:5]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        alloc_cd = min(remaining * 0.25, 40.0)

                                                verbose=False, seed=n_try)
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.10f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()

        mode = 2 + r  # Use asymmetric modes
        a_r = diverse_init(n_current, rng, mode=mode)
        # Blend with best
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / float(a_r.sum())

        alloc = min(remaining * 0.2, 50.0)
        if alloc > 20:

                                              verbose=False, seed=5000 + r * 111)
            obj_r = compute_obj_normalized(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {2+r}): obj={best_score:.10f} **")
                # Follow up
                remaining = deadline - time.monotonic()
                if remaining > 30:
                    a_follow = iterative_refinement(best_seq, time_budget=min(remaining * 0.3, 50.0),
                                                    seed=6000 + r * 222)
                    obj_follow = compute_obj_normalized(a_follow)
                    if obj_follow < best_score:
                        best_score = obj_follow
                        best_seq = a_follow.copy()
            else:
                print(f"  Restart {r} (mode {2+r}): obj={obj_r:.10f}")

    # === Phase 6: Final long CD ===
    print(f"\n=== Phase 6: Final CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 200.0)
        a_final = coordinate_descent_precise(best_seq, time_budget=alloc,
                                              verbose=True, seed=seed + 99999)
        obj_final = compute_obj_normalized(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.10f} **")
        else:
            print(f"  Final CD: obj={obj_final:.10f}")

    # === Phase 7: Final LP + CD ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc_lp = min(remaining * 0.4, 60.0)
        a_lp_final = lp_polish_fast(best_seq, time_budget=alloc_lp,
                                     verbose=False, seed=77777)
        obj_lp_final = compute_obj_normalized(a_lp_final)
        if obj_lp_final < best_score:
            best_score = obj_lp_final
            best_seq = a_lp_final.copy()
            print(f"  Final LP: obj={best_score:.10f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_cd_final = coordinate_descent_precise(best_seq, time_budget=min(remaining * 0.6, 50.0),
                                                 verbose=False, seed=88888)
        obj_cd_final = compute_obj_normalized(a_cd_final)
        if obj_cd_final < best_score:
            best_score = obj_cd_final
            best_seq = a_cd_final.copy()
            print(f"  Final CD2: obj={best_score:.10f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_normalized(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
