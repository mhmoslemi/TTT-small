    """Project v onto the probability simplex."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    i = np.argmax(cond)
    theta = cssv[i] / (i + 1)
    return np.maximum(v - theta, 0)

def solve_exact_qp(a_init, n, tol=1e-8, max_iter=50):
    """Solve min t s.t. conv[k] <= t for all k, sum a = 1, a >= 0.
    Using active-set / cutting-plane method."""
    a = a_init.copy()
    a = a / a.sum()
    
    active_set = []
    
    for iteration in range(max_iter):
        # Find active set
        conv = compute_conv(a)
        conv_max = float(np.max(conv))
        # Active: indices within tol of max
        active_set = list(np.where(conv >= conv_max - tol)[0])
        
        # Solve QCQP with active set
        result = solve_qp(active_set, n)
        if result is None:
            break
        a_new, t_val = result
        
        # Check if all constraints are satisfied
        conv_new = compute_conv(a_new)
        if float(np.max(conv_new)) <= t_val + 1e-10:
            # Success!
            return a_new
        else:
            # Add violating constraints
            a = a_new
    
    return a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS on softmax-parameterized space with LSE-smoothed objective as primary
method, combined with exact active-set QCQP polishing and mirror descent.
Key improvements: second-order L-BFGS optimization, exact QCQP for final precision,
adaptive temperature continuation, and systematic multi-n exploration.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def gradient_from_p(a, p):
    n = len(a)
    b = a[::-1]
    N_fft = next_pow2(3 * n - 1)
    c = irfft(rfft(b, N_fft) * rfft(p, N_fft), N_fft)
    return 2.0 * c[n - 1:2 * n - 1]


def lse_weight(conv, tau):
    shifted = tau * (conv - conv.max())
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    s = float(np.sum(exp_vals))
    if s < 1e-300:
        p = np.zeros_like(conv)
        p[int(np.argmax(conv))] = 1.0
        return p
    return exp_vals / s


def softmax_from_w(w):
    w_shifted = w - w.max()
    ew = np.exp(w_shifted)
    s = float(np.sum(ew))
    if s < 1e-300:
        return np.ones_like(w) / len(w)
    return ew / s


def obj_soft
