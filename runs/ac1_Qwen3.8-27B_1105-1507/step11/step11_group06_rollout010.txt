import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):

    s = float(np.sum(a))
    if s < 0.01:

        eps = 0.01 * max_val
        active = np.where(conv >= max_val - eps)[0]
    else:

                grad[i] += a[j]

"""
Hybrid optimization: Nesterov mirror descent + LP polish + coordinate descent.
Uses multiplicative weights for non-negativity, Nesterov acceleration for speed,
LP for strong descent directions, and coordinate descent for final polish.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def subgradient_conv(a, active_lags=None, tol_frac=0.05):
    """Compute subgradient of max_k (a*a)_k w.r.t. a.
    Uses active lags within tol_frac of the max."""
    n = len(a)
    conv = compute_conv(a)
    max_val = np.max(conv)
    if active_lags is None:
        active = np.where(conv >= max_val * (1 - tol_frac))[0]
    else:
        active = active_lags
    if len(active) == 0:
        active = np.array([int(np.argmax(conv))])
    
    grad = np.zeros(n)
    for k in active:
        # g_i = 2 * a_{k-i} if 0 <= k-i < n, else 0
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # For i in [lo, hi], k-i is in [k-hi, k-lo] = [max(0,k-n+1), min(n-1,k)]... 
        # Actually, for i in [lo, hi], k-i ranges from k-lo to k-hi
        # k-lo = k - max(0, k-n+1)
        # k-hi = k - min(n-1, k)
        # We need 0 <= k-i < n, i.e., k-n+1 <= i <= k
        # So i ranges from max(0, k-n+1) to min(n-1, k)
        valid_i = np.arange(lo, hi + 1)
        valid_k_minus_i = k - valid_i
        grad[valid_i] += 2.0 * a[valid_k_minus_i]
    
    grad /= len(active)
    return grad, active[:20].tolist(), max_val


def mirror_descent_nesterov(a, time_budget=60.0, eta0=0.1, tol_frac=0.05, verbose=False):
    """Nesterov-accelerated mirror descent with multiplicative weights."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 0.01:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    start = time.monotonic()
    a_prev = a.copy()
    it = 0
    
    while time.monotonic() - start < time_budget:
        # Nesterov look-ahead
        momentum = it / (it + 2.0)
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 0.01:
            y = a.copy()
        else:
            y = y / s_y
        
        # Compute subgradient at y
        grad, active, max_val = subgradient_conv(y, tol_frac=tol_frac)
        
        # Adaptive step size
        eta = eta0 / (1.0 + 0.01 * it)
        
        # Multiplicative update
        a_new = y * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 0.01:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        a_prev = a
        a = a_new
        
        # Averaging (Hedge-style)
        # Actually, for Nesterov, we just use a directly
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        it += 1
        if verbose and it % 200 == 0:
            print(f"    MD it={it}: obj={best_obj:.8f}, eta={eta:.6f}")
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A such that A @ g = a * g (linear convolution)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, time_budget=30.0):
    """Solve LP: min M s.t. (a_norm * g)_k <= M for all k, sum(g)=1, g >= 0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    M = cp.Variable()
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, timeout=time_budget)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, direction, n_steps=50, t_max=1.5):
    """Exhaustive line search along direction from a."""
    best_t = 0.0
    best_obj = compute_obj(a)
    ts = np.linspace(0.0, t_max, n_steps)
    for t in ts:
        a_try = (1.0 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    return best_t, best_obj


def lp_polish(a, time_budget=100.0, max_iters=100, verbose=False):
    """LP-based polishing with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 0.01:
        return a, compute_obj(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        g_lp, M_lp = solve_lp(a, time_budget=min(remaining * 0.5, 20.0))
        if g_lp is None:
            break
        
        t_opt, obj_opt = line_search(a, g_lp, n_steps=40, t_max=1.5)
        if t_opt > 1e-6 and obj_opt < best_obj - 1e-12:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                best_obj = compute_obj(a_new)
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            # Try direct jump
            obj_direct = compute_obj(g_lp)
            if obj_direct < best_obj:
                best_obj = obj_direct
                best_a = g_lp.copy()
                a = g_lp.copy()
                stall = 0
            else:
                stall += 1
        
        it += 1
        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, stall={stall}")
        
        if stall >= 3:
            break
    
    return best_a, best_obj
