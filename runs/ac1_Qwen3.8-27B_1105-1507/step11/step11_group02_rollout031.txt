def obj_norm(a):
    """Compute 2n*max(conv) assuming sum(a)=1."""
    n = len(a)
    conv = compute_conv(a)
    return float(2.0 * n * np.max(conv))


def peak_responsibility(a, k_star):
    """Compute gradient of c[k_star] w.r.t. a (the 'responsibility vector')."""
    n = len(a)
    r = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            r[i] += a[j]
            if j != i:
                r[i] += a[j]
    return r


def fast_cd(a, time_budget, seed=42, verbose=False):
    """
    Fast coordinate descent with incremental conv updates and
    precomputed unaffected maximum for accurate 1D subproblems.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0
    no_improve_sweeps = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        obj_before = best_obj
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = i
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            other_idx = ks - i
            other_vals = a[other_idx]
            c_vals = conv[ks].copy()

            k_quad = 2 * i
            is_quad = (ks == k_quad)

            # Precompute max of unaffected conv values
            max_unaffected = float(np.max(conv))
            # If any affected value was the global max, we need the second max
            affected_set_max = float(np.max(c_vals))
            if affected_set_max >= max_unaffected - 1e-15:
                # Need to find max excluding affected positions
                conv_masked = conv.copy()
                conv_masked[ks] = 0.0
                max_unaffected = float(np.max(conv_masked)) if np.any(conv_masked > 0) else 0.0

            S_oth = S_other

            def f1d(x, v=v, c_vals=c_vals, other_vals=other_vals,
                    is_quad=is_quad, max_unaff=max_unaffected, S_oth=S_oth):
                delta = x - v
                max_aff = 0.0
                if np.any(~is_quad):
                    lin = c_vals[~is_quad] + 2.0 * delta * other_vals[~is_quad]
                    max_aff = float(np.max(lin))
                if np.any(is_quad):
                    qv = float(c_vals[np.where(is_quad)[0][0]] + x * x - v * v)
                    max_aff = max(max_aff, qv)
                gm = max(max_unaff, max_aff)
                denom = (S_oth + x) * (S_oth + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * gm / denom

            # Golden section search
            upper = max(2.0 / max(S_oth, 1e-10), 3.0 * max(v, 1.0 / n))
            upper = min(upper, 5.0 / n + 0.1)
            x_lo, x_hi = 0.0, upper

            if x_hi - x_lo < 1e-15:
                continue

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f1d(c_pt)
            fd = f1d(d_pt)

            for _ in range(50):
                if x_hi - x_lo < 1e-14 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt
            S += delta

            # Incremental conv update
            if np.any(~is_quad):
                conv[ks[~is_quad]] += 2.0 * delta * other_vals[~is_quad]
            if np.any(is_quad):
                conv[ks[int(np.where(is_quad)[0][0])]] += x_opt * x_opt - v * v

            obj = 2.0 * n * float(np.max(conv)) / (S * S)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        if best_obj < obj_before - 1e-10:
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
            if no_improve_sweeps > 20:
                # Perturb to escape
                pert = rng.normal(0, 0.005, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                s = float(a.sum())
                a = a / s
                S = 1.0
                conv = compute_conv(a)
                no_improve_sweeps = 0

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def peak_chase(a, time_budget, seed=42, verbose=False):
    """
    Peak-targeted line search: identify the dominant peak and
    move in the direction that most reduces it.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        # Find top peaks
        top_k = np.argsort(conv)[-3:][::-1]
        # Ensure distinct
        peaks = []
        for k in top_k:
            if all(abs(k - p) > n // 10 for p in peaks):
                peaks.append(k)
        if len(peaks) == 0:
            peaks = [int(np.argmax(conv))]

        # Combine responsibility vectors (weighted by peak values)
        r = np.zeros(n)
        for k in peaks:
            w = conv[k] / sum(conv[p] for p in peaks)
            r += w * peak_responsibility(a, k)

        # Direction: -r, projected to sum=0
        r = r - np.mean(r)
        if np.linalg.norm(r) < 1e-15:
            break

        # Line search: a + t * step, maintain non-negativity
        # Max step before hitting zero: t_max = min_i {a[i]/(-step[i]) : step[i] < 0}
        step = -r
        neg_mask = step < 0
        if np.any(neg_mask):
            t_max = float(np.min(a[neg_mask] / (-step[neg_mask])))
        else:
            t_max = 1.0
        t_max = min(t_max * 0.95, 0.5)

        # Golden section for t in [0, t_max]
        if t_max < 1e-12:
            break

        gr = (np.sqrt(5.0) - 1.0) / 2.0
        t_lo, t_hi = 0.0, t_max
        c_pt = t_hi - gr * (t_hi - t_lo)
        d_pt = t_lo + gr * (t_hi - t_lo)

        def f_t(t):
            a_try = a + t * step
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return 1e30
            a_try = a_try / s_try
            return obj_norm(a_try)

        fc = f_t(c_pt)
        fd = f_t(d_pt)
        for _ in range(40):
            if t_hi - t_lo < 1e-12:
                break
            if fc < fd:
                t_hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = t_hi - gr * (t_hi - t_lo)
                fc = f_t(c_pt)
            else:
                t_lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = t_lo + gr * (t_hi - t_lo)
                fd = f_t(d_pt)

        t_opt = (t_lo + t_hi) / 2.0
        a_new = a + t_opt * step
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = obj_norm(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step or random perturbation
            if rng.random() < 0.3:
                pert = rng.normal(0, 0.01, n)
                a_try = a * np.exp(pert)
                a_try = np.maximum(a_try, 0)
                a_try = a_try / float(a_try.sum())
                obj_try = obj_norm(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
            it += 1
            if it > 50:
                break

        it += 1
        if verbose and it % 10 == 0:
            print(f"    PeakChase it{it}: obj={best_obj:.8f}")

    return best_a


def lp_direction(a, time_budget=20.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns optimized direction g or None.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    # Build constraint matrix: A[k, i] = a_norm[k-i] if valid
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for i in range(n):
        r_min = i
        r_max = min(num_rows - 1, i + n - 1)
        A[r_min:r_max + 1, i] = a_norm[0:r_max - r_min + 1]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    start = time.monotonic()
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.PDLP]:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def lp_polish(a, time_budget, seed=42, verbose=False):
    """LP-based direction finding with line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g = lp_direction(a, time_budget=min(remaining * 0.4, 15.0))
        if g is None:
            break

        # Line search over blend factor
        best_t = 0.0
        best_obj_t = best_obj
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = obj_norm(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t

        if best_t > 0 and best_obj_t < best_obj - 1e-12:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / float(a.sum())
            best_obj = best_obj_t
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 3:
                pert = rng.normal(0, 0.02 * stall, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / float(a.sum())
                stall = 0

    return best_a


def diverse_init(n, rng, mode=0, alpha=0.3):
    """Generate diverse initializations with structural properties."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        # Asymmetric exponential
        t = rng.uniform(0.5, 3.0)
        a = np.exp(t * (x - 0.5))
    elif mode == 2:
        # Beta-like asymmetric
        p = rng.uniform(0.3, 0.7)
        a = x ** p * (1 - x + 1e-6) ** (1 - p)
    elif mode == 3:
        # Chirp
        f = rng.uniform(2, 8)
        a = 1.0 + alpha * np.cos(2 * np.pi * f * x + np.pi * x ** 2)
    elif mode == 4:
        # Two bumps
        c1 = rng.uniform(0.2, 0.4)
        c2 = rng.uniform(0.6, 0.8)
        w = rng.uniform(0.05, 0.15)
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
    elif mode == 5:
        # Anti-symmetric pair (scaled to non-negative)
        base = 1.0 + 0.5 * np.sin(2 * np.pi * rng.uniform(1, 5) * x)
        a = base * (1.0 + 0.3 * (x - 0.5))
    elif mode == 6:
        # Geometric
        r = rng.uniform(0.5, 0.9)
        a = r ** (np.arange(n) / n * 10)
    else:
        a = rng.uniform(0.5, 1.5, n)

    a = np.maximum(a, 0.01)
    a = a / float(np.sum(a))
    return a


def resample_sequence(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def multiplicative_update(a, time_budget, tau=50.0, seed=42):
    """
    Multiplicative update based on peak responsibility.
    a[i] *= exp(-eta * r[i]) where r is the responsibility vector.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    eta = 0.3

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        r = peak_responsibility(a, k_star)
        r = (r - np.mean(r)) / (np.std(r) + 1e-10)

        a_new = a * np.exp(-eta * r)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / float(np.sum(a_new))

        obj_new = obj_norm(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta *= 0.98
        else:
            # Try smaller eta
            eta *= 0.7
            if eta < 0.01:
                # Random restart within this method
                pert = rng.normal(0, 0.05, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / float(a.sum())
                eta = 0.2

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining fast CD, peak chase,
    LP polishing, multiplicative updates, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # === Load parent or default ===
    parent_found = False
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                parent_found = True
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if not parent_found:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Multi-initialization (5% of time) ===
    print(f"\n=== Phase 1: Multi-initialization ===")
    p1_budget = min((deadline - time.monotonic()) * 0.05, 60.0)
    if p1_budget > 5:
        for mode in range(7):
            a_try = diverse_init(n_best, rng, mode=mode)
            # Blend with current best
            a_blend = 0.3 * a_try + 0.7 * best_seq
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / float(a_blend.sum())
            obj_try = compute_obj(a_blend)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_blend.copy()
                print(f"  Init mode {mode}: obj={best_score:.8f} **")

    # === Phase 2: Fast CD (35% of time) ===
    print(f"\n=== Phase 2: Fast Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 450.0)
    if p2_budget > 30:
        a_cd = fast_cd(best_seq, time_budget=p2_budget, seed=seed, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            n_best = len(best_seq)
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 3: Peak Chase + CD (20% of time) ===
    print(f"\n=== Phase 3: Peak-Targeted Optimization ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.20, 250.0)
    if p3_budget > 20:
        for round_id in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(p3_budget / (4 - round_id), remaining * 0.5)
            if alloc < 10:
                break

            # Peak chase
            t_chase = min(alloc * 0.4, 30.0)
            a_pc = peak_chase(best_seq, time_budget=t_chase, seed=seed + 100 * round_id)
            obj_pc = compute_obj(a_pc)
            if obj_pc < best_score:
                best_score = obj_pc
                best_seq = a_pc.copy()
                print(f"  PeakChase r{round_id}: obj={best_score:.8f} **")

            # Follow with CD
            remaining = deadline - time.monotonic()
            t_cd = min(alloc * 0.6, 60.0)
            if t_cd > 10:
                a_cd2 = fast_cd(best_seq, time_budget=t_cd, seed=seed + 200 * round_id)
                obj_cd2 = compute_obj(a_cd2)
                if obj_cd2 < best_score:
                    best_score = obj_cd2
                    best_seq = a_cd2.copy()
                    print(f"  CD r{round_id}: obj={best_score:.8f} **")

    # === Phase 4: LP Polishing (20% of time) ===
    print(f"\n=== Phase 4: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.20, 200.0)
    if p4_budget > 15:
        for round_id in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(p4_budget / (5 - round_id), remaining * 0.5)
            if alloc < 8:
                break
            a_lp = lp_polish(best_seq, time_budget=alloc, seed=5000 + round_id * 31)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP r{round_id}: obj={best_score:.8f} **")
            else:
                print(f"  LP r{round_id}: obj={obj_lp:.8f}")

    # === Phase 5: Multiplicative Updates (10% of time) ===
    print(f"\n=== Phase 5: Multiplicative Updates ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.10, 100.0)
    if p5_budget > 10:
        a_mu = multiplicative_update(best_seq, time_budget=p5_budget, seed=seed + 4000)
        obj_mu = compute_obj(a_mu)
        if obj_mu < best_score:
            best_score = obj_mu
            best_seq = a_mu.copy()
            # Follow with CD
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_cd3 = fast_cd(best_seq, time_budget=min(remaining * 0.3, 50.0),
                                seed=seed + 4500)
                obj_cd3 = compute_obj(a_cd3)
                if obj_cd3 < best_score:
                    best_score = obj_cd3
                    best_seq = a_cd3.copy()
            print(f"  MU+CD: obj={best_score:.8f} **")
        else:
            print(f"  MU: obj={obj_mu:.8f}")

    # === Phase 6: Multi-length exploration (10% of time) ===
    print(f"\n=== Phase 6: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    n_options = sorted(set([
        max(400, n_current - 100), n_current + 100, n_current + 200,
        600, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and n != n_current]
    n_options = n_options[:4]

    if n_options:
        print(f"  Trying n values: {n_options}")
        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            a_try = resample_sequence(best_seq, n_try)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try

            # Quick CD
            alloc_cd = min(remaining * 0.15, 40.0)
            if alloc_cd > 10:
                a_try = fast_cd(a_try, time_budget=alloc_cd, seed=n_try)

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final intensive CD (remaining time) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = remaining * 0.6
        a_final = fast_cd(best_seq, time_budget=alloc, seed=seed + 99999, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    # One more peak chase if time
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_pc_final = peak_chase(best_seq, time_budget=min(remaining * 0.5, 30.0),
                                seed=seed + 77777)
        obj_pc_final = compute_obj(a_pc_final)
        if obj_pc_final < best_score:
            best_score = obj_pc_final
            best_seq = a_pc_final.copy()
            print(f"  Final PeakChase: obj={best_score:.8f} **")

    # Final short CD
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_cd_final = fast_cd(best_seq, time_budget=min(remaining * 0.7, 20.0),
                             seed=seed + 88888)
        obj_cd_final = compute_obj(a_cd_final)
        if obj_cd_final < best_score:
            best_score = obj_cd_final
            best_seq = a_cd_final.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
