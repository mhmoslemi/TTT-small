2. Exact coordinate descent with incremental updates

4. Multi-length exploration with resampling
5. Simulated annealing for local minimum escape

import numpy as np
from scipy.fft import fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = fft(a, N)
    return np.real(ifft(fa * fa, N))[:2 * n - 1]


def obj_norm(a):
    """Compute 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def project_simplex(x):
    """Project x onto {a >= 0, sum(a) = 1}."""
    n = len(x)
    # Sort in decreasing order
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)


def compute_gradient(a, tau):
    """Compute smoothed gradient of max(conv(a,a)).
    
    grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
    where p = softmax(tau * (conv - max_conv))
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    # Numerically stable softmax
    shifted = tau * (conv - max_conv)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    p = exp_vals / float(np.sum(exp_vals))
    
    # Compute gradient via FFT
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    grad_full = np.real(ifft(P * np.conj(A), N))
    grad = 2.0 * grad_full[:n]
    return grad, conv, p

    """Project x onto {a >= 0, sum(a) = 1} using sorted algorithm."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho_mask = u * idx > (cssv - 1)
    if not np.any(rho_mask):
        return np.ones(n) / n
    rho = np.nonzero(rho_mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(x - theta, 0.0)


def compute_smoothed_grad(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 2 * n - 1:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    grad_full = np.real(ifft(P * np.conj(A), N))
    grad = 2.0 * grad_full[:n]
    return grad, conv, p


def nesterov_smoothed_descent(a, time_budget=60.0, tau_init=10.0, tau_end=200.0, seed=42):
    """Nesterov-accelerated smoothed gradient descent with simplex projection."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_norm(a)
    start = time.monotonic()

    y = a.copy()
    v = np.zeros(n)
    t_k = 1.0
    lr = 1.0 / np.sqrt(n)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init * (tau_end / tau_init) ** progress

        try:
            grad, conv, p = compute_smoothed_grad(y, tau)
        except Exception:
            break

        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-15:
            lr_eff = min(lr * 0.98 + 0.02 / (g_norm + 1e-10), lr * 2)
        else:
            lr_eff = lr

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        beta = (t_k - 1.0) / t_new

        y_extrap = y + beta * v
        y_new = project_simplex(y_extrap - lr_eff * grad)
        v = y_new - y
        y = y_new
        t_k = t_new

        obj = obj_from_conv(y, compute_conv(y))
        if obj < best_obj:
            best_obj = obj
            best_a = y.copy()

        lr *= 0.999

    return best_a


def mwu_lp_direction(a, T=3000, eta=None):
    """
    MWU to approximate the LP solution:
    min_{g in simplex} max_k (a*g)[k]
    Returns g (the LP direction) and the achieved peak value.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))

    g = np.ones(n) / n
    if eta is None:
        eta = 1.0 / np.sqrt(n)

    for t in range(T):
        # Compute v = a_norm * g (convolution)
        conv = compute_conv(a_norm * g)
        k_star = int(np.argmax(conv))

        # Update: g[i] *= exp(-eta * a_norm[k_star - i]) for valid i
        # a_norm[k_star - i] is non-zero when 0 <= k_star - i < n
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        if i_max < i_min:
            break
        idx = np.arange(i_min, i_max + 1)
        weights = a_norm[k_star - idx]
        g[idx] *= np.exp(-eta * weights)

        # Rescale
        s = float(np.sum(g))
        if s < 1e-30:
            g = np.ones(n) / n
            break
        g = g / s

    # Final evaluation
    conv_final = compute_conv(a_norm * g)
    peak = float(np.max(conv_final))
    return g, peak


def coordinate_descent(a, time_budget=200.0, verbose=False, seed=42):
    """
    Exact coordinate descent with incremental convolution updates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0

    conv = compute_conv(a)
    best_obj = obj_from_conv(a, conv)
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12 and v < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a[ks_lin - i]
            c_k_lin = conv[ks_lin].copy()

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            # Evaluate 1D function
            def f_eval(x):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Candidate points for initial bracket
            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, v * 0.3, v * 0.6, v * 0.8, v,
                          v * 1.2, v * 1.5, v * 2.0, 1.0 / n, 2.0 / n,
                          max(0.0, v - scale * 0.5), max(0.0, v + scale * 0.5)]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = f_eval(v)
            for x in candidates:
                fx = f_eval(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section refinement
            sc = max(best_x, 1.0 / n, 1e-6)
            x_lo = max(0.0, best_x - sc)
            x_hi = best_x + sc * 2.0
            if x_hi - x_lo < 1e-12:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_eval(c_pt)
            fd = f_eval(d_pt)

            for _ in range(40):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_eval(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_eval(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            S += delta

            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def block_cd(a, time_budget=100.0, seed=42):
    """Block coordinate descent: update pairs of coordinates simultaneously."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    conv = compute_conv(a)
    S = 1.0
    best_obj = obj_from_conv(a, conv)
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0

    while time.monotonic() - start < time_budget:
        # Pick a random pair
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue

        v_i, v_j = a[i], a[j]
        S_other = S - v_i - v_j

        # Try a grid of (ratio) values
        total = v_i + v_j
        if total < 1e-12:
            a[i] = 1.0 / (2 * n)
            a[j] = 1.0 / (2 * n)
            S = 1.0
            conv = compute_conv(a)
            continue

        best_ratio = v_i / total
        best_f = 1e30

        # Search over ratio in [0, 1]
        for ratio in np.linspace(0.01, 0.99, 20):
            new_vi = total * ratio
            new_vj = total * (1 - ratio)

            # Compute new conv values
            # conv changes for all k involving i or j
            conv_new = conv.copy()
            # Remove old contributions
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                conv_new[k] -= v_i * a[k - i] if 0 <= k - i < n else 0
            for k in range(max(0, j), min(2 * n - 1, j + n)):
                conv_new[k] -= v_j * a[k - j] if 0 <= k - j < n else 0
            # Add new contributions
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                conv_new[k] += new_vi * a[k - i] if 0 <= k - i < n else 0
            for k in range(max(0, j), min(2 * n - 1, j + n)):
                conv_new[k] += new_vj * a[k - j] if 0 <= k - j < n else 0

            S_new = S_other + new_vi + new_vj
            if S_new < 1e-10:
                continue
            f_val = 2.0 * n * float(np.max(conv_new)) / (S_new * S_new)
            if f_val < best_f:
                best_f = f_val
                best_ratio = ratio

        new_vi = total * best_ratio
        new_vj = total * (1 - best_ratio)

        # Update
        a[i] = new_vi
        a[j] = new_vj
        S = S_other + new_vi + new_vj
        conv = compute_conv(a)
        obj = 2.0 * n * float(np.max(conv)) / (S * S)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        it += 1
        if it % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"    BlockCD it{it}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 1e-4)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.zeros(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Two bumps
        c1, c2 = n // 3, 2 * n // 3
        w = n // 10
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def annealed_search(a, time_budget=80.0, seed=42):
    """Simulated annealing with structured perturbations."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = obj_norm(a)
    cur_obj = best_obj
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    T_init = 0.005
    T_end = 0.0003

    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / max(time_budget, 1e-10))
        T = T_init * (T_end / T_init) ** progress

        r = rng.random()
        a_try = a.copy()

        if r < 0.4:
            scale = 0.03 * (1.0 - 0.5 * progress)
            a_try = a * np.exp(rng.normal(0, scale, n))
        elif r < 0.7:
            bs = int(rng.integers(2, max(3, n // 15)))
            i0 = int(rng.integers(0, n - bs))
            a_try[i0:i0 + bs] *= np.exp(rng.normal(0, 0.08))
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                factor = np.exp(rng.normal(0, 0.15))
                a_try[i] *= factor
                a_try[j] /= factor

        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        obj_try = obj_norm(a_try)
        delta = obj_try - cur_obj

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_try
            cur_obj = obj_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining Nesterov gradient,
    coordinate descent, MWU-LP, multi-n exploration, and annealing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 12.0
    rng = np.random.default_rng(seed)

    # Initialize from parent
    best_seq = None
    best_score = np.inf
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = obj_norm(best_seq)
                print(f"Parent (n={best_seq.size}): obj={best_score:.8f}")
    except Exception:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = obj_norm(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    # === Phase 1: Nesterov Smoothed Descent (fast global) ===
    print(f"\n=== Phase 1: Nesterov Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.06, 60.0)
    if p1_budget > 10:
        a_ns = nesterov_smoothed_descent(best_seq, time_budget=p1_budget,
                                         tau_init=5.0, tau_end=150.0, seed=seed)
        obj_ns = obj_norm(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov: obj={obj_ns:.8f} (no improvement)")

    # === Phase 2: Heavy Coordinate Descent ===
    print(f"\n=== Phase 2: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.30, 350.0)
    if p2_budget > 30:
        a_cd = coordinate_descent(best_seq, time_budget=p2_budget,
                                  verbose=True, seed=seed)
        obj_cd = obj_norm(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 3: CWU-LP Cycles ===
    print(f"\n=== Phase 3: MWU-LP Cycles ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.20, 200.0)
    if p3_budget > 20:
        n_cycles = int(p3_budget // 25)
        n_cycles = max(2, min(n_cycles, 8))
        cycle_time = p3_budget / n_cycles

        for cyc in range(n_cycles):
            if time.monotonic() > deadline:
                break
            cyc_start = time.monotonic()
            cyc_alloc = min(cycle_time, deadline - cyc_start - 5)
            if cyc_alloc < 10:
                break

            # Get MWU direction
            mwu_time = min(cyc_alloc * 0.6, 15.0)
            g_mwu, peak_mwu = mwu_lp_direction(best_seq, T=2000, eta=1.0 / np.sqrt(len(best_seq)))
            print(f"  MWU cycle {cyc}: dir peak={peak_mwu:.6f}, t={time.monotonic()-cyc_start:.1f}s")

            # Try blending
            best_blend_obj = best_score
            best_blend_a = best_seq
            for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
                a_try = (1.0 - t) * best_seq + t * g_mwu
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = obj_norm(a_try)
                if obj_try < best_blend_obj:
                    best_blend_obj = obj_try
                    best_blend_a = a_try.copy()

            if best_blend_obj < best_score - 1e-12:
                best_score = best_blend_obj
                best_seq = best_blend_a.copy()
                print(f"  MWU cycle {cyc}: improved to {best_score:.8f} **")

            # CD polish after LP
            cd_time = deadline - time.monotonic() - 5
            cd_time = min(cd_time * 0.3, 30.0)
            if cd_time > 5:
                a_cd2 = coordinate_descent(best_seq, time_budget=cd_time,
                                           verbose=False, seed=seed + 100 + cyc * 7)
                obj_cd2 = obj_norm(a_cd2)
                if obj_cd2 < best_score:
                    best_score = obj_cd2
                    best_seq = a_cd2.copy()
                    print(f"  CD after LP {cyc}: {best_score:.8f} **")

    # === Phase 4: More CD ===
    print(f"\n=== Phase 4: Extended CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.15, 120.0)
        a_cd3 = coordinate_descent(best_seq, time_budget=alloc,
                                   verbose=True, seed=seed + 5000)
        obj_cd3 = obj_norm(a_cd3)
        if obj_cd3 < best_score:
            best_score = obj_cd3
            best_seq = a_cd3.copy()
            print(f"  Extended CD: obj={best_score:.8f} **")
        else:
            print(f"  Extended CD: obj={obj_cd3:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        n_best - 50, n_best + 50, n_best + 100, n_best - 100,
        600, 700, 800, 900, 1000, 1100, 1200, 1400
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_best) >= 30]
    n_options = n_options[:5]

    print(f"  Trying n values: {n_options}")
    for n_try in n_options:
        if time.monotonic() > deadline - 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # Distribute remaining time among options
        remaining_opts = deadline - time.monotonic() - 60
        n_remaining = len([n2 for n2 in n_options if n2 >= n_try])
        per_opt = min(remaining_opts / max(n_remaining, 1), 80.0)

        alloc_cd = min(per_opt * 0.7, 40.0)
        if alloc_cd > 10:
            a_try = coordinate_descent(a_try, time_budget=alloc_cd,
                                       verbose=False, seed=n_try)

        # Brief Nesterov
        alloc_ns = min(per_opt * 0.2, 15.0)
        if alloc_ns > 5:
            a_try = nesterov_smoothed_descent(a_try, time_budget=alloc_ns,
                                              tau_init=10.0, tau_end=100.0, seed=n_try + 1)

        obj_try = obj_norm(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Annealing + CD ===
    print(f"\n=== Phase 6: Annealing + CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc_sa = min(remaining * 0.3, 60.0)
        a_sa = annealed_search(best_seq, time_budget=alloc_sa, seed=seed + 7777)
        obj_sa = obj_norm(a_sa)
        if obj_sa < best_score:
            best_score = obj_sa
            best_seq = a_sa.copy()
            print(f"  Anneal: obj={best_score:.8f} **")

        # CD after annealing
        remaining = deadline - time.monotonic()
