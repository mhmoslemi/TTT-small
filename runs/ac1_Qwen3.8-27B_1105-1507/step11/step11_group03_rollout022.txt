Let me also think about a key insight: **the problem has a natural connection to the "minimum distance" problem in coding theory**.

For a sequence a of length n with a[i] >= 0, sum(a) = 1:
- c = conv(a, a) has length 2n-1
- We want to minimize max(c)
- This is equivalent to finding a "flat" autocorrelation

In the Wikipedia article on "Barker sequences", the Barker sequence of length 13 has the property that its aperiodic autocorrelation is bounded by 1 for all non-zero lags. The continuous analog would be a function with flat autocorrelation.

For our problem (non-negative, sum=1), the optimal solution should have:
- c[k] approximately constant for all k in the "bulk" (away from edges)
- c[0] and c[2n-2] will be small (since they involve only one term)
- The "bulk" value determines the maximum

If c[k] ≈ C for k in some range, then the objective is approximately 2n*C.
With sum(a) = 1, C ≈ 1/n gives objective ≈ 2.
To get below 2, we need C < 1/n, which means the distribution of X+Y is flatter than uniform.

For a flat distribution of X+Y on {0, ..., 2n-2}, we'd need P(X+Y=k) = 1/(2n-1) for all k. But this requires P(X+Y=k) to be constant, which is impossible for a triangular sumset (the edges must have lower probability).

The best we can do is make the "bulk" as flat as possible while the edges taper off. The maximum will be in the bulk, and we want that to be as low as possible.

For a uniform distribution, the bulk value is 1/n and the max is also 1/n (at the center). The edges go down to 1/n^2.

To get the bulk below 1/n, we need a non-uniform distribution. For example, if a is concentrated at the edges (like a U-shape), then X+Y will be concentrated at the extremes and the center, with a dip in between. But the max might still be at the edge.

Actually, I think the optimal shape is something like a "wavy" distribution that makes the sum distribution as flat as possible. This is related to the "flat-top" window design.

```python
import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smoothed_grad(a, tau):
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    shifted = tau * (conv - exact_max)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max
```

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent with Nesterov acceleration, smoothed max (log-sum-exp),
fictitious play, LP refinement, and multi-scale approach.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_grad_fft(a, p, N):
    """Compute gradient of sum_k p_k * c_k using FFT cross-correlation.
    Returns gradient of length n where c = conv(a,a)."""
    n = len(a)
    A = fft(a, N)
    P = fft(p, N)
    xcorr = ifft(A * np.conj(P)).real
    indices = np.mod(-np.arange(n), N)
    g = 2.0 * xcorr[indices]
    return g


def smoothed_grad(a, tau):
    """Compute gradient of tau-smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    shifted = tau * (conv - exact_max)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    g = smoothed_grad_fft(a, p, N)
    return g, exact_max


def mirror_descent_nesterov(a_init, time_budget, verbose=True, seed=42,
                             tau_start=10, tau_end=5000, 
                             tau_iters_total=3000,
                             lr_base=1.0, beta=0.9):
    """
    Nesterov-accelerated mirror descent on the probability simplex.
    Uses exponential weights parametrization (a[i] > 0 always).
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    s = float(np.sum(a))
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    N = 1
    while N < 3 * n - 1:
        N <<= 1

    # Nesterov momentum
    v = np.zeros(n)
    it = 0
    
    while time.monotonic() - start < time_budget:
        # Adaptive tau
        progress = min(1.0, it / tau_iters_total)
        tau = tau_start * (tau_end / tau_start) ** progress
        # Adaptive learning rate
        lr = lr_base / (np.sqrt(n) * (1.0 + 0.1 * it))

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 1e-12)
        y = y / y.sum()

        # Compute gradient at y
        g, exact_max = smoothed_grad(y, tau)
        
        # Mirror descent step: w = log(a), w_new = w - lr*g, a_new = softmax(w_new)
        # Equivalent: a_new = a * exp(-lr*g) / sum(a * exp(-lr*g))
        lg = -lr * g
        lg -= lg.max()  # numerical stability
        weights = np.exp(lg)
        a_new = a * weights
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        # Check improvement
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        # Update Nesterov velocity
        v = beta * v + (a - a_new)
        a = a_new
        it += 1

        if verbose and it % 200 == 0:
            print(f"    MD-Nest it{it:5d}: obj={best_obj:.10f}, tau={tau:.1f}, lr={lr:.6f}")

    return best_a


def fictitious_play(a_init, time_budget, verbose=True, seed=42,
                    max_iters=500, lr=0.05):
    """
    Fictitious play / no-regret approach:
    Alternate between updating lag weights y and optimizing a for fixed y.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = float(np.sum(a))
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    N = 1
    while N < 3 * n - 1:
        N <<= 1

    # Initialize lag distribution
    conv = compute_conv(a)
    y = np.ones(2*n-1) / (2*n-1)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Update y based on current convolution
        conv = compute_conv(a)
        obj = 2.0 * n * float(np.max(conv))
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Update y: softmax with adaptive temperature
        temp = 0.5 + 0.5 * (it / max_iters)  # start sharp, become smoother
        shifted = conv / (temp * max(1e-10, float(np.max(conv))))
        shifted -= shifted.max()
        y_new = np.exp(shifted)
        y_new = y_new / y_new.sum()
        # Exponential weights update
        y = 0.7 * y + 0.3 * y_new

        # Optimize a for fixed y: minimize a^T Q a where Q[i,j] = y[i+j]
        # Use gradient descent with mirror step
        A = fft(a, N)
        Y = fft(y, N)
        xcorr = ifft(A * np.conj(Y)).real
        indices = np.mod(-np.arange(n), N)
        grad = 2.0 * xcorr[indices]

        # Mirror descent step
        lr_adapt = lr / (1.0 + 0.01 * it)
        lg = -lr_adapt * grad
        lg -= lg.max()
        a = a * np.exp(lg)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        # Occasionally add noise to escape local minima
        if it % 50 == 0 and it > 0:
            a = a + 0.005 * rng.standard_normal(n)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()

        if verbose and it % 50 == 0:
            print(f"    FP it{it:3d}: obj={best_obj:.10f}")

    return best_a


def lp_solve(a):
    """Solve min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None
