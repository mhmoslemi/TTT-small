import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None, tolerance=None):
    """

    n = len(a)

    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass
    
    for solver in ['CLARABEL', 'SCS']:

                kwargs['tol_feas'] = 1e-8

                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:

def lp_alternate_improved(a, time_budget, seed=42, verbose=True, tol=1e-9):
    """Improved LP alternating with bisection line search and better stall handling."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:

                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))

                    t_lo = t_mid
                else:

            improvement = best_obj - best_local_obj
            best_obj = best_local_obj

            a = best_a.copy()

def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 10):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def interpolate_to_length(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(n_new) / n_new
    return a_new / s


def sine_init(n):
    """Sine-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def raised_cosine_init(n):
    """Raised cosine initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def two_lobe_init(n):
    """Two-lobe (Gaussian at each end) initialization."""
    x = np.linspace(0, 1, n)
    sigma = 0.25
    a = np.exp(-0.5 * ((x - 0.15) / sigma) ** 2) + np.exp(-0.5 * ((x - 0.85) / sigma) ** 2)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def triangle_init(n):
    """Triangular (peaked in middle) initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 - np.abs(2.0 * x - 1.0)
    a = np.maximum(a, 1e-6)
    s = float(np.sum(a))
    return a / s

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses enhanced LP alternating minimization with:
- Multiple sequence lengths and diverse initializations (sine, raised-cosine, two-lobe, triangle)
- Bisection line search in the alternating scheme
- Intensive coordinate descent polishing with correct in-place convolution updates
- Adaptive time allocation and multi-restart exploration
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None, gap_tol=1e-8):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass

    for solver in ['CLARABEL', 'SCS']:
        try:
            kwargs = {}
            if solver == 'CLARABEL':
                kwargs['tol_gap_abs'] = gap_tol
                kwargs['tol_feas'] = gap_tol
            prob.solve(solver=getattr(cp, solver), **kwargs)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def cd_polish(a, time_budget, seed=42, verbose=True):
    """
    Coordinate descent with careful in-place convolution updates.
    Tries multiple candidate values per coordinate.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    no_improve = 0
    max_no_improve = 8
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            old_a_j = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)

            # Unaffected max
            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))

            # Build candidate values
            mean_val = 1.0 / n
            candidates = np.array([
                0.0, old_a_j * 0.3, old_a_j * 0.6, old_a_j * 0.85,
                old_a_j * 1.15, old_a_j * 1.4, old_a_j * 2.0,
                mean_val * 0.5, mean_val, mean_val * 1.5, mean_val * 2.5
            ])
            candidates = np.maximum(candidates, 0.0)
            candidates = candidates[candidates <= 0.5]
            candidates = np.unique(candidates)

            best_trial_obj = best_obj
            best_trial_val = old_a_j

            for val in candidates:
                delta = val - old_a_j
                if abs(delta) < 1e-15:
                    continue
                S_new = 1.0 + delta
                if S_new <= 1e-10:
                    continue

                k_indices = np.arange(k_start, k_end + 1)
                k_j = k_indices - j

                # Compute affected entries (using OLD a values)
                affected = conv[k_start:k_end + 1] + 2.0 * delta * a[k_j]
                if 2 * j >= k_start and 2 * j <= k_end:
                    idx = 2 * j - k_start
                    affected[idx] = conv[2 * j] + 2.0 * delta * old_a_j + delta * delta

                max_aff = float(np.max(affected))
                max_conv_new = max(max_unaffected, max_aff)
                obj_trial = 2.0 * n * max_conv_new / (S_new * S_new)

                if obj_trial < best_trial_obj - 1e-14:
                    best_trial_obj = obj_trial
                    best_trial_val = val

            if best_trial_val != old_a_j:
                delta = best_trial_val - old_a_j
                S_new = 1.0 + delta

                # Update conv in-place (BEFORE changing a[j])
                k_indices = np.arange(k_start, k_end + 1)
                k_j = k_indices - j
                conv_update = 2.0 * delta * a[k_j].copy()
                if 2 * j >= k_start and 2 * j <= k_end:
                    idx = 2 * j - k_start
                    conv_update[idx] = 2.0 * delta * old_a_j + delta * delta
                conv[k_start:k_end + 1] += conv_update

                # Now change a[j]
                a[j] = best_trial_val

                # Normalize
                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S

                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")

    return best_a


def lp_alternate(a, time_budget, seed=42, verbose=True, gap_tol=1e-9):
    """LP alternating minimization with bisection line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    iter_count = 0
    stall = 0

    while time.monotonic() - start < time_budget - 5:
        t0 = time.monotonic()
        g_lp, M_lp = solve_lp(a, warm_start=prev_g, gap_tol=gap_tol)
        if g_lp is None:
            break
        prev_g = g_lp
        iter_count += 1
        lp_time = time.monotonic() - t0

        # Bisection line search
        best_local_obj = best_obj
        best_local_a = a.copy()

        # Check full step
        a_try = g_lp.copy()
        obj_try = compute_obj_norm(a_try)
        if obj_try < best_obj - 1e-13:
            best_local_obj = obj_try
            best_local_a = a_try.copy()
        else:
            t_lo, t_hi = 0.0, 1.0
            for _ in range(8):
                t_mid = 0.5 * (t_lo + t_hi)
                a_try = (1.0 - t_mid) * a + t_mid * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    t_hi = t_mid
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj_norm(a_try)
                if obj_try < best_obj - 1e-14:
                    best_local_obj = obj_try
                    best_local_a = a_try.copy()
                    t_lo = t_mid
                else:
                    t_hi = t_mid

        if best_local_obj < best_obj - 1e-14:
            improvement = best_obj - best_local_obj
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall = 0
            if verbose and (iter_count % 30 == 1 or improvement > 0.002):
                elapsed = time.monotonic() - start
                print(f"    LP {iter_count}: obj={best_obj:.10f}, imp={improvement:.6f}, "
                      f"lp_t={lp_time:.2f}s, rem={time_budget-elapsed:.0f}s")
        else:
            stall += 1
            if stall >= 6:
                strength = 0.004 + 0.002 * min(stall - 6, 12)
                pert = rng.standard_normal(n) * strength
                a = best_a * (1.0 + pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                prev_g = None
                stall = 0

        # Relax tolerance if LP is slow
        if lp_time > 8 and gap_tol > 1e-7:
            gap_tol = 1e-7

    if verbose:
        print(f"    LP done: {iter_count} iters, best={best_obj:.10f}")
    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Structured Fourier perturbation for diversity."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 12):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def interpolate_to_length(a, n_new):
    """Interpolate sequence to new length, normalized."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(n_new) / n_new
    return a_new / s


def sine_init(n):
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s if s > 1e-10 else np.ones(n) / n


def raised_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    s = float(np.sum(a))
    return a / s if s > 1e-10 else np.ones(n) / n


def two_lobe_init(n):
    x = np.linspace(0, 1, n)
    sigma = 0.22
    a = np.exp(-0.5 * ((x - 0.12) / sigma) ** 2) + np.exp(-0.5 * ((x - 0.88) / sigma) ** 2)
    s = float(np.sum(a))
    return a / s if s > 1e-10 else np.ones(n) / n


def gaussian_center_init(n):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / 0.3) ** 2)
    s = float(np.sum(a))
    return a / s if s > 1e-10 else np.ones(n) / n
