def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective 2*n*max(conv)/sum^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

    """Assume sum=1, return 2*n*max(conv)."""
    n = len(a)
    if n == 0:

    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def solve_lp_minpeak(a, verbose=False):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None) if infeasible."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:
                    g_val = g_val / g_sum
                    return g_val, float(M.value)
        except Exception:
            continue
    return None, None

                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:

            # Initial bracket

            candidates = [0.0, v * 0.25, v * 0.5, v * 0.75, v,
                          v * 1.25, v * 1.5, v * 2.0, v * 3.0,
                          S / (2 * n), S / n, 2.0 * S / n]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))
            
            best_x = v
            best_f = f_1d(v)
            for x in candidates:
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x
            
            # Golden section
            x_lo = max(0.0, best_x - scale * 0.5)
            x_hi = best_x + scale
            if x_hi <= x_lo:
                x_hi = x_lo + 1e-8
            
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)
            
            for _ in range(35):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)
            
            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            
            if abs(x_opt - v) < 1e-15:
                continue
            
            delta = x_opt - v
            a[i] = x_opt
            
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            
            S += delta
        
        # Track best
        S_new = float(np.sum(a))
        if S_new > 1e-10:
            obj_now = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
            if obj_now < best_obj:
                best_obj = obj_now
                best_a = a.copy()
    
    return best_a

"""
Optimizes step function coefficients using extensive LP cycling, exact coordinate
descent, and multi-n exploration. The main improvement over the previous approach
is much more thorough LP cycling (the fixed-point iteration converges to a tighter
local optimum), combined with faster CD and broader n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective 2*n*max(conv)/sum^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build sparse-friendly matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_lo = max(0, j)
        k_hi = min(2 * n - 1, j + n)
        if k_lo < k_hi:
            A[k_lo:k_hi, j] = a[:(k_hi - k_lo)]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:
                    g_val = g_val / g_sum
                    return g_val, float(M.value)
        except Exception:
            continue
    return None, None


def eval_1d(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i):
    """Evaluate 1D subproblem objective at point x."""
    if len(a_ki_lin) > 0:
        lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
        max_lin = float(np.max(lin_vals))
    else:
        max_lin = 0.0
    quad_val = c_2i + x * x - v * v
    max_val = max(max_lin, quad_val)
    denom = (S_other + x) * (S_other + x)
    if denom < 1e-30:
        return 1e30
    return 2.0 * n * max_val / denom
