import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _fft_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = _next_pow2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def _obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    c = _fft_conv(a_n)
    return 2.0 * n * float(np.max(c))


def _build_sym_matrix(a, n):
    """Build B matrix for symmetric LP. B[k, col] gives coefficient of h[col] in (a*g)_k."""
    half = n // 2
    B = np.zeros((n, half))
    for k in range(n):
        lo1 = max(0, k - n + 1)
        hi1 = min(half - 1, k)
        if hi1 >= lo1:
            cols = np.arange(lo1, hi1 + 1)
            B[k, cols] += a[k - cols]
        lo2 = max(0, n - 1 - k)
        hi2 = min(half - 1, 2 * n - 2 - k)
        if hi2 >= lo2:
            cols = np.arange(lo2, hi2 + 1)
            B[k, cols] += a[k - n + 1 + cols]
    return B


def _solve_lp_minM(a, n, half, B):
    """LP: min M s.t. B@h <= M, 2*sum(h)=1, h>=0."""
    h = cp.Variable(half, nonneg=True)
    M = cp.Variable()
    constraints = [B @ h <= M, 2 * cp.sum(h) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if h.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                h_val = np.clip(h.value, 0, None)
                sh = float(np.sum(h_val))
                if sh > 1e-12:
                    h_val = h_val / sh * 0.5
                    g = np.concatenate([h_val, h_val[::-1]])
                    return g, float(M.value)
        except Exception:
            continue
    return None, None


def _solve_lp_maxsum(a, n, half, B, M_cur):
    """LP: max sum(h) s.t. B@h <= M_cur, h>=0. Returns rescaled g (sum=1)."""
    h = cp.Variable(half, nonneg=True)
    constraints = [B @ h <= M_cur]
    prob = cp.Problem(cp.Maximize(cp.sum(h)), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if h.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                h_val = np.clip(h.value, 0, None)
                sh = float(np.sum(h_val))
                if sh > 1e-12:
                    h_val = h_val / sh * 0.5
                    g = np.concatenate([h_val, h_val[::-1]])
                    return g, float(M_cur / (2.0 * sh))
        except Exception:
            continue
    return None, None


def _line_search(a, g, n_coarse=50, n_fine=30, t_max=1.5):
    """Exhaustive line search between a and g. Returns (best_t, best_obj, best_a)."""
    best_t = 0.0
    best_obj = _obj(a)
    best_a = a.copy()
    ts = np.linspace(0.0, t_max, n_coarse)
    for t in ts:
        a_try = np.maximum((1.0 - t) * a + t * g, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj = _obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
            best_a = a_try.copy()
    if 0.0 < best_t < t_max:
        lo = max(0.0, best_t - 0.05)
        hi = min(t_max, best_t + 0.05)
        ts_fine = np.linspace(lo, hi, n_fine)
        for t in ts_fine:
            a_try = np.maximum((1.0 - t) * a + t * g, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj = _obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t
                best_a = a_try.copy()
    return best_t, best_obj, best_a


def _u_init(n, sharp=4.0):
    """Symmetric U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.03 + sharp * (2 * x - 1) ** 2)
    a = (a + a[::-1]) / 2.0
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _diverse_init(n, rng, mode=0):
    """Various symmetric initializations."""
    half = n // 2
    x = np.linspace(0, 1, half)
    if mode == 0:
        h = 1.0 / (0.02 + 6.0 * (2 * x - 1) ** 2)
    elif mode == 1:
        h = (1 - x) ** 3 + 0.05
    elif mode == 2:
        h = np.exp(-8 * x ** 2) + 0.1
    elif mode == 3:
        h = rng.uniform(0.2, 1.0, half)
        h = (h + h[::-1]) / 2
        h = np.convolve(h, np.ones(7) / 7, mode='same')
    elif mode == 4:
        h = 1.0 / (0.01 + 10.0 * (2 * x - 1) ** 4)
    else:
        h = (1 - 4 * x * (1 - x)) ** 2 + 0.05
    h = np.maximum(h, 0)
    a = np.concatenate([h, h[::-1]])
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _interpolate(a_src, n_dst):
    """Interpolate sequence to new length, preserving symmetry."""
    n_src = len(a_src)
    half_src = n_src // 2
    half_dst = n_dst // 2
    h_src = a_src[:half_src]
    x_src = np.linspace(0, 1, half_src)
    x_dst = np.linspace(0, 1, half_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    h_dst = np.maximum(h_dst, 0)
    a_dst = np.concatenate([h_dst, h_dst[::-1]])
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s


def _lp_polish(a, time_budget=100.0, max_iters=150, verbose=False):
    """Main LP-based optimization loop with symmetry."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = (a + a[::-1]) / 2.0
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    half = n // 2
    best_a = a.copy()
    best_obj = _obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(12345 + n)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        T0 = time.monotonic()
        conv = _fft_conv(a)
        M_cur = float(np.max(conv))

        # Build matrix
        B = _build_sym_matrix(a, n, half)

        # LP 1: min M
        g1, M1 = _solve_lp_minM(a, n, half, B)

        # LP 2: max sum (complementary direction)
        g2, M2 = _solve_lp_maxsum(a, n, half, B, M_cur)

        T1 = time.monotonic()
        lp_time = T1 - T0

        # Try both directions
        best_t = 0.0
        best_obj_ls = best_obj
        best_a_ls = best_a.copy()

        if g1 is not None:
            t1, obj1, a1 = _line_search(a, g1, n_coarse=40, n_fine=20, t_max=1.5)
            if obj1 < best_obj_ls:
                best_obj_ls = obj1
                best_a_ls = a1
                best_t = t1

        if g2 is not None:
            t2, obj2, a2 = _line_search(a, g2, n_coarse=30, n_fine=20, t_max=1.5)
            if obj2 < best_obj_ls:
                best_obj_ls = obj2
                best_a_ls = a2
                best_t = t2

        if best_obj_ls < best_obj - 1e-14:
            a = best_a_ls
            best_a = a.copy()
            best_obj = best_obj_ls
            stall = 0
            if verbose and it % 5 == 0:
                elapsed = time.monotonic() - start
                print(f"    it{it}: obj={best_obj:.8f}, t={best_t:.4f}, "
                      f"lp={lp_time:.2f}s, elapsed={elapsed:.1f}s")
        else:
            stall += 1
            if stall >= 3:
                # Perturbation: multiplicative noise preserving symmetry
                if stall <= 6:
                    freq = rng.integers(1, max(2, half // 40))
                    phase = rng.uniform(0, 2 * np.pi)
                    idx = np.arange(half, dtype=np.float64)
                    pert = 0.02 * stall * np.cos(2 * np.pi * freq * idx / half + phase)
                    h = a[:half] * np.exp(pert)
                    h = np.maximum(h, 0)
                    a_new = np.concatenate([h, h[::-1]])
                    sn = float(np.sum(a_new))
                    if sn > 0.01:
                        a = a_new / sn
                    else:
                        a = best_a.copy()
                else:
                    # Bigger restart: blend with diverse init
                    mode = it % 5
                    a_div = _diverse_init(n, rng, mode=mode)
                    a = 0.25 * a_div + 0.75 * best_a
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                stall = 0
        it += 1

    # Final symmetrize and normalize
    a = (best_a + best_a[::-1]) / 2.0
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    return a, _obj(a)


def _subgradient_polish(a, time_budget=20.0):
    """Fast subgradient descent as final polish."""
    n = len(a)
    a = (a + a[::-1]) / 2.0
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = _obj(a)
    start = time.monotonic()
    a_avg = a.copy()
    momentum = np.zeros(n)
    beta = 0.6
    it = 0

    while time.monotonic() - start < time_budget:
        conv = _fft_conv(a_avg)
        k_max = int(np.argmax(conv))
        # Subgradient of max(conv) w.r.t. a
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_max - i
            j2 = i - k_max
            if 0 <= j1 < n:
                grad[i] += a_avg[j1]
            if 0 <= j2 < n:
                grad[i] += a_avg[j2]
        lr = 0.03 / (1.0 + 0.003 * it)
        momentum = beta * momentum - lr * grad
        a_new = a_avg + momentum
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn > 1e-10:
            a_new = a_new / sn
        a_avg = (it / (it + 1.0)) * a_avg + (1.0 / (it + 1.0)) * a_new
        obj = _obj(a_avg)
        if obj < best_obj:
            best_obj = obj
            best_a = a_avg.copy()
        it += 1
        if it % 500 == 0 and it - 0 > 3000:
            break

    best_a = (best_a + best_a[::-1]) / 2.0
    best_a = np.maximum(best_a, 0)
    best_a = best_a / best_a.sum()
    return best_a, _obj(best_a)


def _try_n_value(n, start_seq, time_budget, verbose=False):
    """Try a specific n value with LP polish."""
    if n < 100 or n > 3000 or n % 2 != 0:
        return None, np.inf
    a = _interpolate(start_seq, n)
    if a is None:
        return None, np.inf
    # Also try diverse inits
    rng = np.random.default_rng(777 + n)
    best = a.copy()
    best_obj = _obj(a)
    for mode in range(3):
        alt = _diverse_init(n, rng, mode=mode)
        obj_alt = _obj(alt)
        if obj_alt < best_obj:
            best = alt
            best_obj = obj_alt
    a_out, obj_out = _lp_polish(best, time_budget=time_budget, max_iters=60, verbose=False)
    if obj_out < best_obj:
        best = a_out
        best_obj = obj_out
    return best, best_obj

    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s > 0.01:
            best_seq = parent / s
            # Symmetrize
            best_seq = (best_seq + best_seq[::-1]) / 2.0
            best_seq = np.maximum(best_seq, 0)
            best_seq = best_seq / best_seq.sum()
            best_score = _obj(best_seq)
            print(f"Parent loaded: n={parent.size}, obj={best_score:.8f}")
        else:
            raise ValueError("Sum too small")
    except Exception:
        n0 = 800
        best_seq = _u_init(n0, sharp=4.0)
        best_score = _obj(best_seq)
        print(f"Default init: n={n0}, obj={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")

    # === Phase 1: Intensive LP on current n ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.35, 400.0)
        print(f"\n=== Phase 1: LP Polish n={n_best} budget={alloc:.0f}s ===")
        a1, obj1 = _lp_polish(best_seq, time_budget=alloc, max_iters=300, verbose=True)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj1:.8f} (no improvement)")

    # === Phase 2: Subgradient polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.05, 30.0)
        print(f"\n=== Phase 2: Subgradient (budget={alloc:.0f}s) ===")
        a2, obj2 = _subgradient_polish(best_seq, time_budget=alloc)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj2:.8f}")

    # === Phase 3: LP again after subgradient ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.15, 120.0)
        print(f"\n=== Phase 3: LP Round 2 (budget={alloc:.0f}s) ===")
        a3, obj3 = _lp_polish(best_seq, time_budget=alloc, max_iters=100, verbose=True)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj3:.8f}")

    # === Phase 4: n Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 4: n Exploration ===")
        n_cur = len(best_seq)
        # Candidate n values (must be even for symmetry)
        n_cands = []
        for delta in [-20, -10, -4, -2, 2, 4, 10, 20, 50, 100, 200, 300, 400]:
            n_try = n_cur + delta
            if 200 <= n_try <= 2500:
                n_cands.append(n_try)
        # Add some fixed values
        for nv in [600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600, 1800, 2000]:
            if 200 <= nv <= 2500 and nv % 2 == 0 and nv not in n_cands:
                n_cands.append(nv)
        # Deduplicate and sort by distance from current
        n_cands = sorted(set(n for n in n_cands if n % 2 == 0))
        n_cands.sort(key=lambda x: abs(x - n_cur))
        n_cands = n_cands[:10]
        print(f"  Trying n: {n_cands}")

        for n_try in n_cands:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            alloc = min(remaining * 0.25, 60.0)

            if a_try is not None:
                if obj_try < best_score:
                    best_score = obj_try
                    best_seq = a_try
                    print(f"  ** n={n_try}: obj={best_score:.8f} **")
                else:
                    print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts at best n ===
    remaining = deadline - time.monotonic()
    if remaining > 50:

        print(f"\n=== Phase 5: Diverse Restarts (n={n_cur}) ===")
        for r in range(5):

            mode = r % 5

            if obj_r < best_score:

                best_seq = a_r2

    remaining = deadline - time.monotonic()

    if remaining > 25:

        a_sg, obj_sg = _subgradient_polish(best_seq, time_budget=min(remaining * 0.3, 15.0))
        if obj_sg < best_score:

            best_seq = a_sg
            print(f"  SG: obj={best_score:.8f} **")
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:

    return result
