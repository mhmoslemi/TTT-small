def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lp_solve(a, peak_lags=None, peak_reduction=0.0):
    """
    Solve min M s.t. (a*g)_k <= M for all k, g >= 0, sum(g) = 1.
    Optionally, for lags in peak_lags, require (a*g)_k <= M*(1-peak_reduction).
    Returns (g, M).
    """
    n = len(a)
    # Build constraint matrix A where row k represents (a*g)_k
    # (a*g)_k = sum_i a_{(k-i) mod ...} ... 
    # Actually (a*g)_k = sum_{i=0}^{n-1} a_i * g_{k-i} where g_j = 0 if j notin [0,n-1]
    # So (a*g)_k = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a_{k-i} * g_i
    #           = sum_{i=0}^{n-1} a[(k-i) % (2n-1) if 0 <= k-i < n else 0] * g_i
    
    # More carefully: (a*g)_k = sum_i a_i * g_{k-i} for 0 <= i <= n-1, 0 <= k-i <= n-1
    # So for each k, the valid i range is max(0, k-n+1) <= i <= min(k, n-1)
    
    # Instead of building the full matrix, let's use a sparse representation
    # A[k, i] = a[k-i] if 0 <= k-i <= n-1, else 0
    
    # For n=800, the matrix is (1599, 800). Let's build it.
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            A[k, i] = a[k - i]
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(g_var) == 1]
    
    if peak_lags is not None and peak_reduction > 0:
        # For peak lags, tighter constraint
        # (a*g)_k <= M * (1 - peak_reduction)
        # This is not linear! We need M * peak_reduction + (a*g)_k <= M
        # => (a*g)_k - M * peak_reduction <= 0
        # Still linear!
        for k in range(2 * n - 1):
            if k in peak_lags:
                constraints.append(A[k] @ g_var - peak_reduction * M_var <= 0)
            else:
                constraints.append(A[k] @ g_var <= M_var)
    else:
        constraints.append(A @ g_var <= M_var)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)

            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None

def dpss_init(n, W):
    """
    Initialize with a Discrete Prolate Spherical Sequence (DPSS).
    These have the best concentration in both time and frequency,
    leading to flat autocorrelation.
    """
    # DPSS is the eigenvector of the bandwidth-limiting matrix
    # For a sequence of length n with bandwidth W (in cycles/sample)
    m = int(W * n)
    if m < 1:
        m = 1
    if m > n // 2:
        m = n // 2
    
    # Build the bandwidth-limiting matrix
    indices = np.arange(n)
    T = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            d = i - j
            if d == 0:
                T[i, j] = n
            else:
                T[i, j] = np.sin(2 * np.pi * W * d) / (np.pi * d)
    
    # Use scipy for eigendecomposition
    from scipy.linalg import eigh
    # T is symmetric
    eigenvalues, eigenvectors = eigh(T)
    # Largest eigenvalue corresponds to best concentration
    v = eigenvectors[:, -1]
    v = np.abs(v)  # Take absolute value for non-negativity
    v = np.maximum(v, 1e-10)
    return v / v.sum()


def dpss_init_ensemble(n, W, num_comps=3):
    """Initialize using sum of multiple DPSS components."""
    result = np.zeros(n)
    for c in range(num_comps):
        W_c = W * (c + 1) / num_comps
        v = dpss_init(n, W_c)
        result += v
    result = np.maximum(result, 0)
    return result / result.sum()


def sinc_init(n, beta=1.0):
    """Initialize with a sinc-like window (truncated and windowed)."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    t = (x - center) / (n * beta / (2 * np.pi))
    a = np.sinc(t / np.pi)  # sinc(x) = sin(pi*x)/(pi*x)
    # Apply a smooth window to avoid Gibbs
    w = 0.54 - 0.46 * np.cos(2 * np.pi * x / (n - 1))
    a = np.maximum(a * w, 1e-6)
    return a / a.sum()

def smoothed_grad(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    shifted = tau * (conv - exact_max)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 2 * n + len(conv):
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def nesterov_gd(a, time_budget, tau_levels=None, verbose=False, beta=0.9, lr_scale=3.0):
    """Nesterov accelerated gradient descent with adaptive tau."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [5, 15, 40, 100, 250, 600, 1500, 3000]

    for tau in tau_levels:
        lr = lr_scale / np.sqrt(n)
        v = np.zeros(n)
        tau_iters = 0
        max_iters_per_level = 3000

        while time.monotonic() - start < time_budget:
            if tau_iters >= max_iters_per_level:
                break

            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
                v = np.zeros(n)
                lr *= 0.5
            else:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new
            tau_iters += 1
            lr = max(1e-7, lr * 0.99999)

        if verbose:
            print(f"    tau={tau:5d}: obj={best_obj:.10f}")

    return best_a


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses DPSS initialization, aggressive LP fixed-point with anti-peak targeting,
Nesterov GD, coordinate local search, and adaptive n exploration.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.linalg import eigh
import scipy.sparse as sp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lp_solve(a, peak_lags=None, peak_reduction=0.0):
    """
    Solve min M s.t. (a*g)_k <= M for all k, g >= 0, sum(g) = 1.
    If peak_lags given, tighten those constraints by peak_reduction fraction.
    Returns (g, M).
    """
    n = len(a)
    # Build constraint matrix
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        if i_min <= i_max:
            A[k, i_min:i_max+1] = a[k - i_min:k - i_max + 1]

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(g_var) == 1]

    if peak_lags is not None and peak_reduction > 0:
        peak_set = set(peak_lags)
        for k in range(2 * n - 1):
            if k in peak_set:
                constraints.append(A[k] @ g_var <= (1.0 - peak_reduction) * M_var)
            else:
                constraints.append(A[k] @ g_var <= M_var)
    else:
        constraints.append(A @ g_var <= M_var)

    prob = cp.Problem(cp.Minimize(M_var), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None


def dpss_init(n, W):
    """Initialize with DPSS (Discrete Prolate Spherical Sequence)."""
    if W * n < 1:
        W = 1.0 / n
    if W >= 0.5:
        W = 0.45
    # Build bandwidth-limiting matrix T where T[i,j] = sinc(2*W*(i-j))
    indices = np.arange(n)
    d = indices[:, None] - indices[None, :]
    T = np.sinc(2 * W * d)  # np.sinc(x) = sin(pi*x)/(pi*x)
    T = np.abs(T)  # Ensure positive
    eigenvalues, eigenvectors = eigh(T)
    v = np.abs(eigenvectors[:, -1])
    v = np.maximum(v, 1e-10)
    return v / v.sum()


def dpss_ensemble(n, W_list, weights=None):
    """Sum of multiple DPSS components."""
    result = np.zeros(n)
    for i, W in enumerate(W_list):
        w = 1.0
        if weights is not None:
            w = weights[i]
        v = dpss_init(n, W)
        result += w * v
    result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return result / s


def sinc_window(n, beta=1.0):

    a = np.maximum(a * w, 1e-6)

            best_a = best_local_a.copy()
            a = best_a
            t_current = max(0.02, min(1.0, best_local_t * 0.85))
            stall_count = 0
            if verbose and it % 50 == 0:
                print(f"    LP it{it:4d}: obj={best_obj:.10f}, t={best_local_t:.3f}, M={M_lp:.6f}")
        else:
            stall_count += 1
            # Perturbation strategies based on stall pattern
            strategy = stall_count % 6
            if strategy == 1:
                # Gaussian perturbation
                a = a + 0.006 * rng.standard_normal(n)
            elif strategy == 2:
                # Targeted subgradient step at peaks
                for lag in peak_lags[:4]:
                    grad = np.zeros(n)
                    i_min = max(0, lag - n + 1)
                    i_max = min(lag, n - 1)
                    for i in range(i_min, i_max + 1):
                        grad[i] = 2.0 * a[lag - i]
                    gn = np.linalg.norm(grad)
                    if gn > 1e-15:
                        a = a - 0.003 * grad / gn
            elif strategy == 3:
                # Low-frequency perturbation
                x = np.arange(n)
                f = rng.integers(1, 6)
                p = rng.uniform(0, 2 * np.pi)
                a = a + 0.012 * np.sin(2 * np.pi * f * x / n + p)
            elif strategy == 4:
                # Smooth (low-pass) perturbation
                ksz = max(5, n // 40)
                kernel = np.ones(ksz) / ksz
                smoothed = np.convolve(a, kernel, mode='same')
                a = 0.8 * a + 0.2 * smoothed
            elif strategy == 5:
                # Antisymmetric perturbation (breaks palindrome)
                ramp = np.linspace(-1, 1, n)
                a = a + 0.008 * ramp * a

            a = np.maximum(a, 0)
            sa = float(np.sum(a))
            if sa > 1e-10:
                a = a / sa
            else:
                a = best_a.copy()

            if stall_count > 60:
                # Major restart
                a = best_a + 0.04 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_count = 0
                t_current = 0.08

    return best_a


def smoothed_grad(a, tau):
    """Smoothed gradient of max conv via log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    exact_max = float(np.max(conv))
    shifted = tau * (conv - exact_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_vals / sum_exp
    N = 1
    while N < 4 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    cross = ifft(P * np.conj(A)).real
    g = 2.0 * cross[:n]
    return g, exact_max


def nesterov_gd(a, time_budget, tau_levels=None, verbose=False, beta=0.9, lr_scale=3.0):
    """Nesterov accelerated gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [5, 15, 40, 100, 250, 600, 1500]

    for tau in tau_levels:
        lr = lr_scale / np.sqrt(n)
        v = np.zeros(n)
        tau_iters = 0
        max_per_level = 2000

        while time.monotonic() - start < time_budget:
            if tau_iters >= max_per_level:
                break
            y = a + beta * v
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy < 1e-10:
                y = best_a.copy()
                v = np.zeros(n)
                sy = 1.0
            y = y / sy

            g, _ = smoothed_grad(y, tau)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-15:
                break

            v = beta * v - lr * g
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s
