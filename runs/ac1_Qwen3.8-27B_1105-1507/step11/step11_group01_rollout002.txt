def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns (g, M, dual_vars)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:
                    g_val = g_val / g_sum
                    # Get dual variables
                    dual = None
                    try:
                        if constraints[0].dual_value is not None:
                            dual = np.array(constraints[0].dual_value)
                    except:
                        pass
                    return g_val, float(prob.value), dual
        except Exception:
            continue
    return None, None, None

def peak_shaving(a, time_budget=100.0, verbose=False):
    """
    Peak shaving: iteratively reduce the coordinates that contribute most
    to the peak convolution positions.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Find peak positions
        peak_threshold = max_conv * 0.95
        peak_positions = np.where(conv >= peak_threshold)[0]
        
        # For each peak position k, identify which a[i] contribute most
        # and reduce them
        for k in peak_positions:
            # conv[k] = sum_i a[i] * a[k-i]
            # The gradient w.r.t. a[i] is a[k-i] (for valid k-i)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo > hi:
                continue
            
            # Reduce a[i] where a[i] * a[k-i] is large
            contrib = a[lo:hi+1] * a[k-hi:k-lo+1]
            # Reduce the ones with highest contribution
            if np.max(contrib) > 1e-10:
                # Proportional reduction
                reduction = contrib / np.sum(contrib) * 0.01
                a[lo:hi+1] -= reduction * a[lo:hi+1]
                # Also reduce a[k-hi:k-lo+1]
                a[k-hi:k-lo+1] -= reduction * a[k-hi:k-lo+1]
        
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s < 1e-10:
            a = best_a.copy()
            s = 1.0
        else:
            a = a / s
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a

                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = obj_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = obj_1d(d_pt)
            
            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            
            if abs(x_opt - v) < 1e-14:
                continue
            
            delta = x_opt - v
            a[i] = x_opt
            S += delta
            
            # Update conv
            if len(a_cross) > 0:
                conv[ks[cross_mask]] += 2.0 * delta * a_cross
            if 0 <= 2*i <= 2*n-2:
                conv[2*i] += x_opt**2 - v**2
            
            sweep_improved = True
            
            obj_now = 2.0 * n * float(np.max(conv)) / (S * S)
            if obj_now < best_obj:
                best_obj = obj_now
                best_a = a.copy()
        
        # Check for stall
        if best_obj < prev_best - 1e-10:
            prev_best = best_obj
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break
        
        if verbose:
            elapsed = time.monotonic() - start
            print(f"    CD: obj={best_obj:.8f}, t={elapsed:.1f}s, stall={stall}")
    
    return best_a

def simulated_annealing_escape(a, time_budget=100.0, seed=42):
    """
    Simulated annealing: accept worse solutions with decreasing probability.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    current_obj = best_obj
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    T = 0.01
    T_min = 1e-6
    cooling = 0.999
    
    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        if T < T_min:
            break
        
        # Propose a perturbation
        pert_type = rng.integers(0, 3)
        a_try = a.copy()
        
        if pert_type == 0:
            # Single coordinate change
            i = rng.integers(0, n)
            a_try[i] = max(0, a_try[i] + rng.normal(0, 0.02))
        elif pert_type == 1:
            # Pair swap
            i, j = rng.integers(0, n, 2)
            swap = min(a_try[i], a_try[j]) * 0.1 * rng.uniform(-1, 1)
            a_try[i] = max(0, a_try[i] + swap)
            a_try[j] = max(0, a_try[j] - swap)
        else:
            # Block perturbation
            b = rng.integers(0, n - 5)
            a_try[b:b+5] += rng.normal(0, 0.01, 5)
            a_try = np.maximum(a_try, 0)
        
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            T *= cooling
            continue
        a_try = a_try / s_try
        
        obj_try = compute_obj(a_try)
        delta = obj_try - current_obj
        
        if delta < 0 or rng.random() < math.exp(-delta / T):
            a = a_try
            current_obj = obj_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
        
        T *= cooling
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via spline interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Sine-modulated
        x = np.arange(n) / n
        a = 1 + 0.5 * np.sin(2 * np.pi * rng.uniform(1, 5) * x + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 3:
        # Cosine bell
        x = np.linspace(-1, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
        a = a / a.sum()
    elif mode == 4:
        # Polynomial
        x = np.linspace(0, 1, n)
        p = rng.uniform(-2, 2, size=3)
        a = 1 + p[0]*x + p[1]*x**2 + p[2]*x**3
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Two peaks
        a = np.zeros(n)
        c1, c2 = n//3, 2*n//3
        w = n//10
        a += np.exp(-((np.arange(n)-c1)**2)/(2*w**2))
        a += np.exp(-((np.arange(n)-c2)**2)/(2*w**2))
        a = a / a.sum()
    elif mode == 6:
        # Arcsine
        x = np.linspace(0.001, 0.999, n)
        a = 1.0 / np.sqrt(x * (1-x))
        a = a / a.sum()
    elif mode == 7:
        # Random walk (clipped)
        a = np.cumsum(rng.normal(0, 1, n))
        a = a - a.min() + 0.1
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

def flatness_aware_lp(a, time_budget=100.0, verbose=False):
    """
    LP that aims for flat convolution: minimize M + lambda * variance(conv).
    Approximated by minimizing M with secondary objective of making conv flat.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a_norm = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(100):
        if time.monotonic() - start > time_budget:
            break
        
        # Solve LP for g that minimizes peak of a*g
        A = build_conv_matrix(a_norm)
        
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        
        # Add flatness penalty: minimize M + lambda * sum((A@g - mean)^2)
        # This makes it a QP instead of LP. Let's stick with LP but add
        # constraints that push towards flatness.
        
        # Alternative: minimize M with tighter bound on second-highest
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        solved = False
        for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    g_sum = float(np.sum(g_val))
                    if g_sum > 1e-10:
                        g_val = g_val / g_sum
                        solved = True
                        break
            except:
                continue
        
        if not solved:
            break
        
        # Blend
        t = max(0.05, 0.3 * (1 - it / 100))
        a_new = (1-t) * a_norm + t * g_val
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a_norm = a_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
    
    return best_a
