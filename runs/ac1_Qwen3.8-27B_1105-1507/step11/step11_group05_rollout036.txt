import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1] if np.any(cond) else 0
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n

def lse_grad_fft(a, T):
    """Compute gradient of 2n * LSE_T(conv(a,a)) using FFT. Returns gradient (before 2n factor)."""
    n = len(a)
    conv = compute_conv(a)
    cmax = conv.max()
    exp_vals = np.exp(T * (conv - cmax))
    w = exp_vals / exp_vals.sum()
    
    # Size for FFT: need at least n + (2n-1) - 1 = 3n-2
    L = 1
    while L < 3 * n - 1:
        L <<= 1
    
    fa = rfft(a, L)
    fw = rfft(w, L)
    
    # T2[i] = (a * w)[i] for i in [0, n-1]
    t2 = irfft(fa * fw, L)[:n]
    
    # T1[i] = (reverse(a) * w)[n-1+i] for i in [0, n-1]
    fa_rev = rfft(a[::-1], L)
    t1_full = irfft(fa_rev * fw, L)
    t1 = t1_full[n-1 : n-1+n]
    
    return 2.0 * n * (t1 + t2)

def lse_value(a, T):
    n = len(a)
    conv = compute_conv(a)
    cmax = conv.max()
    exp_vals = np.exp(T * (conv - cmax))
    lse = cmax + np.log(exp_vals.sum()) / T
    return 2.0 * n * lse

def lse_gd_fast(a, time_budget, T_start=10.0, T_end=100.0, verbose=False):
    """FFT-accelerated LSE gradient descent with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr_base = 2.0 / (n + 1)
    momentum = np.zeros(n)
    beta = 0.9
    it = 0
    stall = 0
    max_stall = 200
    
    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** frac
        
        # Nesterov lookahead
        y = a + beta * momentum
        grad = lse_grad_fft(y, T)
        
        # Backtracking
        obj_y = lse_value(y, T)
        t = lr_base
        improved = False
        for _ in range(15):
            a_try = y - t * grad
            a_try = project_simplex(a_try)
            obj_try = lse_value(a_try, T)
            if obj_try < obj_y - 1e-12:
                momentum = a_try - a
                a = a_try
                improved = True
                break
            t *= 0.5
        
        if not improved:

            grad_a = lse_grad_fft(a, T)
            obj_a = lse_value(a, T)
            t = lr_base

                a_try = a - t * grad_a
                a_try = project_simplex(a_try)
                obj_try = lse_value(a_try, T)
                if obj_try < obj_a:
                    a = a_try
                    momentum = np.zeros(n)
                    improved = True
                    break
                t *= 0.5
        
        if not improved:
            stall += 1
            if stall > max_stall:
                break
        else:
            stall = 0
        
        obj_true = compute_obj(a)
        if obj_true < best_obj:
            best_obj = obj_true
            best_a = a.copy()
        
        it += 1
        if verbose and it % 200 == 0:
            print(f"    LSE-GD iter {it}: obj={best_obj:.8f}, T={T:.1f}, t={time.monotonic()-start:.1f}s")

def full_lp_solve(a, solver_list=None):
    """Solve full LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / float(np.sum(a))
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    # Build constraint matrix: (a*g)[k] <= M for k=0..2n-2
    # (a*g)[k] = sum_j a[j] * g[k-j] for valid j
    # = sum_m A[k,m] * g[m] where A[k,m] = a[k-m] if 0<=k-m<n else 0
    
    # This gives a matrix of size (2n-1) x n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                A[k, m] = a[j]
    
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:
        solver_list = [cp.CLS, cp.CLARABEL, cp.ECOS, cp.SCS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, np.inf

def full_lp_polish(a, time_budget, max_iters=30, verbose=False):
    """Iterative LP polishing with line search."""
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start >= time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        g_lp, m_lp = full_lp_solve(best_a)
        if g_lp is None:
            break
        
        # Line search over blend parameter
        best_t = 0.0
        best_obj_ls = best_obj
        for t in np.linspace(0, 1, 21):
            a_try = (1 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t
        
        if best_t > 0 and best_obj_ls < best_obj - 1e-12:
            a_new = (1 - best_t) * best_a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                stall = 0
                if verbose:
                    print(f"    LP iter {it}: obj={best_obj:.8f}, t={best_t:.3f}, M_lp={m_lp:.6f}")
            else:
                stall += 1
        else:
            stall += 1
            if verbose:
                print(f"    LP iter {it}: no improvement, M_lp={m_lp:.6f}")
        
        if stall >= 5:
            # Perturb
            rng = np.random.default_rng(it * 31 + 7)
            noise = rng.normal(0, 0.002 * stall, len(best_a))
            a_pert = best_a + noise
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            best_a = a_pert
            best_obj = compute_obj(best_a)
            stall = 0
    
    return best_a

def subgradient_fast(a, time_budget, top_k=3, verbose=False):
    """Subgradient descent using top-k peaks, with momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 1.0 / n
    momentum = np.zeros(n)
    beta = 0.8
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:top_k]
        peak_vals = conv[top_idx]
        weights = peak_vals / peak_vals.sum()
        
        # Compute gradient as weighted sum of subgradients
        grad = np.zeros(n)
        for ki, w in zip(top_idx, weights):
            ki = int(ki)
            # a[ki - i] for valid i
            lo1 = max(0, ki - n + 1)
            hi1 = min(n - 1, ki)
            if lo1 <= hi1:
                grad[lo1:hi1+1] += w * a[ki - hi1 : ki - lo1 + 1][::-1]
            lo2 = max(0, ki)
            hi2 = min(n - 1, ki + n - 1)
            if lo2 <= hi2:
                grad[lo2:hi2+1] += w * a[lo2 - ki : hi2 - ki + 1]
        
        # Nesterov step
        y = a + beta * momentum
        # Project y - lr*grad
        a_try = project_simplex(y - lr * grad)
        obj_try = compute_obj(a_try)
        
        if obj_try < best_obj:
            momentum = a_try - a
            a = a_try
            best_obj = obj_try
            best_a = a.copy()
            stall = 0
        else:
            lr *= 0.9
            momentum = 0.5 * momentum
            stall += 1
            if stall > 500:
                break
    
    return best_a

def coordinate_descent_fast(a, time_budget, verbose=False):
    """Coordinate descent with exact 1D optimization per coordinate."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Precompute: for each coordinate i, the "rest" of the convolution
    # (a*a)[k] = a[i]*a[k-i] + a[i]*a[i-k] + rest_k (where rest doesn't involve a[i])
    
    for sweep in range(20):
        if time.monotonic() - start >= time_budget:
            break
        
        improved_sweep = False
        for i in range(n):
            if time.monotonic() - start >= time_budget:
                break
            
            ai = a[i]
            if ai < 1e-12:
                continue
            
            # Binary search for optimal a[i] given others fixed
            # f(x) = max_k [x*(a[k-i] + a[i-k] if valid) + rest_k] / (sum_rest + x)^2 * 2n
            # where rest_k = (a*a)[k] - a[i]*(a[k-i] + a[i-k])
            
            # Simple approach: try a few candidate values
            candidates = [0.0, ai * 0.1, ai * 0.3, ai * 0.5, ai * 0.7, ai, ai * 1.3, ai * 1.6, ai * 2.0, ai * 3.0]
            # Also try the value that would make the peak at the argmax equal to the second peak
            best_val = ai
            best_local = best_obj
            
            for cv in candidates:
                a[i] = cv
                s_new = a.sum()
                if s_new < 0.01:
                    a[i] = ai
                    continue
                a_norm = a / s_new
                obj_try = compute_obj(a_norm)
                if obj_try < best_local:
                    best_local = obj_try
                    best_val = cv
                a[i] = ai  # restore
        
        if best_val != ai:
            a[i] = best_val
            improved_sweep = True
        
        obj_now = compute_obj(a)
        if obj_now < best_obj:
            best_obj = obj_now
            best_a = a.copy()
        
        if not improved_sweep:
            break
    
    return best_a

"""
FFT-accelerated hybrid optimizer for step function coefficient minimization.
Key innovations over previous approach:
1. O(n log n) gradient computation via FFT (80x faster than O(n^2) loops)
2. Full LP polishing (all 2n-1 constraints, not truncated)
3. Exponentiated gradient (mirror descent) as alternate update
4. Adaptive multi-peak subgradient with dynamic top-k
5. Better time allocation and multi-scale n exploration

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cv - 1)
    rho_val = np.nonzero(cond)[0][-1] if np.any(cond) else 0
    theta = (cv[rho_val] - 1.0) / (rho_val + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n


def lse_gradient_fft(a, T):
    """
    Compute gradient of 2n * LSE_T(conv(a,a)) using FFT.
    Returns gradient vector of length n.
    Uses: d(conv[k])/da[i] = 2*a[k-i]*1[0<=k-i<n]
    So grad[i] = 4n * sum_j a[j]*w[i+j] = 4n * (reverse(a) * w)[n-1+i]
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = conv.max()
    # Clip to avoid overflow
    scaled = np.minimum(T * (conv - cmax), 50.0)
    exp_vals = np.exp(scaled)
    w = exp_vals / exp_vals.sum()

    # FFT size: convolution of reverse(a) (length n) and w (length 2n-1)
    # Result length: n + (2n-1) - 1 = 3n-2
    L = 1
    while L < 3 * n - 1:
        L <<= 1

    fa = rfft(a[::-1].copy(), L)
    fw = rfft(w, L)
    t_full = irfft(fa * fw, L)

    # grad_corr[i] = (reverse(a) * w)[n-1+i] for i in [0, n-1]
    grad_corr = t_full[n - 1: 2 * n - 1]

    return 4.0 * n * grad_corr


def lse_value(a, T):
    """Compute 2n * LSE_T(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    cmax = conv.max()
    scaled = np.minimum(T * (conv - cmax), 50.0)
    lse = cmax + np.log(np.sum(np.exp(scaled))) / T
    return 2.0 * n * lse


def lse_gd_fast(a, time_budget, T_start=5.0, T_end=100.0, verbose=False):
    """
    FFT-accelerated LSE gradient descent with Nesterov momentum and backtracking.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / (n + 10)
    momentum = np.zeros(n)
    beta = 0.85
    it = 0
    stall = 0
    max_stall = 300
    best_lr = lr

    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** frac

        # Nesterov lookahead
        y = a + beta * momentum
        grad_y = lse_gradient_fft(y, T)
        obj_y = lse_value(y, T)

        # Backtracking on y
        t = lr
        improved = False
        a_try = None
        for _ in range(12):
            a_try = y - t * grad_y
            a_try = project_simplex(a_try)
            obj_try = lse_value(a_try, T)
            if obj_try < obj_y - 1e-13:
                improved = True
                break
            t *= 0.5

        if improved:
            momentum = a_try - a
            a = a_try
            stall = 0
            # Slightly increase lr
            lr = min(t * 1.05, 10.0 / n)
        else:
            # Try from a directly
            grad_a = lse_gradient_fft(a, T)
            obj_a = lse_value(a, T)
            t = lr * 0.5
            for _ in range(8):
                a_try = a - t * grad_a
                a_try = project_simplex(a_try)
                obj_try = lse_value(a_try, T)
                if obj_try < obj_a - 1e-13:
                    a = a_try
                    momentum = np.zeros(n)
                    improved = True
                    lr = t
                    stall = 0
                    break
                t *= 0.5

            if not improved:
                stall += 1
                momentum *= 0.5
                lr *= 0.8
                if stall > max_stall:
                    break

        obj_true = compute_obj(a)
        if obj_true < best_obj:
            best_obj = obj_true
            best_a = a.copy()

        it += 1
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    LSE-GD it{it}: obj={best_obj:.8f}, T={T:.1f}, lr={lr:.6f}, t={elapsed:.1f}s")

    return best_a


def exp_gradient_descent(a, time_budget, T_start=5.0, T_end=80.0, verbose=False):
    """
    Exponentiated gradient (mirror descent with neg-entropy).
    a[i] <- a[i] * exp(-lr * grad[i]) / Z
    Automatically maintains a >= 0 and sum(a) = 1.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 1.0 / n
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** frac

        grad = lse_gradient_fft(a, T)
        # Center the gradient (remove mean for stability)
        grad_centered = grad - grad.mean()

        # Exponentiated update
        log_a = np.log(a)
        log_a_new = log_a - lr * grad_centered
        # Shift for numerical stability
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.02, 5.0 / n)
        else:
            stall += 1
            lr *= 0.9
            if stall > 200:
                break

        it += 1
        if verbose and it % 200 == 0:
            print(f"    ExpGD it{it}: obj={best_obj:.8f}, T={T:.1f}, lr={lr:.6f}")

    return best_a


def full_lp_solve(a):
    """
    Solve full LP: min M s.t. (a*g)[k] <= M for all k=0..2n-2, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, inf).
    """
    n = len(a)
    a = a / float(np.sum(a))

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()

    # Build constraint matrix A of size (2n-1) x n
    # A[k, m] = a[k-m] if 0 <= k-m < n, else 0
    A = np.zeros((2 * n - 1, n))
    for m in range(n):
        j_start = 0
        j_end = min(n - 1, 2 * n - 2 - m)
        k_start = m
        A[k_start:k_start + (j_end - j_start + 1), m] = a[j_start:j_end + 1]

    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solver_list = [cp.CLARABEL, cp.ECOS, cp.SCS, cp.CVXOPT]
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, np.inf


def full_lp_polish(a, time_budget, max_iters=40, verbose=False):
    """
    Iterative full LP polishing with line search and perturbation.
    """
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, m_lp = full_lp_solve(best_a)
        if g_lp is None:
            break

        # Line search over blend parameter
        best_t = 0.0
        best_obj_ls = best_obj
        # Coarse search
        for t in np.linspace(0, 1, 11):
            a_try = (1 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t

        # Fine search around best_t
        if best_t > 0:
            for t_fine in np.linspace(max(0, best_t - 0.1), min(1, best_t + 0.1), 11):
                a_try = (1 - t_fine) * best_a + t_fine * g_lp
                a_try = np.maximum(a_try, 0)
                s = float(np.sum(a_try))
                if s < 0.01:
                    continue
                a_try = a_try / s
                obj_try = compute_obj(a_try)
                if obj_try < best_obj_ls:
                    best_obj_ls = obj_try
                    best_t = t_fine

        if best_t > 0 and best_obj_ls < best_obj - 1e-14:
            a_new = (1 - best_t) * best_a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                stall = 0
                if verbose:
                    print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, M_lp={m_lp:.6f}")
            else:
                stall += 1
        else:
            stall += 1
            if verbose:
                print(f"    LP it{it}: no improvement, M_lp={m_lp:.6f}")

        if stall >= 4:
            # Perturb to escape local minimum
            rng_local = np.random.default_rng(it * 7919 + 13)
            noise_scale = 0.003 * stall
            noise = rng_local.normal(0, noise_scale, n)
            a_pert = best_a + noise
            a_pert = np.maximum(a_pert, 0)
            s_p = float(np.sum(a_pert))
            if s_p > 0.01:
                best_a = a_pert / s_p
                best_obj = compute_obj(best_a)
            stall = 0

        it += 1

    return best_a


def multi_peak_subgradient(a, time_budget, top_k_start=3, top_k_end=1, verbose=False):
    """
    Subgradient descent using top-k peaks.
    top_k decreases over time (from exploration to exploitation).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 1.5 / n
    momentum = np.zeros(n)
    beta = 0.7
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        top_k = max(1, int(round(top_k_start + frac * (top_k_end - top_k_start))))

        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:top_k]
        peak_vals = conv[top_idx]
        weights = peak_vals / peak_vals.sum()

        # Compute gradient (using FFT-based cross-correlation for speed)
        # Build w
        w = np.zeros(2 * n - 1)
        for k in range(top_k):
            w[top_idx[k]] = weights[k]

        L = 1
        while L < 3 * n - 1:
            L <<= 1
        fa = rfft(a[::-1].copy(), L)
        fw = rfft(w, L)
        t_full = irfft(fa * fw, L)
        grad = 4.0 * n * t_full[n - 1: 2 * n - 1]

        # Nesterov step
        y = a + beta * momentum
        obj_y = compute_obj(y)

        t = lr
        improved = False
        for _ in range(10):
            a_try = y - t * grad
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < obj_y - 1e-13:
                momentum = a_try - a
                a = a_try
                improved = True
                break
            t *= 0.5

        if improved:
            stall = 0
            lr = min(t * 1.02, 3.0 / n)
        else:
            stall += 1
            lr *= 0.9
            momentum *= 0.5
            if stall > 400:
                break

        obj_true = compute_obj(a)
        if obj_true < best_obj:
            best_obj = obj_true
            best_a = a.copy()

        it += 1
        if verbose and it % 500 == 0:
            print(f"    SubGrad it{it}: obj={best_obj:.8f}, top_k={top_k}, lr={lr:.6f}")

    return best_a


def coordinate_descent(a, time_budget, verbose=False):
    """
    Coordinate descent: for each coordinate, try several scaled values.
    Uses early termination and adaptive candidate set.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for sweep in range(50):
        if time.monotonic() - start >= time_budget:
            break
        improved_sweep = False

        for i in range(n):
            if time.monotonic() - start >= time_budget:
                break

            ai = a[i]
            if ai < 1e-12:
                continue

            # Try scaled versions
            candidates = [0.0, ai * 0.2, ai * 0.5, ai * 0.8, ai * 1.2, ai * 1.5, ai * 2.0]
            best_val = ai
            best_local = best_obj

            for cv in candidates:
                a[i] = cv
                s_new = float(a.sum())
                if s_new < 0.01:
                    a[i] = ai
                    continue
                a_norm = a / s_new
                obj_try = compute_obj(a_norm)
                if obj_try < best_local:
                    best_local = obj_try
                    best_val = cv
                a[i] = ai

            if best_val != ai:
                a[i] = best_val
                improved_sweep = True

        obj_now = compute_obj(a)
        if obj_now < best_obj:
            best_obj = obj_now
            best_a = a.copy()
            if verbose:
                print(f"    CD sweep {sweep}: obj={best_obj:.8f}")

        if not improved_sweep:
            break

    return best_a


def make_init(n, mode, rng=None):
    """Various initialization modes."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        # U-shaped
        a = 1.0 / (0.05 + 3.0 * (2 * x - 1) ** 2)
    elif mode == 1:
        # Power law
        d = np.abs(x - 0.5)
        a = np.maximum(d, 1.0 / (2 * n)) ** (-2.0) + 0.01
    elif mode == 2:
        # Two Gaussian peaks
        a = np.exp(-((idx - n / 4.0) ** 2) / (2 * (n / 8.0) ** 2)) + \
            np.exp(-((idx - 3 * n / 4.0) ** 2) / (2 * (n / 8.0) ** 2)) + 0.01
    elif mode == 3:
        # Edge concentrated
        a = (2 * np.abs(x - 0.5)) ** 3 + 0.02
    elif mode == 4:
        # Arcsine-like
        eps = 1e-6
        a = 1.0 / np.sqrt(np.maximum(eps, 1 - (2 * x - 1) ** 2))
    elif mode == 5:
        # W-shaped (3 peaks)
        a = np.exp(-((idx - n * 0.1) ** 2) / (2 * (n / 12.0) ** 2)) + \
            np.exp(-((idx - n * 0.5) ** 2) / (2 * (n / 12.0) ** 2)) + \
            np.exp(-((idx - n * 0.9) ** 2) / (2 * (n / 12.0) ** 2)) + 0.01
    elif mode == 6:
        # Cosine window
        a = 1.0 - np.cos(2 * np.pi * x)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1500):
    """
    FFT-accelerated hybrid optimizer:
    1. FFT-based LSE gradient descent (O(n log n) per iteration)
    2. Full LP polishing (all constraints, no truncation)
    3. Exponentiated gradient descent (mirror descent)
    4. Multi-peak subgradient with adaptive top-k
    5. Coordinate descent for final refinement
    6. Multi-scale n exploration
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 12.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # === Load and evaluate initial candidates ===
    print("=== Initialization ===")

    # Try parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.all(np.isfinite(parent_arr)):
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"  Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        print("  Could not load parent")

    n_base = 800
    if best_seq is not None:
        n_base = len(best_seq)

    # Try multiple initializations
    for mode in range(7):
        a_init = make_init(n_base, mode)
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Init mode {mode} (n={n_base}): obj={obj_init:.8f} **")

    # Also try different n values for initialization
    for n_try in [600, 1000]:
        for mode in [0, 1, 3]:
            a_init = make_init(n_try, mode)
            obj_init = compute_obj(a_init)
            if obj_init < best_score:
                best_score = obj_init
                best_seq = a_init.copy()
                print(f"  Init mode {mode} (n={n_try}): obj={obj_init:.8f} **")

    n_cur = len(best_seq)
    print(f"\n  Best initial: n={n_cur}, obj={best_score:.8f}")

    # === Phase 1: Fast LSE GD (main workhorse) ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 1: FFT LSE-GD (budget={min(150, remaining*0.25):.0f}s) ===")
        alloc = min(150.0, remaining * 0.25)
        a1 = lse_gd_fast(best_seq, time_budget=alloc, T_start=5.0, T_end=80.0, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1 LSE-GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 LSE-GD: obj={obj1:.8f} (no improvement)")

    # === Phase 2: Full LP Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 2: Full LP Polish (budget={min(300, remaining*0.3):.0f}s) ===")
        alloc = min(300.0, remaining * 0.3)
        a2 = full_lp_polish(best_seq, time_budget=alloc, max_iters=50, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj2:.8f}")

    # === Phase 3: Exponentiated Gradient ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 3: Exponentiated GD ===")
        alloc = min(60.0, remaining * 0.12)
        a3 = exp_gradient_descent(best_seq, time_budget=alloc, T_start=10.0, T_end=80.0, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3 ExpGD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 ExpGD: obj={obj3:.8f}")

    # === Phase 4: Multi-peak subgradient ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 4: Multi-Peak Subgradient ===")
        alloc = min(80.0, remaining * 0.12)
        a4 = multi_peak_subgradient(best_seq, time_budget=alloc, top_k_start=5, top_k_end=1, verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4 SubGrad: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 SubGrad: obj={obj4:.8f}")

    # === Phase 5: LSE GD Round 2 (higher T for sharpness) ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 5: LSE-GD Round 2 (high T) ===")
        alloc = min(80.0, remaining * 0.12)
        a5 = lse_gd_fast(best_seq, time_budget=alloc, T_start=30.0, T_end=200.0, verbose=False)
        obj5 = compute_obj(a5)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  Phase5 LSE-GD2: obj={best_score:.8f} **")
        else:
            print(f"  Phase5 LSE-GD2: obj={obj5:.8f}")

    # === Phase 6: Full LP Round 2 ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 6: Full LP Round 2 ===")
        alloc = min(250.0, remaining * 0.25)
        a6 = full_lp_polish(best_seq, time_budget=alloc, max_iters=40, verbose=True)
        obj6 = compute_obj(a6)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Phase6 LP2: obj={best_score:.8f} **")
        else:
            print(f"  Phase6 LP2: obj={obj6:.8f}")

    # === Phase 7: n exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        print(f"\n=== Phase 7: n Exploration ===")
        n_cur = len(best_seq)
        n_options = []
        for delta in [-200, -100, 100, 200, 300, 500]:
            n_try = n_cur + delta
            if 300 <= n_try <= 2000 and n_try != n_cur:
                n_options.append(n_try)
        n_options = sorted(set(n_options))[:5]
        print(f"  Current n={n_cur}, trying: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 100:
                break

            # Interpolate
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = np.maximum(a_try, 0)
            s = float(a_try.sum())
            if s < 0.01:
                continue
            a_try = a_try / s

            # Quick LSE GD
            t1 = min(40.0, remaining * 0.3)
            a_try = lse_gd_fast(a_try, time_budget=t1, T_start=5.0, T_end=50.0)

            # Quick LP
            t2 = min(40.0, remaining * 0.25)
            if t2 > 10:
                a_try = full_lp_polish(a_try, time_budget=t2, max_iters=10)

            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Final LP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 8: Final LP Polish ===")
        alloc = min(remaining - 10, 100.0)
        a8 = full_lp_polish(best_seq, time_budget=alloc, max_iters=20, verbose=False)
        obj8 = compute_obj(a8)
        if obj8 < best_score:
            best_score = obj8
            best_seq = a8.copy()
            print(f"  Phase8 Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase8 Final LP: obj={obj8:.8f}")

    # === Phase 9: Coordinate descent ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 9: Coordinate Descent ===")
        alloc = min(remaining - 5, 25.0)
        a9 = coordinate_descent(best_seq, time_budget=alloc, verbose=False)
        obj9 = compute_obj(a9)
        if obj9 < best_score:
            best_score = obj9
            best_seq = a9.copy()
            print(f"  Phase9 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase9 CD: obj={obj9:.8f}")

    # === Phase 10: One final LP if time permits ===
    remaining = deadline - time.monotonic()
    if remaining > 25:
        a_fin = full_lp_polish(best_seq, time_budget=min(remaining - 3, 20.0), max_iters=5)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"\n  Final touch: obj={best_score:.8f} **")

    # === Validate and return ===
    if best_seq is None:
        best_seq = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"{'='*50}")
    return result
