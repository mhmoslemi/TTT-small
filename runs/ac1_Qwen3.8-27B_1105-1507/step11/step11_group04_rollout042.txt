Uses smoothed L-BFGS (log-sum-exp approximation of max) with annealing,
combined with gradient-aware coordinate descent, LP alternating minimization,
multi-restart exploration, and length variation.
Key innovation: softmax-parameterized smooth optimization with FFT-based gradients.
"""

import numpy as np
import time
import math
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj_full(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def logsumexp(x, beta=1.0):
    """Numerically stable log(sum(exp(beta*x))) / beta."""
    bx = beta * x
    m = np.max(bx)
    if m > 700:
        return m / beta + np.log(np.sum(np.exp(bx - m))) / beta
    return np.log(np.sum(np.exp(bx))) / beta


def softmax_obj_and_grad_z(z, beta=100.0):
    """
    Compute smoothed objective and gradient with respect to z,
    where a = softmax(z). Returns (obj, grad_z).
    """
    n = len(z)
    # Compute a = softmax(z)
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    a = exp_z / np.sum(exp_z)

    # Compute convolution
    conv = compute_conv(a)

    # Smoothed max
    bx = beta * conv
    bx_shifted = bx - np.max(bx)
    log_T = np.max(bx) + np.log(np.sum(np.exp(bx_shifted)))
    g_smooth = log_T / beta
    p = np.exp(bx_shifted) / np.sum(np.exp(bx_shifted))

    # h[j] = sum_k p[k] * a[k-j] for valid k-j in [0, n-1]
    h = np.zeros(n)
    for j in range(n):
        # k-j in [0, n-1] means k in [j, j+n-1], clipped to [0, 2n-2]
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        if k_end > k_start:
            h[j] = np.dot(a[:k_end - k_start], p[k_start:k_end])

    # grad_a[j] = 4n * (h[j] - g_smooth)  [since sum(a)=1]
    grad_a = 4.0 * n * (h - g_smooth)

    # Transform to z-space: grad_z[j] = a[j] * (grad_a[j] - a . grad_a)
    a_dot_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - a_dot_grad)

    obj = 2.0 * n * g_smooth
    return obj, grad_z


def smooth_lbfgs(a_init, time_budget, seed=42, verbose=True):
    """
    Smoothed L-BFGS optimization with annealing beta.
    Start from a_init (sum=1), optimize via softmax parameterization.
    """
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-15)
    a_init = a_init / np.sum(a_init)

    # Initial z from a_init
    z = np.log(a_init + 1e-15)

    best_a = a_init.copy()
    best_obj = compute_obj_norm(a_init)
    start = time.monotonic()

    # Anneal beta from 50 to 5000
    n_phases = 8
    betas = [50, 100, 200, 400, 800, 1600, 3200, 5000][:n_phases]
    per_phase_time = time_budget / n_phases

    for phase, beta in enumerate(betas):
        phase_start = time.monotonic()
        phase_deadline = phase_start + per_phase_time

        def obj_grad(z_val):
            if time.monotonic() > phase_deadline:
                return np.inf, np.zeros(len(z_val))
            o, g = softmax_obj_and_grad_z(z_val, beta=beta)
            return o, g

        try:
            res = minimize(obj_grad, z, method='L-BFGS-B',
                           jac=True, options={'maxiter': 200, 'maxfun': 500,
                                              'ftol': 1e-10, 'gtol': 1e-8,
                                              'gtol_abs': 1e-10})
            if res.x is not None:
                z = res.x
        except Exception:
            pass

        # Compute actual a and objective
        z_shifted = z - np.max(z)
        a = np.exp(z_shifted)
        a = a / np.sum(a)
        a = np.maximum(a, 0)
        a = a / np.sum(a)
        obj = compute_obj_norm(a)

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    S-LBFGS phase {phase+1}/{n_phases} (beta={beta}): obj={best_obj:.10f}, t={elapsed:.1f}s")
        elif verbose:
            print(f"    S-LBFGS phase {phase+1}/{n_phages} (beta={beta}): obj={obj:.10f} (no improvement)")

        if time.monotonic() - start > time_budget - 5:
            break

    return best_a


def gradient_cd(a, time_budget, seed=42, verbose=True):
    """
    Gradient-informed coordinate descent on normalized sequence.
    Uses the subgradient at active (near-max) convolution entries to guide updates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    conv = compute_conv(a)
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    sweep = 0
    no_improve = 0
    max_no_improve = 4

    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)
        max_conv = float(np.max(conv))

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            a_j = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)

            # Unaffected max
            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))

            # Generate candidate values based on current value and target
            mean_val = 1.0 / n
            # Try to reduce the peak: scale down if a_j is large, scale up if small
            candidates = set()
            candidates.add(0.0)
            candidates.add(a_j)
            # Geometric scaling around current value
            for mult in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]:
                candidates.add(a_j * mult)
            # Target values based on mean
            for mult in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 4.0]:
                candidates.add(mean_val * mult)
            # Clip to reasonable range
            candidates = [c for c in candidates if 0 <= c <= 0.5]
            candidates = sorted(set(candidates))

            best_trial_obj = best_obj
            best_trial_val = a_j

            for val in candidates:
                delta = val - a_j
                if abs(delta) < 1e-14:
                    continue

                S_new = 1.0 + delta
                if S_new <= 1e-10:
                    continue

                # Compute max of affected conv entries
                indices = np.arange(k_start, k_end + 1)
                affected = np.zeros(len(indices))
                for idx_i, k in enumerate(indices):
                    if k == 2 * j:
                        affected[idx_i] = conv[k] + (val * val - a_j * a_j)
                    else:
                        other_idx = k - j
                        if 0 <= other_idx < n:
                            affected[idx_i] = conv[k] + 2.0 * delta * a[other_idx]
                        else:
                            affected[idx_i] = conv[k]

                max_conv_new = max(max_unaffected, float(np.max(affected)))
                obj_trial = 2.0 * n * max_conv_new / (S_new * S_new)

                if obj_trial < best_trial_obj - 1e-14:
                    best_trial_obj = obj_trial
                    best_trial_val = val

            if best_trial_val != a_j:
                delta = best_trial_val - a_j
                a[j] = best_trial_val
                S_new = 1.0 + delta

                # Update conv in-place
                for k in range(k_start, k_end + 1):
                    if k == 2 * j:
                        conv[k] += (best_trial_val * best_trial_val - a_j * a_j)
                    else:
                        other_idx = k - j
                        if 0 <= other_idx < n:
                            conv[k] += 2.0 * delta * a[other_idx]

                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S

                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True
                    max_conv = new_obj / (2.0 * n)

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    G-CD sweep {sweep}: obj={best_obj:.10f}, t={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None, None

    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass

    for solver in ['CLARABEL', 'SCS', 'ECOS', 'MOSEK', 'PDLP']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a, time_budget, seed=42, verbose=True):
    """LP alternating minimization with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    prev_g = None
    iter_count = 0
    stall = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget - 5:
        g_lp, M_lp = solve_lp(a, warm_start=prev_g)
        if g_lp is None:
            break
        prev_g = g_lp
        iter_count += 1

        best_local_obj = best_obj
        best_local_a = a.copy()
        best_t = 0.0

        t_values = [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]
        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_local_obj - 1e-15:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_t = t

        if best_local_obj < best_obj - 1e-15:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall = 0
            if verbose and iter_count % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP {iter_count}: obj={best_obj:.10f}, t={best_t:.2f}, elapsed={elapsed:.0f}s")
        else:
            stall += 1
            if stall >= 6:
                strength = 0.01 + 0.005 * min(stall, 5)
                pert = rng.standard_normal(n) * strength
                a = best_a * (1.0 + pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                prev_g = None
                stall = 0

    if verbose:
        print(f"    LP done: {iter_count} iters, best={best_obj:.10f}")
    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 9):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def make_gaussian_init(n, sigma_frac=0.25):
    """Create a Gaussian-like initialization."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = a / np.sum(a)
    return a


def make_tapered_init(n, power=2.0):
    """Create a power-law tapered initialization."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** power
    a = a / np.sum(a)
    return a


def make_bimodal_init(n, sep_frac=0.7):
    """Create a bimodal initialization."""
    x = np.arange(n)
    c1 = (1 - sep_frac) / 2.0 * n
    c2 = (1 + sep_frac) / 2.0 * n
    sigma = 0.1 * n
    a = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    a = a / np.sum(a)
    return a

        print(f"Parent: n={n_parent}, obj={compute_obj_norm(a):.10f}")
    except Exception:

        n_parent = 800
        print(f"Default: n=800, obj={compute_obj_norm(a):.10f}")

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    n = len(best_a)

    # === Phase 1: Smoothed L-BFGS (35% time) ===
    print("\n--- Phase 1: Smoothed L-BFGS ---")
    p1_budget = min(budget_s * 0.35, 500.0)
    a1 = smooth_lbfgs(best_a, time_budget=p1_budget, seed=seed, verbose=True)

    if obj1 < best_obj:
        best_obj = obj1
        best_a = a1.copy()
        print(f"  P1: {best_obj:.10f} **")
    else:
        print(f"  P1: {obj1:.10f}")

    # === Phase 2: Gradient CD (15% time) ===
    print("\n--- Phase 2: Gradient CD ---")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p2_budget = min(remaining * 0.25, 200.0)
        a2 = gradient_cd(best_a, time_budget=p2_budget, seed=seed, verbose=True)
        obj2 = compute_obj_norm(a2)
        if obj2 < best_obj:
            best_obj = obj2

            print(f"  P2: {best_obj:.10f} **")
        else:
            print(f"  P2: {obj2:.10f}")

    # === Phase 3: LP Alternating (25% time) ===
    print("\n--- Phase 3: LP Alternating ---")

            best_a = a3.copy()
            print(f"  P3: {best_obj:.10f} **")
        else:
            print(f"  P3: {obj3:.10f}")

    # === Phase 4: Multi-restart with different initializations (15% time) ===
    print("\n--- Phase 4: Multi-restart ---")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        inits = [
            ("gauss_s02", make_gaussian_init(n, 0.2)),

            ("tapered2", make_tapered_init(n, 2.0)),
