"""
Optimized step function coefficient search using full-constraint LP alternating
method with golden-section line search and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a_norm, n, top_m=None):
    """Build matrix A where row k represents (a_norm * g)[k] = A[k,:] @ g"""
    if top_m is None or top_m >= 2 * n - 1:
        indices = list(range(2 * n - 1))
    else:
        conv_a = compute_conv(a_norm)
        indices = list(np.argsort(conv_a)[::-1][:top_m])
    
    A = np.zeros((len(indices), n))
    for idx, k in enumerate(indices):
        k = int(k)
        m_lo = max(0, k - n + 1)
        m_hi = min(k, n - 1)
        if m_lo <= m_hi:
            A[idx, m_lo:m_hi+1] = a_norm[k - m_hi : k - m_lo + 1][::-1]
    return A


def solve_bottleneck_lp(a, top_m=None, solver_pref=None):
    """
    Solve: min M s.t. (a*g)[k] <= M (top_m or all), sum(g)=1, g>=0
    Returns (g_opt, M_opt) or (None, None)
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm, n, top_m)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]
    if solver_pref:
        solvers = [solver_pref] + [s for s in solvers if s != solver_pref]
    
    for s in solvers:
        try:
            prob.solve(solver=s, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

def golden_section_line_search(a, g, max_iter=50):
    """Find t in [0,1] that minimizes obj((1-t)*a + t*g)"""
    gr = (5.0 ** 0.5 - 1.0) / 2.0  # golden ratio
    
    lo, hi = 0.0, 1.0
    c = lo + (1 - gr) * (hi - lo)
    d = lo + gr * (hi - lo)
    
    f_c = compute_obj((1 - c) * a + c * g)
    f_d = compute_obj((1 - d) * a + d * g)
    
    for _ in range(max_iter):
        if f_c < f_d:
            hi = d
            d = c
            f_d = f_c
            c = lo + (1 - gr) * (hi - lo)
            f_c = compute_obj((1 - c) * a + c * g)
        else:
            lo = c
            c = d
            f_c = f_d
            d = lo + gr * (hi - lo)
            f_d = compute_obj((1 - d) * a + d * g)
    
    t_opt = (lo + hi) / 2.0
    return t_opt


def lp_alternating(a, time_budget, use_full=True, verbose=True):
    """
    LP alternating method:
    1. Solve bottleneck LP for g given current a
    2. Find best blend t
    3. Update a
    Repeat until time runs out or convergence.
    """
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    top_m = 2 * n - 1 if use_full else min(50, 2 * n - 1)
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # Solve LP
        g_opt, M_opt = solve_bottleneck_lp(best_a, top_m=top_m)
        if g_opt is None:
            # Try with fewer constraints
            if top_m > 10:
                top_m = max(10, top_m // 2)
                continue
            break
        
        # Line search
        t_opt = golden_section_line_search(best_a, g_opt, max_iter=30)
        
        if t_opt > 1e-6:
            a_new = (1 - t_opt) * best_a + t_opt * g_opt
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                
                if obj_new < best_obj:
                    improvement = best_obj - obj_new
                    best_obj = obj_new
                    best_a = a_new.copy()
                    stall = 0
                    if verbose:
                        print(f"    LP alt it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                              f"top_m={top_m}, elapsed={time.monotonic()-start:.1f}s")
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1
        
        it += 1
        
        # Adaptive top_m: increase as we converge
        if use_full and it < 5:
            top_m = min(2 * n - 1, 10 + it * 3)
        
        if stall >= 5:
            # Perturb to escape
            rng = np.random.default_rng(it * 104729)
            noise = rng.normal(0, 0.002, n)
            a_pert = best_a + noise
            a_pert = np.maximum(a_pert, 0)
            s_p = float(np.sum(a_pert))
            if s_p > 0.01:
                best_a = a_pert / s_p
                best_obj = compute_obj(best_a)
            stall = 0
            it = 0
    
    return best_a


def coordinate_descent(a, time_budget, verbose=False):
    """
    Coordinate descent: for each i, try several values of a[i] to minimize objective.
    Uses the fact that changing a[i] only affects specific convolution entries.
    """
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for sweep in range(50):
        if time.monotonic() - start >= time_budget:
            break
        improved = False
        
        for i in range(n):
            if time.monotonic() - start >= time_budget:
                break
            
            old_val = best_a[i]
            if old_val < 1e-12:
                # Try setting it to a small positive value
                candidates = [0.001, 0.005, old_val * 1.1]
            else:
                candidates = [0.0, old_val * 0.5, old_val * 0.8, old_val * 1.2, old_val * 1.5,
                              old_val * 0.1, old_val * 2.0]
            
            best_local = best_obj
            best_val = old_val
            
            for cv in candidates:
                best_a[i] = cv
                s_new = float(np.sum(best_a))
                if s_new < 0.01:
                    best_a[i] = old_val
                    continue
                a_try = best_a / s_new
                # Quick objective check
                conv = compute_conv(a_try)
                obj_try = 2.0 * n * float(np.max(conv))
                if obj_try < best_local:
                    best_local = obj_try
                    best_val = cv
            
            if best_val != old_val:
                best_a[i] = best_val
                s_new = float(np.sum(best_a))
                if s_new > 0.01:
                    best_a = best_a / s_new
                    best_obj = compute_obj(best_a)
                    improved = True
            else:
                best_a[i] = old_val
        
        if verbose and sweep % 5 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
        
        if not improved:
            break
    
    return best_a


def u_shaped_init(n, sharpness):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def power_init(n, alpha):
    x = np.linspace(1/n, 1.0, n)
    a = x ** (-alpha)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def arcsine_init(n):
    x = np.linspace(0.005, 0.995, n)
    a = 1.0 / np.sqrt(np.maximum(1e-10, 1 - (2 * x - 1) ** 2))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, mode, rng):
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n / 4.0) ** 2) / (2 * (n / 6.0) ** 2)) + \
            np.exp(-((idx - 3 * n / 4.0) ** 2) / (2 * (n / 6.0) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 4:
        a = (1 + rng.random(n)) * (0.5 + 0.5 * np.cos(2 * np.pi * x))
    elif mode == 5:
        a = np.abs(np.sin(np.pi * x * (1 + rng.random() * 2))) ** 2 + 0.01
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def try_length(best_seq, best_score, n_try, time_budget, rng):
    """Try a different sequence length."""
    n_old = len(best_seq)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_try)
    a_try = np.interp(x_new, x_old, best_seq)
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 0.01:
        return None, np.inf
    a_try = a_try / s
    
    # Quick optimization
    t1 = min(time_budget * 0.5, 60.0)
    if t1 > 10:
        a_try = lp_alternating(a_try, time_budget=t1, use_full=False, verbose=False)
    
    # Quick CD
    t2 = min(time_budget * 0.3, 20.0)
    if t2 > 5:
        a_try = coordinate_descent(a_try, time_budget=t2)
    
    obj_try = compute_obj(a_try)
    return a_try, obj_try
