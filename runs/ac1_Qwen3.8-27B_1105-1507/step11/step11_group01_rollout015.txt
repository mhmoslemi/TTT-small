Multi-strategy optimizer for step function coefficient sequences.
Combines exact coordinate descent, mirror-descent peak shaving, LP-based
direction finding, multi-n exploration, and adaptive restarts to minimize

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def obj_value(a):
    """Compute objective 2n*max(conv)/sum^2 for non-negative a."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def obj_normalized(a):
    """Compute objective assuming sum(a)=1. Returns 2n*max(conv)."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def cd_coordinate_update(a, i, conv, n):
    """
    Exact 1D coordinate descent for coordinate i.
    Returns new value for a[i] that minimizes the 1D subproblem.
    """

    S = float(np.sum(a))
    S_other = S - v
    if S_other < 1e-12:
        return v

    k_min = max(0, i)
    k_max = min(2 * n - 2, i + n - 1)
    ks = np.arange(k_min, k_max + 1)

    # Separate quadratic term (k=2i) from linear terms
    mask_quad = (ks == 2 * i)
    mask_lin = ~mask_quad

    ks_lin = ks[mask_lin]
    a_ki_lin = a[ks_lin - i]
    c_k_lin = conv[ks_lin].copy()

    if mask_quad.any():
        c_2i = float(conv[2 * i])
    else:
        c_2i = 0.0

    def f_1d(x):
        if len(a_ki_lin) > 0:
            lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
            max_lin = float(np.max(lin_vals))
        else:
            max_lin = 0.0
        quad_val = c_2i + x * x - v * v
        max_val = max(max_lin, quad_val)
        denom = (S_other + x) * (S_other + x)
        if denom < 1e-30:
            return 1e30
        return 2.0 * n * max_val / denom

    # Candidate evaluation
    scale = max(v, S / n, 1e-6)
    candidates = [0.0, v * 0.25, v * 0.5, v * 0.75, v,
                  v * 1.25, v * 1.5, v * 2.0,
                  S / (2 * n), S / n, 2.0 * S / n, 3.0 * S / n]
    candidates = sorted(set(max(0.0, float(c)) for c in candidates))

    best_x = v
    best_f = f_1d(v)
    for x in candidates:
        fx = f_1d(x)
        if fx < best_f:
            best_f = fx
            best_x = x

    # Golden section refinement
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    x_lo = max(0.0, best_x - scale * 0.5)
    x_hi = best_x + scale
    if x_hi - x_lo < 1e-14:
        x_hi = x_lo + 1e-8

    c_pt = x_hi - gr * (x_hi - x_lo)
    d_pt = x_lo + gr * (x_hi - x_lo)
    fc = f_1d(c_pt)
    fd = f_1d(d_pt)

    for _ in range(50):
        if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
            break
        if fc < fd:
            x_hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = x_hi - gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
        else:
            x_lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = x_lo + gr * (x_hi - x_lo)
            fd = f_1d(d_pt)

    return max(0.0, (x_lo + x_hi) / 2.0)

def mirror_descent_peaks(a, time_budget=200.0, verbose=False, seed=42):
    """
    Mirror descent with weighted multi-peak subgradient.
    Uses exponential family updates: a_new = a * exp(-eta * grad) / Z.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    eta = 0.5 / np.sqrt(n)
    patience = 0
    it = 0
    top_m_max = min(20, n)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        # Anneal eta
        eta_cur = eta * (1.0 - 0.5 * progress)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Weighted multi-peak subgradient
        # Use top-m peaks with quadratic weighting
        sorted_idx = np.argsort(conv)[::-1]
        m = min(top_m_max, max(3, int(n * 0.05)))
        top_idx = sorted_idx[:m]
        top_vals = conv[top_idx]
        # Weight: (val - mean_top)^2 + 1 to avoid zero weights
        weights = (top_vals - top_vals.min() + 1.0) ** 2
        weights = weights / float(np.sum(weights))

        # Compute subgradient: for each peak k, contribution to a[i] is 2*a[k-i]
        grad = np.zeros(n)
        for wc, k in zip(weights, top_idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * wc * a[k - hi:k - lo + 1]

        # Mirror descent update
        log_a = np.zeros(n)
        pos_mask = a > 1e-15
        log_a[pos_mask] = np.log(a[pos_mask])

        log_a_new = log_a - eta_cur * grad
        log_a_new = np.minimum(log_a_new, 0)  # Cap to prevent explosion
        log_a_new = np.maximum(log_a_new, -30)  # Floor to prevent underflow

        a_new = np.exp(log_a_new)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            eta *= 0.5
            patience += 5
        else:

            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                patience = 0
            else:
                patience += 1
                # Try smaller step

                    eta *= 0.8
                    if eta < 1e-5:
                        # Perturb and reset
                        noise = rng.normal(0, 0.03, n)
                        a_pert = a * np.exp(noise)
                        a_pert = np.maximum(a_pert, 0)
                        sp = float(np.sum(a_pert))

                                best_a = a_pert.copy()
                                a = a_pert
                        eta = 0.5 / np.sqrt(n)
                        patience = 0
                        if patience > 50:

        it += 1
        if verbose and it % 200 == 0:
            print(f"    Mirror it{it}: obj={best_obj:.8f}, eta={eta:.6f}, t={elapsed:.1f}s")

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a_norm = a / s

    # Build convolution matrix A: A[k, i] = a_norm[k-i] if valid
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        lo = max(0, i)
        hi = min(2 * n - 2, i + n - 1)
        for k in range(lo, hi + 1):
            A[k, i] = a_norm[k - i]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:
                    g_val = g_val / g_sum
                    conv_check = A @ g_val
                    M_actual = float(np.max(conv_check))
                    return g_val, M_actual
        except Exception:
            continue
    return None, None

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        # Edge-biased
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * x)
        a = a / a.sum()
    elif mode == 3:
        # Two bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        # Frequency-modulated
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 7:
        # Semi-circle
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(0, 1 - x * x))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def peak_shaft_descent(a, time_budget=200.0, verbose=False, seed=42):
    """
    Direct peak-shaving: iteratively reduce the tallest peaks of the
    convolution by adjusting the responsible coordinates. Uses an adaptive
    step size and non-negative projection.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    alpha = 1.0 / n
    it = 0
    patience = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        mean_conv = float(np.mean(conv))

        # Find peaks significantly above mean
        threshold = max_conv - 0.5 * (max_conv - mean_conv)
        peak_indices = np.where(conv > threshold)[0]
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(conv)])

        # Compute weighted gradient focusing on peaks
        weights = (conv[peak_indices] - threshold) ** 2
        w_sum = float(np.sum(weights))
        if w_sum < 1e-30:
            weights = np.ones(len(peak_indices))
            w_sum = 1.0
        else:
            weights = weights / w_sum

        grad = np.zeros(n)
        for w, k in zip(weights, peak_indices):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                seg_a = a[k - hi:k - lo + 1]
                grad[lo:hi + 1] += 2.0 * w * seg_a

        # Subtract gradient, project
        a_new = a - alpha * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            alpha *= 0.5
            patience += 1
        else:
            a_new = a_new / s_new
            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                patience = 0
                # Slightly increase alpha if improving
                alpha = min(alpha * 1.01, 2.0 / n)
            else:
                patience += 1
                alpha *= 0.9

        if patience > 30:
            # Perturb
            noise = rng.normal(0, 0.02, n)
            a = a * np.exp(noise)
            a = np.maximum(a, 0)
            a = a / a.sum()
            alpha = 1.0 / n
            patience = 0

        it += 1
        if verbose and it % 500 == 0:
            print(f"    PeakShaft it{it}: obj={best_obj:.8f}, alpha={alpha:.6f}, t={elapsed:.1f}s")

    return best_a

        population = new_pop
        scores = new_scores

        if verbose and gen % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    Evo gen{gen}: best={best_score:.8f}, t={elapsed:.1f}s")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: multi-phase optimization combining CD, mirror
    descent peak shaving, LP polishing, evolutionary search, and
    systematic n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent sequence
    parent_n = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = obj_normalized(parent)
                best_seq = parent.copy()
                parent_n = parent.size
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(parent_n) / parent_n
        best_score = 2.0
        print(f"Created default (n={parent_n}): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Coordinate Descent Refinement ===
    print(f"\n=== Phase 1: CD Refinement (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 350.0)
    if p1_budget > 30:
        a_cd = coordinate_descent(best_seq, time_budget=p1_budget, verbose=True, seed=seed)
        sa = float(np.sum(a_cd))
        a_cd = a_cd / sa
        obj_cd = obj_normalized(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 2: Mirror Descent Peak Shaving ===
    print(f"\n=== Phase 2: Mirror Descent Peak Shaving ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 200.0)
    if p2_budget > 20:
        a_md = mirror_descent_peaks(best_seq, time_budget=p2_budget, verbose=False, seed=seed + 100)
        sa = float(np.sum(a_md))
        a_md = a_md / sa
        obj_md = obj_normalized(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Mirror: obj={best_score:.8f} **")
        else:
            print(f"  Mirror: obj={obj_md:.8f}")
            # CD from mirror result
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_cd2 = coordinate_descent(a_md, time_budget=min(40, remaining * 0.1), seed=seed + 200)
                sa2 = float(np.sum(a_cd2))
                a_cd2 = a_cd2 / sa2
                obj_cd2 = obj_normalized(a_cd2)
                if obj_cd2 < best_score:
                    best_score = obj_cd2
                    best_seq = a_cd2.copy()
                    print(f"  CD after Mirror: obj={best_score:.8f} **")

    # === Phase 3: Peak Shaft Descent ===
    print(f"\n=== Phase 3: Peak Shaft Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 150.0)
    if p3_budget > 20:
        a_ps = peak_shaft_descent(best_seq, time_budget=p3_budget, verbose=False, seed=seed + 300)
        sa = float(np.sum(a_ps))
        a_ps = a_ps / sa
        obj_ps = obj_normalized(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  PeakShaft: obj={best_score:.8f} **")
        else:
            print(f"  PeakShaft: obj={obj_ps:.8f}")
            # Follow with CD
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_cd3 = coordinate_descent(a_ps, time_budget=min(30, remaining * 0.1), seed=seed + 400)
                sa3 = float(np.sum(a_cd3))
                a_cd3 = a_cd3 / sa3
                obj_cd3 = obj_normalized(a_cd3)
                if obj_cd3 < best_score:
                    best_score = obj_cd3
                    best_seq = a_cd3.copy()
                    print(f"  CD after PeakShaft: obj={best_score:.8f} **")

    # === Phase 4: LP Polishing ===
    print(f"\n=== Phase 4: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 200.0)
    if p4_budget > 20:
        for rnd in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(p4_budget / (4 - rnd), remaining * 0.4)
            if alloc < 10:
                break
            a_lp = lp_polish(best_seq, time_budget=alloc, verbose=False, seed=1000 + rnd * 100)
            sa = float(np.sum(a_lp))
            a_lp = a_lp / sa
            obj_lp = obj_normalized(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
            # CD follow-up
            remaining = deadline - time.monotonic()
            if remaining > 15 and rnd < 3:
                a_cd4 = coordinate_descent(best_seq, time_budget=min(25, remaining * 0.08), seed=2000 + rnd)
                sa4 = float(np.sum(a_cd4))
                a_cd4 = a_cd4 / sa4
                obj_cd4 = obj_normalized(a_cd4)
                if obj_cd4 < best_score:
                    best_score = obj_cd4
                    best_seq = a_cd4.copy()
                    print(f"  CD after LP{rnd}: obj={best_score:.8f} **")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    # Generate diverse n candidates
    n_cands = []
    for delta in [-300, -200, -100, 100, 200, 300, 500, 700, 1000]:
        nc = n_cur + delta
        if 300 <= nc <= 4000:
            n_cands.append(nc)
    # Add some absolute values
    for nc in [500, 600, 900, 1100, 1500, 2000, 2500, 3000]:
        if 300 <= nc <= 4000:
            n_cands.append(nc)

    # Deduplicate (keep at least 50 apart)
    n_cands = sorted(set(n_cands))
    n_filtered = []
    for nc in n_cands:
        if all(abs(nc - nf) >= 50 for nf in n_filtered):
            n_filtered.append(nc)
    n_filtered = n_filtered[:8]
    print(f"  n candidates: {n_filtered}")

    for n_try in n_filtered:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        sa = float(np.sum(a_try))
        if sa < 1e-10:
            continue
        a_try = a_try / sa

        # CD on resampled
        alloc = min(remaining * 0.12, 50.0)
        if alloc > 15:
            a_try = coordinate_descent(a_try, time_budget=alloc, verbose=False, seed=n_try)
            sa = float(np.sum(a_try))
            if sa > 1e-10:
                a_try = a_try / sa

        obj_try = obj_normalized(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_cur = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Evolutionary Search ===
    print(f"\n=== Phase 6: Evolutionary Search ===")
    remaining = deadline - time.monotonic()
    evo_budget = min(remaining * 0.15, 200.0)
    if evo_budget > 30:
        a_evo = evolutionary_search(best_seq, len(best_seq), time_budget=evo_budget,
                                    rng=rng, verbose=False)
        sa = float(np.sum(a_evo))
        a_evo = a_evo / sa
        obj_evo = obj_normalized(a_evo)
        if obj_evo < best_score:
            best_score = obj_evo
            best_seq = a_evo.copy()
            print(f"  Evo: obj={best_score:.8f} **")
        else:
            print(f"  Evo: obj={obj_evo:.8f}")
            # CD from evo result
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_cd5 = coordinate_descent(a_evo, time_budget=min(30, remaining * 0.1), seed=seed + 600)
                sa5 = float(np.sum(a_cd5))
                a_cd5 = a_cd5 / sa5
                obj_cd5 = obj_normalized(a_cd5)
                if obj_cd5 < best_score:
                    best_score = obj_cd5
                    best_seq = a_cd5.copy()
                    print(f"  CD after Evo: obj={best_score:.8f} **")

    # === Phase 7: Diverse Restarts with CD ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = (r + 1) % 8
        a_r = diverse_init(n_final, rng, mode=mode)
        blend = 0.2 + 0.1 * (r % 4)
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        sr = float(np.sum(a_r))
        if sr > 1e-10:
            a_r = a_r / sr
        else:
            a_r = best_seq.copy()

        alloc = min(remaining * 0.15, 60.0)
        if alloc > 15:
            a_r = coordinate_descent(a_r, time_budget=alloc, verbose=False, seed=5000 + r * 100)
            sa = float(np.sum(a_r))
            a_r = a_r / sa
            obj_r = obj_normalized(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode {mode}): obj={obj_r:.8f}")

    # === Phase 8: Final Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Multiple CD passes with different orderings
        for fp in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.3, 80.0)
            if alloc > 8:
                a_f = coordinate_descent(best_seq, time_budget=alloc, verbose=False,
                                         seed=9000 + fp * 77)
                sa = float(np.sum(a_f))
                a_f = a_f / sa
                obj_f = obj_normalized(a_f)
                if obj_f < best_score:
                    best_score = obj_f
                    best_seq = a_f.copy()
                    print(f"  Final CD {fp}: obj={best_score:.8f} **")

        # Quick LP
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_fl = lp_polish(best_seq, time_budget=min(30, remaining * 0.4), seed=9999)
            sa = float(np.sum(a_fl))
            a_fl = a_fl / sa
            obj_fl = obj_normalized(a_fl)
            if obj_fl < best_score:
                best_score = obj_fl
                best_seq = a_fl.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = 2.0

    best_seq = np.maximum(best_seq, 0.0)
    result = [float(min(1000.0, x)) for x in best_seq]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if np.isinf(val):
        result = [1.0] * 100
        best_score = 2.0
    elif val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
