from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def obj_normalized(a):
    """Objective assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

"""
Optimizes step function coefficients to minimize 2n*max(conv(a,a))/(sum(a))^2.
Uses a multi-phase approach: intensive coordinate descent, LP-based polishing,
subgradient exploration, simulated annealing, and multi-n search.
"""
import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time
import math

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def obj_of(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def obj_norm(a):
    """Compute 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A

def solve_lp(a, solver_list=None):
    """Solve LP: min M s.t. conv(a_norm, g) <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    return g_val, float(M.value)
        except Exception:
            continue
    return None, None

                    fc = fd

                continue
            
            delta = x_opt - v
            a[i] = x_opt
            
            # Update conv incrementally
            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            
            S += delta

                    stall = 0

def subgradient_descent(a, time_budget=100.0, seed=42, verbose=False):
    """Subgradient descent with momentum and adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = float(obj_norm(a))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    eta = 0.5 / np.sqrt(n)
    momentum = 0.7
    v_mom = np.zeros(n)
    patience = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        tol = max_conv * 1e-8
        argmax_set = np.where(conv >= max_conv - tol)[0]
        
        # Subgradient
        v_sub = np.zeros(n)
        for k in argmax_set:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                v_sub[lo:hi+1] += a[k-hi:k-lo+1]
        v_sub /= max(len(argmax_set), 1)
        
        # Target: at optimum, v_sub[i] should be constant (= max_conv/2 for interior points)
        grad = 2.0 * v_sub - 2.0 * max_conv / (2.0 * n) * np.ones(n)
        # Simplified: grad[i] = 2*v_sub[i] - lambda, where lambda = average of 2*v_sub
        # Better: use the KKT condition
        grad = 2.0 * v_sub
        grad -= np.mean(grad)  # Center to preserve sum
        
        v_mom = momentum * v_mom - eta * grad
        a_new = a + v_mom
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = float(obj_norm(a_new))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                patience = 0
            else:
                patience += 1
                if patience > 30:
                    eta *= 0.7
                    if eta < 1e-5:
                        # Restart with perturbation
                        a = best_a * np.exp(rng.normal(0, 0.05, n))
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        eta = 0.5 / np.sqrt(n)
                        v_mom = np.zeros(n)
                        patience = 0
        
    return best_a

def resample(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, solver_list=None):
    """Solve LP: min M s.t. conv(a_norm, g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    return g_val, float(max(M.value, 0))
        except Exception:
            continue
    return None, None


def f_1d_eval(x, n, v, S_other, c_k_lin, a_ki_lin, c_2i):
    """Evaluate 1D subproblem objective for coordinate descent."""
    if len(a_ki_lin) > 0:
        lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
        max_lin = float(np.max(lin_vals))
    else:
        max_lin = 0.0
    quad_val = c_2i + x * x - v * v
    max_val = max(max_lin, quad_val)
    denom = S_other + x
    if denom < 1e-15:
        return 1e30
    return 2.0 * n * max_val / (denom * denom)


def golden_section(f_lo, f_hi, c_pt, d_pt, fc, fd, gr, max_iter=45):
    """Golden section search. Returns optimal point."""
    x_lo = f_lo
    x_hi = f_hi
    for _ in range(max_iter):
        if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
            break
        if fc < fd:
            x_hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = x_hi - gr * (x_hi - x_lo)
            fc = f_1d_eval_wrapper(c_pt)
        else:
            x_lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = x_lo + gr * (x_hi - x_lo)
            fd = f_1d_eval_wrapper(d_pt)
    return max(0.0, (x_lo + x_hi) / 2.0)


# We can't easily use golden_section as a standalone due to the closure issue.
# Instead, inline the golden section in the CD loop.


def coordinate_descent(a, time_budget=300.0, verbose=False, seed=42):
    """Exact coordinate descent with incremental conv updates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = float(obj_norm(a))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    stall = 0
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            v = a[i]
            S_other = S - v

            k_min = i
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            k_quad = 2 * i
            mask_quad = (ks == k_quad)
            mask_lin = ~mask_quad
            ks_lin = ks[mask_lin]
            a_ki_lin = a_ki[mask_lin]
            c_k_lin = c_k[mask_lin].copy()
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            # Candidate points
            scale = max(v, S / n, 1e-6)
            candidates = [0.0, v * 0.5, v, v * 1.5, S / n, 2.0 * S / n]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = f_1d_eval(v, n, v, S_other, c_k_lin, a_ki_lin, c_2i)
            for x in candidates:
                fx = f_1d_eval(x, n, v, S_other, c_k_lin, a_ki_lin, c_2i)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section refinement
            x_lo = max(0.0, best_x - scale * 0.5)
            x_hi = best_x + scale
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-8
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d_eval(c_pt, n, v, S_other, c_k_lin, a_ki_lin, c_2i)
            fd = f_1d_eval(d_pt, n, v, S_other, c_k_lin, a_ki_lin, c_2i)

            for _ in range(40):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d_eval(c_pt, n, v, S_other, c_k_lin, a_ki_lin, c_2i)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d_eval(d_pt, n, v, S_other, c_k_lin, a_ki_lin, c_2i)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-15:
                continue

            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S += delta

        # Periodic verification
        if sweep % 10 == 0:
            conv = compute_conv(a)
            S_new = float(np.sum(a))
            if S_new > 1e-10:
                a = a / S_new
                obj_now = float(obj_norm(a))
                if obj_now < best_obj - 1e-15:
                    best_obj = obj_now
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1
                if stall > 100:
                    break

        if verbose and sweep % 100 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, t={time.monotonic()-start:.1f}s")

    return best_a


def subgradient_descent(a, time_budget=100.0, seed=42, verbose=False):
    """Subgradient descent with momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = float(obj_norm(a))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    eta = 0.3 / np.sqrt(n)
    momentum = 0.7
    v_mom = np.zeros(n)
    patience = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        tol = max( max_conv * 1e-8, 1e-15)
        argmax_set = np.where(conv >= max_conv - tol)[0]

        v_sub = np.zeros(n)
        for k in argmax_set:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                v_sub[lo:hi + 1] += a[k - hi:k - lo + 1]
        v_sub /= max(len(argmax_set), 1)

        grad = 2.0 * v_sub
        grad -= np.mean(grad)

        v_mom = momentum * v_mom - eta * grad
        a_new = a + v_mom
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = float(obj_norm(a_new))
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                patience = 0
            else:
                patience += 1
                if patience % 50 == 0:
                    eta *= 0.8
                if patience > 100:
                    a = best_a * np.exp(rng.normal(0, 0.04, n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    eta = 0.3 / np.sqrt(n)
                    v_mom = np.zeros(n)
                    patience = 0

        if verbose and (time.monotonic() - start) % 5 < 0.1:
            print(f"    SubGD: obj={best_obj:.10f}, eta={eta:.6f}, t={time.monotonic()-start:.1f}s")

    return best_a


def lp_polish(a, time_budget=60.0, verbose=False, seed=42):
    """LP-based polishing with line search on blend parameter."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = float(obj_norm(a))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        best_t_obj = best_obj
        best_t_a = best_a.copy()
        for t in [0.01, 0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = float(obj_norm(a_try))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_a = a_try.copy()

        if best_t_obj < best_obj - 1e-15:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 4:
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = float(obj_norm(a))
                stall = 0

        if verbose:
            print(f"    LP: obj={best_obj:.10f}, t={time.monotonic()-start:.1f}s, stall={stall}")

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode):
    """Generate diverse initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        a = np.zeros(n)
        w = max(1, n // 4)
        a[:w] = np.exp(-((np.arange(w) - w // 2) ** 2) / (2 * max(1, w // 4) ** 2))
        a[-w:] = np.exp(-((np.arange(w) - w // 2) ** 2) / (2 * max(1, w // 4) ** 2))
        a += 0.05
        return a / a.sum()
    elif mode == 3:
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / np.sqrt(1 - x * x)
        return a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        return a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for _ in range(int(rng.integers(3, 7))):
            f = rng.uniform(1, 5)
            p = rng.uniform(0, 2 * np.pi)
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def smoothed_gd(a, time_budget=80.0, seed=42, verbose=False):
    """Smoothed gradient descent with exponential surrogate."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = float(obj_norm(a))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    tau = 50.0
    lr = 2.0 / np.sqrt(n)
    beta = 0.85
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = 30.0 + 400.0 * progress

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[int(np.argmax(conv))] = 1.0
        else:
            p = exp_vals / sum_exp

        N = 1
        while N < 3 * n:
            N <<= 1
        from scipy.fft import fft, ifft
        P = fft(p, N)
        A_fft = fft(y, N)
        g_full = ifft(P * np.conj(A_fft)).real
        grad = 2.0 * g_full[:n]
        grad -= np.mean(grad)

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = float(obj_norm(a_new))
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
        a = a_new
        it += 1

        if verbose and it % 5000 == 0:
            print(f"    SmGD it{it}: obj={best_obj:.10f}, tau={tau:.0f}, t={elapsed:.1f}s")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining exact CD, LP polishing,
    subgradient descent, smoothed GD, simulated annealing, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = float(obj_norm(parent))
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:
        best_seq = np.ones(n_parent) / n_parent
        best_score = float(obj_norm(best_seq))
        print(f"Default init (n={n_parent}): obj={best_score:.10f}")

    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(800) / 800.0
        best_score = float(obj_norm(best_seq))

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.10f}")

    # === Phase 1: Intensive CD ===
    print(f"\n=== Phase 1: Intensive Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 350.0)
    if p1_budget > 30:
        a_cd = coordinate_descent(best_seq, time_budget=p1_budget, verbose=True, seed=seed)
        obj_cd = float(obj_norm(a_cd))
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.10f} **")
        else:
            print(f"  CD: obj={obj_cd:.10f} (no improvement)")

    # === Phase 2: LP Polishing ===
    print(f"\n=== Phase 2: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 300.0)
    if p2_budget > 20:
        for rnd in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(p2_budget / (4 - rnd), remaining * 0.5)
            if alloc < 10:
                break
            a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True, seed=1000 + rnd * 7)
            obj_lp = float(obj_norm(a_lp))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.10f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.10f}")
            # Quick CD after LP
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_cd2 = coordinate_descent(best_seq, time_budget=min(30, remaining * 0.1),
                                           verbose=False, seed=2000 + rnd)
                obj_cd2 = float(obj_norm(a_cd2))
                if obj_cd2 < best_score:
                    best_score = obj_cd2
                    best_seq = a_cd2.copy()
                    print(f"  CD after LP{rnd}: obj={best_score:.10f} **")

    # === Phase 3: Multi-n Exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    n_candidates = sorted(set([
        max(200, n_cur - 100),
        n_cur + 100,
        n_cur + 200,
        n_cur + 400,
        600, 700, 900, 1000, 1200, 1500
    ]))
    n_filtered = []
    for nc in n_candidates:
        if 300 <= nc <= 2000:
            if all(abs(nc - nf) > 80 for nf in n_filtered):
                n_filtered.append(nc)
    n_filtered = n_filtered[:5]
    print(f"  n candidates: {n_filtered}")

    for n_try in n_filtered:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        a_try = resample(best_seq, n_try)
        a_try = a_try / a_try.sum()
        alloc = min(remaining * 0.12, 50.0)
        if alloc > 10:
            a_try = coordinate_descent(a_try, time_budget=alloc, verbose=False, seed=n_try)
        obj_try = float(obj_norm(a_try))
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 4: Subgradient + CD cycles ===
    print(f"\n=== Phase 4: Subgradient Exploration ===")
    remaining = deadline - time.monotonic()
    for cyc in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        # Subgradient to explore
        a_sub = best_seq * np.exp(rng.normal(0, 0.03 + 0.02 * cyc, len(best_seq)))
        a_sub = np.maximum(a_sub, 0)
        a_sub = a_sub / a_sub.sum()
        alloc_sg = min(remaining * 0.15, 60.0)
        if alloc_sg > 15:
            a_sub = subgradient_descent(a_sub, time_budget=alloc_sg, seed=4000 + cyc * 100)
        # CD to polish
        remaining = deadline - time.monotonic()
        alloc_cd = min(remaining * 0.15, 60.0)
        if alloc_cd > 10:
            a_sub = coordinate_descent(a_sub, time_budget=alloc_cd, verbose=False, seed=5000 + cyc * 100)
        obj_sub = float(obj_norm(a_sub))
        if obj_sub < best_score:
            best_score = obj_sub
            best_seq = a_sub.copy()
            print(f"  SubGD cycle {cyc}: obj={best_score:.10f} **")
        else:
            print(f"  SubGD cycle {cyc}: obj={obj_sub:.10f}")

    # === Phase 5: Smoothed GD ===
    print(f"\n=== Phase 5: Smoothed Gradient Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        a_pert = best_seq * np.exp(rng.normal(0, 0.05, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        alloc = min(remaining * 0.15, 100.0)
        if alloc > 15:
            a_gd = smoothed_gd(a_pert, time_budget=alloc, seed=6000)
            obj_gd = float(obj_norm(a_gd))
            if obj_gd < best_score:
                best_score = obj_gd
                best_seq = a_gd.copy()
                print(f"  SmGD: obj={best_score:.10f} **")
            else:
                print(f"  SmGD: obj={obj_gd:.10f}")

    # === Phase 6: Simulated Annealing with CD ===
    print(f"\n=== Phase 6: Simulated Annealing ===")
    remaining = deadline - time.monotonic()
    n_sa = len(best_seq)
    T = 0.02
    sa_accepted = 0
    for sa_iter in range(200):
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        # Perturb
        pert_size = T * (1.0 + sa_iter / 50.0)
        a_pert = best_seq * np.exp(rng.normal(0, pert_size, n_sa))
        a_pert = np.maximum(a_pert, 0)
        s_pert = float(np.sum(a_pert))
        if s_pert < 1e-10:
            T *= 0.95
            continue
        a_pert = a_pert / s_pert

        # Quick CD (just a few sweeps)
        a_pert = coordinate_descent(a_pert, time_budget=min(5.0, remaining * 0.02),
                                    verbose=False, seed=7000 + sa_iter)
        obj_pert = float(obj_norm(a_pert))

        # Accept/reject
        if obj_pert < best_score:
            best_score = obj_pert
            best_seq = a_pert.copy()
            sa_accepted += 1
            T = max(0.005, T * 0.98)
        else:
            # Metropolis acceptance
            delta = obj_pert - best_score
            if delta < T * 0.01:
                best_seq = a_pert.copy()
                sa_accepted += 1
            T *= 0.97

        if sa_iter % 20 == 0 and sa_iter > 0:
            print(f"  SA iter {sa_iter}: best={best_score:.10f}, T={T:.4f}, accepted={sa_accepted}, t={time.monotonic()-start_time:.0f}s")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    for final_round in range(2):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        # Final CD
        cd_alloc = min(remaining * 0.4, 120.0)
        if cd_alloc > 10:
            a_f = coordinate_descent(best_seq, time_budget=cd_alloc,
                                     verbose=True, seed=9000 + final_round)
            obj_f = float(obj_norm(a_f))
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final CD {final_round}: obj={best_score:.10f} **")
        # Final LP
        remaining = deadline - time.monotonic()
        if remaining > 15:
            lp_alloc = min(remaining * 0.3, 60.0)
            if lp_alloc > 10:
                a_f2 = lp_polish(best_seq, time_budget=lp_alloc,
                                 verbose=True, seed=9500 + final_round)
                obj_f2 = float(obj_norm(a_f2))
                if obj_f2 < best_score:
                    best_score = obj_f2
                    best_seq = a_f2.copy()
                    print(f"  Final LP {final_round}: obj={best_score:.10f} **")

    # Final validation and output
    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(200)
        best_score = 2.0

    best_seq = np.maximum(best_seq, 0.0)
    # Ensure sum > 0.01
    s_final = float(np.sum(best_seq))
    if s_final < 0.01:
        best_seq = best_seq * (0.1 / max(s_final, 1e-15))
    # Cap values at 1000
    best_seq = np.minimum(best_seq, 1000.0)

    result = [float(x) for x in best_seq]

    # Validate
    val = evaluate_sequence(result)
    if val < np.inf:
        print(f"\n=== FINAL: n={len(result)}, evaluated_score={val:.10f}, "
              f"computed_score={best_score:.10f}, time={time.monotonic()-start_time:.1f}s ===")
    else:
        print(f"\n=== WARNING: evaluate_sequence returned inf, using fallback ===")
        result = [1.0] * 100

    return result
