import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n


def lse_smoothing(a, T):
    """Compute LSE-smoothed max of conv(a,a)."""
    conv = compute_conv(a)
    cmax = np.max(conv)
    # Clip to avoid overflow
    exp_vals = np.exp(np.clip(T * (conv - cmax), -500, 0))
    lse = cmax + np.log(np.sum(exp_vals)) / T
    return lse

    conv = compute_conv(a)

    w = exp_vals / exp_vals.sum()

    # grad[i] = 2n * sum_k w[k] * (a[k-i] * I(0<=k-i<n) + a[i-k] * I(0<=i-k<n))
    # Term 1: sum_k w[k] * a[k-i] for 0<=k-i<n => sum_m w[i+m] * a[m] for 0<=m<n, 0<=i+m<2n-1
    # Term 2: sum_k w[k] * a[i-k] for 0<=i-k<n => sum_m w[i-m] * a[m] for 0<=m<n, 0<=i-m<2n-1

    # For term 1: (w * a_rev) style correlation
    # Let's compute directly using the structure
    grad = np.zeros(n)
    
    # Term 1: for each i, sum over m where 0<=m<n and 0<=i+m<2n-1
    # i+m < 2n-1 => m < 2n-1-i
    # So m ranges from 0 to min(n-1, 2n-2-i)
    # grad[i] += sum_{m=0}^{min(n-1, 2n-2-i)} w[i+m] * a[m]
    
    # This is a cross-correlation. Let's use FFT.
    # cross_corr1[i] = sum_m w[i+m] * a[m]
    # = sum_m w[m+i] * a[m]
    # Let's reverse w: w_rev[k] = w[2n-2-k]
    # Then w[m+i] = w_rev[2n-2-m-i]
    # cross_corr1[i] = sum_m w_rev[2n-2-m-i] * a[m]
    # = (w_rev * a)[2n-2-i]  (full convolution, index 2n-2-i)
    
    w_rev = w[::-1]  # w_rev[k] = w[2n-2-k]
    # Full convolution of w_rev and a: length (2n-1) + n - 1 = 3n-2
    # conv_full[k] = sum_j w_rev[j] * a[k-j] for valid j
    # We want conv_full[2n-2-i] = sum_j w_rev[j] * a[2n-2-i-j]
    # Let m = 2n-2-i-j, so j = 2n-2-i-m
    # = sum_m w_rev[2n-2-i-m] * a[m] (for 0<=m<n and 0<=2n-2-i-m<2n-1)
    # = sum_m w[2n-1-(2n-2-i-m)] ... hmm this is getting confusing.
    
    # Let me just do it with a direct loop for clarity, since n=800 is small enough
    # Actually, let me use a different FFT approach
    
    # Alternative: compute the two terms as convolutions
    # Term 1: g1[i] = sum_{m: 0<=m<n, 0<=i+m<2n-1} w[i+m] * a[m]
    # This is the cross-correlation of (w, a) evaluated at lag i
    # cross_corr(w, a)[i] = sum_m w[m+i] * a[m] = sum_m w[m+i] * a[m]
    
    # Using FFT: cross_corr can be computed as IFFT(FFT(w) * conj(FFT(a)))
    # But we need to be careful with the length.
    
    # Let's pad both to length 4n-2 (at least 2n-1 + n - 1 = 3n-2)
    size_fft = 1
    while size_fft < 3 * n - 1:
        size_fft <<= 1
    
    W = rfft(w, size_fft)
    A = rfft(a, size_fft)
    
    # cross_corr[i] = sum_m w[m+i] * a[m]
    # = IFFT(conj(W) * A)[size_fft - i] ... no this isn't right either
    # Let me think more carefully.
    # 
    # For sequences x (length Lx) and y (length Ly), the cross-correlation is:
    # rxy[k] = sum_n x[n+k] * conj(y[n])
    # 
    # This equals ifft(fft(x, N) * conj(fft(y, N))) for appropriate N >= Lx + Ly - 1
    # but with a circular shift.
    #
    # Actually, the standard result is:
    # rxy[k] = (x cross-correlate y)[k] = ifft(conj(FFT(x)) * FFT(y))[k] 
    # for 0 <= k < Lx (the "positive lags")
    # and for negative lags, we use the other half.
    #
    # Let me just use np.correlate or do it directly.
    
    # Direct computation (O(n^2), fine for n=800):
    # Term 1: g1[i] = sum_m w[i+m] * a[m] for 0<=m<n and i+m < 2n-1
    for i in range(n):
        # m ranges from 0 to min(n-1, 2n-2-i)
        max_m = min(n - 1, 2 * n - 2 - i)
        if max_m >= 0:
            grad[i] += np.dot(w[i:i + max_m + 1], a[:max_m + 1])
        # Term 2: g2[i] = sum_m w[i-m] * a[m] for 0<=m<n and i-m >= 0 and i-m < 2n-1
        # i-m >= 0 => m <= i
        # i-m < 2n-1 => m > i-(2n-1) (always true since m>=0 and i<n)
        # So m ranges from 0 to min(n-1, i)
        max_m2 = min(n - 1, i)
        if max_m2 >= 0:
            # w[i-m] for m=0,...,max_m2 => w[i], w[i-1], ..., w[i-max_m2]
            # = w[i-max_m2 : i+1][::-1]
            grad[i] += np.dot(w[i - max_m2:i + 1][::-1], a[:max_m2 + 1])
    
    return 2.0 * n * grad


def lse_objective_full(a, T):
    """Compute 2n * LSE_T(conv(a,a))."""
    n = len(a)
    return 2.0 * n * lse_smoothing(a, T)


def full_lp_solve(a_norm, solver_list=None):
    """Solve the full LP: min M s.t. (a_norm * g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a_norm)
    
    # Build constraint matrix: A[k, j] = a_norm[k-j] if 0 <= k-j < n, else 0
    # This is a convolution matrix
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        # (a*g)[k] = sum_j a[j] * g[k-j]
        # Coefficient of g[m] is a[k-m] (if 0 <= k-m < n)
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        for m in range(m_start, m_end + 1):
            A[k, m] = a_norm[k - m]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.ECOS, cp.SCS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def build_conv_matrix(a_norm):
    """Build the convolution matrix A where (A @ g)[k] = (a_norm * g)[k]."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            A[k, m_start:m_end + 1] = a_norm[k - m_end:k - m_start + 1][::-1]
    return A


def lp_alternation(a, time_budget, max_iters=100, verbose=False):
    """
    Iterative LP alternation:
    1. Fix a, solve LP for optimal g
    2. Blend a with g
    3. Repeat
    Uses full constraints.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    blend_factor = 0.5

    # Pre-build the matrix (it changes with a, so we rebuild each time)
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        g_lp = full_lp_solve(a)
        if g_lp is None:
            break
        
        # Find best blend
        best_t = 0.0
        best_obj_try = best_obj
        for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 0.01:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_try:
                best_obj_try = obj_try
                best_t = t
        
        if best_t > 0 and best_obj_try < best_obj - 1e-12:
            a_new = (1 - best_t) * a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        if verbose and it % 5 == 0:
            print(f"      LP-alt it{it}: obj={best_obj:.8f}, t={best_t:.2f}, "
                  f"elapsed={time.monotonic()-start:.1f}s")
        
        it += 1
        
        if stall >= 5:
            # Perturb and reduce blend
            rng_local = np.random.default_rng(it * 31 + 7)
            noise = rng_local.normal(0, 0.002 * stall, n)
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_a = best_a.copy()  # Keep best
            stall = 0
            blend_factor = max(0.1, blend_factor * 0.7)

    return best_a


def lse_gd_optimize(a, time_budget, T_start=5.0, T_end=100.0, verbose=False):
    """LSE gradient descent with backtracking line search and adaptive T."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.5 / n
    it = 0
    best_it = 0
    patience = 500

    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** frac

        grad = lse_gradient(a, T)
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-14:
            break

        # Backtracking line search
        a_old = a.copy()
        obj_old = lse_objective_full(a, T)
        t = lr
        improved = False
        for _ in range(8):
            a_try = a_old - t * grad
            a_try = project_simplex(a_try)
            obj_try = lse_objective_full(a_try, T)
            if obj_try < obj_old - 1e-12:
                a = a_try
                improved = True
                break
            t *= 0.5

        if not improved:
            # Try random perturbation
            for _ in range(5):
                direction = a + np.random.randn(n) * 0.005
                a_try = project_simplex(direction)
                obj_try = lse_objective_full(a_try, T)
                if obj_try < obj_old:
                    a = a_try
                    improved = True
                    break
            if not improved:
                break

        obj_true = compute_obj(a)
        if obj_true < best_obj:
            best_obj = obj_true
            best_a = a.copy()
            best_it = it

        it += 1
        if it - best_it > patience:
            a = best_a.copy()
            lr = lr * 0.5
            patience = max(100, patience // 2)
            best_it = it

        if verbose and it % 200 == 0:
            print(f"      LSE-GD it{it}: obj={best_obj:.8f}, T={T:.1f}, "
                  f"t={time.monotonic()-start:.1f}s")

    return best_a


def coordinate_descent_fast(a, time_budget, verbose=False):
    """
    Fast coordinate descent: for each coordinate, try several candidate values,
    pick the best one. Uses incremental convolution update for speed.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    
    sweep = 0
    while time.monotonic() - start < time_budget:
        sweep_improved = False
        
        for i in range(n):
            if time.monotonic() - start >= time_budget:
                break
            
            ai_old = a[i]
            if ai_old < 1e-12:
                # Try turning on this coordinate
                if sweep % 3 == 0:
                    candidates = [0.001 / n, 0.005 / n, 0.01 / n]
                else:
                    continue
            else:
                candidates = [0.0, ai_old * 0.5, ai_old * 0.75, ai_old * 1.25, ai_old * 1.5, ai_old * 2.0]
            
            best_val = ai_old
            best_local_obj = best_obj
            
            for cv in candidates:
                if cv == ai_old:
                    continue
                # Update conv incrementally
                # conv changes: conv[k] changes by (cv - ai_old) * a[k-i] + (cv - ai_old) * a[i-k]
                # for valid indices
                delta = cv - ai_old
                if abs(delta) < 1e-15:
                    continue
                
                new_max = np.max(conv)
                # We need to check if the max changes
                # Actually, for speed, let's just compute the new conv
                a[i] = cv
                s_new = float(np.sum(a))
                if s_new < 0.01:
                    a[i] = ai_old
                    continue
                a_try = a / s_new
                new_conv = compute_conv(a_try)
                obj_try = 2.0 * n * float(np.max(new_conv))
                a[i] = ai_old
                
                if obj_try < best_local_obj:
                    best_local_obj = obj_try
                    best_val = cv
            
            if best_val != ai_old:
                a[i] = best_val
                s_new = float(np.sum(a))
                if s_new > 0.01:
                    a = a / s_new
                    conv = compute_conv(a)
                    obj_now = 2.0 * n * float(np.max(conv))
                    if obj_now < best_obj:
                        best_obj = obj_now
                        best_a = a.copy()
                    sweep_improved = True

        sweep += 1
        if verbose and sweep % 5 == 0:
            print(f"      CD sweep {sweep}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
        
        if not sweep_improved and sweep > 2:
            break

    return best_a


def diverse_init(n, mode, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n / 4.0) ** 2) / (2 * (n / 6.0) ** 2)) + \
            np.exp(-((idx - 3 * n / 4.0) ** 2) / (2 * (n / 6.0) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 4:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 5:
        a = 1.0 / np.cosh(3 * (2 * x - 1))
    elif mode == 6:
        a = (1 + np.sin(np.pi * x)) ** 2
    elif mode == 7:
        # Random with structure
        base = np.ones(n) * 0.1
        for _ in range(5):
            center = rng.randint(0, n)
            width = rng.randint(n // 10, n // 3)
            height = rng.uniform(0.5, 2.0)
            base += height * np.exp(-((idx - center) ** 2) / (2 * width ** 2))
        a = base
    elif mode == 8:
        # Power law
        d = np.maximum(np.abs(x - 0.5), 1.0 / (2 * n))
        a = d ** (-1.5) + 0.01
    elif mode == 9:
        # Arcsine
        a = 1.0 / np.sqrt(np.maximum(1e-10, 1 - (2 * x - 1) ** 2))
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def palindromic_project(a):
    """Project to palindromic: a[i] = a[n-1-i], normalize."""
    n = len(a)
    a_pal = 0.5 * (a + a[::-1])
    a_pal = np.maximum(a_pal, 0)
    s = float(np.sum(a_pal))
    if s < 1e-10:
        return np.ones(n) / n
    return a_pal / s


def optimize_with_palindromic_constraint(a, time_budget, verbose=False):
    """LP alternation with palindromic constraint on g."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # For palindromic g: g[i] = g[n-1-i]
    # Effective variables: g[0], ..., g[(n-1)//2] (roughly n/2 variables)
    half_n = (n + 1) // 2
    
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget and it < 50:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break
        
        # Build LP with palindromic constraint
        # g is determined by g[0:half_n], with g[n-1-i] = g[i]
        g_half = cp.Variable(half_n, nonneg=True)
        M = cp.Variable()
        
        # Reconstruct full g from g_half
        # g[i] = g_half[i] for i < half_n
        # g[i] = g_half[n-1-i] for i >= half_n
        g_full = np.zeros(n)
        for i in range(n):
            if i < half_n:
                g_full[i] = g_half[i]
            else:
                g_full[i] = g_half[n - 1 - i]
        
        # Build constraint matrix for g_half
        # (a * g_full)[k] = sum_j a[j] * g_full[k-j]
        # = sum_j a[j] * g_half[map(k-j)]
        # where map(m) = m if m < half_n, else n-1-m
        
        A_half = np.zeros((2 * n - 1, half_n))
        for k in range(2 * n - 1):
            for m in range(half_n):
                # g_full[m] = g_half[m]
                # g_full[n-1-m] = g_half[m]
                # Contribution to (a*g)[k] from g_full[m]: a[k-m] (if 0<=k-m<n)
                # Contribution to (a*g)[k] from g_full[n-1-m]: a[k-(n-1-m)] (if 0<=k-(n-1-m)<n)
                j1 = k - m
                if 0 <= j1 < n:
                    A_half[k, m] += a[j1]
                j2 = k - (n - 1 - m)
                if 0 <= j2 < n:
                    A_half[k, m] += a[j2]
        
        constraints = [A_half @ g_half <= M, cp.sum(g_half) == 0.5]  # sum(g_full) = 2*sum(g_half) for even n, need to be careful
        # Actually sum(g_full) = sum_i g_full[i] = 2*sum_{i<half_n} g_half[i] - (g_half[n//2] if n is odd)
        # For simplicity, let's just normalize after
        constraints = [A_half @ g_half <= M]
        # We'll normalize g after the solve
        
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        g_val = None
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g_half.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_half_val = np.maximum(g_half.value, 0)
                    s_half = float(np.sum(g_half_val))
                    if s_half > 1e-10:
                        # Reconstruct full g
                        g_full_val = np.zeros(n)
                        for i in range(n):
                            if i < half_n:
                                g_full_val[i] = g_half_val[i]
                            else:
                                g_full_val[i] = g_half_val[n - 1 - i]
                        s_full = float(np.sum(g_full_val))
                        if s_full > 1e-10:
                            g_val = g_full_val / s_full
                        break
            except Exception:
                continue
        
        if g_val is None:
            break
        
        # Blend
        best_t = 0.0
        best_obj_try = best_obj
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 0.01:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_try:
                best_obj_try = obj_try
                best_t = t
        
        if best_t > 0 and best_obj_try < best_obj - 1e-12:
            a_new = (1 - best_t) * a + best_t * g_val
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        if verbose and it % 5 == 0:
            print(f"      PAL-LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}, "
                  f"elapsed={time.monotonic()-start:.1f}s")
        
        it += 1
        
        if stall >= 4:
            rng_local = np.random.default_rng(it * 41 + 13)
            noise = rng_local.normal(0, 0.003 * stall, n)
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0

    return best_a


def try_n_change(best_seq, best_score, n_try, time_budget, rng):
    """Try a different sequence length by interpolation + quick optimization."""
    n_old = len(best_seq)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_try)
    a_try = np.interp(x_new, x_old, best_seq)
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 0.01:
        return None, np.inf
    a_try = a_try / s

    # Quick optimization
    t1 = min(30.0, time_budget * 0.4)
    if t1 > 5:
        a_try = lse_gd_optimize(a_try, time_budget=t1, T_start=5.0, T_end=50.0)
    
    t2 = min(30.0, time_budget * 0.4)
    if t2 > 5:
        a_try = lp_alternation(a_try, time_budget=t2, max_iters=10, verbose=False)
    
    obj_try = compute_obj(a_try)
    return a_try, obj_try


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-method optimization:
    1. Diverse initialization + parent loading
    2. Full-constraint LP alternation (key improvement over truncated LP)
