import numpy as np

from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def softmax_conv(a, beta):
    """Compute soft-max of convolution entries."""
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    if max_c < 1e-30:
        return 0.0, conv
    shifted = beta * (conv - max_c)
    # Use log-sum-exp trick
    max_shifted = float(np.max(shifted))
    log_sum = max_c + (max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))) / beta
    return log_sum, conv


def gradient_maxconv(a, conv, beta=100.0):
    """
    Compute gradient of soft-max of conv(a,a) w.r.t. a.
    grad[j] = 2 * n * sum_k w[k] * a[k-j] * 1_{0<=k-j<n}
    where w[k] = softmax(beta * conv)[k]
    """
    n = len(a)
    max_c = float(np.max(conv))
    shifted = beta * (conv - max_c)
    max_shifted = float(np.max(shifted))
    w = np.exp(shifted - max_shifted)
    w /= np.sum(w)
    
    # grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j
    # = 2 * cross_correlation(a, w)[j]
    # Where cross_correlation(a, w)[j] = sum_m a[m] * w[m+j]
    
    # Use FFT for efficiency
    size = 1
    while size < n + 2 * n - 1:
        size <<= 1
    F = np.fft.fft(a, size)
    W = np.fft.fft(w, size)
    corr = np.fft.ifft(np.conj(F) * W).real[:n]
    
    return 2.0 * corr


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # We want to find the closest point to v on the simplex
    # Using the standard algorithm
    if np.all(v <= 0):
        return np.full(n, target_sum / n)
    
    sorted_v = np.sort(v)[::-1]
    cssv = np.cumsum(sorted_v)
    rho = np.nonzero(sorted_v * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    # Ensure sum is exactly target_sum
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-12:
        result += diff / n
        result = np.maximum(result, 0)
    return result

    Gradient descent with soft-max approximation of max(conv).

    """
    n = len(a)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Multi-phase approach:
1. Softmax gradient descent for fast initial convergence
2. Precise coordinate descent with vectorized bisection
3. LP alternating minimization for polishing
4. Multi-restart with systematic length exploration

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass
    for solver in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

    no_improve = 0
    max_no_improve = 5

                # conv_k_new = conv_k_old - old*a[k-j] - a[k-j]*old + new*a[k-j] + a[k-j]*new
                # = conv_k_old + 2*(new-old)*a[k-j] (if k-j != j)
                # = conv_k_old - old^2 + new^2 (if k-j == j, i.e., k = 2j)
                
                # But wait, we already changed a[j], so a[others] includes the new value at position j.
                # Let me redo this properly.
                
                # The correct update:
                # For k != 2j: conv_k += 2*(new_val - old_val) * a[k-j]
                #   where a[k-j] is the OLD value (since we're changing a[j], not a[k-j])
                #   BUT if k-j == j (i.e., k=2j), we need special handling
                # For k = 2j: conv_{2j} += new_val^2 - old_val^2
                
                # Since we already set a[j] = best_cand_val, we need to use the old value for a[k-j]
                # when k-j == j. But for k != 2j, a[k-j] is unchanged.
                
                # Let me just recompute the affected conv entries from scratch:
                # conv_k = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a[i] * a[k-i]
                # For affected k (k_start to k_end), the only change is in the term involving a[j]
                
                # Actually, the simplest correct approach:
                # Save old a_j, compute the change, apply it.
                # The change in conv_k due to changing a[j] from old to new:
                #   If k-j is in [0,n-1] and k-j != j: change = 2*(new-old)*a[k-j]
                #   If k-j == j (k=2j): change = new^2 - old^2
                #   (The term a[j]*a[k-j] appears in conv_k when both j and k-j are valid)
                
                # Since we've already updated a[j], let's use the formula with old_val:
                # For k in [k_start, k_end]:
                #   other = k - j
                #   if other == j: conv[k] += new_val^2 - old_val^2
                #   else: conv[k] += 2*(new_val - old_val) * a[other]  (a[other] is unchanged)
                
                # But I already overwrote a[j]! So a[other] for other != j is fine.
                # Let me just redo the conv update:
                
                # First, undo the incorrect update I just made:
                conv[k_start:k_end+1] -= 2.0 * delta * a[others]
                
                # Now do it correctly:
                for k in range(k_start, k_end + 1):
                    other = k - j
                    if other == j:
                        conv[k] = conv[k] - old_val*old_val + best_cand_val*best_cand_val
                    else:
                        conv[k] += 2.0 * delta * a[other]
                
                # Normalize
                a /= S
                conv /= (S * S)
                
                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True
        
        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.12f}, elapsed={elapsed:.1f}s")
    
    return best_a


def softmax_gd(a, time_budget, seed=42, verbose=False):
    """Softmax gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    
    beta = 30.0
    step = 0.05
    momentum = 0.85
    v = np.zeros(n)
    iter_count = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        shifted = beta * (conv - max_c)
        max_sh = float(np.max(shifted))
        w = np.exp(shifted - max_sh)
        w /= np.sum(w)
        
        # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j]
        # = 2 * cross_corr(a, w)[j]
        size = 1
        while size < n + 2 * n - 1:
            size <<= 1
        F = np.fft.fft(a, size)
        W = np.fft.fft(w, size)
        corr = np.fft.ifft(np.conj(F) * W).real[:n]
        grad = 2.0 * corr
        
        # Momentum step
        v = momentum * v - step * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s_new
        
        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            step *= 0.9995
        else:
            a_new = a.copy()  # Revert
            stall += 1
            if stall % 100 == 0:
                step *= 0.7
                v = np.zeros(n)
                beta = min(beta * 1.3, 300.0)
        
        a = a_new
        iter_count += 1
        
        if iter_count % 200 == 0:
            beta = min(beta * 1.1, 200.0)
        
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    SGD {iter_count}: obj={best_obj:.10f}, beta={beta:.0f}, step={step:.6f}, t={elapsed:.0f}s")
    
    return best_a


def lp_alternate(a, time_budget, seed=42, verbose=False):
    """LP alternating minimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    prev_g = None
    iter_count = 0
    stall = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - start < time_budget - 2:
        g_lp, M_lp = solve_lp(a, warm_start=prev_g)
        if g_lp is None:
            break
        prev_g = g_lp
        iter_count += 1
        
        best_local_obj = best_obj
        best_local_a = a.copy()
        
        for t in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try /= s_try
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
        
        if best_local_obj < best_obj - 1e-15:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 5:
                # Perturb
                pert = rng.standard_normal(n) * 0.003
                a = best_a * (1.0 + pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a /= s
                else:
                    a = best_a.copy()
                prev_g = None
                stall = 0
        
        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP {iter_count}: obj={best_obj:.10f}, t={elapsed:.0f}s")
    
    return best_a


def fourier_perturb(a, rng, strength=0.02):
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 8):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s
