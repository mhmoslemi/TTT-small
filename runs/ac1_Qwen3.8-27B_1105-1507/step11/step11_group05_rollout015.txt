from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n

"""
Multi-phase optimization for step function coefficient minimization.
Key innovation: SCA (Successive Convex Approximation) with full LP at each step,
combined with alternating LP polishing, LSE gradient descent, and coordinate descent.
The SCA approach linearizes the quadratic autocorrelation constraints at the current
point and solves the resulting LP, providing a new search direction that differs
from the previous "fix-a-solve-for-g" approach.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2 for any a (normalizes internally)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n


def sca_lp_solve(a_current, top_m=None, use_all=False):
    """
    SCA step: Linearize f_k(a) = (a*a)[k] at a_current, solve the resulting LP.
    Returns the LP solution (normalized to sum=1) or None on failure.
    """
    n = len(a_current)
    a0 = a_current
    conv_a0 = compute_conv(a0)
    num_k = 2 * n - 1

    if use_all:
        k_indices = list(range(num_k))
    elif top_m is not None:
        k_indices = np.argsort(conv_a0)[::-1][:top_m].tolist()
    else:
        # Use constraints where conv > 0.9 * max (active set)
        thresh = 0.9 * conv_a0.max()
        k_indices = np.where(conv_a0 >= thresh)[0].tolist()
        if len(k_indices) < 10:
            k_indices = np.argsort(conv_a0)[::-1][:30].tolist()

    mc = len(k_indices)
    A = np.zeros((mc, n))
    d = np.zeros(mc)

    for idx, k in enumerate(k_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[idx, lo:hi + 1] = 2.0 * a0[k - hi:k - lo + 1][::-1]
        d[idx] = conv_a0[k]

    a_var = cp.Variable(n, nonneg=True)
    q = cp.Variable()
    constraints = [A @ a_var - q <= d, cp.sum(a_var) == 1]
    prob = cp.Problem(cp.Minimize(q), constraints)

    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if a_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                a_new = np.maximum(a_var.value, 0.0)
                s = float(np.sum(a_new))
                if s > 1e-10:
                    return a_new / s
        except Exception:
            continue
    return None


def sca_iteration(a_current, top_m=None, use_all=False, blend_search=True):
    """
    One SCA iteration: solve LP, then line-search blend with current point.
    Returns (new_a, new_obj, old_obj).
    """
    old_obj = compute_obj(a_current)
    a_lp = sca_lp_solve(a_current, top_m=top_m, use_all=use_all)
    if a_lp is None:
        return a_current, old_obj, old_obj

    if not blend_search:
        new_obj = compute_obj(a_lp)
        return a_lp, new_obj, old_obj

    # Line search over blend factor
    best_t = 0.0
    best_obj = old_obj
    for t in [0.3, 0.5, 0.7, 1.0, 0.15, 0.85, 0.2, 0.4, 0.6]:
        a_try = (1.0 - t) * a_current + t * a_lp
        a_try = np.maximum(a_try, 0.0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t

    if best_t == 0.0:
        return a_current, old_obj, old_obj

    a_new = (1.0 - best_t) * a_current + best_t * a_lp
    a_new = np.maximum(a_new, 0.0)
    a_new = a_new / a_new.sum()
    return a_new, best_obj, old_obj


def lse_objective_and_grad(a, T):
    """Compute LSE-smoothed objective and gradient for normalized a."""
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)
    exp_vals = np.exp(T * (conv - cmax))
    w = exp_vals / (exp_vals.sum() + 1e-30)
    lse_val = cmax + np.log(exp_vals.sum()) / T
    obj = 2.0 * n * lse_val

    # Gradient: d/da[i] of LSE = sum_k w[k] * (a[k-i] + a[i-k]) / ... 
    # Actually grad of 2n*LSE_T = 2n * sum_k w[k] * d(conv[k])/da[i]
    # d(conv[k])/da[i] = a[k-i] (if valid) + a[i-k] (if valid)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        # contribution from a[k-i]
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += wk * a[k - hi1:k - lo1 + 1][::-1]
        # contribution from a[i-k]
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] += wk * a[lo2 - k:hi2 - k + 1]
    grad *= 2.0 * n
    return obj, grad


def lse_gradient_descent(a, time_budget, T_start=5.0, T_end=80.0, verbose=False):
    """LSE smoothed gradient descent with backtracking."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.1 / n
    it = 0
    best_it = 0
    patience = 300

    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / max(1e-6, time_budget))
        T = T_start * (T_end / T_start) ** frac

        obj_lse, grad = lse_objective_and_grad(a, T)
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-14:
            break

        a_old = a.copy()
        t = lr
        improved = False
        for _ in range(12):
            a_try = project_simplex(a_old - t * grad)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj + 1e-14:
                a = a_try
                improved = True
                break
            t *= 0.5

        if not improved:
            # Try random perturbations
            for _ in range(3):
                direction = a + np.random.randn(n) * 0.005
                a_try = project_simplex(direction)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    a = a_try
                    improved = True
                    break
            if not improved:
                # Stuck - reduce learning rate and reset
                lr *= 0.5
                if lr < 1e-8 / n:
                    break

        obj_true = compute_obj(a)
        if obj_true < best_obj:
            best_obj = obj_true
            best_a = a.copy()
            best_it = it
            lr = min(lr * 1.05, 0.5 / n)

        it += 1
        if it - best_it > patience:
            a = best_a.copy()
            lr = max(lr * 0.5, 1e-9 / n)
            patience = max(100, patience // 2)
            best_it = it

        if verbose and it % 200 == 0:
            print(f"      LSE-GD it{it}: obj={best_obj:.8f}, T={T:.1f}, "
                  f"lr={lr:.6f}, t={time.monotonic()-start:.1f}s")

    return best_a


def alternating_lp_step(a_current, M_overhead=1.0, top_m=20):
    """
    Alternating LP: fix a, find g that minimizes sum(g) subject to (a*g)[k] <= M.
    This is the approach from the previous state-of-the-art.
    """
    n = len(a_current)
    a_norm = a_current / float(np.sum(a_current))
    conv_a = compute_conv(a_norm)
    M = float(np.max(conv_a)) * M_overhead
    top_indices = np.argsort(conv_a)[::-1][:top_m]

    A = np.zeros((len(top_indices), n))
    for idx, k in enumerate(top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[idx, lo:hi + 1] = a_norm[k - hi:k - lo + 1][::-1]

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0.0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None

def coordinate_descent(a, time_budget, max_sweeps=5, verbose=False):
    """Coordinate descent with adaptive candidate values."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)

        return a

    start = time.monotonic()

    for sweep in range(max_sweeps):

            break
        sweep_improved = False

        for i in range(n):
            if time.monotonic() - start >= time_budget:
                break

            ai_old = a[i]
            if ai_old < 1e-12:
                # Try small positive values
                for cv in [0.001, 0.005]:
                    a[i] = cv
                    s_new = float(np.sum(a))
                    if s_new < 1e-10:
                        a[i] = 0
                        continue
                    obj_try = compute_obj(a)
                    if obj_try < best_obj:
                        a = a / s_new
                        best_obj = obj_try
                        best_a = a.copy()
                        sweep_improved = True
                a[i] = 0
                continue

            # Try various scales
            candidates = [0.0, ai_old * 0.2, ai_old * 0.5, ai_old * 1.5, ai_old * 2.5]
            best_val = ai_old

            for cv in candidates:
                a[i] = cv

                if s_new < 1e-10:
                    a[i] = ai_old
                    continue
                obj_try = compute_obj(a)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_val = cv
                    best_a = a.copy()
                    sweep_improved = True

            a[i] = best_val
            if best_val != ai_old:
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new

        if verbose:
            print(f"      CD sweep {sweep}: obj={best_obj:.8f}, "
                  f"t={time.monotonic()-start:.1f}s")
        if not sweep_improved:
            break

    return best_a


def diverse_init(n, mode):

    x = np.linspace(0, 1, n)

        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n / 4.0) ** 2) / (2 * (n / 8.0) ** 2)) + \

    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = np.exp(3 * (2 * x - 1) ** 2)
    elif mode == 4:

    elif mode == 5:

    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={n_current}): obj={best_score:.8f}")
    except Exception:
        print("Parent unavailable, using diverse inits")

    # Try diverse initializations to find a good starting point
    if best_seq is None:
        best_seq = diverse_init(n_current, 4)
        best_score = compute_obj(best_seq)

    for m in range(7):
        a_init = diverse_init(n_current, m)
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Init mode {m}: obj={obj_init:.8f} **")
        else:
            print(f"  Init mode {m}: obj={obj_init:.8f}")

    print(f"Best initial: obj={best_score:.8f}, n={len(best_seq)}")

    # === Phase 1: SCA iterations (main innovation) ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 1: SCA (budget={min(500, remaining * 0.35):.0f}s) ===")
        alloc = min(500.0, remaining * 0.35)
        a_sca = best_seq.copy()
        obj_sca = best_score
        it = 0
        stall = 0
        s_start = time.monotonic()

        while time.monotonic() - s_start < alloc:
            # Use top-m constraints, expand m as we go
            frac = min(1.0, (time.monotonic() - s_start) / alloc)
            if frac < 0.3:
                top_m = 20
            elif frac < 0.6:
                top_m = 50
            else:
                top_m = None  # use active set

            a_new, obj_new, obj_old = sca_iteration(a_sca, top_m=top_m, blend_search=True)

            if obj_new < obj_sca - 1e-12:
                a_sca = a_new
                obj_sca = obj_new
                stall = 0
                if obj_sca < best_score:
                    best_score = obj_sca
                    best_seq = a_sca.copy()
                    if it % 20 == 0:
                        print(f"      SCA it{it}: obj={best_score:.8f}, "
                              f"t={time.monotonic()-s_start:.1f}s **")
            else:
                stall += 1
                if stall == 5:
                    # Perturb and retry
                    noise = rng.normal(0, 0.002, len(a_sca))
                    a_p = a_sca + noise
                    a_p = np.maximum(a_p, 0)
                    s_p = float(np.sum(a_p))
                    if s_p > 1e-10:
                        a_sca = a_p / s_p
                        obj_sca = compute_obj(a_sca)
                    stall = 0
                elif stall >= 15:
                    break

            it += 1

        print(f"  SCA done: {it} iters, best={best_score:.8f}")

    # === Phase 2: Alternating LP ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 2: Alternating LP (budget={min(300, remaining * 0.25):.0f}s) ===")
        alloc = min(300.0, remaining * 0.25)
        a_alt = alternating_lp_loop(best_seq, time_budget=alloc, verbose=True)
        obj_alt = compute_obj(a_alt)
        if obj_alt < best_score:
            best_score = obj_alt
            best_seq = a_alt.copy()
            print(f"  AltLP: obj={best_score:.8f} **")
        else:
            print(f"  AltLP: obj={obj_alt:.8f}")

    # === Phase 3: SCA round 2 (refined) ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 3: SCA Round 2 (budget={min(250, remaining * 0.2):.0f}s) ===")
        alloc = min(250.0, remaining * 0.2)
        a_sca2 = best_seq.copy()
        obj_sca2 = best_score
        s_start2 = time.monotonic()
        it2 = 0
        stall2 = 0

        while time.monotonic() - s_start2 < alloc:
            top_m = None  # active set
            a_new, obj_new, _ = sca_iteration(a_sca2, top_m=top_m, blend_search=True)
            if obj_new < obj_sca2 - 1e-12:
                a_sca2 = a_new
                obj_sca2 = obj_new
                stall2 = 0
                if obj_sca2 < best_score:
                    best_score = obj_sca2
                    best_seq = a_sca2.copy()
                    if it2 % 10 == 0:
                        print(f"      SCA2 it{it2}: obj={best_score:.8f}, "
                              f"t={time.monotonic()-s_start2:.1f}s **")
            else:
                stall2 += 1
                if stall2 >= 10:
                    break
            it2 += 1

        print(f"  SCA2 done: {it2} iters, best={best_score:.8f}")

    # === Phase 4: LSE Gradient Descent ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 4: LSE GD (budget={min(200, remaining * 0.15):.0f}s) ===")
        alloc = min(200.0, remaining * 0.15)
        a_lse = lse_gradient_descent(best_seq, time_budget=alloc,
                                     T_start=10.0, T_end=100.0, verbose=True)
        obj_lse = compute_obj(a_lse)
        if obj_lse < best_score:
            best_score = obj_lse
            best_seq = a_lse.copy()
            print(f"  LSE-GD: obj={best_score:.8f} **")
        else:
            print(f"  LSE-GD: obj={obj_lse:.8f}")

    # === Phase 5: Coordinate Descent ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 5: Coordinate Descent (budget={min(150, remaining * 0.12):.0f}s) ===")
        alloc = min(150.0, remaining * 0.12)
        a_cd = coordinate_descent(best_seq, time_budget=alloc, max_sweeps=3, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 6: n exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 6: n Exploration ===")
        n_best = len(best_seq)
        n_options = []
        for delta in [-100, -50, 50, 100, 200]:
            n_try = n_best + delta
            if 300 <= n_try <= 2500 and n_try not in n_options:
                n_options.append(n_try)
        n_options = n_options[:4]
        print(f"  Current n={n_best}, trying: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            # Interpolate
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s > 1e-10:
                a_try = a_try / s
                obj_try = compute_obj(a_try)
                print(f"  n={n_try} (interp): obj={obj_try:.8f}")

                # Quick SCA
                t_alloc = min(30.0, remaining * 0.3)
                s_s = time.monotonic()
                while time.monotonic() - s_s < t_alloc:
                    a_new, obj_new, _ = sca_iteration(a_try, top_m=30, blend_search=True)
                    if obj_new < compute_obj(a_try) - 1e-12:
                        a_try = a_new
                    else:
                        break
                obj_try2 = compute_obj(a_try)
                if obj_try2 < obj_try:
                    obj_try = obj_try2
                    print(f"  n={n_try} (SCA): obj={obj_try:.8f}")

                if obj_try < best_score:
                    best_score = obj_try
                    best_seq = a_try.copy()
                    n_best = n_try
                    print(f"  ** n={n_try}: obj={best_score:.8f} **")
                break  # Only try n improvement if it helps, then stop

    # === Phase 7: Final combined polish ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 7: Final Polish (budget={remaining:.0f}s) ===")

        # SCA
        if remaining > 30:
            a_f1 = best_seq.copy()
            s_f = time.monotonic()
            while time.monotonic() - s_f < min(remaining * 0.3, 40.0):
                a_new, obj_new, _ = sca_iteration(a_f1, top_m=None, blend_search=True)
                if obj_new < compute_obj(a_f1) - 1e-13:
                    a_f1 = a_new
                    if obj_new < best_score:
                        best_score = obj_new
                        best_seq = a_f1.copy()
                        print(f"  Final SCA: obj={best_score:.8f} **")
                else:
                    break

        # LSE GD
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_f2 = lse_gradient_descent(best_seq, time_budget=min(remaining * 0.25, 30.0),
                                        T_start=30.0, T_end=150.0)
            obj_f2 = compute_obj(a_f2)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final LSE: obj={best_score:.8f} **")

        # CD
        remaining = deadline - time.monotonic()
        if remaining > 25:
            a_f3 = coordinate_descent(best_seq, time_budget=min(remaining * 0.3, 30.0),
                                      max_sweeps=2)
            obj_f3 = compute_obj(a_f3)
            if obj_f3 < best_score:
                best_score = obj_f3
                best_seq = a_f3.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
        print(f"\n  Validated score: {best_score:.8f}")
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
