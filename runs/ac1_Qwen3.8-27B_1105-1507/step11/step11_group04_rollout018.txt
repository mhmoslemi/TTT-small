def compute_conv(a):

    n = len(a)

    temp = temp_init
    eta = 0.5  # initial step size
    iter_count = 0
    stall = 0
    max_stall = 30

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        conv_max = float(np.max(conv))

        # Compute softmax weights (smooth max)
        shifted = (conv - conv_max) / temp
        shifted -= np.max(shifted)  # for numerical stability
        p = np.exp(shifted)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros_like(conv)
            p[np.argmax(conv)] = 1.0
        else:
            p = p / p_sum

        # Gradient of smooth max w.r.t. a[j] = correlate(p, a, 'valid')[j]
        # Full gradient of objective = 2*n * that
        grad = 2.0 * n * np.correlate(p, a, mode='valid')

        # Backtracking line search
        best_local_obj = best_obj
        best_local_a = a.copy()
        eta_try = eta

        for _ in range(15):
            a_try = a - eta_try * grad
            a_try = simplex_project(a_try)
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_local_obj - 1e-15:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                break
            eta_try *= 0.5

        if best_local_obj < best_obj - 1e-15:
            a = best_local_a.copy()
            best_a = a.copy()
            best_obj = best_local_obj
            stall = 0
            # Slightly increase step size on success
            eta = min(eta * 1.05, 5.0)
        else:
            stall += 1
            eta *= 0.7

            if stall >= max_stall:
                # Random perturbation to escape
                sigma = 0.01 + 0.02 * min(stall / max_stall, 5)
                pert = rng.standard_normal(n) * sigma
                a = best_a * (1.0 + pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                eta = 0.5
                temp = max(temp * 0.5, temp_min)
                stall = 0
                max_stall = min(max_stall + 10, 100)

        # Geometric cooling
        elapsed_frac = (time.monotonic() - start) / time_budget
        temp = max(temp_min, temp_init * (temp_min / temp_init) ** elapsed_frac)

        iter_count += 1
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - start
            rem = time_budget - elapsed
            print(f"    SGD {iter_count}: obj={best_obj:.10f}, temp={temp:.6f}, eta={eta:.3f}, rem={rem:.0f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass

    for solver in ['CLARABEL', 'ECOS', 'SCS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a, time_budget, seed=42, verbose=True):
    """LP alternating minimization with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    iter_count = 0
    stall = 0

    while time.monotonic() - start < time_budget - 3:
        g_lp, M_lp = solve_lp(a, warm_start=prev_g)
        if g_lp is None:
            break
        prev_g = g_lp
        iter_count += 1

        best_local_obj = best_obj
        best_local_a = a.copy()

        t_values = [1.0, 0.8, 0.6, 0.4, 0.2]
        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_local_obj - 1e-15:
                best_local_obj = obj_try
                best_local_a = a_try.copy()

        if best_local_obj < best_obj - 1e-15:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall = 0
            if verbose and iter_count % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP {iter_count}: obj={best_obj:.10f}, rem={time_budget-elapsed:.0f}s")
        else:

            if stall >= 6:
                strength = 0.01
                pert = rng.standard_normal(n) * strength
                a = best_a * (1.0 + pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                prev_g = None
                stall = 0

    if verbose:
        print(f"    LP done: {iter_count} iters, best={best_obj:.10f}")
    return best_a


def coordinate_descent_polish(a, time_budget, seed=42, verbose=False):
    """
    Coordinate descent on normalized sequence (sum=1).
    For each coordinate, try several candidate values and pick the best.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    sweep = 0
    no_improve = 0
    max_no_improve = 4
    mean_val = 1.0 / n

    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            a_j = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)

            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))

            # Try several candidate values
            candidates = np.array([0.0, mean_val * 0.1, mean_val * 0.5, mean_val, mean_val * 1.5, mean_val * 3])
            if a_j > 1e-12:
                extra = np.array([a_j * 0.5, a_j * 0.8, a_j * 1.2, a_j * 2.0])
                candidates = np.concatenate([candidates, extra])
            candidates = np.unique(candidates[(candidates >= 0) & (candidates <= 0.2)])

            best_trial_obj = best_obj
            best_trial_val = a_j

            for val in candidates:
                delta = val - a_j
                if abs(delta) < 1e-15:
                    continue
                S_new = 1.0 + delta
                if S_new <= 1e-10:
                    continue

                indices = np.arange(k_start, k_end + 1)
                affected = conv[k_start:k_end + 1].copy()
                mask = (indices == 2 * j)
                if np.any(~mask):
                    affected[~mask] = conv[k_start:k_end + 1][~mask] + 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    affected[mask] = conv[2 * j] + (val * val - a_j * a_j)

                max_conv_trial = max(max_unaffected, float(np.max(affected)))
                obj_trial = 2.0 * n * max_conv_trial / (S_new * S_new)

                if obj_trial < best_trial_obj - 1e-14:
                    best_trial_obj = obj_trial
                    best_trial_val = val

            if best_trial_val != a_j:
                delta = best_trial_val - a_j
                a[j] = best_trial_val
                S_new = 1.0 + delta

                indices = np.arange(k_start, k_end + 1)
                mask = (indices == 2 * j)
                if np.any(~mask):
                    conv[k_start:k_end + 1][~mask] += 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    conv[2 * j] += (best_trial_val * best_trial_val - a_j * a_j)

                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S

                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

    return best_a


def random_init(n, rng, mode='uniform'):
    """Generate random initial sequence of length n."""
    if mode == 'uniform':
        return np.ones(n) / n
    elif mode == 'exponential':
        a = rng.exponential(1.0, size=n)
    elif mode == 'gaussian_pos':
        a = np.abs(rng.standard_normal(n)) + 0.1
    elif mode == 'sine':
        x = np.arange(n)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * rng.uniform(1, 5) * x / n + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
    elif mode == 'bimodal':
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)  # Wait, x not defined
        x = np.arange(n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * rng.uniform(1, 4) * x / n)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n) / n
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth gradient descent (log-sum-exp smoothing of the max) as the main workhorse,
combined with LP alternating minimization, coordinate descent polishing, and multi-restart
exploration with diverse initializations and length adaptation.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project vector v onto probability simplex {x: sum(x)=1, x>=0} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient_descent(a, time_budget, seed=42, verbose=True):
    """
    Smooth gradient descent minimizing 2*n*max(conv(a,a)) s.t. sum(a)=1, a>=0.
    Uses log-sum-exp smoothing with adaptive relative temperature and backtracking line search.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    alpha = 0.4  # relative temperature: temp = alpha * conv_max
    alpha_min = 0.005
    eta = 1.0
    iter_count = 0
    stall = 0
    max_stall = 40

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        conv_max = float(np.max(conv))
        if conv_max < 1e-20:
            break

        temp = alpha * conv_max
        shifted = (conv - conv_max) / temp
        shifted -= np.max(shifted)
        p = np.exp(shifted)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros_like(conv)
            p[np.argmax(conv)] = 1.0
        else:
            p = p / p_sum

        # Gradient via cross-correlation
        grad = 2.0 * n * np.correlate(p, a, mode='valid')

        # Backtracking line search
        best_local_obj = best_obj
        best_local_a = a.copy()
        eta_try = eta

        for _ in range(20):
            a_try = a - eta_try * grad
            a_try = simplex_project(a_try)
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_local_obj - 1e-15:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                break
            eta_try *= 0.5

        if best_local_obj < best_obj - 1e-15:
            a = best_local_a.copy()
            best_a = a.copy()
            best_obj = best_local_obj
            stall = 0
            eta = min(eta * 1.08, 10.0)
            # Slowly cool
            alpha = max(alpha_min, alpha * 0.998)
        else:
            stall += 1
            eta *= 0.6

            if stall >= max_stall:
                sigma = 0.01 + 0.015 * min(stall / max_stall, 5.0)
                pert = rng.standard_normal(n) * sigma
                a = best_a * (1.0 + pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                eta = 1.0
                alpha = max(alpha * 0.7, alpha_min)
                stall = 0
                max_stall = min(max_stall + 15, 150)

        iter_count += 1
        if verbose and iter_count % 300 == 0:
            elapsed = time.monotonic() - start
            rem = time_budget - elapsed
            print(f"    SGD {iter_count}: obj={best_obj:.10f}, alpha={alpha:.5f}, eta={eta:.2f}, rem={rem:.0f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass

    for solver in ['CLARABEL', 'ECOS', 'SCS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a, time_budget, seed=42, verbose=True):
    """LP alternating minimization with adaptive step size and perturbation restart."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    iter_count = 0
    stall = 0

    while time.monotonic() - start < time_budget - 5:
        g_lp, M_lp = solve_lp(a, warm_start=prev_g)
        if g_lp is None:
            break
        prev_g = g_lp
        iter_count += 1

        best_local_obj = best_obj
        best_local_a = a.copy()

        t_values = [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]
        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_local_obj - 1e-15:
                best_local_obj = obj_try
                best_local_a = a_try.copy()

        if best_local_obj < best_obj - 1e-15:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall = 0
            if verbose and iter_count % 15 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP {iter_count}: obj={best_obj:.10f}, rem={time_budget - elapsed:.0f}s")
        else:
            stall += 1
            if stall >= 5:
                strength = 0.008 + 0.004 * min(stall, 8)
                pert = rng.standard_normal(n) * strength
                a = best_a * (1.0 + pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                prev_g = None
                stall = 0

    if verbose:
        print(f"    LP done: {iter_count} iters, best={best_obj:.10f}")
    return best_a


def coordinate_descent_polish(a, time_budget, seed=42, verbose=False):
    """Coordinate descent on normalized sequence exploiting convolution structure."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    sweep = 0
    no_improve = 0
    max_no_improve = 4
    mean_val = 1.0 / n

    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            a_j = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)

            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))

            candidates = np.array([0.0, mean_val * 0.05, mean_val * 0.2, mean_val * 0.5,
                                   mean_val, mean_val * 1.5, mean_val * 2.5, mean_val * 5.0])
            if a_j > 1e-12:
                extra = np.array([a_j * 0.3, a_j * 0.6, a_j * 0.85, a_j * 1.15, a_j * 1.5, a_j * 2.5])
                candidates = np.concatenate([candidates, extra])
            candidates = np.unique(candidates[(candidates >= 0) & (candidates <= 0.15)])

            best_trial_obj = best_obj
            best_trial_val = a_j

            for val in candidates:
                delta = val - a_j
                if abs(delta) < 1e-15:
                    continue
                S_new = 1.0 + delta
                if S_new <= 1e-10:
                    continue

                indices = np.arange(k_start, k_end + 1)
                affected = conv[k_start:k_end + 1].copy()
                mask = (indices == 2 * j)
                if np.any(~mask):
                    affected[~mask] = conv[k_start:k_end + 1][~mask] + 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    affected[mask] = conv[2 * j] + (val * val - a_j * a_j)

                max_conv_trial = max(max_unaffected, float(np.max(affected)))
                obj_trial = 2.0 * n * max_conv_trial / (S_new * S_new)

                if obj_trial < best_trial_obj - 1e-14:
                    best_trial_obj = obj_trial
                    best_trial_val = val

            if best_trial_val != a_j:
                delta = best_trial_val - a_j
                a[j] = best_trial_val
                S_new = 1.0 + delta

                indices = np.arange(k_start, k_end + 1)
                mask = (indices == 2 * j)
                if np.any(~mask):
                    conv[k_start:k_end + 1][~mask] += 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    conv[2 * j] += (best_trial_val * best_trial_val - a_j * a_j)

                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S

                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

    return best_a


def random_init(n, rng, mode='uniform'):
    """Generate random initial sequence of length n (normalized to sum=1)."""
    if mode == 'uniform':
        a = np.ones(n)
    elif mode == 'exponential':
        a = rng.exponential(1.0, size=n)
    elif mode == 'gaussian':
        a = np.abs(rng.standard_normal(n)) + 0.05
    elif mode == 'sine':
        x = np.arange(n)
        freq = rng.uniform(1.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.6 * np.sin(2.0 * np.pi * freq * x / n + phase)
        a = np.maximum(a, 0.01)
    elif mode == 'cosine_envelope':
        x = np.arange(n)
        a = 0.5 + 0.5 * np.cos(np.pi * x / (n - 1))
        a = a + 0.1 * np.abs(rng.standard_normal(n))
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
    return a / np.sum(a)


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization: smooth gradient descent + LP alternating + CD polishing
    with multi-restart exploration, diverse initializations, and length adaptation.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    rng = np.random.default_rng(seed)

    # Load parent
    try:
        a_parent = np.asarray(height_sequence_1, dtype=np.float64)
        a_parent = np.maximum(a_parent, 0)
        if a_parent.size == 0 or a_parent.sum() < 0.01:
            raise ValueError("Invalid parent")
        a_parent = a_parent / a_parent.sum()
        parent_obj = compute_obj_norm(a_parent)
        print(f"Parent: n={a_parent.size}, obj={parent_obj:.10f}")
    except Exception:
        a_parent = np.ones(800) / 800.0
        parent_obj = 2.0
        print("No parent, using uniform n=800.")

    best_a = a_parent.copy()
    best_obj = compute_obj_norm(best_a)
    n_best = len(best_a)
    print(f"Starting: n={n_best}, obj={best_obj:.10f}")

    # === Phase 1: Smooth GD on parent (35% time) ===
    print("\n--- Phase 1: Smooth GD ---")
    p1_budget = (budget_s - 30) * 0.35
    a1 = smooth_gradient_descent(best_a, time_budget=p1_budget, seed=seed, verbose=True)
    obj1 = compute_obj_norm(a1)
    if obj1 < best_obj:
        best_obj = obj1
        best_a = a1.copy()
        n_best = len(best_a)
        print(f"  P1: {best_obj:.10f} **")
    else:
        print(f"  P1: {obj1:.10f} (no improvement)")

    # === Phase 2: LP Alternating (20% time) ===
    print("\n--- Phase 2: LP ---")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        p2_budget = min(remaining * 0.45, (budget_s - 30) * 0.22)
        a2 = lp_alternate(best_a, time_budget=p2_budget, seed=seed + 100, verbose=True)
        obj2 = compute_obj_norm(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            print(f"  P2: {best_obj:.10f} **")
        else:
            print(f"  P2: {obj2:.10f}")

    # === Phase 3: Smooth GD from perturbed state (15% time) ===
    print("\n--- Phase 3: SGD Restart ---")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        p3_budget = min(remaining * 0.40, (budget_s - 30) * 0.18)
        a_pert = best_a * (1.0 + rng.standard_normal(n_best) * 0.025)
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        a3 = smooth_gradient_descent(a_pert, time_budget=p3_budget, seed=seed + 500, verbose=False)
        obj3 = compute_obj_norm(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            print(f"  P3: {best_obj:.10f} **")
        else:
            print(f"  P3: {obj3:.10f}")

    # === Phase 4: CD Polish (8% time) ===
    print("\n--- Phase 4: CD Polish ---")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p4_budget = min(remaining * 0.5, (budget_s - 30) * 0.10)
        a4 = coordinate_descent_polish(best_a, time_budget=p4_budget, seed=seed + 800, verbose=False)
        obj4 = compute_obj_norm(a4)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()
            print(f"  P4: {best_obj:.10f} **")
        else:
            print(f"  P4: {obj4:.10f}")

    # === Phase 5: Multi-restart with diverse inits and lengths (22% time) ===
    print("\n--- Phase 5: Multi-restart ---")
    remaining = deadline - time.monotonic()
    if remaining > 120:
        n_target = n_best
        attempts = [
            ('exponential', n_target),
            ('gaussian', n_target),
            ('sine', n_target),
            ('cosine_envelope', n_target),
            ('sine', n_target + 100),
            ('exponential', n_target - 60),
        ]

        per_attempt = remaining / (len(attempts) + 1)

        for i, (mode, n_try) in enumerate(attempts):
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            if n_try < 200:
                continue

            a_init = random_init(n_try, rng, mode)
            alloc = min(per_attempt * 0.7, remaining * 0.35)

            if alloc > 30:
                a_r = smooth_gradient_descent(a_init, time_budget=alloc, seed=seed + 2000 + i * 100, verbose=False)
                remaining = deadline - time.monotonic()
                if remaining > 20:
                    a_r = coordinate_descent_polish(a_r, time_budget=min(remaining * 0.25, 30),
                                                    seed=seed + 3000 + i * 100, verbose=False)

            obj_r = compute_obj_norm(a_r)
            if obj_r < best_obj:
                best_obj = obj_r
                best_a = a_r.copy()
                n_best = len(best_a)
                print(f"  Restart {mode} n={n_try}: {best_obj:.10f} **")
            else:
                print(f"  Restart {mode} n={n_try}: {obj_r:.10f}")

    # === Phase 6: Final CD polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n--- Phase 6: Final Polish ---")
        p6_budget = min(remaining * 0.7, 50.0)
        a_fin = coordinate_descent_polish(best_a, time_budget=p6_budget, seed=seed + 77777, verbose=False)
        obj_fin = compute_obj_norm(a_fin)
        if obj_fin < best_obj:
            best_obj = obj_fin
            best_a = a_fin.copy()
            print(f"  Final: {best_obj:.10f} **")
        else:
            print(f"  Final: {obj_fin:.10f}")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        print(f"\n=== FINAL: n={len(result)}, score={val:.10f} ===")
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        result = [1.0] * 100
        val = evaluate_sequence(result)
        print(f"  Fallback: n=100, score={val:.10f}")

    elapsed = time.monotonic() - start_time
    print(f"=== DONE: elapsed={elapsed:.1f}s ===")
    return result
