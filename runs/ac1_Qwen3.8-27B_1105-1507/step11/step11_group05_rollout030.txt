from scipy.optimize import linprog
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj_val(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def compute_obj_normalized(a_norm):
    """For a_norm with sum=1, compute 2n*max(conv(a_norm, a_norm))."""
    n = len(a_norm)
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_nonneg_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Use the standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n

def subgradient_max_conv(a_norm):
    """Compute subgradient of max(conv(a,a)) at a_norm (sum=1)."""
    n = len(a_norm)
    conv = compute_conv(a_norm)
    # Find all positions where conv is within epsilon of max
    max_val = np.max(conv)
    # Use soft max for numerical stability: weight by exp(T*(conv-max))
    T = 100.0
    weights = np.exp(T * (conv - max_val))
    weights /= np.sum(weights)
    
    # grad[i] = sum_k w[k] * (a[k-i] + a[i-k]) for valid indices
    grad = np.zeros(n)
    for i in range(n):
        # Term 1: sum_k w[k] * a[k-i] where 0 <= k-i < n
        lo1 = max(0, i)
        hi1 = min(2*n-2, i+n-1)
        if lo1 <= hi1:
            grad[i] += np.dot(weights[lo1:hi1+1], a_norm[lo1-i:hi1-i+1])
        # Term 2: sum_k w[k] * a[i-k] where 0 <= i-k < n

        if lo2 <= hi2:

            grad[i] += np.dot(weights[lo2:hi2+1], a_norm[i-hi2:i-lo2+1][::-1])
    return grad

"""
Optimized step function coefficient search using:
1. FFT-based fast convolution and subgradient computation
2. Constraint-generation LP (separation oracle) with HiGHS solver
3. LSE smoothed gradient descent
4. Adaptive line search and n-exploration
"""
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = max(len(a), len(b))
    total = len(a) + len(b) - 1
    size = 1
    while size < total:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:total]

def obj_val(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * np.max(conv) / (s * s)

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    return a / s

def project_simplex(v):
    """Project onto {x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    r = idx[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n

def separate_oracle(a, g):
    """Check if g violates any constraint (a*g)[k] > M. Return worst violation index or -1."""
    cross = compute_conv_cross(a, g)
    M = np.max(cross)  # This is the actual max
    # But we need to compare against the M from the LP solution
    # Actually, for the separation oracle, we just need to find if there's a k
    # where (a*g)[k] > M_lp
    # Let me return the index of the max
    return int(np.argmax(cross)), float(M)

def lp_solve_constraints(a, constraint_indices):
    """
    Solve LP: min M
    s.t. (a*g)[k] <= M for k in constraint_indices
         sum(g) = 1
         g >= 0
    Returns (g, M) or (None, None) if infeasible.
    """
    n = len(a)
    m = len(constraint_indices)
    if m == 0:
        return np.ones(n)/n, 0.0
    
    # Variables: g[0..n-1], M (total n+1 variables)
    # Objective: minimize M -> c = [0,0,...,0,1]
    c = np.zeros(n+1)
    c[-1] = 1.0
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Inequalities: (a*g)[k_i] - M <= 0 for each constraint i
    # (a*g)[k] = sum_j a[j]*g[k-j] = sum_j a[k-j]*g[j] (reindexing)
    # So for a given k: coefficient of g[j] is a[k-j] (if 0 <= k-j < n), else 0
    # Coefficient of M is -1
    A_ub = np.zeros((m, n+1))
    A_ub[:, -1] = -1.0
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        for j in range(n):
            jk = k - j
            if 0 <= jk < n:
                A_ub[idx, j] = a[jk]
    b_ub = np.zeros(m)
    
    # Bounds
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = g.sum()
            if s > 1e-10:
                g = g / s
            M = float(res.x[-1])
            return g, M
    except Exception:
        pass
    return None, None

def lp_constraint_generation(a, max_iters=50, init_top_m=10, max_constraints=50):
    """
    Solve the full LP using constraint generation (separation oracle).
    Start with top-m constraints, then add violated ones.
    """
    n = len(a)
    conv_a = compute_conv(a)
    
    # Initialize with top-m constraints
    top_m = min(init_top_m, 2*n-2)
    active_set = list(np.argsort(conv_a)[::-1][:top_m])
    
    best_g = None
    best_M = np.inf
    
    for it in range(max_iters):
        if len(active_set) > max_constraints:
            # Keep only the most important ones
            conv_a = compute_conv(a)
            active_set = list(np.argsort(conv_a)[::-1][:max_constraints])
        
        g, M = lp_solve_constraints(a, active_set)
        if g is None:
            break
        
        if M < best_M:
            best_M = M
            best_g = g.copy()
        
        # Separation: check if g violates any constraint
        cross = compute_conv_cross(a, g)
        max_idx = int(np.argmax(cross))
        max_val = float(cross[max_idx])
        
        if max_val <= M + 1e-10:
            # Feasible - this is the optimal solution
            return g, M
        
        # Add the violated constraint
        if max_idx not in active_set:
            active_set.append(max_idx)
    
    return best_g, best_M

def lse_smoothed_grad_descent(a, time_budget, T_start=3.0, T_end=50.0):
    """LSE smoothed gradient descent with backtracking."""
    n = len(a)
    a = normalize(a)
    best_a = a.copy()
    best_obj = 2.0 * n * np.max(compute_conv(a))
    start = time.monotonic()
    lr = 0.5 / n
    patience_count = 0
    patience = 200
    
    while time.monotonic() - start < time_budget:
        elapsed_frac = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** elapsed_frac
        
        # Compute LSE gradient
        conv = compute_conv(a)
        cmax = np.max(conv)
        exp_vals = np.exp(T * (conv - cmax))
        w = exp_vals / exp_vals.sum()
        
        # Gradient of 2n * LSE_T(conv(a,a))
        # d/d(a[i]) of LSE = sum_k w[k] * d(conv[k])/d(a[i])
        # d(conv[k])/d(a[i]) = a[k-i] + a[i-k] (valid indices)
        grad = np.zeros(n)
        for i in range(n):
            # Term 1: sum_k w[k] * a[k-i] for valid k-i
            lo = max(0, i)
            hi = min(2*n-2, i+n-1)
            if lo <= hi:
                grad[i] += np.dot(w[lo:hi+1], a[lo-i:hi-i+1])
            # Term 2: sum_k w[k] * a[i-k] for valid i-k
            lo2 = max(0, i-n+1)
            hi2 = min(2*n-2, i)
            if lo2 <= hi2:
                grad[i] += np.dot(w[lo2:hi2+1], a[i-hi2:i-lo2+1][::-1])
        
        grad *= 2.0 * n
        
        # Backtracking line search
        obj_old = 2.0 * n * (cmax + np.log(exp_vals.sum()) / T)
        t = lr
        improved = False
        for _ in range(10):
            a_try = a - t * grad
            a_try = project_simplex(a_try)
            conv_try = compute_conv(a_try)
            cmax_try = np.max(conv_try)
            exp_try = np.exp(T * (conv_try - cmax_try))
            obj_try = 2.0 * n * (cmax_try + np.log(exp_try.sum()) / T)
            if obj_try < obj_old - 1e-15:
                a = a_try
                improved = True
                break
            t *= 0.5
        
        if not improved:
            patience_count += 1
            if patience_count > patience:
                break
            lr *= 0.7
        else:
            patience_count = 0
            lr = t * 1.05
        
        # Check true objective
        true_obj = 2.0 * n * np.max(compute_conv(a))
        if true_obj < best_obj:
            best_obj = true_obj
            best_a = a.copy()
    
    return best_a

def fast_subgrad_descent(a, time_budget, top_k=3):
    """Fast subgradient descent on the max-conv objective."""
    n = len(a)
    a = normalize(a)
    best_a = a.copy()
    best_obj = 2.0 * n * np.max(compute_conv(a))
    start = time.monotonic()
    lr = 0.8 / n
    momentum = np.zeros(n)
    beta = 0.5
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = np.max(conv)
        
        # Multi-peak subgradient
        # Use top-k peaks with exponential weighting
        sorted_idx = np.argsort(conv)[::-1][:top_k]
        sorted_vals = conv[sorted_idx]
        T_sq = 50.0
        peak_weights = np.exp(T_sq * (sorted_vals - sorted_vals[0]))
        peak_weights /= peak_weights.sum()
        
        grad = np.zeros(n)
        for ki, pw in zip(sorted_idx, peak_weights):
            ki = int(ki)
            for i in range(n):
                # d(conv[ki])/d(a[i]) = a[ki-i] + a[i-ki] (valid)
                j1 = ki - i
                if 0 <= j1 < n:
                    grad[i] += pw * a[j1]
                j2 = i - ki
                if 0 <= j2 < n:
                    grad[i] += pw * a[j2]
        
        # Update with momentum
        momentum = beta * momentum - lr * grad
        a_try = project_simplex(a + momentum)
        
        obj_try = 2.0 * n * np.max(compute_conv(a_try))
        if obj_try < best_obj - 1e-14:
            a = a_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            momentum *= 0.5
            lr *= 0.85
        
        if stall > 500:
            break
    
    return best_a

def coordinate_descent_fast(a, time_budget):

def diverse_init(n, mode, rng):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = (2*x - 1)**2 + 0.1
    elif mode == 2:
        a = np.exp(-5*(2*x-1)**2)  # Peak in middle
    elif mode == 3:
        a = 1.0 / (0.1 + (2*x-1)**2)  # U-shaped
    elif mode == 4:
        a = np.abs(np.sin(np.pi * x)) ** 0.5
    elif mode == 5:
        a = np.cos(np.pi * (x - 0.5)) ** 2 + 0.05
    elif mode == 6:
        # Random with slight structure
        a = np.abs(rng.normal(0, 1, n)) * (1 + 0.5*np.cos(2*np.pi*x))
    elif mode == 7:
        a = 1.0 / (0.3 + abs(2*x-1)**3)
    else:
        a = np.ones(n)
    return normalize(a)
