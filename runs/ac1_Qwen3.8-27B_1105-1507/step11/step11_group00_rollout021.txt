def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.full(n, 1.0/n)
    
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]

For k < n-1: the sum is over j from max(0, k-(n-1)) to min(n-1, k), i.e., j from 0 to k (since k < n-1 means k-(n-1) < 0).
So (a*a)[k] = sum_{j=0}^{k} a[j]*a[k-j] for 0 ≤ k ≤ n-1.

For k > n-1: j from k-(n-1) to n-1.
(a*a)[k] = sum_{j=k-n+1}^{n-1} a[j]*a[k-j] for n ≤ k ≤ 2n-2.

At k = n-1: sum_{j=0}^{n-1} a[j]*a[n-1-j]. This is the "inner product" of a and its reverse.

Now, the key point: this at the center is <a, a_rev> = sum a[j]*a[n-1-j].

By Cauchy-Schwarz: <a, a_rev> ≤ ||a|| * ||a_rev|| = ||a||^2 = sum a[i]^2.
And by the rearrangement inequality: <a, a_rev> is minimized when a is sorted in one order and a_rev in the opposite, i.e., when a is monotone. So for a sorted in decreasing order, <a, a_rev> is minimized.

So to minimize the center peak, we want a to be monotone (or as monotone as possible)!

But then the edge peaks (k=0: a[0]^2, k=2n-2: a[n-1]^2) would be large if a[0] or a[n-1] is large.

The optimal balance is to have a "smooth" shape that minimizes the maximum of the convolution.

This is actually a well-studied problem! It's related to "minimum peak correlation" sequences and "flat top" filter design.

def conv_full(a):

def smooth_grad(p, tau):

def lp_frank_wolfe(p, time_budget, verbose=False, max_iters=50):
    """
    LP-driven Frank-Wolfe (Conditional Gradient) method.
    Solves: min_g M s.t. 2*(A@g)_k - (conv(p,p))_k <= M, sum(g)=1, g>=0
    Then takes a line search step from p toward g.
    """
    import cvxpy as cp

    n = len(p)
    p = np.maximum(p, 0).copy().astype(np.float64)
    s = float(np.sum(p))
    if s < 1e-10:
        return p
    p = p / s

    best_p = p.copy()
    best_obj = obj_val(p)
    start = time.monotonic()

    # Build constraint matrix A: (A@g)[k] = conv(p,g)[k]
    A = np.zeros((2 * n - 1, n))
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A[valid] = p[diff[valid]]

    conv_pp = conv_full(p)
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    for it in range(max_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break
        remaining = time_budget - elapsed
        if remaining < 3.0:
            break

        # Solve LP: min M s.t. 2*A@g - M <= conv_pp, sum(g)=1, g>=0
        constraints = [2.0 * (A @ g_var) - M_var <= conv_pp, cp.sum(g_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        g_val = None
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver, max_time=5.0)
                if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g_var.value, 0)
                    sg = float(np.sum(g_val))
                    if sg > 1e-10:
                        g_val = g_val / sg
                    break
            except Exception:
                continue

        if g_val is None:
            break

        # Line search: find best t in [0,1] for (1-t)*p + t*g
        ts = np.array([0.1, 0.2, 0.3, 0.5, 0.7, 0.9])
        best_t = 0.0
        best_obj_t = obj_val(p)
        for t in ts:
            mix = (1 - t) * p + t * g_val
            mix = np.maximum(mix, 0)
            sm = float(np.sum(mix))
            if sm > 1e-10:
                mix = mix / sm
                obj_t = obj_val(mix)
                if obj_t < best_obj_t:
                    best_obj_t = obj_t
                    best_t = t

        if best_t > 0 and best_obj_t < best_obj - 1e-12:
            p_new = (1 - best_t) * p + best_t * g_val
            p_new = np.maximum(p_new, 0)
            sn = float(np.sum(p_new))
            if sn > 1e-10:
                p_new = p_new / sn
                best_obj = best_obj_t
                best_p = p_new.copy()
                p = p_new
            if verbose:
                print(f"    FW it{it}: obj={best_obj:.8f} t={best_t:.2f}")
        else:
            # Check if we should also try the "improved" LP
            # min M s.t. (A@g)_k <= M, sum(g)=1, g>=0
            constraints2 = [A @ g_var <= M_var, cp.sum(g_var) == 1]
            prob2 = cp.Problem(cp.Minimize(M_var), constraints2)
            g_val2 = None
            for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
                try:
                    prob2.solve(solver=solver, max_time=3.0)
                    if g_var.value is not None and prob2.status in ('optimal', 'optimal_inaccurate'):
                        g_val2 = np.maximum(g_var.value, 0)
                        sg2 = float(np.sum(g_val2))
                        if sg2 > 1e-10:
                            g_val2 = g_val2 / sg2
                        break
                except Exception:
                    continue

            if g_val2 is not None:
                best_t2 = 0.0
                best_obj2 = obj_val(p)
                for t in [0.1, 0.3, 0.5]:
                    mix2 = (1 - t) * p + t * g_val2
                    mix2 = np.maximum(mix2, 0)
                    sm2 = float(np.sum(mix2))
                    if sm2 > 1e-10:
                        mix2 = mix2 / sm2
                        obj2 = obj_val(mix2)
                        if obj2 < best_obj2:
                            best_obj2 = obj2
                            best_t2 = t
                if best_t2 > 0 and best_obj2 < best_obj - 1e-12:
                    p_new2 = (1 - best_t2) * p + best_t2 * g_val2
                    p_new2 = np maximum(p_new2, 0)  # BUG placeholder
                    p_new2 = np.maximum(p_new2, 0)
                    sn2 = float(np.sum(p_new2))
                    if sn2 > 1e-10:
                        p_new2 = p_new2 / sn2
                        best_obj = best_obj2
                        best_p = p_new2.copy()
                        p = p_new2
            else:
                break  # No improvement, stop

    return best_p


def peak_targeted_update(p, time_budget, k_peaks=3, verbose=False):
    """
    Target the top-k peaks of the convolution specifically.
    Uses a weighted subgradient that emphasizes the top peaks.
    """
    n = len(p)
    p = np.maximum(p, 0).copy().astype(np.float64)
    s = float(np.sum(p))
    if s < 1e-10:
        return p
    p = p / s

    best_p = p.copy()
    best_obj = obj_val(p)
    start = time.monotonic()
    rng = np.random.default_rng(99)

    lr = 1.0 / (10.0 * np.sqrt(n))

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        c = conv_full(p)
        # Find top-k peaks
        top_idx = np.argsort(c)[-k_peaks:]
        w = np.zeros(2 * n - 1)
        # Weight by how much above average each peak is
        c_mean = float(np.mean(c))
        weights = np.maximum(c[top_idx] - c_mean, 0)
        if weights.sum() > 0:
            weights = weights / weights.sum()
        else:
            weights = np.ones(k_peaks) / k_peaks
        for i, idx in enumerate(top_idx):
            w[idx] = weights[i]

        # Compute gradient
        p_rev = p[::-1]
        N = 1
        while N < 3 * n - 1:
            N <<= 1
        gc = irfft(rfft(p_rev, N) * rfft(w, N), N)
        grad = 2.0 * 2.0 * n * gc[n - 1:2 * n - 1]

        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-18:
            break

        # GD step with momentum-like damping
        step = lr / (g_norm / (2.0 * n) + 1e-10)
        p_new = p - step * grad
        p_new = np.maximum(p_new, 0)
        sn = float(np.sum(p_new))
        if sn < 1e-15:
            p_new = best_p.copy()
        else:
            p_new = p_new / sn

        obj_new = obj_val(p_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_p = p_new.copy()
            p = p_new
        else:
            # Try smaller step
            p_try = p - step * 0.3 * grad
            p_try = np.maximum(p_try, 0)
            st = float(np.sum(p_try))
            if st > 1e-15:
                p_try = p_try / st
                ot = obj_val(p_try)
                if ot < best_obj - 1e-14:
                    best_obj = ot
                    best_p = p_try.copy()
                    p = p_try
            else:
                p = p + 0.01 * rng.standard_normal(n)
                p = np.maximum(p, 0)
                ps = float(np.sum(p))
                if ps > 1e-10:
                    p = p / ps
                else:
                    p = best_p.copy()

        # Decay learning rate
        lr *= 0.999

    return best_p


def resample_seq(p, new_n):
    """Resample probability vector to new length."""
    n = len(p)
    if n == new_n:
        return p.copy()
    if n == 1:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    p_new = np.interp(x_new, x_old, p)
    p_new = np.maximum(p_new, 0)
    s = float(np.sum(p_new))
    if s < 1e-15:
        return np.full(new_n, 1.0 / new_n)
    return p_new / s


def propose_candidate():
    """
    Main search function. Multi-phase optimization with FISTA, mirror descent,
    LP Frank-Wolfe, peak targeting, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + 1480.0  # Leave 20s buffer

    # Initialize from parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s > 0.01:
            best_p = parent / s
        else:
            best_p = np.ones(800) / 800.0
    except Exception:
        best_p = np.ones(800) / 800.0
        print("No parent, using uniform init")

    n0 = len(best_p)
    best_score = obj_val(best_p)
    print(f"Initial (n={n0}): obj={best_score:.8f}")

    # === Phase 1: FISTA with continuation (main workhorse) ===
    print(f"\n=== Phase 1: FISTA Continuation (n={n0}) ===")
    remaining = deadline - time.monotonic()
    alloc1 = min(remaining * 0.25, 350.0)
    if alloc1 > 30:
        p1 = fista_continue(best_p.copy(), alloc1, tau_start=100.0, tau_end=30000.0, verbose=True)
        s1 = obj_val(p1)
        if s1 < best_score:
            best_score = s1
            best_p = p1.copy()
            print(f"  Phase1 FISTA: {best_score:.8f} **")
        else:
            print(f"  Phase1 FISTA: {s1:.8f} (no improvement)")

    # === Phase 2: Mirror Descent ===
    print(f"\n=== Phase 2: Mirror Descent (n={n0}) ===")
    remaining = deadline - time.monotonic()
    alloc2 = min(remaining * 0.2, 300.0)
    if alloc2 > 30:
        p2 = mirror_descent_opt(best_p.copy(), alloc2, tau=300.0, eta_scale=0.8, verbose=True)
        s2 = obj_val(p2)
        if s2 < best_score:
            best_score = s2
            best_p = p2.copy()
            print(f"  Phase2 MD: {best_score:.8f} **")
        else:
            print(f"  Phase2 MD: {s2:.8f} (no improvement)")

    # === Phase 3: LP Frank-Wolfe ===
    print(f"\n=== Phase 3: LP Frank-Wolfe (n={n0}) ===")
    remaining = deadline - time.monotonic()
    alloc3 = min(remaining * 0.2, 250.0)
    if alloc3 > 20:
        p3 = lp_frank_wolfe(best_p.copy(), alloc3, verbose=True, max_iters=30)
        s3 = obj_val(p3)
        if s3 < best_score:
            best_score = s3
            best_p = p3.copy()
            print(f"  Phase3 FW: {best_score:.8f} **")
        else:
            print(f"  Phase3 FW: {s3:.8f} (no improvement)")

    # === Phase 4: Peak-targeted GD ===
    print(f"\n=== Phase 4: Peak-Targeted (n={n0}) ===")
    remaining = deadline - time.monotonic()
    alloc4 = min(remaining * 0.1, 100.0)
    if alloc4 > 15:
        p4 = peak_targeted_update(best_p.copy(), alloc4, k_peaks=5, verbose=False)
        s4 = obj_val(p4)
        if s4 < best_score:
            best_score = s4
            best_p = p4.copy()
            print(f"  Phase4 Peak: {best_score:.8f} **")
        else:
            print(f"  Phase4 Peak: {s4:.8f} (no improvement)")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_cur = len(best_p)
    n_candidates = []
    for delta in [-300, -200, -100, -50, 50, 100, 200, 300, 500, 700]:
        nc = n_cur + delta
        if 300 <= nc <= 3000:
            n_candidates.append(nc)
    for na in [400, 500, 600, 800, 1000, 1200, 1500, 2000, 2500]:
        if na not in n_candidates:
            n_candidates.append(na)
    n_candidates = sorted(set(n_candidates))
    # Prioritize n values close to current best
    n_candidates.sort(key=lambda x: abs(x - n_cur))
    n_candidates = n_candidates[:8]
    print(f"  Trying n values: {n_candidates}")

    for nc in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        if nc == n_cur:
            continue

        p_try = resample_seq(best_p, nc)

        # Quick FISTA
        alloc_f = min(remaining * 0.25, 60.0)
        if alloc_f > 15:
            p_try = fista_continue(p_try, alloc_f, tau_start=100.0, tau_end=5000.0, verbose=False)

        # Quick MD
        remaining = deadline - time.monotonic()
        alloc_m = min(remaining * 0.15, 40.0)
        if alloc_m > 10:
            p_try = mirror_descent_opt(p_try, alloc_m, tau=200.0, eta_scale=0.6, verbose=False)

        s_try = obj_val(p_try)
        if s_try < best_score:
            best_score = s_try
            best_p = p_try.copy()
            n_cur = nc
            print(f"  ** n={nc}: obj={best_score:.8f} **")
        else:
            print(f"  n={nc}: obj={s_try:.8f}")

    # === Phase 6: Structured restarts ===
    print(f"\n=== Phase 6: Structured Restarts ===")
    n_final = len(best_p)
    rng = np.random.default_rng(42)

    restarts = []

    # Mirror-symmetric
    p_sym = (best_p + best_p[::-1]) / 2.0
    p_sym = np.maximum(p_sym, 0)
    ss = float(np.sum(p_sym))
    if ss > 1e-10:
        restarts.append(("sym", p_sym / ss))

    # Fourier perturbation (low freq)
    pert = np.zeros(n_final)
    for k in range(1, 4):
        freq = k * 2.0 * np.pi / n_final
        amp = 0.03 / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(freq * np.arange(n_final) + phase)
    p_four = best_p * np.exp(pert)
    p_four = np.maximum(p_four, 0)
    sf = float(np.sum(p_four))
    if sf > 1e-10:
        restarts.append(("fourier", p_four / sf))

    # Exponential tilt (emphasize one side)
    for tilt_dir in [-1, 1]:
        tilt = np.linspace(0, 0.05 * tilt_dir, n_final)
        p_tilt = best_p * np.exp(tilt)
        p_tilt = np.maximum(p_tilt, 0)
        st = float(np.sum(p_tilt))
        if st > 1e-10:
            restarts.append((f"tilt{tilt_dir}", p_tilt / st))

    # Gaussian bump modification
    center = n_final // 2
    sigma = n_final / 4.0
    bumps = np.exp(-((np.arange(n_final) - center) ** 2) / (2 * sigma ** 2))
    p_bump = best_p * (1.0 + 0.1 * bumps)
    p_bump = np.maximum(p_bump, 0)
    sb = float(np.sum(p_bump))
    if sb > 1e-10:
        restarts.append(("bump", p_bump / sb))

    for name, p_r in restarts:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        # FISTA polish
        alloc_r = min(remaining * 0.3, 50.0)
        if alloc_r > 10:
            p_r = fista_continue(p_r, alloc_r, tau_start=200.0, tau_end=10000.0, verbose=False)

        s_r = obj_val(p_r)
        if s_r < best_score:
            best_score = s_r
            best_p = p_r.copy()
            print(f"  Restart ({name}): {best_score:.8f} **")
        else:
            print(f"  Restart ({name}): {s_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_p)

    # High-tau FISTA
    if remaining > 40:
        alloc = min(remaining * 0.35, 200.0)
        p_f = fista_continue(best_p.copy(), alloc, tau_start=500.0, tau_end=100000.0, verbose=True)
        sf = obj_val(p_f)
        if sf < best_score:
            best_score = sf
            best_p = p_f.copy()
            print(f"  Final FISTA: {best_score:.8f} **")
        else:
            print(f"  Final FISTA: {sf:.8f}")

    # MD with high tau
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 100.0)
        p_m = mirror_descent_opt(best_p.copy(), alloc, tau=1000.0, eta_scale=0.5, verbose=True)
        sm = obj_val(p_m)
        if sm < best_score:
            best_score = sm
            best_p = p_m.copy()
            print(f"  Final MD: {best_score:.8f} **")
        else:
            print(f"  Final MD: {sm:.8f}")

    # Last FPW attempt
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.4, 80.0)
        p_lp = lp_frank_wolfe(best_p.copy(), alloc, verbose=True, max_iters=15)
        sl = obj_val(p_lp)
        if sl < best_score:
            best_score = sl
            best_p = p_lp.copy()
            print(f"  Final FW: {best_score:.8f} **")
        else:
            print(f"  Final FW: {sl:.8f}")

    # Final peak targeting
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 40.0)
        p_pk = peak_targeted_update(best_p.copy(), alloc, k_peaks=3, verbose=False)
        spk = obj_val(p_pk)
        if spk < best_score:
            best_score = spk
            best_p = p_pk.copy()
            print(f"  Final Peak: {best_score:.8f} **")

    # Validate with evaluate_sequence
    result = [float(max(0.0, min(1000.0, x))) for x in best_p]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        # Fallback
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
