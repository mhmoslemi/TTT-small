```python
def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    if 0.0 < best_t < t_max:
        lo = max(0.0, best_t - 0.05)
        hi = min(t_max, best_t + 0.05)
        ts_fine = np.linspace(lo, hi, n_fine)
        for t in ts_fine:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t
    return best_t, best_obj


def lp_polish(a, time_budget=60.0, max_iters=50):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(42 + n)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break
        t_opt, obj_opt = line_search(a, g_lp)
        if t_opt > 1e-8 and obj_opt < best_obj - 1e-14:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            obj_direct = compute_obj(g_lp)
            if obj_direct < best_obj:
                best_obj = obj_direct
                best_a = g_lp.copy()
                a = g_lp.copy()
                stall = 0
        it += 1
        if stall >= 3:
            freq = rng.integers(1, max(2, n // 100))
            phase = rng.uniform(0, 2 * np.pi)
            idx = np.arange(n, dtype=np.float64)
            pert = 0.02 * min(stall, 5) * np.cos(2 * np.pi * freq * idx / n + phase)
            a = best_a * np.exp(pert)
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 0.01:
                a = a / s
            else:
                a = best_a.copy()
            stall = 0
    return best_a


# --- Parameterized initialization ---

def parametric_init(n, c, p, shift=0.0):
    """U-shaped: a_i = 1/(1 + c*((2i/(n-1)-1)+shift)^2p)"""
    x = 2.0 * np.arange(n, dtype=np.float64) / (n - 1) - 1.0
    x = x + shift
    a = 1.0 / (1.0 + c * np.abs(x) ** (2.0 * p))
    a = np.maximum(a, 1e-15)
    s = float(np.sum(a))
    return a / s


def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n, dtype=np.float64)
    if mode == 0:
        a = 1.0 / (0.1 + 3.0 * (2 * x - 1) ** 2)
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * (n // 8) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 8) ** 2)) + 0.01
    elif mode == 2:
        a = 4.0 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = np.exp(4.0 * (2 * x - 1) ** 2)
    elif mode == 4:
        a = 1.0 / (0.05 + 8.0 * (2 * x - 1) ** 4)
    elif mode == 5:
        a = np.ones(n)
        a[:n//10] = 3.0
        a[-n//10:] = 3.0
        a = np.convolve(a, np.ones(20)/20, mode='same')
    elif mode == 6:
        a = 1.0 / (0.1 + 5.0 * (2 * x - 1) ** 2)
        a *= (1.0 + 0.1 * np.sin(4 * np.pi * x))
    elif mode == 7:
        a = np.maximum(0, np.cos(np.pi * x)) ** 0.5 + 0.3
    else:
        a = parametric_init(n, 5.0, 1.5)
    a = np.maximum(a, 1e-10)
    s = float(np.sum(a))
    return a / s


def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 1e-15)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s


# --- Subgradient method (direct, no smoothing) ---

def subgradient_descent(a, time_budget=30.0):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    a_avg = a.copy()
    momentum = np.zeros(n)
    beta = 0.8
    it = 0
    best_it = 0
    lr0 = 0.03

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a_avg)
        k_max = int(np.argmax(conv))
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_max - i
            j2 = i - k_max
            if 0 <= j1 < n:
                grad[i] += a_avg[j1]
            if 0 <= j2 < n:
                grad[i] += a_avg[j2]
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            grad = grad / grad_norm
        lr_t = lr0 / (1.0 + 0.01 * it)
        momentum = beta * momentum - lr_t * grad
        a_new = a_avg + momentum
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
        else:
            a_new = a_new / s
        a_avg = (it / (it + 1.0)) * a_avg + (1.0 / (it + 1.0)) * a_new
        obj = compute_obj(a_avg)
        if obj < best_obj:
            best_obj = obj
            best_a = a_avg.copy()
            best_it = it
        it += 1
        if it - best_it > 5000:
            a_avg = best_a.copy()
            momentum = np.zeros(n)
            lr0 = 0.01
            best_it = it
    return best_a, best_obj
```

So term1[i] = cross[i] and term2[i] = conv[i].

```python
def _smooth_grad_fast(x, n, tau):
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    a = exp_x / np.sum(exp_x)
    conv_a = compute_conv(a)
    c_max = float(np.max(conv_a))
    shifted = (conv_a - c_max) / tau
    max_sh = np.max(shifted)
    exp_sh = np.exp(shifted - max_sh)
    weights = exp_sh / np.sum(exp_sh)
    # weights has length 2n-1
    
    # Compute cross-correlation and convolution via FFT
    L = 1
    while L < (2*n - 1) + n - 1:
        L <<= 1
    
    # Pad weights and a to length L
    w_padded = np.zeros(L)
    w_padded[:2*n-1] = weights
    a_padded = np.zeros(L)
    a_padded[:n] = a
    
    W = rfft(w_padded)
    A = rfft(a_padded)
    
    cross = irfft(W * np.conj(A), L)  # cross[m] = sum_j weights[j+m]*a[j]
    conv_w = irfft(W * A, L)  # conv_w[m] = sum_j weights[j]*a[m-j]
    
    d_obj_da = 2.0 * n * (cross[:n] + conv_w[:n])
    
    # Chain rule for softmax parameterization: a = softmax(x)
    # d(a[i])/dx[j] = a[i]*(delta_ij - a[j])
    # d_obj/dx[j] = sum_i (d_obj/da[i]) * (da[i]/dx[j])
    # = sum_i d_obj_da[i] * a[i] * (delta_ij - a[j])
    # = d_obj_da[j]*a[j] - a[j]*sum_i d_obj_da[i]*a[i]
    grad = d_obj_da * a - a * np.dot(d_obj_da, a)
    return grad
```

import numpy as np

def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def get_fft_size2(n):
    """FFT size for cross-correlation of (2n-1) with n: need >= 3n-2."""
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def _smooth_obj(x, n, tau):
    """Top-level smooth objective in log-space."""
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    a = exp_x / np.sum(exp_x)
    conv = compute_conv(a)
    c_max = float(np.max(conv))
    shifted = (conv - c_max) / tau
    max_sh = np.max(shifted)
    exp_sh = np.exp(shifted - max_sh)
    lse = c_max + tau * (max_sh + np.log(np.sum(exp_sh)))
    return 2.0 * n * lse


def _smooth_grad_fast(x, n, tau):

    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)

    conv_a = compute_conv(a)
    c_max = float(np.max(conv_a))

    max_sh = np.max(shifted)
    exp_sh = np.exp(shifted - max_sh)
    weights = exp_sh / np.sum(exp_sh)

    L = get_fft_size2(n)
    w_padded = np.zeros(L)
    w_padded[:2 * n - 1] = weights
    a_padded = np.zeros(L)

    W = rfft(w_padded)

def build_conv_matrix(a):

    A = np.zeros((2 * n - 1, n))

        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):

    a_norm = a / float(np.sum(a))

    g = cp.Variable(n, nonneg=True)
