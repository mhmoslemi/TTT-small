def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:

    return np.ones(n) / n

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]

    if s > 1e-15:

    conv = compute_conv(a)
    best_a = a.copy()
    best_conv = conv.copy()
    best_S = S
    best_obj = 2.0 * n * np.max(conv) / (S * S)
    start = time.monotonic()

    phi = (1.0 + np.sqrt(5.0)) / 2.0
    inv_phi2 = 1.0 / (phi * phi)
    gold_iters = 14

    for sweep in range(max_sweeps):
        if time.monotonic() - start >= time_budget:
            break
        improved_count = 0
        total_coordinats = n
        no_improve_sweeps = 0

        if stochastic and n > 200:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            ai = a[i]
            if ai < 1e-14:
                continue

            lo = -ai
            hi = ai + 1.0 / n

            c = hi - inv_phi2 * (hi - lo)
            d_gold = lo + inv_phi2 * (hi - lo)

            def eval_d(d_val):
                delta = np.zeros(2 * n - 1)
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                if k_lo <= k_hi:
                    length = k_hi - k_lo + 1
                    delta[k_lo:k_hi + 1] = 2.0 * d_val * a[0:length]
                    if k_lo <= 2 * i <= k_hi:
                        delta[2 * i] = 2.0 * a[i] * d_val + d_val * d_val
                conv_new = conv + delta
                S_new = S + d_val
                if S_new < 0.01:
                    return np.inf
                return 2.0 * n * np.max(conv_new) / (S_new * S_new)

            fc = eval_d(c)
            fd = eval_d(d_gold)

            for _ in range(gold_iters):
                if fc < fd:
                    hi = d_gold
                    d_gold = c
                    fd = fc
                    c = hi - inv_phi2 * (hi - lo)
                    fc = eval_d(c)
                else:
                    lo = c
                    c = d_gold
                    fc = fd
                    d_gold = lo + inv_phi2 * (hi - lo)
                    fd = eval_d(d_gold)

            best_d = (lo + hi) / 2.0
            new_ai = ai + best_d
            if new_ai < 0:
                new_ai = 0.0

            a[i] = new_ai
            S_new = S + best_d
            if S_new < 0.01:
                a[i] = ai
                continue

            a = a / S_new
            S = 1.0
            conv = compute_conv(a)

            obj_new = 2.0 * n * np.max(conv)
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a.copy()
                best_conv = conv.copy()
                best_S = S
                improved_count += 1
            elif obj_new < best_obj:
                improved_count += 1
            else:
                a = best_a.copy()
                conv = best_conv.copy()
                S = best_S

        if verbose:
            print(f"      GSCD sweep {sweep}: obj={best_obj:.8f}, improved={improved_count}/{n}, "
                  f"t={time.monotonic()-start:.1f}s")

        if improved_count == 0:
            break

    return best_a


def stochastic_cd(a, time_budget, verbose=False, seed=42):
    """Stochastic coordinate descent - randomly pick coordinates to optimize."""
    n = len(a)
    rng = np.random.default_rng(seed)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    S = 1.0

    conv = compute_conv(a)
    best_a = a.copy()
    best_obj = 2.0 * n * np.max(conv)
    start = time.monotonic()

    phi = (1.0 + np.sqrt(5.0)) / 2.0
    inv_phi2 = 1.0 / (phi * phi)
    gold_iters = 10
    batch_size = max(10, n // 20)

    it = 0
    best_it = 0
    patience = 500

    while time.monotonic() - start < time_budget:
        idx_batch = rng.choice(n, size=min(batch_size, n), replace=False)

        for i in idx_batch:
            if time.monotonic() - start >= time_budget:
                break
            i = int(i)
            ai = a[i]
            if ai < 1e-14:
                continue

            lo = -ai
            hi = ai + 1.0 / n

            c = hi - inv_phi2 * (hi - lo)
            d_gold = lo + inv_phi2 * (hi - lo)

            delta_c = np.zeros(2 * n - 1)
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            if k_lo <= k_hi:
                length = k_hi - k_lo + 1
                delta_c[k_lo:k_hi + 1] = 2.0 * c * a[0:length]
                if k_lo <= 2 * i <= k_hi:
                    delta_c[2 * i] = 2.0 * a[i] * c + c * c
            fc = 2.0 * n * np.max(conv + delta_c) / (S + c) ** 2 if S + c > 0.01 else np.inf

            delta_d = np.zeros(2 * n - 1)
            if k_lo <= k_hi:
                length = k_hi - k_lo + 1
                delta_d[k_lo:k_hi + 1] = 2.0 * d_gold * a[0:length]
                if k_lo <= 2 * i <= k_hi:
                    delta_d[2 * i] = 2.0 * a[i] * d_gold + d_gold * d_gold
            fd = 2.0 * n * np.max(conv + delta_d) / (S + d_gold) ** 2 if S + d_gold > 0.01 else np.inf

            for _ in range(gold_iters):
                if fc < fd:
                    hi = d_gold
                    d_gold = c
                    fd = fc
                    c = hi - inv_phi2 * (hi - lo)
                    delta_c = np.zeros(2 * n - 1)
                    if k_lo <= k_hi:
                        length = k_hi - k_lo + 1
                        delta_c[k_lo:k_hi + 1] = 2.0 * c * a[0:length]
                        if k_lo <= 2 * i <= k_hi:
                            delta_c[2 * i] = 2.0 * a[i] * c + c * c
                    fc = 2.0 * n * np.max(conv + delta_c) / (S + c) ** 2 if S + c > 0.01 else np.inf
                else:
                    lo = c
                    c = d_gold
                    fc = fd
                    d_gold = lo + inv_phi2 * (hi - lo)
                    delta_d = np.zeros(2 * n - 1)
                    if k_lo <= k_hi:
                        length = k_hi - k_lo + 1
                        delta_d[k_lo:k_hi + 1] = 2.0 * d_gold * a[0:length]
                        if k_lo <= 2 * i <= k_hi:
                            delta_d[2 * i] = 2.0 * a[i] * d_gold + d_gold * d_gold
                    fd = 2.0 * n * np.max(conv + delta_d) / (S + d_gold) ** 2 if S + d_gold > 0.01 else np.inf

            best_d = (lo + hi) / 2.0
            new_ai = max(0.0, ai + best_d)
            a[i] = new_ai
            S += best_d
            if S < 0.01:
                a = best_a.copy()
                conv = compute_conv(a)
                S = 1.0
                continue
            a = a / S
            S = 1.0
            conv = compute_conv(a)

            obj_new = 2.0 * n * np.max(conv)
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a.copy()
                best_it = it

        it += 1
        if it - best_it > patience:
            break

        if verbose and it % 50 == 0:
            print(f"      StochCD it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

    return best_a


def build_lp_matrix(a_norm, constraint_indices):
    """Build LP constraint matrix for cross-convolution."""
    n = len(a_norm)
    A = np.zeros((len(constraint_indices), n))
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                A[idx, m] = a_norm[j]
    return A


def lp_solve_direction(a, constraint_indices, solver_list=None):
    """Solve LP: min M s.t. (a*g)[k] <= M for k in constraints, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_lp_matrix(a_norm, constraint_indices)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.SCS, cp.ECOS]

    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def lp_polish(a, time_budget, verbose=False, max_iters=50):
    """Iterative LP polishing with adaptive constraint selection."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        conv = compute_conv(best_a)
        max_c = np.max(conv)

        # Adaptive constraint selection: all k where conv[k] > 0.85 * max_c
        threshold = 0.85 * max_c
        constraint_indices = np.where(conv > threshold)[0]
        if len(constraint_indices) < 5:
            constraint_indices = np.argsort(conv)[::-1][:10]
        elif len(constraint_indices) > 50:
            # Keep top 50
            idx_vals = conv[constraint_indices]
            top_in_constraint = np.argsort(idx_vals)[::-1][:50]
            constraint_indices = constraint_indices[top_in_constraint]

        g_dir = lp_solve_direction(best_a, constraint_indices)
        if g_dir is None:
            break

        # Find best blend
        best_t = 0.0
        best_blend_obj = best_obj
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * best_a + t * g_dir
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 0.01:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_t = t

        if best_t > 0 and best_blend_obj < best_obj - 1e-13:
            a_new = (1 - best_t) * best_a + best_t * g_dir
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 3 == 0:
            print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, "
                  f"n_constraints={len(constraint_indices)}, elapsed={time.monotonic()-start:.1f}s")

        if stall >= 5:
            rng_local = np.random.default_rng(it * 104729)
            noise = rng_local.normal(0, 0.002 * min(stall, 5), n)
            a_pert = best_a + noise
            a_pert = np.maximum(a_pert, 0)
            s_p = float(np.sum(a_pert))
            if s_p > 0.01:
                best_a = a_pert / s_p
                best_obj = compute_obj(best_a)
            stall = 0

    return best_a


def penalty_gradient_descent(a, time_budget, verbose=False):
    """
    Gradient descent on smoothed max using log-sum-exp.
    min 2n * LSE_T(conv(a,a)) subject to sum(a)=1, a>=0
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    T = 10.0
    lr = 0.05 / n
    it = 0
    best_it = 0
    patience = max(500, int(time_budget * 50))

    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        T = 10.0 * (50.0 / 10.0) ** frac

        conv = compute_conv(a)
        cmax = np.max(conv)
        exp_vals = np.exp(T * (conv - cmax))
        w = exp_vals / exp_vals.sum()

        # Gradient of LSE(conv(a,a)) w.r.t. a[i]
        grad = np.zeros(n)
        for i in range(n):
            # d(conv[k])/da[i] = a[k-i] + a[i-k] (with bounds)
            lo1 = max(0, i)
            hi1 = min(2 * n - 2, i + n - 1)
            if lo1 <= hi1:
                grad[i] += np.dot(w[lo1:hi1 + 1], a[lo1 - i:hi1 - i + 1])
            lo2 = max(0, i - n + 1)
            hi2 = min(2 * n - 2, i)
            if lo2 <= hi2:
                grad[i] += np.dot(w[lo2:hi2 + 1], a[i - hi2:i - lo2 + 1][::-1])

        obj_smooth = 2.0 * n * (cmax + np.log(np.sum(exp_vals)) / T)
        grad = 2.0 * n * grad

        # Projected gradient step
        a_try = a - lr * grad
        a_try = project_simplex(a_try)

        # Check true objective
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            best_it = it
            a = a_try.copy()
        else:
            lr *= 0.7

        it += 1
        if it - best_it > patience:
            break

        if verbose and it % 200 == 0:
            print(f"      PGD it{it}: obj={best_obj:.8f}, T={T:.1f}, lr={lr:.6f}, "
                  f"t={time.monotonic()-start:.1f}s")

    return best_a


def explore_n(best_seq, best_score, n_options, time_budget, seed=42):
    """Try different sequence lengths by interpolation + optimization."""
    rng = np.random.default_rng(seed)
    results = []

    for n_try in n_options:
        if time_budget <= 20:
            break
        time_per_n = time_budget / len(n_options)

        x_old = np.linspace(0, 1, len(best_seq))
        x_new = np.linspace(0, 1, n_try)
        a_try = np.interp(x_new, x_old, best_seq)
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s

        # Quick CD
        t_cd = min(time_per_n * 0.5, 30.0)
        a_try = gold_section_cd(a_try, time_budget=t_cd, max_sweeps=3)

        # Quick LP
        t_lp = min(time_per_n * 0.3, 20.0)
        if t_lp > 5:
            a_try = lp_polish(a_try, time_budget=t_lp, max_iters=5)

        obj_try = compute_obj(a_try)
        results.append((n_try, a_try, obj_try))
        print(f"      n={n_try}: obj={obj_try:.8f}")

    return results


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization:
    1. Parametric initialization sweep
    2. Fast golden-section coordinate descent
    3. LP polishing with adaptive constraints
    4. Stochastic CD for exploration
    5. n exploration
    6. Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # === Load parent ===
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.all(np.isfinite(parent_arr)):
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        print("Could not load parent, using defaults")

    # === Phase 1: Parametric initialization sweep ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 1: Parametric Sweep ===")
        sweep_time = min(25.0, remaining * 0.05)
        alphas = [1.0, 2.0, 3.0, 5.0, 8.0, 12.0, 20.0]
        betas = [0.5, 0.8, 1.0, 1.2, 1.5, 2.0]
        n_values = [n_current] if best_seq is not None else [600, 800, 1000, 1200]

        for n_val in n_values:
            for alpha in alphas:
                for beta in betas:
                    if time.monotonic() > start_time + sweep_time:
                        break
                    a_try = parametric_init(n_val, alpha, beta)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_score:
                        best_score = obj_try
                        best_seq = a_try.copy()
                        print(f"  n={n_val}, alpha={alpha}, beta={beta}: obj={obj_try:.8f} **")

        # Also try diverse inits
        for mode in range(9):
            if time.monotonic() > start_time + sweep_time:
                break
            a_try = diverse_init(n_current, mode)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  diverse mode={mode}: obj={obj_try:.8f} **")

        print(f"  After sweep: best={best_score:.8f}")

    if best_seq is None:
        best_seq = parametric_init(800, 5.0, 1.0)
        best_score = compute_obj(best_seq)

    # === Phase 2: Fast Golden Section CD ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 2: Fast GSCD ===")
        alloc = min(180.0, remaining * 0.2)
        a_cd = gold_section_cd(best_seq, time_budget=alloc, max_sweeps=8, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  GSCD: obj={best_score:.8f} **")
        else:
            print(f"  GSCD: obj={obj_cd:.8f}")

    # === Phase 3: LP Polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print("\n=== Phase 3: LP Polishing ===")
        alloc = min(250.0, remaining * 0.25)
        a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True, max_iters=40)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f}")

    # === Phase 4: More CD after LP ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print("\n=== Phase 4: Post-LP GSCD ===")
        alloc = min(120.0, remaining * 0.15)
        a_cd2 = gold_section_cd(best_seq, time_budget=alloc, max_sweeps=5, verbose=True)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  GSCD2: obj={best_score:.8f} **")
        else:
            print(f"  GSCD2: obj={obj_cd2:.8f}")

    # === Phase 5: Stochastic CD for exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print("\n=== Phase 5: Stochastic CD ===")
        alloc = min(80.0, remaining * 0.1)
        a_stoch = stochastic_cd(best_seq, time_budget=alloc, verbose=True, seed=seed + 1)
        obj_stoch = compute_obj(a_stoch)
        if obj_stoch < best_score:
            best_score = obj_stoch
            best_seq = a_stoch.copy()
            print(f"  StochCD: obj={best_score:.8f} **")
        else:
            print(f"  StochCD: obj={obj_stoch:.8f}")

    # === Phase 6: Penalty GD ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 6: Penalty GD ===")
        alloc = min(60.0, remaining * 0.08)
        a_pgd = penalty_gradient_descent(best_seq, time_budget=alloc)
        obj_pgd = compute_obj(a_pgd)
        if obj_pgd < best_score:
            best_score = obj_pgd
            best_seq = a_pgd.copy()
            print(f"  PGD: obj={best_score:.8f} **")
        else:
            print(f"  PGD: obj={obj_pgd:.8f}")

    # === Phase 7: n Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print("\n=== Phase 7: n Exploration ===")
        n_cur = len(best_seq)
        n_options = []
        for dn in [-200, -100, -50, 50, 100, 200, 300]:
            n_try = n_cur + dn
            if 300 <= n_try <= 3000 and n_try != n_cur:
                n_options.append(n_try)
        n_options = sorted(set(n_options))[:5]
        print(f"  Trying n: {n_options}")

        alloc = min(200.0, remaining * 0.2)
        results = explore_n(best_seq, best_score, n_options, alloc, seed=seed + 100)
        for n_try, a_try, obj_try in results:
            if obj
