def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None, solver_list=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """

    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

        solver_list = ['PDLP', 'CLARABEL', 'SCS', 'ECOS', 'MOSEK', 'GLOP']

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

        try:
            g.value = warm_start.copy()
        except Exception:
            pass

            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception as e:

    return None, None

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    fa_fixed = rfft(a_fixed, size)
    
    for it in range(n_iter):
        # Compute current autocorrelation (or cross-correlation with a_fixed)
        fa = rfft(a_work, size)
        cross_spec = fa * np.conj(fa_fixed)  # cross-correlation spectrum
        autocorr = irfft(fa * fa.conj(), size)[:2*n-1]  # self-convolution
        
        # Actually for MPD, we want to minimize peak of autocorrelation
        # The peak is at lag 0 (which is sum(a)^2)
        # We want to reduce the peak sidelobes

        # But this is complex. Let's use a simpler version.

        # the autocorrelation has peak sidelobes
        peak_idx = np.argmax(autocorr[1:]) + 1  # first peak sidelobe

        target = autocorr[0] / (2 * n)  # average level (rough)

        # Frequency domain update

        fa_new = fa * np.exp(-error_spec * 0.01)
        a_work = irfft(fa_new, size)[:n]
        
        # Time domain projection: a >= 0, sum = 1
        a_work = np.maximum(a_work, 0)
        s = float(np.sum(a_work))
        if s < 1e-10:

        a_work = a_work / s
    
    return a_work

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating minimization (PDLP solver), smooth gradient polishing via
FISTA on a log-sum-exp approximation, binary-search coordinate descent,
and multi-restart exploration.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    solvers = ['PDLP', 'CLARABEL', 'SCS', 'ECOS', 'MOSEK', 'GLOP', 'SCS']
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass
    
    seen_solvers = set()
    for solver_name in solvers:
        if solver_name in seen_solvers:
            continue
        seen_solvers.add(solver_name)
        try:
            solver = getattr(cp, solver_name, None)
            if solver is None:
                continue
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def softmax_weights(conv, tau):
    """Compute softmax weights exp(tau*conv)/sum(exp(tau*conv)) with numerical stability."""
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    total = float(np.sum(exp_vals))
    if total < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_vals / total


def smooth_gradient(a, conv, tau):
    """Gradient of 2n * logsumexp(tau * conv(a,a)) / tau w.r.t. a."""
    n = len(a)
    weights = softmax_weights(conv, tau)
    
    grad = np.zeros(n)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    # grad[j] = 2*n * sum_{k=j}^{j+n-1} weights[k] * a[k-j]
    # This is a cross-correlation
    # We can compute it as: grad = 2n * ifft(fft(weights_padded) * conj(fft(a_padded)))[:n]
    # But let's do it directly for clarity (n=800 is small enough)
    
    w_padded = np.zeros(size)
    w_padded[:2*n-1] = weights
    a_padded = np.zeros(size)
    a_padded[:n] = a
    
    # Cross-correlation: sum_k w[k] * a[k-j] for k in [j, j+n-1]
    # This is (w * a_reversed)[j+n-1] in standard convolution notation
    # Or equivalently: ifft(fft(w) * conj(fft(a)))
    cross = irfft(rfft(w_padded, size) * np.conj(rfft(a_padded, size)), size)
    # cross[m] = sum_k w[k] * a[k-m] ... hmm, let me verify
    # Actually ifft(fft(w) * conj(fft(a))) gives the cross-correlation
    # cross[m] = sum_k w[k+m] * conj(a[k]) = sum_k w[k+m] * a[k] (for real)
    # So cross[j] = sum_k w[k+j] * a[k] = sum_i w[i] * a[i-j] (substituting i=k+j)
    # That's what we want: grad[j] = 2n * sum_i w[i] * a[i-j] for i in [j, j+n-1]
    
    grad = 2.0 * n * cross[:n]
    return grad


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    it = 0
    y = a.copy()
    t_k = 1.0
    stall = 0

        # Compute gradient at y
        conv_y = compute_conv(y)
        grad = smooth_gradient(y, conv_y, tau)

        # Actually, the Hessian of the smooth obj is bounded by 2*n*tau*||Q|| where Q is the max convolution matrix
        # For safety, use a conservative step size

        # The Hessian of f at x involves tau * (Quadratic forms) - tau * (weighted sum)^2

        # For safety, use step = 1 / (4 * n * tau)

        z = project_simplex(z)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0

        it += 1

def lp_alternate(a, time_budget, seed=42, verbose=True):
    """LP alternating minimization with adaptive step size and warm starts."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    iter_count = 0
    stall = 0
    
    while time.monotonic() - start < time_budget - 5:
        g_lp, M_lp = solve_lp(a, warm_start=prev_g)
        if g_lp is None:
            break
        prev_g = g_lp
        iter_count += 1
        
        # Adaptive line search
        best_local_obj = best_obj
        best_local_a = a.copy()
        
        t_values = [1.0, 0.8, 0.6, 0.4, 0.25, 0.15, 0.1, 0.05]
        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_local_obj - 1e-15:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
        
        if best_local_obj < best_obj - 1e-15:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall = 0
            if verbose and iter_count % 50 == 0:
                elapsed = time.monotonic() - start
                print(f"      LP {iter_count}: obj={best_obj:.10f}, elapsed={elapsed:.0f}s")
        else:
            stall += 1
            if stall >= 10:
                # Perturb
                strength = 0.003 + 0.002 * min(stall, 15)
                pert = rng.standard_normal(n) * strength
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                prev_g = None
                stall = 0
    
    return best_a
