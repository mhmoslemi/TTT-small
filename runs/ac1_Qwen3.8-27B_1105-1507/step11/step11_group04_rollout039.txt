from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

    """Create a Beta-distributed initialization."""
    j = np.arange(n, dtype=np.float64)
    x = (j + 0.5) / n  # in [0, 1]
    a = x**(alpha-1) * (1-x)**(beta-1)
    a = np.maximum(a, 0)

    return a / s

    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_u_shape(n, power=2.0):
    """U-shape: high at edges."""
    j = np.arange(n, dtype=np.float64)
    x = 2.0 * j / (n - 1) - 1.0 if n > 1 else np.array([0.0])
    a = 0.2 + 0.8 * np.power(np.abs(x), power)
    return a / a.sum()


def make_v_shape(n, power=2.0):
    """V-shape (inverted U): low at edges, high in middle."""
    j = np.arange(n, dtype=np.float64)
    x = 2.0 * j / (n - 1) - 1.0 if n > 1 else np.array([0.0])
    a = 1.0 - 0.6 * np.power(np.abs(x), power)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_two_bump(n, width=8.0):
    """Two Gaussian bumps at 1/4 and 3/4."""
    j = np.arange(n, dtype=np.float64)
    x = j / n
    a = np.exp(-0.5 * ((x - 0.25) * width) ** 2) + np.exp(-0.5 * ((x - 0.75) * width) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_linear_asym(n, slope=1.0):
    """Linearly asymmetric: a_j proportional to 1 + slope*(2j/(n-1) - 1)."""
    j = np.arange(n, dtype=np.float64)
    x = 2.0 * j / (n - 1) - 1.0 if n > 1 else np.array([0.0])
    a = 1.0 + slope * x
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def fourier_perturb(a, rng, strength=0.015):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 9):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / np.sqrt(k)
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def noise_perturb(a, rng, strength=0.02):
    """Add multiplicative Gaussian noise."""
    pert = rng.standard_normal(len(a)) * strength
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def exact_coord_descent(a, time_budget, seed=42, verbose=False, max_sweeps=200):
    """
    Coordinate descent with exact 1D subproblem solving via analytical breakpoints.
    For each coordinate j, finds the optimal delta by evaluating F(delta) at
    analytically derived candidate values.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    no_improve = 0
    max_no_improve = 4
    sweep = 0

    while time.monotonic() - start < time_budget and sweep < max_sweeps:
        sweep += 1
        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            alpha = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)
            k_aff_len = k_end - k_start + 1

            # Unaffected max (vectorized via precomputed would be faster, but this is O(n) per j)
            # Use the fact that only [k_start, k_end] is affected
            max_unaff = 0.0
            if k_start > 0:
                max_unaff = max(max_unaff, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaff = max(max_unaff, float(np.max(conv[k_end + 1:])))

            # Affected conv values
            c_aff = conv[k_start:k_end + 1].copy()
            # a[k-j] for k in [k_start, k_end], i.e., a[0], a[1], ..., a[k_aff_len-1]
            a_kj = a[:k_aff_len]

            # Generate candidate deltas
            candidates = [0.0]

            # Boundary
            if alpha > 1e-12:
                candidates.append(-alpha * 0.999)

            # Critical point of quadratic term (at k = 2j, index = 2j - k_start)
            diag_idx = 2 * j - k_start
            if 0 <= diag_idx < k_aff_len:
                c2j = conv[2 * j]
                if abs(alpha - 1.0) > 1e-10:
                    d_q = (alpha - c2j) / (alpha - 1.0)
                    if -alpha * 0.999 < d_q < 0.5:
                        candidates.append(d_q)

            # Critical points of top linear terms
            # (c_k + 2*delta*a_kj) / (1+delta)^2 has critical point at delta = (a_kj - c_k) / (3*a_kj)
            mask_non_diag = np.ones(k_aff_len, dtype=bool)
            mask_non_diag[diag_idx] = False
            valid_lin = mask_non_diag & (a_kj > 1e-15)
            if np.any(valid_lin):
                c_lin = c_aff[valid_lin]
                a_lin = a_kj[valid_lin]
                # Get top 5 by current value
                top_count = min(5, len(c_lin))
                top_local = np.argpartition(c_lin, -top_count)[-top_count:]
                for ti in top_local:
                    if a_lin[ti] > 1e-15:
                        d_l = (a_lin[ti] - c_lin[ti]) / (3.0 * a_lin[ti])
                        if -alpha * 0.999 < d_l < 0.5:
                            candidates.append(d_l)

            # Delta to balance top-1 and top-2 in affected region
            if k_aff_len >= 2:
                sorted_idx = np.argsort(c_aff)[::-1]
                top1 = sorted_idx[0]
                top2 = sorted_idx[1]
                val1 = c_aff[top1]
                val2 = c_aff[top2]
                if top1 == diag_idx:
                    # Quadratic = linear: c2j + 2αd + d^2 = c_top2 + 2d*a_kj[top2]
                    # d^2 + 2(α - a_kj[top2])d + (c2j - c_top2) = 0
                    A = 1.0
                    B = 2.0 * (alpha - a_kj[top2])
                    C = c2j - val2
                    disc = B * B - 4.0 * A * C
                    if disc >= 0:
                        sq = np.sqrt(disc)
                        for d in [(-B + sq) / (2.0 * A), (-B - sq) / (2.0 * A)]:
                            if -alpha * 0.999 < d < 0.5:
                                candidates.append(d)
                elif top2 == diag_idx:
                    A = 1.0
                    B = 2.0 * (alpha - a_kj[top1])
                    C = c2j - val1
                    disc = B * B - 4.0 * A * C
                    if disc >= 0:
                        sq = np.sqrt(disc)
                        for d in [(-B + sq) / (2.0 * A), (-B - sq) / (2.0 * A)]:
                            if -alpha * 0.999 < d < 0.5:
                                candidates.append(d)
                else:
                    # linear1 = linear2: c1 + 2d*a1 = c2 + 2d*a2
                    da = a_kj[top1] - a_kj[top2]
                    if abs(da) > 1e-15:
                        d_b = (val2 - val1) / (2.0 * da)
                        if -alpha * 0.999 < d_b < 0.5:
                            candidates.append(d_b)

            # Small perturbations around 0
            mean_val = 1.0 / n
            if alpha > 1e-12:
                for frac in [-0.5, -0.2, 0.2, 0.5, 1.0]:
                    candidates.append(frac * alpha * 0.5)
            candidates.extend([0.1 * mean_val, -0.1 * mean_val, 0.3 * mean_val])

            # Evaluate F(delta) at candidates (vectorized where possible)
            best_delta = 0.0
            best_F = best_obj

            # Filter valid candidates
            valid_cands = [d for d in candidates if d > -alpha + 1e-10 and (1.0 + d) > 1e-8]
            if not valid_cands:
                valid_cands = [0.0]

            for d in valid_cands:
                S_new = 1.0 + d
                # New affected conv: c_aff + 2*d*a_kj, with diagonal fix
                new_aff = c_aff + 2.0 * d * a_kj
                # Fix diagonal
                new_aff[diag_idx] = c_aff[diag_idx] + 2.0 * alpha * d + d * d
                # Ensure non-negative (should be if a >= 0 and d > -alpha)
                max_aff = float(np.max(new_aff))
                max_new = max(max_unaff, max_aff)
                F = 2.0 * n * max_new / (S_new * S_new)
                if F < best_F - 1e-14:
                    best_F = F
                    best_delta = d

            if best_delta != 0.0:
                d = best_delta
                a[j] += d
                S_new = 1.0 + d

                # Update conv in-place
                conv[k_start:k_end + 1] += 2.0 * d * a_kj
                conv[2 * j] = conv[k_start + diag_idx] + 2.0 * alpha * d + d * d - 2.0 * d * a_kj[diag_idx]
                # Actually let me redo this more carefully:
                # conv[2j] should be: old_conv[2j] + 2*alpha*d + d^2
                # But we already added 2*d*a_kj[diag_idx] = 2*d*a[2j-j] = 2*d*alpha to conv[2j]
                # So conv[2j] now = old + 2*d*alpha. We need old + 2*alpha*d + d^2.
                # So add d^2:
                conv[2 * j] += d * d

                # Normalize
                a /= S_new
                conv /= (S_new * S_new)

                cur_obj = 2.0 * n * float(np.max(conv))
                if cur_obj < best_obj - 1e-14:
                    best_obj = cur_obj
                    best_a = a.copy()
                    improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"  ECD sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")

    return best_a


def smart_restart_cd(a, time_budget, seed=42, verbose=False):
    """
    Coordinate descent with adaptive restarts on stall.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Run CD, and if it stalls too long, perturb and restart
    sub_budget = min(120.0, time_budget * 0.25)
    num_restarts = max(1, int(time_budget / sub_budget))

    for r in range(num_restarts):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        a_cur = exact_coord_descent(best_a, time_budget=sub_budget, seed=seed + r * 1000,
                                    verbose=verbose, max_sweeps=50)
        obj_cur = compute_obj_norm(a_cur)
        if obj_cur < best_obj:
            best_obj = obj_cur
            best_a = a_cur.copy()
            if verbose:
                print(f"    Restart {r}: obj={best_obj:.10f}")
        else:
            # Perturb for next restart
            strength = 0.01 + 0.005 * r
            best_a = fourier_perturb(best_a, rng, strength=strength)

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: diverse initialization + exact coordinate descent +
    adaptive restarts + length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 15.0

    # === Phase 0: Diverse Initialization ===
    print("=== Phase 0: Initialization ===")
    n_default = 800

    # Try to use parent
    parent_a = None
    try:
        parent_a = np.asarray(height_sequence_1, dtype=np.float64)
        parent_a = np.maximum(parent_a, 0)
        if parent_a.size > 0 and parent_a.sum() > 0.01:
            parent_a = parent_a / parent_a.sum()
            print(f"  Parent loaded: n={parent_a.size}, obj={compute_obj_norm(parent_a):.10f}")
    except Exception:
        pass

    # Generate diverse candidates
    candidates = []

    if parent_a is not None:
        candidates.append((parent_a.copy(), "parent"))

    n_init = 800
    inits = [
        (np.ones(n_init) / n_init, "uniform"),
        (make_beta(n_init, 2.0, 2.0), "beta(2,2)"),
        (make_beta(n_init, 1.5, 1.5), "beta(1.5,1.5)"),
        (make_beta(n_init, 3.0, 3.0), "beta(3,3)"),
        (make_beta(n_init, 1.5, 2.5), "beta(1.5,2.5)"),
        (make_poly_bump(n_init, 1.0), "poly_p1"),
        (make_poly_bump(n_init, 2.0), "poly_p2"),
        (make_poly_bump(n_init, 3.0), "poly_p3"),
        (make_raised_cos(n_init), "raised_cos"),
        (make_u_shape(n_init, 2.0), "u_shape"),
        (make_u_shape(n_init, 1.0), "u_shape_p1"),
        (make_v_shape(n_init, 2.0), "v_shape"),
        (make_two_bump(n_init, 8.0), "two_bump"),
        (make_linear_asym(n_init, 0.5), "lin_asym_05"),
        (make_linear_asym(n_init, 1.0), "lin_asym_10"),
    ]

    # Also try parent with perturbation
    if parent_a is not None:
        rng0 = np.random.default_rng(0)
        candidates.append((fourier_perturb(parent_a, rng0, 0.02), "parent_perturbed"))

    # Evaluate all initializations
    best_init = None
    best_init_obj = np.inf
    for a_init, name in inits:
        a_init = np.maximum(a_init, 0)
        s = float(np.sum(a_init))
        if s < 1e-10:
            continue
        a_init = a_init / s
        obj = compute_obj_norm(a_init)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_init.copy()
        candidates.append((a_init, name))
        if budget_s > 0:
            print(f"  Init [{name:15s}] n={a_init.size}: obj={obj:.10f}")

    # Evaluate parent candidates
    for a_init, name in candidates:
        if name in ("parent", "parent_perturbed"):
            obj = compute_obj_norm(a_init)
            if obj < best_init_obj:
                best_init_obj = obj
                best_init = a_init.copy()
            print(f"  Init [{name:15s}] n={a_init.size}: obj={obj:.10f}")

    best_a = best_init.copy()
    best_obj = best_init_obj
    print(f"  Best init: obj={best_obj:.10f}")

    # === Phase 1: Main ECD on best init (40% budget) ===
    print("\n=== Phase 1: Main ECD ===")
    p1_budget = min(budget_s * 0.35, 500.0)
    p1_end = start_time + p1_budget
    a1 = exact_coord_descent(best_a, time_budget=p1_budget, seed=seed, verbose=True, max_sweeps=300)
    obj1 = compute_obj_norm(a1)
    if obj1 < best_obj:
        best_obj = obj1
        best_a = a1.copy()
    print(f"  P1 result: obj={obj1:.10f} (best={best_obj:.10f})")

    # === Phase 2: Restart ECD from perturbed states (25% budget) ===
    print("\n=== Phase 2: Restart ECD ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = min(remaining * 0.45, 350.0)
        a2 = smart_restart_cd(best_a, time_budget=p2_budget, seed=seed + 7777, verbose=True)
        obj2 = compute_obj_norm(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            print(f"  ** P2: {best_obj:.10f} **")
        else:
            print(f"  P2: {obj2:.10f}")

    # === Phase 3: ECD on second-best init (15% budget) ===
    print("\n=== Phase 3: Second-best init ECD ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        # Find second best init
        second_best = None
        second_obj = best_init_obj + 0.01  # slightly above best
        for a_init, name in inits:
            a_init = np.maximum(a_init, 0)
            s = float(np.sum(a_init))
            if s < 1e-10:
                continue
            a_init = a_init / s
            obj = compute_obj_norm(a_init)
            if obj < second_obj and obj >= best_init_obj:
                second_obj = obj
                second_best = a_init.copy()
        if second_best is None and len(inits) > 1:
            second_best = inits[1][0].copy() / inits[1][0].sum()
        if second_best is not None:
            p3_budget = min(remaining * 0.3, 200.0)
            a3 = exact_coord_descent(second_best, time_budget=p3_budget, seed=seed + 3333,
                                     verbose=False, max_sweeps=150)
            obj3 = compute_obj_norm(a3)
            if obj3 < best_obj:
                best_obj = obj3
                best_a = a3.copy()
                print(f"  ** P3: {best_obj:.10f} **")
            else:
                print(f"  P3: {obj3:.10f}")

    # === Phase 4: Length exploration (15% budget) ===
    print("\n=== Phase 4: Length Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    n_options = sorted(set([n_best - 50, n_best - 20, n_best + 20, n_best + 50,
                            n_best + 100, n_best - 100]))
    n_options = [n for n in n_options if 400 <= n <= 1200 and n != n_best]

    for n_try in n_options[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, n_try)
        a_try = np.interp(x_new, x_old, best_a)
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s

        alloc = min(remaining * 0.25, 120.0)
        if alloc > 20:
            a_try = exact_coord_descent(a_try, time_budget=alloc, seed=n_try * 7 + seed,
                                        verbose=False, max_sweeps=80)

        obj_try = compute_obj_norm(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            print(f"  ** n={n_try}: {best_obj:.10f} **")
        else:
            print(f"  n={n_try}: {obj_try:.10f}")

    # === Phase 5: Final polish ===
    print("\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_fin = exact_coord_descent(best_a, time_budget=min(remaining * 0.7, 60.0),
                                    seed=seed + 99999, verbose=False, max_sweeps=100)
        obj_fin = compute_obj_norm(a_fin)
        if obj_fin < best_obj:
            best_obj = obj_fin
            best_a = a_fin.copy()
        print(f"  Final: obj={obj_fin:.10f} (best={best_obj:.10f})")

    # One more Fourier perturb + short CD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        rng_f = np.random.default_rng(seed + 12345)
        a_pert = fourier_perturb(best_a, rng_f, strength=0.008)
        a_pert = exact_coord_descent(a_pert, time_budget=min(remaining * 0.5, 30.0),
                                     seed=seed + 54321, verbose=False, max_sweeps=50)
        obj_pert = compute_obj_norm(a_pert)
        if obj_pert < best_obj:
            best_obj = obj_pert
            best_a = a_pert.copy()
            print(f"  Perturb+CD: {best_obj:.10f} **")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
