```python
def hankel_mv(y, a):
    """Compute T_y @ a where (T_y)[i,j] = y[i+j].
    y has length 2n-1, a has length n."""
    n = len(a)
    b_rev = a[::-1]
    # conv(b_rev, y) has length n + (2n-1) - 1 = 3n-2
    # We want entries at indices n-1, n, ..., 2n-2 (i.e., indices n-1 to 2n-2)
    size = 1
    while size < 3*n - 2:
        size <<= 1
    F_b = rfft(b_rev, size)
    F_y = rfft(y, size)
    conv = irfft(F_b * F_y, size)
    return conv[n-1:2*n-1]  # indices n-1 to 2n-2 inclusive, that's n entries
```

"""
Primal-dual game-theoretic optimization + FISTA + Full LP for step function minimization.
Key innovations:
1. Primal-dual (MWUA) for global exploration - treats as zero-sum game
2. FISTA with LSE smoothing for fast local convergence
3. Full LP (all 2n-1 constraints) for precise polishing
4. FFT-based Hankel matvec for O(n log n) gradient computation
5. Systematic n exploration
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n


def hankel_matvec(y, a):
    """Compute T_y @ a where (T_y)[i,j] = y[i+j].
    y has length 2n-1, a has length n. Returns vector of length n.
    Uses FFT for O(n log n) complexity."""
    n = len(a)
    b_rev = a[::-1]
    total_len = n + len(y) - 1  # = n + 2n-1 - 1 = 3n-2
    size = 1
    while size < total_len:
        size <<= 1
    F_b = rfft(b_rev, size)
    F_y = rfft(y, size)
    conv = irfft(F_b * F_y, size)
    # We want conv[n-1 : 2n-1] (indices n-1 to 2n-2, total n entries)
    return conv[n-1:2*n-1]


def lse_smooth_obj_and_grad(a, T):
    """Compute LSE-smoothed objective and gradient for normalized a.
    Returns (objective, gradient) where gradient is w.r.t. a (before projection)."""
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)
    # For numerical stability
    exp_vals = np.exp(T * (conv - cmax))
    exp_sum = exp_vals.sum()
    lse = cmax + np.log(exp_sum) / T
    w = exp_vals / exp_sum  # softmax weights
    
    # Gradient: 2n * sum_k w[k] * d(conv[k])/d(a)
    # d(conv[k])/d(a[i]) = 2*a[k-i] if 0<=k-i<n else 0
    # So grad[i] = 2n * sum_k w[k] * 2*a[k-i] * 1[0<=k-i<n]
    #            = 4n * sum_j a[j] * w[j+i] * 1[0<=j+i<2n-1]
    # This is 4n * (Hankel_matvec(w, a))[i]
    grad = 4.0 * n * hankel_matvec(w, a)
    
    obj = 2.0 * n * lse
    return obj, grad


def fista_lse(a_init, time_budget, T_start=5.0, T_end=100.0, verbose=False):
    """FISTA with LSE-smoothed objective."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)

        return a

    best_a = a.copy()
    best_obj = compute_obj(a)

    t_k = 1.0
    a_prev = a.copy()
    y = a.copy()
    lr = 1.0 / (8.0 * n)  # initial step size (conservative)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        # Anneal T
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** frac
