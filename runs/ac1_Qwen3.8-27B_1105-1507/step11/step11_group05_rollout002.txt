import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A

def solve_full_lp(a):
    """Solve the full LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.SCIP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None

        lo, hi = max(0, best_t - 0.1), min(1.0, best_t + 0.1)
        for _ in range(10):
            mid = (lo + hi) / 2
            a_mid = (1 - mid) * a_norm + mid * g_lp
            a_mid = np.maximum(a_mid, 0)
            s_mid = float(np.sum(a_mid))
            if s_mid < 0.01:
                lo = mid
                continue
            a_mid = a_mid / s_mid
            obj_mid = compute_obj(a_mid)
            if obj_mid < best_obj:
                best_obj = obj_mid
                best_t = mid
                hi = mid
            else:
                lo = mid
    
    if best_t > 0:
        a_new = (1 - best_t) * a_norm + best_t * g_lp
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        a_new = a_new / s_new
        return a_new, best_obj
    
    return a, best_obj
