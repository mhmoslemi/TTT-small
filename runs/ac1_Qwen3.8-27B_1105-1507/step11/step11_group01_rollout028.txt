"""
Optimizes step function coefficients using enhanced exact coordinate descent with
KKT-based repair, LP polishing, structured perturbations, and multi-n exploration.
Key improvements: (1) KKT violation detection and repair steps, (2) structured
sinusoidal perturbations for escaping local minima, (3) tighter CD convergence
with adaptive tolerance, (4) more aggressive LP usage with warm-start blending,
(5) targeted n exploration around the current best.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective 2n*max(conv)/sum^2 for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return float(2.0 * n * np.max(conv) / (s * s))


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1. Returns 2n*max(conv)."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return float(2.0 * n * np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:
                    g_val = g_val / g_sum
                    conv_check = A @ g_val
                    M_actual = float(np.max(conv_check))
                    return g_val, M_actual
        except Exception:
            continue
    return None, None


def kkt_violation(a):
    """
    Compute the KKT violation: standard deviation of the subgradient of max_conv
    on the support of a. Returns (violation, subgradient_vector, argmax_set).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    tol = max_conv * 1e-6 + 1e-15
    argmax_set = np.where(conv >= max_conv - tol)[0]

    # Subgradient: v[i] = (1/|S|) * sum_{k in S} 2*a[k-i] for valid k
    v_sub = np.zeros(n)
    for k in argmax_set:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            v_sub[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1]
    v_sub /= max(len(argmax_set), 1)

    # KKT violation: std of v_sub on support
    support = a > 1e-12
    if np.sum(support) > 1:
        v_on_support = v_sub[support]
        violation = float(np.std(v_on_support))
    else:
        violation = 0.0

    return violation, v_sub, argmax_set


def kkt_repair_step(a, v_sub, argmax_set, step_size=0.1):
    """
    Apply a KKT repair step: move a in the direction that equalizes the
    subgradient on the support, while maintaining sum(a)=1 and a>=0.
    """
    n = len(a)
    support = a > 1e-12
    n_support = int(np.sum(support))
    if n_support < 2:
        return a.copy()

    v_on_support = v_sub[support]
    v_mean = float(np.mean(v_on_support))
    # Direction: reduce high subgradient, increase low subgradient
    d = np.zeros(n)
    d[support] = -(v_sub[support] - v_mean)
    # Project onto sum=0
    d[support] -= float(np.sum(d)) / max(n_support, 1)
    # Ensure d >= -a (so a + step*d >= 0)
    max_decrease = step_size
    for i in range(n):
        if a[i] > 0 and d[i] < 0:
            d[i] = max(d[i], -a[i] / step_size)
    # Renormalize to sum=0 after clipping
    d[support] -= float(np.sum(d)) / max(n_support, 1)

    a_new = a + step_size * d
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new

                    d = np.zeros(n)
                    d[support] = -(v_sub[support] - mean_v)
                    d[support] -= float(np.sum(d)) / max(int(np.sum(support)), 1)
                    a_repair = a_repair + step * d
                    a_repair = np.maximum(a_repair, 0)
                    s_r = float(np.sum(a_repair))
                    if s_r > 1e-10:
                        a_repair = a_repair / s_r
                        obj_r = 2.0 * n * float(np.max(compute_conv(a_repair)))
                        if obj_r < best_obj:
                            best_obj = obj_r
                            best_a = a_repair.copy()
                            a = a_repair
                            conv = compute_conv(a)
                            S = float(np.sum(a))

        # Track stalls
        if sweep % 5 == 0:
            S_now = float(np.sum(a))
            if S_now > 1e-10:
                obj_now = 2.0 * n * float(np.max(conv)) / (S_now * S_now)
            if best_obj < prev_obj - 1e-13:
                prev_obj = best_obj
                stall_count = 0
            else:
                stall_count += 1
            if stall_count > 6:
                # Reset with small perturbation
                pert = rng.normal(0, 0.005, n)
                a_pert = a * np.exp(pert)
                a_pert = np.maximum(a_pert, 0)
                s_p = float(np.sum(a_pert))
                if s_p > 1e-10:
                    a_pert = a_pert / s_p
                    obj_p = 2.0 * n * float(np.max(compute_conv(a_pert)))
                    if obj_p < best_obj:
                        best_obj = obj_p
                        best_a = a_pert.copy()
                        a = a_pert
                        conv = compute_conv(a)
                        S = float(np.sum(a))
                stall_count = 0
                prev_obj = best_obj

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s, stall={stall_count}")

    return best_a


def lp_polish(a, time_budget=60.0, verbose=False, seed=42):
    """LP-based polishing with adaptive blending."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 2:
                pert = rng.normal(0, 0.02, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0

    return best_a


def smoothed_gradient_descent(a, time_budget=100.0, tau_init=30.0, tau_end=800.0, verbose=False):
    """Nesterov-accelerated smoothed gradient descent with exponential surrogate."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    beta = 0.9
    lr = 8.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            s_y = 1.0
        else:
            y = y / s_y

        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        N = 1
        while N < 3 * n:
            N <<= 1

        P = fft(p, N)
        A_fft = fft(y, N)
        g_full = ifft(P * np.conj(A_fft)).real
        grad = 2.0 * g_full[:n]

        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-15:
            lr_eff = min(lr, 5.0 / (np.sqrt(n) * (g_norm / np.sqrt(n) + 1e-10)))
        else:
            lr_eff = lr

        v = beta * v - lr_eff * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        it += 1

        if verbose and it % 1000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.1f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x * x + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 10, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        a = np.zeros(n)
        w = n // 5
        a[:w] = np.exp(-((np.arange(w) - w // 2) ** 2) / (2 * (w // 4) ** 2))
        a[-w:] = np.exp(-((np.arange(w) - w // 2) ** 2) / (2 * (w // 4) ** 2))
        a = a / a.sum()
    elif mode == 7:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 8:
        # Sinusoidal modulation of uniform
        f = rng.integers(2, 6)
        amp = rng.uniform(0.1, 0.4)
        x = np.linspace(0, 1, n)
        a = 1.0 + amp * np.cos(2 * np.pi * f * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 9:
        # Two-frequency modulation
        f1, f2 = rng.integers(2, 8, size=2)
        a1, a2 = rng.uniform(0.1, 0.3, size=2)
        x = np.linspace(0, 1, n)
        a = 1.0 + a1 * np.cos(2 * np.pi * f1 * x) + a2 * np.sin(2 * np.pi * f2 * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if n == 2:
        x_new = np.linspace(0, 1, new_n)
        a_new = a[0] + (a[1] - a[0]) * x_new
        return np.maximum(a_new, 0)
    # Use smooth interpolation
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    if new_n < n:
        # Downsampling: use local averaging
        indices = (x_new * (n - 1)).astype(int)
        a_new = np.zeros(new_n)
        for i in range(new_n):
            lo = max(0, indices[i] - 1)
            hi = min(n - 1, indices[i] + 1)
            a_new[i] = np.mean(a[lo:hi + 1])
    else:
        # Upsampling: cubic interpolation via scipy
        from scipy.interpolate import interp1d
        f_interp = interp1d(x_old, a, kind='cubic', fill_value="extrapolate")
        a_new = f_interp(x_new)
    return np.maximum(a_new, 0)


def structured_perturbation(a, rng, mode=0):
    """Apply a structured perturbation to a sequence."""
    n = len(a)
    x = np.arange(n) / n
    if mode == 0:
        # Sinusoidal
        f = rng.integers(2, 8)
        amp = rng.uniform(0.05, 0.15)
        phase = rng.uniform(0, 2 * np.pi)
        a_new = a * (1.0 + amp * np.cos(2 * np.pi * f * x + phase))
    elif mode == 1:
        # Double sinusoidal
        f1, f2 = rng.integers(2, 10, size=2)
        a_new = a * (1.0 + 0.08 * np.cos(2 * np.pi * f1 * x) + 0.05 * np.sin(2 * np.pi * f2 * x))
    elif mode == 2:
        # Bump in the middle
        center = n // 2
        width = n // 8
        amp = rng.uniform(0.1, 0.3)
        bump = np.exp(-((x - 0.5) ** 2) / (2 * (width / n) ** 2))
        a_new = a * (1.0 + amp * bump)
    elif mode == 3:
        # Edge boost
        edge_width = n // 6
        a_new = a.copy()
        a_new[:edge_width] *= (1.0 + rng.uniform(0.1, 0.3))
        a_new[-edge_width:] *= (1.0 + rng.uniform(0.1, 0.3))
    elif mode == 4:
        # Multiplicative noise
        noise = rng.normal(1.0, 0.08, n)
        a_new = a * noise
    elif mode == 5:
        # Periodic pattern
        period = rng.integers(4, 20)
        pattern = np.ones(n)
        for i in range(n):
            pattern[i] = 1.0 + 0.1 * np.cos(2 * np.pi * i / period)
        a_new = a * pattern
    else:
        a_new = a.copy()
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining enhanced CD with KKT repair,
    LP polishing, structured perturbations, smoothed GD, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_default = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = 2.0 * parent.size * float(np.max(compute_conv(parent)))
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(n_default) / n_default
        best_score = 2.0 * n_default * float(np.max(compute_conv(best_seq)))
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(800) / 800.0
        best_score = 2.0

    n_best = len(best_seq)
    print(f"Initial best: n={n_best}, score={best_score:.8f}")

    # === Phase 1: Thorough Coordinate Descent with KKT Repair ===
    print(f"\n=== Phase 1: Enhanced CD (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 450.0)
    if p1_budget > 30:
        a_cd = coordinate_descent(best_seq, time_budget=p1_budget, verbose=True, seed=seed)
        obj_cd = compute_obj_normalized(a_cd / a_cd.sum())
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 2: Second CD pass with different seed ===
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 200.0)
    if p2_budget > 20:
        a_cd2 = coordinate_descent(best_seq, time_budget=p2_budget, verbose=False,
                                   seed=seed + 100, tight_tol=1e-15)
        obj_cd2 = compute_obj_normalized(a_cd2 / a_cd2.sum())
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  CD-2: obj={best_score:.8f} **")
        else:
            print(f"  CD-2: obj={obj_cd2:.8f}")

    # === Phase 3: LP Polishing ===
    print(f"\n=== Phase 3: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 200.0)
    if p3_budget > 20:
        for rnd in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(p3_budget / (4 - rnd), remaining * 0.5)
            if alloc < 10:
                break
            a_lp = lp_polish(best_seq, time_budget=alloc, verbose=False,
                             seed=1000 + rnd * 13)
            obj_lp = compute_obj_normalized(a_lp / a_lp.sum())
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
            # Follow-up CD
            remaining = deadline - time.monotonic()
            if remaining > 20 and rnd < 3:
                a_cd3 = coordinate_descent(best_seq, time_budget=min(20, remaining * 0.1),
                                           verbose=False, seed=2000 + rnd * 7)
                obj_cd3 = compute_obj_normalized(a_cd3 / a_cd3.sum())
                if obj_cd3 < best_score:
                    best_score = obj_cd3
                    best_seq = a_cd3.copy()
                    print(f"  CD after LP{rnd}: obj={best_score:.8f} **")

    # === Phase 4: Structured Perturbations + CD ===
    print(f"\n=== Phase 4: Structured Perturbations ===")
    remaining = deadline - time.monotonic()
    n_pert = 12
    for p in range(n_pert):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        mode = p % 6
        a_pert = structured_perturbation(best_seq, rng, mode=mode)
        alloc = min(remaining * 0.08, 50.0)
        if alloc > 15:
            a_opt = coordinate_descent(a_pert, time_budget=alloc, verbose=False,
                                       seed=5000 + p * 31)
            obj_p = compute_obj_normalized(a_opt / a_opt.sum())
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_opt.copy()
                print(f"  Pert {p} (mode {mode}): obj={best_score:.8f} **")
            else:
                print(f"  Pert {p} (mode {mode}): obj={obj_p:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_candidates = sorted(set([
        max(400, n_current - 200), max(400, n_current - 100),
        n_current + 100, n_current + 200, n_current + 400,
        600, 700, 900, 1000, 1100, 1200, 1400
    ]))
    # Filter to reasonable range
    n_filtered = []
    for nc in n_candidates:
        if 500 <= nc <= 1600:
            if all(abs(nc - nf) > 80 for nf in n_filtered):
                n_filtered.append(nc)
    n_filtered = n_filtered[:5]
    print(f"  n candidates: {n_filtered}")

    for n_try in n_filtered:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        if n_try == n_current:
            continue

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.12, 55.0)
        if alloc > 15:
            a_try = coordinate_descent(a_try, time_budget=alloc, verbose=False,
                                       seed=n_try)
        obj_try = compute_obj_normalized(a_try / a_try.sum())
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Smoothed GD Escape ===
    print(f"\n=== Phase 6: Smoothed GD ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        for gd_attempt in range(2):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            pert = rng.normal(0, 0.03, len(best_seq))
            a_pert = best_seq * np.exp(pert)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()

            alloc = min(remaining * 0.25, 80.0)
            a_gd = smoothed_gradient_descent(a_pert, time_budget=alloc,
                                             tau_init=40.0, tau_end=1000.0)
            obj_gd = compute_obj_normalized(a_gd / a_gd.sum())
            if obj_gd < best_score:
                best_score = obj_gd
                best_seq = a_gd.copy()
                print(f"  GD attempt {gd_attempt}: obj={best_score:.8f} **")
            else:
                print(f"  GD attempt {gd_attempt}: obj={obj_gd:.8f}")
            # CD after GD
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_cd_gd = coordinate_descent(a_gd, time_budget=min(30, remaining * 0.3),
                                             verbose=False, seed=8000 + gd_attempt)
                obj_cd_gd = compute_obj_normalized(a_cd_gd / a_cd_gd.sum())
                if obj_cd_gd < best_score:
                    best_score = obj_cd_gd
                    best_seq = a_cd_gd.copy()
                    print(f"  CD after GD{gd_attempt}: obj={best_score:.8f} **")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        for final_round in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            # CD
            cd_alloc = min(remaining * 0.35, 80.0)
            if cd_alloc > 10:
                a_f = coordinate_descent(best_seq, time_budget=cd_alloc,
                                         verbose=False, seed=9000 + final_round * 17,
                                         tight_tol=1e-15)
                obj_f = compute_obj_normalized(a_f / a_f.sum())
                if obj_f < best_score:
                    best_score = obj_f
                    best_seq = a_f.copy()
                    print(f"  Final CD {final_round}: obj={best_score:.8f} **")
            # LP
            remaining = deadline - time.monotonic()
            if remaining > 10 and final_round < 3:
                lp_alloc = min(remaining * 0.25, 40.0)
                if lp_alloc > 8:
                    a_f2 = lp_polish(best_seq, time_budget=lp_alloc,
                                     verbose=False, seed=9500 + final_round * 11)
                    obj_f2 = compute_obj_normalized(a_f2 / a_f2.sum())
                    if obj_f2 < best_score:
                        best_score = obj_f2
                        best_seq = a_f2.copy()
                        print(f"  Final LP {final_round}: obj={best_score:.8f} **")

    # === Phase 8: One more diverse restart if time permits ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        n_cur = len(best_seq)
        for r in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            mode = 8 + r
            a_r = diverse_init(n_cur, rng, mode=mode)
            blend = 0.2 + 0.15 * r
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

            alloc = min(remaining * 0.3, 50.0)
            if alloc > 15:
                a_r = coordinate_descent(a_r, time_budget=alloc, verbose=False,
                                         seed=10000 + r * 23)
                obj_r = compute_obj_normalized(a_r / a_r.sum())
                if obj_r < best_score:
                    best_score = obj_r
                    best_seq = a_r.copy()
                    print(f"  Diverse restart {r}: obj={best_score:.8f} **")
                else:
                    print(f"  Diverse restart {r}: obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = 2.0

    # Ensure valid output
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    result = [float(x) for x in best_seq]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    print(f"\n  evaluate_sequence says: {val:.8f}")
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        print("  WARNING: invalid sequence, using fallback")
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
