"""
Optimizes non-negative step function coefficients to minimize
2*n*max(convolve(a,a))/(sum(a))^2 using LP fixed-point iteration
with multiple solver backends, coordinate descent polishing,
and diverse perturbation strategies.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def lp_solve_cvxpy(a, timeout=5.0):
    """Solve LP: min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1 using cvxpy."""
    import cvxpy as cp
    n = len(a)
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint matrix
    # conv(a, g)[k] = sum_j a[k-j] * g[j] for valid j
    # = sum_j A[k,j] * g[j]
    # where A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    
    # For efficiency, build A as a sparse matrix
    from scipy.sparse import diags
    
    # A is (2n-1) x n
    # A[k,j] = a[k-j] if 0 <= k-j <= n-1
    # For row k, the non-zero entries are at columns j where 0 <= k-j < n
    # i.e., j in [max(0, k-n+1), min(k, n-1)]
    # A[k,j] = a[k-j]
    
    # This is a banded matrix with bandwidth n-1
    # Diagonal offset d = k - j, ranging from -(n-1) to (n-1)
    # For offset d, the entry is a[d] (where d = k-j, so 0 <= d < n means d is the index into a)
    
    # Actually, let me just build it as a dense matrix for simplicity (n=800 → 1599x800 ≈ 1.3M entries)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None

def lp_solve_mw(a, max_iters=50000):
    """Solve LP using multiplicative weights (Hedge algorithm)."""
    n = len(a)
    g = np.ones(n) / n
    running_sum = np.zeros(n)
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    fa = rfft(a, size)
    eta = np.sqrt(2.0 * np.log(max(n, 2)) / max_iters)
    
    best_avg_max = np.inf
    best_avg_g = g.copy()
    
    for t in range(max_iters):
        fg = rfft(g, size)
        conv = irfft(fa * fg, size)[:2 * n - 1]
        k_star = int(np.argmax(conv))
        
        running_sum += g
        
        # Periodically check the running average
        if (t + 1) % 5000 == 0:
            g_avg = running_sum / (t + 1)
            fg_avg = rfft(g_avg, size)
            conv_avg = irfft(fa * fg_avg, size)[:2 * n - 1]
            avg_max = float(np.max(conv_avg))
            if avg_max < best_avg_max:
                best_avg_max = avg_max
                best_avg_g = g_avg.copy()
        
        # Hedge update
        indices = k_star - np.arange(n)
        valid = (indices >= 0) & (indices < n)
        payoffs = np.zeros(n)
        payoffs[valid] = a[indices[valid]]
        
        log_w = -eta * payoffs
        log_w -= log_w.max()
        w = np.exp(log_w)
        g = g * w
        s = g.sum()
        if s > 0:
            g = g / s
        else:
            g = np.ones(n) / n
    
    # Final running average
    g_avg = running_sum / max_iters
    g_avg = np.maximum(g_avg, 0)
    sg = g_avg.sum()
    if sg > 1e-10:
        g_avg = g_avg / sg
    else:
        g_avg = best_avg_g
    
    return g_avg
