"""
Improved minimax convolution optimizer using:
1. Mirror descent (Bregman projection) for fast initial convergence
2. LP-based direction finding with active set refinement
3. Coordinate descent polish
4. Adaptive n exploration with symmetric sequence exploitation

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def normalize(a):
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a_norm):
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp_direction(a):
    n = len(a)
    a_norm = normalize(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    best_g = None
    best_M = np.inf
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    m_val = float(M.value)
                    if m_val < best_M:
                        best_M = m_val
                        best_g = g_val
        except Exception:
            continue
    return best_g, best_M


def subgradient_at(a):
    n = len(a)
    conv = compute_conv(a)
    max_val = np.max(conv)
    threshold = max_val - 1e-10
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    g = np.zeros(n)
    for k in active:
        for i in range(n):
            j1 = k - i
            j2 = i - k
            if 0 <= j1 < n:
                g[i] += a[j1]
            if 0 <= j2 < n:
                g[i] += a[j2]
    g /= len(active)
    return g


def mirror_descent(a, time_budget=60.0, lr_init=0.1):
    n = len(a)
    a = normalize(a).copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    momentum = np.zeros(n)
    beta = 0.5
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        g = subgradient_at(a)
        lr_t = lr / (1.0 + 0.01 * it)
        momentum = beta * momentum + (1 - beta) * (-lr_t * g)
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                lr *= 0.8
                momentum = np.zeros(n)
                stall = 0
                a = best_a.copy()
        it += 1
    return best_a, best_obj


def golden_line_search(a, direction, t_low=0.0, t_high=1.5, tol=1e-8, max_iter=50):
    best_t = t_low
    best_obj = compute_obj(a)
    gr = (np.sqrt(5) - 1) / 2
    c = t_low + (1 - gr) * (t_high - t_low)
    d = t_low + gr * (t_high - t_low)
    fc = compute_obj(normalize((1 - c) * a + c * direction))
    fd = compute_obj(normalize((1 - d) * a + d * direction))
    for _ in range(max_iter):
        if t_high - t_low < tol:
            break
        if fc < fd:
            t_high = d
            d = c
            fd = fc
            c = t_low + (1 - gr) * (t_high - t_low)
            fc = compute_obj(normalize((1 - c) * a + c * direction))
        else:
            t_low = c
            c = d
            fc = fd
            d = t_low + gr * (t_high - t_low)
            fd = compute_obj(normalize((1 - d) * a + d * direction))
        if fc < best_obj:
            best_obj = fc
            best_t = c
        if fd < best_obj:
            best_obj = fd
            best_t = d
    return best_t, best_obj


def grid_line_search(a, direction, n_points=60, t_max=1.5):
    best_t = 0.0
    best_obj = compute_obj(a)
    ts = np.linspace(0, t_max, n_points)
    for t in ts:
        a_try = normalize((1 - t) * a + t * direction)
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    if 0 < best_t < t_max:
        lo = max(0, best_t - t_max / n_points)
        hi = min(t_max, best_t + t_max / n_points)
        ts_fine = np.linspace(lo, hi, 40)
        for t in ts_fine:
            a_try = normalize((1 - t) * a + t * direction)
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t
    return best_t, best_obj


def lp_optimize(a, time_budget=300.0, max_iters=100, verbose=False):
    n = len(a)
    a = normalize(a).copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(12345 + n)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        T0 = time.monotonic()
        g_lp, M_lp = solve_lp_direction(a)
        if g_lp is None:
            break
        t_opt, obj_opt = grid_line_search(a, g_lp, n_points=40, t_max=1.4)
        if t_opt > 1e-6 and obj_opt < best_obj - 1e-14:
            a_new = normalize((1.0 - t_opt) * a + t_opt * g_lp)
            obj_new = compute_obj(a_new)
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            obj_direct = compute_obj(g_lp)
            if obj_direct < best_obj:
                best_obj = obj_direct
                best_a = g_lp.copy()
                a = g_lp.copy()
                stall = 0
            else:
                stall += 1
        it += 1
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"      LP it{it}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")
        if stall >= 5:
            perturbation = rng.normal(0, 0.02 * stall, n)
            a = best_a * np.exp(perturbation)
            a = normalize(a)
            stall = 0
    return best_a, best_obj


def coordinate_descent_polish(a, time_budget=60.0, n_sweeps=50):
    n = len(a)
    a = normalize(a).copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for sweep in range(n_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            old_val = a[i]
            if old_val < 1e-10:
                continue
            # Try adjusting a[i] by moving mass to a random other coordinate
            j = (i + n // 2) % n
            if i == j:
                j = (i + 1) % n
            # Golden section over delta in [-old_val, a[j]]
            lo = -old_val * 0.99
            hi = a[j] * 0.99
            if hi <= lo:
                continue
            best_delta = 0.0
            best_local_obj = best_obj
            # Quick grid search
            for delta in np.linspace(lo, hi, 15):
                a_try = a.copy()
                a_try[i] = old_val + delta
                a_try[j] = a[j] - delta
                if a_try[i] < 0 or a_try[j] < 0:
                    continue
                a_try = np.maximum(a_try, 0)
                s = float(np.sum(a_try))
                if s < 0.01:
                    continue
                a_try = a_try / s
                obj_try = compute_obj(a_try)
                if obj_try < best_local_obj - 1e-12:
                    best_local_obj = obj_try
                    best_delta = delta
            if best_delta != 0:
                a[i] = old_val + best_delta
                a[j] = a[j] - best_delta
                a = np.maximum(a, 0)
                a = normalize(a)
                obj_now = compute_obj(a)
                if obj_now < best_obj:
                    best_obj = obj_now
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a, best_obj


def u_shaped_init(n, sharpness=3.0):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    return normalize(a)


def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n, dtype=np.float64)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = (np.exp(-((idx - n * 0.2) ** 2) / (2 * (n * 0.12) ** 2)) +
             np.exp(-((idx - n * 0.8) ** 2) / (2 * (n * 0.12) ** 2)) + 0.01)
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.03 + 4.0 * (2 * x - 1) ** 4)
    elif mode == 6:
        a = 1.0 + 0.5 * np.cos(2 * np.pi * idx / n * 3)
        a = np.maximum(a, 0.01)
    else:
        a = 1.0 / (0.15 + 3.0 * np.abs(2 * x - 1))
    return normalize(np.maximum(a, 0))


def make_symmetric(a):
    n = len(a)
    sym = 0.5 * (a + a[::-1])
    return normalize(sym)


def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    if n_src == n_dst:
        return a_src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    return normalize(np.maximum(a_dst, 0))
