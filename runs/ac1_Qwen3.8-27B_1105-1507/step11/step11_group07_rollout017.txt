import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n

        print(f"\n=== Phase 1: LP Polish (symmetric), time={remaining:.0f}s ===")
        alloc = min(remaining * 0.25, 200.0)
        a1, obj1 = lp_polish(best_seq, time_budget=alloc, max_iters=50,
                             use_symmetry=True, verbose=True)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj1:.8f}")

    # === Phase 2: Mirror descent ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 2: Mirror Descent, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.2, 150.0)
        a2, obj2 = mirror_descent_optimize(best_seq, time_budget=alloc,
                                           use_symmetry=True, verbose=True, seed=seed)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2: obj={best_score:.8f} **")
        else:
            print(f"  Phase2: obj={obj2:.8f}")

    # === Phase 3: More LP polish after mirror descent ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 3: LP Polish round 2, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.2, 150.0)
        a3, obj3 = lp_polish(best_seq, time_budget=alloc, max_iters=50,
                             use_symmetry=True, verbose=True)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj3:.8f}")

    # === Phase 4: Try asymmetric LP (remove symmetry constraint) ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 4: Asymmetric LP, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.15, 80.0)
        a4, obj4 = lp_polish(best_seq, time_budget=alloc, max_iters=25,
                             use_symmetry=False, verbose=True)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4 (asym): obj={best_score:.8f} **")
        else:
            print(f"  Phase4 (asym): obj={obj4:.8f}")
            # Re-symmetrize
            best_seq = sym_simplex(best_seq)
            best_score = compute_obj(best_seq)

    # === Phase 5: Length exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 5: Length Exploration, time={remaining:.0f}s ===")
        n_best = len(best_seq)
        
        n_options = []
        for delta in [-200, -100, 100, 200, 300, 400, 500]:
            n_try = n_best + delta
            if 200 <= n_try <= 3000 and n_try != n_best:
                n_options.append(n_try)
        n_options = sorted(set(n_options))[:5]
        print(f"  Current n={n_best}, trying: {n_options}")
        
        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            
            # Interpolate to new length
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = sym_simplex(a_try)
            
            # Quick LP polish
            alloc_lp = min(remaining * 0.3, 40.0)
            if alloc_lp > 10:
                a_try, obj_try = lp_polish(a_try, time_budget=alloc_lp,
                                           max_iters=20, use_symmetry=True, verbose=False)
            
            # Quick mirror descent
            remaining = deadline - time.monotonic()
            if remaining > 20:
                alloc_md = min(remaining * 0.3, 30.0)
                a_try, obj_try = mirror_descent_optimize(
                    a_try, time_budget=alloc_md, use_symmetry=True, verbose=False)
            
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 6: Final Polish, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.6, 100.0)
        a6, obj6 = lp_polish(best_seq, time_budget=alloc, max_iters=40,
                             use_symmetry=True, verbose=True)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a6b, obj6b = mirror_descent_optimize(
                best_seq, time_budget=min(remaining * 0.5, 40.0),
                use_symmetry=True, verbose=False, seed=seed + 999)
            if obj6b < best_score:
                best_score = obj6b
                best_seq = a6b.copy()
                print(f"  Final MD: obj={best_score:.8f} **")

    # === Phase 7: Fine-tuning with small perturbations ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print(f"\n=== Phase 7: Fine-tuning, time={remaining:.0f}s ===")
        rng = np.random.default_rng(seed + 777)
        n = len(best_seq)
        for attempts in range(10):
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            for scale in [0.001, 0.002, 0.005]:
                noise = rng.normal(0, scale, n)
                a_p = sym_simplex(best_seq + noise)
                obj_p = compute_obj(a_p)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    print(f"  Fine-tune {attempts} (scale={scale}): obj={best_score:.8f} **")

    # === Phase 8: One final LP ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_final, obj_final = lp_polish(best_seq, time_budget=min(remaining * 0.7, 30.0),
                                       max_iters=15, use_symmetry=True, verbose=False)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
