Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a multi-phase approach: mirror descent for fast initial convergence,
LP alternating minimization for medium convergence, and exact coordinate
descent (golden section search) for final polishing. Includes basin hopping
and multi-length exploration.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj_full(a):
    """Compute full objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(j, min(j + n, 2 * n - 1)):
            A[k, j] = a[k - j]
    return A


def mirror_descent(a, time_budget, seed=42, verbose=True):
    """Mirror descent (multiplicative weight updates) for fast convergence."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    
    eta = 0.1  # Initial step size
    patience = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        M = float(np.max(conv))
        K = np.where(np.abs(conv - M) < 1e-10 * max(M, 1))[0]
        
        # Compute subgradient of max_k conv[k] w.r.t. a[i]
        # d(conv[k])/da[i] = a[k-i] if 0 <= k-i < n, else 0
        # Subgradient of max at a is average over active set K
        subgrad = np.zeros(n)
        for k in K:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    subgrad[i] += a[j]
        if len(K) > 0:
            subgrad /= len(K)
        
        # Mirror descent update: a[i] *= exp(-eta * subgrad[i])
        # This pushes down coordinates that contribute to the peak
        log_update = -eta * subgrad
        # Center to avoid overflow
        log_update -= np.mean(log_update)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj_norm(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            patience = 0
            # Decay eta slowly
            eta *= 0.999
        else:
            patience += 1
            eta *= 0.95
            if patience > 100:
                # Restart with smaller eta
                eta = max(eta * 0.5, 0.001)
                patience = 0
        
        if verbose and patience % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"  MD: obj={best_obj:.10f}, eta={eta:.6f}, elapsed={elapsed:.1f}s")
    
    return best_a

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Multi-phase approach: subgradient mirror descent for fast initial convergence,
LP alternating minimization with warm starts, exact golden-section coordinate
descent for polishing, and basin-hopping with multi-length exploration.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj_full(a):
    """Compute full objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(j + n, 2 * n - 1)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. (a_norm * g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass

    for solver_name in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=getattr(cp, solver_name))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def coordinate_descent_exact(a, time_budget, seed=42, verbose=True):
    """
    Coordinate descent with golden-section search for exact 1D minimization.
    Works on normalized sequence (sum=1).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)
    S = 1.0

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            a_j = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)

            # Compute max of unaffected conv entries
            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))

            # Precompute affected entries data for fast evaluation
            indices = np.arange(k_start, k_end + 1)
            has_2j = (2 * j >= k_start) and (2 * j <= k_end)

            def eval_F(delta):
                if S + delta <= 1e-12:
                    return np.inf
                if has_2j:
                    quad_val = conv[2 * j] + 2.0 * delta * a_j + delta * delta
                    if k_start <= 2 * j - 1:
                        lin_part = conv[k_start:2 * j] + 2.0 * delta * a[indices[:2 * j - k_start]]
                        max_lin = float(np.max(lin_part)) if len(lin_part) > 0 else 0.0
                    else:
                        max_lin = 0.0
                    if 2 * j + 1 <= k_end:
                        lin_part2 = conv[2 * j + 1:k_end + 1] + 2.0 * delta * a[indices[2 * j + 1 - k_start:]]
                        max_lin2 = float(np.max(lin_part2)) if len(lin_part2) > 0 else 0.0
                    else:
                        max_lin2 = 0.0
                    max_aff = max(max_lin, max_lin2, quad_val)
                else:
                    lin_all = conv[k_start:k_end + 1] + 2.0 * delta * a[indices - j]
                    max_aff = float(np.max(lin_all))
                max_total = max(max_unaffected, max_aff)
                return 2.0 * n * max_total / ((S + delta) * (S + delta))

            # Golden section search
            lo = -a_j + 1e-15
            hi = max(0.05, a_j * 3.0)

            if hi - lo < 1e-12:
                continue

            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = eval_F(c)
            fd = eval_F(f_d := d)

            for _ in range(25):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = eval_F(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = eval_F(d)

            best_delta = (lo + hi) / 2.0
            f_best = eval_F(best_delta)
            f_zero = eval_F(0.0)

            if f_best < f_zero - 1e-14:
                # Apply update
                new_a_j = a_j + best_delta
                if new_a_j < 0:
                    new_a_j = 0.0
                delta = new_a_j - a_j
                a[j] = new_a_j
                S_new = S + delta

                # Update conv in-place
                if has_2j:
                    if k_start <= 2 * j - 1:
                        conv[k_start:2 * j] += 2.0 * delta * a[indices[:2 * j - k_start]]
                    if 2 * j + 1 <= k_end:
                        conv[2 * j + 1:k_end + 1] += 2.0 * delta * a[indices[2 * j + 1 - k_start:]]
                    conv[2 * j] += 2.0 * delta * a_j + delta * delta
                else:
                    conv[k_start:k_end + 1] += 2.0 * delta * a[indices - j]

                # Normalize
                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S
                S = 1.0

                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()

        else:

            if no_improve >= 4:

        return a

    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget - 5:
        g_lp, M_lp = solve_lp(a, warm_start=prev_g)
        if g_lp is None:
            break
        prev_g = g_lp

        best_local_obj = best_obj
        best_local_a = a.copy()

        t_values = [1.0, 0.7, 0.5, 0.3, 0.15, 0.1, 0.05, 0.02]
        for t in t_values:

            if s_try < 1e-10:

            best_a = best_local_a

        else:

                strength = 0.003 + 0.002 * min(stall, 8)
                pert = rng.standard_normal(n) * strength

def peak_reduction(a, time_budget, seed=42, verbose=True):
    """
    Subgradient-based peak reduction via multiplicative weight updates.
    Reduces coordinates that contribute to the convolution peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    eta = 0.3
    stall = 0
    iteration = 0

    while time.monotonic() - start < time_budget:
        iteration += 1
        conv = compute_conv(a)
        M = float(np.max(conv))
        # Find active set (positions within 1% of max)
        threshold = M * 0.99
        K = np.where(conv >= threshold)[0]

        if len(K) == 0:
            break

        # Compute subgradient: subgrad[i] = (1/|K|) * sum_{k in K} a[k-i]
        subgrad = np.zeros(n)
        for k in K:
            i_lo = max(0, k - n + 1)
            i_hi = min(k + 1, n)
            for i in range(i_lo, i_hi):
                j_val = k - i
                if 0 <= j_val < n:
                    subgrad[i] += a[j_val]
        subgrad /= len(K)

        # Multiplicative weight update
        log_update = -eta * subgrad
        log_update -= np.mean(log_update)
        # Clip to avoid overflow
        log_update = np.clip(log_update, -5.0, 5.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        obj_new = compute_obj_norm(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            eta *= 0.995
        else:
            stall += 1
            eta *= 0.9
            if stall > 200:
                eta = max(eta, 0.01)
                # Try a perturbation
                a = best_a.copy()
                pert = np.random.default_rng(iteration).standard_normal(n) * 0.01
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                s_val = float(np.sum(a))
                if s_val > 1e-10:
                    a = a / s_val
                stall = 0

        if verbose and iteration % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"  PR iter {iteration}: obj={best_obj:.10f}, eta={eta:.4f}, t={elapsed:.0f}s")

    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    x = np.arange(n)
    for k in range(1, 10):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / np.sqrt(k)
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * x + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_restart_perturb(a, rng, strength=0.05):
    """Random multiplicative perturbation."""
    n = len(a)
    pert = rng.standard_normal(n) * strength
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: multi-phase optimization combining subgradient
    peak reduction, LP alternating, exact coordinate descent (golden section),
    basin hopping, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 25.0

    # Initialize from parent or uniform
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0)
        if a.size == 0 or a.sum() < 0.01:
            raise ValueError("Invalid parent")
        a = a / a.sum()
        print(f"Parent: n={a.size}, obj={compute_obj_norm(a):.10f}")
    except Exception:
        a = np.ones(800) / 800.0
        print(f"Default: n=800, obj={compute_obj_norm(a):.10f}")

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    rng = np.random.default_rng(seed)

    # === Phase 1: Peak Reduction (subgradient) - 15% time ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p1_budget = min(remaining * 0.15, 180.0)
        print(f"\n--- Phase 1: Peak Reduction ({p1_budget:.0f}s) ---")
        a1 = peak_reduction(best_a, time_budget=p1_budget, seed=seed, verbose=True)
        obj1 = compute_obj_norm(a1)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()
            print(f"  P1 result: {best_obj:.10f} **")
        else:
            print(f"  P1 result: {obj1:.10f}")

    # === Phase 2: LP Alternating - 40% time ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        p2_budget = min(remaining * 0.40, 500.0)
        print(f"\n--- Phase 2: LP Alternating ({p2_budget:.0f}s) ---")
        a2 = lp_alternate(best_a, time_budget=p2_budget, seed=seed, verbose=True)
        obj2 = compute_obj_norm(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            print(f"  P2 result: {best_obj:.10f} **")
        else:
            print(f"  P2 result: {obj2:.10f}")

    # === Phase 3: Exact Coordinate Descent - 30% time ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3_budget = min(remaining * 0.30, 400.0)
        print(f"\n--- Phase 3: Exact CD ({p3_budget:.0f}s) ---")
        a3 = coordinate_descent_exact(best_a, time_budget=p3_budget, seed=seed + 1000, verbose=True)
        obj3 = compute_obj_norm(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            print(f"  P3 result: {best_obj:.10f} **")
        else:
            print(f"  P3 result: {obj3:.10f}")

    # === Phase 4: Basin Hopping - 10% time ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        p4_budget = remaining * 0.10
        print(f"\n--- Phase 4: Basin Hopping ({p4_budget:.0f}s) ---")
        n_restarts = 3
        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            alloc = min(remaining / (n_restarts - r), 120.0)
            if alloc < 30:
                break

            # Different perturbation strategies
            if r == 0:
                a_pert = fourier_perturb(best_a, rng, strength=0.015)
            elif r == 1:
                a_pert = random_restart_perturb(best_a, rng, strength=0.03)
            else:
                # Swap halves with noise
                n_half = len(best_a) // 2
                a_pert = best_a.copy()
                a_pert[:n_half] = best_a[n_half:min(2 * n_half, len(best_a))] * 0.95
                a_pert[n_half:min(2 * n_half, len(best_a))] = best_a[:n_half] * 0.95
                a_pert += rng.standard_normal(len(a_pert)) * 0.005
                a_pert = np.maximum(a_pert, 0)
                s_p = float(np.sum(a_pert))
                if s_p > 1e-10:
                    a_pert = a_pert / s_p

            # Quick LP + CD on perturbed
            a_hp = lp_alternate(a_pert, time_budget=alloc * 0.55, seed=seed + 5000 * (r + 1), verbose=False)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_hp = coordinate_descent_exact(a_hp, time_budget=min(remaining * 0.3, 40.0),
                                                seed=seed + 7000 * (r + 1), verbose=False)

            obj_hp = compute_obj_norm(a_hp)
            if obj_hp < best_obj:
                best_obj = obj_hp
                best_a = a_hp.copy()
                print(f"  Restart {r}: {best_obj:.10f} **")
            else:
                print(f"  Restart {r}: {obj_hp:.10f}")

    # === Phase 5: Multi-length exploration - remaining time ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n--- Phase 5: Multi-Length ({remaining:.0f}s) ---")
        n_best = len(best_a)
        n_options = []
        for delta_n in [50, 100, -50, -100, 200, -200]:
            n_try = n_best + delta_n
            if 300 <= n_try <= 2000:
                n_options.append(n_try)

        for n_try in n_options[:4]:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break

            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_a)
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try

            alloc = min(remaining * 0.35, 100.0)
            if alloc > 30:
                a_try = lp_alternate(a_try, time_budget=alloc * 0.6, seed=n_try * 3, verbose=False)
                remaining = deadline - time.monotonic()
                if remaining > 20:
                    a_try = coordinate_descent_exact(a_try, time_budget=min(remaining * 0.3, 35.0),
                                                     seed=n_try * 7, verbose=False)

            obj_try = compute_obj_norm(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                print(f"  ** n={n_try}: {best_obj:.10f} **")
            else:
                print(f"  n={n_try}: {obj_try:.10f}")

    # === Phase 6: Final polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n--- Phase 6: Final Polish ({remaining:.0f}s) ---")
        a_fin = coordinate_descent_exact(best_a, time_budget=min(remaining * 0.7, 60.0),
                                         seed=seed + 99999, verbose=False)
        obj_fin = compute_obj_norm(a_fin)
        if obj_fin < best_obj:
            best_obj = obj_fin
            best_a = a_fin.copy()
            print(f"  Final: {best_obj:.10f} **")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_obj:
        best_obj = val
    elif val == np.inf:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
