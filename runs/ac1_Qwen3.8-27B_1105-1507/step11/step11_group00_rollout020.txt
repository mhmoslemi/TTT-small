Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses coordinate descent with exact 1D optimization, subgradient descent with
adaptive step sizes, simulated annealing for global exploration, and LP polishing.
Coordinate descent provides fine-grained control that gradient methods miss.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def coord_descent_pass(a, rng, max_evals_per_coord=30):
    """
    One pass of coordinate descent. For each coordinate (in random order),
    find the best value using golden section search on the 1D objective.
    Returns the updated sequence and the number of improvements.
    """
    n = len(a)
    a = a.copy()
    current_obj = compute_obj(a)
    conv = compute_conv(a / np.sum(a))
    s_total = float(np.sum(a))
    
    # Work in normalized space for efficiency
    a_norm = a / s_total
    
    indices = rng.permutation(n)
    improvements = 0
    
    for i in indices:
        # Compute the 1D objective for coordinate i
        # conv_k = base_k + coeff_k * x + quad_k * x^2
        # where x = a_norm[i]
        
        # Pre-compute the base convolution (without coordinate i)
        # Instead of recomputing, compute the contribution of coordinate i to each lag
        old_val = a_norm[i]
        
        # For each lag k, the change when we modify a_norm[i]:
        # delta_conv_k = (new - old) * (a_norm[k-i] + a_norm[k-i] if k!=2i) + (new^2 - old^2) if k==2i
        # Simpler: just compute conv for a few candidate values
        
        # Candidate values to try
        # Current value, 0, and some values around current
        candidates = [0.0, old_val]
        # Add values in a range around old_val
        spread = max(old_val, 1.0/n) * 2
        candidates += list(old_val + spread * np.linspace(-0.5, 0.5, max_evals_per_coord - 2))
        candidates = [max(0.0, c) for c in candidates]
        
        # Evaluate objective for each candidate
        best_val = old_val
        best_obj_i = current_obj

            a_try = a_norm.copy()
            a_try[i] = c
            s_try = float(np.sum(a_try))

                curr_idx = int(np.argmin(np.abs(candidates - old_val)))
                if new_objs[best_idx] > new_objs[curr_idx]:
                    delta_obj = new_objs[best_idx] - new_objs[curr_idx]
                    if delta_obj > anneal_T * 50:
                        best_idx = curr_idx
                    elif rng.random() > np.exp(-delta_obj / anneal_T):
                        best_idx = curr_idx

            new_val = candidates[best_idx]

            if new_objs[best_idx] < current_obj - 1e-14 and abs(new_val - old_val) > 1e-14:
                # Apply update
                diffs_s = new_val - old_val
                deltas_s = lin * diffs_s + quad * (new_val ** 2 - old_val ** 2)
                conv[affected] = conv_aff + deltas_s
                a[i] = new_val
                s_new = 1.0 - old_val + new_val
                if s_new > 1e-15:
                    a = a / s_new
                    conv = conv / (s_new ** 2)
                    current_obj = 2.0 * n * float(np.max(conv))
                    imp += 1

                if current_obj < best_obj:
                    best_obj = current_obj
                    best_a = a.copy()

        total_imp += imp
        if verbose and (it % 20 == 0):
            print(f"      CD it{it}: obj={best_obj:.8f}, imp={imp}")
        if imp == 0 and it > 10:
            break

    return best_a, compute_conv(best_a), best_obj, total_imp


def subgradient_step(a, conv, eta, top_k=5):
    """One subgradient step using top-k active lags. Returns updated a (normalized)."""
    n = len(a)
    idx = np.argsort(conv)[-top_k:]
    p = np.zeros(2 * n - 1)
    p[idx] = 1.0 / top_k

    # Gradient: 2 * (a * p_reversed) projected
    # grad_i = 2 * sum_k p_k * (a_{k-i} + a_{k-i} if k!=2i ...)
    # Simpler: grad = 2 * (a * p)_i where * is cross-correlation
    # Actually: d/d a_i [sum_k p_k * conv_k] = 2 * sum_k p_k * a_{k-i} (with bounds)
    # This is 2 * (p * a_reversed)_{i + n - 1}... let me just use the direct formula

    # conv_k = sum_j a_j * a_{k-j}
    # d conv_k / d a_i = a_{k-i} + a_{k-i} (if k-i != i) or a_{k-i} (if k-i = i)
    # Actually: d conv_k / d a_i = a_{k-i} [0<=k-i<n] + a_{k-i} [0<=k-i<n, k-i!=i]
    # = a_{k-i} * (1 + [k!=2i]) [0<=k-i<n]

    # grad_i = 2 * sum_k p_k * a_{k-i} * [0<=k-i<n] + 2 * sum_k p_k * a_{k-i} * [0<=k-i<n, k!=2i]
    # = 2 * sum_k p_k * a_{k-i} * [0<=k-i<n] + 2 * sum_k p_k * a_{k-i} * [0<=k-i<n] - 2 * p_{2i} * a_i * [0<=2i<2n-1]
    # = 4 * (p_correlated with a at offset i) - 2 * p_{2i} * a_i

    # Simpler to just compute:
    # grad_i = sum_k p_k * (d conv_k / d a_i) * 2
    # d conv_k / d a_i = a_{k-i} if 0<=k-i<n, plus another a_{k-i} if k!=2i
    # So grad_i = 2 * sum_k p_k * a_{k-i} [0<=k-i<n] + 2 * sum_k p_k * a_{k-i} [0<=k-i<n, k!=2i]

    # Using FFT: cross-correlation
    N_fft = 1
    while N_fft < 3 * n - 1:
        N_fft <<= 1
    cross = irfft(rfft(a, N_fft) * rfft(p[::-1], N_fft), N_fft)
    # cross[i + n - 1] = sum_k a[k] * p[k - i + n - 1]... this is getting confusing
    # Let me just do it directly for the active lags (at most top_k of them)

    grad = np.zeros(n)
    for k in idx:
        pk = p[k]
        ki = k - np.arange(n)  # k - i for i = 0..n-1
        valid = (ki >= 0) & (ki < n)
        if k == 2 * i:  # this won't work for all i at once
            pass
        # For lag k: d conv_k / d a_i = a_{k-i} if 0<=k-i<n
        # Plus another a_{k-i} if 0<=k-i<n and k-i != i
        contrib = np.zeros(n)
        contrib[valid] = a[ki[valid]]
        # Add the second term (k-i != i means i != k-i means 2i != k)
        second_valid = valid & (np.arange(n) != (k - np.arange(n)))
        # 2i != k means i != k/2, i.e., i != k - i
        contrib2 = np.zeros(n)
        i_vals = np.arange(n)
        second_valid2 = valid & (2 * i_vals != k)
        contrib2[second_valid2] = a[ki[second_valid2]]
        grad += 2.0 * pk * (contrib + contrib2)

    # Mirror descent step
    g_max = float(np.max(np.abs(grad)))
    if g_max < 1e-15:
        return a.copy()
    eta_eff = eta / (g_max + 1e-10)
    a_new = a * np.exp(-eta_eff * grad)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-15:
        return a.copy()
    return a_new / s


def lp_direction(a):
    """LP direction: min M s.t. 2*conv(a,g)_k - conv(a,a)_k <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    conv_aa = compute_conv(a_norm)

    # Build A matrix: A[k,j] = a_norm[k-j] if 0<=k-j<n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        ks = np.arange(max(0, j), min(2 * n - 1, j + n))
        A[ks, j] = a_norm[ks - j]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [2.0 * (A @ g) - M <= conv_aa, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_g = float(np.sum(g_val))
                if s_g > 1e-10:
                    return g_val / s_g
        except Exception:
            continue
    return None


def resample(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            return
        new_a = new_a / s
        obj = compute_obj_normalized(new_a)
        if obj < best_obj - 1e-14:

            best_obj = obj
