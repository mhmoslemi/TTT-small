    return irfft(fa * fa, size)[:2 * n - 1]

    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp_minpeak(a, time_budget=20.0):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    start = time.monotonic()
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            if time.monotonic() - start > time_budget:
                break
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget, seed=42, verbose=False):
    """LP-based polishing with multiple blending factors and restarts."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp_minpeak(a, time_budget=min(remaining * 0.4, 18.0))
        if g_lp is None:
            break

        best_t_obj = best_obj
        best_a_local = best_a.copy()
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 3:
                # More aggressive perturbation
                pert = rng.normal(0, 0.02 * stall, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / float(a.sum())
                stall = 0

    return best_a


def subgradient_method(a, time_budget, seed=42, verbose=False):
    """
    Projected subgradient method with averaging for the non-smooth problem.
    Minimizes max_k conv(a,a)[k] subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    a_avg = np.zeros(n)
    lr_base = 5.0 / np.sqrt(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        lr = lr_base * (1.0 - 0.6 * progress)

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        # Find near-maximum positions (for smoothing subgradient)
        threshold = max_val * 0.99
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Compute subgradient
        sg = np.zeros(n)
        for k in active:
            k_lo = max(0, k - n + 1)
            k_hi = min(n - 1, k)
            idx = np.arange(k_lo, k_hi + 1)
            sg[idx] += 2.0 * a[k - idx]
        sg /= len(active)

        # Projected update
        a_new = a - lr * sg
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        a_avg = a_avg + a_new
        it += 1

        # Check averaged iterate
        a_check = a_avg / it
        obj_avg = 2.0 * n * float(np.max(compute_conv(a_check)))
        if obj_avg < best_obj - 1e-10:
            best_obj = obj_avg
            best_a = a_check.copy()

        obj_cur = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_cur < best_obj - 1e-10:
            best_obj = obj_cur
            best_a = a_new.copy()

        a = a_new

    # Final: polish the averaged iterate
    a_avg_final = a_avg / max(it, 1)
    obj_final = 2.0 * n * float(np.max(compute_conv(a_avg_final)))
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = a_avg_final.copy()

    return best_a


def palindromic_cd(a, time_budget, seed=42):
    """
    Coordinate descent with palindromic constraint: a[i] = a[n-1-i].
    Reduces effective dimension by half.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    # Enforce palindromic
    half = (n + 1) // 2
    a_pal = np.zeros(n)
    for i in range(half):
        a_pal[i] = (a[i] + a[n - 1 - i]) / 2
        a_pal[n - 1 - i] = a_pal[i]
    a_pal = a_pal / float(np.sum(a_pal))

    best_a = a_pal.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a_pal)))

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    conv = compute_conv(a_pal)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a_pal)
        S = float(np.sum(a_pal))
        obj = 2.0 * n * float(np.max(conv)) / (S * S)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_pal.copy()

        for i in range(half):
            if time.monotonic() - start >= time_budget:
                break

            j = n - 1 - i
            # For palindromic, changing a[i] means changing a[j] too
            # This is a 2D subproblem, but we can treat it approximately
            # by fixing a[j] = a[i] (the constraint)
            v = a_pal[i]
            S_other = S - 2.0 * v

            # Affected positions: those involving index i or j
            # conv[k] has terms a[i]*a[k-i] and a[j]*a[k-j]
            # With a[i] = a[j] = x:
            # conv[k] changes by (x-v)*(a[k-i] + a[k-j]) for k where both terms exist
            # Plus quadratic terms at k=2i and k=2j

            # Simplified: just do 1D on a[i] with a[j] fixed at same value
            # Actually for true palindromic, we need to track both

            # For simplicity, do individual coordinate updates but then re-enforce symmetry
            k_min_i = i
            k_max_i = min(2 * n - 2, i + n - 1)
            k_min_j = j
            k_max_j = min(2 * n - 2, j + n - 1)

            # Just do a simple 1D optimization on a[i] (ignoring the symmetry for now)
            # Then re-enforce at the end
            k_q = 2 * i
            all_ks = np.arange(k_min_i, k_max_i + 1)
            mask = (all_ks != k_q)
            ks_lin = all_ks[mask]
            a_ki = a_pal[ks_lin - i]
            c_lin = conv[ks_lin].copy()
            c_q = float(conv[k_q]) if 0 <= k_q <= 2 * n - 2 else 0.0

            affected = np.zeros(2 * n - 1, dtype=bool)
            affected[k_min_i:k_max_i + 1] = True
            c_unaff = float(np.max(conv[~affected])) if np.any(~affected) else 0.0

            x_lo = 0.0
            x_hi = max(3.0 * v, 5.0 / n)
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f1d_val(c_pt, v, S - v, n, c_lin, a_ki, c_q, c_unaff)
            fd = f1d_val(d_pt, v, S - v, n, c_lin, a_ki, c_q, c_unaff)

            for _ in range(25):
                if x_hi - x_lo < 1e-12 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt; d_pt = c_pt; fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f1d_val(c_pt, v, S - v, n, c_lin, a_ki, c_q, c_unaff)
                else:
                    x_lo = c_pt; c_pt = d_pt; fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f1d_val(d_pt, v, S - v, n, c_lin, a_ki, c_q, c_unaff)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            a_pal[i] = x_opt
            a_pal[j] = x_opt
            S = float(np.sum(a_pal))

    # Final re-enforce and normalize
    for i in range(half):
        val = (a_pal[i] + a_pal[n - 1 - i]) / 2
        a_pal[i] = val
        a_pal[n - 1 - i] = val
    a_pal = np.maximum(a_pal, 0)
    a_pal = a_pal / float(np.sum(a_pal))

    obj_final = compute_obj(a_pal)
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = a_pal.copy()

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        # Bimodal
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * (n // 10) ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Semicircle (optimal for certain spectral problems)
        x = np.linspace(-0.999, 0.999, n)
        a = 1.0 / np.sqrt(1.0 - x ** 2)
        a = a / a.sum()
    elif mode == 4:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 5:
        # Cosine-weighted
        x = np.linspace(0, np.pi, n)
        a = np.abs(np.cos(x)) ** rng.uniform(1, 5)
        a = a / a.sum()
    elif mode == 6:
        # Sinc-like (optimal for bandlimited signals)
        x = np.linspace(-5, 5, n)
        a = np.sinc(x / 5.0) ** 2
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    else:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    return a

            s = float(np.sum(parent_arr))

                parent = parent_arr / s

                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")

        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    n_cur = len(best_seq)

    print(f"\n=== Phase 1: Coordinate Descent (n={n_cur}) ===")

                print(f"  LP round {rnd}: obj={obj_lp:.8f}")

    # === Phase 3: CD after LP ===

    remaining = deadline - time.monotonic()

    if p3_budget > 15:
        a_cd2 = coordinate_descent(best_seq, time_budget=p3_budget, seed=777)
        obj_cd2 = compute_obj(a_cd2)

            print(f"  Subgrad: obj={obj_sg:.8f}")

    # === Phase 5: Palindromic optimization ===
    print(f"\n=== Phase 5: Palindromic CD ===")

    remaining = deadline - time.monotonic()
