def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming a is normalized (sum=1). Returns 2n*max(conv)."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp_active(a, active_lags):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for k in active_lags, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a_norm = a / s
    A_full = build_conv_matrix(a_norm)
    A = A_full[active_lags, :]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.GLOP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:
                    g_val = g_val / g_sum
                    conv_check = A_full @ g_val
                    M_actual = float(np.max(conv_check))
                    return g_val, M_actual
        except Exception:
            continue
    return None, None

            bb_step *= 0.5
        else:
            a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1
            if patience > 30:
                # Escape: add perturbation
                pert = rng.normal(0, 0.03, n)
                a_pert = a * np.exp(pert)
                a_pert = np.maximum(a_pert, 0)
                sp = float(np.sum(a_pert))
                if sp > 1e-10:
                    a_pert /= sp
                    obj_pert = 2.0 * n * float(np.max(compute_conv(a_pert)))
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert.copy()
                        a = a_pert
                    v = np.zeros(n)
                    bb_step = 1.0 / np.sqrt(n)
                patience = 0

        it += 1
        if verbose and it % 200 == 0:
            print(f"    NSGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.1f}s")

    return best_a


def lp_polish_enhanced(a, time_budget=60.0, verbose=False, seed=42):
    """LP polishing with active-set identification and adaptive blending."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Identify active lags (within 1% of max)
        threshold = max_conv * 0.999
        active_lags = np.where(conv >= threshold)[0]
        # Also include lags within 5% for robustness
        threshold2 = max_conv * 0.995
        active_lags2 = np.where(conv >= threshold2)[0]
        use_lags = active_lags2 if len(active_lags2) < n else active_lags

        g_lp, M_lp = solve_lp_active(a, use_lags)
        if g_lp is None:
            break

        # Adaptive blending: try multiple t values
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        t_values = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 3:
                # Perturb and retry
                pert = rng.normal(0, 0.015, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0

    return best_a


def resample_sequence(a, new_n, method='cubic'):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * (w ** 2)))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / np.sqrt(1 - x * x + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        a = np.zeros(n)
        w = n // 5
        a[:w] = np.exp(-((np.arange(w) - w // 2) ** 2) / (2 * (w // 4) ** 2))
        a[-w:] = np.exp(-((np.arange(w) - w // 2) ** 2) / (2 * (w // 4) ** 2))
        a = a / a.sum()
    elif mode == 7:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def pair_coordinate_update(a, i, j, conv, n):
    """Optimize coordinates i and j simultaneously via 2D search."""
    v_i, v_j = a[i], a[j]
    S = float(np.sum(a))
    S_other = S - v_i - v_j
    if S_other < 1e-12:
        return v_i, v_j

    def f_2d(xi, xj, v_i=v_i, v_j=v_j, S_other=S_other, n=n, i=i, j=j, a=a, conv=conv):
        # Compute max conv after changing a[i] and a[j]
        # Only affected conv entries: those involving i or j
        # For simplicity, compute the full conv (slower but correct)
        a_new = a.copy()
        a_new[i] = xi
        a_new[j] = xj
        conv_new = conv.copy()
        # Update conv entries affected by i
        for k in range(max(0, i), min(2 * n - 1, i + n)):
            old_contrib = conv_new[k]
            # This is complex; just recompute affected parts
            pass
        # Simpler: just recompute the full conv for this 2D eval
        conv_full = compute_conv(a_new)
        max_c = float(np.max(conv_full))
        denom = (S_other + xi + xj) ** 2
        if denom < 1e-30:
            return 1e30
        return 2.0 * n * max_c / denom

    # Grid search over a small region
    scale_i = max(v_i, 1.0 / n, 1e-4)
    scale_j = max(v_j, 1.0 / n, 1e-4)

    best_val = f_2d(v_i, v_j)
    best_xi, best_xj = v_i, v_j

    # Coarse grid
    for di in [-1.0, -0.5, -0.25, 0.0, 0.25, 0.5, 1.0, 2.0]:
        xi = max(0.0, v_i + di * scale_i)
        for dj in [-1.0, -0.5, -0.25, 0.0, 0.25, 0.5, 1.0, 2.0]:
            xj = max(0.0, v_j + dj * scale_j)
            val = f_2d(xi, xj)
            if val < best_val:
                best_val = val
                best_xi, best_xj = xi, xj

    return best_xi, best_xj


def propose_candidate(seed=42, budget_s=1500):

            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = 2.0 * parent.size * float(np.max(compute_conv(parent)))
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(n_default) / n_default
        best_score = 2.0 * n_default * float(np.max(compute_conv(best_seq)))
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

        best_score = 2.0

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")

    # === Phase 1: Quick CD warmup ===
    print(f"\n=== Phase 1: CD Warmup ===")

        else:
