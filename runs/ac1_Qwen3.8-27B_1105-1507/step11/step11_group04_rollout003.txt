"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses peak-targeted coordinate descent with 1D line search, active peak tracking,
adaptive perturbations, and multi-restart exploration with structured initializations.

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def line_search_coord(a, conv, j, n, num_points=30):
    """
    For coordinate j, find the best new value via 1D search.
    Returns (new_value, new_objective).
    """
    a_j = a[j]
    k_start = j
    k_end = min(j + n - 1, 2 * n - 2)
    
    # Unaffected max
    max_unaffected = 0.0
    if k_start > 0:
        max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
    if k_end < 2 * n - 2:
        max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))
    
    # Precompute the "offset" and "slope" for each affected conv entry
    indices = np.arange(k_start, k_end + 1)
    # For k in affected range:
    # conv_new[k] = conv[k] + 2*(w - a_j)*a[k-j] (for k != 2j)
    # conv_new[k] = conv[k] - a_j^2 + w^2 (for k = 2j, i.e., the diagonal)
    
    # Precompute: base[k] and slope[k] such that conv_new[k] ≈ base[k] + slope[k]*w
    # For k != 2j: conv_new[k] = conv[k] - 2*a_j*a[k-j] + 2*a[k-j]*w
    #   base = conv[k] - 2*a_j*a[k-j], slope = 2*a[k-j]
    # For k = 2j: conv_new[k] = conv[k] - a_j^2 + w^2
    #   This is quadratic, handle separately
    
    offsets = indices - j  # These are the values a[k-j]
    a_offsets = a[offsets]
    
    # Linear part (for k != 2j)
    base = conv[k_start:k_end+1].copy() - 2.0 * a_j * a_offsets
    slopes = 2.0 * a_offsets
    
    # Handle the diagonal (k = 2j)
    diag_idx = 2 * j - k_start
    if 0 <= diag_idx < len(indices):
        # At k=2j: conv_new = conv[2j] - a_j^2 + w^2
        # We'll handle this by adding w^2 separately
        base[diag_idx] = conv[2*j] - a_j * a_j  # This will be base + 0*w + w^2
        slopes[diag_idx] = 0.0
    
    # Now conv_new[k] ≈ base + slopes * w (linear approx, except diagonal has w^2)
    # For the search, we'll evaluate exactly.
    
    # Range for w: [0, upper_bound]
    # Upper bound: we don't want w to be too large
    upper = max(a_j * 5, 1.0 / n * 10, 0.1)
    upper = min(upper, 1.0)  # Since sum = 1, no single element should exceed 1
    
    # Grid search
    w_values = np.linspace(0, upper, num_points)
    # Add some points around the current value
    if a_j > 0:
        local = np.linspace(max(0, a_j * 0.5), min(upper, a_j * 2.0), 15)
        w_values = np.unique(np.concatenate([w_values, local]))
    
    best_obj = np.inf
    best_w = a_j
    
    for w in w_values:
        if abs(w - a_j) < 1e-15:
            continue
        S_new = 1.0 + (w - a_j)
        if S_new <= 1e-10:
            continue
        
        # Compute affected conv values
        affected = base + slopes * w
        # Add the quadratic term for the diagonal
        if 0 <= diag_idx < len(indices):
            affected[diag_idx] += w * w
        
        max_conv = max(max_unaffected, float(np.max(affected)))
        obj = 2.0 * n * max_conv / (S_new * S_new)
        
        if obj < best_obj:
            best_obj = obj
            best_w = w
    
    if best_w == a_j:
        return a_j, None
    
    return best_w, best_obj


def apply_update(a, conv, j, new_val):
    """Apply the update to a[j] and update conv in-place. Then renormalize."""
    n = len(a)
    old_val = a[j]
    delta = new_val - old_val
    
    k_start = j
    k_end = min(j + n - 1, 2 * n - 2)
    indices = np.arange(k_start, k_end + 1)
    
    # Update conv
    # For k != 2j: conv[k] += 2*delta*a[k-j]
    # For k = 2j: conv[k] += 2*delta*a[j] + delta^2 = 2*delta*old_val + delta^2
    offsets = indices - j
    conv[k_start:k_end+1] += 2.0 * delta * a[offsets]
    
    # Fix the diagonal
    if k_start <= 2*j <= k_end:
        conv[2*j] += delta * delta  # The extra delta^2 term
    
    a[j] = new_val
    
    # Renormalize
    S_new = 1.0 + delta
    if S_new > 1e-10:
        inv = 1.0 / S_new
        a *= inv
        conv *= inv * inv


def cd_sweep(a, conv, rng, verbose=False):
    """One sweep of coordinate descent with line search."""
    n = len(a)
    order = rng.permutation(n)
    improved = 0
    
    for j in order:
        new_val, _ = line_search_coord(a, conv, j, n, num_points=25)
        if new_val != a[j]:
            apply_update(a, conv, j, new_val)
            improved += 1
    
    return improved


def fourier_perturb(a, rng, strength=0.01):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 10):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.0) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_perturb(a, rng, strength=0.01):
    """Add random perturbation."""
    n = len(a)
    pert = rng.standard_normal(n) * strength
    a_new = a * (1.0 + pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def make_ramp(n, alpha=1.0):
    """Create a ramp sequence: a[j] = (j+1)^alpha, normalized."""
    a = np.power(np.arange(1, n+1), alpha)
    return a / a.sum()


def make_triangle(n):
    """Create a triangular sequence."""
    if n % 2 == 0:
        a = np.concatenate([np.arange(1, n//2 + 1), np.arange(n//2, 0, -1)])
    else:
        a = np.concatenate([np.arange(1, n//2 + 2), np.arange(n//2, 0, -1)])
    return a / a.sum()


def make_exp_decay(n, rate=0.5):
    """Create an exponential decay sequence."""
    a = np.exp(-rate * np.arange(n) / n)
    return a / a.sum()


def make_bell(n, width=0.3):
    """Create a bell-shaped sequence."""
    x = np.arange(n) / n
    a = np.exp(-((x - 0.5) ** 2) / (2 * width ** 2))
    return a / a.sum()
