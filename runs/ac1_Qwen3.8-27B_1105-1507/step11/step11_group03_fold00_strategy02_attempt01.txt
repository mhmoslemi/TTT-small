def evaluate_sequence(seq):
    n = len(seq)
    # Compute the convolution if the sum of seq is within a reasonable range
    total = np.sum(seq)
    if not is_finite(seq) or total < 0.01:
        return np.inf

    # Clip each element to be between 0 and 1000
    seq = np.clip(seq, 0, 1000)

    # Normalize the sequence if the total sum is greater than 0.01
    if total > 0.01:
        normalized_seq = seq / total
    else:
        normalized_seq = seq

    # Compute the convolution = self-convolution of the normalized sequence
    convolved = np.convolve(normalized_seq, normalized_seq, mode='full')
    max_val = np.max(convolved)
    return 2 * n * max_val / (total ** 2)
