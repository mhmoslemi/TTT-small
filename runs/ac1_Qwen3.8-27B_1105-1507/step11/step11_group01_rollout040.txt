import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective 2*n*max(conv)/sum^2 for non-negative a."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return float(2.0 * n * np.max(conv) / (s * s))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a, solver_preference=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M, binding_indices) or (None, None, None) on failure.
    binding_indices are indices k where the constraint is tight at optimum.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    if solver_preference:
        solvers = [solver_preference] + solvers
    
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:
                    g_val = g_val / g_sum
                    conv_check = A @ g_val
                    M_actual = float(np.max(conv_check))
                    # Find binding constraints
                    binding = np.where(conv_check >= M_actual * (1 - 1e-6))[0]
                    return g_val, M_actual, binding
        except Exception:
            continue
    return None, None, None


def golden_section_line_search(a, direction, f_current, n, max_iter=50):

        if s < 1e-10:
            return 1e30
        a_new = a_new / s
        conv = compute_conv(a_new)
        return float(2.0 * n * np.max(conv))
    
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f_of_t(c)
    fd = f_of_t(d)

            d = c
            fd = fc

        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)

def targeted_peak_reduction(a, time_budget=60.0, seed=42):
    """
    Targeted approach: identify the peak convolution position and
    specifically reduce contributions to it.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    for it in range(100):
        if time.monotonic() - start >= time_budget:
            break

        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]

        # Find contributing pairs
        # conv[peak_idx] = sum_j a[j] * a[peak_idx - j] for valid j
        j_min = max(0, peak_idx - n + 1)
        j_max = min(n - 1, peak_idx)
        
        # Contribution of each pair to the peak
        contributions = np.zeros(n)
        for j in range(j_min, j_max + 1):
            partner = peak_idx - j
            if 0 <= partner < n:
                contrib = a[j] * a[partner]
                contributions[j] += contrib
                if partner != j:
                    contributions[partner] += contrib
    
        # Find the "cheapest" coordinate to reduce (contributes to peak
        # but is less critical for other high convolution values)
        # Score: contribution to peak / (contribution to other high values + epsilon)
        other_high = conv.copy()
        other_high[peak_idx] = 0
        threshold = peak_val * 0.8
        
        high_other_idx = np.where(other_high > threshold)[0]
        
        other_contrib = np.zeros(n)
        for k in high_other_idx:
            j_min_k = max(0, k - n + 1)
            j_max_k = min(n - 1, k)
            for j in range(j_min_k, j_max_k + 1):
                partner = k - j
                if 0 <= partner < n:
                    other_contrib[j] += a[j] * a[partner]
                    if partner != j:
                        other_contrib[partner] += a[j] * a[partner]
        
        # Score: how much we can reduce peak per unit "damage" elsewhere
        safe_reduce = np.zeros(n)
        for i in range(n):
            peak_contrib = contributions[i]
            other_danger = other_contrib[i]
            if peak_contrib > 1e-12:
                safe_reduce[i] = peak_contrib / (other_danger + 1e-8)
        
        # Reduce top few "safest" contributors
        top_k = min(5, n)
        top_coords = np.argsort(-safe_reduce)[:top_k]
        
        for i in top_coords:
            if a[i] < 1e-10:
                continue
            # Try reducing coordinate i
            for factor in [0.7, 0.5, 0.3]:
                a_try = a.copy()
                a_try[i] *= factor
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
                if obj_try < best_obj:
                    best_obj = obj_try
                    a = a_try.copy()
                    best_a = a.copy()
                    break
    
    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two bumps
        x = np.arange(n)
        a = np.exp(-((x - n*0.25)**2) / (2*(n*0.1)**2)) + np.exp(-((x - n*0.75)**2) / (2*(n*0.1)**2))
        a = a / a.sum()
    elif mode == 3:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2*np.pi*x))
        a = a / a.sum()
    elif mode == 4:
        # Periodic modulation
        x = np.arange(n)
        freq = rng.uniform(2, 8)
        a = 1.0 + 0.5 * np.sin(2*np.pi*freq*x/n)
        a = np.maximum(a, 0.1)
        a = a / a.sum()
    elif mode == 5:
        # Logarithmic bumps
        x = np.arange(n)
        a = 1.0
        for _ in range(3):
            c = rng.integers(0, n)
            w = rng.integers(n//10, n//3)
            a += np.exp(-((x-c)**2) / (2*w*w))
        a = a / a.sum()
    elif mode == 6:
        # Alternating high-low
        a = np.ones(n)
        a[::2] = rng.uniform(0.3, 0.7)
        a[1::2] = rng.uniform(1.0, 2.0)
        a = a / a.sum()
    elif mode == 7:
        # Horseshoe
        x = np.linspace(-1, 1, n)
        a = 1.0 / (1.0 + x*x)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients using enhanced LP-direction finding with
golden-section line search, interleaved coordinate descent, KKT-guided
subgradient polishing, and systematic multi-n exploration.
Key improvements: (1) adaptive LP-line-search hybrid as primary optimizer with
convergence detection, (2) KKT subgradient for exploitation between LP steps,
(3) broader n-sweep with warm-start resampling and per-n LP+CD,
(4) targeted peak reduction via pair-coordinate updates,
(5) diverse restarts with LP-based polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time
import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective 2*n*max(conv)/sum^2 for non-negative a."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return float(2.0 * n * np.max(conv) / (s * s))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M, binding_indices) or (None, None, None) on failure.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:
                    g_val = g_val / g_sum
                    conv_check = A @ g_val
                    M_actual = float(np.max(conv_check))
                    binding = np.where(conv_check >= M_actual * (1.0 - 1e-6))[0]
                    return g_val, M_actual, binding
        except Exception:
            continue
    return None, None, None


def f_1d_eval(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i):
    """Evaluate 1D objective for coordinate descent subproblem."""
    if len(a_ki_lin) > 0:
        lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
        max_lin = float(np.max(lin_vals))
    else:
        max_lin = 0.0
    quad_val = c_2i + x * x - v * v
    max_val = max(max_lin, quad_val)
    denom = (S_other + x) * (S_other + x)
    if denom < 1e-30:
        return 1e30
    return 2.0 * n * max_val / denom


def line_search_eval(a, direction, t, n):
    """Evaluate objective at step t along direction."""
    a_new = (1.0 - t) * a + t * direction
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 1e30
    a_new = a_new / s
    conv = compute_conv(a_new)
    return float(2.0 * n * np.max(conv))


def golden_line_search(a, direction, n, max_iter=45):
    """
    Golden section search for optimal t in [0,1] minimizing
    f(t) = 2n * max(conv((1-t)*a + t*direction, ...))
    Returns (t_star, f_star).
    """
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = line_search_eval(a, direction, c, n)
    fd = line_search_eval(a, direction, d, n)

    for _ in range(max_iter):
        if hi - lo < 1e-11:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = line_search_eval(a, direction, c, n)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = line_search_eval(a, direction, d, n)

    t_star = (lo + hi) / 2.0
    f_star = line_search_eval(a, direction, t_star, n)

    # Check a few additional points
    for t_check in [0.0, 0.01, 0.03, 0.1, 0.3, 0.5, 0.7, 0.9, 0.99, 1.0]:
        fc_check = line_search_eval(a, direction, t_check, n)
        if fc_check < f_star:
            f_star = fc_check
            t_star = t_check

    return t_star, f_star


def coordinate_descent_fast(a, time_budget=60.0, seed=42, priority=None):
    """
    Fast coordinate descent with optional priority ordering.
    Priority: array where smaller values = higher priority (updated first).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0

    if priority is not None:
        order_base = np.argsort(priority)[:n]
    else:
        order_base = None

    while time.monotonic() - start < time_budget:
        sweep += 1

        if sweep % 3 == 1:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    a = a / S_check

        if order_base is not None:
            order = order_base.copy()
            shuffle_start = int(n * 0.7)
            if shuffle_start < n:
                rng.shuffle(order[shuffle_start:])
        else:
            order = rng.permutation(n)

        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue

            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            # Quick candidates
            scale = max(v, S / n, 1e-6)
            candidates_arr = np.array([0.0, v * 0.5, v * 0.75, v, v * 1.25,
                                       v * 1.5, v * 2.0, v * 3.0,
                                       S / (2 * n), S / n, 2.0 * S / n])
            candidates_arr = np.unique(np.maximum(candidates_arr, 0.0))

            best_x = v
            best_f = f_1d_eval(v, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
            for x in candidates_arr:
                fx = f_1d_eval(x, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section refinement
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale
            if x_hi <= x_lo:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d_eval(c_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
            fd = f_1d_eval(d_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)

            for _ in range(35):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d_eval(c_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d_eval(d_pt, v, S_other, n, c_k_lin, a_ki_lin, c_2i)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) > 1e-14:
                delta = x_opt - v
                a[i] = x_opt

                if 0 <= k_quad <= 2 * n - 2:
                    conv[k_quad] += x_opt * x_opt - v * v
                if len(ks_lin) > 0:
                    conv[ks_lin] += 2.0 * delta * a_ki_lin

                S += delta

    return best_a


def kkt_subgradient_step(a, alpha, n):
    """
    One KKT-guided subgradient step. Returns updated a.
    Uses the exact subgradient of max(conv(a,a)) with uniform weights.
    """
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    argmax_set = np.where(conv >= max_conv * (1.0 - 1e-10))[0]

    v_sub = np.zeros(n)
    for k in argmax_set:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            v_sub[lo:hi + 1] += a[k - hi:k - lo + 1]
    v_sub /= max(len(argmax_set), 1)

    # Target: at KKT optimum, v_sub[i] should be constant for all i with a[i]>0
    # Gradient of max(conv) w.r.t. a is 2*v_sub (by chain rule)
    # We want to reduce v_sub where it's high and increase where it's low
    target = np.mean(v_sub[a > 1e-10]) if np.any(a > 1e-10) else np.mean(v_sub)
    grad = 2.0 * v_sub - target

    # Gradient descent step with projection
    a_new = a - alpha * grad
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        return a.copy()
    a_new = a_new / s_new
    return a_new


def enhanced_lp_iteration(a, time_budget=300.0, verbose=False, seed=42):
    """
    Main workhorse: iterate LP direction finding + golden section line search.
    Intersperses KKT subgradient steps and CD for exploitation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    no_improve_count = 0
    max_no_improve_lp = 8
    max_no_improve_kw = 15
    last_improve_obj = best_obj

    it = 0
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 5:
            break

        it += 1

        # Solve LP for direction
        lp_start = time.monotonic()
        g_lp, M_lp, binding = solve_lp_direction(a)
        lp_time = time.monotonic() - lp_start

        if g_lp is None:
            if verbose:
                print(f"    LP failed at iter {it}, t={elapsed:.1f}s")
            break

        # Line search
        t_star, f_star = golden_line_search(a, g_lp, n, max_iter=40)

        if f_star < last_improve_obj - 1e-11:
            a_new = (1.0 - t_star) * a + t_star * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
            a = a_new
            best_obj = f_star
            best_a = a.copy()
            last_improve_obj = best_obj
            no_improve_count = 0
            if verbose:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={t_star:.4f}, "
                      f"M_lp={M_lp:.6f}, lp_t={lp_time:.2f}s, elapsed={elapsed:.1f}s")
        else:
            no_improve_count += 1

            if no_improve_count >= max_no_improve_lp:
                # Try KKT subgradient steps
                if verbose:
                    print(f"    LP stalled at it{it}, trying KKT steps...")
                alpha = 1.0 / (n * 0.5)
                kkt_improved = False
                for kkt_it in range(50):
                    if time.monotonic() - start >= time_budget:
                        break
                    a_kkt = kkt_subgradient_step(a, alpha, n)
                    obj_kkt = 2.0 * n * float(np.max(compute_conv(a_kkt)))
                    if obj_kkt < best_obj - 1e-12:
                        best_obj = obj_kkt
                        best_a = a_kkt.copy()
                        last_improve_obj = best_obj
                        a = a_kkt.copy()
                        kkt_improved = True
                    alpha *= 0.95

                if kkt_improved:
                    no_improve_count = 0
                    if verbose:
                        print(f"    KKT improved: obj={best_obj:.8f}")

                if no_improve_count >= max_no_improve_kw:
                    # Try CD escape
                    if remaining > 30:
                        priority = None
                        if binding is not None and len(binding) > 0:
                            priority = np.zeros(n)
                            for k in binding:
                                lo = max(0, k - n + 1)
                                hi = min(n - 1, k)
                                if lo <= hi:
                                    priority[lo:hi + 1] += 1.0
                            priority = -priority
                        a_cd = coordinate_descent_fast(a, time_budget=min(40, remaining * 0.25),
                                                       seed=seed + it, priority=priority)
                        obj_cd = 2.0 * a_cd.size * float(np.max(compute_conv(a_cd / a_cd.sum())))
                        if obj_cd < last_improve_obj - 1e-12:
                            best_obj = obj_cd
                            best_a = a_cd.copy()
                            last_improve_obj = best_obj
                            a = a_cd.copy()
                            no_improve_count = 0
                            if verbose:
                                print(f"    CD escape: obj={best_obj:.8f}")
                        else:
                            # Add perturbation
                            pert = rng.normal(0, 0.02, n)
                            a_pert = a * np.exp(pert)
                            a_pert = np.maximum(a_pert, 0)
                            a_pert = a_pert / a_pert.sum()
                            a = a_pert
                            no_improve_count = 0

                    if no_improve_count >= 20:
                        if verbose:
                            print(f"    Giving up LP loop at it{it}")
                        break

    return best_a


def targeted_pair_reduction(a, time_budget=60.0, seed=42):
    """
    Targeted peak reduction: identify the peak convolution position and
    try paired coordinate updates to reduce it.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    for it in range(200):
        if time.monotonic() - start >= time_budget:
            break

        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))
        peak_val = float(conv[peak_idx])

        # Find contributing pairs for the peak
        j_min = max(0, peak_idx - n + 1)
        j_max = min(n - 1, peak_idx)

        # Try reducing individual coordinates that contribute to peak
        for j in range(j_min, j_max + 1):
            partner = peak_idx - j
            if 0 <= partner < n and a[j] > 1e-10:
                for factor in [0.7, 0.5, 0.3]:
                    a_try = a.copy()
                    a_try[j] *= factor
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        break

        # Also try small random perturbations
        if it % 10 == 5:
            pert = rng.normal(0, 0.01, n)
            a_pert = a * np.exp(pert)
            a_pert = np.maximum(a_pert, 0)
            s_pert = float(np.sum(a_pert))
            if s_pert > 1e-10:
                a_pert = a_pert / s_pert
                obj_pert = 2.0 * n * float(np.max(compute_conv(a_pert)))
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                    a = a_pert.copy()

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.arange(n)
        a = (np.exp(-((x - n * 0.25) ** 2) / (2 * (n * 0.1) ** 2)) +
             np.exp(-((x - n * 0.75) ** 2) / (2 * (n * 0.1) ** 2)))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
        a = a / a.sum()
    elif mode == 4:
        x = np.arange(n)
        freq = rng.uniform(2, 8)
        a = 1.0 + 0.4 * np.sin(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0.1)
        a = a / a.sum()
    elif mode == 5:
        x = np.arange(n)
        a = 1.0
        for _ in range(4):
            c = rng.integers(0, n)
            w = rng.integers(max(1, n // 10), max(2, n // 3))
            a += np.exp(-((x - c) ** 2) / (2.0 * w * w))
        a = a / a.sum()
    elif mode == 6:
        a = np.ones(n)
        a[::2] = rng.uniform(0.3, 0.7, size=(n + 1) // 2)
        a[1::2] = rng.uniform(1.0, 2.0, size=n // 2)
        a = a / a.sum()
    elif mode == 7:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (1.0 + x * x)
        a = a / a.sum()
    elif mode == 8:
        # Geometric spacing of peaks
        x = np.arange(n)
        a = 1.0
        ratio = rng.uniform(1.3, 2.0)
        pos = 0
        amp = 1.0
        while pos < n:
            a[max(0, pos - 2):min(n, pos + 3)] += amp * np.exp(-((x[max(0, pos-2):min(n, pos+3)] - pos) ** 2) / 2.0)
            pos = int(pos * ratio)
            amp *= 0.6
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Enhanced LP-CD hybrid with KKT polishing, targeted peak
    reduction, multi-n exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 12.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = 2.0 * parent.size * float(np.max(compute_conv(parent)))
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(800) / 800.0
        best_score = 2.0
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Enhanced LP Iteration (main workhorse, 35% budget) ===
    print(f"\n=== Phase 1: Enhanced LP Iteration (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 500.0)
    if p1_budget > 60:
        t0 = time.monotonic()
        a_lp1 = enhanced_lp_iteration(best_seq, time_budget=p1_budget,
                                      verbose=True, seed=seed)
        obj_lp1 = 2.0 * a_lp1.size * float(np.max(compute_conv(a_lp1 / a_lp1.sum())))
        if obj_lp1 < best_score:
            best_score = obj_lp1
            best_seq = a_lp1.copy()
            print(f"  ** LP Phase 1: obj={best_score:.8f} (t={time.monotonic()-t0:.1f}s)")
        else:
            print(f"  LP Phase 1: obj={obj_lp1:.8f} (no improvement)")

    # === Phase 2: CD Polish (10% budget) ===
    print(f"\n=== Phase 2: CD Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.10, 120.0)
    if p2_budget > 20:
        a_cd = coordinate_descent_fast(best_seq, time_budget=p2_budget, seed=seed + 100)
        obj_cd = 2.0 * a_cd.size * float(np.max(compute_conv(a_cd / a_cd.sum())))
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  ** CD Phase 2: obj={best_score:.8f}")
        else:
            print(f"  CD Phase 2: obj={obj_cd:.8f}")

    # === Phase 3: Second LP Round (20% budget) ===
    print(f"\n=== Phase 3: Second LP Round ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.20, 280.0)
    if p3_budget > 30:
        t0 = time.monotonic()
        a_lp2 = enhanced_lp_iteration(best_seq, time_budget=p3_budget,
                                      verbose=True, seed=seed + 200)
        obj_lp2 = 2.0 * a_lp2.size * float(np.max(compute_conv(a_lp2 / a_lp2.sum())))
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  ** LP Phase 3: obj={best_score:.8f} (t={time.monotonic()-t0:.1f}s)")
        else:
            print(f"  LP Phase 3: obj={obj_lp2:.8f}")

    # === Phase 4: Targeted Peak Reduction (5% budget) ===
    print(f"\n=== Phase 4: Targeted Peak Reduction ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.05, 60.0)
    if p4_budget > 15:
        a_tpr = targeted_pair_reduction(best_seq, time_budget=p4_budget, seed=seed + 300)
        obj_tpr = 2.0 * a_tpr.size * float(np.max(compute_conv(a_tpr / a_tpr.sum())))
        if obj_tpr < best_score:
            best_score = obj_tpr
            best_seq = a_tpr.copy()
            print(f"  ** TPR Phase 4: obj={best_score:.8f}")
        else:
            print(f"  TPR Phase 4: obj={obj_tpr:.8f}")

    # === Phase 5: Multi-n Exploration (20% budget) ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_candidates = sorted(set([
        max(100, n_current - 300), max(100, n_current - 150),
        n_current + 150, n_current + 300, n_current + 500,
        n_current + 800, n_current + 1200,
        300, 400, 500, 600, 700, 900, 1100, 1400, 1700, 2000, 2500, 3000
    ]))
    n_filtered = []
    for nc in n_candidates:
        if 200 <= nc <= 3500:
            if all(abs(nc - nf) > 100 for nf in n_filtered):
                n_filtered.append(nc)
    n_filtered = n_filtered[:7]
    print(f"  n candidates: {n_filtered}")

    for n_try in n_filtered:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Optimize: CD then LP
        alloc = min(remaining * 0.18, 90.0)
        if alloc > 10:
            a_try = coordinate_descent_fast(a_try, time_budget=alloc * 0.4,
                                           seed=n_try)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_try = enhanced_lp_iteration(a_try,
                                              time_budget=min(alloc * 0.6, remaining * 0.4),
                                              verbose=False, seed=n_try + 500)

        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = 2.0 * a_try.size * float(np.max(compute_conv(a_try)))
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts (10% budget) ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        mode = 1 + r
        a_r = diverse_init(n_cur, rng, mode=mode)
        blend = 0.35 + 0.1 * r
        a_r = (1.0 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r < 1e-10:
            continue
        a_r = a_r / s_r

        alloc = min(remaining * 0.25, 100.0)
        if alloc > 15:
            a_r = coordinate_descent_fast(a_r, time_budget=alloc * 0.35,
                                         seed=3000 + r * 100)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_r = enhanced_lp_iteration(a_r,
                                           time_budget=min(alloc * 0.65, remaining * 0.5),
                                           verbose=False, seed=3500 + r * 100)

        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r < 1e-10:
            continue
        a_r = a_r / s_r
        obj_r = 2.0 * a_r.size * float(np.max(compute_conv(a_r)))
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  ** Restart {r} (mode {mode}): obj={best_score:.8f}")
        else:
            print(f"  Restart {r} (mode {mode}): obj={obj_r:.8f}")

    # === Phase 7: Final Polish (remaining) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        # Final LP
        lp_alloc = min(remaining * 0.55, 120.0)
        if lp_alloc > 15:
            a_f = enhanced_lp_iteration(best_seq, time_budget=lp_alloc,
                                        verbose=True, seed=5000)
            obj_f = 2.0 * a_f.size * float(np.max(compute_conv(a_f / a_f.sum())))
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  ** Final LP: obj={best_score:.8f}")

        # Final CD
        remaining = deadline - time.monotonic()
        if remaining > 10:
            cd_alloc = min(remaining * 0.8, 60.0)
            if cd_alloc > 8:
                a_f2 = coordinate_descent_fast(best_seq, time_budget=cd_alloc,
                                               seed=6000)
                obj_f2 = 2.0 * a_f2.size * float(np.max(compute_conv(a_f2 / a_f2.sum())))
                if obj_f2 < best_score:
                    best_score = obj_f2
                    best_seq = a_f2.copy()
                    print(f"  ** Final CD: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = 2.0

    best_seq = np.maximum(best_seq, 0.0)
    result = [float(min(1000.0, x)) for x in best_seq]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
        print(f"  Evaluation confirms: {best_score:.8f}")
    elif val == np.inf:
        print("  WARNING: evaluation returned inf, using fallback")
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
