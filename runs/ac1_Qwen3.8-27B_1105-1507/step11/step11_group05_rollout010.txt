def spectral_flatten(a, n_fft):
    """One step of spectral flattening."""
    A = np.fft.fft(a, n_fft)
    mag = np.abs(A)
    phase = np.angle(A)
    
    # Target: flat magnitude
    # But we need to preserve the DC component (sum of a = 1)
    # A[0] = sum(a) = 1, so |A[0]| = 1
    # For other frequencies, we want |A[k]| to be small and equal
    
    # Simple approach: geometric mean of target
    target_mag = np.ones(n_fft)
    target_mag[0] = mag[0]  # Preserve DC
    # For real signal, A[n_fft-k] = conj(A[k])
    
    # Blend: A_new = (1-alpha)*A + alpha * target (in magnitude)
    new_mag = 0.7 * mag + 0.3 * target_mag
    new_mag[0] = mag[0]
    
    A_new = new_mag * np.exp(1j * phase)
    # Ensure conjugate symmetry
    A_new[n_fft//2:] = np.conj(A_new[n_fft//2:-1][::-1])  # approximate
    A_new_real = np.fft.irfft(A_new[:n_fft//2+1])[:len(a)]
    
    return np.maximum(A_new_real, 0)

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_flatten(a, n_iters=50, alpha=0.3):
    """Iteratively flatten the power spectrum to reduce correlation peaks."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    for it in range(n_iters):
        # FFT of a (zero-padded to size)
        A = np.fft.fft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: geometric mean magnitude (flat spectrum)
        # Preserve DC component
        mag_nonzero = mag[mag > 1e-10]
        target_mag = np.full(size, np.sqrt(mag[0] * (mag_nonzero.mean() + 1e-10)))
        target_mag[0] = mag[0]
        
        # Blend
        new_mag = (1 - alpha) * mag + alpha * target_mag
        new_mag[0] = mag[0]
        
        # Reconstruct with conjugate symmetry
        A_new = new_mag * np.exp(1j * phase)
        # For real signal: A[k] = conj(A[size-k])
        for k in range(1, size // 2 + 1):
            if k == size - k:
                continue
            # Average with conjugate
            conj_val = np.conj(A_new[size - k])
            A_new[k] = (A_new[k] + conj_val) / 2
            A_new[size - k] = np.conj(A_new[k])
        
        a_new = np.real(np.fft.ifft(A_new, size))[:n]
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a = a_new / s_new
    
    return a

def exp_gradient_descent(a, time_budget, eta_init=0.5, eta_min=0.01, verbose=False):
    """Exponentiated gradient descent (multiplicative weight updates)."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    eta = eta_init
    patience = 200
    best_it = 0
    
    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        cmax = np.max(conv)
        
        # Use log-sum-exp for smooth gradient
        T = 30.0
        exp_vals = np.exp(T * (conv - cmax))
        w = exp_vals / exp_vals.sum()
        
        # Gradient: grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
        grad = np.zeros(n)
        for i in range(n):
            # a[k-i] for valid k: k from i to i+n-1, so a[k-i] = a[0..n-1]
            lo = max(0, i)
            hi = min(2*n-2, i+n-1)
            if lo <= hi:
                grad[i] += np.dot(w[lo:hi+1], a[lo-i:hi-i+1])
            # a[i-k] for valid k: k from max(0,i-n+1) to min(i, 2n-2)
            lo2 = max(0, i-n+1)
            hi2 = min(2*n-2, i)
            if lo2 <= hi2:
                grad[i] += np.dot(w[lo2:hi2+1], a[i-hi2:i-lo2+1][::-1])
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad / cmax)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        else:
            eta *= 0.8
        
        # Slowly decrease eta
        eta = max(eta_min, eta_init * 0.995 ** it)
        
        it += 1
        if it - best_it > patience:
            a = best_a.copy()
            eta = max(eta_min, eta * 0.5)
            patience = max(50, patience // 2)
            best_it = it
        
        if verbose and it % 100 == 0:
            print(f"      EGD it{it}: obj={best_obj:.8f}, eta={eta:.4f}, t={time.monotonic()-start:.1f}s")
    
    return best_a

The last condition is interesting. By Cauchy-Schwarz:
sum_j a[j]*a[n-1-j] ≤ sqrt(sum a[j]^2 * sum a[n-1-j]^2) = sum a[j]^2

And sum a[j]^2 ≥ 1/n. So sum a[j]*a[n-1-j] ≤ sum a[j]^2, which is always true.

But we need sum a[j]*a[n-1-j] < 1/n. This requires the sequence to be "anti-correlated" with its reverse, i.e., a[j] should be large where a[n-1-j] is small.

This is the key! The sequence should be asymmetric: if a[j] is large, then a[n-1-j] should be small.

For example: a[j] ∝ (j+1)^alpha for some alpha > 0. Then a[j] increases with j, and a[n-1-j] decreases with j. Their product is maximized in the middle.

Let me check: a[j] = (j+1)^alpha / sum_k (k+1)^alpha for j=0,...,n-1.
- a[0] = 1/sum, a[n-1] = n^alpha/sum
- For large n and alpha=1: sum ≈ n^2/2, so a[0] ≈ 2/n^2, a[n-1] ≈ 2/n
- conv[n-1] = sum_j (j+1)(n-j) / sum^2
  = (1/sigma^2) * sum_j (j+1)(n-j) where sigma = sum_k (k+1) = n(n+1)/2
  = (1/sigma^2) * [n*sum(j+1) - sum(j(j+1))]
  = (1/sigma^2) * [n * n(n+1)/2 - n(n+1)(2n+1)/6]
  = (1/sigma^2) * [n^2(n+1)/2 - n(n+1)(2n+1)/6]
  = n(n+1)/sigma^2 * [n/2 - (2n+1)/6]
  = n(n+1)/sigma^2 * [3n - 2n - 1]/6
  = n(n+1)/(6*sigma^2) * (n-1)
  = n(n+1)(n-1) / (6 * n^2(n+1)^2/4)
  = 4(n-1) / (6n(n+1))
  ≈ 2/(3n) for large n

So conv[n-1] ≈ 2/(3n) < 1/n. 

And the edge values:
- conv[0] = a[0]^2 = 1/sigma^2 ≈ 4/n^4 (very small)
- conv[2n-2] = a[n-1]^2 = n^{2*alpha}/sigma^2 = n^2/(n^2(n+1)^2/4) = 4/(n+1)^2 ≈ 4/n^2

So max is at the center: ≈ 2/(3n). Objective ≈ 2n * 2/(3n) = 4/3 ≈ 1.33.
