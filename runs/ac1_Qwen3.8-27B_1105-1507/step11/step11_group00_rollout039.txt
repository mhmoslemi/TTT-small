```python
import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.ndimage import uniform_filter1d
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def irls_qp_step(a, p_power=1.0, eps=1e-8, max_iter=30, lam=0.0):
    """
    IRLS step: iteratively solve QP to flatten autocorrelation.
    p_power: how aggressively to weight (higher = more aggressive)
    lam: regularization to ensure PSD (0 = no regularization)
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    for _ in range(max_iter):
        conv = compute_conv(a)
        # Weights: inversely proportional to autocorrelation values
        c = 1.0 / (np.maximum(conv, eps) ** p_power)
        # Normalize c so that the QP isn't too extreme
        c = c / float(np.sum(c))
        
        # Build Hankel matrix C[j,l] = c[j+l]
        idx = np.add.outer(np.arange(n), np.arange(n))
        C = c[idx]
        
        # Add regularization for PSD
        if lam > 0:
            C = C + lam * np.eye(n)
        
        # Make sure C is symmetric (it should be by construction, but numerical errors)
        C = (C + C.T) / 2
        
        # Solve QP: minimize a^T C a s.t. sum(a) = 1, a >= 0
        a_var = cp.Variable(n)
        objective = cp.Minimize(cp.quad_form(a_var, cp.psd_wrap(C)))
        constraints = [cp.sum(a_var) == 1, a_var >= 0]
        prob = cp.Problem(objective, constraints)
        
        solved = False
        for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if a_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    a_new = np.maximum(a_var.value, 0)
                    s_new = float(np.sum(a_new))
                    if s_new > 1e-10:
                        a_new = a_new / s_new
                        # Check if this is better
                        if compute_obj(a_new) <= compute_obj(a) + 1e-10:
                            a = a_new
                        else:
                            # Damped update
                            a = 0.5 * a + 0.5 * a_new
                            a = np.maximum(a, 0)
                            s = float(np.sum(a))
                            if s > 1e-10:
                                a = a / s
                        solved = True
                        break
            except Exception:
                continue
        
        if not solved:
            break
    
    return a
```

```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses IRLS-weighted gradient descent (new), adaptive-momentum Nesterov GD,
pairwise local search, and multi-n exploration with diverse restarts.
Key innovation: IRLS gradient that reweights subgradients by inverse
autocorrelation, providing smoother descent than max-only gradients.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def gradient_from_p(a, p):
    """
    Compute 2 * sum_j a[j] * p[i+j] for i=0..n-1.
    p has length 2n-1 (distribution over convolution lags).
    Uses FFT: g[i] = (b * p)[i+n-1] where b = a[::-1].
    """
    n = len(a)
    b = a[::-1]
    N_fft = 1
    while N_fft < 2 * n - 1 + len(p) - 1:
        N_fft <<= 1
    c = irfft(rfft(b, N_fft) * rfft(p, N_fft), N_fft)
    return 2.0 * c[n - 1:2 * n - 1]


def irls_gradient(a, conv, p_power=1.0, eps=1e-10):
    """
    IRLS-weighted gradient: weights each lag inversely to its autocorrelation.
    This provides a smoother descent direction than using only the max lag.
    """
    n = len(a)
    c = 1.0 / (np.maximum(conv, eps) ** p_power)
    c_sum = float(np.sum(c))
    if c_sum < 1e-300:
        # Fallback: uniform weight
        p = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        p = c / c_sum
    return gradient_from_p(a, p)


def smoothed_gradient(a, conv, tau):
    """Compute gradient of LSE(tau, conv(a,a)) with respect to a."""
    n = len(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    if shifted.max() > 500:
        shifted = shifted - shifted.max()
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(2 * n - 1)
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    return gradient_from_p(a, p)


def topk_subgradient(a, conv, k=3):
    """Subgradient using only the top-k peaks."""
    n = len(a)
    idx = np.argsort(conv)[-k:]
    p = np.zeros(2 * n - 1)
    p[idx] = 1.0 / k
    return gradient_from_p(a, p)


def mirror_descent_step(a, grad, eta):
    """Multiplicative weights update."""
    exponent = np.clip(-eta * grad, -50, 50)
    a_new = a * np.exp(exponent)
    s = float(np.sum(a_new))
    if s < 1e-300:
        return a.copy()
    return a_new / s


def lp_direction(a):
    """
    Solve LP: maximize sum(g) s.t. conv(a,g) <= max(conv(a,a)), g >= 0.
    Returns g normalized to sum 1, or None.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv_aa = compute_conv(a_norm)
    M = float(np.max(conv_aa))

    A = np.zeros((2 * n - 1, n))
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A[valid] = a_norm[diff[valid]]

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def nesterov_optimize(a, time_budget=100.0, tau_init=30.0, tau_end=2000.0,
                        use_irls=True, p_power=0.5, verbose=False):
    """
    Nesterov accelerated gradient with IRLS gradient and adaptive learning rate.
    """

    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)

    beta = 0.75
    base_lr = 3.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.3:
            break

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.3

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        conv_y = compute_conv(y)

        # Choose gradient: blend IRLS with smoothed
        if use_irls:
            g_irls = irls_gradient(y, conv_y, p_power=p_power)
            g_smooth = smoothed_gradient(y, conv_y, tau)
            # Blend: more IRLS early, more smoothed later

            g = alpha_irls * g_irls + (1.0 - alpha_irls) * g_smooth
        else:
            g = smoothed_gradient(y, conv_y, tau)

        g_norm = float(np.linalg.norm(g))

        # Adaptive LR with Barzilai-Borwein
        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm + 1e-10)
            adapt = min(max(adapt, 0.2 * base_lr), 3.0 * base_lr)
            lr = 0.85 * lr + 0.15 * adapt

        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-8), 5.0 / np.sqrt(n))
                lr = 0.75 * lr + 0.25 * bb

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.3
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall_count = 0
        else:
            stall_count += 1

        # Backtracking if stalling
        if stall_count > 50:
            # Try halving
            a_half = y - 0.3 * lr * g
            a_half = np.maximum(a_half, 0)
            s_h = float(np.sum(a_half))
            if s_h > 1e-10:
                a_half = a_half / s_h
                obj_h = compute_obj(a_half)
                if obj_h < best_obj:
                    best_obj = obj_h
                    best_a = a_half.copy()
                    a = a_half
                    v = np.zeros(n)
                    stall_count = 0
                else:
                    lr *= 0.5
                    v = np.zeros(n)
                    stall_count = 0
            else:
                a = best_a.copy()
                v = np.zeros(n)
                stall_count = 0

        # Periodic perturbation to escape local minima
        if it % 100 == 80 and progress < 0.85:
            temp = 0.008 * (1.0 - progress)
            a_pert = a * np.exp(temp * rng.standard_normal(n))
            a_pert = np.maximum(a_pert, 0)
            s_p = float(np.sum(a_pert))
            if s_p > 1e-10:
                a_pert = a_pert / s_p
                obj_p = compute_obj(a_pert)
                if obj_p < best_obj:
                    best_obj = obj_p
                    best_a = a_pert.copy()
                a = a_pert
            v = np.zeros(n)

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-12, lr * 0.9998)

        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.3
            best_it = it

        if verbose and (it % 200 == 0 or it == 1):

    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(456)

    conv = compute_conv(a)
    current_max = float(np.max(conv))

    # Track top lags for quick evaluation
    top_lags = set(np.argsort(conv)[-5:].tolist())

    max_delta = 0.02  # max fraction of mass to transfer
    min_delta = 0.0005
    delta = max_delta
    no_improve = 0
    total_moves = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        # Pick random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Determine direction: transfer from larger to smaller
        if a[i] < a[j]:
            i, j = j, i
        # Transfer delta * a[i] from i to j
        d = delta * a[i]
        if d < min_delta * float(np.sum(a)) or d >= a[i]:
            d = 0.5 * a[i]
            if d < 1e-10:
                continue

        # Compute new conv efficiently
        # conv_new[k] = conv[k] + 2*d*(a[k-j]-a[k-i]) + d^2*(indicator terms)
        # For quick check, just compute conv at top lags
        change = 2.0 * d * (np.roll(a, j) - np.roll(a, i))  # approx (circular)
        # Actually let me just compute the full conv (it's fast with FFT)
        a_trial = a.copy()
        a_trial[i] -= d
        a_trial[j] += d
        conv_trial = compute_conv(a_trial)
        max_trial = float(np.max(conv_trial))

        if max_trial < current_max * (1.0 - 1e-8):
            # Accept
            a = a_trial
            conv = conv_trial
            current_max = max_trial
            obj_new = 2.0 * n * current_max
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a.copy()
            no_improve = 0
            delta = min(delta * 1.1, max_delta)
        else:
            no_improve += 1
            if no_improve > 100:
                delta *= 0.7
                no_improve = 0
                if delta < min_delta:
                    break

        total_moves += 1

        # Update top lags periodically
        if total_moves % 200 == 0:
            top_lags = set(np.argsort(conv)[-5:].tolist())

        if verbose and total_moves % 5000 == 0:
            print(f"      LS moves={total_moves}: obj={best_obj:.8f}, "
                  f"delta={delta:.5f}, t={time.monotonic()-start:.0f}s")

    return best_a


def lp_driven_optimize(a, time_budget=150.0, verbose=False):
    """LP-driven optimization with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        g = lp_direction(a)
        it += 1

        if g is None:
            conv = compute_conv(a)
            g = irls_gradient(a, conv, p_power=1.0)
            g = np.maximum(g, 0)
            s_g = float(np.sum(g))
            if s_g < 1e-10:
                break
            g = g / s_g

        # Line search
        best_t = 0.0
        best_obj_t = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            a_try = np.maximum((1 - t) * a + t * g, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                obj_try = compute_obj(a_try / s_try)
                if obj_try < best_obj_t:
                    best_obj_t = obj_try
                    best_t = t

        if best_t > 0 and best_obj_t < best_obj - 1e-12:
            a = np.maximum((1 - best_t) * a + best_t * g, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            best_obj = best_obj_t
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                # Perturb
                conv_p = compute_conv(a)
                g_p = irls_gradient(a, conv_p, p_power=1.0)
                lr_p = 0.05 / (float(np.linalg.norm(g_p)) + 1e-10)
                a = np.maximum(a - lr_p * g_p + 0.005 * rng.standard_normal(n), 0)
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new
                else:
                    a = best_a.copy()
                no_improve = 0
                if it > 100:
                    break

        if verbose and it % 5 == 0:
            print(f"      LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}, "
                  f"time={time.monotonic()-start:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: IRLS-Nesterov GD + Mirror Descent + Local Search + LP + multi-n.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Initialize from parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: IRLS Nesterov GD (primary) ===
    print(f"\n=== Phase 1: IRLS Nesterov GD (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 350.0)
    if p1_budget > 30:
        a1 = nesterov_optimize(best_seq, time_budget=p1_budget,
                               tau_init=30.0, tau_end=3000.0,
                               use_irls=True, p_power=0.5, verbose=True)
        obj1 = compute_obj(a1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj1:.8f}")

    # === Phase 2: Mirror Descent ===
    print(f"\n=== Phase 2: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 250.0)
    if p2_budget > 20:
        a2 = mirror_descent_optimize(best_seq, time_budget=p2_budget,
                                     tau_init=50.0, tau_end=3000.0,
                                     p_power=0.7, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2: obj={best_score:.8f} **")
        else:
            print(f"  Phase2: obj={obj2:.8f}")

    # === Phase 3: LP-Driven ===
    print(f"\n=== Phase 3: LP-Driven ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 180.0)
    if p3_budget > 15:
        a3 = lp_driven_optimize(best_seq, time_budget=p3_budget, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj3:.8f}")

    # === Phase 4: Local Search ===
    print(f"\n=== Phase 4: Pairwise Local Search ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 200.0)
    if p4_budget > 20:
        a4 = pairwise_local_search(best_seq, time_budget=p4_budget, verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4: obj={best_score:.8f} **")
        else:
            print(f"  Phase4: obj={obj4:.8f}")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = []
    for delta in [-200, -100, 100, 200, 400, 600, 1000]:
        n_try = n_best + delta
        if 400 <= n_try <= 5000:
            n_options.append(n_try)
    for n_abs in [600, 1000, 1500, 2000, 3000]:
        if n_abs not in n_options:
            n_options.append(n_abs)
    n_options = sorted(set(n_options))
    print(f"  Candidate n values: {n_options}")

    # Try top 4 n values
    n_to_try = n_options[:4]
    for n_try in n_to_try:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Nesterov GD
        alloc = min(remaining * 0.4, 100.0)
        if alloc > 20:
            a_try = nesterov_optimize(a_try, time_budget=alloc,
                                      tau_init=50.0, tau_end=2000.0,
                                      use_irls=True, p_power=0.5, verbose=False)

        # Quick local search
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.2, 50.0)
        if alloc > 10:
            a_try = pairwise_local_search(a_try, time_budget=alloc, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    restart_types = ["mirror_sym", "fourier_perturb", "gaussian_smooth", "bimodal"]
    for r, name in enumerate(restart_types):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        if name == "mirror_sym":
            a_r = (best_seq + best_seq[::-1]) / 2.0
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r
        elif name == "fourier_perturb":
            pert = np.zeros(n_cur)
            for k in range(1, 8):
                freq = k * 2.0 * np.pi / n_cur
                amp = 0.05 * rng.uniform(0.3, 1.5) / k
                phase = rng.uniform(0, 2.0 * np.pi)
                pert += amp * np.cos(freq * np.arange(n_cur) + phase)
            a_r = best_seq * np.exp(pert)
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r
        elif name == "gaussian_smooth":
            sigma = max(3, n_cur // 40)
            kernel = np.exp(-0.5 * (np.arange(-sigma*3, sigma*3+1)/sigma)**2)
            kernel /= kernel.sum()
            a_smooth = np.convolve(best_seq, kernel, mode='same')
            a_r = 0.6 * best_seq + 0.4 * a_smooth
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r
        else:  # bimodal
            center = n_cur // 2
            width = n_cur // 4
            bump1 = np.exp(-0.5 * ((np.arange(n_cur) - center*0.3) / width)**2)
            bump2 = np.exp(-0.5 * ((np.arange(n_cur) - center*1.7) / width)**2)
            a_bimodal = 0.5 * bump1 + 0.5 * bump2
            a_bimodal /= a_bimodal.sum()
            a_r = 0.7 * best_seq + 0.3 * a_bimodal
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r

        # Optimize
        alloc = min(remaining * 0.35, 80.0)
        if alloc > 15:
            a_r = nesterov_optimize(a_r, time_budget=alloc,
                                    tau_init=50.0, tau_end=1500.0,
                                    use_irls=True, p_power=0.5, verbose=False)
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.15, 40.0)
        if alloc > 10:
            a_r = pairwise_local_search(a_r, time_budget=alloc, verbose=False)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart ({name}): obj={best_score:.8f} **")
        else:
            print(f"  Restart ({name}): obj={obj_r:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau Nesterov
    if remaining > 40:
        alloc = min(remaining * 0.35, 300.0)
        a_f = nesterov_optimize(best_seq, time_budget=alloc,
                                tau_init=500.0, tau_end=10000.0,
                                use_irls=True, p_power=0.3, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final Nesterov: obj={best_score:.8f} **")

    # Final local search
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 200.0)
        a_f2 = pairwise_local_search(best_seq, time_budget=alloc, verbose=True)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final LS: obj={best_score:.8f} **")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.3, 100.0)
        a_f3 = lp_driven_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final tiny Nesterov
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 80.0)
        a_f4 = nesterov_optimize(best_seq, time_budget=alloc,
                                 tau_init=2000.0, tau_end=20000.0,
                                 use_irls=True, p_power=0.2, verbose=False)
        obj_f4 = compute_obj(a_f4)
        if obj_f4 < best_score:
            best_score = obj_f4
            best_seq = a_f4.copy()
            print(f"  Final Nesterov2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
