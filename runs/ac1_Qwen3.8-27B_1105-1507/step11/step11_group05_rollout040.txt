"""
Optimization of step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements:
1. Flat-convolution (renewal) sequence initialization
2. Fast custom multi-peak projected subgradient (no cvxpy overhead)
3. Newton refinement on active-set KKT system
4. Systematic n exploration

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))

def flat_conv_init(n):
    """Generate the flat-convolution (renewal) sequence of length n."""
    a = np.ones(n)
    for k in range(1, n):
        a[k] = a[k-1] * (2*k - 1) / (2*k)
    s = a.sum()
    return a / s

def power_init(n, alpha):
    """a[k] = k^(-alpha) (with a[0]=1), normalized."""
    k = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    a[1:] = np.power(k[1:], -alpha)
    s = a.sum()
    return a / s

def custom_init(n, decay):
    """a[k] = exp(-decay * k / n), normalized."""
    k = np.arange(n, dtype=np.float64)
    a = np.exp(-decay * k / n)
    s = a.sum()
    return a / s

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    return proj / proj.sum()
