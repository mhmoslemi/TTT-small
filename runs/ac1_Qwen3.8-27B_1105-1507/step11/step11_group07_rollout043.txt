Combines fast FFT-accelerated gradients with precise LP and subgradient methods.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def softmax_gradient_fft(a, p, gamma=2.0):
    """Compute gradient of (1/gamma)*log(sum exp(gamma*conv)) w.r.t. a, using FFT."""
    n = len(a)
    p_len = len(p)
    # grad[i] ~ 2 * sum_k p[k] * a[k-i] (cross-correlation)
    # This is: (a_rev * p)[n-1 : n-1+n]
    a_rev = a[::-1]
    size = 1
    while size < n + p_len - 1:
        size <<= 1
    A = rfft(a_rev, size)
    P = rfft(p, size)
    full = irfft(A * P, size)
    q = full[n - 1: n - 1 + n]
    return gamma * 2.0 * q


def adam_softmax_gd(a, time_budget=200.0, beta_start=10.0, beta_end=150.0,
                     lr_start=0.01, lr_end=0.0001, beta1=0.9, beta2=0.999,
                     verbose=True, seed=42):
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Exponential beta annealing
        beta = beta_start * (beta_end / beta_start) ** progress
        # Cosine learning rate
        lr = lr_end + 0.5 * (lr_start - lr_end) * (1 + np.cos(np.pi * progress))
        
        # Compute convolution and softmax
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Compute gradient
        grad = softmax_gradient_fft(a, p, gamma=beta)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = t
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Reset and perturb
                a = project_simplex(best_a + rng.normal(0, 0.003, n))
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                stall = 0
        
        a = a_new
        
        if verbose and t % 300 == 0:
            print(f"    Adam it{t}: obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.6f}, t={elapsed:.1f}s")
        
        if t - best_it > 8000 and t > 2000:
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = best_it
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j: j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g_lp, time_budget=3.0):
    """Find optimal blend factor t for (1-t)*a + t*g_lp."""
    n = len(a)
    
    def obj_at(t):
        if t < 0:
            t = 0.0
        a_try = (1 - t) * a + t * g_lp
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)
    
    # Dense grid search first
    best_t = 0.0
    best_val = obj_at(0.0)
    for t_try in np.linspace(0, 1.0, 21):
        val = obj_at(t_try)
        if val < best_val:
            best_val = val
            best_t = t_try
    
    # Refine with golden section around best_t
    if best_t > 0 and best_t < 1:
        lo = max(0, best_t - 0.05)
        hi = min(1.2, best_t + 0.05)
        phi = (1 + np.sqrt(5)) / 2
        gr = 2 - phi
        c = lo + gr * (hi - lo)
        d = lo + (1 - gr) * (hi - lo)
        fc = obj_at(c)
        fd = obj_at(d)
        start = time.monotonic()
        for _ in range(15):
            if time.monotonic() - start > time_budget:
                break
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = lo + gr * (hi - lo)
                fc = obj_at(c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
                fd = obj_at(d)
        t_cand = c if fc < fd else d
        v_cand = min(fc, fd)
        if v_cand < best_val:
            best_val = v_cand
            best_t = t_cand
    
    return best_t, best_val


def lp_polish(a, time_budget=120.0, max_iters=50, verbose=False):
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        t_opt, obj_opt = line_search_blend(a, g_lp, time_budget=min(2.0, remaining * 0.03))
        
        if t_opt > 1e-6 and obj_opt < best_obj - 1e-10:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = project_simplex(np.maximum(a_new, 0))
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        it += 1
        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, elapsed={time.monotonic()-start:.1f}s")
        
        if stall >= 5:
            noise = rng.normal(0, 0.005, n)
            a = project_simplex(best_a + noise)
            stall = 0
    
    return best_a, best_obj


def subgradient_polish(a, time_budget=60.0, top_k=5, verbose=False):
    """Polish using exact subgradient of max(conv) with top-K peaks."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.01
    stall = 0
    rng = np.random.default_rng(999)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        
        # Find top-K peak indices
        top_idx = np.argsort(conv)[::-1][:top_k]
        weights = np.zeros(len(conv))
        for idx in top_idx:
            weights[idx] = 1.0 / top_k
        
        # Compute subgradient
        # grad[i] = sum_k weights[k] * dc[k]/da[i]
        # dc[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n and k-i!=i-k)
        # Simplified: grad[i] ≈ 2 * sum_k weights[k] * a[k-i] (for k-i in range)
        # Use FFT: this is 2 * cross_correlation(weights, a)
        # Actually: sum_k weights[k] * a[k-i] is the cross-correlation at lag i
        # = (a_rev * weights)[n-1 : n-1+n]
        a_rev = a[::-1]
        size = 1
        while size < n + len(weights) - 1:
            size <<= 1
        A = rfft(a_rev, size)
        W = rfft(weights, size)
        full = irfft(A * W, size)
        grad = 2.0 * full[n - 1: n - 1 + n]
        
        # Line search
        for _ in range(20):
            a_new = project_simplex(a - lr * grad)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                lr = min(lr * 1.1, 0.1)
                stall = 0
                break
            else:
                lr *= 0.7
        else:
            stall += 1
            if stall > 10:
                lr = 0.01
                a = project_simplex(best_a + rng.normal(0, 0.002, n))
                stall = 0
        
        if verbose:
            print(f"    SubGrad: obj={best_obj:.8f}, lr={lr:.4f}, t={time.monotonic()-start:.1f}s")
    
    return best_a, best_obj


def coordinate_descent(a, time_budget=30.0, verbose=False):
    """Coordinate descent: update one coordinate at a time."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(555)
    
    while time.monotonic() - start < time_budget:
        # Try a random set of coordinates
        n_try = max(10, n // 10)
        coords = rng.choice(n, size=n_try, replace=False)
        
        improved = False
        for i in coords:
            # Try to optimize a[i] while keeping others fixed
            # After changing a[i], we need to re-normalize
            # Simpler: try a few values for a[i]
            old_val = a[i]
            for factor in [0.5, 0.7, 1.3, 2.0, 0.3, 3.0]:
                a[i] = old_val * factor
                a = np.maximum(a, 0)
                a = project_simplex(a)
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break
            if not improved:
                a[i] = old_val
                a = project_simplex(a)
        
        if verbose and time.monotonic() - start < time_budget:
            pass
    
    return best_a, best_obj


def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = (np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) +
             np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01)
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 6:
        a = 1.0 / np.cosh(3.0 * (2 * x - 1))
    elif mode == 7:
        a = np.zeros(n)
        a[0] = 1.0
        a[-1] = 1.0
        a[1:min(3, n)] = 0.3
        a[max(0, n - 4): n - 1] = 0.3
        if n > 4:
            a[1: -1] += 0.05
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

Wait, I need to double-check a few things:

1. The `compute_obj` function: it normalizes to sum=1, then computes 2*n*max(conv). This matches the evaluation formula.

2. The gradient computation: I'm using `softmax_gradient_fft` which computes `gamma * 2 * cross_correlation(a, p)`. Let me verify this is correct.

The soft-max objective is: `f(a) = (1/gamma) * log(sum_k exp(gamma * c[k]))` where `c[k] = conv(a,a)[k]`.

`df/da[i] = sum_k p[k] * dc[k]/da[i]` where `p[k] = exp(gamma*c[k]) / sum_j exp(gamma*c[j])`

Now `dc[k]/da[i]`:
- `c[k] = sum_j a[j] a[k-j]`
- The terms involving `a[i]`:
  - `j=i`: `a[i] * a[k-i]` (valid if `0 <= k-i < n`)
  - `k-j=i`, i.e., `j=k-i`: `a[k-i] * a[i]` (valid if `0 <= k-i < n`)

"""
Adam-based soft-max gradient descent with L-BFGS refinement, LP polishing,
and subgradient methods for minimizing peak autocorrelation of non-negative step functions.
Uses FFT-accelerated gradients, adaptive beta annealing, and multi-phase optimization.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy import optimize as sci_opt
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Normalizes internally."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def softmax_grad_cross(a, p):
    """Compute 2 * cross_correlation(a, p) via FFT. This is the gradient of soft-max objective."""
    n = len(a)
    p_len = len(p)
    a_rev = a[::-1]
    size = 1
    while size < n + p_len - 1:
        size <<= 1
    A = rfft(a_rev, size)
    P = rfft(p, size)
    full = irfft(A * P, size)
    q = full[n - 1: n - 1 + n]
    return 2.0 * q


def adam_softmax_gd(a, time_budget=200.0, beta_start=10.0, beta_end=150.0,
                     lr_start=0.01, lr_end=0.0001, adam_b1=0.9, adam_b2=0.999,
                     verbose=True, seed=42):
    """Adam optimizer on soft-max objective with simplex projection."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)
    it_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        # Exponential beta annealing
        beta = beta_start * (beta_end / beta_start) ** progress
        # Cosine learning rate
        lr = lr_end + 0.5 * (lr_start - lr_end) * (1.0 + np.cos(np.pi * progress))

        # Compute convolution and softmax weights
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p = exp_vals / float(np.sum(exp_vals))

        # Gradient (no extra gamma factor - it cancels in the soft-max derivative)
        grad = softmax_grad_cross(a, p)

        # Adam update
        t += 1
        m = adam_b1 * m + (1.0 - adam_b1) * grad
        v = adam_b2 * v + (1.0 - adam_b2) * (grad * grad)
        m_hat = m / (1.0 - adam_b1 ** t)
        v_hat = v / (1.0 - adam_b2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it_count
            stall = 0
        else:
            stall += 1
            if stall > 300:
                a = project_simplex(best_a + rng.normal(0, 0.003, n))
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                stall = 0

        a = a_new
        it_count += 1

        if verbose and it_count % 500 == 0:
            print(f"    Adam it{it_count}: obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.6f}, t={elapsed:.1f}s")

        if it_count - best_it > 10000 and it_count > 3000:
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            it_count = best_it

    return best_a, best_obj


def lbfgs_refine(a, beta=50.0, max_iter=1000, verbose=False):
    """L-BFGS-B refinement on soft-max objective with softmax parameterization."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))

    def obj_grad(z):
        z_shifted = z - np.max(z)
        a_soft = np.exp(z_shifted)
        a_soft = a_soft / np.sum(a_soft)

        conv = compute_conv(a_soft)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p = exp_vals / float(np.sum(exp_vals))
        obj_val = float(np.sum(exp_vals) + c_max * beta) / beta  # log-sum-exp / beta

        grad_a = softmax_grad_cross(a_soft, p)
        mean_grad = float(np.sum(a_soft * grad_a))
        grad_z = a_soft * (grad_a - mean_grad)

        return obj_val, grad_z

    a_safe = np.maximum(a, 1e-12)
    z0 = np.log(a_safe)

    try:
        result = sci_opt.minimize(
            obj_grad, z0, method='L-BFGS-B',
            jac=True, options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                               'ftol': 1e-14, 'gtol': 1e-9}
        )
        z_opt = result.x
        a_opt = np.exp(z_opt - np.max(z_opt))
        a_opt = a_opt / np.sum(a_opt)
        obj_opt = compute_obj(a_opt)
    except Exception as e:
        if verbose:
            print(f"    L-BFGS error: {e}")
        return a, compute_obj(a)

    if verbose:
        print(f"    L-BFGS: obj={obj_opt:.8f}, nfev={result.nfev}")

    return a_opt, obj_opt


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j: j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g_lp, time_budget=3.0):

        a_try = (1.0 - t) * a + t * g_lp
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)

    best_t = 0.0
    best_val = obj_at(0.0)

    # Coarse grid
    for t_try in np.linspace(0, 1.0, 21):
        val = obj_at(t_try)
        if val < best_val:
            best_val = val
            best_t = t_try

    # Fine golden section around best
    if best_t > 0.01 and best_t < 0.99:
        lo = max(0.0, best_t - 0.05)
        hi = min(1.2, best_t + 0.05)
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        gr = 2.0 - phi
        c = lo + gr * (hi - lo)
        d = lo + (1.0 - gr) * (hi - lo)
        fc = obj_at(c)
        fd = obj_at(d)
        start = time.monotonic()
        for _ in range(20):
            if time.monotonic() - start > time_budget:
                break
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = lo + gr * (hi - lo)
                fc = obj_at(c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
                fd = obj_at(d)
        t_cand = c if fc < fd else d
        v_cand = min(fc, fd)
        if v_cand < best_val:
            best_val = v_cand
            best_t = t_cand

    return best_t, best_val


def lp_polish(a, time_budget=120.0, max_iters=50, verbose=False):
    """LP-based iterative polishing with line search blending."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        t_opt, obj_opt = line_search_blend(a, g_lp, time_budget=min(2.0, remaining * 0.03))

        if t_opt > 1e-6 and obj_opt < best_obj - 1e-12:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp
            a_new = project_simplex(np.maximum(a_new, 0))
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, elapsed={time.monotonic()-start:.1f}s")

        if stall >= 5:
            noise = rng.normal(0, 0.004 * (1 + stall // 3), n)
            a = project_simplex(best_a + noise)
            stall = 0

    return best_a, best_obj


def subgradient_polish(a, time_budget=60.0, top_k=5, verbose=False):
    """Polish using exact subgradient of max(conv) with top-K peak averaging."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.005
    stall = 0
    rng = np.random.default_rng(999)
    it_count = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        c_max = float(np.max(conv))

        top_idx = np.argsort(conv)[::-1][:top_k]
        weights = np.zeros(len(conv))
        for idx in top_idx:
            weights[idx] = 1.0 / top_k

        # Subgradient via FFT cross-correlation
        a_rev = a[::-1]
        size = 1
        while size < n + len(weights) - 1:
            size <<= 1
        A = rfft(a_rev, size)
        W = rfft(weights, size)
        full = irfft(A * W, size)
        grad = 2.0 * full[n - 1: n - 1 + n]

        # Backtracking line search
        improved = False
        for attempt in range(25):
            a_new = project_simplex(a - lr * grad)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                lr = min(lr * 1.15, 0.05)
                stall = 0
                improved = True
                break
            else:
                lr *= 0.7

        if not improved:
            stall += 1
            if stall > 15:
                lr = 0.005
                a = project_simplex(best_a + rng.normal(0, 0.002, n))
                stall = 0

        it_count += 1
        if verbose and it_count % 20 == 0:
            print(f"    SubGrad it{it_count}: obj={best_obj:.8f}, lr={lr:.5f}, t={time.monotonic()-start:.1f}s")

    return best_a, best_obj


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = (np.exp(-((idx - n // 4) ** 2) / (2.0 * max(1, n // 6) ** 2)) +
             np.exp(-((idx - 3 * n // 4) ** 2) / (2.0 * max(1, n // 6) ** 2)) + 0.01)
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 6:
        a = 1.0 / np.cosh(3.0 * (2 * x - 1))
    elif mode == 7:
        a = np.zeros(n)
        a[0] = 1.0
        a[-1] = 1.0
        a[1:min(3, n)] = 0.3
        a[max(0, n - 4): n - 1] = 0.3
        if n > 4:
            a[1: -1] += 0.05
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization: Adam soft-max GD -> L-BFGS -> LP polish ->
    high-beta Adam GD -> subgradient -> final LP + length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = diverse_init(n_current, rng, mode=0)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")

    # Try arcsine as alternative init
    x = np.linspace(0.01, 0.99, n_current)
    a_arc = 1.0 / np.sqrt(np.maximum(1e-6, 1 - (2 * x - 1) ** 2))
    a_arc = project_simplex(a_arc)
    obj_arc = compute_obj(a_arc)
    if obj_arc < best_score:
        best_score = obj_arc
        best_seq = a_arc.copy()
        print(f"Arcsine init better: obj={best_score:.8f}")

    # === Phase 1: Adam soft-max GD with moderate beta ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        print(f"\n=== Phase 1: Adam SoftMax GD (beta 10->100), ~{remaining*0.35:.0f}s ===")
        alloc = remaining * 0.35
        a1, obj1 = adam_softmax_gd(
            best_seq, time_budget=alloc,
            beta_start=10.0, beta_end=100.0,
            lr_start=0.01, lr_end=0.0001,
            adam_b1=0.9, adam_b2=0.999,
            verbose=True, seed=seed
        )
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1 improved: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj1:.8f} (no improvement)")

    # === Phase 2: L-BFGS quick refinement ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 2: L-BFGS Refinement ===")
        a2, obj2 = lbfgs_refine(best_seq, beta=50.0, max_iter=500, verbose=True)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 improved: obj={best_score:.8f} **")
        else:
            print(f"  Phase2: obj={obj2:.8f}")

    # === Phase 3: LP Polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 3: LP Polish, ~{remaining*0.20:.0f}s ===")
        alloc = remaining * 0.20
        a3, obj3 = lp_polish(best_seq, time_budget=alloc, max_iters=50, verbose=True)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3 improved: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj3:.8f}")

    # === Phase 4: High-beta Adam GD ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 4: Adam GD High Beta (50->250), ~{remaining*0.18:.0f}s ===")
        alloc = remaining * 0.18
        a4, obj4 = adam_softmax_gd(
            best_seq, time_budget=alloc,
            beta_start=50.0, beta_end=250.0,
            lr_start=0.004, lr_end=0.00005,
            adam_b1=0.9, adam_b2=0.999,
            verbose=True, seed=seed + 1
        )
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4 improved: obj={best_score:.8f} **")
        else:
            print(f"  Phase4: obj={obj4:.8f}")

    # === Phase 5: Subgradient polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 5: Subgradient Polish, ~{remaining*0.10:.0f}s ===")
        alloc = remaining * 0.10
        a5, obj5 = subgradient_polish(best_seq, time_budget=alloc, top_k=5, verbose=True)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  Phase5 improved: obj={best_score:.8f} **")
        else:
            print(f"  Phase5: obj={obj5:.8f}")

    # === Phase 6: LP Round 2 ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 6: LP Round 2, ~{remaining*0.15:.0f}s ===")
        alloc = remaining * 0.15
        a6, obj6 = lp_polish(best_seq, time_budget=alloc, max_iters=30, verbose=True)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Phase6 improved: obj={best_score:.8f} **")
        else:
            print(f"  Phase6: obj={obj6:.8f}")

    # === Phase 7: Length exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 7: Length Exploration, {remaining:.0f}s ===")
        n_best = len(best_seq)
        n_options = sorted(set([n_best - 50, n_best + 50, n_best + 100, n_best - 100]))
        n_options = [n for n in n_options if 400 <= n <= 1500]
        # Sort by closeness to current best
        n_options.sort(key=lambda x: abs(x - n_best))
        n_options = n_options[:3]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = project_simplex(np.maximum(a_try, 0))

            alloc_sgd = min(remaining * 0.35, 50.0)
            if alloc_sgd > 15:
                a_try, obj_try = adam_softmax_gd(
                    a_try, time_budget=alloc_sgd,
                    beta_start=30.0, beta_end=120.0,
                    lr_start=0.005, lr_end=0.0005,
                    verbose=False, seed=seed + 200
                )

            remaining = deadline - time.monotonic()
            if remaining > 25:
                alloc_lp = min(remaining * 0.35, 35.0)
                a_try, obj_try = lp_polish(a_try, time_budget=alloc_lp, max_iters=15, verbose=False)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 8: Final Polish, {remaining:.0f}s ===")

        # Subgradient with top-3
        alloc1 = min(remaining * 0.3, 30.0)
        if alloc1 > 8:
            a_f1, obj_f1 = subgradient_polish(best_seq, time_budget=alloc1, top_k=3, verbose=False)
            if obj_f1 < best_score:
                best_score = obj_f1
                best_seq = a_f1.copy()
                print(f"  Final SubGrad: obj={best_score:.8f} **")

        # Final LP
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc2 = min(remaining * 0.5, 40.0)
            a_f2, obj_f2 = lp_polish(best_seq, time_budget=alloc2, max_iters=20, verbose=False)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        # L-BFGS final
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_f3, obj_f3 = lbfgs_refine(best_seq, beta=100.0, max_iter=300, verbose=False)
            if obj_f3 < best_score:
                best_score = obj_f3
                best_seq = a_f3.copy()
                print(f"  Final L-BFGS: obj={best_score:.8f} **")

    # Final validation with evaluate_sequence
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
