```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2
using a minimax QP approach with FFT-accelerated Hankel matrix operations.
The key insight is that the problem can be reformulated as a convex minimax game:
  min_a max_k c_k(a)  =  max_p min_a a^T H_p a
where c_k is the k-th element of conv(a,a) and H_p is a Hankel matrix.
The inner QP is solved via projected gradient with Nesterov acceleration,
and the outer loop uses exponential tilting to concentrate p on active lags.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def hankel_mv(p, a):
    """Compute H_p @ a where H_{j,m} = p[j+m], using FFT.
    p has length 2n-1, a has length n. Returns vector of length n.
    """
    n = len(a)
    N = 1
    while N < 4 * n:
        N <<= 1
    P = rfft(p, N)
    A_rev = rfft(a[::-1], N)
    conv = irfft(P * A_rev, N)
    return conv[n-1:2*n-1]


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_inner_qp(p, a_init, n_iters=50, lr=None):
    """Solve min a^T H_p a s.t. a in simplex using projected Nesterov gradient."""
    n = len(a)
    # Estimate step size
    if lr is None:
        # Use the fact that lambda_max(H_p) <= 1 (for p in simplex)
        lr = 0.49
    else:
        lr = min(lr, 0.49)
    
    beta = 0.75  # momentum
    a = a_init.copy()
    v = np.zeros(n)
    
    for i in range(n_iters):
        # Nesterov lookahead
        y = a + beta * v
        y = project_simplex(y)
        
        # Gradient: 2 * H_p @ y
        grad = 2.0 * hankel_mv(p, y)
        
        # Update
        v_new = beta * (a - y) - lr * grad
        a_new = y + v_new
        a_new = project_simplex(a_new)
        
        # Adaptive: if we went out of bounds too much, reduce lr
        if np.sum(np.abs(a_new - a)) > 2.0:
            lr *= 0.5
        
        v = v_new
        a = a_new
    
    return a


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2*n-1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def minimax_optimize(a_init, time_budget=300.0, verbose=True):
    """
    Minimax QP optimization.
    Outer loop: update p (distribution over lags) via exponential tilting.
    Inner loop: solve QP min a^T H_p a s.t. a in simplex.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    n_lags = 2 * n - 1
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    
    # Initialize p
    p = np.ones(n_lags) / n_lags
    
    outer_it = 0
    stall = 0
    temperature = 1.0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1.0:
            break
        
        # Adaptive inner iterations: more when not converged, fewer when close to time limit
        progress = 1.0 - remaining / time_budget
        if progress < 0.3:
            n_inner = 80
        elif progress < 0.7:
            n_inner = 50
        else:
            n_inner = 30
        
        # Solve inner QP (hot-start from current a)
        a_new = solve_inner_qp(p, a, n_iters=n_inner)
        
        # Compute convolution and objective
        conv = compute_conv(a_new)
        obj = 2.0 * n * float(np.max(conv))
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        # Update p via exponential tilting
        # Concentrate p on peaks of conv
        temperature = 1.0 + 5.0 * progress  # increasing temperature
        c_shifted = conv - np.max(conv)
        # Avoid underflow
        c_shifted = np.clip(c_shifted, -50.0 / temperature, 0.0)
        p_new = np.exp(temperature * c_shifted)
        p_new = p_new / np.sum(p_new)
        # Mix with old p for stability
        alpha_p = 0.3 + 0.4 * progress
        p = (1 - alpha_p) * p + alpha_p * p_new
        p = p / np.sum(p)
        
        a = a_new
        outer_it += 1
        
        if verbose and (outer_it % 5 == 0 or outer_it == 1):
            print(f"    Minimax it{outer_it}: obj={best_obj:.8f}, max_c={np.max(conv):.6e}, "
                  f"min_c={np.min(conv):.6e}, ratio={np.max(conv)/max(np.min(conv),1e-15):.1f}, "
                  f"temp={temperature:.1f}, time={time.monotonic()-start:.0f}s")
        
        if stall > 30 and progress > 0.5:
            # Perturb to escape
            a = a * np.exp(0.01 * rng.standard_normal(n))
            a = np.maximum(a, 0)
            s_a = float(np.sum(a))
            if s_a > 1e-10:
                a = a / s_a
            stall = 0
    
    return best_a


def mirror_descent_polish(a, time_budget=60.0, tau=500.0, verbose=False):
    """Polish using mirror descent with high temperature (nearly exact max)."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99)
    
    conc = np.zeros(n)  # cumulative gradient for step size
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        shifted = tau * (conv - max_c)
        shifted = np.clip(shifted, -500, 0)
        exp_vals = np.exp(shifted)
        p = exp_vals / np.sum(exp_vals)
        
        # Gradient: 2 * H_p @ a (where H_p has H[j,m] = p[j+m])
        grad = 2.0 * hankel_mv(p, a)
        
        # Multiplicative weights update
        eta = 2.0 / (float(np.max(np.abs(grad))) + 1e-10)
        eta = min(eta, 1.0)
        
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
```

The projection onto the non-negative simplex `{x : x >= 0, sum(x) = 1}`:
- Sort v in decreasing order: v_(1) >= v_(2) >= ... >= v_(n).
- Find the largest rho such that v_(rho) - (sum_{i=1}^{rho} v_(i) - 1) / rho > 0.
- theta = (sum_{i=1}^{rho} v_(i) - 1) / rho.
- project(v)_i = max(0, v_i - theta).

This is O(n log n) due to sorting.

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2
using a minimax QP approach with FFT-accelerated Hankel matrix operations.
Key idea: reformulate as min_a max_k c_k(a) = max_p min_a a^T H_p a (convex game),
solve inner QP via projected Nesterov gradient, outer loop via exponential tilting.
Adds mirror-descent polish and local block search for final improvements.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def hankel_mv(p, a):
    """Compute H_p @ a where H_{j,m} = p[j+m]. p has length 2n-1, a has length n."""
    n = len(a)
    N = 1
    while N < 4 * n:
        N <<= 1
    P = rfft(p, N)
    A_rev = rfft(a[::-1], N)
    conv = irfft(P * A_rev, N)
    return conv[n-1:2*n-1]


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if np.max(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_inner_qp(p, a_init, n_iters=60, lr=0.45, beta=0.7):
    """Solve min a^T H_p a s.t. a in simplex using projected Nesterov gradient."""
    n = len(a)
    a = a_init.copy()
    v = np.zeros(n)

    for i in range(n_iters):
        y = a + beta * (a - _prev_a if hasattr(solve_inner_qp, '_prev_a') else 0)
        # Simpler: use momentum
        y = a + beta * v
        y = project_simplex(y)

        grad = 2.0 * hankel_mv(p, y)

        v_new = beta * v - lr * grad
        a_new = y + v_new
        a_new = project_simplex(a_new)

        v = v_new
        a = a_new

    return a


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2*n-1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def minimax_optimize(a_init, time_budget=300.0, verbose=True):
    """
    Minimax QP optimization.
    Outer: update p via exponential tilting toward peaks.
    Inner: solve QP min a^T H_p a s.t. a in simplex via projected Nesterov gradient.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    n_lags = 2 * n - 1
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)

    # Initialize p: mix of uniform and peak-concentrated
    conv_init = compute_conv(a)
    initial_temp = 5.0
    shifted = conv_init - np.max(conv_init)
    shifted = np.clip(shifted, -50.0 / initial_temp, 0.0)
    p_conc = np.exp(initial_temp * shifted)
    p_conc = p_conc / np.sum(p_conc)
    p = 0.5 * (np.ones(n_lags) / n_lags) + 0.5 * p_conc
    p = p / np.sum(p)

    outer_it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1.0:
            break

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Adaptive inner iterations
        if progress < 0.2:
            n_inner = 100
            lr_inner = 0.4
            beta = 0.7
        elif progress < 0.5:
            n_inner = 60
            lr_inner = 0.35
            beta = 0.75
        elif progress < 0.8:
            n_inner = 40
            lr_inner = 0.3
            beta = 0.8
        else:
            n_inner = 25
            lr_inner = 0.25
            beta = 0.85

        # Solve inner QP
        a_new = solve_inner_qp(p, a, n_iters=n_inner, lr=lr_inner, beta=beta)

        # Compute objective
        conv = compute_conv(a_new)
        obj = 2.0 * n * float(np.max(conv))

            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        # Update p via exponential tilting
        temperature = 2.0 + 12.0 * progress
        c_shifted = conv - np.max(conv)
        c_shifted = np.clip(c_shifted, -50.0 / temperature, 0.0)
        p_new = np.exp(temperature * c_shifted)
        p_sum = np.sum(p_new)
        if p_sum < 1e-300:
            p_new = np.ones(n_lags) / n_lags
        else:
            p_new = p_new / p_sum

        alpha_p = 0.3 + 0.5 * progress
        p = (1 - alpha_p) * p + alpha_p * p_new
        p = p / np.sum(p)

        a = a_new
        outer_it += 1

        if verbose and (outer_it % 10 == 0 or outer_it == 1):
            c_ratio = np.max(conv) / max(np.median(conv), 1e-15)
            print(f"    MM it{outer_it}: obj={best_obj:.8f}, max_c={np.max(conv):.4e}, "
                  f"ratio={c_ratio:.2f}, temp={temperature:.1f}, "
                  f"stall={stall}, t={time.monotonic()-start:.0f}s")

        # Escape stagnation
        if stall > 25 and progress > 0.3:
            strength = 0.015 * (1.0 + progress)
            a = a * np.exp(strength * rng.standard_normal(n))
            a = np.maximum(a, 0)
            s_a = float(np.sum(a))
            if s_a > 1e-10:
                a = a / s_a
            else:
                a = best_a.copy()
            stall = 0

    return best_a


def mirror_descent_polish(a, time_budget=60.0, tau_init=200.0, tau_end=3000.0, verbose=False):
    """Polish using mirror descent with increasing temperature (near-exact max)."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99)

    it = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        progress = min(1.0, (time.monotonic() - start) / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress

        conv = compute_conv(a)
        max_c = float(np.max(conv))
        shifted = tau * (conv - max_c)
        shifted = np.clip(shifted, -500.0, 0.0)
        exp_vals = np.exp(shifted)
        p = exp_vals / np.sum(exp_vals)

        # Gradient
        grad = 2.0 * hankel_mv(p, a)

        # Multiplicative weights
        g_max = float(np.max(np.abs(grad)))
        if g_max < 1e-15:
            break
        eta = 1.5 / (g_max + 1e-10)
        eta = min(eta, 0.8)

        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 50 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={time.monotonic()-start:.0f}s")

    return best_a


def block_local_search(a, time_budget=60.0, verbose=False):
    """Local search with random block multiplicative perturbations."""
    n = len(a)

        return a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
