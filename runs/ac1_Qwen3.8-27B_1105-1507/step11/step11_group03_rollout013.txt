```python
import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def build_lp_matrix_sparse(a):
    """Build sparse LP constraint matrix for min M s.t. conv(a,g) <= M, sum(g)=1, g>=0"""
    n = len(a)
    # A is (2n-1) x n, A[k, j] = a[j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j+n):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def lp_solve_sparse(a, solver_list=None):
    """Solve LP: min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1"""
    n = len(a)
    A = build_lp_matrix_sparse(a)
    g_var = cp.Variable(n, nonneg=True)

    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    if solver_list is None:
```

```python
def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def lp_solve(a):
    """Solve min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1."""
    n = len(a)
    # Build convolution matrix A (dense, since n is moderate)
    # A[k, j] = a[j] if 0 <= k-j < n
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a[j]
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None
```

```python
def build_conv_matrix(a):
    n = len(a)
    first_col = np.zeros(2*n-1)
    first_col[:n] = a
    first_row = np.zeros(n)
    first_row[0] = a[0]
    return toeplitz(first_col, first_row)
```
