def evaluate_sequence(seq):
    if not isinstance(seq, list) or not all(isinstance(x, (int, float)) for x in seq):
        return np.inf
    if len(seq) < 1 or np.finfo(float).max in np.array(seq) or np.isnan(np.array(seq).sum()):
        return np.inf
    arr = np.array(seq)
    shape = arr.shape
    if not np.all(shape) or not np.all(np.array([0.0, 1000.0]) >= arr) or not np.all(arr >= 0) or not np.isfinite(arr).all():
        return np.inf
    total_sum = np.sum(arr)
    if total_sum < 0.01:
        return np.inf
    # Core evaluation: convolution and max-normalization
    conv_result = linear_convolve(seq)  # returns convolve as a list for mode='full'
    max_val = np.max(conv_result)      
    return float(2.0 * len(seq) * max_val / (total_sum ** 2))
