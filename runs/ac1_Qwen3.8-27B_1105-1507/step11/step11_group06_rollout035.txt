```python
    """Build sparse matrix A where A[k,i] = a[k-i] if 0<=k-i<n else 0."""
    n = len(a)
    rows_list = []
    cols_list = []
    data_list = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_end < i_start:
            continue
        rows_list.append(np.full(i_end - i_start + 1, k, dtype=np.int32))
        cols_list.append(np.arange(i_start, i_end + 1, dtype=np.int32))
        data_list.append(a[k - i_start:k - i_end + n][::-1])
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    data = np.concatenate(data_list)
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))


def solve_lp(a):
    """Solve the LP: min M s.t. A@g <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix_sparse(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    best_g = None
    best_M = np.inf
    
    for solver in [cp.PDLP, cp.MOSEK, cp.SCS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    if M_val < best_M:
                        best_M = M_val
                        best_g = g_val
        except Exception:
            continue
    
    if best_g is not None:
        return best_g, best_M
    return None, None


def golden_section_line_search(a, g_lp, t_max=1.5, tol=1e-5, max_iter=50):
    """Golden section search for optimal t in [0, t_max]."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    
    def obj_at_t(t):
        a_try = (1.0 - t) * a + t * g_lp
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        a_try = a_try / s
        return compute_obj(a_try)
    
    # First check endpoints and a few points to bracket
    obj_0 = obj_at_t(0)
    obj_max = obj_at_t(t_max)
    
    # If the function is monotonically decreasing, use t_max
    if obj_max < obj_0:
        # Check a few interior points
        best_t = t_max
        best_obj = obj_max
        for t in [0.1, 0.3, 0.5, 0.7, 0.9]:
            o = obj_at_t(t)
            if o < best_obj:
                best_obj = o
                best_t = t
        return best_t, best_obj
    
    # Golden section search
    a_bound = 0.0
    b_bound = t_max
    c = b_bound - gr * (b_bound - a_bound)
    d = a_bound + gr * (b_bound - a_bound)
    fc = obj_at_t(c)
    fd = obj_at_t(d)
    
    for _ in range(max_iter):
        if b_bound - a_bound < tol:
            break
        if fc < fd:
            b_bound = d
            d = c
            fd = fc
            c = b_bound - gr * (b_bound - a_bound)
            fc = obj_at_t(c)
        else:
            a_bound = c
            c = d
            fc = fd
            d = a_bound + gr * (b_bound - a_bound)
            fd = obj_at_t(d)
    
    best_t = (a_bound + b_bound) / 2
    best_obj = obj_at_t(best_t)
    return best_t, best_obj


def lp_polish(a, time_budget=100.0, max_iters=200, verbose=False):
    """Iterative LP polishing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        T0 = time.monotonic()
        g_lp, M_lp = solve_lp(a)
        T1 = time.monotonic()
        
        if g_lp is None:
            break
        
        t_opt, obj_opt = golden_section_line_search(a, g_lp, t_max=1.5)
        
        if t_opt > 1e-8 and obj_opt < best_obj - 1e-14:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            # Try t=1 directly
            obj_direct = compute_obj(g_lp)
            if obj_direct < best_obj - 1e-14:
                best_obj = obj_direct
                best_a = g_lp.copy()
                a = g_lp.copy()
                stall = 0
            else:
                stall += 1
        
        it += 1
        if verbose and it % 10 == 0:
            print(f"    LP it={it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                  f"lp_time={T1-T0:.3f}s, elapsed={time.monotonic()-start:.1f}s")
        
        if stall >= 3:
            # Perturbation strategies
            strategy = stall % 5
            if strategy == 0:
                # Frequency perturbation
                freq = rng.integers(1, max(2, n // 50))
                phase = rng.uniform(0, 2 * np.pi)
                idx = np.arange(n, dtype=np.float64)
                pert = 0.02 * np.cos(2 * np.pi * freq * idx / n + phase)
                a = best_a * np.exp(pert)
            elif strategy == 1:
                # Random smooth perturbation
                noise = rng.normal(0, 0.01, n)
                kernel = np.ones(3) / 3
                noise = np.convolve(noise, kernel, mode='same')
                a = best_a + noise
            elif strategy == 2:
                # Mix with U-shape
                x = np.linspace(0, 1, n)
                u = 1.0 / (0.05 + 3.0 * (2*x - 1)**2)
                a = 0.5 * best_a + 0.5 * u / u.sum()
            elif strategy == 3:
                # Mirror perturbation
                half = n // 2
                a[:half] = best_a[:half] * (1 + 0.03 * rng.normal(0, 1, half))
                a[half:] = best_a[half:][::-1] * (1 + 0.03 * rng.normal(0, 1, n-half))
                # Actually just perturb
                a = best_a * (1 + 0.03 * rng.normal(0, 1, n))
            else:
                # Larger restart
                x = np.linspace(0, 1, n)
                a = 1.0 / (0.02 + 5.0 * (2*x - 1)**4)
                a = 0.3 * a / a.sum() + 0.7 * best_a
            
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 0.01:
                a = a / s
            else:
                a = best_a.copy()
            stall = 0
    
    return best_a


def u_shaped_init(n, sharpness=3.0):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.02 + 5.0 * (2 * x - 1) ** 4)
    elif mode == 6:
        # Arcsine-like
        a = 1.0 / np.sqrt(np.maximum(x * (1 - x), 1e-6))
    elif mode == 7:
        # Beta(0.5, 0.5) = arcsine
        a = 1.0 / np.sqrt(np.maximum(x * (1 - x), 1e-6))
        a = a * (1 + 0.1 * np.sin(2*np.pi*3*x))
    else:
        a = 1.0 / (0.1 + 2.0 * np.abs(2 * x - 1))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s


def subgradient_descent(a, time_budget=30.0):
    """Exact subgradient descent on the normalized objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    a_avg = a.copy()
    lr0 = 0.03
    momentum = np.zeros(n)
    beta = 0.8
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a_avg)
        k_max = int(np.argmax(conv))
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_max - i
            j2 = i - k_max
            if 0 <= j1 < n:
                grad[i] += a_avg[j1]
            if 0 <= j2 < n:
                grad[i] += a_avg[j2]
        
        lr_t = lr0 / (1.0 + 0.003 * it)
        momentum = beta * momentum - lr_t * grad
        a_new = a_avg + momentum
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
        else:
            a_new = a_new / s
        a_avg = (it / (it + 1.0)) * a_avg + (1.0 / (it + 1.0)) * a_new
        obj = compute_obj(a_avg)
        if obj < best_obj:
            best_obj = obj
            best_a = a_avg.copy()
            best_it = it
        it += 1
        if it - best_it > 2000:
            a_avg = best_a.copy()
            momentum = np.zeros(n)
            lr0 = max(0.01, lr0 * 0.5)
            best_it = it
    
    return best_a, best_obj
```

```python
"""
LP-based optimization with PDLP, sparse matrices, golden-section line search
with extrapolation, adaptive perturbation, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]
```

```python
"""
LP-based optimization with PDLP solver, sparse matrix construction,
golden-section line search with extrapolation, adaptive perturbation,
and aggressive multi-n exploration. Targets objective < 1.5027.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_pattern(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def build_conv_matrix(a, pattern):
    rows, cols = pattern
    data = a[rows - cols]
    n = len(a)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(a, pattern):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm, pattern)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    best_g = None
    best_M = np.inf
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    if M_val < best_M:
                        best_M = M_val
                        best_g = g_val
        except Exception:
            continue
    if best_g is not None:
        return best_g, best_M
    return None, None


def golden_section_line_search(a, g_lp, t_max=2.0, tol=1e-5, max_iter=40):
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    def obj_at_t(t):
        a_try = (1.0 - t) * a + t * g_lp
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        a_try = a_try / s
        return compute_obj(a_try)

    obj_0 = obj_at_t(0.0)
    # Scan to find approximate minimum region
    best_t = 0.0
    best_obj = obj_0
    n_scan = 20
    for i in range(1, n_scan):
        t = t_max * i / n_scan
        o = obj_at_t(t)
        if o < best_obj:
            best_obj = o
            best_t = t

    if best_t <= 0.01 or best_t >= t_max - 0.01:
        return best_t, best_obj

    # Golden section in neighborhood
    a_bound = max(0.0, best_t - 0.15)
    b_bound = min(t_max, best_t + 0.15)
    c = b_bound - gr * (b_bound - a_bound)
    d = a_bound + gr * (b_bound - a_bound)
    fc = obj_at_t(c)
    fd = obj_at_t(d)

    for _ in range(max_iter):
        if b_bound - a_bound < tol:
            break
        if fc < fd:
            b_bound = d
            d = c
            fd = fc
            c = b_bound - gr * (b_bound - a_bound)
            fc = obj_at_t(c)
        else:
            a_bound = c
            c = d
            fc = fd
            d = a_bound + gr * (b_bound - a_bound)
            fd = obj_at_t(d)

    best_t = (a_bound + b_bound) / 2.0
    best_obj = obj_at_t(best_t)
    return best_t, best_obj

def u_shaped_init(n, sharpness=3.0):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = (np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) +
             np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01)
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        a = np.convolve(a, np.ones(7) / 7.0, mode='same')
    elif mode == 4:
        a = np.exp(6 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.02 + 6.0 * (2 * x - 1) ** 4)
    elif mode == 6:
        a = 1.0 / np.sqrt(np.maximum(x * (1 - x), 1e-6))
    elif mode == 7:
        a = 1.0 / (0.01 + 10.0 * (2 * x - 1) ** 2)
    elif mode == 8:
        a = np.sin(np.pi * x) ** 2 + 0.01
    else:
        a = 1.0 / (0.1 + 3.0 * np.abs(2 * x - 1))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n

def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s


def subgradient_descent(a, time_budget=30.0):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    a_avg = a.copy()
    lr0 = 0.04
    momentum = np.zeros(n)
    beta = 0.75
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a_avg)
        k_max = int(np.argmax(conv))
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_max - i
            j2 = i - k_max
            if 0 <= j1 < n:
                grad[i] += a_avg[j1]
            if 0 <= j2 < n:
                grad[i] += a_avg[j2]

        lr_t = lr0 / (1.0 + 0.004 * it)
        momentum = beta * momentum - lr_t * grad
        a_new = a_avg + momentum
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
        else:
            a_new = a_new / s
        a_avg = (it / (it + 1.0)) * a_avg + (1.0 / (it + 1.0)) * a_new
        obj = compute_obj(a_avg)
        if obj < best_obj:
            best_obj = obj
            best_a = a_avg.copy()
            best_it = it
        it += 1
        if it - best_it > 1500:
            a_avg = best_a.copy()
            momentum = np.zeros(n)
            lr0 = max(0.005, lr0 * 0.5)
            best_it = it

    return best_a, best_obj

            a_r = a_r / a_r.sum()
            alloc = min(remaining * 0.3, 70.0)
            if alloc > 20:
                a_r = lp_polish(a_r, time_budget=alloc, max_iters=80, verbose=False)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 6: Final intensive LP ===
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining - 10, 250.0)
        print(f"\n=== Phase 6: Final Polish (budget={alloc:.0f}s) ===")
        if alloc > 20:
            a_final = lp_polish(best_seq, time_budget=alloc, max_iters=500, verbose=True)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final: obj={best_score:.8f} **")

    # === Phase 7: Last squeeze ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_sq, obj_sq = subgradient_descent(best_seq, time_budget=min(remaining * 0.35, 20.0))
        if obj_sq < best_score:
            best_score = obj_sq
            best_seq = a_sq.copy()
            print(f"  Squeeze SG: obj={best_score:.8f} **")
        remaining2 = deadline - time.monotonic()
        if remaining2 > 12:
            a_sq2 = lp_polish(best_seq, time_budget=min(remaining2 - 5, 35.0),
                              max_iters=80, verbose=False)
            obj_sq2 = compute_obj(a_sq2)
            if obj_sq2 < best_score:
                best_score = obj_sq2
                best_seq = a_sq2.copy()
                print(f"  Squeeze LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
```
