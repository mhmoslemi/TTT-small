"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated smoothed gradient descent, sparsification,
exact coordinate descent polishing, and adaptive length exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def smooth_max_and_grad(a, tau):
    """
    Compute smoothed max of conv(a,a) and its gradient w.r.t. a.
    Returns (smooth_max, grad) where grad has same shape as a.
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Softmax weights for smoothing
    conv_shifted = conv - np.max(conv)
    w = np.exp(conv_shifted / tau)
    W = float(np.sum(w))
    w = w / W  # now sum(w) = 1
    
    smooth_max = tau * np.log(W) + float(np.max(conv))
    
    # Gradient: d(smooth_max)/d(a[i]) = 2 * sum_k w[k] * a[k-i] (for valid k-i)
    # = 2 * g[i] where g[i] = sum_m a[m] * w[m+i]
    
    # Compute g via FFT cross-correlation
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(w, size)
    # g = ifft(fw * conj(fa))[:n]
    g = irfft(fw * np.conj(fa), size)[:n]
    
    # Make sure g is real (numerical issues)
    g = np.real(g)
    
    return smooth_max, 2.0 * g


def objective_and_grad(a):
    """
    Compute f(a) = 2n * smooth_max / S^2 and its gradient.
    """
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf, np.zeros(n)
    
    # Use a small tau for near-exact max
    tau = 1e-4
    smooth_max, grad_sm = smooth_max_and_grad(a, tau)
    
    # f = 2n * smooth_max / S^2
    f = 2.0 * n * smooth_max / (S * S)
    
    # grad_f = 2n/S^2 * grad_sm - 2 * f / S * ones
    factor1 = 2.0 * n / (S * S)
    factor2 = 2.0 * f / S
    grad_f = factor1 * grad_sm - factor2 * np.ones(n)
    
    return f, grad_f


def smooth_gradient_descent(a, time_budget, tau_init=0.005, tau_min=1e-5,
                            lr_init=0.005, verbose=True, seed=42):

    Gradient descent on the smoothed objective with adaptive tau and step size.
    """

    a = np.maximum(a, 0).astype(np.float64).copy()

    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    tau = tau_init
    lr = lr_init
    n_iter = 0
    stall = 0
    f_prev = best_obj
    
    while time.monotonic() - start < time_budget:
        n_iter += 1
        
        # Compute gradient with current tau
        smooth_max, grad_sm = smooth_max_and_grad(a, tau)
        S_cur = float(np.sum(a))
        f_cur = 2.0 * n * smooth_max / (S_cur * S_cur)
        factor1 = 2.0 * n / (S_cur * S_cur)
        factor2 = 2.0 * f_cur / S_cur
        grad = factor1 * grad_sm - factor2 * np.ones(n)
        
        # Line search
        best_local_obj = f_cur
        best_local_a = a.copy()
        
        for step_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = lr * step_mult
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses smoothed gradient descent (log-sum-exp), exact 1D coordinate descent with
ternary search, sparsification, multi-initialization, and adaptive length exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def smooth_max_and_grad(a, tau):
    """
    Compute smoothed max of conv(a,a) and gradient w.r.t. a.
    Uses log-sum-exp smoothing with parameter tau.
    Returns (smooth_max_value, gradient_array).
    """
    n = len(a)
    conv = compute_conv(a)

    # Softmax weights
    conv_shifted = conv - np.max(conv)
    w = np.exp(conv_shifted / tau)
    W = float(np.sum(w))
    w = w / W  # now sum(w) = 1

    smooth_max = tau * np.log(W) + float(np.max(conv))

    # Gradient of smooth_max w.r.t. a[i] = 2 * sum_k w[k] * a[k-i]
    # = 2 * g[i] where g[i] = sum_m a[m] * w[m+i]
    n = len(a)
    g = np.empty(n)
    for i in range(n):
        g[i] = np.dot(a, w[i:i + n])

    return smooth_max, 2.0 * g


def objective_grad(a, tau):
    """Compute f(a) = 2n*smooth_max/S^2 and its gradient."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf, np.zeros(n)
    smooth_max, grad_sm = smooth_max_and_grad(a, tau)
    f = 2.0 * n * smooth_max / (S * S)
    factor1 = 2.0 * n / (S * S)
    factor2 = 2.0 * f / S
    grad = factor1 * grad_sm - factor2 * np.ones(n)
    return f, grad


def smooth_gradient_descent(a, time_budget, tau_init=0.003, tau_min=1e-5,
                            lr_init=0.003, verbose=True, seed=42):
    """Gradient descent on smoothed objective with adaptive tau and step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    tau = tau_init
    lr = lr_init
    n_iter = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        n_iter += 1

        smooth_max, grad_sm = smooth_max_and_grad(a, tau)
        S_cur = float(np.sum(a))
        f_cur = 2.0 * n * smooth_max / (S_cur * S_cur)
        factor1 = 2.0 * n / (S_cur * S_cur)
        factor2 = 2.0 * f_cur / S_cur
        grad = factor1 * grad_sm - factor2 * np.ones(n)

        # Gradient norm for adaptive step
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-12:
            stall += 10
        else:
            grad = grad / grad_norm * float(np.max(np.abs(grad)))

        # Line search
        improved = False
        for step_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            step = lr * step_mult
            a_new = a - step * grad

            S_new = float(np.sum(a_new))

        else:
            stall += 1
            lr *= 0.6

            if S_new > 1e-10:

            else:

            stall = 0
            lr = lr_init * 0.5
            tau = max(tau * 2, tau_init * 0.5)

def coordinate_descent_exact(a, time_budget, seed=42, verbose=True):
    """
    Exact 1D coordinate descent using ternary search on the 1D minimax problem.
    Works with normalized sequence (sum=1).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    sweep = 0
    no_improve = 0
    max_no_improve = 6

    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            a_j = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)

            if k_start > k_end:
                continue

            # Unaffected max (computed once per j)
            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))

            # Precompute for affected entries
            affected_indices = np.arange(k_start, k_end + 1)
            n_affected = len(affected_indices)
            base = np.zeros(n_affected)
            coeff = np.zeros(n_affected)
            quad_mask = np.zeros(n_affected, dtype=bool)

            for idx, k in enumerate(affected_indices):
                if k == 2 * j:
                    base[idx] = conv[k] - a_j * a_j
                    quad_mask[idx] = True
                else:
                    ki = k - j
                    if 0 <= ki < n:
                        base[idx] = conv[k] - 2.0 * a_j * a[ki]
                        coeff[idx] = 2.0 * a[ki]
                    else:
                        base[idx] = conv[k]

            def obj_1d(v):
                S_new = 1.0 + v - a_j
                if S_new < 1e-10:
                    return np.inf
                vals = base + coeff * v
                if np.any(quad_mask):
                    vals = vals.copy()
                    vals[quad_mask] = base[quad_mask] + v * v
                max_affected = float(np.max(vals))
                max_total = max(max_unaffected, max_affected)
                return 2.0 * n * max_total / (S_new * S_new)

            # Evaluate at strategic points to find good bracket
            current_obj = obj_1d(a_j)
            test_points = [0.0, a_j * 0.5, a_j * 1.5, 1.0 / n, 0.5 / n, 2.0 / n,
                           a_j * 0.2, a_j * 3.0, 0.01, 0.05]
            best_v = a_j
            best_f = current_obj
            for tp in test_points:
                tp = max(0.0, min(tp, 0.3))
                if abs(tp - best_v) < 1e-12:
                    continue
                f_tp = obj_1d(tp)
                if f_tp < best_f:
                    best_f = f_tp
                    best_v = tp

            # Ternary search refinement around best_v
            if best_v > 0.01:
                w = max(0.01, a_j * 0.5)
                tlo = max(0.0, best_v - w)
                thi = min(best_v + w, 0.3)
            else:
                tlo = 0.0
                thi = 0.05
            if thi - tlo < 1e-6:
                thi = tlo + 0.01

            for _ in range(12):
                m1 = tlo + (thi - tlo) / 3.0
                m2 = thi - (thi - tlo) / 3.0
                f1 = obj_1d(m1)
                f2 = obj_1d(m2)
                if f1 < f2:
                    thi = m2
                else:
                    tlo = m1

            v_opt = (tlo + thi) / 2.0
            f_opt = obj_1d(v_opt)
            if f_opt < best_f:
                best_f = f_opt
                best_v = v_opt

            if best_f < current_obj - 1e-12:
                v = best_v
                delta = v - a_j
                S_new = 1.0 + delta

                # Update conv in-place
                for idx, k in enumerate(affected_indices):
                    if quad_mask[idx]:
                        conv[k] = base[idx] + v * v
                    else:
                        conv[k] = base[idx] + coeff[idx] * v

                a[j] = v
                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S

                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")

    return best_a


def sparsify(a, threshold_frac=0.001):
    """Set small coordinates to 0 and redistribute mass."""
    n = len(a)
    a = a.copy()
    mean_val = np.mean(a)
    threshold = threshold_frac * mean_val
    small_mask = a < threshold
    if np.sum(small_mask) == 0 or np.sum(small_mask) >= n:
        return a
    mass = float(np.sum(a[small_mask]))
    a[small_mask] = 0
    large_mask = ~small_mask
    a[large_mask] += mass / np.sum(large_mask) * (a[large_mask] / a[large_mask].max())
    S = float(np.sum(a))
    if S > 1e-10:
        a = a / S
    return a


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 10):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    S = float(np.sum(a_new))
    if S < 1e-10:
        return a.copy()
    return a_new / S

    sorted_idx = np.argsort(a)

    recipients = sorted_idx[n // 2:]

    while time.monotonic() - start < time_budget:
        iters += 1
        i = int(rng.choice(donors))
        j = int(rng.choice(recipients))
        if i == j or a[i] < 1e-8:
            continue

        # Try moving delta from i to j
        # Affected range: from min(i,j) to max(i,j)+n-1
        k_start = min(i, j)
        k_end = min(max(i, j) + n - 1, 2 * n - 2)

        a_i = a[i]
        a_j = a[j]

        # Try several delta values
        best_delta = 0.0
        best_trial = best_obj

        max_delta = a_i * 0.9
        for frac in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
            delta = frac * max_delta
            if delta < 1e-10:
                continue

            # Compute new conv in affected range
            affected = np.zeros(k_end - k_start + 1)
            for idx, k in enumerate(range(k_start, k_end + 1)):
                # Original conv[k]
                val = conv[k]
                # Change due to a[i] -> a[i]-delta, a[j] -> a[j]+delta
                # Terms involving a[i]: positions where i or k-i is involved
                # delta_a_i = -delta, delta_a_j = +delta
                change = 0.0
                # Contribution from a[i] change: -delta * (a[k-i] if k!=2i else a[i]+delta/2)
                ki = k - i
                if 0 <= ki < n:
                    change += -delta * a[ki]
                    if ki == i:
                        change += -delta * a[i]  # second occurrence
                        change += delta * delta  # cross term
                kj = k - j
                if 0 <= kj < n:
                    change += delta * (a[j] + delta)  # new a[j]
                    if kj != j:
                        change += delta * a[kj]  # wait this is getting wrong

                # Let me just compute directly
                affected[idx] = val  # placeholder

            # Direct computation for correctness
            a_test = a.copy()
            a_test[i] = a_i - delta
            a_test[j] = a_j + delta
            new_conv_entries = np.zeros(k_end - k_start + 1)
            for idx, k in enumerate(range(k_start, k_end + 1)):
                s_val = 0.0
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for m in range(lo, hi + 1):
                    s_val += a_test[m] * a_test[k - m]
                new_conv_entries[idx] = s_val

            max_new_conv = max(float(np.max(new_conv_entries)),
                              float(np.max(conv[:k_start])) if k_start > 0 else 0,
                              float(np.max(conv[k_end + 1:])) if k_end < 2 * n - 2 else 0)
            trial_obj = 2.0 * n * max_new_conv  # sum is still 1

            if trial_obj < best_trial - 1e-13:
                best_trial = trial_obj
                best_delta = delta

        if best_delta > 0:
            a[i] -= best_delta
            a[j] += best_delta
            # Recompute conv
            conv = compute_conv(a)
            new_obj = 2.0 * n * float(np.max(conv))
            if new_obj < best_obj - 1e-13:
                best_obj = new_obj
                best_a = a.copy()

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    PairMove {iters}: obj={best_obj:.10f}, t={elapsed:.1f}s")

    return best_a


def make_initialization(n, seed=42):
    """Create a diverse initial sequence."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 5)
    if choice == 0:
        a = np.ones(n)
    elif choice == 1:
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
        a += 0.1 * rng.standard_normal(n)
    elif choice == 2:
        a = np.exp(-0.5 * ((np.arange(n) - n / 2) / (n / 4)) ** 2)
        a += 0.05 * rng.standard_normal(n)
    elif choice == 3:
        a = np.abs(np.sin(2 * np.pi * np.arange(n) / (2 * n))) * 0.7 + 0.3
        a += 0.05 * rng.standard_normal(n)
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)
    a = np.maximum(a, 0)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n)
        S = n
    return a / S


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: smoothed gradient descent + exact coordinate descent +
    pair moves + sparsification + length exploration + multi-restart.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    # Load parent
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0)
        if a.size == 0 or a.sum() < 0.01:
            raise ValueError("Invalid parent")
        a = a / a.sum()
        print(f"Parent: n={a.size}, obj={compute_obj_norm(a):.10f}")
    except Exception:
        a = np.ones(800)
        a = a / a.sum()
        print(f"Default: n=800, obj={compute_obj_norm(a):.10f}")

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    n_best = len(best_a)
    rng = np.random.default_rng(seed)

    # === Phase 1: Smoothed Gradient Descent (35% time) ===
    print("\n--- Phase 1: Smoothed Gradient Descent ---")
    p1_budget = min(budget_s * 0.35, 480.0)
    a1 = smooth_gradient_descent(best_a, time_budget=p1_budget, tau_init=0.003,
                                 tau_min=1e-5, lr_init=0.003, verbose=True, seed=seed)
    obj1 = compute_obj_norm(a1)
    if obj1 < best_obj:
        best_obj = obj1
        best_a = a1.copy()
        print(f"  P1: {best_obj:.10f} **")
    else:
        print(f"  P1: {obj1:.10f}")

    # === Phase 2: Exact Coordinate Descent (20% time) ===
    print("\n--- Phase 2: Exact CD ---")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = min(remaining * 0.45, 300.0)
        a2 = coordinate_descent_exact(best_a, time_budget=p2_budget, seed=seed, verbose=True)
        obj2 = compute_obj_norm(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            print(f"  P2: {best_obj:.10f} **")
        else:
            print(f"  P2: {obj2:.10f}")

    # === Phase 3: Sparsify + Re-optimize (15% time) ===
    print("\n--- Phase 3: Sparsify + Re-optimize ---")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        a_spar = sparsify(best_a, threshold_frac=0.002)
        obj_spar = compute_obj_norm(a_spar)
        print(f"  After sparsify: {obj_spar:.10f} (nnz={np.sum(a_spar > 1e-10)})")

        # Re-optimize sparsified version
        p3a_budget = min(remaining * 0.3, 150.0)
        a3a = smooth_gradient_descent(a_spar, time_budget=p3a_budget, tau_init=0.002,
                                       tau_min=1e-5, lr_init=0.002, verbose=False, seed=seed + 100)
        obj3a = compute_obj_norm(a3a)
        if obj3a < best_obj:
            best_obj = obj3a
            best_a = a3a.copy()
            print(f"  P3a (sparsified GD): {best_obj:.10f} **")

        remaining = deadline - time.monotonic()
        if remaining > 30:
            p3b_budget = min(remaining * 0.4, 120.0)
            a3b = coordinate_descent_exact(best_a, time_budget=p3b_budget,
                                            seed=seed + 200, verbose=False)
            obj3b = compute_obj_norm(a3b)
            if obj3b < best_obj:
                best_obj = obj3b
                best_a = a3b.copy()
                print(f"  P3b (sparsified CD): {best_obj:.10f} **")

    # === Phase 4: Multi-restart with perturbations (20% time) ===
    print("\n--- Phase 4: Multi-Restart ---")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_restarts = 3
        restart_budget = remaining / (n_restarts + 1)
        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            alloc = min(restart_budget, remaining * 0.5)
            strength = 0.01 + 0.005 * r
            a_pert = fourier_perturb(best_a, rng, strength=strength)

            a_r = smooth_gradient_descent(a_pert, time_budget=alloc * 0.5,
                                           tau_init=0.002, tau_min=1e-5,
                                           lr_init=0.002, verbose=False,
                                           seed=seed + 5000 + r * 7)
            obj_r = compute_obj_norm(a_r)

            remaining = deadline - time.monotonic()
            if remaining > 20 and obj_r < best_obj + 0.01:
                a_r = coordinate_descent_exact(a_r, time_budget=min(remaining * 0.3, 60.0),
                                                seed=seed + 6000 + r * 11, verbose=False)
                obj_r = compute_obj_norm(a_r)

            if obj_r < best_obj:
                best_obj = obj_r
                best_a = a_r.copy()
                print(f"  Restart {r}: {best_obj:.10f} **")
            else:
                print(f"  Restart {r}: {obj_r:.10f}")

    # === Phase 5: Length exploration (15% time) ===
    print("\n--- Phase 5: Length Exploration ---")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    n_options = sorted(set([n_best - 50, n_best - 20, n_best + 20, n_best + 50,
                            n_best + 100, n_best - 100, n_best + 50, n_best - 50]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and n != n_best]

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        # Resample from best_a to new length
        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, n_try)
        a_try = np.interp(x_new, x_old, best_a)
        a_try = np.maximum(a_try, 0)
        S_try = float(np.sum(a_try))
        if S_try < 1e-10:
            continue
        a_try = a_try / S_try

        alloc = min(remaining * 0.35, 120.0)
        if alloc > 30:
            a_try = smooth_gradient_descent(a_try, time_budget=alloc * 0.5,
                                             tau_init=0.002, tau_min=1e-5,
                                             lr_init=0.002, verbose=False,
                                             seed=n_try * 3 + 1)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_try = coordinate_descent_exact(a_try, time_budget=min(remaining * 0.3, 50.0),
                                                  seed=n_try * 7 + 3, verbose=False)

        obj_try = compute_obj_norm(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: {best_obj:.10f} **")
        else:
            print(f"  n={n_try}: {obj_try:.10f}")

    # === Phase 6: Final polish ===
    print("\n--- Phase 6: Final Polish ---")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_fin = coordinate_descent_exact(best_a, time_budget=min(remaining * 0.7, 80.0),
                                          seed=seed + 99999, verbose=False)
        obj_fin = compute_obj_norm(a_fin)
        if obj_fin < best_obj:
            best_obj = obj_fin
            best_a = a_fin.copy()
            print(f"  Final CD: {best_obj:.10f} **")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_fin2 = smooth_gradient_descent(best_a, time_budget=min(remaining * 0.6, 40.0),
                                              tau_init=0.001, tau_min=1e-6,
                                              lr_init=0.001, verbose=False,
                                              seed=seed + 12345)
            obj_fin2 = compute_obj_norm(a_fin2)
            if obj_fin2 < best_obj:
                best_obj = obj_fin2
                best_a = a_fin2.copy()
                print(f"  Final GD: {best_obj:.10f} **")

    # === Final validation ===
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val != np.inf:
        best_obj = val
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
