"""
LP-based alternating optimization for minimizing 2*n*max(conv(a))/(sum(a))^2.
Uses parameterized monotonic initializations, intensive LP polishing with warm starts,
n-exploration, and multi-restart strategy to push below the 1.5027 target.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, warm_start=None):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if warm_start is not None:
        try:
            g.value = warm_start
            M.value = float(np.max(A @ warm_start))
        except Exception:
            pass
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver, eps_abs=1e-9, eps_rel=1e-9, max_iter=100000)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, g_lp, t_max=1.5, n_coarse=50, n_fine=40):
    a_norm = a / float(np.sum(a))
    g_norm = g_lp / float(np.sum(g_lp))
    best_t = 0.0
    best_obj = compute_obj(a_norm)
    ts = np.linspace(0.0, t_max, n_coarse)
    best_coarse_i = 0
    for i, t in enumerate(ts):
        a_try = (1.0 - t) * a_norm + t * g_norm
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
            best_coarse_i = i
    if 0.0 < best_t < t_max:
        lo = max(0.0, ts[max(0, best_coarse_i - 1)])
        hi = min(t_max, ts[min(n_coarse - 1, best_coarse_i + 1)])
        ts_fine = np.linspace(lo, hi, n_fine)
        for t in ts_fine:
            a_try = (1.0 - t) * a_norm + t * g_norm
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t
    return best_t, best_obj


def subgradient_step(a, time_budget=20.0):
    """Modifier subgradient method on the exact quadratic objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    a_avg = a.copy()
    momentum = np.zeros(n)
    beta = 0.8
    it = 0
    best_it = 0
    lr0 = 0.03

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a_avg)
        k_max = int(np.argmax(conv))
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_max - i
            j2 = i - k_max
            if 0 <= j1 < n:
                grad[i] += a_avg[j1]
            if 0 <= j2 < n:
                grad[i] += a_avg[j2]
        lr_t = lr0 / (1.0 + 0.01 * it)
        momentum = beta * momentum - lr_t * grad
        a_new = a_avg + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            momentum = np.zeros(n)
        else:
            a_new = a_new / s_new
        a_avg = (it / (it + 1.0)) * a_avg + (1.0 / (it + 1.0)) * a_new
        obj = compute_obj(a_avg)
        if obj < best_obj:
            best_obj = obj
            best_a = a_avg.copy()
            best_it = it
        it += 1
        if it - best_it > 2000:
            a_avg = best_a.copy()
            momentum = np.zeros(n)
            lr0 = 0.01
            best_it = it
    return best_a


def parametric_init(n, p, direction='increasing'):
    """Generate a monotonic sequence a[i] ∝ (i+1)^p or (n-i)^p."""
    idx = np.arange(n, dtype=np.float64)
    if direction == 'increasing':
        a = (idx + 1) ** p
    else:
        a = (n - idx) ** p
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, mode=0):

    if mode == 0:

    elif mode == 1:

    elif mode == 2:

    elif mode == 3:
        a = 1.0 / (0.3 + idx / n)
    elif mode == 4:
        a = (idx + 1) ** 0.4
    elif mode == 5:
        a = np.exp(1.5 * idx / n)
    elif mode == 6:
        a = (idx + 1) ** 2.0
    else:
        a = (idx + 1) ** rng.uniform(0.1, 3.0)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    if n_src == n_dst:
        return a_src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s
