    x = np.linspace(-1, 1, n)
    idx = np.arange(n, dtype=np.float64)
    if mode == 0:
        a = 1.0 / (0.1 + 3.0 * x**2)
    elif mode == 1:
        a = 1.0 / (0.05 + 5.0 * x**4)
    elif mode == 2:
        a = np.exp(3.0 * x**2)
    elif mode == 3:
        a = 1.0 / (0.08 + 4.0 * np.abs(x))
    elif mode == 4:
        a = 0.5 + 0.5 * np.cos(np.pi * x)
        a = a / (a + 0.1)
    elif mode == 5:
        # Two-hump
        a = np.exp(-((idx - n*0.2)**2)/(2*(n*0.1)**2)) + np.exp(-((idx - n*0.8)**2)/(2*(n*0.1)**2)) + 0.05
    elif mode == 6:
        # Fourier-based flat spectrum attempt
        freqs = rng.choice(range(1, n//2), size=min(20, n//4), replace=False)
        a = np.ones(n)
        for f in freqs:
            a += np.cos(2 * np.pi * f * idx / n) * rng.uniform(0.3, 0.7)
        a = np.maximum(a, 0)
    else:
        a = 1.0 / (0.1 + 2.0 * x**2 + 0.5 * x**4)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a_src, n_dst):
    n_src = len(a_src)
    if n_src == n_dst:
        return a_src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: parameterized init + Nesterov subgrad + LP polish + n sweep.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s > 0.01:
            best_seq = parent / s
            best_mc = max_conv(best_seq)
            n_best = len(best_seq)
            print(f"Parent: n={n_best}, mc={best_mc:.10f}, obj={2*n_best*best_mc:.8f}")
        else:
            raise ValueError
    except Exception:
        n_best = 800
        best_seq = 1.0 / (0.1 + 3.0 * np.linspace(-1, 1, n_best)**2)
        best_seq = best_seq / best_seq.sum()
        best_mc = max_conv(best_seq)
        print(f"Default init: n={n_best}, mc={best_mc:.10f}")

    best_obj = 2.0 * n_best * best_mc

    # === Phase 1: Parameterized shape search (global) ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        print(f"\n=== Phase 1: Parameterized Shape Search (budget={min(remaining, 200):.0f}s) ===")
        alloc = min(remaining * 0.12, 150.0)
        t0 = time.monotonic()

        # Search over several n values with parameterized families
        n_list = [n_best]
        for dn in [-200, -100, 0, 100, 200, 300, 400]:
            n_try = n_best + dn
            if 300 <= n_try <= 2500:
                n_list.append(n_try)
        n_list = sorted(set(n_list))

        for n_try in n_list:
            if time.monotonic() - t0 > alloc:
                break
            # Fine parameter search
            a_p, mc_p, params = param_search(
                n_try,
                alpha_range=(0.01, 2.0),
                beta_range=(0.0, 10.0),
                gamma_range=(0.0, 5.0),
                n_alpha=15, n_beta=12, n_gamma=6
            )
            if a_p is not None:
                obj_p = 2.0 * n_try * mc_p
                if obj_p < best_obj:
                    best_obj = obj_p
                    best_mc = mc_p
                    best_seq = a_p.copy()
                    n_best = n_try
                    print(f"  Param search n={n_try}: obj={obj_p:.8f} ** params={params}")

        print(f"  Phase 1 done. Best obj so far: {best_obj:.8f}")

    # === Phase 2: Nesterov subgradient (main refinement) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 2: Nesterov Subgradient (budget={min(remaining*0.3, 300):.0f}s) ===")
        alloc = min(remaining * 0.25, 250.0)
        a_ng, mc_ng = nesterov_subgrad(best_seq, time_budget=alloc, verbose=True)
        obj_ng = 2.0 * n_best * mc_ng
        if obj_ng < best_obj:
            best_obj = obj_ng
            best_mc = mc_ng
            best_seq = a_ng.copy()
            print(f"  Nesterov: obj={obj_ng:.8f} **")
        else:
            print(f"  Nesterov: obj={obj_ng:.8f} (no improvement)")

    # === Phase 3: LP Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 3: LP Polish (budget={min(remaining*0.25, 200):.0f}s) ===")
        alloc = min(remaining * 0.2, 180.0)
        a_lp, mc_lp = lp_iterate(best_seq, n_iters=80, time_budget=alloc, verbose=True)
        obj_lp = 2.0 * n_best * mc_lp
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_mc = mc_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={obj_lp:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 4: Nesterov again (shorter) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 4: Nesterov Round 2 ===")
        alloc = min(remaining * 0.1, 80.0)
        a_ng2, mc_ng2 = nesterov_subgrad(best_seq, time_budget=alloc, verbose=False)
        obj_ng2 = 2.0 * n_best * mc_ng2
        if obj_ng2 < best_obj:
            best_obj = obj_ng2
            best_mc = mc_ng2
            best_seq = a_ng2.copy()
            print(f"  Nesterov2: obj={obj_ng2:.8f} **")
        else:
            print(f"  Nesterov2: obj={obj_ng2:.8f}")

    # === Phase 5: n Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 5: n Exploration ===")
        n_cur = len(best_seq)
        # Try nearby n values and some far ones
        n_candidates = []
        for dn in range(-300, 501, 50):
            n_try = n_cur + dn
            if 300 <= n_try <= 2500 and abs(dn) >= 30:
                n_candidates.append(n_try)
        # Also add some strategic values
        for n_strat in [400, 500, 600, 700, 900, 1000, 1200, 1500, 1800, 2000]:
            if 300 <= n_strat <= 2500 and abs(n_strat - n_cur) >= 50:
                n_candidates.append(n_strat)
        n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - n_cur))
        n_candidates = n_candidates[:8]
        print(f"  Exploring n: {n_candidates}")

        for n_try in n_candidates:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            a_try = interpolate_seq(best_seq, n_try)
            if a_try is None:
                continue
            # Quick subgradient refinement
            alloc = min(remaining * 0.25, 60.0)
            a_try2, mc_try = nesterov_subgrad(a_try, time_budget=alloc, verbose=False)
            obj_try = 2.0 * n_try * mc_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_mc = mc_try
                best_seq = a_try2.copy()
                n_cur = n_try
                print(f"  ** n={n_try}: obj={obj_try:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: LP Polish on best n ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 6: Final LP Polish ===")
        alloc = min(remaining * 0.2, 100.0)
        a_final, mc_final = lp_iterate(best_seq, n_iters=50, time_budget=alloc, verbose=True)
        obj_final = 2.0 * len(best_seq) * mc_final
        if obj_final < best_obj:
            best_obj = obj_final
            best_mc = mc_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={obj_final:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # === Phase 7: Final squeeze (short Nesterov + short LP) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 7: Final Squeeze ===")
        alloc_sg = min(remaining * 0.3, 40.0)
        a_sq, mc_sq = nesterov_subgrad(best_seq, time_budget=alloc_sg, verbose=False)
        obj_sq = 2.0 * len(best_seq) * mc_sq
        if obj_sq < best_obj:
            best_obj = obj_sq
            best_seq = a_sq.copy()
            print(f"  Squeeze SG: obj={obj_sq:.8f} **")
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            alloc_lp = min(remaining2 * 0.4, 40.0)
            a_sq2, mc_sq2 = lp_iterate(best_seq, n_iters=20, time_budget=alloc_lp, verbose=False)
            obj_sq2 = 2.0 * len(best_seq) * mc_sq2
            if obj_sq2 < best_obj:
                best_obj = obj_sq2
                best_seq = a_sq2.copy()
                print(f"  Squeeze LP: obj={obj_sq2:.8f} **")

    # Try diverse restarts if time permits
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 8: Diverse Restarts ===")
        n_cur = len(best_seq)
        for r in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            mode = (r + 3) % 7
            a_r = diverse_init(n_cur, rng, mode=mode)
            # Blend with best
            blend = 0.3 + 0.1 * r
            a_r = (1.0 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
            alloc = min(remaining * 0.3, 40.0)
            a_r2, mc_r = nesterov_subgrad(a_r, time_budget=alloc, verbose=False)
            obj_r = 2.0 * n_cur * mc_r
            if obj_r < best_obj:
                best_obj = obj_r
                best_seq = a_r2.copy()
                print(f"  Restart {r} mode={mode}: obj={obj_r:.8f} **")
            else:
                print(f"  Restart {r} mode={mode}: obj={obj_r:.8f}")

    # Final validation
    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(500) / 500.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 200

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
