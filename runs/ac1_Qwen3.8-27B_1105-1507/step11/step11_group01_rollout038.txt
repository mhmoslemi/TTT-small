Key improvements: (1) vectorized CD with O(n) per-coordinate updates,

import numpy as np

def fft_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def seq_obj(a):
    """Objective = 2n * max(conv(a,a)) assuming a is normalized (sum=1)."""
    n = len(a)
    if n == 0:
        return np.inf
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    a_n = a / S
    return 2.0 * n * float(np.max(fft_conv(a_n)))


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def cd_optimize(a, t_budget, seed=42, verbose=False):
    """
    Fast coordinate descent with golden section search.
    Uses incremental convolution updates for O(n) per coordinate.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    conv = fft_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0
    stall = 0
    prev_best = best_obj

    while time.monotonic() - t0 < t_budget:
        # Periodic FFT recomputation to avoid drift
        if sweep > 0 and sweep % 12 == 0:
            S = float(np.sum(a))
            if S > 1e-10:
                a = a / S
                conv = fft_conv(a)
                obj = 2.0 * n * float(np.max(conv))
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        order = rng.permutation(n)
        S = float(np.sum(a))
        sweep_improved = False

        for i in order:
            if time.monotonic() - t0 >= t_budget:
                break
            v = a[i]
            S_o = S - v
            if v < 1e-15 and S_o < 1e-10:
                continue

            # Local 1D function: objective if a[i] = x
            # conv changes at positions [i, i+n-1]:
            #   linear: conv[k] + 2*(x-v)*a[k-i] for k != 2i
            #   quad:   conv[2i] - v^2 + x^2
            # In slice [i, i+n): position i corresponds to global 2i

            # Quick candidate scan
            scale = max(v, 1.0 / n, 1e-8)
            cands = [0.0, v * 0.25, v * 0.5, v * 0.75, v,
                     v * 1.25, v * 1.5, v * 2.0, v * 3.0,
                     0.5 / n, 1.0 / n, 2.0 / n, 5.0 / n]
            cands = sorted(set(max(0.0, float(c)) for c in cands))

            best_x = v
            best_f = 2.0 * n * float(np.max(conv)) / (S * S)

            for x in cands:
                d = x - v
                if d == 0.0:
                    continue

                new_slice[i] += d * d

                    a = a / S_now
                    conv = fft_conv(a)

    S_f = float(np.sum(a))

        obj_f = 2.0 * n * float(np.max(fft_conv(a_f)))

def sa_perturb(a, t_budget, seed=42, verbose=False):
    """Simulated annealing to escape local minima."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(fft_conv(a)))
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    T = 0.02
    T_min = 0.001
    it = 0

    while time.monotonic() - t0 < t_budget and T > T_min:
        # Random perturbation
        sigma = T * np.sqrt(n)
        pert = a * np.exp(rng.normal(0, sigma, n))
        pert = np.maximum(pert, 0.0)
        S_p = float(np.sum(pert))
        if S_p < 1e-10:
            T *= 0.95
            continue
        pert = pert / S_p
        obj_p = 2.0 * n * float(np.max(fft_conv(pert)))

        delta = obj_p - seq_obj(a)
        if delta < 0 or rng.random() < np.exp(-max(delta, -100) / max(T, 1e-10)):
            a = pert
            if obj_p < best_obj:
                best_obj = obj_p
                best_a = a.copy()

        T *= 0.98
        it += 1
        if verbose and it % 500 == 0:
            print(f"    SA it{it}: T={T:.5f}, best={best_obj:.8f}")

    return best_a


def lp_refine(a, t_budget, seed=42):
    """LP-based refinement: solve min-peak LP and blend."""
    if not HAS_CVXPY:
        return a
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(fft_conv(a)))
    t0 = time.monotonic()

    # Build constraint matrix A (2n-1 x n)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]

    for _ in range(3):
        if time.monotonic() - t0 >= t_budget:
            break
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        solved = False
        for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    gs = float(np.sum(g_val))
                    if gs > 1e-10:
                        g_val = g_val / gs
                        solved = True
                    break
            except Exception:
                continue
        if not solved:
            break

        # Try blending
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(fft_conv(a_try)))
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initial sequences."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x * x + 0.01)
        a = a / a.sum()
    elif mode == 3:
        a = np.zeros(n)
        w = n // 4
        centers = [n // 4, 3 * n // 4]
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * (w // 2) ** 2))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(2, 15, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 4)
        a = a / a.sum()
    elif mode == 6:
        # Triangular
        a = 1.0 - np.abs(2.0 * np.arange(n) / n - 1.0)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

        obj_try = seq_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 3: CD on best found (15% time) ===
    print(f"\n=== Phase 3: CD Polish (n={n_best}) ===")
    rem = deadline - time.monotonic()
    p3_time = min(rem * 0.15, 200.0)
    if p3_time > 15:
        a3 = cd_optimize(best_a, t_budget=p3_time, seed=seed + 100, verbose=True)
        obj3 = seq_obj(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            print(f"  Polish: obj={best_obj:.8f} **")
        else:
            print(f"  Polish: obj={obj3:.8f}")

    # === Phase 4: LP Refinement (10% time) ===
    print(f"\n=== Phase 4: LP Refinement ===")
    rem = deadline - time.monotonic()
    p4_time = min(rem * 0.10, 120.0)
    if p4_time > 10 and HAS_CVXPY:
        a4 = lp_refine(best_a, t_budget=p4_time, seed=seed + 200)
        obj4 = seq_obj(a4)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()
            print(f"  LP: obj={best_obj:.8f} **")
        else:
            print(f"  LP: obj={obj4:.8f}")
        # CD after LP
        rem = deadline - time.monotonic()
        if rem > 20:
            a4b = cd_optimize(best_a, t_budget=min(rem * 0.3, 60), seed=seed + 300)
            obj4b = seq_obj(a4b)
            if obj4b < best_obj:
                best_obj = obj4b
                best_a = a4b.copy()
                print(f"  CD after LP: obj={best_obj:.8f} **")
    elif not HAS_CVXPY:
        print("  cvxpy not available, skipping LP")

    # === Phase 5: SA Escape + CD (remaining time) ===
    print(f"\n=== Phase 5: SA + CD ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Try SA perturbation
        sa_time = min(rem * 0.3, 80.0)
        if sa_time > 10:
            a5 = sa_perturb(best_a, t_budget=sa_time, seed=seed + 400)
            obj5 = seq_obj(a5)
            if obj5 < best_obj:
                best_obj = obj5
                best_a = a5.copy()
                print(f"  SA: obj={best_obj:.8f} **")
            else:
                print(f"  SA: obj={obj5:.8f}")

        # Final CD
        rem = deadline - time.monotonic()
        if rem > 15:
            a6 = cd_optimize(best_a, t_budget=rem * 0.7, seed=seed + 500, verbose=True)
            obj6 = seq_obj(a6)
            if obj6 < best_obj:
                best_obj = obj6
                best_a = a6.copy()
                print(f"  Final CD: obj={best_obj:.8f} **")
            else:
                print(f"  Final CD: obj={obj6:.8f}")

    # === Phase 6: Diverse restarts if time remains ===
    rem = deadline - time.monotonic()
    if rem > 40:
        print(f"\n=== Phase 6: Diverse Restarts ===")
        for mode in [1, 3, 5]:
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            a_r = diverse_init(n_best, rng, mode=mode)
            blend = 0.4
            a_r = (1 - blend) * a_r + blend * best_a
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
            alloc = min(rem * 0.3, 50)
            a_r = cd_optimize(a_r, t_budget=alloc, seed=seed + 1000 + mode)
            obj_r = seq_obj(a_r)
            if obj_r < best_obj:
                best_obj = obj_r
                best_a = a_r.copy()
                print(f"  Restart mode {mode}: obj={best_obj:.8f} **")
            else:
                print(f"  Restart mode {mode}: obj={obj_r:.8f}")

    # Final validation
    best_a = np.maximum(best_a, 0.0)
    result = [float(min(1000.0, x)) for x in best_a]

    val = evaluate_sequence(result)
    if val < np.inf:
        print(f"  Validated: {val:.8f}")
        if val < best_obj:
            best_obj = val
    else:
        print("  WARNING: evaluation returned inf, using fallback")
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - t_start
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")
    return result
