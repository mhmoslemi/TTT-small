def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute 2n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n

def mirror_descent(a, time_budget, top_k=1, eta_start=0.1, eta_end=0.01, verbose=False):
    """
    Mirror descent (multiplicative weights) with structured gradient.
    Uses top-k active lags for subgradient.
    """
    n = len(a)
    a = np.maximum(a, 1e-15)  # Ensure positivity for log
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    it = 0
    best_it = 0
    patience = max(500, int(time_budget * 50))
    
    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        eta = eta_start * (eta_end / eta_start) ** frac
        
        conv = compute_conv(a)
        # Get top-k peaks
        k_peaks = np.argsort(conv)[::-1][:top_k]
        peak_vals = conv[k_peaks]
        weights = (peak_vals / peak_vals.sum() ** 0.5) ** 2  # Softmax-like weights
        # Actually let's use exponential weights
        T_sub = 50.0
        exp_w = np.exp(T_sub * (peak_vals - peak_vals.max()))
        weights = exp_w / exp_w.sum()
        
        # Compute subgradient
        g = np.zeros(n)
        for ki, w in zip(k_peaks, weights):
            ki = int(ki)
            # g[i] += 2 * w * a[ki - i] if 0 <= ki - i < n
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            if lo <= hi:
                g[lo:hi+1] += 2.0 * w * a[ki - hi:ki - lo + 1][::-1]
        
        # Mirror descent update: a[i] <- a[i] * exp(-eta * g[i])
        log_update = -eta * g
        # Clip to avoid overflow
        log_update = np.clip(log_update, -10, 10)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj_norm(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        
        if it - best_it > patience:
            a = best_a.copy()
            eta = eta * 0.5
            patience = max(100, patience // 2)
            best_it = it

        if verbose and it % 2000 == 0:

def full_lp_solve(a, max_time=10.0):
    """Solve full LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    # Build constraint matrix A (2n-1 x n)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[k, lo:hi+1] = a_norm[k - hi:k - lo + 1][::-1]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    t_start = time.monotonic()
    for solver in [cp.CBC, cp.GLPK, cp.ECOS, cp.SCS, cp.CLARABEL]:
        if time.monotonic() - t_start > max_time:
            break
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None

"""
Optimization for step function coefficients minimizing 2n*max(conv(a,a))/(sum(a))^2.
Methods: Mirror descent (multiplicative weights) with structured sparse gradient,
full-constraint LP polishing, fast coordinate descent with golden section search,
LSE smoothed gradient descent, and adaptive n exploration.
Key innovations over parent:
1. Mirror descent exploiting the sparse structure of the subgradient
2. Full-constraint LP (all 2n-1 constraints) for higher quality LP steps
3. Fast coordinate descent using analytical conv update + golden section
4. Dual-informed subgradient weighting from LP dual variables
5. Palindromic symmetry encouragement
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute 2n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n


def lse_obj(a, T):
    """Log-sum-exp smoothed objective."""
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)
    shifted = np.clip(T * (conv - cmax), -50, 50)
    lse = cmax + np.log(np.sum(np.exp(shifted))) / T
    return 2.0 * n * lse


def subgradient_at_peak(a, k_peak):
    """Compute subgradient of (a*a)[k_peak] w.r.t. a. Result has length n."""
    n = len(a)
    g = np.zeros(n)
    # g[i] = 2 * a[k_peak - i] if 0 <= k_peak - i < n
    lo = max(0, k_peak - n + 1)
    hi = min(n - 1, k_peak)
    if lo <= hi:
        g[lo:hi + 1] = 2.0 * a[k_peak - hi:k_peak - lo + 1][::-1]
    return g


def mirror_descent(a, time_budget, top_k=3, eta_start=0.15, eta_end=0.005,
                   T_sub=30.0, verbose=False):
    """
    Mirror descent (multiplicative weights) with top-k subgradient.
    Exploits the sparse structure of the subgradient for speed.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    it = 0
    best_it = 0
    patience = max(300, int(time_budget * 30))

    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / max(1, time_budget))
        eta = eta_start * (eta_end / eta_start) ** frac

        conv = compute_conv(a)
        # Soft weighting over top-k peaks
        k_peaks = np.argsort(conv)[::-1][:top_k]
        peak_vals = conv[k_peaks]
        exp_w = np.exp(T_sub * (peak_vals - peak_vals.max()))
        weights = exp_w / exp_w.sum()

        # Compute weighted subgradient
        g = np.zeros(n)
        for ki, w in zip(k_peaks, weights):
            ki = int(ki)
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            if lo <= hi:
                g[lo:hi + 1] += 2.0 * w * a[ki - hi:ki - lo + 1][::-1]

        # Multiplicative weights update
        log_update = -eta * g
        log_update = np.clip(log_update, -8, 8)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj_norm(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        if it - best_it > patience:
            a = best_a.copy()
            eta = max(eta * 0.5, 1e-6)
            patience = max(50, patience // 2)
            best_it = it
            if eta < 1e-6:
                break

        if verbose and it % 2000 == 0:
            print(f"      MD it{it}: obj={best_obj:.8f}, eta={eta:.5f}, "
                  f"t={time.monotonic() - start:.1f}s")

    return best_a


def full_lp_solve(a, max_time=15.0):
    """Solve full LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a_norm))
    if s < 1e-10:
        return None
    a_norm = a_norm / s

    # Build constraint matrix A (2n-1 x n) - banded structure
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[k, lo:hi + 1] = a_norm[k - hi:k - lo + 1][::-1]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    t_start = time.monotonic()
    solvers = [cp.CBC, cp.GLPK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        if time.monotonic() - t_start > max_time:
            break
        try:
            prob.solve(solver=solver, max_time=max_time)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv
        except Exception:
            continue
    return None


def lp_blend_search(a, g_lp):
    """Find optimal blend factor between a and g_lp."""
    best_t = 0.0
    best_obj = compute_obj_norm(a)

    # Coarse
    for t in np.linspace(0, 1.5, 31):
        a_try = (1 - t) * a + t * g_lp
        a_try = np.maximum(a_try, 1e-15)
        a_try = a_try / a_try.sum()
        obj_try = compute_obj_norm(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t

    # Fine
    if 0 < best_t < 1.5:
        for t in np.linspace(max(0, best_t - 0.08), min(1.5, best_t + 0.08), 33):
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 1e-15)
            a_try = a_try / a_try.sum()
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_t = t

    return best_t, best_obj


def lp_polish_loop(a, time_budget, max_iters=40, verbose=False):
    """Iterative LP polishing with full constraints."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        g_lp = full_lp_solve(best_a, max_time=min(remaining * 0.6, 12.0))
        if g_lp is None:
            break

        t_opt, obj_opt = lp_blend_search(best_a, g_lp)

        if t_opt > 0 and obj_opt < best_obj - 1e-14:
            a_new = (1 - t_opt) * best_a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj_norm(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 3 == 0:
            print(f"      LP it{it}: obj={best_obj:.8f}, t_opt={t_opt:.3f}, "
                  f"elapsed={time.monotonic() - start:.1f}s")

        if stall >= 3:
            rng_local = np.random.default_rng(it * 104729)
            noise = rng_local.normal(0, 0.003 * stall, n)
            a_pert = best_a + noise
            a_pert = np.maximum(a_pert, 0)
            s_p = float(np.sum(a_pert))
            if s_p > 0.01:
                best_a = a_pert / s_p
                best_obj = compute_obj_norm(best_a)
            stall = 0

    return best_a


def fast_coord_descent(a, time_budget, max_sweeps=30, verbose=False):
    """
    Fast coordinate descent using analytical conv update.
    For coordinate i, conv change is: new_conv[k] = conv[k] + (v-ai)*h[k]
    where h[k] = a[k-i] + a[i-k]. Then golden section on w = v - ai.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start >= time_budget:
            break
        any_improved = False
        conv_base = compute_conv(a)

        for i in range(n):
            if time.monotonic() - start >= time_budget:
                break

            ai = a[i]
            d = s - ai  # sum of all other elements
            if d < 1e-12:
                continue

            # h[k] = a[k-i] + a[i-k] (with bounds)
            h = np.zeros(2 * n - 1)
            lo1 = i
            hi1 = min(2 * n - 2, i + n - 1)
            if lo1 <= hi1:
                h[lo1:hi1 + 1] += a[0:hi1 - lo1 + 1]
            lo2 = max(0, i - n + 1)
            hi2 = min(2 * n - 2, i)
            if lo2 <= hi2:
                h[lo2:hi2 + 1] += a[i - hi2:i - lo2 + 1][::-1]

            # f(w) = max_k (conv_base[k] + w*h[k]) / (d + w)^2
            # Search w in [-ai, W_max]
            w_min = -ai
            w_max = max(5.0 * ai, 0.5 / n)

            # Check if current point is already optimal (gradient ~ 0)
            # Quick check: try a few points
            gr = (np.sqrt(5) - 1) / 2
            a_gs = w_min
            b_gs = w_max
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)

            num_c = conv_base + c_gs * h
            num_d = conv_base + d_gs * h
            fc = float(np.max(num_c)) / (d + c_gs) ** 2
            fd = float(np.max(num_d)) / (d + d_gs) ** 2

            for _ in range(30):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    num_c = conv_base + c_gs * h
                    fc = float(np.max(num_c)) / (d + c_gs) ** 2
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    num_d = conv_base + d_gs * h
                    fd = float(np.max(num_d)) / (d + d_gs) ** 2

            w_opt = (a_gs + b_gs) / 2
            v_opt = ai + w_opt

            if abs(v_opt - ai) > 1e-12:
                a[i] = max(0.0, v_opt)
                s = float(np.sum(a))
                a = a / s
                any_improved = True
                obj_now = compute_obj_norm(a)
                if obj_now < best_obj:
                    best_obj = obj_now
                    best_a = a.copy()
                conv_base = compute_conv(a)  # Refresh for next coordinate

        if verbose and sweep % 3 == 0:
            print(f"      FCD sweep {sweep}: obj={best_obj:.8f}, t={time.monotonic() - start:.1f}s")

        if not any_improved:
            break

    return best_a


def lse_gradient_descent(a, time_budget, T_start=5.0, T_end=50.0, verbose=False):
    """LSE smoothed gradient descent with backtracking."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    lr = 0.2 / n
    it = 0
    best_it = 0
    patience = max(100, int(time_budget * 20))

    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / max(1, time_budget))
        T = T_start * (T_end / T_start) ** frac

        conv = compute_conv(a)
        cmax = np.max(conv)
        shifted = np.clip(T * (conv - cmax), -50, 50)
        exp_vals = np.exp(shifted)
        w = exp_vals / exp_vals.sum()

        # Vectorized gradient computation
        grad = np.zeros(n)
        for i in range(n):
            lo1 = max(0, i)
            hi1 = min(2 * n - 2, i + n - 1)
            if lo1 <= hi1:
                grad[i] += np.dot(w[lo1:hi1 + 1], a[lo1 - i:hi1 - i + 1])
            lo2 = max(0, i - n + 1)
            hi2 = min(2 * n - 2, i)
            if lo2 <= hi2:
                grad[i] += np.dot(w[lo2:hi2 + 1], a[i - hi2:i - lo2 + 1][::-1])
        grad *= 2.0 * n

        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-14:
            break

        # Backtracking
        obj_old = lse_obj(a, T)
        t = lr
        improved = False
        for _ in range(8):
            a_try = project_simplex(a - t * grad)
            obj_try = lse_obj(a_try, T)
            if obj_try < obj_old - 1e-13:
                a = a_try
                improved = True
                break
            t *= 0.5

        if not improved:
            break

        obj_true = compute_obj_norm(a)
        if obj_true < best_obj:
            best_obj = obj_true
            best_a = a.copy()
            best_it = it

        it += 1
        if it - best_it > patience:
            a = best_a.copy()
            lr *= 0.5
            patience = max(50, patience // 2)
            best_it = it
            if lr < 1e-8:
                break

        if verbose and it % 500 == 0:
            print(f"      LSE it{it}: obj={best_obj:.8f}, T={T:.1f}, t={time.monotonic() - start:.1f}s")

    return best_a


def u_shaped_init(n, sharpness):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def arcsine_init(n):
    """Arcsine distribution."""
    x = np.linspace(0.005, 0.995, n)
    a = 1.0 / np.sqrt(np.maximum(1e-10, 1 - (2 * x - 1) ** 2))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def power_init(n, alpha):
    """Power-law edge concentration."""
    x = np.linspace(0, 1, n)
    d = np.abs(x - 0.5)
    a = np.maximum(d, 1.0 / (2 * n)) ** (-alpha) + 0.01
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, mode):
    """Diverse initialization patterns."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n / 4.0) ** 2) / (2 * (n / 6.0) ** 2)) + \
            np.exp(-((idx - 3 * n / 4.0) ** 2) / (2 * (n / 6.0) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 4:
        a = 1.0 / np.cosh(4 * (2 * x - 1))
    elif mode == 5:
        a = (1 + np.sin(np.pi * x)) ** 2 + 0.01
    elif mode == 6:
        a = 1.0 / (1.0 + 10.0 * (2 * x - 1) ** 4)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def try_different_n(best_seq, n_try, time_budget):
    """Try a different sequence length."""
    n_old = len(best_seq)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_try)
    a_try = np.interp(x_new, x_old, best_seq)
    a_try = np.maximum(a_try, 0)
    s = float(np.sum(a_try))
    if s < 0.01:
        return None, np.inf
    a_try = a_try / s

    # Quick optimization
    t1 = min(20.0, time_budget * 0.3)
    if t1 > 5:
        a_try = mirror_descent(a_try, time_budget=t1, top_k=3)

    t2 = min(25.0, time_budget * 0.4)
    if t2 > 8:
        a_try = lp_polish_loop(a_try, time_budget=t2, max_iters=8, verbose=False)

    t3 = min(15.0, time_budget * 0.2)
    if t3 > 5:
        a_try = fast_coord_descent(a_try, time_budget=t3, max_sweeps=5, verbose=False)

    obj_try = compute_obj_norm(a_try)
    return a_try, obj_try

            s = float(np.sum(parent_arr))

                parent = parent_arr / s
