"""
Step function optimization using soft-max gradient descent, 
LP polishing with constraint tracking, and mirror descent.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a))) / (s * s)

def compute_obj_unit(a):
    """Assumes sum(a) = 1."""
    n = len(a)
    if n == 0:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a)))

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    valid = u * rho > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    idx = np.nonzero(valid)[0][-1]
    theta = (cssv[idx] - 1) / (idx + 1.0)
    return np.maximum(v - theta, 0)

def cross_correlate(a, b):
    """Compute sum_j a_j * b_{i+j} for i = 0..n-1. O(n log n)."""
    n = len(a)
    m = len(b)
    a_rev = a[::-1]
    size = 1
    while size < n + m - 1:
        size <<= 1
    A = rfft(a_rev, size)
    B = rfft(b, size)
    full = irfft(A * B, size)
    return full[n - 1:n - 1 + n]

def soft_max_sgd(a, time_budget, beta_start, beta_end, lr_start, momentum, seed, verbose=False):
    a = project_simplex(a.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj_unit(a)
    v = np.zeros(n)
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    it = 0
    best_it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress
        lr = lr_start * (1.0 - 0.3 * progress)
        
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p_sum = float(np.sum(exp_vals))
        if p_sum < 1e-30:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / p_sum
        
        grad = 2.0 * cross_correlate(a, p)
        v = momentum * v + grad
        a_new = project_simplex(a - lr * v)
        obj_new = compute_obj_unit(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall > 80:
                v = np.zeros(n)
                pert = rng.normal(0, 0.003, n)
                a = project_simplex(best_a + pert)
                a_new = a
                obj_new = compute_obj_unit(a)
                stall = 0
            elif stall > 30:
                lr *= 0.8
        
        a = a_new
        it += 1
        
        if verbose and it % 500 == 0:
            print(f"    SGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, t={elapsed:.1f}s")
    
    return best_a, best_obj

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a, use_pdlp=False):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    import cvxpy as cp
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def line_search(a, g, max_t=1.5, n_iter=25):
    """Golden section search for optimal blend factor."""
    gr = (3.0 - np.sqrt(5.0)) / 2.0
    lo, hi = 0.0, max_t
    
    def obj_at(t):
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            return np.inf
        return compute_obj_unit(a_try / s)
    
    c = lo + gr * (hi - lo)
    d = lo + (1.0 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)
    
    for _ in range(n_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)
    
    t_best = c if fc < fd else d
    obj_best = min(fc, fd)
    
    for t_try in np.linspace(0.05, 1.5, 15):
        obj_try = obj_at(t_try)
        if obj_try < obj_best:
            obj_best = obj_try
            t_best = t_try
    
    return t_best, obj_best

            a_new = project_simplex(np.maximum(a_new, 0))

    return best_a, best_obj

def mirror_descent(a, time_budget, lr_start=0.01, lr_end=0.001, seed=42, verbose=False):
    """Mirror descent (multiplicative weights) on the simplex."""
    a = project_simplex(a.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj_unit(a)
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    it = 0
    best_it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        lr = lr_start * (lr_end / lr_start) ** progress
        beta = 10.0 + 30.0 * progress
        
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p_sum = float(np.sum(exp_vals))
        if p_sum < 1e-30:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / p_sum
        
        grad = 2.0 * cross_correlate(a, p)
        
        # Multiplicative update
        log_update = -lr * grad

def u_shaped_init(n, sharpness=3.0):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    return project_simplex(a)

def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 6:
        a = 1.0 / np.cosh(3.0 * (2 * x - 1))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

"""
Step function optimization combining soft-max gradient descent, aggressive LP polishing,
mirror descent (multiplicative weights), and adaptive length exploration.
Key improvements: longer LP polishing budget, mirror descent as complementary optimizer,
better restart strategies, and targeted length search.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_unit(a):
    n = len(a)
    if n == 0:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a)))


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a))) / (s * s)


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    valid = u * rho > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    idx = np.nonzero(valid)[0][-1]
    theta = (cssv[idx] - 1) / (idx + 1.0)
    return np.maximum(v - theta, 0)


def cross_correlate(a, b):
    n = len(a)
    m = len(b)
    a_rev = a[::-1]
    size = 1
    while size < n + m - 1:
        size <<= 1
    A = rfft(a_rev, size)
    B = rfft(b, size)
    full = irfft(A * B, size)
    return full[n - 1:n - 1 + n]


def get_softmax_probs(conv, beta):
    c_max = float(np.max(conv))
    exp_vals = np.exp(beta * (conv - c_max))
    p_sum = float(np.sum(exp_vals))
    if p_sum < 1e-30:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
        return p
    return exp_vals / p_sum


def soft_max_sgd(a, time_budget, beta_start, beta_end, lr_start, momentum, seed, verbose=False):
    a = project_simplex(a.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj_unit(a)
    v = np.zeros(n)
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    it = 0
    best_it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress
        lr = lr_start * (1.0 - 0.3 * progress)

        conv = compute_conv(a)
        p = get_softmax_probs(conv, beta)
        grad = 2.0 * cross_correlate(a, p)
        v = momentum * v + grad
        a_new = project_simplex(a - lr * v)
        obj_new = compute_obj_unit(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall > 80:
                v = np.zeros(n)
                pert = rng.normal(0, 0.003, n)
                a = project_simplex(best_a + pert)
                a_new = a
                stall = 0
            elif stall > 30:
                lr *= 0.85

        a = a_new
        it += 1

        if verbose and it % 500 == 0:
            print(f"    SGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, t={elapsed:.1f}s")

    return best_a, best_obj


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_topk(a, k=100):
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv = compute_conv(a_norm)
    k_indices = np.argsort(conv)[::-1][:k]

    A = np.zeros((k, n))
    for idx, ki in enumerate(k_indices):
        for j in range(n):
            r = ki - j
            if 0 <= r < n:
                A[idx, j] = a_norm[r]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, g, max_t=1.5, n_iter=25):
    gr = (3.0 - np.sqrt(5.0)) / 2.0
    lo, hi = 0.0, max_t

    def obj_at(t):
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            return np.inf
        return compute_obj_unit(a_try / s)

    c = lo + gr * (hi - lo)
    d = lo + (1.0 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    for _ in range(n_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)

    t_best = c if fc < fd else d
    obj_best = min(fc, fd)

    for t_try in np.linspace(0.02, 1.5, 20):
        obj_try = obj_at(t_try)
        if obj_try < obj_best:
            obj_best = obj_try
            t_best = t_try

    return t_best, obj_best


def lp_polish_loop(a, time_budget, max_iters=200, verbose=True, seed=42, use_topk=False):
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj_unit(a)
    start = time.monotonic()

    stall = 0

        if remaining < 5:
            break

        if g_lp is None:

        t_opt, obj_opt = line_search(a, g_lp, max_t=1.0, n_iter=20)
        elapsed = time.monotonic() - start

        improved = False
        if t_opt > 0.001 and obj_opt < best_obj - 1e-12:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp
            a_new = project_simplex(np.maximum(a_new, 0))

            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new

                improved = True

        else:
            stall += 1

            last_improve_obj = compute_obj_unit(a)

def mirror_descent(a, time_budget, lr_start, lr_end, seed, verbose=False):

    return best_a, best_obj


def u_shaped_init(n, sharpness=3.0, eps=0.05):
    x = np.linspace(0, 1, n)
    a = 1.0 / (eps + sharpness * (2.0 * x - 1.0) ** 2)

def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2.0 * x - 1.0) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2.0 * max(1, n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2.0 * max(1, n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4.0 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(4.0 * (2.0 * x - 1.0) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2.0 * x - 1.0)) ** 2
    elif mode == 6:
        a = 1.0 / np.cosh(3.0 * (2.0 * x - 1.0))
    elif mode == 7:
        a = np.zeros(n)
        a[0] = 1.0
        a[-1] = 1.0
        if n > 2:
            a[1] = 0.2
            a[-2] = 0.2
        if n > 4:
            a[1:-1] += 0.03
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Perturb {attempt} (scale={scale:.4f}): obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj_unit(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
