def proj_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def gradient_lse(a, conv, tau):
    """Gradient of LSE_tau(conv(a)) w.r.t. a, where LSE uses sharpness tau."""
    n = len(a)
    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(2 * n - 1)
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    # grad = 2 * (a[::-1] * p)[n-1:2n-1]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    c = irfft(rfft(a[::-1], size) * rfft(p, size), size)
    return 2.0 * c[n - 1:2 * n - 1]

def gradient_topk(a, conv, k=3):
    """Subgradient using top-k peaks."""
    n = len(a)
    idx = np.argsort(conv)[-k:]
    p = np.zeros(2 * n - 1)
    p[idx] = 1.0 / k
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    c = irfft(rfft(a[::-1], size) * rfft(p, size), size)
    return 2.0 * c[n - 1:2 * n - 1]

def proj_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def fista_optimize(a_init, time_budget=100.0, tau_start=50.0, tau_end=5000.0,
                   verbose=False):
    """FISTA on the simplex for minimizing 2n*LSE_tau(conv(a,a))."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    t = 1.0
    a_prev = a.copy()
    L = 2.0 * n * tau_start  # Initial Lipschitz estimate
    
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.1:
            break
        
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        
        # Adaptive temperature
        if progress < 0.3:
            tau = tau_start
        elif progress < 0.7:
            tau = tau_start + (tau_end * 0.3 - tau_start) * (progress - 0.3) / 0.4
        else:
            tau = tau_end * 0.3 + (tau_end - tau_end * 0.3) * (progress - 0.7) / 0.3
        
        # FISTA extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a + (t - 1.0) / t_new * (a - a_prev)
        y = proj_simplex(y)
        
        # Compute gradient at y
        conv_y = compute_conv(y)
        g = gradient_lse(y, conv_y, tau)
        
        # Update Lipschitz estimate
        L = max(L * 0.999, 2.0 * n * tau)
        
        # Gradient step with projection
        step = 1.0 / L
        a_new = proj_simplex(y - step * g)
        
        # Check convergence
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a_prev = a
        a = a_new
        t = t_new
        it += 1
        
        # Restart momentum if stuck
        if it - best_it > 5000:
            a_prev = a.copy()
            t = 1.0
        
        if verbose and it % 500 == 0:
            print(f"    FISTA it{it}: obj={best_obj:.8f}, tau={tau:.0f}, "
                  f"L={L:.1f}, time={time.monotonic()-start:.1f}s")
    
    return best_a

def peak_targeting(a, time_budget=100.0, verbose=False):
    """Iteratively target the peak lags of the autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    
    it = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break
        
        conv = compute_conv(a)
        top_k = 3
        idx = np.argsort(conv)[-top_k:]
        p = np.zeros(2 * n - 1)
        p[idx] = 1.0 / top_k
        
        # Gradient direction
        size = 1
        while size < 3 * n - 1:
            size <<= 1
        c = irfft(rfft(a[::-1], size) * rfft(p, size), size)
        g = 2.0 * c[n - 1:2 * n - 1]
        
        # Line search
        best_t = 0.0
        best_f = best_obj
        for t_try in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:
            a_try = proj_simplex(a - t_try * g)
            f_try = compute_obj(a_try)
            if f_try < best_f:
                best_f = f_try
                best_t = t_try
        
        if best_t > 0:
            a = proj_simplex(a - best_t * g)
            if best_f < best_obj - 1e-14:
                best_obj = best_f
                best_a = a.copy()
        else:
            # Perturb to escape
            a = proj_simplex(a + 0.01 * rng.standard_normal(n))
        
        it += 1
        if verbose and it % 50 == 0:
            print(f"    Peak it{it}: obj={best_obj:.8f}, time={time.monotonic()-start:.1f}s")
    
    return best_a

        return np.ones(n) / n
