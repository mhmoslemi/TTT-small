    """Project onto probability simplex."""
    n = len(v)

"""

Uses LP-based direction finding with active set management, cutting plane
refinement, multiplicative weights subgradient method, and length exploration.
Key improvements over previous: exact subgradients (no soft-max approx),

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.linalg import toeplitz


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return np.array([])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    valid = u * rho > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    idx = np.nonzero(valid)[0][-1]
    theta = (cssv[idx] - 1) / (idx + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that (A @ g)[k] = sum_i a[k-i]*g[i]."""
    n = len(a)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = a
    first_row = np.zeros(n)
    first_row[0] = a[0]
    A = toeplitz(first_col, first_row)
    return A


def lp_direction(a, M_override=None, time_budget=5.0):
    """
    Solve LP: max sum(g) s.t. conv(a, g)[k] <= M, g >= 0.
    Returns (g, M_opt) or (None, None) if failed.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    if M_override is None:
        conv_a = compute_conv(a_norm)
        M = float(np.max(conv_a))
    else:
        M = M_override

    A = build_conv_matrix(a_norm)

    g_var = cp.Variable(n, nonneg=True)
    constraints = [A @ g_var <= M, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, M
                return None, None
        except Exception:
            continue
    return None, None


def lp_direction_peak_targeted(a, n_peaks=20, time_budget=5.0):
    """
    Solve LP that targets reducing the peak convolution entries.
    Minimize weighted sum of conv(a,g) emphasizing peaks.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv_a = compute_conv(a_norm)
    M = float(np.max(conv_a))

    # Weight: emphasize top convolution entries
    sorted_idx = np.argsort(conv_a)[::-1]
    weights = np.zeros(2 * n - 1)
    for i in range(min(n_peaks, 2 * n - 1)):
        weights[sorted_idx[i]] = 1.0
    # Also add small weight to all
    weights += 0.01

    A = build_conv_matrix(a_norm)

    g_var = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(g_var) == 1, A @ g_var <= M * 1.01]
    prob = cp.Problem(cp.Minimize(weights @ (A @ g_var)), constraints)

    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def line_search_blend(a, g, time_budget=2.0):
    """Find optimal blend factor t in a + t*(g-a) to minimize objective."""
    def obj_at(t):
        if t < 0:
            t = 0.0
        a_try = a + t * (g - a)
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try)

    # Golden section search
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi
    lo, hi = 0.0, 1.5
    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    start = time.monotonic()
    for _ in range(25):
        if time.monotonic() - start >= time_budget:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)

    t_best = c if fc < fd else d
    obj_best = min(fc, fd)

    # Check some specific values
    for t_try in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.2, 1.5]:
        obj_try = obj_at(t_try)
        if obj_try < obj_best:
            obj_best = obj_try
            t_best = t_try

    return t_best, obj_best


def lp_refinement(a, time_budget=200.0, max_iters=200, verbose=True, seed=42):
    """
    LP-based refinement: iteratively find best direction and blend.
    Uses both standard LP and peak-targeted LP.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Alternate between standard and peak-targeted LP
        use_peak = (it % 3 == 2)

        if use_peak:
            g_lp = lp_direction_peak_targeted(a, n_peaks=min(30, n), time_budget=min(3.0, remaining * 0.1))
        else:
            g_lp, _ = lp_direction(a, time_budget=min(3.0, remaining * 0.1))

        if g_lp is None:
            stall += 1
            it += 1
            continue

        t_opt, obj_opt = line_search_blend(a, g_lp, time_budget=min(2.0, remaining * 0.05))

        if t_opt > 0 and obj_opt < best_obj - 1e-10:
            a_new = a + t_opt * (g_lp - a)
            a_new = project_simplex(np.maximum(a_new, 0))
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1

        if verbose and it % 10 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, stall={stall}, t={time.monotonic()-start:.1f}s")

        if stall >= 5:
            # Perturb and restart
            noise_scale = 0.01 * (stall - 3)
            noise = rng.normal(0, noise_scale, n)
            a = project_simplex(best_a + noise)
            stall = 0
            if stall >= 15:
                # More aggressive restart
                x = np.linspace(0, 1, n)
                a_div = 1.0 / (0.05 + 3.0 * (2 * x - 1) ** 2)
                a = project_simplex(0.4 * a_div + 0.6 * best_a)
                stall = 0

    return best_a, best_obj


def cutting_plane_refinement(a, time_budget=150.0, max_iters=40, verbose=True, seed=42):
    """
    Cutting plane method: accumulate linear under-estimators of the
    max-convolution objective and solve LPs with those constraints.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    cuts = []  # List of (gradient_vector, offset) representing grad^T x <= t + offset

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        current_obj = 2.0 * n * max_val
        if current_obj < best_obj:
            best_obj = current_obj
            best_a = a.copy()

        # Add cutting planes for top saturation entries
        sorted_idx = np.argsort(conv)[::-1]
        n_add = min(3, 2 * n - 1)
        for si in range(n_add):
            k = int(sorted_idx[si])
            # Subgradient of conv_k at a: grad_i = 2*a[k-i] if 0 <= k-i < n
            grad = np.zeros(n)
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            if j_max >= j_min:
                grad[k - j_max:k - j_min + 1] = 2.0 * a[j_min:j_max + 1][::-1]

            # Tangent: conv_k(a') >= conv_k(a) + grad^T (a' - a)
            # = grad^T a' - grad^T a + conv_k(a)
            # = grad^T a' - 2*conv_k(a) + conv_k(a)  [since grad^T a = 2*conv_k(a)]
            # = grad^T a' - conv_k(a)
            # So constraint: grad^T a' - conv_k(a) <= t
            # i.e., grad^T a' <= t + conv_k(a)
            offset = float(conv[k])
            cuts.append((grad, offset))

        # Keep only most recent cuts to bound LP size
        max_cuts = 80
        if len(cuts) > max_cuts:
            cuts = cuts[-max_cuts:]

        # Solve LP: minimize t s.t. grad_j^T a <= t + offset_j, sum(a)=1, a>=0
        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]
        for grad_j, offset_j in cuts:
            constraints.append(grad_j @ a_var <= t_var + offset_j)

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        solved = False
        for solver in [cp.CLARABEL, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if a_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    a_new = np.maximum(a_var.value, 0)
                    s = float(np.sum(a_new))
                    if s > 1e-10:
                        a = a_new / s
                        solved = True
                        break
            except Exception:
                continue

        if not solved:
            # Fallback: small perturbation
            noise = rng.normal(0, 0.005, n)
            a = project_simplex(a + noise)
            continue

        obj_new = compute_obj(a)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()

        if verbose and it % 5 == 0:
            print(f"    CP it{it}: obj={best_obj:.8f}, cuts={len(cuts)}, t={time.monotonic()-start:.1f}s")

    return best_a, best_obj


def multiplicative_weights_opt(a, time_budget=100.0, lr=0.05, verbose=True, seed=42):
    """
    Multiplicative weights / mirror descent with exact subgradients.
    Uses the entropy-regularized simplex update.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    it = 0
    while time.monotonic() - start < time_budget and it < 20000:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        current_obj = 2.0 * n * max_val
        if current_obj < best_obj:
            best_obj = current_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 2000:
            noise = rng.normal(0, 0.01, n)
            a = project_simplex(best_a + noise)
            stall = 0

        # Compute subgradient using top few max entries (soft version)
        # For better stability, use weighted combination of top entries
        top_k = np.where(conv >= max_val * 0.95)[0]
        if len(top_k) == 0:
            top_k = np.array([np.argmax(conv)])

        grad = np.zeros(n)
        for k in top_k[:10]:  # Use at most 10 top entries
            j_min = max(0, int(k) - (n - 1))
            j_max = min(int(k), n - 1)
            if j_max >= j_min:
                grad[k - j_max:k - j_min + 1] += 2.0 * a[j_min:j_max + 1][::-1]

        # Normalize gradient
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm > 1e-10:
            grad = grad / grad_norm

        # Project onto tangent space (for simplex)
        grad = grad - np.mean(grad)

        # Adaptive learning rate
        progress = (time.monotonic() - start) / time_budget
        lr_adapt = lr * (1.0 - 0.5 * progress)

        # Multiplicative update (mirror descent on simplex)
        a = a * np.exp(-lr_adapt * grad)
        a = project_simplex(a)

        if verbose and it % 500 == 0:
            print(f"    MW it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

        it += 1

    return best_a, best_obj


def coordinate_descent(a, time_budget=60.0, max_sweeps=5, verbose=False):
    """
    Coordinate descent: for each a[i], find optimal value given others fixed.
    Uses the fact that the objective is convex in each coordinate.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        for i in range(n):
            if time.monotonic() - start > time_budget:
                    break

            # Try small perturbations of a[i]
            base = a.copy()
            best_local = float(np.sum(a))
            best_val = base[i]

            # Binary search for optimal a[i] in [0, sum(a)]
            lo, hi = 0.0, float(np.sum(a)) * 0.5
            # Evaluate at a few points
            candidates = [0.0, base[i] * 0.5, base[i] * 0.7, base[i],
                          base[i] * 1.3, base[i] * 1.5, hi]
            best_cand_obj = compute_obj(base)

            for val_try in candidates:
                a_try = base.copy()
                a_try[i] = max(0, val_try)
                s = float(np.sum(a_try))
                if s < 0.01:
                    continue
                a_try = a_try / s
                obj_try = compute_obj(a_try)
                if obj_try < best_cand_obj:
                    best_cand_obj = obj_try
                    best_val = val_try
                    best_local = a_try.copy()

            if best_cand_obj < best_obj - 1e-10:
                a = best_local
                best_obj = best_cand_obj
                best_a = a.copy()

    return best_a, best_obj


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    return project_simplex(a)


def arcsine_init(n):
    """Arcsine distribution initialization."""
    x = np.linspace(0.001, 0.999, n)
    a = 1.0 / np.sqrt(1 - (2 * x - 1) ** 2)
    return project_simplex(a)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = (np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) +
             np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01)
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 6:
        a = 1.0 / np.cosh(3.0 * (2 * x - 1))
    elif mode == 7:
        a = np.zeros(n)
        a[0] = 1.0
        a[-1] = 1.0
        a[1:min(3, n)] = 0.3
        a[max(0, n - 4):n - 1] = 0.3
        if n > 4:
            a[1:-1] += 0.05
    elif mode == 8:
        # Beta distribution with high edge concentration
        a = x ** 0.3 + (1 - x) ** 0.3
    elif mode == 9:
        # Double U-shape
        a = 1.0 / (0.1 + 4.0 * (4 * x - 3) ** 2 * (4 * x + 1) ** 2) + 0.01
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 3: Multiplicative Weights, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.20, 200.0)
        a3, obj3 = multiplicative_weights_opt(best_seq, time_budget=alloc, lr=0.03, verbose=True, seed=seed + 2)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj3:.8f} (no improvement)")

    # === Phase 4: More LP from current best (10% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: LP Round 2, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.15, 150.0)
        a4, obj4 = lp_refinement(best_seq, time_budget=alloc, max_iters=100, verbose=True, seed=seed + 10)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4: obj={best_score:.8f} **")
        else:
            print(f"  Phase4: obj={obj4:.8f} (no improvement)")

    # === Phase 5: Length Exploration (15% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 5: Length Exploration, time={remaining:.0f}s ===")
        n_best = len(best_seq)

        n_options = []
        for offset in [-200, -100, -50, 50, 100, 200, 300, 400, 500]:
            n_try = n_best + offset
            if 300 <= n_try <= 2500 and n_try not in n_options:
                n_options.append(n_try)

        # Also try some standard values
        for n_std in [500, 600, 700, 800, 900, 1000, 1200, 1500]:
            if 300 <= n_std <= 2500 and n_std not in n_options:
                n_options.append(n_std)

        n_options = sorted(n_options)[:6]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            # Interpolate to new length
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = project_simplex(np.maximum(a_try, 0))

            # Quick LP refinement
            alloc_lp = min(remaining * 0.3, 40.0)
            if alloc_lp > 10:
                a_try, obj_try = lp_refinement(a_try, time_budget=alloc_lp, max_iters=30, verbose=False, seed=seed + 100)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts (10% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 6: Diverse Restarts, time={remaining:.0f}s ===")
        n_cur = len(best_seq)

        for r in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            mode = (seed + r) % 10
            a_r = diverse_init(n_cur, rng, mode=mode)
            # Blend with current best
            blend = 0.5
            a_r = project_simplex((1 - blend) * a_r + blend * best_seq)

            alloc = min(remaining * 0.3, 60.0)
            if alloc > 15:
                a_r, obj_r = lp_refinement(a_r, time_budget=alloc, max_iters=30, verbose=False, seed=seed + 200 + r)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 7: Final Polish, time={remaining:.0f}s ===")

        # LP polish
        alloc1 = min(remaining * 0.5, 80.0)
        if alloc1 > 15:
            a_f, obj_f = lp_refinement(best_seq, time_budget=alloc1, max_iters=50, verbose=True, seed=seed + 500)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()

        # Cutting plane
        alloc2 = min(remaining * 0.4, 40.0)
        if alloc2 > 15:
            a_f2, obj_f2 = cutting_plane_refinement(best_seq, time_budget=alloc2, max_iters=20, verbose=True, seed=seed + 600)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final CP: obj={best_score:.8f} **")

    # === Final: Tiny perturbations + LP ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        for attempt in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            noise = rng.normal(0, 0.002, len(best_seq))
            a_p = project_simplex(best_seq + noise)
            alloc = min(remaining * 0.4, 15.0)
            if alloc > 5:
                a_p, obj_p = lp_refinement(a_p, time_budget=alloc, max_iters=10, verbose=False, seed=seed + 700 + attempt)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Final pert {attempt}: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
