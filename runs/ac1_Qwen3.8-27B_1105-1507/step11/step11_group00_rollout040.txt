"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a novel primal-dual multiplicative weights algorithm combined with
spectral analysis and coordinate descent polishing.
The key innovation is an explicit dual variable (distribution over lags) 
updated via multiplicative weights, which drives the primal variable 
(non-negative sequence) to flatten the autocorrelation.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_grad(a, p):
    """
    Compute gradient of sum_k p_k * conv(a,a)_k with respect to a.
    grad[i] = 2 * sum_j a[j] * p[i+j]
    p has length 2n-1.
    """
    n = len(a)
    # Build index matrix
    idx = np.arange(n)[:, None] + np.arange(n)[None, :]
    M = p[idx]
    return 2.0 * (M @ a)


def primal_dual_mirror_descent(a_init, time_budget=600.0, eta_d_init=1.0, 
                                eta_p_init=0.5, verbose=True):
    """
    Primal-dual multiplicative weights algorithm.
    Dual: distribution p over lags, updated proportionally to conv(a,a).
    Primal: non-negative sequence a, updated via mirror descent on dual gradient.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    # Initialize dual variable (uniform over lags)
    p = np.ones(2 * n - 1) / (2 * n - 1)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    it = 0
    stall = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.1:
            break

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Compute convolution
        conv = compute_conv(a)
        conv_max = float(np.max(conv))
        conv_mean = float(np.mean(conv))

        # Dual update: p_k <- p_k * exp(eta_d * (conv_k - conv_max))
        # This focuses p on the peaks
        # Adaptive eta_d: larger early (more exploration), smaller later (more exploitation)
        eta_d = eta_d_init * (1.0 + 2.0 * progress)
        
        shifted = eta_d * (conv - conv_max)
        exp_vals = np.exp(shifted)
        sum_exp = float(np.sum(exp_vals))
        
        if sum_exp < 1e-300:
            p_new = np.zeros(2 * n - 1)
            p_new[int(np.argmax(conv))] = 1.0
        else:
            p_new = exp_vals / sum_exp
        
        # Mix with previous p for stability
        alpha_p = 0.5 + 0.5 * progress  # More weight on new p as we progress
        p = alpha_p * p_new + (1.0 - alpha_p) * p
        p = p / np.sum(p)

        # Primal update: compute gradient
        grad = compute_grad(a, p)
        grad_max = float(np.max(np.abs(grad)))
        if grad_max < 1e-15:
            break

        # Adaptive step size for primal
        eta_p = eta_p_init / (1.0 + progress)
        # Scale by gradient magnitude
        eta_p_scaled = eta_p / grad_max
        
        # Mirror descent step
        exponent = -eta_p_scaled * grad
        exponent = np.clip(exponent, -50, 50)
        a_new = a * np.exp(exponent)
        s_new = float(np.sum(a_new))
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Evaluate
        obj_new = compute_obj(a_new)
        a = a_new

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall > 100:
                # Reheat: add noise to p and a
                p = p + 0.1 * rng.dirichlet(np.ones(2*n-1))
                p = p / np.sum(p)
                a = a + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                s_re = float(np.sum(a))
                if s_re > 1e-10:
                    a = a / s_re
                stall = 0

        it += 1
        if verbose and (it % 500 == 0 or it == 1 or it == 2):
            print(f"    PD it{it}: obj={best_obj:.8f}, conv_max={conv_max:.8f}, "
                  f"eta_d={eta_d:.2f}, eta_p={eta_p_scaled:.6f}, "
                  f"stall={stall}, time={elapsed:.0f}s")

    return best_a


def primal_dual_topk(a_init, time_budget=300.0, k=1, verbose=True):
    """
    Primal-dual with top-k subgradient (no softmax smoothing).
    Instead of exp(eta * conv), use uniform distribution over top-k peaks.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99999)

    it = 0
    stall = 0
    eta_p = 0.3

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.1:
            break

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        conv = compute_conv(a)
        
        # Top-k subgradient
        top_idx = np.argsort(conv)[-k:]
        p = np.zeros(2 * n - 1)
        p[top_idx] = 1.0 / k

        grad = compute_grad(a, p)
        grad_max = float(np.max(np.abs(grad)))
        if grad_max < 1e-15:
            break

        eta_p_scaled = eta_p / (grad_max * (1.0 + progress))
        
        exponent = -eta_p_scaled * grad
        exponent = np.clip(exponent, -50, 50)
        a_new = a * np.exp(exponent)
        s_new = float(np.sum(a_new))
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new
        obj_new = compute_obj(a)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                # Perturb
                a = best_a * np.exp(0.05 * rng.standard_normal(n))
                a = np.maximum(a, 0)
                s_pert = float(np.sum(a))
                if s_pert > 1e-10:
                    a = a / s_pert
                stall = 0
            # Decrease step size when stalling
            eta_p *= 0.95

        it += 1
        if verbose and (it % 200 == 0):
            print(f"    PD-topk it{it}: obj={best_obj:.8f}, eta_p={eta_p:.6f}, "
                  f"stall={stall}, time={elapsed:.0f}s")

    return best_a


def coordinate_descent_polish(a_init, time_budget=120.0, verbose=False):
    """
    Coordinate descent: for each coordinate, find the optimal value
    that minimizes the maximum convolution value.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    it = 0
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 1.0:
            break

        # Random coordinate ordering
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - start > time_budget - 0.5:
                break
            
            # Current conv
            # Try a few values of a[i]
            old_val = a[i]
            candidates = [old_val * 0.5, old_val * 0.8, old_val * 1.2, 
                         old_val * 1.5, old_val * 0.3, 0.0]
            # Also try values based on the "target" flat level
            conv = compute_conv(a)
            target = float(np.mean(conv))
            candidates.append(np.sqrt(target))
            
            best_cand = old_val
            best_obj_cand = best_obj  # current obj with a[i] = old_val
            
            for cand in candidates:
                if cand < -1e-10:
                    continue
                cand = max(0.0, cand)
                a[i] = cand
                s_new = float(np.sum(a))
                if s_new < 1e-10:
                    a[i] = old_val
                    continue
                obj_try = compute_obj(a)
                if obj_try < best_obj_cand:
                    best_obj_cand = obj_try
                    best_cand = cand
            
            a[i] = best_cand
            s_new = float(np.sum(a))
            if s_new > 1e-10:
                a = a / s_new
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
        
        it += 1
        if not improved and it > 20:
            break
        
        if verbose and it % 5 == 0:
            print(f"    CD it{it}: obj={best_obj:.8f}, time={elapsed:.0f}s")

    return best_a
