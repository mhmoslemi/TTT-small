        if it % 100 == 0 and verbose:
            elapsed = time.monotonic() - start
            print(f"        NLSE it{it}: obj={best_obj:.8f}, T={T:.1f}, lr={lr:.6f}, t={elapsed:.1f}s")

        if it - best_it > patience:
            a = best_a.copy()
            y = a.copy()
            v = np.zeros(n)
            lr *= 0.7
            T = min(T * 1.5, T_end * 2)
            patience = max(100, patience // 2)
            best_it = it

    return best_a


def standard_lse_descent(a, time_budget, T_start=5.0, T_end=60.0, verbose=False):
    """Standard LSE smoothed gradient descent with backtracking."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.3 / n
    it = 0
    best_it = 0
    patience = 300

    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** frac

        obj_lse, grad = lse_objective_and_gradient(a, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        t = lr
        improved = False
        for _ in range(12):
            a_try = project_simplex(a - t * grad)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14 or obj_try < obj_lse - 1e-10:
                a = a_try
                improved = True
                break
            t *= 0.5

        if not improved:
            # Try small random perturbation
            for _ in range(5):
                noise = np.random.randn(n) * 0.005
                a_try = project_simplex(a + noise)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    a = a_try
                    improved = True
                    break
            if not improved:
                break

        obj_true = compute_obj(a)
        if obj_true < best_obj:
            best_obj = obj_true
            best_a = a.copy()
            best_it = it
        lr = t * 1.05

        it += 1
        if it - best_it > patience:
            a = best_a.copy()
            lr = 0.1 / n
            patience = max(50, patience // 2)
            best_it = it

        if verbose and it % 200 == 0:
            print(f"        LSE it{it}: obj={best_obj:.8f}, T={T:.1f}, lr={t:.6f}, t={time.monotonic()-start:.1f}s")

    return best_a


def coordinate_descent_golden(a, time_budget, verbose=False):
    """Coordinate descent with golden section 1D search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    phi = (np.sqrt(5) - 1) / 2

    sweeps = 0
    while time.monotonic() - start < time_budget:
        sweeps += 1
        sweep_improved = False

        for i in range(n):
            if time.monotonic() - start >= time_budget:
                break

            ai_old = a[i]
            if ai_old < 1e-8:
                continue

            # Golden section search for optimal a[i]
            lo = 0.0
            hi = ai_old * 5.0 + 0.01

            def f_1d(x):
                a[i] = x
                s_new = float(np.sum(a))
                if s_new < 0.01:
                    a[i] = ai_old
                    return np.inf
                a_norm = a / s_new
                obj_val = compute_obj(a_norm)
                a[i] = ai_old  # restore
                return obj_val

            # Sample a few points first to get better bounds
            c = hi - phi * (hi - lo)
            d = lo + phi * (hi - lo)
            fc = f_1d(c)
            fd = f_1d(d)
            f0 = f_1d(ai_old)

            for _ in range(25):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - phi * (hi - lo)
                    fc = f_1d(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + phi * (hi - lo)
                    fd = f_1d(d)

            x_opt = (lo + hi) / 2
            f_opt = f_1d(x_opt)

            if f_opt < f0 - 1e-12:
                a[i] = x_opt
                s_new = float(np.sum(a))
                if s_new > 0.01:
                    a = a / s_new
                    sweep_improved = True

            a[i] = ai_old  # restore for next iteration check
            # Actually set to the best
            if f_opt < f0 - 1e-12:
                a[i] = x_opt
                s_new = float(np.sum(a))
                if s_new > 0.01:
                    a = a / s_new

        obj_now = compute_obj(a)
        if obj_now < best_obj:
            best_obj = obj_now
            best_a = a.copy()

        if verbose:
            print(f"        CD sweep {sweeps}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

        if not sweep_improved:
            break

    return best_a


def iterative_lp_optimize(a, time_budget, start_top_m, end_top_m, max_iters, verbose=False):
    """Iterative LP with golden section blend search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        frac = min(1.0, it / max(1, max_iters - 1))
        top_m = int(round(start_top_m + frac * (end_top_m - start_top_m)))
        top_m = max(10, min(top_m, 2 * n - 2))

        t0 = time.monotonic()
        g_lp = solve_lp(best_a, top_m=top_m)
        lp_time = time.monotonic() - t0

        if g_lp is None:
            break

        # Golden section blend search
        t_opt, obj_opt = golden_section_blend(best_a, g_lp, compute_obj, time_limit=min(3.0, remaining * 0.1))

        if t_opt > 0.01 and obj_opt < best_obj - 1e-12:
            a_new = (1 - t_opt) * best_a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 3 == 0:
            print(f"        LP it{it}: obj={best_obj:.8f}, top_m={top_m}, t={t_opt:.3f}, "
                  f"lp_time={lp_time:.2f}s, elapsed={time.monotonic()-start:.1f}s")

        if stall >= 3:
            # Perturb
            rng_local = np.random.default_rng(it * 104729)
            noise = rng_local.normal(0, 0.002 * (stall + 1), n)
            a_pert = best_a + noise
            a_pert = np.maximum(a_pert, 0)
            s_p = float(np.sum(a_pert))
            if s_p > 0.01:
                best_a = a_pert / s_p
                best_obj = compute_obj(best_a)
            stall = 0

    return best_a


def diverse_init(n, mode):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = 1.0 / (0.03 + 3.0 * (2 * x - 1) ** 2)
    elif mode == 1:
        a = 1.0 / np.sqrt(np.maximum(1e-10, 1 - (2 * x - 1) ** 2))
    elif mode == 2:
        d = np.abs(x - 0.5)
        a = np.maximum(d, 1.0 / (2 * n)) ** (-2.0) + 0.01
    elif mode == 3:
        a = np.exp(-((idx - n / 4.0) ** 2) / (2 * (n / 8.0) ** 2)) + \
            np.exp(-((idx - 3 * n / 4.0) ** 2) / (2 * (n / 8.0) ** 2)) + 0.01
    elif mode == 4:
        a = (1 + 0.5 * np.sin(2 * np.pi * x)) ** 2
    elif mode == 5:
        a = 4 * np.abs(x - 0.5) ** 3 + 0.02
    elif mode == 6:
        a = np.exp(3 * (2 * x - 1) ** 2)
    elif mode == 7:
        a = 1.0 / (1.0 + np.exp(-8 * (x - 0.3)) + np.exp(-8 * (0.7 - x)))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-method optimization combining full LP, LSE gradient descent,

    best_seq = None
    best_score = np.inf

    # === Load parent ===
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.all(np.isfinite(parent_arr)):
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        print("Could not load parent, generating inits")

    if best_seq is None:
        n_current = 800

    for mode in inits_modes:

        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Init mode={mode}: obj={obj_init:.8f} **")
        else:
            print(f"  Init mode={mode}: obj={obj_init:.8f}")

        best_seq = diverse_init(n_current, 0)
        best_score = compute_obj(best_seq)
