Optimizes step function coefficients using a combination of:
1. Softmax-smoothed gradient descent with temperature annealing on the simplex
2. LP alternating minimization (Dinkelbach-style)
3. Vectorized coordinate descent polishing
4. Multi-restart with length exploration

The key new contribution is the softmax gradient method, which provides a
smooth approximation to the max-convolution objective and allows efficient
O(n log n) iterations via FFT-based convolution.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}. O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    Softmax-smoothed gradient descent on the simplex.
    Minimizes t * log(sum exp(b[k]/t)) which approximates max(b) as t -> 0.
    Uses annealing: t decreases over time to sharpen the max approximation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    if t_start is None:
        b = compute_conv(a)
        t_start = float(np.std(b)) * 2.0 + 1e-6
    if t_end is None:
        t_end = t_start * 1e-4
    if eta_start is None:
        eta_start = 0.1 / n
        
    best_a = a.copy()
    b = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(b))
    
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Precompute FFT sizes
    conv_size = 1
    while conv_size < 2 * n - 1:
        conv_size <<= 1
    grad_size = 1
    while grad_size < 3 * n - 2:
        grad_size <<= 1
    
    iter_count = 0
    stall = 0
    no_improve_count = 0
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    adam_t = 0
    
    while time.monotonic() - start < time_budget - 1.0:
        iter_count += 1
        adam_t += 1
        
        # Anneal temperature
        progress = min(1.0, iter_count / max(1, int(time_budget / 0.05)))  # rough estimate
        t = t_start * (t_end / t_start) ** (progress ** 0.7)
        t = max(t, t_end)
        
        # Recompute conv if needed (we have it from last iteration)
        if iter_count == 1:
            b = compute_conv(a)
        
        # Softmax weights
        b_max = float(np.max(b))
        exp_b = np.exp((b - b_max) / t)
        p = exp_b / float(np.sum(exp_b))
        
        # Gradient: g[j] = 2 * conv(p, a)[j]
        fp = rfft(p, grad_size)
        fa = rfft(a, grad_size)
        g = 2.0 * irfft(fp * fa, grad_size)[:n]
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** adam_t)
        v_hat = v / (1 - beta2 ** adam_t)
        eta_adapt = eta_start / (np.sqrt(v_hat) + eps)
        
        a_new = a - eta_adapt * m_hat
        a_new = project_simplex(a_new)
        
        # Evaluate
        b = compute_conv(a_new)
        obj = 2.0 * n * float(np.max(b))
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve_count = 0
        else:
            a = a_new  # still move (momentum helps)
            no_improve_count += 1
        
        # Reset if stuck
        if no_improve_count > 200:
            # Perturb
            pert = rng.standard_normal(n) * 0.01
            a = best_a + pert * best_a
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            else:
                a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve_count = 0
            
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    SGD iter {iter_count}: obj={best_obj:.10f}, t={t:.2e}, elapsed={elapsed:.1f}s")
    
    # Final recompute
    b = compute_conv(best_a)
    final_obj = 2.0 * n * float(np.max(b))
    if verbose:
        print(f"    SGD done: {iter_count} iters, best={final_obj:.10f}")
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    try:
        import cvxpy as cp
    except ImportError:
        return None, None
    
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass
    
    for solver in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a, time_budget, seed=42, verbose=True):
    """LP alternating minimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    iter_count = 0
    stall = 0
    
    while time.monotonic() - start < time_budget - 3:
        t0 = time.monotonic()
        g_lp, M_lp = solve_lp(a, warm_start=prev_g)
        if g_lp is None:
            break
        prev_g = g
        iter_count += 1
        
        best_local_obj = best_obj
        best_local_a = a.copy()
        best_t = 0.0
        
        t_values = [1.0, 0.7, 0.5, 0.3, 0.15, 0.1, 0.05, 0.02]
        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_t = t
        
        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall = 0
            if verbose and iter_count % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP {iter_count}: obj={best_obj:.10f}, t={best_t:.3f}, elapsed={elapsed:.0f}s")
        else:
            stall += 1
            if stall >= 6:
                strength = 0.005 + 0.003 * min(stall, 10)
                pert = rng.standard_normal(n) * strength
                a = best_a + pert * best_a
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                prev_g = None
                stall = 0
    
    if verbose:
        print(f"    LP done: {iter_count} iters, best={best_obj:.10f}")
    return best_a


def coordinate_descent_polish(a, time_budget, seed=42, verbose=False):
    """Vectorized coordinate descent on normalized sequence (sum=1)."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    sweep = 0
    no_improve = 0
    max_no_improve = 8
    mean_val = 1.0 / n
    
    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - start > time_budget:
                break
            
            a_j = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)
            
            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))
            
            if a_j < 1e-12:
                candidates = np.array([0.0, mean_val * 0.1, mean_val * 0.5, mean_val, mean_val * 2])
            else:
                candidates = np.array([
                    a_j * 0.3, a_j * 0.5, a_j * 0.7,
                    a_j * 1.3, a_j * 1.7, a_j * 2.5,
                    mean_val * 0.5, mean_val * 1.5, mean_val * 3
                ])
                candidates = candidates[(candidates >= 0) & (candidates <= 0.5)]
                candidates = np.unique(candidates)
            
            best_trial_obj = best_obj
            best_trial_val = a_j
            
            for val in candidates:
                delta = val - a_j
                if abs(delta) < 1e-15:
                    continue
                S_new = 1.0 + delta
                if S_new <= 1e-10:
                    continue
                
                indices = np.arange(k_start, k_end + 1)
                affected = conv[k_start:k_end + 1].copy()
                mask = (indices == 2 * j)
                if np.any(~mask):
                    affected[~mask] += 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    affected[mask] = conv[2 * j] + (val * val - a_j * a_j)
                
                max_conv_new = max(max_unaffected, float(np.max(affected)))
                obj_trial = 2.0 * n * max_conv_new / (S_new * S_new)
                
                if obj_trial < best_trial_obj - 1e-13:
                    best_trial_obj = obj_trial
                    best_trial_val = val
            
            if best_trial_val != a_j:
                delta = best_trial_val - a_j
                a[j] = best_trial_val
                S_new = 1.0 + delta
                indices = np.arange(k_start, k_end + 1)
                mask = (indices == 2 * j)
                if np.any(~mask):
                    conv[k_start:k_end + 1] += 2.0 * delta * a[indices - j]
                if np.any(mask):
                    conv[2 * j] += (best_trial_val ** 2 - a_j ** 2)
                
                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S
                
                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True
        
        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")
    
    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 9):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s
