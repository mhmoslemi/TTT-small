"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses projected subgradient descent on the max-convolution objective, combined

Key innovation: direct subgradient optimization of the actual objective (max of

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v, target_sum=1.0):
    """Project vector v onto the simplex {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    # Scale to make it a standard simplex projection
    w = v / target_sum if target_sum > 0 else v
    # Standard: project onto {x: sum(x)=1, x>=0}
    # Algorithm: find theta such that sum(max(w - theta, 0)) = 1
    w_sorted = np.sort(w)[::-1]
    cv = np.cumsum(w_sorted)
    rho = np.nonzero(w_sorted * np.arange(1, n + 1) > (cv - 1))[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cv[rho] - 1) / (rho + 1)
    result = np.maximum(w - theta, 0) * target_sum
    # Fix rounding
    diff = target_sum - float(np.sum(result))
    if abs(diff) > 1e-15 and n > 0:
        idx = np.argmax(result)
        if result[idx] + diff > 0:
            result[idx] += diff
        else:
            result += diff / n
            result = np.maximum(result, 0)
    return result

        return a

    best_a = a.copy()
    best_obj = compute_obj_norm(a)

    conv = compute_conv(a)
    max_val = float(np.max(conv))

    avg_grad_norm = 2.0  # initial estimate

    stall = 0
    
    while time.monotonic() - start < time_budget - 2:
        t += 1
        
        # Compute subgradient
        g, active = subgradient_step(a, conv)
        g_norm = float(np.linalg.norm(g))
        
        # Update running average of gradient norm (exponential moving average)
        avg_grad_norm = 0.9 * avg_grad_norm + 0.1 * max(g_norm, 1e-10)
        
        # Step size: c / sqrt(t) with backtracking
        base_step = 0.5 / (np.sqrt(t) * avg_grad_norm)
        step = base_step
        
        # Backtracking line search
        best_trial_obj = best_obj
        best_trial_a = a.copy()
        for back in range(20):
            a_try = a - step * g
            a_try = project_simplex(a_try, 1.0)
            if float(np.sum(a_try)) < 0.01:
                step *= 0.5
                continue
            conv_try = compute_conv(a_try)
            obj_try = 2.0 * n * float(np.max(conv_try))
            if obj_try < best_obj - 1e-14:
                best_trial_obj = obj_try
                best_trial_a = a_try.copy()
                break
            step *= 0.5
        
        if best_trial_obj < best_obj - 1e-14:
            best_obj = best_trial_obj
            best_a = best_trial_a
            a = best_trial_a
            conv = compute_conv(a)
            stall = 0
        else:
            # Try a smaller fixed step without backtracking (just move a little)
            a_try = a - base_step * 0.1 * g
            a_try = project_simplex(a_try, 1.0)
            conv_try = compute_conv(a_try)
            obj_try = 2.0 * n * float(np.max(conv_try))
            if obj_try < best_obj - 1e-14:

                best_a = a_try
                a = a_try
                conv = conv_try
                stall = 0
            else:

                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                    else:

                    conv = compute_conv(a)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass

    for solver in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_qp(a):
    """
    Solve QP: min sum(conv(a,g)^2) s.t. sum(g)=1, g>=0.
    This gives a different direction than LP.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    c = A @ g
    constraints = [cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(cp.sum_squares(c)), constraints)
    
    for solver in ['CLARABEL', 'SCS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def lp_alternate(a, time_budget, seed=42, verbose=True, use_qp=False):
    """LP/QP alternating minimization with adaptive step size and warm starts."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    iter_count = 0
    stall = 0

    while time.monotonic() - start < time_budget - 3:
        t0 = time.monotonic()
        
        if use_qp and iter_count % 3 == 2:
            g_lp = solve_qp(a)
            M_lp = None
        else:
            g_lp, M_lp = solve_lp(a, warm_start=prev_g)
        
        if g_lp is None:
            break
        prev_g = g_lp
        iter_count += 1

        # Adaptive line search
        best_local_obj = best_obj
        best_local_a = a.copy()
        best_t = 0.0

        t_values = [1.0, 0.7, 0.5, 0.3, 0.15, 0.1, 0.05, 0.03, 0.02]
        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_t = t

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall = 0
            if verbose and iter_count % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP {iter_count}: obj={best_obj:.10f}, t={best_t:.3f}, elapsed={elapsed:.0f}s")
        else:
            stall += 1
            if stall >= 6:
                strength = 0.003 + 0.002 * min(stall, 15)
                pert = rng.standard_normal(n) * strength
                a = best_a * (1 + pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                prev_g = None
                stall = 0

        if verbose and iter_count % 40 == 0:
            elapsed = time.monotonic() - start
            remaining = time_budget - elapsed
            print(f"    LP iter {iter_count}: best={best_obj:.10f}, rem={remaining:.0f}s")

    if verbose:
        print(f"    LP done: {iter_count} iters, best={best_obj:.10f}")
    return best_a


def coordinate_descent_polish(a, time_budget, seed=42, verbose=False):
    """
    Coordinate descent on normalized sequence (sum=1).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    mean_val = 1.0 / n

    sweep = 0
    no_improve = 0
    max_no_improve = 4

    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            a_j = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)

            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))

            if a_j < 1e-12:
                candidates = np.array([0.0, mean_val * 0.1, mean_val * 0.5, mean_val, mean_val * 2, mean_val * 5])
            else:
                candidates = np.array([
                    a_j * 0.3, a_j * 0.5, a_j * 0.7, a_j * 0.9,
                    a_j * 1.1, a_j * 1.3, a_j * 1.5, a_j * 2.0,
                    mean_val * 0.3, mean_val * 0.7, mean_val * 1.3, mean_val * 2, mean_val * 4
                ])
                candidates = candidates[(candidates >= 0) & (candidates <= 0.3)]
                candidates = np.unique(candidates)

            best_trial_obj = best_obj
            best_trial_val = a_j

            for val in candidates:
                delta = val - a_j
                if abs(delta) < 1e-15:
                    continue

                S_new = 1.0 + delta
                if S_new <= 1e-10:
                    continue

                indices = np.arange(k_start, k_end + 1)
                affected = conv[k_start:k_end + 1].copy()
                mask = (indices == 2 * j)
                if np.any(~mask):
                    affected[~mask] = conv[k_start:k_end + 1][~mask] + 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    affected[mask] = conv[2 * j] + (val * val - a_j * a_j)

                max_conv_unscaled = max(max_unaffected, float(np.max(affected)))
                obj_trial = 2.0 * n * max_conv_unscaled / (S_new * S_new)

                if obj_trial < best_trial_obj - 1e-13:
                    best_trial_obj = obj_trial
                    best_trial_val = val

            if best_trial_val != a_j:
                delta = best_trial_val - a_j
                old_a_j = a_j
                a[j] = best_trial_val
                S_new = 1.0 + delta

                indices = np.arange(k_start, k_end + 1)
                mask = (indices == 2 * j)
                if np.any(~mask):
                    conv[k_start:k_end + 1][~mask] += 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    conv[2 * j] += (best_trial_val * best_trial_val - old_a_j * old_a_j)

                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S

                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")

    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 10):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_restart_perturb(a, rng, strength=0.05):
    """Random multiplicative perturbation."""
    n = len(a)
    pert = rng.standard_normal(n) * strength
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: Direct projected subgradient descent on the max-convolution objective
using the full active set with momentum acceleration, combined with LP alternating
minimization for larger jumps, QP-based diverse directions, and coordinate descent
polishing. Multi-restart with length exploration to escape local minima.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v, target_sum=1.0):
    """Project vector v onto the simplex {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    w = v / target_sum
    w_sorted = np.sort(w)[::-1]
    cv = np.cumsum(w_sorted)
    rho_candidates = np.nonzero(w_sorted * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho_candidates) == 0:
        theta = 0.0
    else:
        rho = rho_candidates[-1]
        theta = (cv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(w - theta, 0.0) * target_sum
    diff = target_sum - float(np.sum(result))
    if abs(diff) > 1e-12 and n > 0:
        idx = int(np.argmax(result))
        if result[idx] + diff > 0:
            result[idx] += diff
        else:
            result = np.maximum(result + diff / n, 0.0)
    return result


def compute_subgradient(a, conv, active_fraction=0.05, min_active=10):
    """
    Compute subgradient of max(conv(a,a)) at point a.
    g[m] = (2/|S|) * sum_{k in S} a[k-m]
    Returns (gradient vector of length n, active set size).
    """
    n = len(a)
    max_val = float(np.max(conv))
    threshold = max_val * (1.0 - active_fraction)
    active = np.where(conv >= threshold)[0]

    if len(active) < min_active:
        sorted_idx = np.argsort(conv)[::-1]
        active = sorted_idx[:min_active]

    active_count = len(active)
    active_ind = np.zeros(2 * n - 1)
    active_ind[active] = 1.0

    # Cross-correlation: corr[m] = sum_l a[l] * active_ind[l+m]
    # Need FFT size >= n + (2n-1) - 1 = 3n - 2 to avoid wraparound
    size = 1
    while size < 3 * n - 2:
        size <<= 1

    fa = rfft(a, size)
    f_act = rfft(active_ind, size)
    corr = irfft(fa * np.conjugate(f_act), size)

    g = (2.0 / active_count) * corr[:n]
    return g, active_count


def subgradient_optimize(a, time_budget, seed=42, verbose=True, momentum=True):
    """
    Projected subgradient descent with momentum on the max-conv objective.
    Uses Polyak-like step size with backtracking.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    conv = compute_conv(a)
    t = 0
    stall = 0
    v = np.zeros(n)  # momentum

    # Adaptive parameters
    avg_obj_gap = best_obj - lower_bound
    step_scale = 1.0

    while time.monotonic() - start < time_budget - 2:
        t += 1

        # Compute subgradient
        g, active_count = compute_subgradient(a, conv, active_fraction=0.05, min_active=max(10, n // 100))

        if momentum:
            beta = min(0.9, 0.3 + 0.005 * t)
            v = beta * v + (1.0 - beta) * g
            direction = v
        else:
            direction = g

        g_norm = float(np.linalg.norm(direction))
        if g_norm < 1e-15:
            break

        # Polyak-like step size with scaling
        obj_gap = best_obj - lower_bound
        obj_gap = max(obj_gap, 1e-6)
        base_step = step_scale * obj_gap / (g_norm * g_norm + 1e-10)
        # Cap step size to avoid overshooting
        max_step = 0.5 / g_norm
        base_step = min(base_step, max_step)

        # Backtracking line search
        found_improvement = False
        for back in range(25):
            trial_step = base_step * (0.9 ** back)
            a_try = a - trial_step * direction
            a_try = project_simplex(a_try, 1.0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            conv_try = compute_conv(a_try)
            obj_try = 2.0 * n * float(np.max(conv_try))
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                conv = conv_try
                found_improvement = True
                break

        if found_improvement:
            stall = 0
            step_scale = min(step_scale * 1.05, 5.0)
        else:
            step_scale = max(step_scale * 0.7, 0.01)
            stall += 1
            if stall >= 40:
                # Perturb to escape

                pert = rng.standard_normal(n) * strength
                a = best_a * (1.0 + pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                conv = compute_conv(a)
                v = np.zeros(n)
                step_scale = 1.0
                stall = 0

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - start
            remaining = time_budget - elapsed
            print(f"    SubGrad {t}: obj={best_obj:.10f}, active={active_count}, "
                  f"step_scale={step_scale:.3f}, rem={remaining:.0f}s")

    if verbose:
        elapsed = time.monotonic() - start
        print(f"    SubGrad done: {t} iters, best={best_obj:.10f}, time={elapsed:.1f}s")
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass

    for solver in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_qp(a):
    """
    Solve QP: min sum(conv(a,g)^2) s.t. sum(g)=1, g>=0.
    Provides a different (smoother) direction than LP.
    """
    n = len(a)
    if n > 600:
        # QP is too expensive for large n, subsample
        return None
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    c = A @ g
    constraints = [cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(cp.sum_squares(c)), constraints)

    for solver in ['CLARABEL', 'SCS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def solve_lp_max_plus_smooth(a, smooth_weight=0.01):
    """
    Solve: min M + w * sum(conv(a,g)^2) s.t. sum(g)=1, g>=0.
    Hybrid between LP (min max) and QP (min sum of squares).
    """
    n = len(a)
    if n > 800:
        return None
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c = A @ g
    constraints = [cp.sum(g) == 1, c <= M]
    prob = cp.Problem(cp.Minimize(M + smooth_weight * cp.sum_squares(c)), constraints)

    for solver in ['CLARABEL', 'SCS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None

def coordinate_descent_polish(a, time_budget, seed=42, verbose=False):

    start = time.monotonic()
    rng = np.random.default_rng(seed)

    no_improve = 0
    max_no_improve = 4

    while time.monotonic() - start < time_budget:

        sweep_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)

                S_new = 1.0 + delta

                mask = (indices == 2 * j)
                if np.any(~mask):
                    conv[k_start:k_end + 1][~mask] += 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    conv[2 * j] += (best_trial_val * best_trial_val - old_a_j * old_a_j)

                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S

                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")

    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 12):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_perturb(a, rng, strength=0.03):
    """Random multiplicative perturbation."""
    n = len(a)
    pert = rng.standard_normal(n) * strength
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining:
    1. Projected subgradient descent with momentum (direct objective optimization)
    2. LP alternating minimization (proven large jumps)
    3. Hybrid LP+QP for diverse directions
    4. Coordinate descent polishing
    5. Multi-restart with perturbation and length exploration
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 25.0

    # Load parent
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0)
        if a.size == 0 or a.sum() < 0.01:
            raise ValueError("Invalid parent")
        a = a / a.sum()
        print(f"Parent: n={a.size}, obj={compute_obj_norm(a):.10f}")
    except Exception:
        a = np.ones(800) / 800.0
        print(f"Default: n=800, obj={compute_obj_norm(a):.10f}")

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    rng = np.random.default_rng(seed)

    # === Phase 1: Subgradient with momentum (30% time) ===
    print("\n--- Phase 1: Subgradient Descent ---")
    p1_budget = min(budget_s * 0.30, 400.0)
    a1 = subgradient_optimize(best_a, time_budget=p1_budget, seed=seed, verbose=True, momentum=True)
    obj1 = compute_obj_norm(a1)
    if obj1 < best_obj:
        best_obj = obj1
        best_a = a1.copy()
        print(f"  P1 SubGrad: {best_obj:.10f} **")
    else:
        print(f"  P1 SubGrad: {obj1:.10f} (no improvement)")

    # === Phase 2: LP Alternating (25% time) ===
    print("\n--- Phase 2: LP Alternating ---")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        p2_budget = min(remaining * 0.4, 350.0)
        a2 = lp_alternate(best_a, time_budget=p2_budget, seed=seed, verbose=True, use_hybrid=True)
        obj2 = compute_obj_norm(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            print(f"  P2 LP: {best_obj:.10f} **")
        else:
            print(f"  P2 LP: {obj2:.10f}")

    # === Phase 3: Subgradient polish after LP (15% time) ===
    print("\n--- Phase 3: SubGrad after LP ---")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p3_budget = min(remaining * 0.3, 200.0)
        a3 = subgradient_optimize(best_a, time_budget=p3_budget, seed=seed + 1000, verbose=True, momentum=True)
        obj3 = compute_obj_norm(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            print(f"  P3 SubGrad: {best_obj:.10f} **")
        else:
            print(f"  P3 SubGrad: {obj3:.10f}")

    # === Phase 4: CD Polish (10% time) ===
    print("\n--- Phase 4: CD Polish ---")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = min(remaining * 0.3, 150.0)
        a4 = coordinate_descent_polish(best_a, time_budget=p4_budget, seed=seed + 2000, verbose=False)
        obj4 = compute_obj_norm(a4)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()
            print(f"  P4 CD: {best_obj:.10f} **")
        else:
            print(f"  P4 CD: {obj4:.10f}")

    # === Phase 5: LP from Fourier-perturbed state (10% time) ===
    print("\n--- Phase 5: LP from Perturbed State ---")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_pert = fourier_perturb(best_a, rng, strength=0.012)
        p5_budget = min(remaining * 0.35, 200.0)
        a5 = lp_alternate(a_pert, time_budget=p5_budget, seed=seed + 5000, verbose=False, use_qp=True)
        obj5 = compute_obj_norm(a5)
        if obj5 < best_obj:
            best_obj = obj5
            best_a = a5.copy()
            print(f"  P5 LP+QP perturbed: {best_obj:.10f} **")
        else:
            print(f"  P5 LP+QP perturbed: {obj5:.10f}")

    # === Phase 6: Multi-length exploration (10% time) ===
    print("\n--- Phase 6: Length Exploration ---")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    n_options = sorted(set([
        n_best - 40, n_best - 20, n_best + 20, n_best + 40,
        n_best - 80, n_best + 80, n_best + 150, n_best - 150,
        max(400, n_best - 200), min(2000, n_best + 200),
    ]))
    n_options = [n for n in n_options if 350 <= n <= 2000 and n != n_best]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_a, n_try)
        alloc = min(remaining * 0.3, 120.0)

        if alloc > 30:
            # Subgradient for new length
            a_try = subgradient_optimize(a_try, time_budget=alloc * 0.5, seed=n_try * 7, verbose=False)
            remaining = deadline - time.monotonic()
            if remaining > 25:
                a_try = lp_alternate(a_try, time_budget=min(remaining * 0.35, 60.0),
                                     seed=n_try * 11, verbose=False)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_try = coordinate_descent_polish(a_try, time_budget=min(remaining * 0.3, 30.0),
                                                  seed=n_try * 13, verbose=False)

        obj_try = compute_obj_norm(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            print(f"  ** n={n_try}: {best_obj:.10f} **")
        else:
            print(f"  n={n_try}: {obj_try:.10f}")

    # === Phase 7: Another perturbed restart with subgradient ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n--- Phase 7: Perturbed SubGrad Restart ---")
        a_pert2 = random_perturb(best_a, rng, strength=0.02)
        p7_budget = min(remaining * 0.4, 150.0)
        a7 = subgradient_optimize(a_pert2, time_budget=p7_budget, seed=seed + 7777, verbose=False)
        obj7 = compute_obj_norm(a7)
        if obj7 < best_obj:
            best_obj = obj7
            best_a = a7.copy()
            print(f"  P7: {best_obj:.10f} **")
        else:
            print(f"  P7: {obj7:.10f}")

    # === Phase 8: Final polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("\n--- Phase 8: Final Polish ---")
        a_fin = coordinate_descent_polish(best_a, time_budget=min(remaining * 0.5, 50.0),
                                          seed=seed + 42424, verbose=False)
        obj_fin = compute_obj_norm(a_fin)
        if obj_fin < best_obj:
            best_obj = obj_fin
            best_a = a_fin.copy()
            print(f"  Final CD: {best_obj:.10f} **")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_fin2 = subgradient_optimize(best_a, time_budget=min(remaining * 0.7, 25.0),
                                          seed=seed + 99999, verbose=False)
            obj_fin2 = compute_obj_norm(a_fin2)
            if obj_fin2 < best_obj:
                best_obj = obj_fin2
                best_a = a_fin2.copy()
                print(f"  Final SubGrad: {best_obj:.10f} **")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
