Multi-trajectory soft-max gradient descent with Nesterov momentum,
adaptive learning rate, and LP polishing for step function optimization.
Uses FFT for O(n log n) convolution and gradient computations.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective assuming sum(a) ~ 1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1) / (idx + 1.0)
    return np.maximum(v - theta, 0)


def compute_gradient(a, p):
    """
    Gradient of soft objective. p is the softmax weights over convolution indices.
    grad_i = 2 * sum_k p_k * a_{k-i} (with appropriate bounds)
    Uses cross-correlation via FFT.
    """
    n = len(a)
    p_len = len(p)  # 2n-1
    # We need: for each i, sum over k of p_k * a_{k-i} where 0<=k-i<n
    # This is a cross-correlation of a and p
    # cross_corr[i] = sum_k p_k * a_{k-i} = (p * a_rev)[n-1+i] where * is convolution
    a_rev = a[::-1]
    size = 1
    while size < n + p_len - 1:
        size <<= 1
    A = rfft(a_rev, size)
    P = rfft(p, size)
    full = irfft(A * P, size)
    q = full[n - 1:n - 1 + n]
    return 2.0 * q

    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    it = 0
    stall = 0

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        if schedule == 'linear':
            beta = beta_start + (beta_end - beta_start) * progress
        elif schedule == 'geometric':
            beta = beta_start * (beta_end / beta_start) ** progress
        else:
            beta = beta_start + (beta_end - beta_start) * progress**2
        
        lr = lr_start * (1.0 - 0.5 * progress)
        
        conv = compute_conv(a)

    stall = 0

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        a_norm = a / float(np.sum(a))
        # Build constraint matrix: [A | -1] @ [g; M] <= 0, sum(g)=1, g>=0
        # A[k,i] = a_norm[k-i] if 0<=k-i<n

        for k in range(2 * n - 1):

            i_hi = min(n - 1, k)

                A[k, i_lo:i_hi + 1] = a_norm[js]

        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Assumes sum(a)=1 for speed."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1) / (idx + 1.0)
    return np.maximum(v - theta, 0)


def compute_gradient(a, p):
    """
    Compute gradient of soft objective.
    p is softmax weights over convolution indices (length 2n-1).
    grad_i = 2 * sum_j a_j * p_{i+j}
    Uses FFT cross-correlation for O(n log n) efficiency.
    """
    n = len(a)
    p_len = len(p)
    f = a[::-1]
    size = 1
    while size < n + p_len - 1:
        size <<= 1
    F = rfft(f, size)
    P = rfft(p, size)
    conv = irfft(F * P, size)
    q = conv[n - 1:2 * n - 1]
    return 2.0 * q


def subgradient_step(a, lr):
    """
    Take a subgradient step on the max-convolution objective.
    Returns new a and the active indices.
    """
    n = len(a)
    conv = compute_conv(a)
    c_max = float(np.max(conv))
    active = np.where(conv >= c_max - 1e-12)[0]
    # Average subgradient over active set
    grad = np.zeros(n)
    for k in active:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            js = k - np.arange(i_lo, i_hi + 1)
            grad[i_lo:i_hi + 1] += 2.0 * a[js] / len(active)
    a_new = project_simplex(a - lr * grad)
    return a_new, active


def softmax_gd(a, time_budget, beta_start, beta_end, lr_start, momentum,
               verbose=False, seed=42, schedule='geometric'):
    """
    Soft-max gradient descent with Nesterov momentum on the simplex.
    Minimizes smooth approximation of max-convolution objective.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    it = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        if schedule == 'geometric':
            ratio = beta_end / beta_start
            beta = beta_start * ratio ** progress
        elif schedule == 'quadratic':
            beta = beta_start + (beta_end - beta_start) * progress ** 2
        else:
            beta = beta_start + (beta_end - beta_start) * progress

        lr = lr_start * (1.0 - 0.4 * progress)

        conv = compute_conv(a)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p_sum = float(np.sum(exp_vals))
        if p_sum < 1e-300:
            p = np.zeros(len(conv))
            p[int(np.argmax(conv))] = 1.0
        else:
            p = exp_vals / p_sum

        grad = compute_gradient(a, p)

        v = momentum * v + grad
        a_new = project_simplex(a - lr * v)

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall > 150:
                v = np.zeros(n)
                stall = 0
                # Antithetic perturbation: try both directions
                pert = rng.normal(0, 0.004, n)
                a_plus = project_simplex(best_a + pert)
                a_minus = project_simplex(best_a - pert)
                obj_plus = compute_obj(a_plus)
                obj_minus = compute_obj(a_minus)
                if obj_plus <= obj_minus:
                    a = a_plus
                else:
                    a = a_minus

        a = a_new
        it += 1

        if verbose and it % 500 == 0:
            print(f"      SmaxGD it{it}: best={best_obj:.8f}, beta={beta:.1f}, "
                  f"lr={lr:.5f}, t={elapsed:.1f}s")

        # Reset if stuck too long
        if it - best_it > 6000 and it > 2000:
            a = best_a.copy()
            v = np.zeros(n)

    return best_a, best_obj


def subgradient_polish(a, time_budget=30.0, lr_start=0.01, verbose=False):
    """
    Subgradient method with decreasing step size on the max-convolution.
    Complements soft-max by directly targeting the active constraints.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        lr = lr_start / (1.0 + 0.01 * it)

        a_new, _ = subgradient_step(a, lr)

        # Add small noise to escape degeneracy
        noise = rng.normal(0, lr * 0.1, n)
        a_new = project_simplex(a_new + noise)

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                a = project_simplex(best_a + rng.normal(0, 0.002, n))
                stall = 0

        a = a_new
        it += 1

        if verbose and it % 100 == 0:
            print(f"      SubGrad it{it}: best={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

        if it > 500 and time.monotonic() - start > time_budget * 0.8:
            break

    return best_a, best_obj


def lp_polish(a, time_budget=60.0, max_iters=30, verbose=False):
    """
    LP-based polishing: iteratively find best direction g minimizing
    max conv(a_norm, g), then blend with line search.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(777)

    # Pre-build matrix structure (indices only, values change with a)
    # A[k, i] = a[k-i] if 0 <= k-i < n
    k_indices = []
    i_indices = []
    rel_indices = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            k_indices.append(k)
            i_indices.append(i)
            rel_indices.append(k - i)

    k_indices = np.array(k_indices)
    i_indices = np.array(i_indices)
    rel_indices = np.array(rel_indices)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        a_norm = a / float(np.sum(a))

        # Build sparse-like matrix using the precomputed structure
        A_vals = a_norm[rel_indices]
        A = np.zeros((2 * n - 1, n))
        A[k_indices, i_indices] = A_vals

        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        solved = False
        for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        g_val = g_val / s
                        solved = True
                        break
            except Exception:
                continue

        if not solved:
            break

        # Line search for blend factor
        best_t = 0.0
        best_obj_t = best_obj
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = project_simplex((1 - t) * a + t * g_val)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t

        if best_t > 0 and best_obj_t < best_obj - 1e-13:
            a = project_simplex((1 - best_t) * a + best_t * g_val)
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 5 == 0:
            print(f"      LP it{it}: best={best_obj:.8f}, t={best_t:.3f}, "
                  f"elapsed={time.monotonic() - start:.1f}s")

        if stall >= 4:
            # Perturb and restart
            noise_scale = 0.003 + 0.002 * (stall - 4)
            noise = rng.normal(0, noise_scale, n)
            a = project_simplex(best_a + noise)
            stall = 0

    return best_a, best_obj


def block_refine(a, time_budget=30.0, block_size=20, verbose=False):
    """
    Block coordinate descent: optimize small blocks while fixing the rest.
    For each block, solve a small QP/LP to minimize the max convolution.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(456)
    n_blocks = (n + block_size - 1) // block_size

    while time.monotonic() - start < time_budget:
        # Shuffle block order
        order = rng.permutation(n_blocks)

        for b_idx in order:
            if time.monotonic() - start >= time_budget:
                break

            lo = b_idx * block_size
            hi = min(n, lo + block_size)
            if lo >= hi:
                continue

            # Optimize block [lo:hi] while fixing rest
            # Use a few gradient steps on just this block
            block = a[lo:hi].copy()
            fixed = a.copy()

            for inner in range(10):
                conv = compute_conv(fixed)
                c_max = float(np.max(conv))
                # Soft gradient with high beta
                beta = 50.0
                exp_vals = np.exp(beta * (conv - c_max))
                p = exp_vals / float(np.sum(exp_vals))
                grad = compute_gradient(fixed, p)
                # Only update the block
                block -= 0.01 * grad[lo:hi]
                block = np.maximum(block, 0)
                # Renormalize the whole vector
                fixed = a.copy()
                fixed[lo:hi] = block
                fix_sum = float(np.sum(fixed))
                if fix_sum > 0:
                    fixed = fixed / fix_sum

            obj_block = compute_obj(fixed)
            if obj_block < best_obj - 1e-14:
                best_obj = obj_block
                best_a = fixed.copy()
                a = fixed.copy()

    return best_a, best_obj


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    return project_simplex(a)


def arcsine_init(n):
    """Arcsine distribution initialization."""
    x = np.linspace(0.001, 0.999, n)
    a = 1.0 / np.sqrt(1 - (2 * x - 1) ** 2)
    return project_simplex(a)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = (np.exp(-((idx - n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) +
             np.exp(-((idx - 3 * n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) + 0.01)
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 6:
        a = 1.0 / np.cosh(3.0 * (2 * x - 1))
    elif mode == 7:
        a = np.zeros(n)
        a[0] = 1.0
        a[-1] = 1.0
        a[1:min(3, n)] = 0.3
        a[max(0, n - 4):n - 1] = 0.3
        if n > 4:
            a[1:-1] += 0.05
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization combining:
    1. Soft-max GD with geometric annealing
    2. Subgradient polishing
    3. LP-based direction finding with line search
    4. Block coordinate refinement
    5. Multi-trajectory exploration with diverse inits
    6. Length exploration
    7. Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = u_shaped_init(n_current, sharpness=3.0)
        best_score = compute_obj(best_seq)
        print(f"Default U-shape init (n={n_current}): obj={best_score:.8f}")

    # Try multiple initializations
    for init_fn, name in [(lambda: arcsine_init(n_current), "arcsine"),
                          (lambda: u_shaped_init(n_current, 5.0), "sharp_u"),
                          (lambda: u_shaped_init(n_current, 2.0), "wide_u")]:
        a_try = init_fn()
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"{name} init: obj={best_score:.8f} **")

    # === Phase 1: Soft-max GD with geometric beta (smooth start) ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 1: SoftMax GD geometric beta 3->30, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.22, 320.0)
        a1, obj1 = softmax_gd(
            best_seq, time_budget=alloc,
            beta_start=3.0, beta_end=30.0,
            lr_start=0.012, momentum=0.90,
            verbose=True, seed=seed, schedule='geometric'
        )
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj1:.8f} (no improvement)")

    # === Phase 2: Subgradient polish ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 2: Subgradient polish, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.08, 80.0)
        a2, obj2 = subgradient_polish(best_seq, time_budget=alloc, lr_start=0.008, verbose=True)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2: obj={best_score:.8f} **")
        else:
            print(f"  Phase2: obj={obj2:.8f} (no improvement)")

    # === Phase 3: Soft-max GD higher beta (sharper) ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 3: SoftMax GD geometric beta 30->100, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.20, 280.0)
        a3, obj3 = softmax_gd(
            best_seq, time_budget=alloc,
            beta_start=30.0, beta_end=100.0,
            lr_start=0.005, momentum=0.92,
            verbose=True, seed=seed + 1, schedule='geometric'
        )
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj3:.8f} (no improvement)")

    # === Phase 4: LP Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: LP Polish, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.18, 200.0)
        a4, obj4 = lp_polish(best_seq, time_budget=alloc, max_iters=40, verbose=True)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4: obj={best_score:.8f} **")
        else:
            print(f"  Phase4: obj={obj4:.8f} (no improvement)")

    # === Phase 5: Multi-trajectory with different hyperparams ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 5: Multi-trajectory exploration, time={remaining:.0f}s ===")
        n_cur = len(best_seq)
        configs = [
            # (beta_start, beta_end, lr, momentum, schedule)
            (5.0, 25.0, 0.010, 0.88, 'geometric'),
            (15.0, 50.0, 0.007, 0.91, 'geometric'),
            (40.0, 120.0, 0.004, 0.93, 'quadratic'),
            (8.0, 40.0, 0.009, 0.89, 'geometric'),
            (20.0, 80.0, 0.006, 0.90, 'geometric'),
        ]

        for ci, (bs, be, lr, mom, sched) in enumerate(configs):
            remaining = deadline - time.monotonic()
            if remaining < 35:
                break

            # Different perturbation strategy for each
            if ci % 2 == 0:
                noise = rng.normal(0, 0.004, n_cur)
                a_c = project_simplex(best_seq + noise)
            else:
                a_c = diverse_init(n_cur, rng, mode=ci % 8)
                a_c = project_simplex(0.5 * a_c + 0.5 * best_seq)

            alloc_sgd = min(remaining * 0.28, 90.0)
            if alloc_sgd > 15:
                a_c, obj_c = softmax_gd(
                    a_c, time_budget=alloc_sgd,
                    beta_start=bs, beta_end=be,
                    lr_start=lr, momentum=mom,
                    verbose=False, seed=seed + 100 + ci, schedule=sched
                )

            remaining = deadline - time.monotonic()
            if remaining > 25:
                alloc_lp = min(remaining * 0.25, 45.0)
                a_c, obj_c = lp_polish(a_c, time_budget=alloc_lp, max_iters=15, verbose=False)

            if obj_c < best_score:
                best_score = obj_c
                best_seq = a_c.copy()
                print(f"  Traj {ci} (beta {bs}->{be}): obj={best_score:.8f} **")
            else:
                print(f"  Traj {ci} (beta {bs}->{be}): obj={obj_c:.8f}")

    # === Phase 6: Block refinement ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 6: Block refinement, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.12, 60.0)
        a6, obj6 = block_refine(best_seq, time_budget=alloc, block_size=16, verbose=True)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Phase6: obj={best_score:.8f} **")
        else:
            print(f"  Phase6: obj={obj6:.8f} (no improvement)")

    # === Phase 7: Length exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 70:
        print(f"\n=== Phase 7: Length exploration, time={remaining:.0f}s ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            n_best - 100, n_best + 100, n_best + 200,
            600, 700, 900, 1000, 1100, 1200, 1400
        ]))
        n_options = [n for n in n_options if 400 <= n <= 1800 and abs(n - n_best) >= 50]
        n_options = n_options[:3]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break

            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = project_simplex(np.maximum(a_try, 0))

            alloc_sgd = min(remaining * 0.35, 55.0)
            if alloc_sgd > 15:
                a_try, obj_try = softmax_gd(
                    a_try, time_budget=alloc_sgd,
                    beta_start=10.0, beta_end=60.0,
                    lr_start=0.008, momentum=0.90,
                    verbose=False, seed=seed + 200, schedule='geometric'
                )

            remaining = deadline - time.monotonic()
            if remaining > 25:
                alloc_lp = min(remaining * 0.30, 40.0)
                a_try, obj_try = lp_polish(a_try, time_budget=alloc_lp, max_iters=20, verbose=False)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 8: Final polish, time={remaining:.0f}s ===")
        # High-beta soft-max GD
        alloc1 = min(remaining * 0.40, 100.0)
        if alloc1 > 20:
            a_f, obj_f = softmax_gd(
                best_seq, time_budget=alloc1,
                beta_start=50.0, beta_end=200.0,
                lr_start=0.003, momentum=0.93,
                verbose=True, seed=seed + 300, schedule='geometric'
            )
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final SG: obj={best_score:.8f} **")

        # Subgradient
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc2 = min(remaining * 0.20, 30.0)
            a_s, obj_s = subgradient_polish(best_seq, time_budget=alloc2, lr_start=0.005, verbose=False)
            if obj_s < best_score:
                best_score = obj_s
                best_seq = a_s.copy()
                print(f"  Final SubGrad: obj={best_score:.8f} **")

        # LP
        remaining = deadline - time.monotonic()
        if remaining > 25:
            alloc3 = min(remaining * 0.40, 70.0)
            a_l, obj_l = lp_polish(best_seq, time_budget=alloc3, max_iters=35, verbose=True)
            if obj_l < best_score:
                best_score = obj_l
                best_seq = a_l.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

    # === Phase 9: Last-ditch perturbations ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print(f"\n=== Phase 9: Last-ditch, time={remaining:.0f}s ===")
        for attempt in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            # Try different perturbation scales
            scales = [0.002, 0.005, 0.01, 0.003, 0.008]
            noise = rng.normal(0, scales[attempt % len(scales)], len(best_seq))
            a_p = project_simplex(best_seq + noise)
            alloc = min(remaining * 0.35, 20.0)
            if alloc > 5:
                a_p, obj_p = lp_polish(a_p, time_budget=alloc, max_iters=10, verbose=False)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Last-ditch {attempt}: obj={best_score:.8f} **")

    # Final validation with actual evaluator
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
