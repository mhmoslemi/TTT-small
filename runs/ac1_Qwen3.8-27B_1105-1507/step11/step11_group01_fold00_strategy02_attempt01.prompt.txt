def build_conv_matrix(a):

        r = min(n, 2 * n - 1 - j)

def kkt_subgradient_descent(a, time_budget=200.0, verbose=False, seed=42):
    """
    KKT-guided subgradient descent. Uses the exact subgradient of max(conv(a,a))

        return a

    start = time.monotonic()
    rng = np.random.default_rng(seed)

    momentum = 0.5
    v_mom = np.zeros(n)
    it = 0
    patience = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining_frac = 1.0 - elapsed / max(time_budget, 1e-10)

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        argmax_set = np.where(conv >= max_conv * (1.0 - 1e-10))[0]

        # Compute subgradient: v[i] = sum_{k in argmax} a[k-i] * 1{0<=k-i<n} / |argmax|
        v_sub = np.zeros(n)
        for k in argmax_set:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                v_sub[lo:hi + 1] += a[k - hi:k - lo + 1]
        v_sub /= max(len(argmax_set), 1)

        # Target: v_sub[i] should equal max_conv/2 (from KKT: 2*v_sub[i] = lambda)
        target = max_conv / 2.0

        # Gradient direction
        grad = 2.0 * v_sub - target * np.ones(n) * 2.0 / n

        # Momentum
        v_mom = momentum * v_mom - eta * grad
        a_new = a + v_mom
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
            v_mom = np.zeros(n)
            eta *= 0.5
        else:
            a_new = a_new / s_new

            best_a = a_new.copy()
            a = a_new

                        best_a = a_pert.copy()

def smoothed_gradient_descent(a, time_budget=100.0, tau_init=30.0, tau_end=500.0, verbose=False):
    """Nesterov-accelerated smoothed gradient descent with exponential surrogate."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    beta = 0.9
    lr = 5.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            s_y = 1.0
        else:
            y = y / s_y

        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        N = 1
        while N < 3 * n:
            N <<= 1

        P = fft(p, N)
        A_fft = fft(y, N)
        g_full = ifft(P * np.conj(A_fft)).real
        grad = 2.0 * g_full[:n]

        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-15:
            lr_eff = min(lr, 3.0 / (np.sqrt(n) * (g_norm / np.sqrt(n) + 1e-10)))
        else:
            lr_eff = lr

        v = beta * v - lr_eff * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:

            best_a = a_new.copy()
        a = a_new
        it += 1

        if verbose and it % 2000 == 0:
            print(f"    GD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, t={elapsed:.1f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x * x + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 10, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Two-bump with gap
        a = np.zeros(n)
        w = n // 5
        a[:w] = np.exp(-((np.arange(w) - w // 2) ** 2) / (2 * (w // 4) ** 2))
        a[-w:] = np.exp(-((np.arange(w) - w // 2) ** 2) / (2 * (w // 4) ** 2))
        a = a / a.sum()
    elif mode == 7:
        # Cosine bump
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
