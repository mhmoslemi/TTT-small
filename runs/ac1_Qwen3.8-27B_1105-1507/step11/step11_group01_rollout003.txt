import cvxpy as cp
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A

def solve_lp(a_norm):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a_norm)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val
        except:
            continue
    return None

    x_old = np.linspace(0, 1, n)

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""

    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Compute linear convolution a*b via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj_full(a):
    """Compute full objective 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a))) / (s * s)


def build_conv_matrix(a_norm):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp(a_norm, verbose=False):
    """Solve LP: min M s.t. conv(a_norm,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a_norm)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    if verbose:
                        M_val = float(np.max(A @ g_val))
                        print(f"      LP solved: M={M_val:.8f}")
                    return g_val
        except Exception:
            continue
    return None


def line_search_blend(a, g, n):
    """
    Find best t in [0,1] minimizing objective of (1-t)*a + t*g.
    Both a and g assumed normalized (sum=1).
    Uses fine grid + golden section on best region.
    """
    # Precompute convolutions
    aa = compute_conv(a)
    ag = compute_conv_cross(a, g)
    gg = compute_conv(g)
    
    def f_of_t(t):
        vals = (1-t)**2 * aa + 2*t*(1-t) * ag + t**2 * gg
        return 2.0 * n * float(np.max(vals))
    
    # Coarse grid
    ts = np.linspace(0, 1, 51)
    fs = np.array([f_of_t(t) for t in ts])
    i_best = int(np.argmin(fs))
    
    # Refine around best
    t_lo = max(0.0, ts[max(0, i_best-1)])
    t_hi = min(1.0, ts[min(50, i_best+1)])
    
    if t_hi - t_lo < 1e-6:
        return ts[i_best], fs[i_best]
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = t_hi - gr * (t_hi - t_lo)
    d = t_lo + gr * (t_hi - t_lo)
    fc = f_of_t(c)
    fd = f_of_t(d)
    
    for _ in range(40):
        if t_hi - t_lo < 1e-8:
            break
        if fc < fd:
            t_hi = d
            d = c
            fd = fc
            c = t_hi - gr * (t_hi - t_lo)
            fc = f_of_t(c)
        else:
            t_lo = c
            c = d
            fc = fd
            d = t_lo + gr * (t_hi - t_lo)
            fd = f_of_t(d)
    
    t_opt = (t_lo + t_hi) / 2.0
    f_opt = f_of_t(t_opt)
    return t_opt, f_opt


def coordinate_descent(a, time_budget=30.0, seed=42, verbose=False):
    """
    Exact coordinate descent with golden section on each 1D subproblem.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0
    
    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        S = float(np.sum(a))
        
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            
            v = a[i]
            S_other = S - v
            if S_other < 1e-12:
                continue
            
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()
            
            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask].copy()
            
            k_quad = 2 * i
            c_2i = float(conv[k_quad])
            
            def f_1d(x, v=v, S_other=S_other, c_k_lin=c_k_lin,
                     a_ki_lin=a_ki_lin, c_2i=c_2i):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom
            
            best_x = v
            best_f = f_1d(v)
            scale = max(v, 1.0 / n, 1e-6)
            
            for mult in [0, 0.3, 0.5, 0.7, 1.0, 1.3, 1.7, 2.5, 4.0]:
                x = v * mult
                if x < 0:
                    continue
                fx = f_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x
            
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + 2.0 * scale
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-8
            
            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)
            
            for _ in range(35):
                if x_hi - x_lo < 1e-13 * max(1.0, abs(x_hi)):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)
            
            x_opt = max(0.0, (x_lo + x_hi) / 2.0)
            
            if abs(x_opt - v) < 1e-15:
                continue
            
            delta = x_opt - v
            a[i] = x_opt
            
            conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin
            
            S += delta
            S_new = float(np.sum(a))
            if S_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
        
        if sweep % 3 == 0:
            conv = compute_conv(a)
            S = float(np.sum(a))
            if S > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                a = a / S
    
    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n == 1 else 1.0/new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def parametric_init(n, mode=0, rng=None):
    """Generate parametric initializations."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.linspace(0, 1, n)
    xm = x - 0.5
    
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        # Semicircle
        a = 1.0 / np.sqrt(np.maximum(1 - (2*xm)**2, 0.001))
    elif mode == 2:
        # Gaussian
        sigma = rng.uniform(0.15, 0.4)
        a = np.exp(-xm**2 / (2*sigma**2))
    elif mode == 3:
        # Polynomial bump
        p = rng.uniform(0.5, 3.0)
        a = np.maximum(1 - (2*xm)**2, 0)**p
    elif mode == 4:
        # Flat top with rounded edges
        w = rng.uniform(0.3, 0.6)
        a = np.where(np.abs(xm) < w, 1.0,
                    np.maximum(1 - (np.abs(xm)-w)/(0.5-w), 0)**2)
    elif mode == 5:
        # Cosine window
        a = np.cos(np.pi * xm)**2
    elif mode == 6:
        # raised cosine squared
        a = np.cos(np.pi * xm / 2)**4
    elif mode == 7:
        # Two peaks
        sep = rng.uniform(0.15, 0.35)
        a = np.exp(-((xm)**2)/(2*0.08**2)) + np.exp(-((xm)**2 - 4*sep**2)/(2*0.08**2))
        a = np.exp(-(np.abs(xm)**2 - sep**2)**2 / (0.05**2))
    else:
        a = rng.uniform(0.5, 1.5, n)
    
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def lp_iteration_loop(a, time_budget, seed=42, verbose=True):
    """
    Main LP iteration: repeatedly solve LP, line search, update.
    Intersperse CD polishing. Restart if stuck.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        it += 1
        
        # Solve LP
        g_lp = solve_lp(a, verbose=False)
        if g_lp is None:
            break
        
        # Line search
        t_opt, f_opt = line_search_blend(a, g_lp, n)
        
        if f_opt < best_obj - 1e-12:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                best_a = a.copy()
                best_obj = f_opt
                stall = 0
                if verbose and it % 5 == 0:
                    elapsed = time.monotonic() - start
                    print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, time={elapsed:.1f}s")
        else:
            stall += 1
            if verbose and it % 5 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it{it}: no improvement (stall={stall}), obj={best_obj:.8f}, time={elapsed:.1f}s")
            
            if stall >= 3:
                # CD polish
                if remaining > 15:
                    cd_budget = min(remaining * 0.05, 20.0)
                    a_cd = coordinate_descent(best_a, time_budget=cd_budget, seed=seed + it)
                    obj_cd = compute_obj_normalized(a_cd)
                    if obj_cd < best_obj:
                        best_obj = obj_cd
                        best_a = a_cd.copy()
                        a = a_cd.copy()
                        if verbose:
                            print(f"    CD polish: obj={best_obj:.8f}")
                        stall = 1
                    else:
                        # Random perturbation
                        pert = rng.normal(0, 0.03, n)
                        a_p = best_a * np.exp(pert)
                        a_p = np.maximum(a_p, 0)
                        a_p = a_p / a_p.sum()
                        obj_p = compute_obj_normalized(a_p)
                        if obj_p < best_obj * 1.05:
                            a = a_p
                        else:
                            a = best_a.copy()
                        stall = 0
        
        # Periodic CD polish every 8 iterations
        if it % 8 == 0 and remaining > 20:
            cd_budget = min(remaining * 0.03, 15.0)
            a_cd = coordinate_descent(a, time_budget=cd_budget, seed=seed + 100 + it)
            obj_cd = compute_obj_normalized(a_cd)
            if obj_cd < best_obj:
                best_obj = obj_cd
                best_a = a_cd.copy()
                a = a_cd.copy()
                if verbose:
                    print(f"    Periodic CD: obj={best_obj:.8f}")
    
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP-iteration-based optimization with CD polishing,
    multi-n exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj_normalized(parent)
                best_seq = parent.copy()
                parent_n = parent.size
                print(f"Parent (n={parent_n}): obj={best_score:.8f}")
            else:
                raise ValueError
        else:
            raise ValueError
    except Exception:
        parent_n = 800
        best_seq = np.ones(parent_n) / parent_n
        best_score = compute_obj_normalized(best_seq)
        print(f"Default init (n={parent_n}): obj={best_score:.8f}")
    
    n_best = len(best_seq)
    
    # === Phase 1: Quick CD from parent ===
    print(f"\n=== Phase 1: Initial CD (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 60.0)
    if p1_budget > 10:
        a_cd = coordinate_descent(best_seq, time_budget=p1_budget, seed=seed)
        obj_cd = compute_obj_normalized(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)")
    
    # === Phase 2: Main LP iteration loop ===
    print(f"\n=== Phase 2: LP Iteration (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.50, 700.0)
    if p2_budget > 30:
        a_lp = lp_iteration_loop(best_seq, time_budget=p2_budget, seed=seed + 10, verbose=True)
        obj_lp = compute_obj_normalized(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP loop: obj={best_score:.8f} **")
        else:
            print(f"  LP loop: obj={obj_lp:.8f} (no improvement over CD)")
    
    # === Phase 3: Multi-n exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    
    # Generate n candidates
    n_candidates = sorted(set([
        max(400, n_cur - 300),
        max(400, n_cur - 150),
        n_cur + 100,
        n_cur + 200,
        n_cur + 400,
        n_cur + 600,
        500, 600, 700, 900, 1000, 1100, 1200, 1400, 1600, 1800, 2000
    ]))
    # Filter to reasonable range, ensure spread
    n_filtered = []
    for nc in n_candidates:
        if 400 <= nc <= 2200:
            if all(abs(nc - nf) >= 80 for nf in n_filtered):
                n_filtered.append(nc)
    # Limit to top candidates, prioritize those near current best
    n_filtered = n_filtered[:8]
    print(f"  n candidates: {n_filtered}")
    
    for n_try in n_filtered:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        
        a_try = resample(best_seq, n_try)
        a_try = a_try / a_try.sum()
        
        # Quick CD
        cd_budget = min(remaining * 0.04, 25.0)
        if cd_budget > 8:
            a_try = coordinate_descent(a_try, time_budget=cd_budget, seed=n_try)
        
        # Brief LP
        lp_budget = min(remaining * 0.06, 60.0)
        if lp_budget > 10:
            a_try = lp_iteration_loop(a_try, time_budget=lp_budget, seed=n_try + 500, verbose=False)
        
        obj_try = compute_obj_normalized(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_cur = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")
    
    # === Phase 4: Diverse restarts at best n ===
    print(f"\n=== Phase 4: Diverse Restarts (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    n_r = len(best_seq)
    
    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        
        mode = 1 + r
        a_r = parametric_init(n_r, mode=mode, rng=rng)
        # Blend with best
        blend = 0.4 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()
        
        # CD + LP
        alloc_total = min(remaining * 0.2, 100.0)
        if alloc_total > 15:
            cd_b = alloc_total * 0.4
            lp_b = alloc_total * 0.6
            a_r = coordinate_descent(a_r, time_budget=cd_b, seed=3000 + r * 100)
            a_r = lp_iteration_loop(a_r, time_budget=lp_b, seed=4000 + r * 100, verbose=False)
            
            obj_r = compute_obj_normalized(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode {mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode {mode}): obj={obj_r:.8f}")
    
    # === Phase 5: Final aggressive LP on best ===
    print(f"\n=== Phase 5: Final LP Polish (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p5_budget = min(remaining * 0.5, 200.0)
        a_final = lp_iteration_loop(best_seq, time_budget=p5_budget, seed=seed + 9999, verbose=True)
        obj_final = compute_obj_normalized(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")
    
    # === Phase 6: Final CD ===
    print(f"\n=== Phase 6: Final CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_final_cd = coordinate_descent(best_seq, time_budget=min(remaining * 0.8, 100.0),
                                        seed=seed + 77777)
        obj_final_cd = compute_obj_normalized(a_final_cd)
        if obj_final_cd < best_score:
            best_score = obj_final_cd
            best_seq = a_final_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final_cd:.8f}")
    
    # Convert to list for return
    best_seq = np.maximum(best_seq, 0.0)
    result = [float(min(1000.0, max(0.0, x))) for x in best_seq]
    
    # Validate
    val = evaluate_sequence(result)
    if val == np.inf:
        print("Warning: evaluate_sequence returned inf, using fallback")
        result = [1.0] * 100
        best_score = 2.0
    elif val < best_score:
        best_score = val
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
