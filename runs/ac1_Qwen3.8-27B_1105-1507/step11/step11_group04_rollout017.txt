"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Projected gradient descent with adaptive softmax temperature (annealing)
2. LP alternating minimization
3. Coordinate descent polishing
4. Multi-restart exploration

The gradient method exploits the connection between the softmax-smoothed
max and the dual of the LP subproblem, using FFT-based gradient computation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_grad_softmax(a, tau):
    """Compute gradient of tau*log(sum(exp(conv/tau))) w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    # Softmax weights
    conv_shifted = conv - np.max(conv)
    p = np.exp(conv_shifted / tau)
    p /= p.sum()
    # Gradient: 2 * (Q * a_rev)[j+n-1] where Q = p padded
    a_rev = a[::-1]
    # Full convolution of p (len 2n-1) and a_rev (len n): length 3n-2
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    fp = rfft(p, size)
    fa = rfft(a_rev, size)
    full_conv = irfft(fp * fa, size)
    grad = 2.0 * full_conv[n-1 : 2*n-1]  # positions j+n-1 for j=0..n-1
    return grad, p

def project_simplex(v):
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    a is normalized to sum=1.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP alternating minimization (polishing)
2. Projected gradient descent with softmax annealing (primary)
3. Coordinate descent (final polish)
4. Multi-restart exploration

The gradient method uses centered gradients and FFT-based computation,
exploiting the connection between softmax-smoothed max and the LP dual.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_grad_and_weights(a, tau):
    """
    Compute gradient of tau*log(sum(exp(conv/tau))) and softmax weights.
    Returns (grad, p, conv).
    """
    n = len(a)
    conv = compute_conv(a)
    conv_max = float(np.max(conv))
    conv_shifted = conv - conv_max
    p = np.exp(conv_shifted / tau)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        p /= p_sum
    
    # Gradient: 2 * (p * a_rev)[j+n-1]
    a_rev = a[::-1]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    fp = rfft(p, size)
    fa = rfft(a_rev, size)
    full_conv = irfft(fp * fa, size)
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    
    return grad, p, conv


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def gradient_optimize(a, time_budget, seed=42, verbose=True):
    """
    Projected gradient descent with softmax annealing and centered gradients.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    conv = compute_conv(a)
    tau = max(float(np.max(conv)) * 3.0, 1e-8)
    
    prev_s = None
    prev_gc = None
    lr = 1.0 / (2 * n)
    
    no_improve = 0
    max_no_improve = 500
    iter_count = 0
    
    while time.monotonic() - start < time_budget and no_improve < max_no_improve:
        # Compute gradient and softmax weights
        grad, p, conv = compute_grad_and_weights(a, tau)
        
        # Center the gradient (remove mean to keep on hyperplane sum=1)
        grad_mean = float(np.mean(grad))
        grad_c = grad - grad_mean
        
        # Barzilai-Borwein step size
        if prev_s is not None:
            y = grad_c - prev_gc
            sy = float(np.dot(prev_s, y))
            yy = float(np.dot(y, y))
            ss = float(np.dot(prev_s, prev_s))
            if sy > 1e-14 and yy > 1e-14:
                bb1 = sy / yy
                if ss > 1e-14:
                    bb2 = ss / sy
                else:
                    bb2 = bb1
                lr = max(1e-12, min(1.0, 0.5 * (bb1 + bb2)))
        
        # Take step
        a_new = a - lr * grad_c
        a_new = project_simplex(a_new)
        
        # Evaluate
        conv_new = compute_conv(a_new)
        obj_new = 2.0 * n * float(np.max(conv_new))
        
        prev_s = a_new - a
        prev_gc = grad_c
        a = a_new
        iter_count += 1
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Anneal temperature
        if no_improve > 0 and no_improve % 30 == 0:
            tau *= 0.85

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None
    a_norm = a / s
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass
    
    for solver in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    return g_val / s_val
        except Exception:
            continue
    return None


def lp_alternate(a, time_budget, seed=42, verbose=False):
    """LP alternating minimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    stall = 0
    iter_count = 0
    
    while time.monotonic() - start < time_budget - 5:
        g_lp = solve_lp(a, warm_start=prev_g)
        if g_lp is None:
            break
        prev_g = g_lp
        iter_count += 1
        
        best_local_obj = best_obj
        best_local_a = a.copy()
        
        for t in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
        
        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 10:
                strength = 0.005
                a = best_a + strength * rng.standard_normal(n) * best_a
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                prev_g = None
                stall = 0
    
    if verbose:
        print(f"    LP done: {iter_count} iters, best={best_obj:.10f}")
    return best_a
