Actually, |A(e^{iθ})|^2 = sum_k conv[k] e^{ikθ} (up to indexing). So max conv[k] >= (1/(2n-1)) * sum conv[k] = (1/(2n-1)) * (sum a_j)^2.

For a polynomial of degree n-1 with nonnegative coefficients summing to 1, the minimum possible max autocorrelation is related to the Bose-Einstein condensate / flat spectrum problem.

"""
Optimizes step function coefficients using LP-based iterative minimax
with coordinate descent polishing. Alternates between solving an LP
to find the optimal response direction and exact CD sweeps.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Assume a is normalized (sum=1). Return 2n*max(conv)."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_minpeak(a):
    """
    Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    a should be the current (normalized) sequence.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(M.value) if M.value is not None else None
                    return g_val, M_val
        except Exception:
            continue
    return None, None


def f_1d_eval(x, n, v, c_quad, a_lin, c_lin):
    """Evaluate 1D subproblem for CD on a normalized vector (sum=1)."""
    if len(a_lin) > 0:
        lin_vals = c_lin + 2.0 * (x - v) * a_lin
        max_lin = float(np.max(lin_vals))
    else:
        max_lin = 0.0
    quad_val = c_quad + x * x - v * v
    max_val = max(max_lin, quad_val)
    denom = 1.0 - v + x
    if denom < 1e-15:
        return 1e30
    return 2.0 * n * max_val / (denom * denom)


def coordinate_descent(a, time_budget=60.0, seed=42, verbose=False):
    """
    Exact coordinate descent. Input should be normalized (sum=1).
    Returns best normalized a found.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    
    conv = compute_conv(a)
    best_obj = compute_obj_normalized(a)
    best_a = a.copy()
    
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0
    no_improve_sweeps = 0
    
    while time.monotonic() - start < time_budget:
        sweep += 1
        elapsed = time.monotonic() - start
        
        # Recompute conv at start of each sweep
        conv = compute_conv(a)
        S = float(np.sum(a))
        if S > 1e-10:
            a = a / S
        conv = compute_conv(a)  # Fresh conv after normalization
        
        order = rng.permutation(n)
        sweep_start_obj = best_obj
        
        for i in order:
            if time.monotonic() - start >= time_budget:
                break
            
            v = a[i]
            k_quad = 2 * i
            
            # Linear terms: k in [i, i+n-1], k != 2i
            ks_all = np.arange(i, i + n)
            mask = ks_all != k_quad
            ks_lin = ks_all[mask]
            
            if len(ks_lin) > 0:
                a_lin = a[ks_lin - i]
                c_lin = conv[ks_lin].copy()
            else:
                a_lin = np.array([])
                c_lin = np.array([])
            
            c_quad = float(conv[k_quad])
            
            # Golden section search on [0, upper]
            upper = max(3.0 * v, 3.0 / n) + 0.01
            
            a_gs = 0.0
            b_gs = upper
            c_pt = b_gs - gr * (b_gs - a_gs)
            d_pt = a_gs + gr * (b_gs - a_gs)
            fc = f_1d_eval(c_pt, n, v, c_quad, a_lin, c_lin)
            fd = f_1d_eval(d_pt, n, v, c_quad, a_lin, c_lin)
            
            for _ in range(45):
                if b_gs - a_gs < 1e-13:
                    break
                if fc < fd:
                    b_gs = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = b_gs - gr * (b_gs - a_gs)
                    fc = f_1d_eval(c_pt, n, v, c_quad, a_lin, c_lin)
                else:
                    a_gs = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = a_gs + gr * (b_gs - a_gs)
                    fd = f_1d_eval(d_pt, n, v, c_quad, a_lin, c_lin)
            
            x_opt = max(0.0, (a_gs + b_gs) / 2.0)
            
            # Check boundary x=0
            f0 = f_1d_eval(0.0, n, v, c_quad, a_lin, c_lin)
            fx = f_1d_eval(x_opt, n, v, c_quad, a_lin, c_lin)
            if f0 < fx:
                x_opt = 0.0
            
            if abs(x_opt - v) < 1e-14:
                continue
            
            delta = x_opt - v
            a[i] = x_opt
            
            # Update conv in-place
            conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_lin
            
            # Track best (using unnormalized conv, will normalize at end of sweep)
            S_curr = float(np.sum(a))
            if S_curr > 1e-10:
                obj_curr = 2.0 * n * float(np.max(conv)) / (S_curr * S_curr)
                if obj_curr < best_obj:
                    best_obj = obj_curr
                    candidate = a / S_curr
                    best_a = candidate.copy()
        
        # End of sweep
        S = float(np.sum(a))
        if S > 1e-10:
            a = a / S
            obj_now = compute_obj_normalized(a)
            if obj_now < best_obj:
                best_obj = obj_now
                best_a = a.copy()
        
        if best_obj < sweep_start_obj - 1e-12:
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
        
        # Perturbation to escape stalemate
        if no_improve_sweeps >= 10 and sweep % 15 == 0:
            strength = 0.005 + 0.005 * (no_improve_sweeps // 15)
            pert = rng.normal(0, strength, n)
            a_try = a * np.exp(pert)
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st > 1e-10:
                a_try = a_try / st
                obj_try = compute_obj_normalized(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                a = a_try
            no_improve_sweeps = 0
        
        if verbose and sweep % 10 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation, return normalized."""
    n = len(a)
    if n == new_n:
        a = a.copy()
    elif n == 1:
        a = np.full(new_n, a[0])
    else:
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, new_n)
        a = np.interp(x_new, x_old, a)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    return a


def lp_iterate(a, time_budget=300.0, seed=42, verbose=False):
    """
    LP-based iteration: solve LP, blend with current, polish with CD.
    Input a normalized. Returns best a found.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 20:
            break
        
        # Step 1: Solve LP
        t_lp_start = time.monotonic()
        g_lp, M_lp = solve_lp_minpeak(a)
        t_lp = time.monotonic() - t_lp_start
        
        if g_lp is None:
            # LP failed, do CD instead
            if remaining > 30:
                a = coordinate_descent(a, time_budget=min(remaining * 0.5, 60.0), seed=seed + it)
                obj = compute_obj_normalized(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                it += 1
            continue
        
        # Step 2: Line search on blending parameter t
        best_t_obj = best_obj
        best_a_blend = best_a.copy()
        
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_blend = a_try.copy()
            if obj_try > best_t_obj + 0.01:
                break
        
        if best_t_obj < best_obj:
            a = best_a_blend
            best_obj = best_t_obj
            best_a = a.copy()
            it += 1
            if verbose:
                print(f"    LP it{it}: obj={best_obj:.8f}, t=${t_lp:.2f}s")
        else:
            # No improvement from LP blend, try CD
            if remaining > 30:
                cd_budget = min(remaining * 0.3, 45.0)
                a = coordinate_descent(a, time_budget=cd_budget, seed=seed + 1000 + it)
                obj = compute_obj_normalized(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                it += 1
            else:
                break
    
    return best_a

"""
Optimizes step function coefficients using LP-based iterative minimax with
coordinate descent polishing. Alternates between solving an LP to find the
optimal response direction and exact golden-section CD sweeps. Includes
simulated annealing for escape, multi-n exploration, and dual LP variants.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Assume a is normalized (sum=1). Return 2n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_minpeak(a):
    """
    Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(M.value) if M.value is not None else None
                    return g_val, M_val
        except Exception:
            continue
    return None, None


def solve_lp_maxsum(a):
    """
    Solve: max sum(g) s.t. conv(a,g)[k] <= M0 for all k, g>=0.
    where M0 = max(conv(a,a)). Returns (g, sum_g) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)
    M0 = float(np.max(compute_conv(a)))
    if M0 < 1e-15:
        return None, None
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    return g_val, gs
        except Exception:
            continue
    return None, None


def f_1d_eval(x, n, v, c_quad, a_lin, c_lin):
    """Evaluate 1D subproblem for CD on a unit-sum vector."""
    if len(a_lin) > 0:
        lin_vals = c_lin + 2.0 * (x - v) * a_lin
        max_lin = float(np.max(lin_vals))
    else:
        max_lin = 0.0
    quad_val = c_quad + x * x - v * v
    max_val = max(max_lin, quad_val)
    denom = 1.0 - v + x
    if denom < 1e-15:
        return 1e30
    return 2.0 * n * max_val / (denom * denom)


def coordinate_descent(a, time_budget=60.0, seed=42, verbose=False):
    """
    Exact coordinate descent with golden section search.
    Input should be normalized (sum=1). Returns best normalized a.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = compute_obj_normalized(a)
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        elapsed = time.monotonic() - start

        conv = compute_conv(a)
        S = float(np.sum(a))
        if S > 1e-10:
            a = a / S

        order = rng.permutation(n)
        sweep_start_obj = best_obj

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            k_quad = 2 * i

            ks_all = np.arange(i, i + n)
            mask = ks_all != k_quad
            ks_lin = ks_all[mask]

            if len(ks_lin) > 0:
                a_lin = a[ks_lin - i]
                c_lin = conv[ks_lin].copy()
            else:
                a_lin = np.array([])
                c_lin = np.array([])

            c_quad = float(conv[k_quad])

            # Golden section on [0, upper]
            upper = max(3.0 * v, 3.0 / n) + 0.01
            a_gs = 0.0
            b_gs = upper
            c_pt = b_gs - gr * (b_gs - a_gs)
            d_pt = a_gs + gr * (b_gs - a_gs)
            fc = f_1d_eval(c_pt, n, v, c_quad, a_lin, c_lin)
            fd = f_1d_eval(d_pt, n, v, c_quad, a_lin, c_lin)

            for _ in range(50):
                if b_gs - a_gs < 1e-13:
                    break
                if fc < fd:
                    b_gs = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = b_gs - gr * (b_gs - a_gs)
                    fc = f_1d_eval(c_pt, n, v, c_quad, a_lin, c_lin)
                else:
                    a_gs = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = a_gs + gr * (b_gs - a_gs)
                    fd = f_1d_eval(d_pt, n, v, c_quad, a_lin, c_lin)

            x_opt = max(0.0, (a_gs + b_gs) / 2.0)

            # Check x=0 boundary
            f0 = f_1d_eval(0.0, n, v, c_quad, a_lin, c_lin)
            fx = f_1d_eval(x_opt, n, v, c_quad, a_lin, c_lin)
            if f0 < fx:
                x_opt = 0.0

            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt
            conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_lin

            # Track best
            S_curr = float(np.sum(a))
            if S_curr > 1e-10:
                obj_curr = 2.0 * n * float(np.max(conv)) / (S_curr * S_curr)
                if obj_curr < best_obj:
                    best_obj = obj_curr
                    best_a = (a / S_curr).copy()

        # End of sweep
        S = float(np.sum(a))
        if S > 1e-10:
            a = a / S
            obj_now = compute_obj_normalized(a)
            if obj_now < best_obj:
                best_obj = obj_now
                best_a = a.copy()

        if best_obj < sweep_start_obj - 1e-12:
            no_improve = 0
        else:
            no_improve += 1

        # Annealing perturbation
        if no_improve >= 8 and sweep % 12 == 0:
            strength = 0.003 + 0.003 * min(no_improve // 12, 5)
            pert = rng.normal(0, strength, n)
            a_try = a * np.exp(pert)
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st > 1e-10:
                a_try = a_try / st
                obj_try = compute_obj_normalized(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                a = a_try
            no_improve = 0

        if verbose and sweep % 5 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length, return normalized."""
    n = len(a)
    if n == new_n:
        a = a.copy()
    elif n == 1:
        a = np.full(new_n, float(a[0]))
    else:
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, new_n)
        a = np.interp(x_new, x_old, a)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    return a


def lp_iteration(a, time_budget=300.0, seed=42, verbose=False):
    """
    LP-based iteration: solve LP, line-search blend, CD polish.
    Input normalized. Returns best a.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 25:
            break

        # Solve LP (use min-peak formulation)
        g_lp, M_lp = solve_lp_minpeak(a)

        if g_lp is None:
            # Fallback: max-sum formulation
            g_lp, _ = solve_lp_maxsum(a)
            if g_lp is None:
                if remaining > 30:
                    a = coordinate_descent(a, time_budget=min(remaining * 0.4, 60.0),
                                           seed=seed + it)
                    obj = compute_obj_normalized(a)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()
                    it += 1
                continue

        # Line search on blend parameter
        best_t_obj = best_obj
        best_a_blend = best_a.copy()

        for t in [0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_blend = a_try.copy()
            if obj_try > best_t_obj + 0.005:
                break

        if best_t_obj < best_obj - 1e-14:
            a = best_a_blend
            best_obj = best_t_obj
            best_a = a.copy()
            it += 1
            if verbose:
                print(f"    LP it{it}: obj={best_obj:.8f}, rem={remaining:.0f}s")
        else:
            # No improvement: do CD
            if remaining > 30:
                cd_b = min(remaining * 0.25, 50.0)
                a = coordinate_descent(a, time_budget=cd_b, seed=seed + 1000 + it)
                obj = compute_obj_normalized(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
            it += 1
            if verbose:
                print(f"    LP it{it} (CD): obj={best_obj:.8f}, rem={remaining:.0f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations, all normalized."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:

        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):

        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x * x + 1e-6)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:

def subgradient_step(a, step_size, rng):
    """
    One subgradient step for min max(conv(a,a)) s.t. sum(a)=1, a>=0.
    Uses the KKT subgradient of the max function.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    argmax_set = np.where(conv >= max_conv * (1.0 - 1e-9))[0]

    # Subgradient of max_k conv(a,a)[k] w.r.t. a
    # d(max conv)/d(a[i]) = sum_{k in argmax} 2*a[k-i] (if valid) / |argmax|
    v_sub = np.zeros(n)
    for k in argmax_set:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            v_sub[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1]
    v_sub /= max(len(argmax_set), 1)

    # Projected gradient step: minimize max_conv + <v_sub, a_new - a>
    # subject to sum(a_new)=1, a_new >= 0
    # Equivalent to: a_new = proj_simplex(a - step_size * v_sub)
    a_new = a - step_size * v_sub
    # Project onto simplex {x: sum(x)=1, x>=0}
    a_new = a_new - np.min(a_new - (1.0 - np.min(a_new + np.sort(a_new)[::-1] - np.arange(1, n + 1)[::-1]) / n))
    # Simpler: use watermark algorithm
    u = np.sort(a_new)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) > 0:
        theta = (cssv[rho[-1]] - 1.0) / (rho[-1] + 1.0)
        a_new = np.maximum(a_new - theta, 0)
    else:
        a_new = np.ones(n) / n

    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def subgradient_descent(a, time_budget=60.0, seed=42):
    """Subgradient descent with decreasing step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    eta = 0.5 / np.sqrt(n)
    patience = 0

    while time.monotonic() - start < time_budget:
        a_new = subgradient_step(a, eta, rng)
        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1
        eta *= 0.995
        if patience > 30:
            eta *= 0.5
            pert = rng.normal(0, 0.01, n)
            a = a * np.exp(pert)
            a = np.maximum(a, 0)
            a = a / a.sum()
            patience = 0
        if eta < 1e-6:
            break

    return best_a

    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 3: CD Polish (budget={min(remaining*0.15, 120):.0f}s) ===")
    p3 = min(remaining * 0.15, 120.0)
    if p3 > 15:
        a_cd2 = coordinate_descent(best_seq, time_budget=p3, seed=300, verbose=False)
        obj_cd2 = compute_obj_normalized(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  CD: obj={best_score:.8f} **")

    # === Phase 4: Multi-n exploration ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 4: Multi-n (budget={remaining:.0f}s) ===")
    n_cur = len(best_seq)
    n_candidates = sorted(set([
        n_cur - 100, n_cur + 50, n_cur + 100, n_cur + 200, n_cur + 400,
        600, 700, 900, 1000, 1200, 1500, 2000
    ]))
    n_candidates = [nc for nc in n_candidates if 400 <= nc <= 2500]
    n_filt = []
    for nc in n_candidates:
        if all(abs(nc - nf) > 70 for nf in n_filt):
            n_filt.append(nc)
    n_filt = n_filt[:5]
    print(f"  n candidates: {n_filt}")

    for n_try in n_filt:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break
        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.12, 45.0)
        if alloc > 10:
            a_try = coordinate_descent(a_try, time_budget=alloc, seed=n_try)
        obj_try = compute_obj_normalized(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Subgradient escape + CD ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 5: Subgradient + CD (budget={remaining:.0f}s) ===")
    if remaining > 40:
        n_cur = len(best_seq)
        for r in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break
            mode = 1 + r
            a_r = diverse_init(n_cur, rng, mode=mode)
            blend = 0.2 + 0.15 * r
            a_r = (1.0 - blend) * best_seq + blend * a_r
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

            alloc = min(remaining * 0.2, 40.0)
            if alloc > 10:
                a_r = subgradient_descent(a_r, time_budget=alloc * 0.5, seed=1000 + r)
                a_r = coordinate_descent(a_r, time_budget=alloc * 0.5, seed=2000 + r)
                obj_r = compute_obj_normalized(a_r)
                if obj_r < best_score:
                    best_score = obj_r
                    best_seq = a_r.copy()
                    print(f"  Restart {r}: obj={best_score:.8f} **")
                else:
                    print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 6: Final LP + CD polish ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 6: Final Polish (budget={remaining:.0f}s) ===")
    if remaining > 40:
        p6a = min(remaining * 0.4, 100.0)
        if p6a > 20:
            a_f = lp_iteration(best_seq, time_budget=p6a, seed=5000, verbose=False)
            obj_f = compute_obj_normalized(a_f)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        p6b = min(remaining * 0.6, 100.0)
        if p6b > 15:
            a_f2 = coordinate_descent(best_seq, time_budget=p6b, seed=6000, verbose=False)
            obj_f2 = compute_obj_normalized(a_f2)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    best_seq = np.maximum(best_seq, 0.0)
    result = [float(min(1000.0, max(0.0, x))) for x in best_seq]

    val = evaluate_sequence(result)
    if np.isinf(val):
        result = [1.0 / 100.0] * 100
        best_score = 2.0
    else:
        best_score = min(best_score, val)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
