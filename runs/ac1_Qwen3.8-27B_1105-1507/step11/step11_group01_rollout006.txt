                    else:
                        x_lo = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = x_lo + gr * (x_hi - x_lo)
                        fd = eval_1d(d_pt)

                x_opt = max(0.0, (x_lo + x_hi) / 2.0)
                f_opt = eval_1d(x_opt)
                if f_opt < best_f:
                    best_x = x_opt
                    best_f = f_opt

            if abs(best_x - v) > 1e-14:
                delta = best_x - v
                a[i] = best_x
                # Update conv
                if 0 <= k_2i <= 2 * n - 2:
                    conv[k_2i] += best_x * best_x - v * v
                for k in range(k_min, k_max + 1):
                    if k == k_2i:
                        continue
                    if 0 <= k <= 2 * n - 2:
                        j = k - i
                        if 0 <= j < n and j != i:
                            conv[k] += 2.0 * delta * a[j]
                S += delta
                improved = True

        if not improved:
            stall += 1
            if stall > 3:
                break
        else:
            if best_obj < prev_best - 1e-12:
                stall = 0
            prev_best = best_obj

        if verbose:
            elapsed = time.monotonic() - start
            print(f"    CD: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def simulated_annealing(a, time_budget=100.0, seed=42, verbose=False):
    """Simulated annealing with structured perturbations."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    cur_obj = best_obj
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    T_init = 0.05
    T_end = 0.001
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        T = T_init * (T_end / T_init) ** progress

        # Choose perturbation type
        ptype = rng.random()
        a_try = a.copy()

        if ptype < 0.3:
            # Single coordinate perturbation
            i = rng.integers(n)
            scale = max(a[i], S / n * 0.5)
            a_try[i] = max(0.0, a[i] + rng.normal(0, scale))
        elif ptype < 0.5:
            # Block perturbation
            block_len = rng.integers(1, max(2, n // 10))
            start_i = rng.integers(0, n - block_len)
            a_try[start_i:start_i + block_len] += rng.normal(0, S / n * 0.3, block_len)
            a_try = np.maximum(a_try, 0)
        elif ptype < 0.7:
            # Gaussian perturbation (small)
            a_try = a * np.exp(rng.normal(0, 0.05, n))
        elif ptype < 0.85:
            # Mirror/symmetry operation
            if n >= 4:
                half = n // 2
                swap_prob = 0.3
                for i in range(half):
                    if rng.random() < swap_prob:
                        a_try[i], a_try[n - 1 - i] = a_try[n - 1 - i], a_try[i]
            a_try = np.maximum(a_try, 0)
        else:
            # RBFS-like: keep best half, perturb worst half
            idx = np.argsort(a)
            worst_half = idx[:n // 2]
            a_try[worst_half] = rng.uniform(0, S / n, n // 2)

        S_try = float(np.sum(a_try))
        if S_try < 1e-10:
            continue
        a_try = a_try / S_try
        obj_try = compute_obj_normalized(a_try)

        # Acceptance
        if obj_try < cur_obj or rng.random() < math.exp(-(obj_try - cur_obj) / max(T, 1e-10)):
            a = a_try
            cur_obj = obj_try
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        it += 1
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    SA it{it}: best={best_obj:.8f}, T={T:.6f}, t={elapsed:.1f}s")

        if no_improve > 5000:
            break

    return best_a


def gradient_descent_smoothed(a, time_budget=100.0, seed=42, verbose=False):
    """
    Smoothed gradient descent using softmax surrogate for the max.
    With Nesterov acceleration and projection onto non-negative simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    beta = 0.8
    v = np.zeros(n)
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = 20.0 + 500.0 * progress  # Temperature increases over time
        lr = 2.0 / (np.sqrt(n) * (1.0 + progress))

        y = a + beta * v
        y = np.maximum(y, 0)
        Sy = float(np.sum(y))
        if Sy < 1e-10:
            y = best_a.copy()
            Sy = 1.0
        else:
            y = y / Sy

        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        shifted = conv - max_conv
        exp_vals = np.exp(tau * shifted)
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient of max(conv) w.r.t. a using FFT
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A_fft = fft(y, N)
        g_full = ifft(P * np.conj(A_fft)).real
        grad = 2.0 * g_full[:n]

        # Normalize gradient
        g_norm = float(np.linalg.norm(grad))
        if g_norm > 1e-15:
            grad = grad / g_norm

        v = beta * v - lr * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        S_new = float(np.sum(a_new))
        if S_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / S_new

        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        it += 1

        if no_improve > 3000:
            break

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    GD it{it}: best={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def lp_minpeak(a, time_budget=30.0):
    """
    LP: minimize M subject to conv(a, g)[k] <= M, sum(g) = 1, g >= 0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a_norm = a / s

    # Build constraint matrix
    num_constraints = 2 * n - 1
    A = np.zeros((num_constraints, n))
    for j in range(n):
        for k in range(max(0, j), min(2 * n - 1, j + n)):
            i = k - j
            if 0 <= i < n:
                A[k, j] = a_norm[i]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    return g_val, float(M.value) if M.value is not None else float(np.max(A @ g_val))
        except Exception:
            continue
    return None, None


def lp_polish_loop(a, time_budget=60.0, seed=42, verbose=False):
    """Iterative LP polishing with blending."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        g_lp, M_lp = lp_minpeak(a, time_budget=min(remaining * 0.5, 20.0))
        if g_lp is None:
            break

        best_t_obj = best_obj
        best_t_a = best_a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_t_obj - 1e-14:
                best_t_obj = obj_try
                best_t_a = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 3:
                pert = rng.normal(0, 0.03, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        c1, c2 = n // 4, 3 * n // 4
        w = n // 8
        x = np.arange(n)
        a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine distribution
        x = np.linspace(-1 + 1e-6, 1 - 1e-6, n)
        a = 1.0 / np.sqrt(1 - x * x)
        a = a / a.sum()
    elif mode == 4:
        # Cosine window
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    elif mode == 5:
        # Power-law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        # Frequency modulation
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.sin(2 * np.pi * rng.integers(1, 5) * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 7:
        # Triangular
        x = np.linspace(0, 1, n)
        a = 1 - np.abs(2 * x - 1)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines multiple optimization strategies:
    coordinate descent, simulated annealing, smoothed gradient descent,
    LP polishing, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj_normalized(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj_normalized(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting with n={n_best}, score={best_score:.8f}")

    def try_update(a_try, label=""):
        nonlocal best_seq, best_score
        if a_try is None or len(a_try) == 0:
            return
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            return
        a_try = a_try / s
        obj = compute_obj_normalized(a_try)
        if obj < best_score - 1e-14:
            best_score = obj
            best_seq = a_try.copy()
            print(f"  ** {label}: n={len(a_try)}, obj={best_score:.8f} **")
        else:
            print(f"  {label}: obj={obj:.8f} (no improvement)")

    # === Phase 1: Coordinate Descent (main workhorse) ===
    print(f"\n=== Phase 1: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 400.0)
    if p1_budget > 30:
        a_cd = coordinate_descent_fast(best_seq, time_budget=p1_budget, seed=seed, verbose=True)
        try_update(a_cd, "CD")

    # === Phase 2: Simulated Annealing ===
    print(f"\n=== Phase 2: Simulated Annealing ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 200.0)
    if p2_budget > 30:
        a_sa = simulated_annealing(best_seq, time_budget=p2_budget, seed=seed + 100, verbose=True)
        try_update(a_sa, "SA")
        # Polish SA result with CD
        remaining = deadline - time.monotonic()
        if remaining > 30 and best_seq is not a_sa:
            a_cd2 = coordinate_descent_fast(a_sa, time_budget=min(60, remaining * 0.1), seed=seed + 200)
            try_update(a_cd2, "CD after SA")
        elif remaining > 30:
            a_cd2 = coordinate_descent_fast(a_sa, time_budget=min(60, remaining * 0.1), seed=seed + 200)
            try_update(a_cd2, "CD after SA")

    # === Phase 3: Smoothed Gradient Descent ===
    print(f"\n=== Phase 3: Smoothed Gradient Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 150.0)
    if p3_budget > 20:
        a_gd = gradient_descent_smoothed(best_seq, time_budget=p3_budget, seed=seed + 300, verbose=True)
        try_update(a_gd, "GD")
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_cd3 = coordinate_descent_fast(a_gd, time_budget=min(40, remaining * 0.1), seed=seed + 400)
            try_update(a_cd3, "CD after GD")

    # === Phase 4: LP Polishing ===
    print(f"\n=== Phase 4: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 200.0)
    if p4_budget > 30:
        a_lp = lp_polish_loop(best_seq, time_budget=p4_budget, seed=seed + 500, verbose=True)
        try_update(a_lp, "LP")
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_cd4 = coordinate_descent_fast(a_lp, time_budget=min(40, remaining * 0.1), seed=seed + 600)
            try_update(a_cd4, "CD after LP")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_list = sorted(set([
        max(100, n_current - 100),
        n_current,
        n_current + 100,
        n_current + 200,
        n_current + 400,
    ]))
    # Add some fixed values
    for nv in [200, 300, 400, 500, 600, 700, 800, 900, 1000, 1200, 1500]:
        if not any(abs(nv - x) < 50 for x in n_list):
            n_list.append(nv)
    n_list = [x for x in n_list if 100 <= x <= 2000]
    n_list = sorted(n_list)[:8]
    print(f"  n candidates: {n_list}")

    for n_try in n_list:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        if n_try == len(best_seq):
            continue

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.12, 50.0)
        if alloc > 10:
            a_try = coordinate_descent_fast(a_try, time_budget=alloc, seed=n_try + 1000)
        try_update(a_try, f"n={n_try}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = (r + 1) % 8
        a_r = diverse_init(n_cur, rng, mode=mode)
        # Blend with current best
        blend = 0.4 + 0.05 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.12, 50.0)
        if alloc > 10:
            a_r = coordinate_descent_fast(a_r, time_budget=alloc, seed=5000 + r * 100)
        try_update(a_r, f"Restart{r}(mode{mode})")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Alternating CD and SA
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            # CD
            alloc_cd = min(remaining * 0.35, 80.0)
            if alloc_cd > 10:
                a_f = coordinate_descent_fast(best_seq, time_budget=alloc_cd, seed=9000 + rnd)
                try_update(a_f, f"FinalCD{rnd}")
            # SA (short)
            remaining = deadline - time.monotonic()
            alloc_sa = min(remaining * 0.2, 30.0)
            if alloc_sa > 10:
                a_f2 = simulated_annealing(best_seq, time_budget=alloc_sa, seed=9500 + rnd)
                try_update(a_f2, f"FinalSA{rnd}")
            # CD again
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_f3 = coordinate_descent_fast(best_seq, time_budget=min(20, remaining * 0.3),
                                                seed=9800 + rnd)
                try_update(a_f3, f"FinalCD2_{rnd}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = 2.0

    best_seq = np.maximum(best_seq, 0.0)
    result = [float(min(1000.0, max(0.0, x))) for x in best_seq]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        print(f"\n  evaluate_sequence says: {val:.8f}")
        if val != best_score:
            print(f"  (computed: {best_score:.8f})")
    else:
        print(f"\n  evaluate_sequence returned inf! Falling back to uniform.")
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
