One more thought: the objective `2n * max_conv / sum^2` for normalized (sum=1) sequences is just `2n * max_conv`. The key insight is that we want to distribute the convolution mass as evenly as possible.

For a sequence with "bandwidth" W (i.e., most of its energy is in frequencies up to W), the convolution has support of width about 2W. If W << n, then the convolution is concentrated in a small region, making the max large. If W ≈ n/2, the convolution is spread out.

So we want a sequence with "wide" frequency content, i.e., one that oscillates a lot. But it must be non-negative! This is a key constraint.

A non-negative sequence with wide frequency content is like a "windowed" version of a high-frequency signal. The best such sequences are related to prolate spheroidal wave functions (PSWFs) - they maximize the energy in a given bandwidth.

But we want the OPPOSITE: we want to MINIMIZE the peak of the autocorrelation, which means we want the autocorrelation to be as flat as possible. This is the "flat-top" or "low PAPR" problem.

For non-negative sequences, the optimal shapes are known to be somewhat "U-shaped" or "bimodal" - concentrated at the edges but not too much. This matches what the previous approach found.

Let me also consider: what if we use a sequence that's approximately constant on two intervals (left and right) with a gap in the middle? Like a[0..m-1] = c1, a[m..n-m-1] = c2, a[n-m..n-1] = c1 with c1 > c2.

This would give a bi-modal distribution. The convolution would have:
- A peak at 0 (from c1^2 * m terms)
- A peak at n-1 (from the cross terms)
- A dip in the middle

The trade-off is between the edge peaks and the middle peak.

"""
Fast coordinate descent with incremental convolution updates + LP polishing.
Key innovation: O(n) per coordinate update via incremental conv maintenance,
allowing thousands of sweeps in the time budget. Combined with periodic LP
polishing and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n


def fast_coordinate_descent(a, time_budget, verbose=False, candidate_strategy="auto"):
    """
    Fast coordinate descent with incremental convolution updates.
    Each coordinate update is O(n) instead of O(n log n).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    s = 1.0

    conv = compute_conv(a)
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    sweep = 0
    no_improve = 0
    patience = 5

    while time.monotonic() - start < time_budget:
        obj_at_sweep_start = best_obj

        for i in range(n):
            if time.monotonic() - start >= time_budget:
                break

            ai = a[i]

            # Determine range of affected conv entries
            lo = i  # k-i >= 0 => k >= i
            hi = min(2 * n - 2, i + n - 1)  # k-i <= n-1 => k <= i+n-1

            # Save the slice of a that will be needed
            if lo <= hi:
                a_slice = a[lo - i: hi - i + 1].copy()
                conv_slice = conv[lo: hi + 1].copy()
            else:
                a_slice = None
                conv_slice = None

            # Get unchanged max (for comparison)
            if lo > 0 or hi < 2 * n - 2:
                if lo > 0 and hi < 2 * n - 2:
                    unchanged_max = max(float(np.max(conv[:lo])), float(np.max(conv[hi + 1:])))
                elif lo > 0:
                    unchanged_max = float(np.max(conv[:lo]))
                else:
                    unchanged_max = float(np.max(conv[hi + 1:]))
            else:
                unchanged_max = float('-inf')

            # Generate candidates
            if ai < 1e-12:
                candidates = [0.0, 0.3 / n, 1.0 / n, 2.0 / n, 5.0 / n]
            else:
                candidates = [ai, 0.0, ai * 0.5, ai * 1.5]
                # Add equalization candidate
                if lo <= hi:
                    cur_max = float(np.max(conv))
                    # Find second peak
                    conv_copy = conv.copy()
                    conv_copy[int(np.argmax(conv))] = -np.inf
                    second_max = float(np.max(conv_copy)) if np.any(conv_copy > -np.inf) else 0.0
                    # Try to find delta that equalizes
                    # conv[peak] + 2*delta*a[peak-i] = second_max
                    peak_idx = int(np.argmax(conv))
                    if 0 <= peak_idx - i < n:
                        coeff = 2.0 * a[peak_idx - i]
                        if coeff > 1e-15:
                            delta_eq = (second_max - cur_max) / coeff
                            if delta_eq < -ai + 1e-12:
                                delta_eq = -ai + 0.01 * ai  # Can't go below 0
                            elif delta_eq > 0:
                                delta_eq = min(delta_eq, 3.0 * ai)
                            else:
                                delta_eq = 0.5 * ai
                            candidates.append(ai + delta_eq)
                # Gradient-based: reduce if this coordinate contributes to peak
                candidates.append(ai * 0.8)
                candidates.append(ai * 1.2)

            best_val = ai
            best_local = 2.0 * n * float(np.max(conv))  # current objective (s=1)

            for cv in candidates:
                if cv < -1e-15:
                    continue
                cv = max(0.0, cv)
                delta = cv - ai
                if abs(delta) < 1e-15:
                    continue

                s_new = s + delta
                if s_new < 0.01:
                    continue

                if lo <= hi:
                    new_changed = conv_slice + 2.0 * delta * a_slice
                    if 2 * i >= lo and 2 * i <= hi:
                        new_changed[2 * i - lo] += delta * delta
                    changed_max = float(np.max(new_changed))
                    new_max = max(changed_max, unchanged_max)
                else:
                    new_max = float(np.max(conv))

                obj_try = 2.0 * n * new_max / (s_new * s_new)
                if obj_try < best_local - 1e-15:
                    best_local = obj_try
                    best_val = cv

            if best_val != ai:
                delta = best_val - ai
                if lo <= hi:
                    conv[lo: hi + 1] += 2.0 * delta * a_slice
                    if 2 * i >= lo and 2 * i <= hi:
                        conv[2 * i] += delta * delta
                a[i] = best_val
                s += delta

        # Renormalize and recompute conv to avoid drift
        if s > 0.01:
            a = a / s
            s = 1.0
            conv = compute_conv(a)

        obj_now = 2.0 * n * float(np.max(conv))
        if obj_now < best_obj - 1e-14:
            best_obj = obj_now
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        sweep += 1
        if verbose and sweep % 5 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

        if no_improve >= patience:
            break

    return best_a


def lse_gradient_descent(a, time_budget, T_start=5.0, T_end=50.0, verbose=False):
    """LSE smoothed gradient descent with FFT-based gradient."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    lr = 0.5 / n
    it = 0
    best_it = 0
    patience = 500

    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** frac

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        exp_vals = np.exp(T * (conv - cmax))
        w = exp_vals / exp_vals.sum()

        # Gradient via correlation (O(n^2) but fast for n=800)
        grad = np.zeros(n)
        for i in range(n):
            lo1 = max(0, i)
            hi1 = min(2 * n - 2, i + n - 1)
            if lo1 <= hi1:
                grad[i] += np.dot(w[lo1:hi1 + 1], a[lo1 - i:hi1 - i + 1])
            lo2 = max(0, i - n + 1)
            hi2 = min(2 * n - 2, i)
            if lo2 <= hi2:
                grad[i] += np.dot(w[lo2:hi2 + 1], a[lo2:hi2 + 1][::-1])
        grad *= 2.0 * n

        # Projected gradient step with backtracking
        a_old = a.copy()
        obj_old = 2.0 * n * (cmax + np.log(float(np.sum(exp_vals))) / T)
        t = lr
        improved = False
        for _ in range(8):
            a_try = a_old - t * grad
            a_try = project_simplex(a_try)
            conv_try = compute_conv(a_try)
            cmax_try = float(np.max(conv_try))
            exp_try = np.exp(T * (conv_try - cmax_try))
            obj_try = 2.0 * n * (cmax_try + np.log(float(np.sum(exp_try))) / T)
            if obj_try < obj_old - 1e-12:
                a = a_try
                improved = True
                break
            t *= 0.5

        if not improved:
            break

        obj_true = compute_obj_norm(a)
        if obj_true < best_obj:
            best_obj = obj_true
            best_a = a.copy()
            best_it = it

        it += 1
        if it - best_it > patience:
            a = best_a.copy()
            lr *= 0.7
            patience = max(100, patience // 2)
            best_it = it

    return best_a


def truncated_lp_solve(a, top_m):
    """Solve truncated LP for better direction."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv_a = compute_conv(a_norm)
    top_indices = np.argsort(conv_a)[::-1][:top_m]

    A = np.zeros((len(top_indices), n))
    for idx, k in enumerate(top_indices):
        k = int(k)
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                A[idx, m] = a_norm[j]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def lp_polish(a, time_budget, top_m=20, verbose=False):
    """LP polishing loop."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget and it < 30:
        g_lp = truncated_lp_solve(best_a, top_m)
        if g_lp is None:
            break

        best_t = 0.0
        best_obj_t = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * best_a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj_try = compute_obj_norm(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t

        if best_t > 0 and best_obj_t < best_obj - 1e-14:
            a_new = (1 - best_t) * best_a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            best_obj = best_obj_t
            best_a = a_new
            stall = 0
        else:
            stall += 1

        it += 1
        if stall >= 3:
            break

    return best_a


def u_shaped_init(n, sharpness):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    return a / a.sum()


def arcsine_init(n):
    x = np.linspace(0.005, 0.995, n)
    a = 1.0 / np.sqrt(np.maximum(1e-10, 1 - (2 * x - 1) ** 2))
    return a / a.sum()


def power_init(n, alpha):
    x = np.linspace(0, 1, n)
    d = np.abs(x - 0.5)
    a = np.maximum(d, 1.0 / (2 * n)) ** (-alpha) + 0.01
    return a / a.sum()

                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  Perturb {r}: obj={best_score:.8f} **")
                else:
                    print(f"  Perturb {r}: obj={obj_pert:.8f}")

            phase = 0
            cd_cycles += 1

            # After 3 full cycles, do only CD
            if cd_cycles >= 3:
                phase = 0

    # === Final polish ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print(f"\n=== Final Polish (remaining={remaining:.0f}s) ===")
        a_final = fast_coordinate_descent(best_seq, time_budget=min(remaining - 5, 60.0))
        obj_final = compute_obj_norm(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_norm(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
