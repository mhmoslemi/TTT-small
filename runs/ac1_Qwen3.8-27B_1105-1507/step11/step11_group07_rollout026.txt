    size = 1
    while size < n + len(p) - 1:
        size <<= 1
    A = rfft(a_rev, size)
    P = rfft(p, size)
    full = irfft(A * P, size)
    # The cross-correlation at lag 0 is at index n-1
    # grad[i] = 2 * full[n-1 + i] ... let me verify
    # full[m] = sum_j a_rev[j] * p[m-j] = sum_j a[n-1-j] * p[m-j]
    # Let i' = n-1-j, so j = n-1-i':
    # full[m] = sum_i' a[i'] * p[m-(n-1-i')] = sum_i' a[i'] * p[m-n+1+i']
    # We want grad[i] = 2 * sum_k p[k] * a[k-i] (for valid k-i, i.e., 0<=k-i<n)
    # = 2 * sum_{k-i} a[k-i] * p[k]
    # Let j = k-i: = 2 * sum_j a[j] * p[j+i]
    # From full[m] with m = n-1+i: full[n-1+i] = sum_j a[j] * p[n-1+i-(n-1-j)] = sum_j a[j] * p[i+j]
    # Yes! So grad[i] = 2 * full[n-1+i]
    
    q = full[n-1:n-1+n]
    return 2.0 * q

def build_conv_matrix(a):
    n = len(a)
    # A is (2n-1) x n, A[k, i] = a[k-i] if 0<=k-i<n else 0
    # conv(a, g)[k] = sum_i A[k,i] * g[i]
    A = np.zeros((2*n-1, n))
    for i in range(n):
        # A[k, i] = a[k-i] for valid k
        # k-i must be in [0, n-1], so k in [i, i+n-1]
        lo = i
        hi = min(2*n-1, i+n)
        A[lo:hi, i] = a[:hi-lo]
    return A

def solve_lp(a, M_target=None, time_budget=5.0):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum g = 1, g >= 0"""
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    if M_target is None:
        conv = compute_conv(a_norm)
        M_target = float(np.max(conv)) * 0.999
    
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    if M_target is not None:
        constraints.append(M <= M_target)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver, time_limit=time_budget)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

Refined optimization for step function coefficient minimization.
Strategy: High-beta soft-max gradient descent with adaptive restarts,
subgradient polishing using exact max, and intensive LP refinement.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective assuming we'll normalize. Returns 2*n*max(conv) for sum=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_obj_and_conv(a):
    """Returns (objective, conv, max_position) for normalized a."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf, None, 0
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_pos = int(np.argmax(conv))
    return 2.0 * n * float(conv[max_pos]), conv, max_pos


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1.0)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_max_grad(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*conv_k)) at a (sum(a)=1).
    Uses FFT cross-correlation for O(n log n) efficiency.
    """
    n = len(a)
    conv = compute_conv(a)
    c_max = float(np.max(conv))
    exp_vals = np.exp(beta * (conv - c_max))
    p = exp_vals / float(np.sum(exp_vals))
    # grad[i] = 2 * sum_j a[j] * p[i+j] (for valid i+j < 2n-1)
    a_rev = a[::-1]
    p_len = len(p)
    size = 1
    while size < n + p_len - 1:
        size <<= 1
    A = rfft(a_rev, size)
    P = rfft(p, size)
    full = irfft(A * P, size)
    q = full[n - 1:n - 1 + n]
    return 2.0 * q


def subgradient_step(a, lr):
    """
    Subgradient step for the EXACT objective (non-smooth max).
    Uses the subgradient at the argmax position.
    """
    n = len(a)
    conv = compute_conv(a)
    max_pos = int(np.argmax(conv))
    # subgrad[i] = 2 * a[max_pos - i] if 0 <= max_pos - i < n, else 0
    subgrad = np.zeros(n)
    lo = max(0, max_pos - n + 1)
    hi = min(n - 1, max_pos)
    for i in range(lo, hi + 1):
        subgrad[i] = 2.0 * a[max_pos - i]
    # Also, positions where max is achieved contribute. For ties, average.
    # For simplicity, just use argmax.
    a_new = a - lr * subgrad
    return project_simplex(a_new), max_pos

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        lo = i
        hi = min(2 * n - 1, i + n)
        A[lo:hi, i] = a[:hi - lo]
    return A


def solve_lp(a, M_frac=1.0, time_limit=5.0):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M * current_max * M_frac,
    sum g = 1, g >= 0.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv_current = compute_conv(a_norm)
    M_current = float(np.max(conv_current))

    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M * M_frac * M_current, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver, time_limit=time_limit)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_unconstrained(a, time_limit=5.0):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum g = 1, g >= 0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver, time_limit=time_limit)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget=120.0, max_iters=40, verbose=False, m_fracs=None):
    """
    LP-based polishing with multiple M fractions for aggressive improvement.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(999)

    if m_fracs is None:
        m_fracs = [1.0, 0.99, 0.98, 0.97, 0.95]

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Cycle through M fractions, starting permissive
        frac_idx = min(stall, len(m_fracs) - 1)
        m_frac = m_fracs[frac_idx]

        if m_frac == 1.0:
            g_lp, M_lp = solve_lp_unconstrained(a, time_limit=min(5.0, remaining * 0.08))
        else:
            g_lp, M_lp = solve_lp(a, M_frac=m_frac, time_limit=min(5.0, remaining * 0.08))

        if g_lp is None:
            stall += 1
            it += 1
            continue

        # Line search for optimal blend
        best_t = 0.0
        best_blend_obj = best_obj
        for t in np.linspace(0, 1.0, 13):
            a_try = (1 - t) * a + t * g_lp
            a_try = project_simplex(np.maximum(a_try, 0))
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_t = t

        if best_t > 0 and best_blend_obj < best_obj - 1e-14:
            a_new = project_simplex(np.maximum((1 - best_t) * a + best_t * g_lp, 0))
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 5 == 0:
            print(f"        LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, "
                  f"frac={m_frac:.2f}, elapsed={time.monotonic()-start:.1f}s")

        if stall >= 6:
            noise = rng.normal(0, 0.002 * (1 + stall), n)
            a = project_simplex(best_a + noise)
            stall = 0

    return best_a, best_obj


def subgradient_polish(a, time_budget=60.0, lr_start=0.001, verbose=False):
    """Polish using exact subgradient of the max objective."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_start
    it = 0
    stall = 0
    rng = np.random.default_rng(555)

    while time.monotonic() - start < time_budget:
        a_new, max_pos = subgradient_step(a, lr)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            lr = min(lr * 1.05, lr_start)  # slightly increase if making progress
        else:
            stall += 1
            lr *= 0.95
            if stall > 100:
                a = project_simplex(best_a + rng.normal(0, 0.0005, n))
                stall = 0
                lr = lr_start * 0.5

        a = a_new
        it += 1

        if verbose and it % 5000 == 0:
            print(f"        SubGrad it{it}: obj={best_obj:.8f}, lr={lr:.6f}, "
                  f"t={time.monotonic()-start:.1f}s")

        if it % 5000 == 0 and it > 10000:
            lr *= 0.8

    return best_a, best_obj


def u_shaped_init(n, sharpness=3.0):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    return project_simplex(a)


def poly_init(n, p=2.0, alpha=1.0):
    """a[i] proportional to 1 - alpha*(2i/n - 1)^p, clamped to positive."""
    x = np.linspace(0, 1, n)
    t = 2 * x - 1
    a = 1.0 - alpha * (t ** p)
    a = np.maximum(a, 0.01)
    return project_simplex(a)


def gaussian_init(n, sigma=0.3, center=0.5):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return project_simplex(a)


def bimodal_init(n, sigma=0.1, centers=(0.3, 0.7)):
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    for c in centers:
        a += np.exp(-0.5 * ((x - c) / sigma) ** 2)
    return project_simplex(a)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Multi-phase optimization combining soft-max GD (high beta),
    subgradient polishing, and intensive LP refinement.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        best_seq = u_shaped_init(800, 3.0)
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    # Try diverse parametric initializations at current length
    print("\n=== Trying parametric inits ===")
    inits = [
        u_shaped_init(n_current, 2.0),
        u_shaped_init(n_current, 4.0),
        u_shaped_init(n_current, 5.0),
        poly_init(n_current, 2, 0.5),
        poly_init(n_current, 2, 1.0),
        poly_init(n_current, 2, 1.5),
        poly_init(n_current, 2, 2.0),
        poly_init(n_current, 3, 1.0),
        poly_init(n_current, 4, 1.5),
        gaussian_init(n_current, 0.25, 0.5),
        gaussian_init(n_current, 0.35, 0.5),
        bimodal_init(n_current, 0.12, (0.25, 0.75)),
        bimodal_init(n_current, 0.15, (0.3, 0.7)),
    ]
    for init in inits:
        obj_i = compute_obj(init)
        if obj_i < best_score:
            best_score = obj_i
            best_seq = init.copy()
            print(f"  Better init: obj={best_score:.8f} **")

    # Phase 1: Soft-max GD moderate beta
    remaining = deadline - time.monotonic()
    if remaining > 150:
        alloc = min(remaining * 0.22, 320.0)
        print(f"\n=== Phase 1: SmaxGD beta 5->40, alloc={alloc:.0f}s ===")
        a1, obj1 = softmax_gd(best_seq, alloc, 5.0, 40.0, 0.012, 0.9,
                              seed=seed, verbose=True, stall_threshold=250)
        if obj1 < best_score:
            best_score, best_seq = obj1, a1.copy()
            print(f"  Phase1: {best_score:.8f} **")
        else:
            print(f"  Phase1: {obj1:.8f} (no improvement)")

    # Phase 2: Soft-max GD high beta
    remaining = deadline - time.monotonic()
    if remaining > 120:
        alloc = min(remaining * 0.25, 350.0)
        print(f"\n=== Phase 2: SmaxGD beta 40->150, alloc={alloc:.0f}s ===")
        a2, obj2 = softmax_gd(best_seq, alloc, 40.0, 150.0, 0.006, 0.93,
                              seed=seed + 1, verbose=True, stall_threshold=300)
        if obj2 < best_score:
            best_score, best_seq = obj2, a2.copy()
            print(f"  Phase2: {best_score:.8f} **")
        else:
            print(f"  Phase2: {obj2:.8f} (no improvement)")

    # Phase 3: Subgradient polish
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.1, 120.0)
        print(f"\n=== Phase 3: Subgradient, alloc={alloc:.0f}s ===")
        a3, obj3 = subgradient_polish(best_seq, alloc, lr_start=0.0008, verbose=True)
        if obj3 < best_score:
            best_score, best_seq = obj3, a3.copy()
            print(f"  Phase3: {best_score:.8f} **")
        else:
            print(f"  Phase3: {obj3:.8f} (no improvement)")

    # Phase 4: LP Polish (intensive)
    remaining = deadline - time.monotonic()
    if remaining > 80:
        alloc = min(remaining * 0.2, 250.0)
        print(f"\n=== Phase 4: LP Polish, alloc={alloc:.0f}s ===")
        a4, obj4 = lp_polish(best_seq, alloc, max_iters=50, verbose=True,
                             m_fracs=[1.0, 0.995, 0.99, 0.985, 0.98])
        if obj4 < best_score:
            best_score, best_seq = obj4, a4.copy()
            print(f"  Phase4: {best_score:.8f} **")
        else:
            print(f"  Phase4: {obj4:.8f} (no improvement)")

    # Phase 5: Length exploration
    remaining = deadline - time.monotonic()
    if remaining > 120:
        print(f"\n=== Phase 5: Length Exploration ===")
        n_best = len(best_seq)
        n_options = [n for n in [600, 700, 900, 1000, 1100, 1200, 1400, 1600]
                     if abs(n - n_best) >= 100]
        n_options = n_options[:4]
        print(f"  Current n={n_best}, trying: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 80:
                break
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = project_simplex(a_try)

            alloc_sgd = min(remaining * 0.35, 100.0)
            if alloc_sgd > 20:
                a_try, obj_try = softmax_gd(a_try, alloc_sgd, 10.0, 60.0,
                                            0.008, 0.91, seed=seed + 500,
                                            verbose=False, stall_threshold=200)

            remaining = deadline - time.monotonic()
            if remaining > 30:
                alloc_lp = min(remaining * 0.25, 60.0)
                a_try, obj_try = lp_polish(a_try, alloc_lp, max_iters=20, verbose=False,
                                           m_fracs=[1.0, 0.99, 0.98])

            if obj_try < best_score:
                best_score, best_seq = obj_try, a_try.copy()
                print(f"  ** n={n_try}: {best_score:.8f} **")
            else:
                print(f"  n={n_try}: {obj_try:.8f}")

    # Phase 6: Final very high beta soft-max
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(remaining * 0.35, 180.0)
        print(f"\n=== Phase 6: SmaxGD beta 100->300, alloc={alloc:.0f}s ===")
        a6, obj6 = softmax_gd(best_seq, alloc, 100.0, 300.0, 0.002, 0.95,
                              seed=seed + 800, verbose=True, stall_threshold=400)
        if obj6 < best_score:
            best_score, best_seq = obj6, a6.copy()
            print(f"  Phase6: {best_score:.8f} **")
        else:
            print(f"  Phase6: {obj6:.8f} (no improvement)")

    # Phase 7: Final LP + subgradient
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc_lp = min(remaining * 0.5, 100.0)
        print(f"\n=== Phase 7: Final LP+SubGrad, alloc={alloc_lp:.0f}s ===")
        a7, obj7 = lp_polish(best_seq, alloc_lp * 0.6, max_iters=25, verbose=True,
                             m_fracs=[1.0, 0.995, 0.99])
        if obj7 < best_score:
            best_score, best_seq = obj7, a7.copy()
            print(f"  Final LP: {best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a7b, obj7b = subgradient_polish(best_seq, min(remaining * 0.5, 40.0),
                                            lr_start=0.0003, verbose=False)
            if obj7b < best_score:
                best_score, best_seq = obj7b, a7b.copy()
                print(f"  Final SubGrad: {best_score:.8f} **")

    # Phase 8: Last-ditch perturbations
    remaining = deadline - time.monotonic()
    if remaining > 20:
        for attempt in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            for noise_scale in [0.001, 0.002, 0.003]:
                noise = rng.normal(0, noise_scale, len(best_seq))
                a_p = project_simplex(best_seq + noise)
                obj_p = compute_obj(a_p)
                if obj_p < best_score:
                    best_score, best_seq = obj_p, a_p.copy()
                    print(f"  Perturb {attempt} (s={noise_scale}): {best_score:.8f} **")
                # Quick LP on the perturbed version
                remaining = deadline - time.monotonic()
                if remaining > 10 and (obj_p < best_score + 0.001):
                    a_p2, obj_p2 = lp_polish(a_p, min(remaining * 0.4, 15.0),
                                             max_iters=8, verbose=False,
                                             m_fracs=[1.0, 0.99])
                    if obj_p2 < best_score:
                        best_score, best_seq = obj_p2, a_p2.copy()
                        print(f"  Perturb+LP {attempt}: {best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
