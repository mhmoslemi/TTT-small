```python
def frank_wolfe(g, n_steps=1000):
    n = len(g)
    g = g / sum(g)  # normalize
    for t in range(n_steps):
        # Compute convolution
        C = convolve(g, g)  # length 2n-1
        max_idx = argmax(C)
        max_val = C[max_idx]
        
        # Subgradient: gradient of C[max_idx] w.r.t. g
        # dC[k]/dg[j] = 2*g[k-j] if 0<=k-j<n and k!=2j
        #              = 2*g[j] if k==2j
        #              = 0 otherwise
        k = max_idx
        sg = np.zeros(n)
        for j in range(n):
            if k - j >= 0 and k - j < n:
                if k == 2*j:
                    sg[j] = 2 * g[j]
                else:
                    sg[j] = 2 * g[k-j]
        
        # Linear min oracle: find j with smallest sg[j]
        j_min = argmin(sg)
        v = np.zeros(n)
        v[j_min] = 1
        
        # Step size
        gamma = 2 / (t + 2)
        
        # Update
        g = (1 - gamma) * g + gamma * v
    
    return g
```

```python
"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth gradient descent (log-sum-exp) with Adam optimizer, temperature annealing,
correlation-based gradient computation, coordinate descent polishing, and multi-restart
exploration with length variation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_gradient(g, tau):
    """
    Compute gradient of f_tau(g) = (1/tau)*log(sum_k exp(tau*C[k]))
    where C = convolve(g,g).
    Returns gradient vector of length n.
    """
    n = len(g)
    C = compute_conv(g)
    
    # Numerically stable softmax
    C_shifted = tau * C - tau * C.max()
    weights = np.exp(C_shifted)
    weights /= weights.sum()
    
    # Gradient: grad[j] = 2 * sum_i weights[i+j] * g[i]
    # This is the cross-correlation of weights and g
    h = np.correlate(weights, g, mode='valid')
    return 2.0 * h


def adam_smooth_descent(g, time_budget, seed=42, tau_start=2.0, tau_end=200.0,
                        lr=0.01, verbose=True):
    """
    Adam-based smooth gradient descent on the log-sum-exp approximation.
    g is normalized to sum=1. Returns best g found.
    """
    n = len(g)
    g = np.maximum(g, 0).astype(np.float64).copy()
    s = float(np.sum(g))
    if s < 1e-10:
        return g
    g = g / s

    best_g = g.copy()
    best_obj = compute_obj_norm(g)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    patience = 0
    max_patience = 2000

    while time.monotonic() - start < time_budget:
        # Anneal tau
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        tau = tau_start * (tau_end / tau_start) ** progress

        # Compute gradient
        grad = compute_gradient(g, tau)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # Adaptive learning rate (decay over time)
        current_lr = lr * 0.95 ** (t // 500)
        g_new = g - current_lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project: non-negative and normalize
        g_new = np.maximum(g_new, 0)
        s_new = float(np.sum(g_new))
        if s_new < 1e-10:
            break
        g_new = g_new / s_new

        # Evaluate
        if t % 10 == 0 or t < 10:
            obj = compute_obj_norm(g_new)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_g = g_new.copy()
                patience = 0
            else:
                patience += 10
            g = g_new
        else:
            g = g_new

        if verbose and t % 2000 == 0:
            obj = compute_obj_norm(g)
            elapsed = time.monotonic() - start
            print(f"      Adam t={t}: obj={obj:.10f}, best={best_obj:.10f}, tau={tau:.1f}, time={elapsed:.1f}s")

        if patience > max_patience and t > 5000:
            # Perturb to escape
            pert = rng.standard_normal(n) * 0.01
            g = best_g * np.exp(pert)
            g = np.maximum(g, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
            m = np.zeros(n)
            v = np.zeros(n)
            patience = 0

    return best_g


def coordinate_descent_polish(a, time_budget, seed=42, verbose=False):
    """
    Coordinate descent on normalized sequence (sum=1).
    Uses smart candidate values based on current state.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    no_improve = 0
    max_no_improve = 8

    while time.monotonic() - start < time_budget:
        sweep_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            a_j = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)

            # Unaffected max
            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))

            # Generate candidates
            if a_j < 1e-12:
                candidates = [0.0, 1e-4, 5e-4, 1e-3, 3e-3, 1e-2]
            else:
                candidates = [
                    a_j * 0.3, a_j * 0.6, a_j * 0.8, a_j * 0.9,
                    a_j * 1.1, a_j * 1.3, a_j * 1.7,
                    a_j * 0.1, a_j * 3.0,
                    1.0/n * 0.5, 1.0/n * 1.5, 1.0/n * 3.0
                ]
                candidates = [c for c in candidates if 0 <= c <= 0.3]
                # Add a fine grid around current value
                for mult in [0.95, 0.97, 1.03, 1.05]:
                    candidates.append(a_j * mult)
                candidates = list(set(candidates))

            best_trial_obj = best_obj
            best_trial_val = a_j

            for val in candidates:
                delta = val - a_j
                if abs(delta) < 1e-15:
                    continue
                S_new = 1.0 + delta
                if S_new <= 1e-10:
                    continue

                indices = np.arange(k_start, k_end + 1)
                mask = (indices == 2 * j)
                if np.any(~mask):
                    affected = conv[k_start:k_end + 1].copy()
                    affected[~mask] = conv[k_start:k_end + 1][~mask] + 2.0 * delta * a[indices[~mask] - j]
                else:
                    affected = conv[k_start:k_end + 1].copy()
                if np.any(mask):
                    affected[mask] = conv[2 * j] + (val * val - a_j * a_j)

                max_conv = max(max_unaffected, float(np.max(affected)))
                obj_trial = 2.0 * n * max_conv / (S_new * S_new)

                if obj_trial < best_trial_obj - 1e-14:
                    best_trial_obj = obj_trial
                    best_trial_val = val

            if best_trial_val != a_j:
                delta = best_trial_val - a_j
                a[j] = best_trial_val
                S_new = 1.0 + delta

                indices = np.arange(k_start, k_end + 1)
                mask = (indices == 2 * j)
                if np.any(~mask):
                    conv[k_start:k_end + 1][~mask] += 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    conv[2 * j] += (best_trial_val * best_trial_val - a_j * a_j)

                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S

                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose:
            elapsed = time.monotonic() - start
            print(f"    CD: obj={best_obj:.10f}, time={elapsed:.1f}s, no_improve={no_improve}")

    return best_a


def fourier_perturb(a, rng, strength=0.015):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 20):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.3, 1.0) / np.sqrt(k)
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_perturb(a, rng, strength=0.01):
    """Add random multiplicative perturbation."""
    n = len(a)
    pert = 1.0 + strength * rng.standard_normal(n)
    pert = np.maximum(pert, 0.1)
    a_new = a * pert
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s
```
