def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)

    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None, tol=1e-10):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
        except Exception:
            pass

    for solver in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=getattr(cp, solver), **{'tol': tol, 'max_iter': 100000})
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def _cd_eval(j, val, a_j, a, conv, max_unaffected, n):
    """Evaluate CD objective for coordinate j at value val."""
    delta = val - a_j
    S_new = 1.0 + delta
    if S_new <= 1e-10:
        return np.inf
    k_start = j
    k_end = min(j + n - 1, 2 * n - 2)
    indices = np.arange(k_start, k_end + 1)
    affected = conv[k_start:k_end + 1].copy()
    mask = (indices == 2 * j)
    if np.any(~mask):
        affected[~mask] = conv[k_start:k_end + 1][~mask] + 2.0 * delta * a[indices[~mask] - j]
    if np.any(mask):
        affected[mask] = conv[2 * j] + (val * val - a_j * a_j)
    max_conv_unscaled = max(max_unaffected, float(np.max(affected)))
    return 2.0 * n * max_conv_unscaled / (S_new * S_new)

def fourier_perturb(a, rng, strength=0.015):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 10):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses improved LP alternating with binary search line search, smooth gradient
polishing via log-sum-exp with projected gradient descent, enhanced coordinate
descent with golden section refinement, and multi-length exploration.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = np.asarray(warm_start, dtype=np.float64).copy()
        except Exception:
            pass

    for solver in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            solver_enum = getattr(cp, solver)
            prob.solve(solver=solver_enum)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def smooth_gradient_polish(a, time_budget, seed=42, verbose=True):
    """
    Polish using smooth approximation of max (log-sum-exp) with projected gradient descent.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    temps = [0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001]

    for temp in temps:
        if time.monotonic() - start > time_budget:
            break

        max_iters = 300
        lr = 0.05 / (n ** 0.5)
        momentum = 0.9
        v_mom = np.zeros(n)

        for it in range(max_iters):
            if time.monotonic() - start > time_budget:
                break

            conv = compute_conv(a)
            max_c = float(np.max(conv))
            shifted = (conv - max_c) / temp
            # Clip for numerical stability
            shifted = np.clip(shifted, -50, 50)
            log_sum = np.log(np.sum(np.exp(shifted)))
            weights = np.exp(shifted - log_sum)

            # Gradient: 2 * full_conv(weights, a)[:n]
            size = 1
            while size < 2 * n - 1:
                size <<= 1
            fw = rfft(weights, size)
            fa = rfft(a, size)
            conv_wa = irfft(fw * fa, size)[:n]
            grad = 2.0 * conv_wa

            # Momentum update
            v_mom = momentum * v_mom + (1 - momentum) * grad
            a_new = a - lr * v_mom
            a_new = project_simplex(a_new)

            obj = compute_obj_norm(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
            a = a_new

            # Adaptive: reduce lr if no improvement for a while
            if it > 50 and it % 50 == 0:
                lr *= 0.7

        if verbose:
            elapsed = time.monotonic() - start
            print(f"    SG temp={temp}: obj={best_obj:.10f}, time={elapsed:.1f}s")

    return best_a


def _cd_eval(j, val, a_j, a, conv, max_unaffected, n):
    """Evaluate CD objective for coordinate j at value val."""
    delta = val - a_j
    S_new = 1.0 + delta
    if S_new <= 1e-10:
        return np.inf
    k_start = j
    k_end = min(j + n - 1, 2 * n - 2)
    if k_end < k_start:
        max_conv = max_unaffected
    else:
        indices = np.arange(k_start, k_end + 1)
        affected = np.zeros(k_end - k_start + 1)
        mask = (indices == 2 * j)
        if np.any(~mask):
            affected[~mask] = conv[k_start:k_end + 1][~mask] + 2.0 * delta * a[indices[~mask] - j]
        if np.any(mask):
            affected[mask] = conv[2 * j] + (val * val - a_j * a_j)
        max_conv = max(max_unaffected, float(np.max(affected)))
    return 2.0 * n * max_conv / (S_new * S_new)


def cd_polish(a, time_budget, seed=42, verbose=True):
    """
    Enhanced coordinate descent with golden section refinement.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    mean_val = 1.0 / n

    sweep = 0
    no_improve = 0
    max_no_improve = 6

    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            a_j = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)

            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))

            if a_j < 1e-12:
                candidates = [0.0, mean_val * 0.05, mean_val * 0.1, mean_val * 0.3,
                              mean_val * 0.5, mean_val, mean_val * 2, mean_val * 5]
            else:
                candidates = [
                    0.0, a_j * 0.05, a_j * 0.2, a_j * 0.4, a_j * 0.6, a_j * 0.8,
                    a_j * 1.2, a_j * 1.5, a_j * 2.0, a_j * 3.0, a_j * 5.0,
                    mean_val * 0.1, mean_val * 0.3, mean_val * 0.5, mean_val,
                    mean_val * 2, mean_val * 3, mean_val * 5
                ]
                candidates = [c for c in candidates if 0 <= c <= 0.5]
                seen = set()
                unique_candidates = []
                for c in candidates:
                    if c not in seen:
                        seen.add(c)
                        unique_candidates.append(c)
                candidates = unique_candidates

            best_trial_obj = best_obj
            best_trial_val = a_j

            for val in candidates:
                delta = val - a_j
                if abs(delta) < 1e-15:
                    continue
                obj_trial = _cd_eval(j, val, a_j, a, conv, max_unaffected, n)
                if obj_trial < best_trial_obj - 1e-14:
                    best_trial_obj = obj_trial
                    best_trial_val = val

            # Golden section refinement
            if best_trial_val != a_j and best_trial_obj < best_obj - 1e-13:
                spread = max(a_j * 0.5, mean_val * 0.2, 1e-4)
                lo = max(0.0, best_trial_val - spread)
                hi = best_trial_val + spread
                for _ in range(10):
                    if time.monotonic() - start > time_budget:
                        break
                    if hi - lo < 1e-13:
                        break
                    ratio = 1.0 / (1.0 + np.sqrt(5.0))
                    m1 = lo + ratio * (hi - lo)
                    m2 = hi - ratio * (hi - lo)
                    obj_m1 = _cd_eval(j, m1, a_j, a, conv, max_unaffected, n)
                    obj_m2 = _cd_eval(j, m2, a_j, a, conv, max_unaffected, n)
                    if obj_m1 < obj_m2:
                        hi = m2
                    else:
                        lo = m1
                mid_val = 0.5 * (lo + hi)
                mid_obj = _cd_eval(j, mid_val, a_j, a, conv, max_unaffected, n)
                if mid_obj < best_trial_obj:
                    best_trial_val = mid_val
                    best_trial_obj = mid_obj

            if best_trial_obj < best_obj - 1e-14 and best_trial_val != a_j:
                delta = best_trial_val - a_j
                a[j] = best_trial_val
                S_new = 1.0 + delta

                indices = np.arange(k_start, k_end + 1)
                mask = (indices == 2 * j)
                if np.any(~mask):
                    conv[k_start:k_end + 1][~mask] += 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    conv[2 * j] += (best_trial_val * best_trial_val - a_j * a_j)

                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S

                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")

    return best_a


def lp_alternate(a, time_budget, seed=42, verbose=True):
    """LP alternating minimization with binary search line search and periodic CD."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    iter_count = 0
    stall = 0

    while time.monotonic() - start < time_budget - 5:
        g_lp, M_lp = solve_lp(a, warm_start=prev_g)
        if g_lp is None:
            break
        prev_g = g_lp
        iter_count += 1

        # Binary search line search for best t
        obj_at_1 = compute_obj_norm(g_lp)

        if obj_at_1 < best_obj - 1e-14:
            lo, hi = 0.0, 1.0
            best_t = 0.0
            best_line_obj = best_obj
            for _ in range(14):
                t_mid = 0.5 * (lo + hi)
                a_try = (1.0 - t_mid) * a + t_mid * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    hi = t_mid
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj_norm(a_try)
                if obj_try < best_line_obj - 1e-14:
                    best_line_obj = obj_try
                    best_t = t_mid
                    lo = t_mid
                else:
                    hi = t_mid
        else:
            best_t = 0.0
            best_line_obj = best_obj

        if best_t > 0 and best_line_obj < best_obj - 1e-14:
            a_try = (1.0 - best_t) * a + best_t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                line_obj = compute_obj_norm(a_try)
            else:
                line_obj = np.inf
                a_try = a.copy()

            if line_obj < best_obj - 1e-14:
                best_obj = line_obj
                best_a = a_try.copy()
                a = best_a.copy()
                stall = 0
                if verbose and iter_count % 20 == 0:
                    elapsed = time.monotonic() - start
                    print(f"    LP {iter_count}: obj={best_obj:.10f}, t={best_t:.3f}, elapsed={elapsed:.0f}s")
            else:
                stall += 1
        else:
            stall += 1

        if stall >= 5:
            strength = 0.002 + 0.001 * min(stall, 20)
            if stall > 10:
                # More aggressive: Fourier perturbation
                a_pert = fourier_perturb(best_a, rng, strength=0.008 + 0.004 * min(stall - 10, 5))
                a = a_pert
            else:
                pert = rng.standard_normal(n) * strength
                a_candidate = best_a * (1.0 + pert)
                a_candidate = np.maximum(a_candidate, 0)
                s_c = float(np.sum(a_candidate))
                if s_c > 1e-10:
                    a = a_candidate / s_c
                else:
                    a = best_a.copy()
            prev_g = None
            stall = 0

        # Periodic CD interspersed
        if iter_count % 12 == 0 and time.monotonic() - start < time_budget - 40:
            cd_time = min(25.0, (time_budget - (time.monotonic() - start)) * 0.08)
            if cd_time > 5:
                a_cd = cd_polish(a, time_budget=cd_time, seed=seed + iter_count * 31, verbose=False)
                obj_cd = compute_obj_norm(a_cd)
                if obj_cd < best_obj:
                    best_obj = obj_cd
                    best_a = a_cd.copy()
                    a = best_a.copy()

    if verbose:
        print(f"    LP done: {iter_count} iters, best={best_obj:.10f}")
    return best_a


def fourier_perturb(a, rng, strength=0.015):
    """Add structured Fourier perturbation to escape local minima."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 12):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_perturb(a, rng, strength=0.05):
    """Add random multiplicative perturbation."""
    n = len(a)
    pert = 1.0 + rng.standard_normal(n) * strength
    a_new = a * pert
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s
