"""
Multi-phase optimization for step function sequence minimization.
Combines FFT-accelerated soft-max/power-sum gradient descent with Nesterov momentum,
Frank-Wolfe LP polishing with active set management, subgradient refinement,

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_soft_gradient(a, p):
    """
    Compute gradient of soft-max objective.
    grad_i = 2 * sum_j a_j * p_{i+j}
    Uses FFT cross-correlation.
    """
    n = len(a)
    p_len = len(p)
    a_rev = a[::-1]
    size = 1
    while size < n + p_len - 1:
        size <<= 1
    A = rfft(a_rev, size)
    P = rfft(p, size)
    full = irfft(A * P, size)
    q = full[n - 1:n - 1 + n]
    return 2.0 * q


def compute_power_gradient(a, conv, p):
    """
    Compute gradient of power-sum objective.
    grad_i = 2 * sum_j a_j * conv_{i+j}^{p-1} / sum(conv_k^{p-1})
    """
    n = len(a)
    if p <= 1:
        # Fallback to uniform weighting
        w = np.ones(len(conv)) / len(conv)
    else:
        c_safe = np.maximum(conv, 1e-30)
        w = c_safe ** (p - 1)
        w = w / float(np.sum(w))
    return compute_soft_gradient(a, w)


def sgd_smooth(a, time_budget, method='softmax', beta_start=10.0, beta_end=100.0,
               p_start=4.0, p_end=50.0, lr_start=0.01, momentum=0.9,
               nesterov=True, verbose=True, seed=42):
    """
    Smooth gradient descent (soft-max or power-sum) with momentum/Nesterov.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    it = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Adaptive parameters
        if method == 'softmax':
            beta = beta_start + (beta_end - beta_start) * (progress ** 0.5)
            lr = lr_start * (1.0 - 0.5 * progress)
        else:
            p = p_start + (p_end - p_start) * (progress ** 0.5)
            lr = lr_start * (1.0 - 0.5 * progress)
        
        # Compute convolution
        conv = compute_conv(a)
        
        # Compute weights
        if method == 'softmax':
            c_max = float(np.max(conv))
            exp_vals = np.exp(beta * (conv - c_max))
            w = exp_vals / float(np.sum(exp_vals))
        else:
            c_safe = np.maximum(conv, 1e-30)
            w = c_safe ** (p - 1)
            w = w / float(np.sum(w))
        
        # Compute gradient
        grad = compute_soft_gradient(a, w)
        
        # Nesterov or standard momentum

            v_new = momentum * v + grad

            a_lookahead = project_simplex(a_lookahead)
            # Recompute gradient at lookahead (expensive, so we skip for simplicity)
            # For efficiency, use the gradient at current point
            v = v_new
            a_new = a - lr * v
        else:
            v = momentum * v + grad
            a_new = a - lr * v
        
        a_new = project_simplex(a_new)
        
        # Evaluate
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0

        it += 1

            print(f"      SGD({method}) it{it}: obj={best_obj:.8f}, t={elapsed:.1f}s")
        
        # Early reset if long stall
        if it - best_it > 3000 and it > 500:
            a = best_a.copy()
            v = np.zeros(n)
            best_it = it
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, active_set=None):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k in active_set (or all k),
    sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    if active_set is not None:
        A = build_conv_matrix(a_norm)[active_set]
    else:
        A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def fw_line_search(a, g, max_iters=30):
    """
    Strong Wolfe line search for FW step: find t in [0, 1.5] minimizing obj((1-t)*a + t*g).
    """
    def obj_at(t):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)
    
    # First, find the bracket using golden section
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi
    lo, hi = 0.0, 1.5
    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)
    
    for _ in range(max_iters):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)
    
    t_best = c if fc < fd else d
    obj_best = min(fc, fd)
    
    # Refine with local search around best
    for dt in [0.05, -0.05, 0.02, -0.02, 0.01, -0.01]:
        t_try = t_best + dt
        if 0 <= t_try <= 2:
            obj_try = obj_at(t_try)
            if obj_try < obj_best:
                obj_best = obj_try
                t_best = t_try
    
    return t_best, obj_best

                j2 = i - k  # This would be for the symmetric term, but for linear conv
                # Actually for linear convolution (a*a)_k = sum_j a_j * a_{k-j} where both indices in [0,n-1]
                # d/d a_i = a_{k-i} if 0 <= k-i < n, else 0
                # Plus for the case where i appears as the second factor: a_{k-i}... 
                # Wait, (a*a)_k = sum_{j: 0<=j<n, 0<=k-j<n} a_j * a_{k-j}
                # d/d a_i: i appears as j (giving a_{k-i}) and as k-j (giving a_j = a_{k-i}... no)
                # If i = j: term is a_i * a_{k-i}, derivative w.r.t. a_i is a_{k-i}
                # If i = k-j, i.e., j = k-i: term is a_{k-i} * a_i, derivative w.r.t. a_i is a_{k-i}
                # So total: a_{k-i} + a_{k-i} = 2*a_{k-i} if both i and k-i are in [0,n-1]
                # Or just a_{k-i} if only one of them is valid
                # But since conv is symmetric in the sense that a*b = b*a, we have:
                # (a*a)_k = sum_j a_j * a_{k-j}, and d/d a_i = a_{k-i} + a_{k-i} if i != k-i
                # or 2*a_{k-i} if i == k-i (which means 2i = k)
                # Actually it's always 2*a_{k-i} when both i and k-i are valid.
                j = k - i
                if 0 <= j < n:
                    sg[i] += 2.0 * a[j]
        
        sg /= n_active
        
        # Polyak step: step = (current_obj - best_obj) / ||sg||^2
        # But we need to be careful: the objective is 2n * max(conv) not just max(conv)
        # And we're on the simplex, so we need to project
        # Let's use a diminishing step size instead for safety
        elapsed = time.monotonic() - start
        progress = elapsed / time_budget
        step = 0.01 * (1 - 0.8 * progress)
        
        a_new = a - step * sg
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        it += 1
        
        if verbose and it % 50 == 0:
            print(f"      SubGrad it{it}: obj={best_obj:.8f}, n_active={n_active}, t={elapsed:.1f}s")
    
    return best_a, best_obj


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    return project_simplex(a)


def arcsine_init(n):
    """Arcsine distribution initialization."""
    x = np.linspace(0.01, 0.99, n)
    a = 1.0 / np.sqrt(np.maximum(1 - (2 * x - 1) ** 2, 1e-10))
    return project_simplex(a)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = (np.exp(-((idx - n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) +
             np.exp(-((idx - 3 * n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) + 0.01)
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 6:
        a = 1.0 / np.cosh(3.0 * (2 * x - 1))
    elif mode == 7:
        a = np.zeros(n)
        a[0] = 1.0
        a[-1] = 1.0
        if n > 4:
            a[1:min(4, n)] = 0.3
            a[max(0, n-5):n-1] = 0.3
            a[1:-1] += 0.05
    elif mode == 8:
        # Cosine bell
        a = np.cos(np.pi * (x - 0.5)) ** 2 + 0.1
    elif mode == 9:
        # Double peak
        a = np.exp(-20 * (x - 0.15) ** 2) + np.exp(-20 * (x - 0.85) ** 2) + 0.05
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search combining FFT smooth gradient descent, FW LP polishing,
    subgradient refinement, and length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Try to load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = u_shaped_init(n_current, sharpness=3.0)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")
    
    # Try several initializations
    for init_fn, name in [(lambda: arcsine_init(n_current), "arcsine"),
                          (lambda: u_shaped_init(n_current, 5.0), "u_sharp"),
                          (lambda: u_shaped_init(n_current, 1.5), "u_soft"),
                          (lambda: diverse_init(n_current, rng, 8), "cosine_bell"),
                          (lambda: diverse_init(n_current, rng, 9), "double_peak")]:
        try:
            a_try = init_fn()
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  {name} init: obj={best_score:.8f} **")
        except Exception:
            pass
