"""

dual-guided perturbation, subgradient polishing, coordinate descent, and multi-n exploration.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lp_solve_precomputed(a, A_mat, solver_list=None):
    """Solve LP with precomputed constraint matrix. Returns (g, M, dual_values)."""
    n = len(a)
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A_mat @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.MOSEK, cp.SCS]
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    # Try to get dual values
                    duals = None
                    try:
                        duals = np.array(constraints[0].dual_value)
                    except Exception:
                        pass
                    return g_val / sg, float(M_var.value), duals
        except Exception:
            continue
    return None, None, None


def build_constraint_matrix(n):
    """Build the (2n-1) x n constraint matrix for the LP."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            A[k, i] = 0  # will be filled by a
    return A

def subgradient_polish(a, time_budget, verbose=False, seed=123):
    """
    Subgradient descent with diminishing step size for final polishing.
    Uses the exact subgradient of max(conv(a,a)).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Diminishing step size schedule
    step0 = 0.02
    decay = 0.9995
    step = step0
    momentum = np.zeros(n)
    beta = 0.9

    for it in range(50000):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        # Subgradient: average over (near-)max lags
        threshold = max_val * (1.0 - 1e-4)
        active_lags = np.where(conv >= threshold)[0]
        
        subgrad = np.zeros(n)
        for lag in active_lags:
            for i in range(n):
                j = lag - i
                if 0 <= j < n:
                    subgrad[i] += a[j]
        subgrad /= len(active_lags)

        # Project onto simplex tangent
        subgrad_tangent = subgrad - np.mean(subgrad)
        gn = np.linalg.norm(subgrad_tangent)
        if gn < 1e-15:
            break

        # Nesterov momentum
        momentum = beta * momentum - step * subgrad_tangent / gn
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn < 1e-10:
            a_new = a
        else:
            a_new = a_new / sn

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        step *= decay

        # Periodic random restart if stuck
        if it % 500 == 0 and it > 0:
            if compute_obj(a) > best_obj * (1 + 1e-6):
                a = best_a + 0.001 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                momentum = np.zeros(n)

    return best_a


def coordinate_descent_polish(a, time_budget, verbose=False, seed=456):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate i, the objective as a function of a[i] is:
    2*n*max_k (a*a)[k] / (sum(a))^2
    where (a*a)[k] is linear in a[i] for k != i*2 (quadratic at k=2i).
    Actually (a*a)[k] = sum_j a[j]*a[k-j], which includes a[i]*a[k-i] term.
    So for fixed other coordinates, (a*a)[k] = c_k + a[i]*d_k where d_k = a[k-i] (if valid).
    This makes it a 1D nonconvex problem (max of linear functions / quadratic).
    We use golden section or grid search in 1D.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    # Work with sum=1 normalization but track actual values
    a_orig = a.copy()
    a = a / s  # normalize

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    coord_order = list(range(n))
    rng.shuffle(coord_order)

    # Precompute: for each i, the "base" convolution without a[i]'s contribution
    # conv_no_i[k] = sum_{j != i} a[j]*a[k-j] (excluding terms involving a[i])
    # (a*a)[k] = conv_no_i[k] + a[i]*a[k-i] + (if 2i==k: a[i]^2 already counted? No)
    # Actually: (a*a)[k] = sum_j a[j]*a[k-j]. The term with j=i is a[i]*a[k-i].
    # The term with k-j=i, i.e., j=k-i, is a[k-i]*a[i] - same term!
    # So (a*a)[k] = sum_{j != i} a[j]*a[k-j] + a[i]*a[k-i] (counted once)
    # Wait no: sum_j a[j]*a[k-j] includes j=i giving a[i]*a[k-i], AND j=k-i giving a[k-i]*a[i].
    # If i != k-i (i.e., 2i != k), these are two separate terms.
    # If 2i == k, it's one term: a[i]*a[i] = a[i]^2.
    
    # This is getting complicated. Let's just do a simple 1D search.
    
    max_rounds = 20
    for round_num in range(max_rounds):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1.0:
            break
        
        improved = False
        for idx in range(n):
            if (time.monotonic() - start) > time_budget - 1:
                    break
            i = coord_order[idx]
            old_val = a[i]
            
            # Try a few values around old_val
            candidates = [0.0, old_val * 0.5, old_val * 0.75, old_val, old_val * 1.25, old_val * 1.5, old_val * 2.0]
            # Also try absolute values
            candidates.extend([max(0, old_val - 0.01), max(0, old_val - 0.005), max(0, old_val + 0.005), max(0, old_val + 0.01)])
            candidates = [max(0, c) for c in candidates]
            
            best_cand_obj = best_obj
            best_cand_val = old_val
            
            for cv in candidates:
                if abs(cv - old_val) < 1e-12:
                    continue
                a[i] = cv
                s_temp = float(np.sum(a))
                if s_temp < 1e-10:
                    a[i] = old_val
                    continue
                a_temp = a / s_temp
                obj_temp = compute_obj(a_temp)
                if obj_temp < best_cand_obj - 1e-12:
                    best_cand_obj = obj_temp
                    best_cand_val = cv
            
            if best_cand_val != old_val:
                a[i] = best_cand_val
                a = a / a.sum()
                improved = True
                if best_cand_obj < best_obj - 1e-14:
                    best_obj = best_cand_obj
                    best_a = a.copy()
                    if verbose and round_num % 5 == 0:
                        print(f"    CD round{round_num}: obj={best_obj:.10f}")
            else:
                a[i] = old_val
        
        if not improved:
            break
    
    # Restore original scaling
    result = best_a * s
    return result


def spectral_init(n, mode=0, seed=42):
    """Generate various structured initializations."""
    rng = np.random.default_rng(seed)
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.maximum(np.sinc(x), 0.001)
        return a / a.sum()
    elif mode == 2:
        a = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / max(1, n - 1))
        return a / a.sum()
    elif mode == 3:
        x = np.arange(n) / max(1, n - 1)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        return np.maximum(a, 0.001) / np.sum(np.maximum(a, 0.001))
    elif mode == 4:
        x = np.linspace(-3, 3, n)
        a = np.exp(-x ** 2)
        return a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
        return a / a.sum()
    elif mode == 6:
        a = rng.uniform(0.1, 1.0, n)
        return a / a.sum()
    elif mode == 7:
        # Optimal-like: concentrated in center
        x = np.arange(n) - n / 2
        a = np.exp(-0.5 * (x / (n * 0.15)) ** 2)
        return a / a.sum()
    elif mode == 8:
        # Multi-scale
        x = np.arange(n) / n
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        x = np.arange(n) / max(1, n - 1)
        a = np.sin(np.pi * x) ** 2
        return a / a.sum()


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def propose_candidate(seed=42, budget_s=1500):

    best_seq = None
    best_score = np.inf

    try:

        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent_arr.copy()
                print(f"Parent loaded (n={parent_arr.size}): obj={best_score:.10f}")
    except Exception:
        parent = None

    n_parent = len(best_seq) if best_seq is not None else 800

    # === Phase 1: Try multiple n values with quick LP ===
    print(f"\n=== Phase 1: Multi-n Quick Screening ===")
    remaining = deadline - time.monotonic()
    
    # n values to try (centered around parent)
    n_candidates = []
    if best_seq is not None:
        n_candidates.append(n_parent)
    for n_try in [n_parent - 100, n_parent - 50, n_parent + 50, n_parent + 100, n_parent + 200]:
        if 400 <= n_try <= 2000:
            n_candidates.append(n_try)
    # Add some fixed values
    for nv in [600, 700, 800, 900, 1000, 1200, 1400]:
        if 400 <= nv <= 2000 and nv not in n_candidates:
            n_candidates.append(nv)
    
    n_candidates = sorted(set(n_candidates))
    n_candidates = [n for n in n_candidates if 400 <= n <= 2000]
    
    # Sort by proximity to parent n
    n_candidates.sort(key=lambda x: abs(x - n_parent))
    n_candidates = n_candidates[:8]  # Limit to 8
    print(f"  n candidates: {n_candidates}")
    
    # First pass: quick LP (short budget) for each n
    quick_results = []
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        # Initialize
        if best_seq is not None and n_try != n_parent:

            # Actually just store the normalized version scaled
            best_seq = a_try.copy()
    
    # === Phase 2: Deep LP on best n ===
    print(f"\n=== Phase 2: Deep LP Fixed-Point ===")
    remaining = deadline - time.monotonic()
    
    # Use the best n from quick screening
    if quick_results:
        best_n_try, best_quick_obj, best_quick_a = min(quick_results, key=lambda x: x[1])
    else:

        best_quick_obj = best_score
        best_quick_a = best_seq

                n_best = len(a_alt)
    
    # === Phase 7: Final polishing ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    
    if remaining > 20:

        a_final = subgradient_polish(best_seq, time_budget=alloc, seed=31415)

    return result
