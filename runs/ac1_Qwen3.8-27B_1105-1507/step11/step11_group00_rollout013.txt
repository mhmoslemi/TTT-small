"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of:
1. Mirror descent with iterate averaging for fast convergence
2. Coordinate descent exploiting 1D structure for polishing
3. Active-set detection for focusing optimization effort
4. Multi-n exploration to find optimal sequence length
5. Simulated annealing perturbations to escape local minima
Key improvement over previous: proper iterate averaging, active-set QP refinement,
and more efficient time allocation.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def gradient_from_p(a, p):
    """
    Compute 2 * sum_j a[j] * p[i+j] for i=0..n-1.
    p has length 2n-1 (distribution over convolution lags).
    """
    n = len(a)
    b = a[::-1]
    N_fft = 1
    while N_fft < 3 * n - 1:
        N_fft <<= 1
    c = irfft(rfft(b, N_fft) * rfft(p, N_fft), N_fft)
    return 2.0 * c[n - 1:2 * n - 1]


def smoothed_gradient(a, conv, tau):
    """Compute gradient of LSE(tau, conv(a,a)) with respect to a."""
    n = len(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(2 * n - 1)
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    return gradient_from_p(a, p)


def active_set_gradient(a, conv, k_active=3, tol=1e-4):
    """
    Subgradient using only the active set (top lags near the max).
    More focused than full softmax when active set is small.
    """
    n = len(a)
    max_conv = float(np.max(conv))
    if max_conv < 1e-15:
        return np.zeros(n)
    # Active set: lags within tol of max
    threshold = max_conv * (1.0 - tol)
    active_idx = np.where(conv >= threshold)[0]
    # Cap at k_active largest
    if len(active_idx) > k_active:
        top_vals = np.sort(conv)[-k_active:]
        threshold = top_vals[0]
        active_idx = np.where(conv >= threshold)[0]
    p = np.zeros(2 * n - 1)
    p[active_idx] = 1.0 / len(active_idx)
    return gradient_from_p(a, p)


def mirror_descent_with_averaging(a, time_budget=300.0, tau_init=50.0, tau_end=2000.0,
                                  verbose=True, use_averaging=True):
    """
    Mirror descent (multiplicative weights) with iterate averaging.
    Key improvement: maintains running average of all iterates for better convergence.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    # Averaging: maintain weighted sum of iterates
    avg_a = np.zeros(n)
    avg_weight = 0.0
    best_avg_a = a.copy()
    best_avg_obj = best_obj

    it = 0
    stall = 0
    gamma_0 = 5.0  # Initial step size parameter

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 0.8

        conv = compute_conv(a)

        # Choose gradient: use active set when it's concentrated
        max_conv = float(np.max(conv))
        ratio = conv[np.argsort(conv)[-min(5, 2*n-1):]].sum() / max(1e-30, conv.sum())
        if ratio > 0.5:
            # Concentrated: use active set gradient
            g = active_set_gradient(a, conv, k_active=3, tol=1e-3)
        else:
            # Spread out: use smoothed gradient
            g = smoothed_gradient(a, conv, tau)

        g_max = float(np.max(np.abs(g)))
        if g_max < 1e-15:
            break

        # Diminishing step size: gamma_t = gamma_0 / (1 + t * gamma_0)
        gamma_t = gamma_0 / (1.0 + it * 0.01 * gamma_0)
        eta = gamma_t / (g_max + 1e-10)

        # Mirror descent step
        exponent = -eta * g
        exponent = np.clip(exponent, -30, 30)
        a_new = a * np.exp(exponent)
        s_new = float(np.sum(a_new))
        if s_new < 1e-300:
            a_new = a.copy()
            s_new = float(np.sum(a))
        a_new = a_new / s_new

        # Update average (use 1/t weighting for best rate)
        t_weight = 1.0 / (it + 1)
        avg_a = (1.0 - t_weight) * avg_a + t_weight * a_new
        avg_weight += 1.0

        # Evaluate current and averaged
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        if use_averaging and it > 10:
            obj_avg = compute_obj(avg_a)
            if obj_avg < best_avg_obj - 1e-14:
                best_avg_obj = obj_avg
                best_avg_a = avg_a.copy()
            if obj_avg < best_obj - 1e-12:
                best_obj = obj_avg
                best_a = avg_a.copy()
                stall = 0

        # Perturbation to escape local minima
        if stall >= 30 and it % 50 == 0:
            strength = 0.03 * (1.0 + 0.5 * progress)
            a_pert = a * np.exp(strength * rng.standard_normal(n))
            a_pert = np.maximum(a_pert, 0)
            s_pert = float(np.sum(a_pert))
            if s_pert > 1e-10:
                a = a_pert / s_pert
                avg_a = a.copy()
                stall = 0
            else:
                a = best_a.copy()
                stall = 0

        a = a_new
        it += 1

        if verbose and (it % 200 == 0 or it == 1):
            print(f"    MD avg it{it}: obj={best_obj:.8f}, avg_obj={best_avg_obj:.8f}, "
                  f"tau={tau:.0f}, stall={stall}, time={time.monotonic()-start:.0f}s")

    return best_a


def coordinate_descent(a, time_budget=200.0, sweeps_max=50, verbose=True):
    """
    Coordinate descent: for each element, find optimal value via 1D search.
    Exploits the structure that changing one element affects conv in a simple way.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)

    for sweep in range(sweeps_max):
        if time.monotonic() - start > time_budget:
            break
        obj_before = best_obj

        # Randomize coordinate order for diversity
        order = rng.permutation(n) if sweep % 3 != 0 else np.arange(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                    break

            # Try several candidate values for a[i]
            x_old = a[i]
            s_others = s - x_old

            # Candidates: 0, current, and values that might equalize top lags
            candidates = [0.0, x_old]

            # Value to make this element match the mean of others
            if s_others > 1e-10:
                candidates.append(s_others / (n - 1) if n > 1 else 0.0)

            # Values that might reduce the max conv
            # Try scaling: x * factor for various factors
            for factor in [0.5, 0.7, 1.3, 1.5, 2.0, 0.3]:
                candidates.append(x_old * factor)

            # Small perturbation
            candidates.append(max(0, x_old + 0.01 * s / n * rng.standard_normal()))

            # Evaluate each candidate
            best_cand_obj = compute_obj(a)
            best_cand_x = x_old

            for x_cand in candidates:
                if x_cand < 0 or x_cand > 10 * s / n:
                    continue
                a[i] = x_cand
                s_new = float(np.sum(a))
                if s_new < 0.01:
                    a[i] = x_old
                    continue
                obj_cand = compute_obj(a / s_new)
                if obj_cand < best_cand_obj - 1e-14:
                    best_cand_obj = obj_cand
                    best_cand_x = x_cand

            a[i] = best_cand_x
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
                s = 1.0

        obj_after_sweep = compute_obj(a)
        if obj_after_sweep < best_obj - 1e-14:
            best_obj = obj_after_sweep
            best_a = a.copy()

        if verbose and (sweep % 5 == 0):
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, "
                  f"time={time.monotonic()-start:.0f}s")

        # Early stop if no improvement in 5 sweeps
        if obj_after_sweep >= best_obj - 1e-12 and sweep > 5:
            if sweep - 5 > 10:
                break

    return best_a


def coordinate_descent_focused(a, time_budget=200.0, verbose=True):
    """
    Focused coordinate descent: only optimize coordinates that affect the active lags.
    Much faster when the active set is small.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    no_improve = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Find active lags
        threshold = max_conv * (1.0 - 1e-3)
        active_lags = np.where(conv >= threshold)[0]

        # For each active lag k, the elements that affect it are
        # a[j] and a[k-j] for valid j. Collect all such indices.
        affected = set()
        for k in active_lags:
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                affected.add(j)

        # Also include elements that might affect new lags (near the max)
        top_lags = np.argsort(conv)[-5:]
        for k in top_lags:
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                affected.add(j)

        affected_list = list(affected)
        rng.shuffle(affected_list)

        # Optimize each affected coordinate
        for i in affected_list:
            if time.monotonic() - start > time_budget:
                break
            x_old = a[i]
            s_others = 1.0 - x_old  # since sum(a) = 1

            # Grid search around current value
            x_scale = max(x_old, s / n) * 0.5
            candidates = [max(0, x_old + delta * x_scale)
                          for delta in [-2, -1, -0.5, -0.2, 0, 0.2, 0.5, 1, 2]]
            candidates.append(0.0)
            candidates.append(min(s / 3, x_old * 3))

            best_cand_obj = compute_obj(a)
            best_cand_x = x_old

            for x_cand in candidates:
                if x_cand < 0:

                a[i] = x_cand
                s_new = float(np.sum(a))
                if s_new < 0.01:
                    a[i] = x_old
                    continue
                obj_cand = compute_obj(a / s_new)
                if obj_cand < best_cand_obj - 1e-14:
                    best_cand_obj = obj_cand
                    best_cand_x = x_cand

            a[i] = best_cand_x
            s_new = float(np.sum(a))
            if s_new > 1e-10:
                a = a / s_new

        obj_new = compute_obj(a)
        if obj_new < best_obj - 1e-14:

def nesterov_polish(a, time_budget=100.0, tau=500.0, verbose=False):
    """Nesterov accelerated gradient descent for final polishing."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 1.0 / np.sqrt(n)
    v = np.zeros(n)
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
            s_y = 1.0
        else:
            y = y / s_y

        conv_y = compute_conv(y)
        g = smoothed_gradient(y, conv_y, tau)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            lr = lr * 0.999
            adapt = 0.5 / (g_norm + 1e-10)
            lr = min(max(lr, 0.5 * adapt), 2.0 * adapt)

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1

        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr *= 0.5
            best_it = it

    if verbose:
        print(f"    Nesterov: obj={best_obj:.8f}, iters={it}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def make_triang_init(n):
    """Create a triangular initialization (peak in center)."""
    center = n / 2.0
    a = 1.0 - np.abs(np.arange(n) - center) / center
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_mseq_init(n):

            print(f"  Phase1 MD: obj={obj_md:.8f} (no improvement)")

    # === Phase 2: Focused Coordinate Descent ===
    print(f"\n=== Phase 2: Focused Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 250.0)
    if p2_budget > 20:
        a_cd = coordinate_descent_focused(best_seq, time_budget=p2_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase2 CD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 3: Nesterov Polish ===
    print(f"\n=== Phase 3: Nesterov Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 120.0)
    if p3_budget > 15:
        a_gd = nesterov_polish(best_seq, time_budget=p3_budget, tau=500.0, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            print(f"  Phase3 Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 Nesterov: obj={obj_gd:.8f} (no improvement)")

    # === Phase 4: Second MD Round (different parameters) ===
    print(f"\n=== Phase 4: MD Round 2 (sharper gradient) ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 150.0)
    if p4_budget > 20:
        # Use active-set gradient more aggressively
        a_md2 = mirror_descent_with_averaging(best_seq, time_budget=p4_budget,
                                              tau_init=200.0, tau_end=3000.0,
                                              verbose=True)
        obj_md2 = compute_obj(a_md2)
        if obj_md2 < best_score:
            best_score = obj_md2
            best_seq = a_md2.copy()
            print(f"  Phase4 MD2: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 MD2: obj={obj_md2:.8f} (no improvement)")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = []
    # Try nearby n values
    for delta in [-100, -50, 50, 100, 200, 300]:
        n_try = n_best + delta
        if 200 <= n_try <= 3000:
            n_options.append(n_try)
    # Some absolute values
    for n_abs in [400, 600, 1000, 1200, 1500, 2000]:
        if n_abs not in n_options:
            n_options.append(n_abs)
    n_options = sorted(set(n_options))
    # Limit to 4 options based on time
    max_n_opts = min(4, max(1, int(remaining / 120)))
    n_options = n_options[:max_n_opts]
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)

        # MD
        alloc_md = min(remaining * 0.3, 60.0)
        if alloc_md > 15:
            a_try = mirror_descent_with_averaging(a_try, time_budget=alloc_md,
                                                  tau_init=50.0, tau_end=1000.0, verbose=False)

        # CD
        remaining = deadline - time.monotonic()
        alloc_cd = min(remaining * 0.25, 40.0)
        if alloc_cd > 10:
            a_try = coordinate_descent_focused(a_try, time_budget=alloc_cd, verbose=False)

        # Nesterov
        remaining = deadline - time.monotonic()
        alloc_gd = min(remaining * 0.1, 20.0)
        if alloc_gd > 8:
            a_try = nesterov_polish(a_try, time_budget=alloc_gd, tau=500.0)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    restart_configs = [
        ("symmetric", lambda: (best_seq + best_seq[::-1]) / 2.0),
        ("triangular_blend", lambda: 0.6 * best_seq + 0.4 * make_triang_init(n_cur)),
        ("mseq_blend", lambda: 0.7 * best_seq + 0.3 * make_mseq_init(n_cur)),
    ]

    for name, maker in restart_configs:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        try:
            a_r = maker()
        except Exception:
            continue
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r < 1e-10:
            a_r = best_seq.copy()
        else:
            a_r = a_r / s_r

        # Optimize
        alloc_md = min(remaining * 0.4, 50.0)
        if alloc_md > 12:
            a_r = mirror_descent_with_averaging(a_r, time_budget=alloc_md,
                                                tau_init=50.0, tau_end=800.0, verbose=False)

        remaining = deadline - time.monotonic()
        alloc_cd = min(remaining * 0.2, 30.0)
        if alloc_cd > 10:
            a_r = coordinate_descent_focused(a_r, time_budget=alloc_cd, verbose=False)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart ({name}): obj={best_score:.8f} **")
        else:
            print(f"  Restart ({name}): obj={obj_r:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Final MD with very high tau (nearly exact max)
    if remaining > 40:
        alloc = min(remaining * 0.35, 120.0)
        a_final = mirror_descent_with_averaging(best_seq, time_budget=alloc,
                                                tau_init=1000.0, tau_end=10000.0,
                                                verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # Final CD
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.35, 80.0)
        a_final2 = coordinate_descent_focused(best_seq, time_budget=alloc, verbose=True)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final Nesterov
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 50.0)
        a_final3 = nesterov_polish(best_seq, time_budget=alloc, tau=2000.0, verbose=True)
        obj_final3 = compute_obj(a_final3)
        if obj_final3 < best_score:
            best_score = obj_final3
            best_seq = a_final3.copy()
            print(f"  Final Nesterov: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
