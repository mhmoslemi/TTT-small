"""
FFT-accelerated LSE gradient descent with adaptive temperature, 
Remez-inspired peak reduction, and multi-start optimization
for minimizing 2n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

TOTAL_BUDGET = 1485.0  # Leave 15s margin

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj_conv(a_norm):
    """Compute 2n*max(conv) for already-normalized a (sum=1)."""
    n = len(a_norm)
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n

def lse_gradient_fft(a, T):
    """
    Compute gradient of 2n * LSE_T(conv(a,a)) for normalized a.
    Uses FFT for O(n log n) computation.
    
    LSE_T(x) = (1/T) * log(sum_k exp(T * x_k))
    grad LSE = sum_k p_k * grad x_k where p_k = softmax(T*x)
    
    For x = conv(a,a), grad x[k] wrt a[i] = a[k-i] + a[i-k] (with bounds)
    So grad[i] = sum_k p[k]*(a[k-i] + a[i-k]) = (a *⋆ p)[i] + (a * p)[i]
    where *⋆ is cross-correlation and * is convolution.
    """
    n = len(a)
    conv = compute_conv(a)
    # Compute softmax weights
    cmax = np.max(conv)
    exp_vals = np.exp(T * (conv - cmax))
    p = exp_vals / exp_vals.sum()  # length 2n-1
    
    # FFT size for convolution of length-n and length-(2n-1) arrays
    fft_size = 1
    while fft_size < n + (2*n - 1) - 1:
        fft_size <<= 1
    
    A = rfft(a, fft_size)
    P = rfft(p, fft_size)
    
    # term2: (a * p)[i] = convolution
    conv_ap = irfft(A * P, fft_size)[:n]
    
    # term1: (a ⋆ p)[i] = cross-correlation = IFFT(conj(FFT(a)) * FFT(p))
    # For real signals: cross_corr = IFFT(conj(A) * P)
    cross_ap = irfft(np.conj(A) * P, fft_size)[:n]
    
    grad = cross_ap + conv_ap
    return 2.0 * n * grad

def lse_objective(a, T):
    """Compute 2n * LSE_T(conv(a,a)) for normalized a."""
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)
    exp_vals = np.exp(T * (conv - cmax))
    lse = cmax + np.log(np.sum(exp_vals)) / T
    return 2.0 * n * lse

def lse_gradient_descent(a, time_budget, T_start=20.0, T_end=200.0, verbose=False):
    """
    LSE smoothed gradient descent with FFT-based gradient and backtracking.
    T anneals from T_start (smooth) to T_end (sharp, close to true max).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_conv(a)
    start = time.monotonic()
    lr = 1.0 / (n * 10.0)
    it = 0
    best_it = 0
    stall_count = 0
    max_stall = 50

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / time_budget)
        T = T_start * (T_end / T_start) ** frac

        grad = lse_gradient_fft(a, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        a_old = a.copy()
        obj_old = lse_objective(a, T)
        t = lr
        improved = False
        for _ in range(12):
            a_try = a_old - t * grad
            a_try = project_simplex(a_try)
            obj_try = lse_objective(a_try, T)
            if obj_try < obj_old - 1e-14:
                a = a_try.copy()
                improved = True
                break
            t *= 0.5

        if not improved:
            # Try random perturbation
            rng_local = np.random.default_rng(it)
            for attempt in range(5):
                direction = a + rng_local.normal(0, 0.01, n)
                a_try = project_simplex(direction)
                obj_try = lse_objective(a_try, T)
                if obj_try < obj_old:
                    a = a_try.copy()
                    improved = True
                    break
            if not improved:
                stall_count += 1
                if stall_count > max_stall:
                    lr *= 0.5
                    if lr < 1e-10:
                        break
                    stall_count = 0
                continue
        else:
            stall_count = 0
            lr = t * 1.05  # slightly increase step size if it worked

        # Check true objective
        obj_true = compute_obj_conv(a)
        if obj_true < best_obj:
            best_obj = obj_true
            best_a = a.copy()
            best_it = it

        it += 1
        if verbose and it % 200 == 0:
            print(f"      LSE-GD it{it}: obj={best_obj:.8f}, T={T:.1f}, lr={lr:.2e}, t={time.monotonic()-start:.1f}s")

    return best_a

"""
FFT-accelerated LSE gradient descent with adaptive temperature, LP polishing,
and multi-start for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Key: O(n log n) gradient via correct cross-correlation FFT formula.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

TOTAL_BUDGET = 1480.0

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj_conv(a_norm):
    """Compute 2n*max(conv) for normalized a (sum=1)."""
    n = len(a_norm)
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n

def lse_grad_and_obj(a, T):
    """
    Compute gradient and value of 2n*LSE_T(conv(a,a)) for normalized a.
    grad[i] = 4n * sum_j a[j]*p[j+i] = 4n * (p cross-corr a)[i]
    Uses FFT: grad = 4n * IFFT(conj(FFT(p)) * FFT(a))[:n]
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)
    exp_vals = np.exp(T * (conv - cmax))
    p = exp_vals / exp_vals.sum()
    lse_val = cmax + np.log(np.sum(exp_vals)) / T
    obj = 2.0 * n * lse_val

    fft_size = 1
    while fft_size < 3 * n:
        fft_size <<= 1
    A = rfft(a, fft_size)
    P = rfft(p, fft_size)
    cross = irfft(np.conj(P) * A, fft_size)[:n]
    grad = 4.0 * n * cross
    return grad, obj

def lse_gradient_descent(a, time_budget, T_start=50.0, T_end=300.0, verbose=False):
    """LSE smoothed gradient descent with FFT gradient and backtracking."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_conv(a)
    start = time.monotonic()
    lr = 2.0 / (n * n)
    it = 0
    best_it = 0
    stall = 0
    max_stall = 80

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / time_budget)
        T = T_start * (T_end / T_start) ** frac

        grad, obj_cur = lse_grad_and_obj(a, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        a_old = a.copy()
        t = lr
        improved = False
        for _ in range(14):
            a_try = a_old - t * grad
            a_try = project_simplex(a_try)
            _, obj_try = lse_grad_and_obj(a_try, T)
            if obj_try < obj_cur - 1e-14:
                a = a_try.copy()
                improved = True
                break
            t *= 0.5

        if not improved:
            rng_l = np.random.default_rng(it + n)
            found = False
            for _ in range(4):
                pert = a + rng_l.normal(0, 0.005 * (1 + stall * 0.5), n)
                a_try = project_simplex(pert)
                _, obj_try = lse_grad_and_obj(a_try, T)
                if obj_try < obj_cur:
                    a = a_try.copy()
                    improved = True
                    found = True
                    break
            if not found:
                stall += 1
                if stall > max_stall:
                    lr *= 0.5
                    if lr < 1e-12:
                        break
                    stall = 0
                continue
        else:
            stall = 0
            lr = min(t * 1.1, 10.0 / (n * n))

        obj_true = compute_obj_conv(a)
        if obj_true < best_obj:
            best_obj = obj_true
            best_a = a.copy()
            best_it = it

        it += 1
        if verbose and it % 500 == 0:
            print(f"      LSE it{it}: obj={best_obj:.8f}, T={T:.1f}, lr={lr:.2e}, t={time.monotonic()-start:.1f}s")

    return best_a

def lp_polish(a, time_budget, max_iters=30, top_m_start=25, top_m_end=10, verbose=False):
    """LP-based Remez polishing: iteratively reduce top peaks."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj_conv(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start >= time_budget - 5:
            break
        frac = min(1.0, it / max(1, max_iters - 1))
        top_m = int(round(top_m_start + frac * (top_m_end - top_m_start)))
        top_m = max(5, min(top_m, 2 * n - 2))

        conv_cur = compute_conv(best_a)
        top_idx = np.argsort(conv_cur)[::-1][:top_m]

        A = np.zeros((top_m, n))
        for idx, k in enumerate(top_idx):
            k = int(k)
            for m in range(n):
                j = k - m
                if 0 <= j < n:
                    A[idx, m] = best_a[j]

        g_var = cp.Variable(n, nonneg=True)
        c_var = cp.Variable()
        constraints = [A @ g_var <= c_var, cp.sum(g_var) == 1]
        prob = cp.Problem(cp.Minimize(c_var), constraints)

        solvable = False
        for solver in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
            try:
                prob.solve(solver=solver)
                if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g_var.value, 0)
                    s_g = float(np.sum(g_val))
                    if s_g > 1e-10:
                        g_val = g_val / s_g
                        solvable = True
                        break
            except Exception:
                continue

        if not solvable:
            break

        best_t = 0.0
        best_obj_blend = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * best_a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 0.01:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_conv(a_try)
            if obj_try < best_obj_blend:
                best_obj_blend = obj_try
                best_t = t

        if best_t > 0 and best_obj_blend < best_obj - 1e-14:
            a_new = (1 - best_t) * best_a + best_t * g_val
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj_conv(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1

        if verbose and (it + 1) % 5 == 0:
            print(f"      LP it{it+1}: obj={best_obj:.8f}, top_m={top_m}, t={best_t:.2f}, elapsed={time.monotonic()-start:.1f}s")

        if stall >= 5:
            rng_p = np.random.default_rng(it * 104729)
            noise = rng_p.normal(0, 0.002 * stall, n)
            a_pert = best_a + noise
            a_pert = np.maximum(a_pert, 0)
            s_p = float(np.sum(a_pert))
            if s_p > 0.01:
                best_a = a_pert / s_p
                best_obj = compute_obj_conv(best_a)
            stall = 0

    return best_a

def coordinate_descent(a, time_budget, max_sweeps=2):
    """Coordinate descent: try scaling each coordinate."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj_conv(a)
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start >= time_budget:
            break
        any_improved = False
        for i in range(n):
            if time.monotonic() - start >= time_budget:
                break
            ai = a[i]
            if ai < 1e-8:
                continue
            best_val = ai
            best_local = best_obj
            for factor in [0.0, 0.3, 0.6, 1.4, 2.0]:
                a[i] = ai * factor
                s_new = float(np.sum(a))
                if s_new < 0.01:
                    a[i] = ai
                    continue
                a_try = a / s_new
                obj_try = compute_obj_conv(a_try)
                if obj_try < best_local:
                    best_local = obj_try
                    best_val = ai * factor
            a[i] = best_val
            if best_val != ai:
                s_new = float(np.sum(a))
                if s_new > 0.01:
                    a = a / s_new
                    any_improved = True
        obj_now = compute_obj_conv(a)
        if obj_now < best_obj:
            best_obj = obj_now
            best_a = a.copy()
        if not any_improved:
            break
    return best_a

def u_shaped_init(n, sharpness):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.01 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def power_init(n, alpha):
    """Edge-concentrated power law."""
    x = np.linspace(0, 1, n)
    d = np.abs(x - 0.5)
    a = np.maximum(d, 1.0 / (2 * n)) ** (-alpha) + 0.001
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def diverse_init(n, mode):

    s = float(np.sum(a))

            noise_scale = 0.003 * (round_num + 1)
            noise = rng.normal(0, noise_scale, n_cur)
            a_pert = best_seq + noise
            a_pert = np.maximum(a_pert, 0)
            s_p = float(np.sum(a_pert))
            if s_p < 0.01:
                continue
            a_pert = a_pert / s_p

            t_lse = min(60.0, remaining * 0.25)
            if t_lse > 10:
                a_pert = lse_gradient_descent(a_pert, time_budget=t_lse, T_start=50.0, T_end=200.0)
            t_lp = min(40.0, remaining * 0.15)
            if t_lp > 10:
                a_pert = lp_polish(a_pert, time_budget=t_lp, max_iters=10, top_m_start=20, top_m_end=10)

            obj_pert = compute_obj_conv(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                print(f"  ** Perturb {round_num}: obj={best_score:.10f}")
            else:
                print(f"     Perturb {round_num}: obj={obj_pert:.10f}")

    # === Phase 6: LP round 2 ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 6: LP R2 (budget={min(150, remaining*0.2):.0f}s) ===")
        alloc = min(150.0, remaining * 0.2)
        a_lp2 = lp_polish(best_seq, time_budget=alloc, max_iters=30, top_m_start=20, top_m_end=8, verbose=True)
        obj_lp2 = compute_obj_conv(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  ** LP2: obj={best_score:.10f}")
        else:
            print(f"     LP2: obj={obj_lp2:.10f}")

    # === Phase 7: Length exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 7: Length Exploration ===")
        n_best = len(best_seq)
        n_options = []
        for delta in [50, 100, 150, -50, -100]:
            n_try = n_best + delta
            if 300 <= n_try <= 2500:
                n_options.append(n_try)
        n_options = list(set(n_options))[:5]
        print(f"  n_current={n_best}, trying: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            a_try, obj_try = try_length(best_seq, n_try, remaining * 0.6)
            if a_try is not None and obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.10f}")
            else:
                print(f"     n={n_try}: obj={obj_try:.10f}")

    # === Phase 8: Final intensive LSE ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 8: Final LSE (budget={min(100, remaining*0.3):.0f}s) ===")
        alloc = min(100.0, remaining * 0.3)
        a_fin = lse_gradient_descent(best_seq, time_budget=alloc, T_start=200.0, T_end=1000.0, verbose=True)
        obj_fin = compute_obj_conv(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  ** Final LSE: obj={best_score:.10f}")
        else:
            print(f"     Final LSE: obj={obj_fin:.10f}")

    # === Phase 9: Final LP ===
    remaining = deadline - time.monotonic()
    if remaining > 25:
        print(f"\n=== Phase 9: Final LP ===")
        alloc = min(80.0, remaining * 0.4)
        a_flp = lp_polish(best_seq, time_budget=alloc, max_iters=20, top_m_start=15, top_m_end=6)
        obj_flp = compute_obj_conv(a_flp)
        if obj_flp < best_score:
            best_score = obj_flp
            best_seq = a_flp.copy()
            print(f"  ** Final LP: obj={best_score:.10f}")

    # === Phase 10: Coordinate descent ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 10: Coordinate Descent ===")
        alloc = min(60.0, remaining * 0.5)
        a_cd = coordinate_descent(best_seq, time_budget=alloc, max_sweeps=2)
        obj_cd = compute_obj_conv(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  ** CD: obj={best_score:.10f}")
        else:
            print(f"     CD: obj={obj_cd:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")
    return result
