```python
def multiplicative_update(a, eta=0.1, n_iters=1000):
    n = len(a)
    a = a.copy()
    for it in range(n_iters):
        conv = compute_conv(a)
        k_max = np.argmax(conv)
        # Gradient of conv[k_max] wrt a[i]
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_max - i
            j2 = i - k_max
            if 0 <= j1 < n:
                grad[i] += a[j1]
            if 0 <= j2 < n:
                grad[i] += a[j2]
        # Multiplicative update
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-10)  # keep positive
        a = a / a.sum()  # normalize
    return a
```

```python
def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size

def compute_conv(a):
    n = len(a)
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))

def gradient_at_max(a):
    """Compute gradient of max convolution lag wrt a."""
    n = len(a)
    conv = compute_conv(a)
    k_max = int(np.argmax(conv))
    grad = np.zeros(n)
    for i in range(n):
        j = k_max - i
        if 0 <= j < n:
            grad[i] += a[j]
        j2 = i - k_max
        if 0 <= j2 < n:
            grad[i] += a[j2]
    return grad, k_max

def solve_lp(a):

    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ b <= M, cp.sum(b) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:

            if b.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(b.value, 0)
                s = float(np.sum(b_val))
```

softer gradient via temperature-based lag weighting, diverse perturbations.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def compute_conv(a):
    n = len(a)
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def gradient_at_max(a):
    n = len(a)
    conv = compute_conv(a)
    k_max = int(np.argmax(conv))
    grad = np.zeros(n)
    for i in range(n):
        j = k_max - i
        if 0 <= j < n:
            grad[i] += a[j]
        j2 = i - k_max
        if 0 <= j2 < n:
            grad[i] += a[j2]
    return grad, k_max, conv


def gradient_soft(a, temperature=0.05):
    """Weighted gradient using softmax over top lags for smoother descent."""
    n = len(a)
    conv = compute_conv(a)
    # Threshold to top lags
    threshold = conv.max() - temperature
    active = conv[conv >= threshold]
    if len(active) == 0:
        active = np.array([conv.max()])
    # Softmax weights over active lags
    shifted = conv - conv.max()
    weights = np.exp(shifted / (temperature * 0.1 + 1e-10))
    weights = weights / weights.sum()
    # Only use significant weights
    weights[weights < 1e-6] = 0
    weights = weights / weights.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-8:
            continue
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += w * a[j]
            j2 = i - k
            if 0 <= j2 < n:
                grad[i] += w * a[j2]
    return grad


def multiplicative_updates(a, time_budget=60.0, eta0=0.05, decay=0.9995, use_soft=False):
    """Mirror descent (exponentiated gradient) on the simplex with entropy regularizer."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta0
    it = 0
    patience = 0
    min_obj = best_obj

    while time.monotonic() - start < time_budget:
        batch = 5
        for _ in range(batch):
            if use_soft and it % 10 == 0:
                grad = gradient_soft(a, temperature=0.05)
            else:
                grad, _, _ = gradient_at_max(a)
            a = a * np.exp(-eta * grad)
            a = np.maximum(a, 1e-15)
            s = a.sum()
            if s < 1e-10:
                a = best_a.copy()
            else:
                a = a / s
        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        eta *= decay
        it += 1
        if patience > 300 and eta < eta0 * 0.005:
            rng = np.random.default_rng(42 + it)
            freq = rng.integers(1, max(2, n // 40))
            phase = rng.uniform(0, 2 * np.pi)
            idx = np.arange(n, dtype=np.float64)
            pert = 0.04 * np.cos(2 * np.pi * freq * idx / n + phase)
            a = best_a * np.exp(pert)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            eta = eta0 * 0.5
            patience = 0

    return best_a, best_obj


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ b <= M, cp.sum(b) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if b.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(b.value, 0)
                s = float(np.sum(b_val))
                if s > 1e-10:
                    return b_val / s
        except Exception:
            continue
    return None


def line_search(a, direction, t_max=2.0):
    best_t = 0.0
    best_obj = compute_obj(a)
    ts = np.linspace(0, t_max, 40)
    for t in ts:
        a_try = (1 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 0.01:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    if 0 < best_t < t_max:
        step = t_max / 40
        lo = max(0.0, best_t - 2 * step)
        hi = min(t_max, best_t + 2 * step)
        ts_fine = np.linspace(lo, hi, 25)
        for t in ts_fine:
            a_try = (1 - t) * a + t * direction
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 0.01:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t
    return best_t, best_obj


def lp_optimize(a, time_budget=200.0):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        direction = solve_lp(a)
        if direction is None:
            break
        t_opt, obj_opt = line_search(a, direction)
        improved = False
        if t_opt > 1e-8 and obj_opt < best_obj - 1e-15:
            a_new = (1 - t_opt) * a + t_opt * direction
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 0.01:
                a_new = a_new / s_new
                a = a_new
                best_obj = compute_obj(a)
                best_a = a.copy()
                stall = 0
                improved = True
        if not improved:
            obj_direct = compute_obj(direction)
            if obj_direct < best_obj - 1e-15:
                a = direction.copy()
                best_obj = obj_direct
                best_a = a.copy()
                stall = 0
                improved = True
        if not improved:
            stall += 1

        if stall >= 3:
            rng = np.random.default_rng(2000 + it)
            mode = stall % 4
            if mode == 0:
                freq = rng.integers(1, max(2, n // 30))
                phase = rng.uniform(0, 2 * np.pi)
                idx = np.arange(n, dtype=np.float64)
                pert = 0.03 * min(stall, 5) * np.cos(2 * np.pi * freq * idx / n + phase)
                a = best_a * np.exp(pert)
            elif mode == 1:
                pert = rng.normal(0, 0.015 * min(stall, 5), n)
                kernel = np.ones(3) / 3
                pert = np.convolve(pert, kernel, mode='same')
                a = best_a * np.exp(pert)
            elif mode == 2:
                x = np.linspace(0, 1, n)
                sharp = rng.uniform(2, 8)
                u = 1.0 / (0.05 + sharp * (2 * x - 1) ** 2)
                u = u / u.sum()
                blend = rng.uniform(0.2, 0.5)
                a = (1 - blend) * best_a + blend * u
            else:
                # Mirror around center
                a = best_a[::-1].copy()
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stall = 0
        it += 1

    return best_a


def u_shaped_init(n, sharpness=3.0):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 0 else np.ones(n) / n


def diverse_init(n, rng, mode=0):
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.02 + 5.0 * (2 * x - 1) ** 4)
    elif mode == 6:
        a = 1.0 / (0.01 + 10.0 * (2 * x - 1) ** 6)
    elif mode == 7:
        a = (2 * np.abs(x - 0.5)) ** 8 + 0.01
    elif mode == 8:
        # Two-bump
        a = np.exp(-((idx - n * 0.2) ** 2) / (2 * (n * 0.08) ** 2)) + \
            np.exp(-((idx - n * 0.8) ** 2) / (2 * (n * 0.08) ** 2)) + 0.005
    elif mode == 9:
        # Power-law U
        a = np.abs(x - 0.5) ** 0.5 + 0.1
    else:
        a = 1.0 / (0.1 + 2.0 * np.abs(2 * x - 1))
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 0 else np.ones(n) / n


def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = a_dst.sum()
    if s < 0.01:
        return None
    return a_dst / s
