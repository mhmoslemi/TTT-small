"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses analytical coordinate descent with golden-section search for exact per-coordinate
optimization, combined with LP-based global moves, Fourier perturbations for restarts,
and length exploration. Key improvement over previous approach: instead of trying
discrete candidate values per coordinate, we analytically optimize each coordinate
via 1D golden-section search on the exact objective, yielding tighter convergence.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def find_optimal_coord(a, conv, j, n, n_iter=45):
    """
    Find optimal value of a[j] via golden-section search.
    Returns the optimal non-negative value.
    """
    old = a[j]
    S = float(np.sum(a))
    S_prime = S - old

    k_lo = j
    k_hi = min(j + n - 1, 2 * n - 2)

    # Unaffected max
    M_unaff = 0.0
    if k_lo > 0:
        M_unaff = max(M_unaff, float(np.max(conv[:k_lo])))
    if k_hi < 2 * n - 2:
        M_unaff = max(M_unaff, float(np.max(conv[k_hi + 1:])))

    # Affected entries: k in [k_lo, k_hi], where k-j in [0, n-1]
    ks = np.arange(k_lo, k_hi + 1)
    idx = ks - j  # a[k-j] indices, all in [0, n-1]

    # Separate quadratic (k=2j) from linear
    is_quad = (ks == 2 * j)
    lin_ks = ks[~is_quad]
    lin_idx = idx[~is_quad]

    # Precompute arrays for fast eval
    if len(lin_idx) > 0:
        A_prep = 2.0 * a[lin_idx]  # slopes
        B_prep = conv[lin_ks]      # current conv values (intercepts at x=old)
    quad_base = conv[2 * j] - old * old  # so b[2j](x) = quad_base + x^2

    def eval_g(x):
        """max convolution entry as function of x=a[j]"""
        max_val = M_unaff
        # Quadratic
        qv = quad_base + x * x
        if qv > max_val:
            max_val = qv
        # Linear
        if len(lin_idx) > 0:
            lin_max = float(np.max(B_prep + A_prep * (x - old)))
            if lin_max > max_val:
                max_val = lin_max
        return max_val

    def eval_f(x):
        """Objective as function of x=a[j]"""
        S_new = S_prime + x
        if S_new <= 1e-15:
            return np.inf
        return 2.0 * n * eval_g(x) / (S_new * S_new)

    # Determine search bounds
    f0 = eval_f(0.0)
    f_old = eval_f(old)

    # Upper bound: expand until objective exceeds f0
    x_max = max(4.0 / n, 3.0 * old + 1.0 / n)
    while eval_f(x_max) < f0 and x_max < 50.0:
        x_max *= 2.0

    if x_max < 1e-12:
        return old  # can't improve

    # Golden section search on [0, x_max]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    ga = 0.0
    gb = x_max
    gc = gb - gr * (gb - ga)
    gd = ga + gr * (gb - ga)
    fc = eval_f(gc)
    fd = eval_f(gd)

    for _ in range(n_iter):
        if fc < fd:
            gb = gd
            gd = gc
            fd = fc
            gc = gb - gr * (gb - ga)
            fc = eval_f(gc)
        else:
            ga = gc
            gc = gd
            fc = fd
            gd = ga + gr * (gb - ga)
            fd = eval_f(gd)

    x_opt = (ga + gb) / 2.0
    return max(0.0, x_opt)


def cd_sweep(a, conv, n, n_iter=45, rng=None, order='random'):
    """
    One full sweep of analytical coordinate descent.
    Returns (new_a, new_conv, improvement).
    """
    if order == 'random' and rng is not None:
        order = list(rng.permutation(n))
    elif order == 'random':
        order = list(np.random.permutation(n))

    total_improvement = 0.0

    for j in order:
        old_val = a[j]
        new_val = find_optimal_coord(a, conv, j, n, n_iter=n_iter)

        delta = new_val - old_val
        if abs(delta) < 1e-15:
            continue

        # Update a
        a[j] = new_val

        # Update conv in-place
        k_lo = j
        k_hi = min(j + n - 1, 2 * n - 2)

        # Linear updates for k != 2j
        ks = np.arange(k_lo, k_hi + 1)
        idx = ks - j
        lin_mask = (ks != 2 * j)
        if np.any(lin_mask):
            conv[ks[lin_mask]] += 2.0 * delta * a[idx[lin_mask]]

        # Quadratic update for k = 2j
        conv[2 * j] = conv[2 * j] - old_val * old_val + new_val * new_val

        # Clip tiny negatives from numerical error
        if conv[2 * j] < 0:
            conv[2 * j] = 0.0

    return a, conv, total_improvement


def cd_loop(a, time_budget, rng, verbose=True, n_iter=45, stall_patience=8):
    """
    Run coordinate descent until time_budget expires or stalled.
    Returns best (a, obj) found.
    """
    n = len(a)
    a = a.copy()
    conv = compute_conv(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return a, np.inf
    a = a / S
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    sweep = 0
    stall = 0
    prev_obj = best_obj

    while time.monotonic() - start < time_budget:
        sweep += 1
        a, conv, _ = cd_sweep(a, conv, n, n_iter=n_iter, rng=rng, order='random')

        # Periodic recompute to avoid drift
        if sweep % 50 == 0:
            conv = compute_conv(a)

        curr_obj = 2.0 * n * float(np.max(conv))

        if curr_obj < best_obj - 1e-14:
            best_obj = curr_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, curr={curr_obj:.10f}, "
                  f"stall={stall}, elapsed={elapsed:.1f}s")

        if stall >= stall_patience:
            # Perturb and reset
            break

    return best_a, best_obj


def lp_step(a, max_iters=30):
    """
    Simple LP-based step: solve min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Then blend. Returns improved a or None.
    Uses a simple iterative projection method instead of full LP for speed.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    M = float(np.max(compute_conv(a_norm)))

    # Initialize g = a_norm
    g = a_norm.copy()

    for it in range(max_iters):
        # Compute conv(a_norm, g)
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        fa = rfft(a_norm, size)
        fg = rfft(g, size)
        ag_conv = irfft(fa * fg, size)[:2 * n - 1]

        # Find violations
        excess = ag_conv - M
        if np.max(excess) < 1e-10:
            break

        # Simple multiplicative update: for g[j], the contribution to ag_conv[k]
        # for k in [j, j+n-1] is a_norm[k-j]*g[j]
        # Reduce g[j] proportional to excess where it contributes
        reduction = np.zeros(n)
        for k in range(2 * n - 1):
            if excess[k] > 0:
                j_lo = max(0, k - n + 1)
                j_hi = min(n - 1, k)
                # g[j] contributes a_norm[k-j]*g[j] to ag_conv[k]
                indices = np.arange(j_lo, j_hi + 1)
                coeffs = a_norm[k - indices]
                reduction[indices] += excess[k] * coeffs

        # Update: g *= (1 - lr * reduction / max_reduction)
        max_red = np.max(reduction)
        if max_red > 0:
            lr = 0.5
            g *= (1.0 - lr * reduction / max_red)
            g = np.maximum(g, 0)
            s = float(np.sum(g))
            if s < 1e-10:
                g = a_norm.copy()
            else:
                g /= s

    # Blend
    best_a = a_norm.copy()
    best_obj = 2.0 * n * M
    for t in [0.3, 0.5, 0.7, 1.0]:
        a_try = (1.0 - t) * a_norm + t * g
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try /= s_try
        obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a


def fourier_perturb(a, rng, strength=0.02):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 8):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def noise_perturb(a, rng, strength=0.03):
    """Add multiplicative noise perturbation."""
    n = len(a)
    noise = rng.standard_normal(n) * strength
    a_new = a * np.exp(noise)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s
