def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:

    size = 1
    while size < 2 * n - 1:

    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lp_solve_active(a, active_lags, solver_list=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for k in active_lags, g >= 0, sum(g)=1.
    Much faster than full LP when active_lags is small.
    Returns (g, M, all_violated_lags).
    """
    n = len(a)
    m = len(active_lags)
    
    # Build sparse constraint matrix for active lags
    # (a*g)[k] = sum_i a[i]*g[k-i] for valid i
    rows = []
    cols = []
    vals = []
    
    for idx, k in enumerate(active_lags):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            j = k - i
            if 0 <= j < n and a[i] > 1e-12:
                rows.append(idx)
                cols.append(j)
                vals.append(a[i])
    
    if len(vals) == 0:
        g = np.ones(n) / n
        return g, 0.0, []
    
    import scipy.sparse as sp
    A_mat = sp.csr_matrix((vals, (rows, cols)), shape=(m, n))
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A_mat @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    g_val = g_val / sg
                    M_val = float(M_var.value)
                    # Check for violated non-active constraints
                    conv_full = compute_conv(g_val)
                    M_current = float(np.max(conv))
                    violated = np.where(conv_full > M_val * 1.001 + 1e-10)[0]
                    return g_val, M_val, list(violated)
        except Exception:
            continue
    return None, None, []


def lp_solve_full(a, solver_list=None):
    """Solve full LP: min M s.t. (a*g)[k] <= M for all k, g >= 0, sum(g)=1."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.MOSEK, cp.SCS]
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None


def active_set_lp_iteration(a, time_budget, verbose=False, seed=42, max_iters=2000,
                            active_frac=0.05, min_active=10, max_active=200):
    """
    LP fixed-point with adaptive active set.
    Only solve LP with constraints near the maximum, then expand active set.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall_count = 0
    t_current = 0.15
    consecutive_fails = 0

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        
        # Determine active lags: those where conv >= (1 - active_frac) * max_val
        threshold = max_val * (1.0 - active_frac)
        active_lags = list(np.where(conv >= threshold)[0])
        
        # Ensure minimum number of active lags
        if len(active_lags) < min_active:
            # Add nearest lags to reach minimum
            all_lags = list(range(2 * n - 1))
            active_set = set(active_lags)
            remaining_lags = [l for l in all_lags if l not in active_set]
            remaining_lags.sort(key=lambda l: max_val - conv[l])
            for l in remaining_lags[:max(0, min_active - len(active_lags))]:
                active_lags.append(l)
        
        # Cap active set size
        if len(active_lags) > max_active:
            active_lags = sorted(active_lags, key=lambda l: -conv[l])[:max_active]

        g_lp, M_lp, violated = lp_solve_active(a, active_lags)
        
        if g_lp is None:
            consecutive_fails += 1
            if consecutive_fails > 10:
                break
            continue
        consecutive_fails = 0

        # Try blending with various t values
        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t_current

        t_candidates = [0.01, 0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj - 1e-13:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t

        if best_local_obj < best_obj - 1e-13:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a
            t_current = max(0.02, min(1.0, best_local_t * 0.9))
            stall_count = 0
            if verbose and it % 50 == 0:
                print(f"    AS-LP it{it:4d}: obj={best_obj:.10f}, t={best_local_t:.3f}, active={len(active_lags)}")
        else:
            stall_count += 1
            # Perturbation strategies
            if stall_count % 5 == 1:
                # Target the peak lags directly
                peak_lags = np.where(conv >= max_val * 0.999)[0]
                for lag in peak_lags[:2]:
                    grad = np.zeros(n)
                    i_min = max(0, lag - (n - 1))
                    i_max = min(lag, n - 1)
                    for i in range(i_min, i_max + 1):
                        j = lag - i
                        if 0 <= j < n:
                            grad[i] += 2.0 * a[j]
                    gn = np.linalg.norm(grad)
                    if gn > 1e-15:
                        a = a - 0.005 * grad / gn
            elif stall_count % 5 == 3:
                # Random perturbation
                a = a + 0.01 * rng.standard_normal(n)
            else:
                # Low-frequency perturbation
                freqs = rng.integers(1, 8, size=2)
                phases = rng.uniform(0, 2 * np.pi, size=2)
                x = np.arange(n)
                pert = np.zeros(n)
                for f, p in zip(freqs, phases):
                    pert += 0.01 * np.sin(2 * np.pi * f * x / n + p)
                a = a + pert
            
            a = np.maximum(a, 0)
            sa = float(np.sum(a))
            if sa > 1e-10:
                a = a / sa
            else:
                a = best_a.copy()

        if stall_count > 30:
            # Larger restart
            a = best_a.copy()
            strength = min(0.05, 0.01 * (stall_count // 10))
            a = a + strength * rng.standard_normal(n)
            a = np.maximum(a, 0)
            sa = float(np.sum(a))
            if sa > 1e-10:
                a = a / sa
            stall_count = 0
            t_current = 0.05

    return best_a


def smoothed_gd(a, time_budget, verbose=False, seed=42):
    """
    Adaptive log-sum-exp smoothed gradient descent with Nesterov acceleration.
    Uses adaptive tau that increases over time for sharper max approximation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Adaptive tau schedule: start smooth, get sharper
    tau = 5.0
    lr = 3.0 / np.sqrt(n)
    beta = 0.9
    v = np.zeros(n)
    no_improve = 0

    while time.monotonic() - start < time_budget:
        y = a + beta * v
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy < 1e-10:
            y = best_a.copy()
            v = np.zeros(n)
            sy = 1.0
        y = y / sy

        # Compute smoothed gradient
        conv = compute_conv(y)
        exact_max = float(np.max(conv))
        shifted = tau * (conv - exact_max)
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = exp_vals / sum_exp

        # Gradient via FFT
        N = 1
        while N < 2 * n + len(conv):
            N <<= 1
        P = fft(p, N)
        A = fft(y, N)
        cross = ifft(P * np.conj(A)).real
        g = 2.0 * cross[:n]
        
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
            lr *= 0.5
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Increase tau for sharper gradient
                tau *= 1.5
                lr *= 0.7
                v = np.zeros(n)
                no_improve = 0
                if tau > 5000:
                    # Add perturbation to escape
                    a = best_a + 0.02 * rng.standard_normal(n)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    best_a = a.copy()

        a = a_new
        lr = max(1e-8, lr * (1 - 1e-6))

    return best_a


def minimax_game(a, time_budget, verbose=False, seed=42, max_iters=500):
    """
    Minimax / zero-sum game approach.
    Player 1: choose probability y over lags to maximize min_i (a*y_conv)_i
    Player 2: choose a (non-negative, sum=1) to minimize max_k (a*a)[k]
    
    This is equivalent to the primal-dual of the LP but implemented as
    an alternating optimization.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Initialize y (dual variable) uniform over lags
    m = 2 * n - 1
    y = np.ones(m) / m

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        # Step 1: Given y, find best response for a
        # min_a max_k (a*a)[k] -> approximately min_a sum_k y_k * (a*a)[k]
        # = min_a sum_k y_k * sum_i a[i]*a[k-i]
        # = min_a a^T (Y) a where Y[i,j] = sum_k y_k * A[k,i] * A[k',j]... 
        # Actually: sum_k y_k * (a*a)[k] = sum_k y_k * sum_i a[i]*a[k-i]
        # = sum_i sum_j a[i]*a[j] * y[|i-j| + something]... 
        # This is a quadratic form: a^T Q a where Q[i,j] = y[|i-j|] (approximately)
        
        # Simpler: use subgradient of sum_k y_k * (a*a)[k]
        conv = compute_conv(a)
        weighted = np.sum(y * conv)
        
        # Gradient of sum_k y_k * conv(a,a)[k] with respect to a
        # d/da[i] sum_k y_k * conv(a,a)[k] = 2 * sum_k y_k * (a shifted)[k-i]
        # = 2 * (y_conv_with_a)[i] where y_conv_with_a is correlation of y and a
        N2 = 1
        while N2 < 2 * n + m:
            N2 <<= 1
        Y = fft(y, N2)
        A = fft(a, N2)
        cross = ifft(np.conj(Y) * A).real
        grad = 2.0 * cross[:n]
        
        # Normalize and step
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn
        
        # Line search for step size
        improved = False
        for step in [0.001, 0.003, 0.005, 0.01, 0.02, 0.03]:
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        # Step 2: Given a, update y (follow the max of conv(a,a))
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        # Sharpen y towards the peaks
        tau_y = 50.0 + 50.0 * (it / max_iters)
        shifted_y = tau_y * (conv - max_val)
        shifted_y = np.clip(shifted_y, -500, 500)
        exp_y = np.exp(shifted_y)
        y_new = exp_y / exp_y.sum()
        # Blend for stability
        y = 0.7 * y + 0.3 * y_new

        if not improved and it % 10 == 0 and it > 50:
            # Perturb
            a = best_a + 0.01 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_init(n, mode, rng):
    """Create structured initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.maximum(np.sinc(x), 0.001)
        return a / a.sum()
    elif mode == 2:
        a = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / max(1, n - 1))
        return a / a.sum()
    elif mode == 3:
        x = np.arange(n) / max(1, n - 1)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        return a / a.sum()
    elif mode == 4:
        x = np.linspace(-3, 3, n)
        a = np.exp(-x ** 2)
        return a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
        return a / a.sum()
    elif mode == 6:
        # Bell curve with wider base
        x = np.linspace(-4, 4, n)
        a = np.exp(-0.5 * x ** 2)
        return a / a.sum()
    elif mode == 7:
        # Piecewise: flat center, tapered edges
        a = np.ones(n)
        taper = 0.15 * n
        for i in range(n):
            d = min(i, n - 1 - i)
            if d < taper:
                a[i] = 0.5 + 0.5 * np.cos(np.pi * (taper - d) / taper)
        return a / a.sum()
    elif mode == 8:
        # Two-humped
        x = np.linspace(-3, 3, n)
        a = np.exp(-((x - 1) ** 2)) + np.exp(-((x + 1) ** 2))
        return a / a.sum()
    elif mode == 9:
        # Random smooth
        a = rng.uniform(0.5, 1.5, n)
        # Smooth it
        kernel_size = max(3, n // 20)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        a = rng.uniform(0.1, 1.0, n)
        return a / a.sum()

    best_seq = None
    best_score = np.inf

    # --- Load parent ---
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj_parent = compute_obj(parent)
                best_score = obj_parent
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.10f}")
    except Exception:

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0

                best_seq = a_screen.copy()

            else:

            best_score = obj3
            best_seq = a3.copy()

    # === Phase 4: Minimax game ===
    print(f"\n=== Phase 4: Minimax Game ===")

    if remaining > 80:
        alloc = min(remaining * 0.25, 200.0)
        a6 = active_set_lp_iteration(best_seq, time_budget=alloc, verbose=True,
                                     seed=9999, active_frac=0.02, min_active=5, max_active=100)
        obj6 = compute_obj(a6)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Phase6 AS-LP2: obj={best_score:.10f} ** IMPROVED")
        else:
            print(f"  Phase6 AS-LP2: obj={obj6:.10f}")

    # === Phase 7: Final GD with high tau ===
    print(f"\n=== Phase 7: Final High-Tau GD ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.3, 80.0)
        a7 = smoothed_gd(best_seq, time_budget=alloc, seed=42424)
        obj7 = compute_obj(a7)
        if obj7 < best_score:
            best_score = obj7
            best_seq = a7.copy()
            print(f"  Phase7 GD: obj={best_score:.10f} ** IMPROVED")
        else:
            print(f"  Phase7 GD: obj={obj7:.10f}")

    # === Phase 8: Full LP verification on best few candidates ===
    print(f"\n=== Phase 8: Full LP Verification ===")
    remaining = deadline - time.monotonic()
    if remaining > 30 and len(best_seq) <= 1000:
        g_full, M_full = lp_solve_full(best_seq)
        if g_full is not None:
            # Blend with full LP solution
            for t in [0.1, 0.2, 0.3, 0.5]:
                a_try = (1 - t) * best_seq + t * g_full
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                obj_try = compute_obj(a_try)
                if obj_try < best_score:
                    best_score = obj_try
                    best_seq = a_try.copy()
                    print(f"  Full LP blend t={t}: obj={best_score:.10f} ** IMPROVED")
                    break
        # Quick AS-LP one more time
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a8 = active_set_lp_iteration(best_seq, time_budget=min(remaining - 5, 30.0),
                                         verbose=False, seed=77777)
            obj8 = compute_obj(a8)
            if obj8 < best_score:
                best_score = obj8
                best_seq = a8.copy()
                print(f"  Final AS-LP: obj={best_score:.10f} ** IMPROVED")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, internal_score={best_score:.10f}, "
          f"eval_score={val:.10f}, time={elapsed:.1f}s ===")
    return result
