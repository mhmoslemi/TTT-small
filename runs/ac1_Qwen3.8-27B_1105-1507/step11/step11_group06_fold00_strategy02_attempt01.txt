                new_vertical = int(800*expansion_factor)
                try:
                    start_candidate = variation_mode(height_sequence_1, variation_type) if any_validated_dimension_avail else np.ones(new_vertical)/new_vertical
                    score = evaluate_sequence(start_candidate)
                    if score < best_score:
                        best_score = score
                        best_sequence = np.array(start_candidate)
                except:
                    start_candidate = np.ones(new_vertical)/new_vertical
                    score = evaluate_sequence(start_candidate)
            elif n == 800:  # Already handled parent
                continue
            else:
                start_candidate = variation_mode(parent, variation_type) if any_validated_dimension_avail else np.ones(new_vertical)/new_vertical

    elif area == 'secondary':
        # More systematic expansion
        new_vertical = int(800 * expansion_factor)
        for factor in variation_range:
            if new_vertical < 300 or new_vertical > 2500:
                continue
            # Will do diverse initialization including magnitude changes
            pass

    return best_sequence.tolist().copy()

def propose_candidate(seed=42, budget_s=1500):
    allocated_time = budget_s
    family = None

    # --- Primary Optimization ---
    try:
        # Check if parent is available
        if height_sequence_1 is not None:
            n0 = len(height_sequence_1)
            s0 = np.sum([float(x) for x in height_sequence_1])
            if s0 > 0.01:
                parent_arr = np.asarray(height_sequence_1)
                norm_parent = parent_arr / s0
                score_parent = 2.0 * n0 * float(np.max(compute_conv(norm_parent))) / s0**2
            else:
                norm_parent = np.ones(n0) / n0
                score_parent = evaluate_sequence(norm_parent)
        # But note: the parent might be normalized already
    except Exception:
        # Resolute fallback: no parent available
        n0 = 800
        family = 'initial'

    best_score = score_parent if family == 'initial' else np.inf
    best_sequence = norm_parent if family == 'initial' and score_parent is not np.inf else None

    # --- Localizing and Structuring ---
    if best_sequence is not None:
        # Use advanced search to improve
        try:
            time_budget = 1450  # Leave 50 seconds for the cleanup
            optimize_result = minimize(score_parent, best_sequence, 
                                      method='L-BFGS-B', 
                                     options={'disp': True, 'maxfev': 10000, 'gtol': 1e-5, 'eps': 1e-8})
            new_seq = optimize_result.x
            new_score = evaluate_sequence(new_seq)
            if new_score < best_score:
                best_score = new_score
                best_sequence = new_seq
        except Exception as e:
            raise e  # Rely on scip return codes

    # --- Ensure Length and Computation ---
    try:
        optimize_result = dual_annealing(lambda x: evaluate_sequence(x), 
                                        bounds=[(0, 1000) for _ in range(n0)], 
                                        max_evals= int(1400/time_per_eval)  # Adjust for time_per_eval being the duration of evaluate_sequence call
        new_seq = optimize_result.x
        best_sequence = new_seq
    except:
        pass

    # --- Adjust Geometrically ---
    def geometric_center(seq, center=0.5):
        n = len(seq)
        # Normalize and bring back to lower scores
        s_val = np.sum(seq)
        if s_val < 0.01:
            seq = np.ones(n)/n * (0.01/len(seq))
        return geometric_shrink(seq, 2) if center<0.5 else geometric_enlarge(seq, 2)

    def geometric_shrink(seq, factor):
        n = len(seq)
        new_seq = np.zeros(n)
        for i in range(n):
            if i // (n//factor) < factor and factor*factor>=0.9*0.9:
                new_seq[i] = seq[i] / factor
            elif i == n-1:
                new_seq[i] = seq[i] / factor * decrease
        # Then check sum, redo until sum>=0.01
        while np.sum(new_seq) < 0.01:
            factor *= 0.99  # Decrease geometrically until we find a valid configuration
            new_seq = np.zeros(n)
            for i in range(n):
                new_seq[i] = seq[i] / factor
        return new_seq

    best_sequence = geometric_center(best_sequence, center=0.7)  # Higher center for more overall size

    # Final defensive check
    s_val = np.sum(best_sequence)
    if s_val < 0.01:
        best_sequence = np.ones(len(best_sequence)) / len(best_sequence) * 0.01  # Safe fallback
    else:
        best_sequence = best_sequence / s_val * 1  # Normalization to safe sum is handled by evaluate_sequence elsewhere

    # --- Final Return ---
    return [float(x) for x in best_sequence]
