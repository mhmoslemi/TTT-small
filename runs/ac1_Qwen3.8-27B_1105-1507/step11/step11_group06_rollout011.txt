    rho = np.argmax(1 - u / np.arange(1, n + 1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_descent(a, time_budget=60.0, lr0=0.1, decay=0.999):
    """Subgradient descent on the simplex for min 2n*max_k c[k]"""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr0
    
    # Use mirror average (like in previous code but with proper subgradient)
    a_avg = a.copy()
    momentum = np.zeros(n)
    beta = 0.5
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a_avg)
        k_max = int(np.argmax(conv))
        grad = compute_subgrad(a_avg, k_max)
        
        lr_t = lr / (1.0 + 0.001 * it)
        momentum = beta * momentum + lr_t * grad
        a_new = a_avg - momentum
        a_new = proj_simplex(a_new)
        
        a_avg = (it / (it + 1.0)) * a_avg + (1.0 / (it + 1.0)) * a_new
        
        obj = compute_obj(a_avg)
        if obj < best_obj:
            best_obj = obj
            best_a = a_avg.copy()
            best_it = it
        it += 1
        
        if it % 5000 == 0:
            lr *= decay
        
        if it - best_it > 5000:
            a_avg = best_a.copy()
            momentum = np.zeros(n)
            best_it = it
    
    return best_a


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, direction, t_max=1.3):
    """Coarse + fine line search for best t in [0, t_max]"""
    best_t = 0.0
    best_obj = compute_obj(a)
    
    # Coarse
    ts = np.linspace(0.0, t_max, 30)
    for t in ts:
        a_try = (1.0 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    # Fine
    if 0.0 < best_t < t_max:
        lo = max(0.0, best_t - 0.05)
        hi = min(t_max, best_t + 0.05)
        ts_fine = np.linspace(lo, hi, 25)
        for t in ts_fine:
            a_try = (1.0 - t) * a + t * direction
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t
    
    return best_t, best_obj


def lp_polish(a, time_budget=100.0, max_iters=100):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break
        
        t_opt, obj_opt = line_search(a, g_lp)
        if t_opt > 1e-6 and obj_opt < best_obj - 1e-12:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                best_obj = compute_obj(a_new)
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            obj_direct = compute_obj(g_lp)
            if obj_direct < best_obj:
                best_obj = obj_direct
                best_a = g_lp.copy()
                a = g_lp.copy()
                stall = 0
        
        it += 1
        if stall >= 5:
            # Perturb
            freq = rng.integers(1, max(2, n // 100))
            phase = rng.uniform(0, 2 * np.pi)
            idx = np.arange(n, dtype=np.float64)
            pert = 0.02 * np.cos(2 * np.pi * freq * idx / n + phase)
            a = best_a * np.exp(pert)
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 0.01:
                a = a / s
            else:
                a = best_a.copy()
            stall = 0
    
    return best_a


def coord_descent(a, time_budget=30.0, block_size=10):
    """Block coordinate descent: optimize blocks of consecutive elements"""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    
    while time.monotonic() - start < time_budget:
        improved = False
        # Random block
        block_start = rng.integers(0, n - block_size)
        idx = np.arange(block_start, block_start + block_size)
        
        # Try to optimize this block via small perturbations
        for _ in range(5):
            a_try = a.copy()
            # Add small random perturbation to block
            pert = rng.normal(0, 0.01, block_size)
            a_try[idx] = np.maximum(a_try[idx] + pert, 0)
            s_try = float(np.sum(a_try))
            if s_try < 0.01:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
        
        if not improved and rng.random() < 0.1:
            break
    
    return best_a


def u_shaped_init(n, sharpness=3.0):
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def param_family(n, p, alpha):
    """Parameterized family: a[i] = 1 + alpha*(2*i/n - 1)^p"""
    x = np.linspace(0, 1, n)
    t = 2 * x - 1
    a = 1.0 + alpha * np.abs(t) ** p
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def param_family2(n, alpha, beta, gamma):
    """More general: a[i] = alpha + beta*(2i/n-1)^2 + gamma*(2i/n-1)^4"""
    x = np.linspace(0, 1, n)
    t = 2 * x - 1
    a = alpha + beta * t**2 + gamma * t**4
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(a_src, n_dst):
    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s
