"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive mirror descent (multiplicative weights) with subgradient estimation,
LP-based direction finding, and multi-n exploration with simulated annealing restarts.
Key improvements: natural gradient updates, adaptive temperature scheduling,
efficient FFT-based convolution, and diverse initialization strategies.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def gradient_from_p(a, p):
    """
    Compute sum_j a[j] * p[i+j] for i=0..n-1, scaled by 2.
    p has length 2n-1 (distribution over convolution lags).
    This gives the gradient of sum_k p[k]*conv(a,a)[k] wrt a.
    """
    n = len(a)
    b = a[::-1]
    N_fft = 1
    while N_fft < 3 * n - 1:
        N_fft <<= 1
    c = irfft(rfft(b, N_fft) * rfft(p, N_fft), N_fft)
    return 2.0 * c[n - 1:2 * n - 1]


def lse_gradient(a, conv, tau):
    """Compute gradient of tau * logsumexp(conv/tau) which approximates max(conv)."""
    n = len(a)
    max_conv = float(np.max(conv))
    shifted = (conv - max_conv) / tau
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(2 * n - 1)
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    return gradient_from_p(a, p), p


def topk_subgradient(a, conv, k=5):
    """Subgradient using top-k peaks with equal weights."""
    n = len(a)
    idx = np.argsort(conv)[-k:]
    p = np.zeros(2 * n - 1)
    p[idx] = 1.0 / k
    return gradient_from_p(a, p)


def weighted_topk_subgradient(a, conv, k=5, beta=10.0):
    """Subgradient using top-k peaks with softmax weights."""
    n = len(a)
    top_vals = np.sort(conv)[-k:]
    top_vals_minus = top_vals - top_vals[-1]
    weights = np.exp(beta * top_vals_minus)
    weights /= weights.sum()
    idx = np.argsort(conv)[-k:]
    p = np.zeros(2 * n - 1)
    p[idx] = weights
    return gradient_from_p(a, p)


def mirror_descent_step(a, grad, eta):
    """Multiplicative weights update with normalization."""
    exponent = -eta * grad
    exponent = np.clip(exponent, -50, 50)
    a_new = a * np.exp(exponent)
    s = float(np.sum(a_new))
    if s < 1e-300:
        return a.copy()
    return a_new / s


def lp_direction(a, use_improved=True):
    """
    Solve LP to find descent direction.
    Standard: min M s.t. conv(a,g)_k <= M, sum(g)=1, g>=0.
    Improved: min M s.t. 2*conv(a,g)_k - conv(a,a)_k <= M.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv_aa = compute_conv(a_norm)

    A = np.zeros((2 * n - 1, n))
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A[valid] = a_norm[diff[valid]]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    if use_improved:
        constraints = [2.0 * (A @ g) - M <= conv_aa, cp.sum(g) == 1]
    else:
        constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, g, n_evals=30):
    """Line search for best t in [0,1]."""
    n = len(a)
    ts = np.linspace(0.05, 0.95, n_evals)
    objs = np.full(n_evals, np.inf)
    for i, t in enumerate(ts):
        a_try = np.maximum((1 - t) * a + t * g, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            objs[i] = compute_obj(a_try / s)
    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]
    return best_t, best_obj


def mirror_descent_optimize(a, time_budget, tau_init=50.0, tau_end=2000.0,
                            verbose=True, blend_k=5, blend_alpha=0.3, seed=42):
    """
    Adaptive mirror descent with mixed gradient strategies.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    it = 0
    best_it = 0
    stall = 0
    eta_running = 0.1

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))

        # Temperature schedule
        tau = tau_init + (tau_end - tau_init) * progress ** 0.8

        conv = compute_conv(a)

        # Mixed gradient strategy
        phase = (it // 10) % 4
        if phase == 0:
            # Pure top-k subgradient
            g = topk_subgradient(a, conv, k=blend_k)
        elif phase == 1:
            # LSE smoothed gradient
            g, _ = lse_gradient(a, conv, tau)
        elif phase == 2:
            # Weighted top-k
            g = weighted_topk_subgradient(a, conv, k=blend_k, beta=5.0 + 20.0 * progress)
        else:
            # Blend of LSE and top-k
            g_lse, _ = lse_gradient(a, conv, tau)
            g_top = topk_subgradient(a, conv, k=blend_k)
            g = blend_alpha * g_top + (1.0 - blend_alpha) * g_lse

        # Adaptive step size
        g_max = float(np.max(np.abs(g)))
        if g_max < 1e-15:
            break
        eta = 2.0 / (g_max * np.sqrt(1.0 + it * 0.1) + 1e-10)
        eta_running = 0.9 * eta_running + 0.1 * eta
        eta = eta_running * (1.5 - 0.5 * progress)

        a_new = mirror_descent_step(a, g, eta)

        # Periodic perturbation for diversity
        if it % 50 == 30 and progress < 0.85:
            temp = 0.005 * (1.0 - progress)
            pert = rng.standard_normal(n)
            a_new = a_new * np.exp(temp * pert)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall >= 30:
                # Escape: try smaller step
                a_half = mirror_descent_step(a, g, eta * 0.2)
                obj_half = compute_obj(a_half)
                if obj_half < best_obj - 1e-14:
                    best_obj = obj_half
                    best_a = a_half.copy()
                    a = a_half
                    stall = 0
                else:
                    # Larger perturbation
                    strength = 0.03 * (1.0 + progress)
                    a_pert = a * np.exp(strength * rng.standard_normal(n))
                    a_pert = np.maximum(a_pert, 0)
                    s_pert = float(np.sum(a_pert))
                    if s_pert > 1e-10:
                        a = a_pert / s_pert
                    stall = 0

        a = a_new
        it += 1

        if verbose and (it % 200 == 0 or it == 1):
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, "
                  f"eta={eta:.5f}, stall={stall}, t={time.monotonic()-start:.0f}s")

    return best_a


def lp_driven_optimize(a, time_budget, verbose=True, seed=123):
    """LP-driven optimization with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        use_imp = (it % 2 == 0)
        g, M = lp_direction(a, use_improved=use_imp)

        if g is None:
            conv = compute_conv(a)
            g_lse, _ = lse_gradient(a, conv, 300.0)
            g = np.maximum(g_lse, 0)
            s_g = float(np.sum(g))
            if s_g < 1e-10:
                break
            g = g / s_g

        best_t, obj_t = line_search(a, g, n_evals=25)

        if obj_t < best_obj - 1e-13:
            a_try = np.maximum((1 - best_t) * a + best_t * g, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                best_obj = obj_t
                best_a = a_try.copy()
                a = a_try
                stall = 0
                if verbose and (it % 3 == 0):
                    print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, "
                          f"M={M:.6f}, t_elapsed={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall >= 5:
                # Perturb and reset
                conv_p = compute_conv(a)
                g_p, _ = lse_gradient(a, conv_p, 500.0)
                lr = 0.02 / (float(np.linalg.norm(g_p)) + 1e-10)
                a = np.maximum(a - lr * g_p + 0.005 * rng.standard_normal(n), 0)
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new
                else:
                    a = best_a.copy()
                stall = 0
                if it > 100:
                    if verbose:
                        print(f"    LP: stopping after {it} iterations")
                    break

        it += 1

    return best_a


def gradient_descent_polish(a, time_budget, tau=1000.0, verbose=False):
    """Nesterov-accelerated gradient descent polishing."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.8
    base_lr = 1.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau_cur = tau + tau * 2.0 * progress

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        conv_y = compute_conv(y)
        g, _ = lse_gradient(y, conv_y, tau_cur)
        g_norm = float(np.linalg.norm(g))

        if g_norm > 1e-15:
            adapt = base_lr / (g_norm / base_lr + 1.0)
            lr = 0.85 * lr + 0.15 * adapt

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr *= 0.999

        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
            best_it = it

    if verbose:
        print(f"    GD polish: obj={best_obj:.8f}, iters={it}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        a_final2 = lp_driven_optimize(best_seq, time_budget=alloc, verbose=True, seed=666)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 40.0)
        a_final3 = gradient_descent_polish(best_seq, time_budget=alloc,
                                           tau=2000.0, verbose=True)
        obj_final3 = compute_obj(a_final3)
        if obj_final3 < best_score:
            best_score = obj_final3
            best_seq = a_final3.copy()
            print(f"  Final GD: obj={best_score:.8f} **")

    # Final validation with evaluate_sequence
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
