Optimizes step function coefficients using L-BFGS continuation method with
adaptive smoothing, LP fixed-point iteration, and multi-n exploration
to minimize 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
from scipy.signal import correlate as sp_correlate


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_objective_and_grad(z, tau):
    """
    Compute smoothed objective and gradient for L-BFGS optimization.
    z: parameter vector (will be converted to simplex via softmax-like normalization)
    tau: smoothing parameter (larger = smoother)
    
    We use the parameterization a = max(z, 0) / sum(max(z, 0)) to stay on the simplex.
    Actually, for L-BFGS-B, let's just work with a directly and use bounds.
    """
    # z is the raw variable, we clip to non-negative and normalize
    a = np.maximum(z, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        n = len(z)
        return 1e10, np.zeros(n)
    a = a / s
    n = len(a)
    
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    # Smoothed log-sum-exp
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    
    if sum_exp < 1e-300:
        smooth_term = 1.0 / tau
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        smooth_term = (1.0 / tau) * np.log(sum_exp)
        p = exp_vals / sum_exp
    
    obj_val = 2.0 * n * (max_conv + smooth_term)
    
    # Gradient computation using scipy.signal.correlate
    # G[m] = sum_k p[k] * a[k-m] for valid k
    # This is the cross-correlation of p and a
    G = sp_correlate(p, a, mode='full')
    # G has length (2n-1) + n - 1 = 3n-2
    # G[m] for m in [0, n-1] gives sum_k p[k] * a[k-m+n-1]... 
    # Hmm, I need to check the indexing of scipy.signal.correlate
    
    # scipy.signal.correlate(in1, in2, mode='full'):
    # result[n] = sum_k in1[k] * conj(in2[k - n + len(in2) - 1])
    # Actually, let me just check: for in1 of length L1 and in2 of length L2,
    # result has length L1+L2-1, and
    # result[k] = sum_n in1[n] * in2[n + k - (L2-1)] for valid n
    # Hmm, this is confusing. Let me just use a direct computation.
    
    # Direct gradient:
    # grad[m] = 2n * sum_k p[k] * d conv_k / d a_m
    # d conv_k / d a_m = 2*a[k-m] (if 0<=k-m<n and k!=2m) or a_m (if k=2m)
    #                  = 2*a[k-m]*[0<=k-m<n] - a_m*[k==2m]
    
    # So: grad[m] = 2n * (2*sum_{k=max(m,0)}^{min(m+n-1, 2n-2)} p[k]*a[k-m] - a[m]*p[2*m]*[2m<2n-1])
    
    grad = np.zeros(n)
    for m in range(n):
        k_start = m  # k - m >= 0 => k >= m
        k_end = min(m + n - 1, 2 * n - 2)  # k - m < n and k <= 2n-2
        if k_end >= k_start:
            indices = np.arange(k_start, k_end + 1)
            a_indices = indices - m  # 0 to a_indices[-1]
            grad[m] = 2.0 * np.dot(p[k_start:k_end+1], a[0:len(indices)])
        if 2 * m < 2 * n - 1:
            grad[m] -= a[m] * p[2 * m]
        grad[m] *= 2.0 * n
    
    # Account for the normalization (a = z / sum(z)):
    # d a_i / d z_j = (delta(i,j) - a_i) / s = delta(i,j) - a_i (since s=1 after normalization)
    # Wait, we have a = max(z,0)/sum(max(z,0)). The gradient through the max is tricky.
    # For L-BFGS-B with bounds [0, inf], we can just use the gradient of the objective
    # with respect to a (assuming a is in the interior of the feasible set).
    # At the boundary (a_i = 0), the gradient is only relevant for the subgradient.
    # L-BFGS-B handles this via the bounds.
    
    # But we need to account for the normalization. Since we normalize inside the
    # objective function, the "effective" gradient with respect to z (before normalization)
    # is: d obj / d z_j = sum_i (d obj / d a_i) * (d a_i / d z_j)
    # where a_i = z_i / sum(z) (assuming z_i > 0 for all i)
    # d a_i / d z_j = delta(i,j)/sum(z) - z_i / sum(z)^2 = a_i * (delta(i,j) - a_i) / a_i... 
    # Actually: a_i = z_i / S where S = sum(z).
    # d a_i / d z_j = delta(i,j)/S - z_j * z_i / S^2 = a_i/S * delta(i,j) - a_i * a_j / ... 
    # = (delta(i,j) - a_i * S / S) * ... 
    # Let me just do it carefully:
    # a_i = z_i / S, S = sum_k z_k
    # d a_i / d z_j = delta(i,j) / S - z_i * 1 / S^2 = delta(i,j)/S - a_i/S = (delta(i,j) - a_i) / S
    # Since S = sum(z) and a_i = z_i/S: (delta(i,j) - a_i) / S = delta(i,j)/S - a_i/S
    # Hmm, this has a 1/S factor. Since S is fixed during gradient computation (we already normalized),
    # for the L-BFGS purpose, we can just return the gradient with respect to a (the normalized variable)
    # and let L-BFGS-B handle the bounds. The normalization is just a reparameterization.
    
    # Actually, the simplest and most correct approach: since the objective is homogeneous
    # (f(ca) = f(a) for c > 0), the gradient on the simplex is the projection of the
    # Euclidean gradient onto the tangent space of the simplex.
    # Tangent space: {v : sum(v) = 0}
    # Projected gradient: grad - mean(grad) * 1 (subtract the mean to make it sum to 0)
    # But for L-BFGS-B with box constraints, we don't need the equality constraint handling.
    # We just need the gradient in the direction of feasible movements.
    
    # I'll just return the gradient as-is. L-BFGS-B will handle the rest.
    # The key insight: since we normalize a inside the objective, and L-BFGS-B works
    # with the raw variable z, we need the chain rule. But for simplicity and since
    # the function is homogeneous, the direction of improvement is the same whether
    # we work with a or z (as long as we maintain sum=1).
    
    return obj_val, grad


def lbfgs_continuation(a_init, time_budget, verbose=False, seed=42):
    """
    L-BFGS with continuation method on tau for the smoothed objective.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    tau_schedule = [200, 100, 50, 20, 10, 5, 2, 1, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
    
    for tau in tau_schedule:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        # Wrap the objective for L-BFGS
        def obj_wrapper(z):
            val, grad = smooth_objective_and_grad(z, tau)
            return val, grad
        
        # Run L-BFGS-B
        try:
            result = minimize(
                obj_wrapper, a, method='L-BFGS-B',
                jac=True,
                bounds=[(0, None)] * n,
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 50}
            )
            a_new = np.maximum(result.x, 0)
            sn = float(np.sum(a_new))
            if sn > 1e-10:
                a_new = a_new / sn
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    if verbose:
                        print(f"    tau={tau:6.2f}: obj={best_obj:.10f}, iters={result.nit}")
                else:
                    if verbose:
                        print(f"    tau={tau:6.2f}: obj={obj_new:.10f} (no improvement)")
            else:
                a = best_a.copy()
        except Exception as e:
            if verbose:
                print(f"    tau={tau:6.2f}: L-BFGS failed: {e}")
    
    return best_a


def lp_solve(a, solver_list=None):
    """Solve min M s.t. conv(a,g) <= M, g >= 0, sum(g) = 1."""
    n = len(a)
    # Build constraint matrix: (a*g)_k <= M for k = 0, ..., 2n-2
    # (a*g)_k = sum_i a_i * g_{k-i} (valid indices)
    # For each k, the constraint is: sum_{i: 0<=k-i<n} a_i * g_{k-i} <= M
    # Let j = k-i: sum_{j: 0<=j<n, 0<=k-j<n} a_{k-j} * g_j <= M
    
    # Build the matrix A where A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # For each g_j, it appears in constraints k = j, j+1, ..., j+n-1
        # with coefficients a[0], a[1], ..., a[n-1]
        A[j:j+n, j] = a
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None


def lp_fixed_point_v2(a, time_budget, verbose=False, seed=42, max_iters=300):
    """
    Improved LP fixed-point with line search and adaptive perturbation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall_count = 0
    t_current = 0.3
    consecutive_lp_fails = 0
    
    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        g_lp, M_lp = lp_solve(a)
        if g_lp is None:
            consecutive_lp_fails += 1
            if consecutive_lp_fails > 8:
                break
            continue
        consecutive_lp_fails = 0
        
        # Line search along direction d = g_lp - a
        d = g_lp - a
        best_local_obj = best_obj
        best_local_t = 0.0
        
        # Try a range of t values
        t_candidates = np.logspace(-3, 0, 20)  # 0.001 to 1.0, log-spaced
        for t in t_candidates:
            a_try = a + t * d
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_t = t
        
        if best_local_obj < best_obj - 1e-12:
            best_obj = best_local_obj
            # Recompute best_a with the best t
            a_try = a + best_local_t * d
            a_try = np.maximum(a_try, 0)
            best_a = a_try / np.sum(a_try)
            a = best_a.copy()
            t_current = best_local_t
            stall_count = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it:3d}: obj={best_obj:.10f}, t={best_local_t:.4f}, M_lp={M_lp:.6f}")
        else:
            stall_count += 1
            # Perturbation strategies
            strategy = stall_count % 5
            if strategy == 1:
                # Random perturbation
                a = a + 0.02 * rng.standard_normal(n)
            elif strategy == 2:
                # Target active set: reduce max lags
                conv = compute_conv(a)
                max_val = float(np.max(conv))
                active = np.where(conv >= max_val * 0.999)[0]
                for lag in active[:5]:
                    grad_lag = np.zeros(n)
                    for i in range(n):
                        j = int(lag) - i
                        if 0 <= j < n:
                            if i != j:
                                grad_lag[i] += 2.0 * a[j]
                            else:
                                grad_lag[i] += a[j]
                    gn = np.linalg.norm(grad_lag)
                    if gn > 1e-15:
                        a = a - 0.005 * grad_lag / gn
            elif strategy == 3:
                # Blend with uniform
                a = 0.95 * a + 0.05 / n
            elif strategy == 4:
                # Spectral perturbation
                x = np.arange(n)
                f = rng.integers(1, n//4 + 1)
                phase = rng.uniform(0, 2*np.pi)
                a = a + 0.01 * np.sin(2*np.pi*f*x/n + phase)
            
            a = np.maximum(a, 0)
            sa = float(np.sum(a))
            if sa > 1e-10:
                a = a / sa
            else:
                a = best_a.copy()
        
        if stall_count > 30:
            a = best_a + 0.05 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            sa = float(np.sum(a))
            if sa > 1e-10:
                a = a / sa
            stall_count = 0
    
    return best_a


def resample(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    from scipy.interpolate import PchipInterpolator
    try:
        interp = PchipInterpolator(x_old, a)
        a_new = interp(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s
