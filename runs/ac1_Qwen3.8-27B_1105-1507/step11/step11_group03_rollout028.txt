def lp_solve_fast(a, time_limit=5.0):
    """Solve min M s.t. (a*g)_i <= M, g >= 0, sum(g) = 1."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    start = time.monotonic()
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        if time.monotonic() - start > time_limit:
            break
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None

            return 2.0 * n * val, grad
        
        # Use L-BFGS-B

                obj_and_grad, a, method='L-BFGS-B',

            )
            a_new = np.maximum(result.x, 0)
            sn = float(np.sum(a_new))
            if sn > 1e-10:
                a_new = a_new / sn
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                a = a_new
        except Exception:
            pass
        
        if verbose:
            print(f"      tau={tau:8.2f}: obj={best_obj:.10f}")
    
    return best_a

def coordinate_descent(a, time_budget, verbose=False, seed=42, max_sweeps=50):
    """
    Cyclic coordinate descent: for each coordinate, find the best value
    that minimizes the objective (1D minimization).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        # Randomize order for diversity
        order = rng.permutation(n)
        improved_this_sweep = False
        
        for idx in order:
            if time.monotonic() - start > time_budget:
                    break
            
            # Try perturbation of a[idx]
            base = a[idx].copy()
            candidates = [base * 0.5, base * 0.75, base * 1.25, base * 1.5,
                         base + 0.001, base - 0.001,
                         base + 0.005, base - 0.005,
                         0.0, base * 2.0]
            
            best_cand = base
            best_cand_obj = best_obj
            
            for cand in candidates:
                cand = max(0.0, cand)
                a[idx] = cand
                s_new = float(np.sum(a))
                if s_new < 1e-10:
                    a[idx] = base
                    continue
                a_norm = a / s_new
                obj = compute_obj(a_norm)
                if obj < best_cand_obj:
                    best_cand_obj = obj
                    best_cand = cand
            
            a[idx] = best_cand
            s_new = float(np.sum(a))
            if s_new > 1e-10:
                a = a / s_new
            
            obj_now = compute_obj(a)
            if obj_now < best_obj:
                best_obj = obj_now
                best_a = a.copy()
                improved_this_sweep = True
        
        if verbose and sweep % 5 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, improved={improved_this_sweep}")
    
    return best_a


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def periodic_extension(a, factor):
    """Create a longer sequence by periodic repetition (not resampling)."""
    n = len(a)
    new_n = n * factor
    a_new = np.tile(a, factor)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

Optimizes step function coefficients using L-BFGS-B quasi-Newton methods with
log-sum-exp smoothing, improved LP fixed-point with golden-section t-search,
dual-guided perturbation, and multi-n exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_obj_and_grad(a, tau):
    """
    Compute smoothed max of conv(a,a) and its gradient.
    Uses direct O(n^2) computation for the gradient (fine for n<=2000).
    """
    n = len(a)
    conv = compute_conv(a)

    shifted = np.clip(shifted, -700, 700)

    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
        smooth_val = max_val
    else:
        p = exp_vals / sum_exp
        smooth_val = max_val + tau * np.log(sum_exp)

        N <<= 1
    Af = fft(a, N)
    Pf = fft(p, N)

    # term1[j] = sum_k p[k]*a[k-j] (valid)
    # = sum_i a[i]*p[i+j] = cross-correlation
    # = (a * p_rev)[2n-2-j] where p_rev = p[::-1]
    p_rev = p[::-1]
    Pfr = fft(p_rev, N)
    conv_apr = ifft(Af * Pfr).real[:2 * n - 1]
    
    grad = conv_pa[:n] + conv_apr[n - 1:2 * n - 1][::-1]
    
    return 2.0 * n * smooth_val, 2.0 * n * grad


def lbfgs_optimize(a, time_budget, tau_levels=None, max_iter_per_tau=2000, verbose=False):
    """
    L-BFGS-B optimization with progressive tau reduction.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if tau_levels is None:
        tau_levels = [50, 20, 8, 3, 1, 0.5, 0.2, 0.1, 0.05]

    for tau in tau_levels:
        if time.monotonic() - start > time_budget:
            break

        def obj_and_grad(x):
            x = np.maximum(x, 0)
            sx = float(np.sum(x))
            if sx < 1e-10:
                return 1e10, np.zeros(n)
            x = x / sx
            val, grad = smooth_obj_and_grad(x, tau)
            return val, grad

        try:
            result = minimize(
                obj_and_grad, a.copy(), method='L-BFGS-B',
                options={'maxiter': max_iter_per_tau, 'gtol': 1e-10, 'ftol': 1e-12,
                         'maxls': 50}
            )
            a_new = np.maximum(result.x, 0)
            sn = float(np.sum(a_new))
            if sn > 1e-10:
                a_new = a_new / sn
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                a = a_new
        except Exception:
            pass

        if verbose:
            print(f"      tau={tau:8.2f}: obj={best_obj:.10f}")

    return best_a


def lp_solve_fast(a, time_limit=5.0):
    """Solve min M s.t. (a*g)_i <= M, g >= 0, sum(g) = 1. Returns (g, M, duals)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    start = time.monotonic()
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        if time.monotonic() - start > time_limit:
            break
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    g_val = g_val / sg
                    return g_val, float(M_var.value)
        except Exception:
            continue
    return None, None


def lp_fixed_point_v2(a, time_budget, verbose=False, seed=42, max_iters=400):
    """
    LP fixed-point with adaptive t-search and smart perturbation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall_count = 0
    t_current = 0.15
    consecutive_fails = 0

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, M_lp = lp_solve_fast(a, time_limit=min(remaining * 0.2, 8.0))
        if g_lp is None:
            consecutive_fails += 1
            if consecutive_fails > 8:
                break
            a = best_a + 0.01 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            continue
        consecutive_fails = 0

        # Adaptive t-search: try a set of candidates
        obj_at_1 = compute_obj(g_lp)
        if obj_at_1 >= best_obj:
            t_candidates = [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.15, 0.2, 0.3,
                           max(0.005, t_current * 0.5), max(0.005, t_current * 0.8),
                           t_current]
        else:
            t_candidates = [0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0,
                           min(1.0, t_current * 1.5), min(1.0, t_current * 2.0)]

        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_t = t_current

        for t in sorted(set(t_candidates)):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_t = t

        if best_local_obj < best_obj * (1.0 - 1e-13):
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a
            t_current = max(0.01, min(1.0, best_local_t * 0.8))
            stall_count = 0
            if verbose and it % 10 == 0:
                print(f"    LP it{it:3d}: obj={best_obj:.10f}, t={best_local_t:.3f}, M={M_lp:.6f}")
        else:
            stall_count += 1

        # Smart perturbation on stall
        if stall_count > 0 and stall_count % 4 == 0:
            strategy = (stall_count // 4) % 7
            a = best_a.copy()

            if strategy == 0:
                # Target top lags directly
                conv = compute_conv(a)
                max_val = float(np.max(conv))
                top_lags = np.argsort(conv)[-5:]
                for lag in top_lags:
                    grad = np.zeros(n)
                    for i in range(n):
                        j = lag - i
                        if 0 <= j < n:
                            grad[i] += 2.0 * a[j]
                    gn = np.linalg.norm(grad)
                    if gn > 1e-15:
                        for step in [0.0005, 0.002, 0.005, 0.01, 0.02]:
                            a_try = a - step * grad / gn
                            a_try = np.maximum(a_try, 0)
                            st = float(np.sum(a_try))
                            if st < 1e-10:
                                continue
                            a_try = a_try / st
                            obj_try = compute_obj(a_try)
                            if obj_try < best_obj:
                                best_obj = obj_try
                                best_a = a_try.copy()
                                break
                        a = best_a.copy()
                        break
            elif strategy == 1:
                # Low-freq Fourier
                x = np.arange(n)
                for f in rng.integers(1, max(2, n // 15), size=2):
                    a += 0.015 * np.sin(2 * np.pi * f * x / n + rng.uniform(0, 2 * np.pi))
            elif strategy == 2:
                # Random
                a += 0.01 * rng.standard_normal(n)
            elif strategy == 3:
                # Smooth
                k = max(3, n // 50)
                kernel = np.ones(k) / k
                a = 0.85 * a + 0.15 * np.convolve(a, kernel, mode='same')
            elif strategy == 4:
                # Boost minima
                median_val = np.median(a)
                a = a + 0.005 * np.maximum(median_val - a, 0)
            elif strategy == 5:
                # Anti-symmetric
                half = n // 2
                pert = rng.uniform(-1, 1, half) * 0.015
                if n % 2 == 0:
                    a[:half] += pert
                    a[half:] -= pert[::-1]
                else:
                    a[:half] += pert
                    a[half+1:] -= pert[::-1]
            elif strategy == 6:
                # Block perturbation
                block = max(5, n // 20)
                for _ in range(3):
                    start_idx = rng.integers(0, n - block)
                    a[start_idx:start_idx + block] += rng.standard_normal(block) * 0.01

            a = np.maximum(a, 0)
            sa = float(np.sum(a))
            if sa > 1e-10:
                a = a / sa
            else:
                a = best_a.copy()

        if stall_count > 40:
            a = best_a + min(0.08, 0.02 * (stall_count // 10)) * rng.standard_normal(n)
            a = np.maximum(a, 0)
            sa = float(np.sum(a))
            if sa > 1e-10:
                a = a / sa
            stall_count = 0
            t_current = 0.05

    return best_a


def resample(a, new_n):
    """Resample via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def subgradient_refine(a, time_budget, verbose=False, seed=42, max_iters=200):
    """Targeted subgradient descent on the active set."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        threshold = max_val * 0.9995
        active_lags = np.where(conv >= threshold)[0]

        if len(active_lags) == 0:
            active_lags = np.array([np.argmax(conv)])

        # Compute subgradient (average over active set)
        subgrad = np.zeros(n)
        for lag in active_lags:
            for i in range(n):
                j = lag - i
                if 0 <= j < n:
                    subgrad[i] += 2.0 * a[j]
        subgrad /= len(active_lags)
        gn = np.linalg.norm(subgrad)
        if gn < 1e-15:
            break
        subgrad /= gn

        # Backtracking line search
        improved = False
        step = 0.001
        for _ in range(10):
            a_try = a - step * subgrad
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                step *= 0.5
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                step = min(0.05, step * 1.5)
                break
            else:
                step *= 0.5

        if not improved:
            # Perturb
            a = best_a + 0.005 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and it % 20 == 0:
            print(f"    SubGrad it{it:3d}: obj={best_obj:.10f}, |active|={len(active_lags)}")

    return best_a
