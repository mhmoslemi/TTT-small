import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2n * max(conv) assuming sum(a) = 1."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def build_cross_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # (conv(a,g))[k] = sum_j a[j] * g[k-j]
        # For fixed j, the contribution to A[k,j] is a[j] when k-j is in [0,n-1]
        # i.e., k in [j, j+n-1]
        k_start = j
        k_end = min(2 * n - 1, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_fw_direction(a, time_budget=10.0):
    """
    Solve FW direction: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns g or None.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_cross_matrix(a_norm)

def exact_line_search_fw(a, g):

    s = g - a  # direction
    
    # B_k = conv(a,a)[k], C_k = conv(a,s)[k], D_k = conv(s,s)[k]
    # conv(a+s, a+s)[k] = conv(a,a)[k] + 2*conv(a,s)[k] + conv(s,s)[k]
    # But we need conv(a+t*s, a+t*s) = conv(a,a) + 2t*conv(a,s) + t^2*conv(s,s)
    # where conv(a,s) is the cross-correlation sum_j a[j]*s[k-j]
    
    # Compute B = conv(a,a)
    # Compute C = cross_corr(a, s) = sum_j a[j]*s[k-j]
    # Compute D = conv(s,s)
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    fs = rfft(s, size)
    f_s = rfft(s, size)

    C = irfft(fa * fs, size)[:2*n-1]  # cross_corr(a,s) -- this gives sum_j a[j]*s[k-j]
    D = irfft(fs * fs, size)[:2*n-1]  # conv(s,s)
    
    # q_k(t) = B[k] + 2*t*C[k] + t^2*D[k]
    # h(t) = max_k q_k(t)
    # Find min of h(t) on [0,1]
    
    candidates = [0.0, 1.0]
    
    # For each k, the minimum of q_k is at t_k = -C[k]/(2*D[k]) if D[k] > 0
    # Check if this is in [0,1] and if q_k is the max there
    for k in range(2*n-1):
        if D[k] > 1e-15:
            tk = -C[k] / (2.0 * D[k])
            if 0.0 <= tk <= 1.0:
                candidates.append(tk)
    
    # For pairs (k,l) where q_k(t) = q_l(t):
    # (B_k - B_l) + 2t(C_k - C_l) + t^2(D_k - D_l) = 0
    # Only check top active constraints for efficiency
    active = np.argsort(-B)[:20]  # top 20 by B value
    for i_idx in range(len(active)):
        for j_idx in range(i_idx + 1, len(active)):
            k, l = active[i_idx], active[j_idx]
            a2 = D[k] - D[l]
            a1 = 2.0 * (C[k] - C[l])
            a0 = B[k] - B[l]
            if abs(a2) < 1e-15:
                if abs(a1) > 1e-15:
                    tk = -a0 / a1
                    if 0.0 <= tk <= 1.0:
                        candidates.append(tk)
            else:
                disc = a1*a1 - 4.0*a2*a0
                if disc >= 0:
                    sqrt_disc = np.sqrt(disc)
                    for tk in [(-a1 - sqrt_disc)/(2.0*a2), (-a1 + sqrt_disc)/(2.0*a2)]:
                        if 0.0 <= tk <= 1.0:
                            candidates.append(tk)
    
    # Also check t where h(t) has a kink from any pair
    # (already covered above)
    
    # Evaluate h(t) at all candidates
    best_t = 0.0
    best_val = float('inf')
    for t in candidates:
        t = max(0.0, min(1.0, t))
        vals = B + 2.0 * t * C + t * t * D
        h = float(np.max(vals))
        if h < best_val:
            best_val = h
            best_t = t
    
    # Also do a fine grid search around the best candidate
    t_coarse = best_t
    for dt in [0.01, 0.001, 0.0001]:
        t_lo = max(0.0, t_coarse - 5*dt)

            h = float(np.max(vals))

                best_t = t
        t_coarse = best_t

"""
Optimizes step function coefficients to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses a hybrid of: (1) fast vectorized coordinate descent, (2) Frank-Wolfe with
LP direction finding and exact line search, (3) multiplicative subgradient updates,
(4) multi-n exploration. Key improvements over previous: exact FW line search,

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_cross(a, b):
    """Cross-correlation: sum_j a[j]*b[k-j] via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj_norm(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_cross_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = cross_corr(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def fw_exact_line_search(a, g):
    """

    Uses quadratic structure for efficiency.
    """
    n = len(a)
    s = g - a

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    fa = rfft(a, size)
    fs = rfft(s, size)

            val += a_norm[k_peak - i]
        if 0 <= i + k_peak <= n - 1 and (i + k_peak) != k_peak:
            # This is the other term: a_norm[i] appears when k-j=i, j=k-i
            # Actually let me redo: conv[k] = sum_j a[j]*a[k-j]
            # d(conv[k])/d(a[i]) = a[k-i] (if 0<=k-i<=n-1) + a[i-k] (if 0<=i-k<=n-1 and i-k != k-i)
            # The two terms are: a[k-i] and a[i-k], but they're the same when k-i = i-k, i.e., i = k/2
            pass
        # Let me just compute it properly:
    # conv[k] = sum_j a[j] * a[k-j]
    # d/d(a[i]) conv[k] = a[k-i] * (1 if 0<=k-i<=n-1) + a[i-k] * (1 if 0<=i-k<=n-1)
    # Wait, that's not right either. Let me be more careful.
    # conv[k] = sum_{j: 0<=j<=n-1, 0<=k-j<=n-1} a[j]*a[k-j]
    # The terms involving a[i] are:
    #   j=i (if 0<=k-i<=n-1): a[i]*a[k-i]
    #   k-j=i => j=k-i (if 0<=k-i<=n-1): a[k-i]*a[i]
    # When i = k-i (i.e., 2i=k), these are the same term, so the derivative is 2*a[i] for that case... no.
    # Actually: d/d(a[i]) [a[i]*a[k-i]] = a[k-i] (treat a[k-i] as constant when differentiating w.r.t. a[i])
    # And d/d(a[i]) [a[k-i]*a[i]] = a[k-i]
    # When 2i=k: the term is a[i]*a[i] = a[i]^2, derivative is 2*a[i].
    # But in the sum, when 2i=k, the j=i term and the j=k-i=i term are the same, so we count it once.
    # So: d/d(a[i]) conv[k] = a[k-i] if 0<=k-i<=n-1 (this covers both cases, and when k-i=i, it gives a[i])
    # Wait no. Let me just enumerate:
    # If k-i is in [0, n-1]: the term a[i]*a[k-i] appears in the sum (with j=i)
    #   Its derivative w.r.t. a[i] is a[k-i]
    # If k-i is in [0, n-1] AND k-i != i: the term a[k-i]*a[i] also appears (with j=k-i)
    #   Its derivative w.r.t. a[i] is a[k-i]
    # If k-i == i (i.e., 2i=k): only ONE term a[i]^2 appears, derivative is 2*a[i]
    #   But this is the same as a[k-i] + a[k-i] when k-i=i, so it's 2*a[i]. Consistent.

    grad = np.zeros(n)
    for i in range(n):
        if 0 <= k_peak - i <= n - 1:
            grad[i] += a_norm[k_peak - i]
        if 0 <= i - k_peak + 2 * k_peak <= n - 1:  # This is wrong, let me just do it directly
            pass

    # Simpler: just use FFT
    # The gradient of conv(a,a) w.r.t. a is related to the cross-correlation of a with a
    # Actually d/d(a[i]) conv(a,a)[k] = a[k-i] + a[i-k] (with zero-padding)
    # which is 2*a[k-i] when |k-i| < n and k != i... 
    # Let me just compute it via the matrix:
    # grad[i] = sum over the two contributions
    # The cleanest way: cross_corr(a, a) gives conv(a,a), and the "Jacobian" is:
    # d(conv[a,a])[k]/d(a[i]) = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n and i-k != k-i)
    # For the peak k_peak:
    for i in range(n):
        g = 0.0
        ki = k_peak - i
        if 0 <= ki < n:
            g += a_norm[ki]
        ik = i - k_peak
        if 0 <= ik < n and ik != ki:
            g += a_norm[ik]
        # Wait this double-counts when ik==ki (i.e., 2i=k_peak)
        # When 2i = k_peak: ki = k_peak - i = i = ik, so we'd add a_norm[i] twice
        # But the correct derivative of a[i]^2 is 2*a[i], so adding twice is correct!
        # Actually no: when ki == ik, the single term a[i]^2 has derivative 2*a[i],
        # and my formula gives a_norm[ki] + a_norm[ik] = a_norm[i] + a_norm[i] = 2*a_norm[i]. Correct!
        # When ki != ik: two separate terms a[i]*a[ki] and a[ik]*a[i], each with derivative a[ki] and a[ik].
        # Total: a[ki] + a[ik]. Correct!
        grad[i] = g

    # Project onto tangent of simplex: subtract mean
    grad_centered = grad - np.mean(grad)
    # Multiplicative update direction
    eta = 1.0 / (np.max(np.abs(grad_centered)) + 1e-10)
    direction = a_norm * np.exp(-eta * grad_centered)
    direction = np.maximum(direction, 0)
    ds = float(np.sum(direction))
    if ds > 1e-10:
        direction = direction / ds
    return direction, k_peak


def fast_cd(a, time_budget=200.0, seed=42, max_sweeps=None):
    """
    Fast coordinate descent for minimizing 2n*max(conv)/sum^2.
    Works with unnormalized a, tracks best normalized objective.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        if max_sweeps is not None and sweep >= max_sweeps:
            break
        sweep += 1
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]
            S_other = 1.0 - v  # since sum(a) = 1

            if S_other < 1e-10:
                continue

            # Affected convolution indices: k in [i, i+n-1] intersect [0, 2n-2]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)

            # Linear part: for k != 2i, conv[k] += 2*(x-v)*a[k-i]
            ks = np.arange(k_lo, k_hi + 1)
            mask = ks != 2 * i
            ks_lin = ks[mask]
            if len(ks_lin) > 0:
                a_ki = a[ks_lin - i]
                c_lin_base = conv[ks_lin]
            else:
                a_ki = np.array([])
                c_lin_base = np.array([])

            c_2i = float(conv[2 * i]) if 0 <= 2 * i <= 2 * n - 2 else 0.0

            # 1D minimization: min_x max(max(c_lin + 2*(x-v)*a_ki), c_2i + x^2 - v^2)
            # (we ignore the denominator since sum=1 means S_other+x = 1)
            # Wait, if we change a[i] from v to x, the new sum is 1-x+v = 1. Good!
            # So we just minimize max(max_lin(x), quad(x)) directly.

            # Use a smart bracket
            # The minimum is near where the linear and quadratic parts balance
            # Linear part at x=v: max(c_lin_base)
            # Quadratic at x=v: c_2i
            # If c_2i > max(c_lin_base): the quadratic is the binding constraint
            #   min of quad: x = 0 (since quad = c_2i + x^2 - v^2 is decreasing for x < v)
            #   Actually quad(x) = c_2i + x^2 - v^2, minimum at x=0 with value c_2i - v^2
            # If max(c_lin) > c_2i: linear is binding, we want to reduce it
            #   Linear is c_lin_base[k] + 2*(x-v)*a_ki[k]
            #   To reduce, we want x < v if a_ki > 0

            if len(ks_lin) > 0:
                lin_at_v = float(np.max(c_lin_base))
            else:
                lin_at_v = 0.0

            curr_peak = max(lin_at_v, c_2i)

            # Initial bracket: try a few candidate x values
            candidates = [0.0, v, min(1.0, v + 0.05 / max(n, 1))]
            if v > 0:
                candidates.extend([v * 0.3, v * 0.7, v * 1.3, v * 1.5])
            candidates.extend([1.0 / n, 2.0 / n, 0.5 / n])
            # Add gradient-informed point
            if curr_peak > 0 and v > 0:
                # If quad is binding: push x toward 0
                # If linear is binding: push x in direction that reduces max linear
                if c_2i >= lin_at_v:
                    candidates.append(max(0.0, v * 0.5))
                else:
                    if len(ks_lin) > 0 and len(a_ki) > 0:
                        # The most positive a_ki limits how much we can increase x
                        # The most negative... all a_ki >= 0, so increasing x increases all linear terms
                        # To decrease linear terms, decrease x
                        candidates.append(max(0.0, v * 0.5))

            candidates = [max(0.0, min(1.0, float(c))) for c in candidates]
            candidates = list(set(candidates))

            def f1d(x):
                if x < 0 or x > 1.0:
                    return 1e30
                if len(ks_lin) > 0:
                    max_lin = float(np.max(c_lin_base + 2.0 * (x - v) * a_ki))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                return max(max_lin, quad_val)

            best_x = v
            best_f = f1d(v)
            for xc in candidates:
                fc = f1d(xc)
                if fc < best_f:
                    best_f = fc
                    best_x = xc

            # Golden section refinement
            span = max(best_x * 0.8, 0.3 / n)
            lo = max(0.0, best_x - span)
            hi = min(1.0, best_x + span * 2.0)
            if hi - lo < 1e-15:
                hi = lo + 1e-10

            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = f1d(c)
            fd = f1d(d)

            for _ in range(35):
                if hi - lo < 1e-15:
                    break
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = f1d(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = f1d(d)

            x_opt = max(0.0, min(1.0, (lo + hi) / 2.0))
            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt

            # Update conv incrementally
            if 0 <= 2 * i <= 2 * n - 2:
                conv[2 * i] = c_2i + x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] = c_lin_base + 2.0 * delta * a_ki

        # Check improvement every few sweeps
        if sweep % 5 == 0:
            max_c = float(np.max(conv))
            if max_c < best_obj - 1e-10:
                best_obj = max_c
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 5
            # Re-normalize occasionally to prevent drift
            if sweep % 20 == 0:
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    conv = compute_conv(a)

    return best_a


def fw_iteration(a, use_lp=True, time_budget_lp=8.0):
    """One Frank-Wolfe iteration. Returns updated a."""
    n = len(a)
    s = float(np.sum(a))
    a_norm = a / s

    if use_lp:
        g = solve_lp_direction(a_norm, time_budget=time_budget_lp)
    else:
        g, _ = anti_peak_direction(a_norm)

    if g is None:
        return a_norm

    t = fw_exact_line_search(a_norm, g)
    a_new = (1.0 - t) * a_norm + t * g
    a_new = np.maximum(a_new, 0)
    sn = float(np.sum(a_new))
    if sn > 1e-10:
        a_new = a_new / sn
    return a_new


def multiplicative_step(a, eta=0.5, time_budget=5.0):
    """One multiplicative subgradient step."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv = compute_conv(a_norm)
    k_peak = int(np.argmax(conv))

    # Gradient
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        ki = k_peak - i
        if 0 <= ki < n:
            g += a_norm[ki]
        ik = i - k_peak
        if 0 <= ik < n and ik != ki:
            g += a_norm[ik]
        grad[i] = g

    # Center
    grad -= np.mean(grad)
    # Multiplicative update
    a_new = a_norm * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / float(np.sum(a_new))
    return a_new


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x * rng.uniform(2, 5))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(4 * np.pi * x + rng.uniform(0, 6.28)) + 0.2 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Two bumps
        c1, c2 = rng.uniform(0.2, 0.4), rng.uniform(0.6, 0.8)
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - c1) ** 2) / 0.01) + np.exp(-((x - c2) ** 2) / 0.01) + 0.1
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines CD, FW, multiplicative steps, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    # Load parent
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj_norm(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}", flush=True)
    except Exception:
        pass

    if best_seq is None:
        n_default = 800
        best_seq = np.ones(n_default) / n_default
        best_score = compute_obj_norm(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.8f}", flush=True)

    n_best = len(best_seq)

    # === Phase 1: Initial CD ===
    print(f"\n=== Phase 1: Initial CD (n={n_best}) ===", flush=True)
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 30:
        a_cd = fast_cd(best_seq, time_budget=p1_budget, seed=seed)
        obj_cd = compute_obj_norm(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **", flush=True)
        else:
            print(f"  CD: obj={obj_cd:.8f} (no improvement)", flush=True)

    # === Phase 2: FW Iterations ===
    print(f"\n=== Phase 2: Frank-Wolfe ===", flush=True)
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 200.0)
    if p2_budget > 20:
        a_fw = best_seq.copy()
        obj_fw = best_score
        n_fw_iters = 0
        while time.monotonic() < start_time + (deadline - start_time) - p2_budget and n_fw_iters < 20:
            if time.monotonic() + 15 > deadline:
                break
            used_lp = n_fw_iters < 8  # Use LP for first 8, then anti-peak
            a_fw = fw_iteration(a_fw, use_lp=used_lp,
                                time_budget_lp=min(8.0, max(2.0, deadline - time.monotonic() - 30)))
            obj_fw = compute_obj_norm(a_fw)
            n_fw_iters += 1
            improved = " ** " if obj_fw < best_score else ""
            print(f"  FW iter {n_fw_iters} ({'LP' if used_lp else 'AP'}): obj={obj_fw:.8f}{improved}", flush=True)
            if obj_fw < best_score:
                best_score = obj_fw
                best_seq = a_fw.copy()
        # CD polish after FW
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_polish = fast_cd(a_fw, time_budget=min(remaining * 0.3, 80.0), seed=seed + 100)
            obj_polish = compute_obj_norm(a_polish)
            if obj_polish < best_score:
                best_score = obj_polish
                best_seq = a_polish.copy()
                print(f"  Post-FW CD: obj={best_score:.8f} **", flush=True)

    # === Phase 3: Aggressive CD ===
    print(f"\n=== Phase 3: Aggressive CD ===", flush=True)
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.25, 350.0)
    if p3_budget > 30:
        # Multiple CD runs with different seeds
        for cd_run in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(p3_budget / (3 - cd_run), remaining * 0.6)
            if alloc < 20:
                break
            s = seed + 500 + cd_run * 100
            a_cd = fast_cd(best_seq, time_budget=alloc, seed=s)
            obj_cd = compute_obj_norm(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  CD run {cd_run}: obj={best_score:.8f} **", flush=True)
            else:
                print(f"  CD run {cd_run}: obj={obj_cd:.8f}", flush=True)

    # === Phase 4: Multiplicative + CD alternating ===
    print(f"\n=== Phase 4: Multiplicative + CD ===", flush=True)
    remaining = deadline - time.monotonic()
    if remaining > 40:
        for m_round in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            # Several multiplicative steps
            a_m = best_seq.copy()
            for step in range(50):
                if time.monotonic() > deadline - 20:
                    break
                eta = 0.3 * (1.0 - step / 50.0) + 0.05
                a_m = multiplicative_step(a_m, eta=eta)
            obj_m = compute_obj_norm(a_m)
            # Follow with short CD
            if time.monotonic() < deadline - 15:
                a_m = fast_cd(a_m, time_budget=min(30.0, remaining * 0.15), seed=seed + 2000 + m_round * 7)
                obj_m = compute_obj_norm(a_m)
            if obj_m < best_score:
                best_score = obj_m
                best_seq = a_m.copy()
                print(f"  M+CD round {m_round}: obj={best_score:.8f} **", flush=True)
            else:
                print(f"  M+CD round {m_round}: obj={obj_m:.8f}", flush=True)

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===", flush=True)
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 100),
        max(400, n_current - 50),
        n_current + 50,
        n_current + 100,
        n_current + 150,
        700, 750, 850, 900, 950, 1000, 1100
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500 and abs(n - n_current) >= 30]
    n_options = n_options[:5]

    print(f"  Trying n values: {n_options}", flush=True)
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # CD on new length
        alloc = min(40.0, remaining * 0.25)
        if alloc > 10:
            a_try = fast_cd(a_try, time_budget=alloc, seed=n_try)

        obj_try = compute_obj_norm(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **", flush=True)
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}", flush=True)

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===", flush=True)
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        mode = 1 + r
        a_r = diverse_init(n_final, rng, mode=mode)
        # Blend with current best
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / float(a_r.sum())

        alloc = min(40.0, remaining * 0.2)
        if alloc > 10:
            a_r = fast_cd(a_r, time_budget=alloc, seed=9000 + r * 111)
            obj_r = compute_obj_norm(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **", flush=True)
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}", flush=True)

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===", flush=True)
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 100.0)
        a_final = fast_cd(best_seq, time_budget=alloc, seed=seed + 99999)
        obj_final = compute_obj_norm(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **", flush=True)
        else:
            print(f"  Final CD: obj={obj_final:.8f}", flush=True)

    # One more FW at the very end if time
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = fw_iteration(best_seq, use_lp=True, time_budget_lp=8.0)
        obj_last = compute_obj_norm(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            # Quick CD
            if deadline - time.monotonic() > 10:
                a_last = fast_cd(a_last, time_budget=10.0, seed=seed + 111111)
                obj_last = compute_obj_norm(a_last)
                if obj_last < best_score:
                    best_score = obj_last
                    best_seq = a_last.copy()
            print(f"  Final FW: obj={best_score:.8f} **", flush=True)

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===", flush=True)
    return result
