import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Cross convolution a*b via FFT. Returns array of length len(a)+len(b)-1."""
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:na + nb - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho - 1] - 1) / rho
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = len(a)
    A = np.zeros((2 * n - 1, m))
    for j in range(m):
        for k in range(2 * n - 1):
            i = k - j
            if 0 <= i < n:
                A[k, j] = a[i]
    return A


def build_conv_matrix_sparse(a):
    """Build convolution matrix more efficiently using structure."""
    n = len(a)
    m = len(a)
    A = np.zeros((2 * n - 1, m))
    for j in range(m):
        # A[k, j] = a[k-j] if 0 <= k-j < n
        k_min = max(0, j)
        k_max = min(2 * n - 2, j + n - 1)
        idx = np.arange(k_min, k_max + 1)
        A[idx, j] = a[idx - j]
    return A


def solve_lp_minpeak(a, solver=None):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix_sparse(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.SCIPS, cp.SCS, cp.ECOS, cp.GLPK]
    if solver is not None:
        solvers = [solver] + solvers

    for sv in solvers:
        try:
            prob.solve(solver=sv)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def exact_line_search(b, c, d):
    """
    Exact 1D line search: minimize f(t) = max_k [b_k(1-t)^2 + 2c_k*t*(1-t) + d_k*t^2]
    over t in [0, 1].
    """
    def f(t):
        t = np.clip(t, 0.0, 1.0)
        vals = b * (1 - t) ** 2 + 2 * c * t * (1 - t) + d * t ** 2
        return float(np.max(vals))

    # Use scipy's bounded minimizer
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10})
    if res.success:
        return res.x, res.fun

    # Fallback: grid search
    ts = np.linspace(0, 1, 200)
    vals = np.array([f(t) for t in ts])
    idx = np.argmin(vals)
    # Refine
    lo, hi = ts[max(0, idx - 1)], ts[min(len(ts) - 1, idx + 1)]
    res2 = minimize_scalar(f, bounds=(lo, hi), method='bounded',
                           options={'xatol': 1e-12})
    if res2.success:
        return res2.x, res2.fun
    return ts[idx], vals[idx]


def soft_max_gd(a, time_budget=60.0, beta_start=10.0, beta_end=80.0,
                lr_start=0.01, momentum=0.9, verbose=False, seed=42):
    """
    Soft-max gradient descent with cosine annealing and adaptive LR.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    rng = np.random.default_rng(seed)
    it = 0
    stall = 0
    best_it = 0
    patience = 800

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        # Cosine annealing for beta
        beta = beta_start + (beta_end - beta_start) * (1 - np.cos(np.pi * progress)) / 2

        # Adaptive LR: start high, decay with progress, boost if stuck
        lr = lr_start * (1.0 - 0.5 * progress)
        if stall > 200:
            lr *= 0.5

        # Compute convolution
        conv = compute_conv(a)
        c_max = float(np.max(conv))

        # Numerical stability
        scale = beta * (conv - c_max)
        scale -= np.max(scale)
        exp_vals = np.exp(scale)
        p_sum = float(np.sum(exp_vals))
        if p_sum < 1e-300:
            p = np.zeros(len(conv))
            p[int(np.argmax(conv))] = 1.0
        else:
            p = exp_vals / p_sum

        # Gradient via FFT cross-correlation
        # grad_i = 2 * sum_j a_j * p_{i+j}
        # This is the cross-correlation of a with p, evaluated at specific lags
        a_rev = a[::-1]
        p_len = len(p)
        size = 1
        while size < n + p_len - 1:
            size <<= 1
        A_fft = rfft(a_rev, size)
        P_fft = rfft(p, size)
        full = irfft(A_fft * P_fft, size)
        q = full[n - 1:n - 1 + n]
        grad = 2.0 * q

        # Momentum update
        v = momentum * v + grad
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall > patience:
                v = np.zeros(n)
                pert = rng.normal(0, 0.003, n)
                a = project_simplex(best_a + pert)
                stall = 0
                it = best_it
                lr = lr_start * 1.5

        a = a_new
        it += 1

        if verbose and it % 200 == 0:
            print(f"    SoftMaxGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, "
                  f"lr={lr:.5f}, t={elapsed:.1f}s")

    return best_a, best_obj


def multiplicative_updates(a, time_budget=60.0, alpha=0.5, verbose=False, seed=42):
    """
    Log-domain multiplicative updates for minimizing max-convolution.
    Update: a_i <- a_i * exp(-alpha * grad_i) where grad comes from soft-max.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        beta = 10.0 + 50.0 * progress

        conv = compute_conv(a)
        c_max = float(np.max(conv))
        scale = beta * (conv - c_max)
        scale -= np.max(scale)
        exp_vals = np.exp(scale)
        p_sum = float(np.sum(exp_vals))
        if p_sum < 1e-300:
            p = np.zeros(len(conv))
            p[int(np.argmax(conv))] = 1.0
        else:
            p = exp_vals / p_sum

        # Gradient
        a_rev = a[::-1]
        p_len = len(p)
        size = 1
        while size < n + p_len - 1:
            size <<= 1
        A_fft = rfft(a_rev, size)
        P_fft = rfft(p, size)
        full = irfft(A_fft * P_fft, size)
        q = full[n - 1:n - 1 + n]
        grad = 2.0 * q

        # Normalized gradient for multiplicative update
        grad_mean = float(np.mean(grad))
        grad_std = float(np.std(grad)) + 1e-10
        grad_norm = (grad - grad_mean) / grad_std

        # Multiplicative update in log domain
        step = alpha * 0.1 * (1 - 0.5 * progress)
        log_a = np.log(a + 1e-300) - step * grad_norm
        a_new = np.exp(log_a)
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                pert = rng.normal(0, 0.005, n)
                a = project_simplex(best_a + pert)
                stall = 0

        a = a_new
        it += 1

        if verbose and it % 200 == 0:
            print(f"    MulUpd it{it}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a, best_obj


def lp_polish(a, time_budget=120.0, max_iters=50, verbose=False, seed=777):
    """
    LP-based polishing with exact 1D line search.
    At each iteration:
    1. Solve LP to find g minimizing max_k conv(a, g)_k
    2. Find optimal blend t via exact 1D search
    3. Update a
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(seed)
    prev_t = 0.5

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Solve LP
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Compute convolutions for line search
        b = compute_conv(a)       # conv(a, a)
        c = compute_conv_cross(a, g_lp)  # conv(a, g)
        d = compute_conv(g_lp)    # conv(g, g)

        # Ensure same length (pad if needed)
        max_len = max(len(b), len(c), len(d))
        if len(b) < max_len:
            b = np.pad(b, (0, max_len - len(b)))
        if len(c) < max_len:
            c = np.pad(c, (0, max_len - len(c)))
        if len(d) < max_len:
            d = np.pad(d, (0, max_len - len(d)))

        # Exact 1D line search
        t_opt, obj_opt = exact_line_search(b, c, d)

        if t_opt > 1e-6 and obj_opt < best_obj - 1e-10:
            a_new = (1 - t_opt) * a + t_opt * g_lp
            a_new = project_simplex(np.maximum(a_new, 0))
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                prev_t = t_opt
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 3 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                  f"M_lp={M_lp:.6f}, elapsed={time.monotonic()-start:.1f}s")

        if stall >= 5:
            if stall <= 8:
                noise = rng.normal(0, 0.004 * (stall - 4), n)
                a = project_simplex(best_a + noise)
            else:
                # More aggressive restart
                x = np.linspace(0, 1, n)
                a_div = 1.0 / (0.03 + 5.0 * (2 * x - 1) ** 2)
                a = project_simplex(0.4 * a_div + 0.6 * best_a)
            stall = 0

    return best_a, best_obj


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.03 + sharpness * (2 * x - 1) ** 2)
    return project_simplex(a)


def arcsine_init(n):
    """Arcsine distribution initialization."""
    x = np.linspace(0.005, 0.995, n)
    a = 1.0 / np.sqrt(np.maximum(1e-10, 1 - (2 * x - 1) ** 2))
    return project_simplex(a)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.03
    elif mode == 1:
        a = (np.exp(-((idx - n // 4) ** 2) / (2 * (n // 8) ** 2)) +
             np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 8) ** 2)) + 0.01)
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.03
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(9) / 9
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(6 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 4
    elif mode == 6:
        a = 1.0 / np.cosh(4.0 * (2 * x - 1))
    elif mode == 7:
        # Very sharp edges
        a = np.zeros(n)
        a[0] = 1.0
        a[-1] = 1.0
        a[1:min(5, n)] = 0.4
        a[max(0, n - 5):n - 1] = 0.4
        if n > 6:
            a[1:-1] += 0.03
    elif mode == 8:
        # W-shaped
        a = np.exp(-((2 * x - 1) ** 2 - 0.3) ** 2 / 0.1) + 0.05
    elif mode == 9:
        # Power law at edges
        a = np.abs(2 * x - 1) ** (-0.3) + 0.02
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def try_length(a_best, n_new, time_budget=60.0, verbose=False):
    """Interpolate to new length and optimize."""
    n_old = len(a_best)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_try = np.interp(x_new, x_old, a_best)
    a_try = project_simplex(np.maximum(a_try, 0))

    # Quick soft-max GD
    alloc1 = min(time_budget * 0.5, 40.0)
    if alloc1 > 10:
        a_try, obj_try = soft_max_gd(
            a_try, time_budget=alloc1,
            beta_start=10.0, beta_end=50.0,
            lr_start=0.008, momentum=0.9,
            verbose=False, seed=12345
        )

    # LP polish
    remaining = time_budget - (time.monotonic() - (time.monotonic() - time_budget))
    alloc2 = min(time_budget * 0.4, 30.0)
    if alloc2 > 5:
        a_try, obj_try = lp_polish(a_try, time_budget=alloc2, max_iters=15, verbose=False)

    return a_try, obj_try


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining multiple optimization strategies.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # --- Load parent ---
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    # --- Initialize if no parent ---
    if best_seq is None:
        best_seq = u_shaped_init(n_current, sharpness=3.0)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")

    # Try arcsine
    a_arc = arcsine_init(n_current)
    obj_arc = compute_obj(a_arc)
    if obj_arc < best_score:
        best_score = obj_arc
        best_seq = a_arc.copy()
        print(f"Arcsine init: obj={best_score:.8f}")

    # --- Phase 1: Soft-max GD (smooth, low beta) ---
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 1: SoftMax GD (beta 8->30), budget={remaining:.0f}s ===")
        alloc = min(remaining * 0.20, 240.0)
        a1, obj1 = soft_max_gd(
            best_seq, time_budget=alloc,
            beta_start=8.0, beta_end=30.0,
            lr_start=0.012, momentum=0.91,
            verbose=True, seed=seed
        )
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1 SoftMaxGD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 SoftMaxGD: obj={obj1:.8f} (no improvement)")

    # --- Phase 2: Multiplicative updates ---
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 2: Multiplicative Updates, budget={remaining:.0f}s ===")
        alloc = min(remaining * 0.12, 120.0)
        a2, obj2 = multiplicative_updates(
            best_seq, time_budget=alloc,
            alpha=0.8, verbose=True, seed=seed + 1
        )
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2 MulUpd: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 MulUpd: obj={obj2:.8f} (no improvement)")

    # --- Phase 3: Soft-max GD (sharper) ---
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 3: SoftMax GD (beta 30->100), budget={remaining:.0f}s ===")
        alloc = min(remaining * 0.18, 200.0)
        a3, obj3 = soft_max_gd(
            best_seq, time_budget=alloc,
            beta_start=30.0, beta_end=100.0,
            lr_start=0.006, momentum=0.93,
            verbose=True, seed=seed + 2
        )
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3 SoftMaxGD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 SoftMaxGD: obj={obj3:.8f} (no improvement)")

    # --- Phase 4: LP Polish (main refinement) ---
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: LP Polish, budget={remaining:.0f}s ===")
        alloc = min(remaining * 0.22, 250.0)
        a4, obj4 = lp_polish(best_seq, time_budget=alloc, max_iters=40, verbose=True)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 LP: obj={obj4:.8f} (no improvement)")

    # --- Phase 5: Diverse restarts ---
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 5: Diverse Restarts, budget={remaining:.0f}s ===")
        n_cur = len(best_seq)

        for r in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            mode = r % 10
            a_r = diverse_init(n_cur, rng, mode=mode)
            blend = 0.5 + 0.1 * r
            a_r = project_simplex((1 - blend) * a_r + blend * best_seq)

            # Quick optimization
            alloc_sgd = min(remaining * 0.22, 60.0)
            if alloc_sgd > 15:
                a_r, obj_r = soft_max_gd(
                    a_r, time_budget=alloc_sgd,
                    beta_start=15.0, beta_end=60.0,
                    lr_start=0.008, momentum=0.9,
                    verbose=False, seed=seed + 100 + r
                )

            remaining = deadline - time.monotonic()
            if remaining > 25:
                alloc_lp = min(remaining * 0.22, 40.0)
                if alloc_lp > 5:
                    a_r, obj_r = lp_polish(a_r, time_budget=alloc_lp, max_iters=15, verbose=False)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # --- Phase 6: Length exploration ---
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 6: Length Exploration, budget={remaining:.0f}s ===")
        n_best = len(best_seq)

        n_options = sorted(set([
            max(400, n_best - 100),
            n_best + 100, n_best + 200,
            600, 700, 900, 1000, 1200, 1400, 1600
        ]))
        n_options = [n for n in n_options if 300 <= n <= 2000 and abs(n - n_best) >= 80]
        n_options = n_options[:3]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            alloc = min(remaining * 0.4, 80.0)
            a_try, obj_try = try_length(best_seq, n_try, time_budget=alloc, verbose=True)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # --- Phase 7: Final intensive LP polish ---
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 7: Final LP Polish, budget={remaining:.0f}s ===")
        alloc1 = min(remaining * 0.5, 100.0)
        if alloc1 > 20:
            a_f, obj_f = lp_polish(best_seq, time_budget=alloc1, max_iters=30, verbose=True)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        # One more soft-max round
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc2 = min(remaining * 0.5, 60.0)
            a_f2, obj_f2 = soft_max_gd(
                best_seq, time_budget=alloc2,
                beta_start=50.0, beta_end=150.0,
                lr_start=0.004, momentum=0.94,
                verbose=True, seed=seed + 500
            )
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final SoftMax: obj={best_score:.8f} **")

    # --- Phase 8: Last-ditch ---
    remaining = deadline - time.monotonic()
    if remaining > 20:
        for attempt in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            noise = rng.normal(0, 0.002, len(best_seq))
            a_p = project_simplex(best_seq + noise)
            alloc = min(remaining * 0.4, 15.0)
            if alloc > 5:
                a_p, obj_p = lp_polish(a_p, time_budget=alloc, max_iters=8, verbose=False)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Last-ditch {attempt}: obj={best_score:.8f} **")

    # --- Final validation ---
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
