```python
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

    s = float(np.sum(a))

    a_norm = a / s

def project_simplex(v):
    # Project onto probability simplex
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    valid = u * rho > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    idx = np.nonzero(valid)[0][-1]
    theta = (cssv[idx] - 1) / (idx + 1.0)
    return np.maximum(v - theta, 0)

def gerchberg_saxton(a, num_iters=200, alpha_start=0.1, alpha_end=0.5, time_budget=60.0):
    """GS iterations to flatten the spectrum."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s  # normalize to sum 1
    
    size = 1
    while size < 2*n - 1:

        if time.monotonic() - start > time_budget:
            break

        # Flatten magnitude
        mag = np.abs(A)
        mag_sq = mag ** 2

        # Compress: new_mag_sq = (1-alpha)*mag_sq + alpha*mean_mag_sq
```

```python
"""
Hybrid spectral-temporal optimization for step function minimization.
Combines Gerchberg-Saxton spectral flattening, soft-max gradient descent,
coordinate descent, and LP polishing for global + local optimization.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    valid = u * rho > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    idx = np.nonzero(valid)[0][-1]
    theta = (cssv[idx] - 1) / (idx + 1.0)
    return np.maximum(v - theta, 0)


def gerchberg_saxton(a, time_budget=120.0, alpha_start=0.05, alpha_end=0.6,
                     verbose=False, seed=42):
    """
    Gerchberg-Saxton iterations to flatten the power spectrum.
    Alternates between:
    - Frequency domain: compress |A|^2 towards uniform
    - Time domain: enforce non-negativity
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    a = a / s
    s = float(np.sum(a))
    a = a / s
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        
        # FFT
        A = fft(a, size)
        mag = np.abs(A)
        mag_sq = mag ** 2
        mean_mag_sq = float(np.mean(mag_sq))
        
        # Compress dynamic range
        # new_mag = mag^(1-alpha) * sqrt(mean_mag_sq)^alpha
        # This reduces the peak-to-average ratio
        new_mag = mag ** (1.0 - alpha) * (np.sqrt(mean_mag_sq)) ** alpha
        
        # Reconstruct with original phase
        phase = A / (mag + 1e-30)
        A_new = new_mag * phase
        a_new = np.real(ifft(A_new, size))[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            # Failed, try additive method
            a_new = a + 0.1 * rng.normal(0, 1, n)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            a_new = a_new / s_new
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                # Perturb and restart
                noise = rng.normal(0, 0.01, n)
                a = project_simplex(best_a + noise)
                stall = 0
                if verbose:
                    print(f"    GS restart at it={it}")
        
        a = a_new
        it += 1
        
        if verbose and it % 100 == 0:
            print(f"    GS it{it}: obj={best_obj:.8f}, alpha={alpha:.3f}, t={time.monotonic()-start:.1f}s")
    
    return best_a, best_obj


def soft_max_gd(a, time_budget=200.0, beta_start=10.0, beta_end=80.0,
                lr_start=0.008, momentum=0.9, verbose=False, seed=42):
    """Soft-max gradient descent with Nesterov momentum."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    it = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress
        lr = lr_start * (1.0 - 0.4 * progress)
        
        # Compute convolution
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient via FFT cross-correlation
        a_rev = a[::-1]
        size = 1
        while size < n + len(p) - 1:
            size <<= 1
        A = rfft(a_rev, size)
        P = rfft(p, size)
        full = irfft(A * P, size)
        q = full[n - 1:n - 1 + n]
        grad = 2.0 * q
        
        # Momentum update
        v = momentum * v + grad
        a_new = project_simplex(a - lr * v)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall > 100:
                v = np.zeros(n)
                stall = 0
                pert = rng.normal(0, 0.003, n)
                a = project_simplex(best_a + pert)
        
        a = a_new
        it += 1
        
        if verbose and it % 500 == 0:
            print(f"    SMGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.5f}")
        
        if it - best_it > 5000 and it > 1000:
            a = best_a.copy()
            v = np.zeros(n)
            lr *= 0.7
    
    return best_a, best_obj
```

```python
"""
Hybrid spectral-temporal optimization for step function minimization.
Combines Gerchberg-Saxton spectral flattening, soft-max gradient descent,
coordinate descent with exact 1D subproblem solving, and LP polishing.
Key innovation: coordinate descent exploits the piecewise-linear structure
of the max-convolution objective for exact 1D minimization, combined with
spectral domain operations for basin escaping.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return np.array([])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    valid = u * rho > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    idx = np.nonzero(valid)[0][-1]
    theta = (cssv[idx] - 1) / (idx + 1.0)
    return np.maximum(v - theta, 0)


def gs_iteration(a, size, alpha):
    """Single Gerchberg-Saxton iteration."""
    n = len(a)
    A = fft(a, size)
    mag = np.abs(A)
    mag_sq = mag ** 2
    mean_mag_sq = float(np.mean(mag_sq))
    if mean_mag_sq < 1e-30:
        return a
    new_mag = mag ** (1.0 - alpha) * (np.sqrt(mean_mag_sq)) ** alpha
    phase = A / (mag + 1e-30)
    A_new = new_mag * phase
    a_new = np.real(ifft(A_new, size))[:n]
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        return a
    return a_new / s_new

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 6:
        a = 1.0 / np.cosh(3.0 * (2 * x - 1))
    elif mode == 7:
        a = np.zeros(n)
        a[0] = 1.0
        a[-1] = 1.0
        if n > 4:
            a[1:-1] = 0.05
    elif mode == 8:
        # W-shaped
        a = np.abs(np.sin(2 * np.pi * x)) + 0.05
    elif mode == 9:
        # Power law
        a = (x + 0.01) ** (-0.3) + (1 - x + 0.01) ** (-0.3)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    return project_simplex(a)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining multiple optimization strategies:
    1. Multi-initialization + GS for basin finding
    2. Soft-max GD for smooth refinement
    3. Coordinate descent for exact local optimization
    4. LP polishing for structured improvements
    5. Length exploration
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = u_shaped_init(n_current, sharpness=3.0)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")

    # Quick GS to try to improve the starting point
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.1, 60.0)
        a_gs, obj_gs = gerchberg_saxton(best_seq, time_budget=alloc,
                                        alpha_start=0.05, alpha_end=0.4,
                                        verbose=False, seed=seed)
        if obj_gs < best_score:
            best_score = obj_gs
            best_seq = a_gs.copy()
            print(f"After quick GS: obj={best_score:.8f}")

    # === Phase 1: Multi-restart with diverse initializations + GS ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 1: Diverse init + GS, time={remaining:.0f}s ===")
        n_cur = len(best_seq)
        num_restarts = 4
        alloc_each = min(remaining / (num_restarts + 2), 40.0)

        for r in range(num_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            mode = r % 10
            a_r = diverse_init(n_cur, rng, mode=mode)
            # Blend with current best for stability
            a_r = project_simplex(0.4 * a_r + 0.6 * best_seq)

            a_r, obj_r = gerchberg_saxton(a_r, time_budget=alloc_each,
                                          alpha_start=0.1, alpha_end=0.5,
                                          verbose=False, seed=seed + r)
            # Quick GD
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_r, obj_r = soft_max_gd(a_r, time_budget=min(alloc_each * 0.5, 30.0),
                                         beta_start=10.0, beta_end=30.0,
                                         lr_start=0.01, momentum=0.9,
                                         verbose=False, seed=seed + 10 + r)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 2: Soft-max GD (main refinement) ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 2: SoftMax GD, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.25, 250.0)
        a2, obj2 = soft_max_gd(best_seq, time_budget=alloc,
                               beta_start=10.0, beta_end=60.0,
                               lr_start=0.008, momentum=0.9,
                               verbose=True, seed=seed + 100)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2: obj={best_score:.8f} **")
        else:
            print(f"  Phase2: obj={obj2:.8f} (no improvement)")

    # === Phase 3: Coordinate descent (exact local optimization) ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 3: Coordinate Descent, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.25, 200.0)
        a3, obj3 = coordinate_descent(best_seq, time_budget=alloc,
                                      verbose=True, seed=seed + 200)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj3:.8f} (no improvement)")

    # === Phase 4: LP Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 4: LP Polish, time={remaining:.0f}s ===")
        alloc = min(remaining * 0.2, 150.0)
        a4, obj4 = lp_polish(best_seq, time_budget=alloc, max_iters=30, verbose=True)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4: obj={best_score:.8f} **")
        else:
            print(f"  Phase4: obj={obj4:.8f} (no improvement)")

    # === Phase 5: Second round of CD + LP (alternating) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 5: CD+LP alternating, time={remaining:.0f}s ===")
        # CD
        alloc_cd = min(remaining * 0.4, 100.0)
        if alloc_cd > 20:
            a5, obj5 = coordinate_descent(best_seq, time_budget=alloc_cd,
                                          verbose=False, seed=seed + 300)
            if obj5 < best_score:
                best_score = obj5
                best_seq = a5.copy()
                print(f"  CD round2: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        # LP
        if remaining > 30:
            alloc_lp = min(remaining * 0.4, 80.0)
            a5b, obj5b = lp_polish(best_seq, time_budget=alloc_lp, max_iters=20, verbose=False)
            if obj5b < best_score:
                best_score = obj5b
                best_seq = a5b.copy()
                print(f"  LP round2: obj={best_score:.8f} **")

        # Another CD
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a5c, obj5c = coordinate_descent(best_seq, time_budget=min(remaining * 0.5, 60.0),
                                            verbose=False, seed=seed + 400)
            if obj5c < best_score:
                best_score = obj5c
                best_seq = a5c.copy()
                print(f"  CD round3: obj={best_score:.8f} **")

    # === Phase 6: Length exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 6: Length Exploration, time={remaining:.0f}s ===")
        n_best = len(best_seq)

        n_options = sorted(set([
            max(400, n_best - 100), max(400, n_best + 100),
            n_best + 200, n_best - 200, 1000, 1200, 1500
        ]))
        n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_best) >= 80]
        n_options = n_options[:3]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break

            # Interpolate to new length
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = project_simplex(np.maximum(a_try, 0))

            # Quick optimization
            alloc_gd = min(remaining * 0.3, 40.0)
            if alloc_gd > 10:
                a_try, obj_try = soft_max_gd(a_try, time_budget=alloc_gd,
                                             beta_start=10.0, beta_end=40.0,
                                             lr_start=0.008, momentum=0.9,
                                             verbose=False, seed=seed + 500)
            remaining = deadline - time.monotonic()
            if remaining > 25:
                a_try, obj_try = coordinate_descent(a_try, time_budget=min(remaining * 0.4, 40.0),
                                                    verbose=False, seed=seed + 600)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 7: Final Polish, time={remaining:.0f}s ===")

        # GS with very high alpha (aggressive flattening)
        alloc = min(remaining * 0.25, 60.0)
        if alloc > 10:
            a_f, obj_f = gerchberg_saxton(best_seq, time_budget=alloc,
                                          alpha_start=0.3, alpha_end=0.7,
                                          verbose=False, seed=seed + 700)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final GS: obj={best_score:.8f} **")

        # CD
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_f2, obj_f2 = coordinate_descent(best_seq, time_budget=min(remaining * 0.4, 60.0),
                                              verbose=False, seed=seed + 800)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

        # LP
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_f3, obj_f3 = lp_polish(best_seq, time_budget=min(remaining * 0.5, 50.0),
                                     max_iters=15, verbose=False)
            if obj_f3 < best_score:
                best_score = obj_f3
                best_seq = a_f3.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        # One more CD
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_f4, obj_f4 = coordinate_descent(best_seq, time_budget=remaining - 5,
                                              verbose=False, seed=seed + 900)
            if obj_f4 < best_score:
                best_score = obj_f4
                best_seq = a_f4.copy()
                print(f"  Final CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
