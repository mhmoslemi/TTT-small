import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.ndimage import uniform_filter1d

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Assumes a normalized or not."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))

def gradient_from_p(a, p):
    """
    Compute gradient of sum_k p[k]*conv[k](a) with respect to a.
    Returns 2 * (a_reversed * p)[n-1:2n-1] which equals d/d a_i sum_k p[k] conv[k].
    """
    n = len(a)
    b = a[::-1]
    N_fft = 1
    while N_fft < 3 * n - 1:
        N_fft <<= 1
    c = irfft(rfft(b, N_fft) * rfft(p, N_fft), N_fft)
    return 2.0 * c[n - 1:2 * n - 1]


def lse_smoothed_gradient(a, conv, tau):
    """Compute gradient of the LSE(tau, conv) approximation to max(conv)."""
    n = len(a)
    max_conv = float(np.max(conv))
    weighted = tau * (conv - max_conv)
    # Numerical stability

        return gradient_from_topk(a, conv, k=1)
    if weighted.max() < -700:

        p = np.zeros(2 * n - 1)

def gradient_from_topk(a, conv, k=5):

def gradient_active_set(a, conv, threshold=0.999):
    """Gradient from all lags within threshold of the max."""
    n = len(a)
    max_c = float(np.max(conv))
    if max_c < 1e-300:
        return np.zeros(n)
    active = conv >= threshold * max_c
    count = int(np.sum(active))
    if count == 0:
        return gradient_from_topk(a, conv, k=1)
    p = np.zeros(2 * n - 1)
    p[active] = 1.0 / count
    return gradient_from_p(a, p)


def backtracking_mirror_descent(a, grad, max_eta=10.0, min_eta=1e-8, phi=0.5, rho=0.5, max_iter=50):
    """
    Mirror descent with backtracking line search.
    Find largest eta in [min_eta, max_eta] such that obj decreases.
    """
    n = len(a)
    base_obj = compute_obj(a)
    eta = max_eta
    for _ in range(max_iter):
        exponent = -eta * grad
        exponent = np.clip(exponent, -100, 100)
        a_new = a * np.exp(exponent)
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-300:
            eta *= rho
            continue
        a_new = a_new / s
        new_obj = compute_obj(a_new)
        if new_obj < base_obj - 1e-15:
            return a_new, eta, new_obj
        eta *= rho
        if eta < min_eta:
            break
    # If no improvement found, return original
    return a, eta, base_obj


def fast_lp_direction(a, n_constraints=15):
    """
    LP direction with reduced constraint set (only top n_constraints peaks).
    min M s.t. 2*conv(a,g)[k] - conv(a,a)[k] <= M for active k, sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv_aa = compute_conv(a_norm)

    # Select top constraints
    top_idx = np.argsort(conv_aa)[-n_constraints:]
    top_idx = np.sort(top_idx)

    # Build matrix A (only for selected rows)
    A = np.zeros((len(top_idx), n))
    k_idx = top_idx[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A[valid] = a_norm[diff[valid]]

    rhs = conv_aa[top_idx]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [2.0 * (A @ g) - M <= rhs, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None

def line_search(a, direction, n_evals=25):
    """Line search for best t in [0,1] to minimize obj((1-t)*a + t*direction)."""
    n = len(a)
    ts = np.linspace(0.0, 0.99, n_evals)
    objs = np.full(n_evals, np.inf)
    for i, t in enumerate(ts):
        a_try = np.maximum((1 - t) * a + t * direction, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            objs[i] = compute_obj(a_try / s)

    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]

    # Refinement
    if 0 < best_idx < n_evals - 1:
        for _ in range(3):
            t_lo = ts[max(0, best_idx - 1)]
            t_hi = ts[min(n_evals - 1, best_idx + 1)]
            ts_fine = np.linspace(t_lo, t_hi, 11)
            objs_fine = np.full(11, np.inf)
            for i, t in enumerate(ts_fine):
                a_try = np.maximum((1 - t) * a + t * direction, 0)
                s = float(np.sum(a_try))
                if s > 1e-10:
                    objs_fine[i] = compute_obj(a_try / s)
            fi = int(np.argmin(objs_fine))
            if objs_fine[fi] < best_obj:
                best_t = ts_fine[fi]
                best_obj = objs_fine[fi]
            t_lo = ts_fine[max(0, fi - 1)]
            t_hi = ts_fine[min(10, fi + 1)]

    return best_t, best_obj

        it += 1
        if verbose and (it % 200 == 0 or it == 1):
            print(f"    Polish it{it}: obj={best_obj:.8f}, eta={eta:.2e}, stall={stall}")

    return best_a


def fourier_init(n, mode=1, seed=42):
    """
    Create initialization based on Fourier modes.
    a_j = 1 + sum_k amp_k * cos(2*pi*k*j/n + phase_k)
    """
    rng = np.random.default_rng(seed)
    x = np.arange(n)
    a = np.ones(n)
    for k in range(1, mode + 1):
        amp = 0.1 / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * x / n + phase)
    a = np.maximum(a, 0.001)
    s = float(np.sum(a))
    if s > 0:
        a = a / s
    return a


def smooth_window_init(n, window_type='raised_cosine'):
    """Create initialization from a smooth window function."""
    x = np.arange(n)
    if window_type == 'raised_cosine':
        a = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    elif window_type == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-6)
    s = float(np.sum(a))
    if s > 0:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization with diverse strategies.
    Phase 1: Fast mirror descent (20% time)
    Phase 2: LP-driven refinement (20% time)
    Phase 3: Exact subgradient polishing (25% time)
    Phase 4: Diverse restarts + n exploration (25% time)
    Phase 5: Final polish (10% time)
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Initialize from parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Mirror Descent ===
    print(f"\n=== Phase 1: Mirror Descent (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.20, 280.0)
    if p1_budget > 30:
        a_md = mirror_descent_phase(best_seq, time_budget=p1_budget,
                                     tau_start=20.0, tau_end=5000.0, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Phase1 MD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 MD: obj={obj_md:.8f} (no improvement)")

    # === Phase 2: LP-Driven ===
    print(f"\n=== Phase 2: LP-Driven ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.18, 250.0)
    if p2_budget > 20:
        a_lp = lp_phase(best_seq, time_budget=p2_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Exact Subgradient Polish (long) ===
    print(f"\n=== Phase 3: Exact Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.25, 370.0)
    if p3_budget > 20:
        a_pol = exact_subgrad_polish(best_seq, time_budget=p3_budget, verbose=True)
        obj_pol = compute_obj(a_pol)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  Phase3 Polish: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 Polish: obj={obj_pol:.8f} (no improvement)")

    # === Phase 4: Diverse Restarts + n Exploration ===
    print(f"\n=== Phase 4: Diverse Restarts + n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Try different n values
    n_options = []
    for delta in [-100, -50, 50, 100, 200]:
        n_try = n_best + delta
        if 500 <= n_try <= 2000 and n_try not in n_options:
            n_options.append(n_try)
    for n_abs in [600, 700, 900, 1000, 1200]:
        if n_abs not in n_options and abs(n_abs - n_best) <= 400:
            n_options.append(n_abs)
    n_options = sorted(set(n_options))[:4]
    print(f"  n options: {n_options}")

    n_alloc = max(60.0, remaining * 0.4)
    per_n_budget = n_alloc / max(1, len(n_options))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)
        b_alloc = min(per_n_budget, remaining * 0.45)

        # Mirror descent
        if b_alloc > 15:
            a_try = mirror_descent_phase(a_try, time_budget=b_alloc * 0.4,
                                         tau_start=50.0, tau_end=2000.0, verbose=False)

        # Polish
        remaining = deadline - time.monotonic()
        b_rem = min(remaining * 0.3, b_alloc * 0.4)
        if b_rem > 10:
            a_try = exact_subgrad_polish(a_try, time_budget=b_rem, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # Fourier-based diverse restarts
    remaining = deadline - time.monotonic()
    restart_budget = min(remaining * 0.5, 200.0)
    if restart_budget > 40:
        per_restart = restart_budget / 3

        for r, name in enumerate(["fourier_mode2", "asymmetric", "two_peak"]):
            remaining = deadline - time.monotonic()
            if remaining < 30 or restart_budget - r * per_restart < 20:
                break

            n_cur = len(best_seq)
            if name == "fourier_mode2":
                a_r = fourier_init(n_cur, mode=4, seed=seed + 100)
                a_r = 0.6 * best_seq + 0.4 * a_r
            elif name == "asymmetric":
                # Shift the sequence to break symmetry
                shift = n_cur // 5
                a_r = np.roll(best_seq, shift)
                a_r = np.maximum(a_r, 0)
            else:
                # Two-peak modification
                a_r = best_seq.copy()
                peak1 = n_cur // 3
                peak2 = 2 * n_cur // 3
                bump = np.zeros(n_cur)
                bump[max(0, peak1 - 5):min(n_cur, peak1 + 5)] = 0.05
                bump[max(0, peak2 - 5):min(n_cur, peak2 + 5)] = 0.05
                a_r = a_r + bump
                a_r = np.maximum(a_r, 0)

            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r
            else:
                a_r = best_seq.copy()

            # Optimize
            b_r = min(remaining * 0.25, per_restart)
            if b_r > 15:
                a_r = mirror_descent_phase(a_r, time_budget=b_r * 0.5,
                                           tau_start=50.0, tau_end=1000.0, verbose=False)
                remaining = deadline - time.monotonic()
                b_r2 = min(remaining * 0.2, per_restart * 0.35)
                if b_r2 > 10:
                    a_r = exact_subgrad_polish(a_r, time_budget=b_r2, verbose=False)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart ({name}): obj={best_score:.8f} **")
            else:
                print(f"  Restart ({name}): obj={obj_r:.8f}")

    # === Phase 5: Final Polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        alloc = min(remaining * 0.5, 150.0)
        a_final = exact_subgrad_polish(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final Polish: obj={best_score:.8f} **")

    # Final LP if time permits
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.4, 80.0)
        a_final2 = lp_phase(best_seq, time_budget=alloc, verbose=False)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # One more polish after LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 50.0)
        a_final3 = exact_subgrad_polish(best_seq, time_budget=alloc, verbose=False)
        obj_final3 = compute_obj(a_final3)
        if obj_final3 < best_score:
            best_score = obj_final3
            best_seq = a_final3.copy()
            print(f"  Final Polish2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
