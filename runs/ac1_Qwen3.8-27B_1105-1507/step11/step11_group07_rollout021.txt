"""
LP-based iterative optimization with adaptive step sizes, constraint subsets,
and direct peak targeting for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = (u * rho_candidates) > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a, subset=None):
    """Build matrix A (|subset| x n) such that A @ g = conv(a, g) at subset positions."""
    n = len(a)
    if subset is None:
        subset = range(2 * n - 1)
    A = np.zeros((len(subset), n))
    for row_idx, k in enumerate(subset):
        # conv_k(a, g) = sum_i a_i * g_{k-i} for valid i
        # g_j appears with coefficient a_{k-j} when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[row_idx, j_start:j_end+1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_direction(a, time_limit=5.0, subset_size=None, verbose=False):
    """
    Solve LP: min M s.t. conv(a_norm, g)_k <= M for all k (or subset),
    sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    num_constr = 2 * n - 1
    
    # Use subset of constraints for speed
    if subset_size is not None and subset_size < num_constr:
        conv_current = compute_conv(a_norm)
        # Use the top subset_size positions + some random
        top_idx = np.argsort(conv_current)[::-1][:subset_size]
        # Add some random positions for diversity
        remaining = np.setdiff1d(np.arange(num_constr), top_idx)
        rand_idx = np.random.choice(remaining, size=min(50, len(remaining)), replace=False)
        subset = np.concatenate([top_idx, rand_idx])
    else:
        subset = np.arange(num_constr)
    
    A = build_conv_matrix(a_norm, subset)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, time_limit=time_limit if hasattr(prob, 'solve') else None, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g, t_values=None, max_time=3.0):
    """Find optimal blend factor t for a_new = (1-t)*a + t*g."""
    start = time.monotonic()
    if t_values is None:
        t_values = np.concatenate([
            np.linspace(0, 1, 20),
            np.array([1.2, 1.5, 2.0, 3.0])
        ])
    
    best_t = 0.0
    best_obj = compute_obj(a)
    
    for t in t_values:
        if time.monotonic() - start > max_time:
            break
        if t < 0:
            continue
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        obj = compute_obj(a_try / s)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    return best_t, best_obj


def lp_iteration(a, step_candidates=None, subset_size=None, verbose=False):
    """
    One LP iteration: find direction g, then line search for best blend.
    Returns (new_a, new_obj, g, t_used).
    """
    g, M = solve_lp_direction(a, subset_size=subset_size)
    if g is None:
        return a, compute_obj(a), None, 0.0
    
    t, obj = line_search_blend(a, g, step_candidates=step_candidates)
    
    if t > 0:
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s > 1e-10:
            a_new = a_new / s
        else:
            a_new = a
    else:
        a_new = a
    
    return a_new, obj, g, t

    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)

def soft_max_gradient_descent(a, time_budget=100.0, beta_start=10.0, beta_end=50.0,
                              lr_start=0.008, momentum=0.9, verbose=False, seed=42):
    """Soft-max gradient descent with momentum on the simplex."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    it = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress
        lr = lr_start * (1.0 - 0.3 * progress)
        
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient via cross-correlation
        a_rev = a[::-1]
        size = 1
        while size < n + len(p) - 1:
            size <<= 1
        A = rfft(a_rev, size)
        P = rfft(p, size)
        full = irfft(A * P, size)
        q = full[n - 1:n - 1 + n]
        grad = 2.0 * q
        
        v = momentum * v + grad
        a_new = project_simplex(a - lr * v)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall > 100:
                v = np.zeros(n)
                stall = 0
                pert = rng.normal(0, 0.002, n)
                a = project_simplex(best_a + pert)
        
        a = a_new
        it += 1
        
        if verbose and it % 200 == 0:
            print(f"    SmaxGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, "
                  f"lr={lr:.5f}, t={elapsed:.1f}s")
    
    return best_a, best_obj

"""
LP-driven iterative optimization with adaptive constraint subsets, over-relaxation,
multiplicative weights, and diverse length exploration for minimizing
2*n*max(conv(a,a))/(sum(a))^2 on non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Assumes sum(a)=1 for speed."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = (u * rho_candidates) > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix_rows(a, positions):
    """Build matrix rows for given convolution positions. A[row] @ g = conv(a,g)_position."""
    n = len(a)
    A = np.zeros((len(positions), n))
    for row_idx, k in enumerate(positions):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            coeffs = a[k - j_end:k - j_start + 1][::-1]
            A[row_idx, j_start:j_end + 1] = coeffs
    return A


def solve_lp_direction(a, positions, time_limit=5.0):
    """
    Solve LP: min M s.t. conv(a_norm, g)_k <= M for k in positions,
    sum(g)=1, g>=0. Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix_rows(a_norm, positions)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g, max_time=2.0):

        it += 1
        if verbose and (it % 10 == 0 or obj_opt < best_obj):
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                  f"K={len(top_idx)}, stall={stall}, t_elapsed={time.monotonic()-start:.1f}s")

    return best_a, best_obj


def multiplicative_weights(a, time_budget=60.0, lr_start=0.05, lr_end=0.01,
                           verbose=False, seed=42):
    """
    Multiplicative weights / mirror descent on the simplex.
    Updates: a[i] *= exp(-lr * subgrad[i]), then project to simplex.
    Subgradient of max_k conv_k(a,a) at a.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        lr = lr_start * (lr_end / lr_start) ** progress

        conv = compute_conv(a)
        c_max = float(np.max(conv))

        # Compute subgradient: sum over peak positions
        # conv_k(a,a) = sum_i a_i * a_{k-i}, d/da_j = a_{k-j} + a_{j-k}
        peak_thresh = c_max - 1e-10
        peak_positions = np.where(conv >= peak_thresh)[0]

        # Use top few peaks for efficiency
        if len(peak_positions) > 20:
            peak_positions = np.argsort(conv)[::-1][:20]

        subgrad = np.zeros(n)
        for k in peak_positions:
            for j in range(n):
                # Contribution from a_j to conv_k:
                # a_j appears as a_i with i=j: coefficient is a_{k-j} if 0<=k-j<n
                # a_j appears as a_{k-i} with i=k-j: coefficient is a_j... wait
                # conv_k = sum_i a_i * a_{k-i}
                # d(conv_k)/da_j = a_{k-j} (if 0 <= k-j < n) + a_{j-k} (if 0 <= j-k < n)
                kj = k - j
                if 0 <= kj < n:
                    subgrad[j] += a[kj]
                jk = j - k
                if 0 <= jk < n:
                    subgrad[j] += a[jk]

        # Multiplicative update
        a_new = a * np.exp(-lr * subgrad)
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                noise = rng.normal(0, 0.002, n)
                a = project_simplex(best_a + noise)
                stall = 0

        a = a_new
        it += 1

        if verbose and it % 100 == 0:
            print(f"    MW it{it}: obj={best_obj:.8f}, lr={lr:.5f}, t={elapsed:.1f}s")

    return best_a, best_obj


def soft_max_gradient_descent(a, time_budget=100.0, beta_start=10.0, beta_end=50.0,
                              lr_start=0.008, momentum=0.9, verbose=False, seed=42):
    """Soft-max gradient descent with Nesterov momentum on the simplex."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    it = 0
    stall = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress
        lr = lr_start * (1.0 - 0.4 * progress)

        conv = compute_conv(a)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p = exp_vals / float(np.sum(exp_vals))

        # Gradient via cross-correlation (FFT)
        a_rev = a[::-1]
        size = 1
        while size < n + len(p) - 1:
            size <<= 1
        A = rfft(a_rev, size)
        P = rfft(p, size)
        full = irfft(A * P, size)
        q = full[n - 1:n - 1 + n]
        grad = 2.0 * q

        v = momentum * v + grad
        a_new = project_simplex(a - lr * v)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 150:
                v = np.zeros(n)
                stall = 0
                pert = rng.normal(0, 0.003, n)
                a = project_simplex(best_a + pert)

        a = a_new
        it += 1

        if verbose and it % 200 == 0:
            print(f"    SmaxGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, "
                  f"lr={lr:.5f}, t={elapsed:.1f}s")

    return best_a, best_obj


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    return project_simplex(a)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = (np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) +
             np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01)
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 2
    elif mode == 6:
        a = 1.0 / np.cosh(4.0 * (2 * x - 1)) ** 2 + 0.01
    elif mode == 7:
        a = np.zeros(n)
        a[0] = 1.0
        a[-1] = 1.0
        a[1:min(5, n)] = 0.4
        a[max(0, n - 5):n - 1] = 0.4
        if n > 6:
            a[1:-1] += 0.03
    elif mode == 8:
        a = 1.0 / (0.01 + 20.0 * (2 * x - 1) ** 2)
    else:
        a = (2 * x - 1) ** 6 + 0.02
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def peak_clipping_iteration(a, n_steps=5):
    """
    Direct peak clipping: identify peak, transfer mass from contributing pairs.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64))

    for _ in range(n_steps):
        conv = compute_conv(a)
        peak_pos = int(np.argmax(conv))
        peak_val = conv[peak_pos]

        # Find pairs (i, peak_pos-i) that contribute to the peak
        # Transfer mass from larger to smaller in each pair
        for i in range(n):
            j = peak_pos - i
            if 0 <= j < n and j != i:
                if a[i] > a[j]:
                    delta = (a[i] - a[j]) * 0.01
                    a[i] -= delta
                    a[j] += delta
                elif a[j] > a[i]:
                    delta = (a[j] - a[i]) * 0.01
                    a[j] -= delta
                    a[i] += delta
                break  # Only do a few per iteration

        a = np.maximum(a, 0)
        a = project_simplex(a)

    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining LP polishing, multiplicative weights, soft-max GD,
    peak clipping, diverse restarts, and length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = u_shaped_init(n_current, sharpness=3.0)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_current}): obj={best_score:.8f}")

    # Try various initializations
    for sharp in [1.0, 2.0, 4.0, 6.0, 10.0, 15.0]:
        a_try = u_shaped_init(n_current, sharpness=sharp)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"U-shape (sharp={sharp}): obj={best_score:.8f}")

    for mode in range(9):
        a_try = diverse_init(n_current, rng, mode=mode)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"Init mode {mode}: obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Soft-max GD for fast initial progress ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 1: SoftMax GD ({remaining:.0f}s avail) ===")
        alloc = min(remaining * 0.15, 120.0)
        a1, obj1 = soft_max_gradient_descent(
            best_seq, time_budget=alloc,
            beta_start=5.0, beta_end=30.0,
            lr_start=0.01, momentum=0.9,
            verbose=True, seed=seed
        )
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1: obj={best_score:.8f} **")

    # === Phase 2: Adaptive LP polishing (main workhorse) ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 2: Adaptive LP ({remaining:.0f}s avail) ===")
        alloc = min(remaining * 0.35, 400.0)
        a2, obj2 = adaptive_lp_polish(best_seq, time_budget=alloc, max_iters=100, verbose=True)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2: obj={best_score:.8f} **")

    # === Phase 3: Multiplicative weights ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 3: Multiplicative Weights ({remaining:.0f}s avail) ===")
        alloc = min(remaining * 0.12, 100.0)
        a3, obj3 = multiplicative_weights(best_seq, time_budget=alloc,
                                          lr_start=0.05, lr_end=0.005,
                                          verbose=True, seed=seed + 50)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3: obj={best_score:.8f} **")

    # === Phase 4: More LP polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: More LP ({remaining:.0f}s avail) ===")
        alloc = min(remaining * 0.25, 250.0)
        a4, obj4 = adaptive_lp_polish(best_seq, time_budget=alloc, max_iters=80, verbose=True)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4: obj={best_score:.8f} **")

    # === Phase 5: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 5: Diverse Restarts ({remaining:.0f}s avail) ===")
        n_cur = len(best_seq)

        for r in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break

            mode = r % 9
            a_r = diverse_init(n_cur, rng, mode=mode)
            # Blend with current best
            for blend in [0.4, 0.6, 0.8]:
                a_r = project_simplex((1 - blend) * diverse_init(n_cur, rng, mode=mode) + blend * best_seq)
                obj_r = compute_obj(a_r)
                if obj_r < best_score:
                    # Quick polish
                    alloc_q = min(remaining * 0.15, 40.0)
                    if alloc_q > 10:
                        a_r, obj_r = adaptive_lp_polish(a_r, time_budget=alloc_q, max_iters=20, verbose=False)
                    if obj_r < best_score:
                        best_score = obj_r
                        best_seq = a_r.copy()
                        print(f"  R{r} blend={blend} mode={mode}: obj={best_score:.8f} **")
                    break

    # === Phase 6: Length exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 6: Length Exploration ({remaining:.0f}s avail) ===")
        n_best = len(best_seq)

        n_options = sorted(set([
            max(300, n_best - 100), max(300, n_best - 200),
            n_best + 100, n_best + 200, n_best + 400,
            500, 700, 900, 1100, 1300
        ]))
        n_options = [n for n in n_options if 300 <= n <= 2500 and abs(n - n_best) >= 50]
        n_options = n_options[:5]
        print(f"  Trying n: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break

            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = project_simplex(np.maximum(a_try, 0))

            # Quick optimization
            alloc_sgd = min(remaining * 0.25, 40.0)
            if alloc_sgd > 10:
                a_try, obj_try = soft_max_gradient_descent(
                    a_try, time_budget=alloc_sgd,
                    beta_start=10.0, beta_end=40.0,
                    lr_start=0.008, momentum=0.9,
                    verbose=False, seed=seed + 200
                )

            remaining = deadline - time.monotonic()
            if remaining > 30:
                alloc_lp = min(remaining * 0.35, 50.0)
                a_try, obj_try = adaptive_lp_polish(a_try, time_budget=alloc_lp, max_iters=20, verbose=False)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 7: Final Polish ({remaining:.0f}s avail) ===")

        # LP polish
        alloc1 = min(remaining * 0.5, 150.0)
        if alloc1 > 20:
            a_f, obj_f = adaptive_lp_polish(best_seq, time_budget=alloc1, max_iters=50, verbose=True)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        # Multiplicative weights
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc2 = min(remaining * 0.3, 60.0)
            if alloc2 > 10:
                a_f2, obj_f2 = multiplicative_weights(best_seq, time_budget=alloc2,
                                                      lr_start=0.03, lr_end=0.005,
                                                      verbose=False, seed=seed + 500)
                if obj_f2 < best_score:
                    best_score = obj_f2
                    best_seq = a_f2.copy()
                    print(f"  Final MW: obj={best_score:.8f} **")

        # One more LP
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc3 = min(remaining * 0.6, 60.0)
            if alloc3 > 10:
                a_f3, obj_f3 = adaptive_lp_polish(best_seq, time_budget=alloc3, max_iters=30, verbose=False)
                if obj_f3 < best_score:
                    best_score = obj_f3
                    best_seq = a_f3.copy()
                    print(f"  Final LP2: obj={best_score:.8f} **")

    # === Phase 8: Last-ditch perturbations ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        for attempt in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break
            noise = rng.normal(0, 0.002, len(best_seq))
            a_p = project_simplex(best_seq + noise)
            alloc = min(remaining * 0.4, 15.0)
            if alloc > 5:
                a_p, obj_p = adaptive_lp_polish(a_p, time_budget=alloc, max_iters=10, verbose=False)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Last-ditch {attempt}: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
