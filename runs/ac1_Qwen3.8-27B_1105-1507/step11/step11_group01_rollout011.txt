"""
Optimizes step function coefficients to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses LP-based block coordinate descent (state-of-the-art approach), exact
coordinate descent with FFT updates, multi-n exploration, and random restarts.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def obj_value(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2 (matches evaluate_sequence)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return float(2.0 * n * np.max(conv) / (s * s))


def obj_normalized(a):
    """Compute 2n * max(conv) assuming sum(a)=1."""
    n = len(a)
    conv = compute_conv(a)
    return float(2.0 * n * np.max(conv))


def build_conv_matrix(a):
    """Build matrix C (2n-1 x n) such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        row_start = j
        row_end = min(2 * n - 1, j + n)
        C[row_start:row_end, j] = a[:row_end - j]
    return C


def solve_lp_minpeak(a, timeout=30):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    C = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [C @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0.0)
                g_sum = float(np.sum(g_val))
                if g_sum > 1e-10:
                    g_val = g_val / g_sum
                    return g_val, float(M.value)
        except Exception:
            continue
    return None, None

    max_val = max(max_lin, quad)
    denom = (S_other + x) * (S_other + x)
    if denom < 1e-30:
        return 1e30
    return 2.0 * n * max_val / denom

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)

def subgradient_step(a, eta, momentum, v_mom, seed=42):
    """Single subgradient step using KKT conditions."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    argmax_set = np.where(conv >= max_conv * (1.0 - 1e-10))[0]

    v_sub = np.zeros(n)
    for k in argmax_set:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            v_sub[lo:hi + 1] += a[k - hi:k - lo + 1]
    v_sub /= max(len(argmax_set), 1)

    target = max_conv / 2.0
    grad = 2.0 * v_sub - target
    grad -= np.mean(grad)  # Center the gradient (invariance to scaling)

    v_mom = momentum * v_mom - eta * grad
    a_new = a + v_mom
    a_new = np.maximum(a_new, 0.0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        a_new = a.copy()
        v_mom = np.zeros(n)
    else:
        a_new = a_new / s_new

    return a_new, v_mom

                print(f"Parent: n={len(parent)}, normalized_obj={best_score:.10f}")
    except Exception:
        pass

    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(800) / 800.0
        best_score = 2.0
        print(f"Default init: n=800, obj=2.0")

    n_cur = len(best_seq)

    # === Phase 1: LP Block Descent ===
    print(f"\n=== Phase 1: LP Block Descent (n={n_cur}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 450.0)
    if p1_budget > 30:
        a_lp = lp_block_descent(best_seq, n_iters=40, time_budget=p1_budget,
                                verbose=True, seed=100)
        obj_lp = obj_normalized(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** LP: {best_score:.10f} **")
        else:
            print(f"  LP: {obj_lp:.10f} (no improvement)")

    # === Phase 2: Exact CD ===
    print(f"\n=== Phase 2: Exact Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.20, 250.0)
    if p2_budget > 20:
        a_cd = coordinate_descent(best_seq, time_budget=p2_budget,
                                  verbose=True, seed=200)
        obj_cd = obj_normalized(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  ** CD: {best_score:.10f} **")
        else:
            print(f"  CD: {obj_cd:.10f}")

    # === Phase 3: LP + CD Alternation ===
    print(f"\n=== Phase 3: LP + CD Alternation ===")
    remaining = deadline - time.monotonic()
    n_rounds = 3
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        # LP step
        lp_alloc = min(remaining * 0.25, 120.0)
        if lp_alloc > 15:
            a = lp_block_descent(best_seq, n_iters=10, time_budget=lp_alloc,
                                 verbose=False, seed=500 + rnd)
            obj = obj_normalized(a)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  ** Round {rnd} LP: {best_score:.10f} **")
        # CD step
        remaining = deadline - time.monotonic()
        cd_alloc = min(remaining * 0.25, 100.0)
        if cd_alloc > 15:
            a = coordinate_descent(best_seq, time_budget=cd_alloc,
                                   verbose=False, seed=600 + rnd)
            obj = obj_normalized(a)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  ** Round {rnd} CD: {best_score:.10f} **")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    n_options = sorted(set([
        max(200, n_cur - 200), max(200, n_cur - 100),
        n_cur + 100, n_cur + 200, n_cur + 400,
        500, 700, 1000, 1200, 1500
    ]))
    # Filter
    n_filtered = []
    for nc in n_options:
        if 300 <= nc <= 3000:
            if all(abs(nc - nf) > 80 for nf in n_filtered):
                n_filtered.append(nc)
    n_filtered = n_filtered[:5]
    print(f"  n candidates: {n_filtered}")

    for nc in n_filtered:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, nc)

        # LP on resampled
        lp_alloc = min(remaining * 0.15, 60.0)
        if lp_alloc > 10:
            a_try = lp_block_descent(a_try, n_iters=5, time_budget=lp_alloc,
                                     verbose=False, seed=nc)

        # CD on resampled
        cd_alloc = min(remaining * 0.10, 40.0)
        if cd_alloc > 10:
            a_try = coordinate_descent(a_try, time_budget=cd_alloc,
                                       verbose=False, seed=nc + 1)

        obj_try = obj_normalized(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_cur = nc
            print(f"  ** n={nc}: {best_score:.10f} **")
        else:
            print(f"  n={nc}: {obj_try:.10f}")

    # === Phase 5: Subgradient Escape + CD ===
    print(f"\n=== Phase 5: Subgradient Escape ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        n_cur = len(best_seq)
        a_sub = best_seq.copy()
        v_mom = np.zeros(n_cur)
        eta = 1.0 / np.sqrt(n_cur)
        momentum = 0.7
        pdl_start = time.monotonic()
        pdl_budget = min(remaining * 0.15, 120.0)
        st = 0

        for it in range(5000):
            if time.monotonic() - pdl_start >= pdl_budget:
                break
            a_sub, v_mom = subgradient_step(a_sub, eta, momentum, v_mom)
            obj_sub = obj_normalized(a_sub)
            if obj_sub < best_score:
                best_score = obj_sub
                best_seq = a_sub.copy()
                st = 0
            else:
                st += 1
            if st > 100:
                eta *= 0.5
                v_mom = np.zeros(n_cur)
                st = 0
                if eta < 1e-6:
                    break

        # CD after subgradient
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_cd = coordinate_descent(a_sub, time_budget=min(remaining * 0.2, 60.0),
                                      verbose=False, seed=700)
            obj_cd = obj_normalized(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  ** SG+CD: {best_score:.10f} **")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)
    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = 1 + r
        a_r = diverse_init(n_cur, rng, mode=mode)
        blend = 0.2 + 0.15 * r
        a_r = (1.0 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0.0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.25, 80.0)
        if alloc > 15:
            # Quick LP
            a_r = lp_block_descent(a_r, n_iters=3, time_budget=alloc * 0.4,
                                   verbose=False, seed=1000 + r)
            # CD
            alloc2 = min(remaining * 0.20, alloc * 0.5)
            if alloc2 > 10:
                a_r = coordinate_descent(a_r, time_budget=alloc2,
                                         verbose=False, seed=1100 + r)

        obj_r = obj_normalized(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  ** Restart {r} (mode {mode}): {best_score:.10f} **")
        else:
            print(f"  Restart {r} (mode {mode}): {obj_r:.10f}")

    # === Phase 7: Intensive Final Polish ===
    print(f"\n=== Phase 7: Intensive Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Long CD
        cd_alloc = min(remaining * 0.5, 300.0)
        if cd_alloc > 20:
            a_f = coordinate_descent(best_seq, time_budget=cd_alloc,
                                     verbose=True, seed=9999)
            obj_f = obj_normalized(a_f)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  ** Final CD: {best_score:.10f} **")
            else:
                print(f"  Final CD: {obj_f:.10f}")

        # Final LP
        remaining = deadline - time.monotonic()
        if remaining > 20:
            lp_alloc = min(remaining * 0.4, 100.0)
            if lp_alloc > 10:
                a_f2 = lp_block_descent(best_seq, n_iters=8, time_budget=lp_alloc,
                                        verbose=False, seed=12345)
                obj_f2 = obj_normalized(a_f2)
                if obj_f2 < best_score:
                    best_score = obj_f2
                    best_seq = a_f2.copy()
                    print(f"  ** Final LP: {best_score:.10f} **")

    # === Prepare and validate output ===
    best_seq = np.maximum(best_seq, 0.0)
    result = [float(min(1000.0, x)) for x in best_seq]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if not np.isinf(val) and val < best_score:
        print(f"  evaluate_sequence gives {val:.10f} (better than internal)")
    elif np.isinf(val):
        print("  Warning: evaluate_sequence returned inf, using fallback")
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, internal_score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
