"""
Subgradient descent with exact active-set gradients for step function optimization.
Uses Polyak step sizes, Nesterov momentum, and LP polishing for final refinement.
Key innovations: exact subgradient computation (no soft-max approximation),
adaptive active set size, and pairwise coordinate optimization.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective assuming a is already normalized (sum=1)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    valid = u * rho > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    idx = np.nonzero(valid)[0][-1]
    theta = (cssv[idx] - 1) / (idx + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_topk(a, top_k=5):
    """
    Compute subgradient of max(conv(a,a)) using top_k active indices.
    Returns (gradient, conv_values, active_indices, max_val).
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    
    # Get top_k indices
    if top_k >= len(conv):
        idx = np.arange(len(conv))
        weights = np.ones(len(idx)) / len(idx)
    else:
        # Get top_k
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        # Get their values for weighting
        top_vals = conv[top_idx]
        max_conv = float(np.max(top_vals))
        # Weight by proximity to max (or uniform)
        weights = np.ones(len(top_idx)) / len(top_idx)
        idx = top_idx
    
    # Compute gradient: for each active index k, grad_k[i] = 2 * a[(k-i) mod ...]
    # More precisely: grad_k[i] = 2 * a[k-i] if 0 <= k-i <= n-1, else 0
    grad = np.zeros(n)
    for t in range(len(idx)):
        k = idx[t]
        w = weights[t]
        # grad contribution from peak at index k
        # For each i in [0, n-1]: grad[i] += w * 2 * a[k-i] if 0 <= k-i <= n-1
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # grad[lo:hi+1] += w * 2 * a[k-hi:k-lo+1][::-1]
        # a[k-i] for i in [lo, hi] is a[k-hi : k-lo+1] reversed
        segment = a[k - hi : k - lo + 1][::-1]
        grad[lo:hi + 1] += w * 2.0 * segment
    
    return grad, conv, idx, max_val


def polyak_step(obj_val, obj_lb, grad_norm_sq):
    """Compute Polyak step size."""
    if grad_norm_sq < 1e-15:
        return 1e-6
    step = (obj_val - obj_lb) / grad_norm_sq
    # Cap the step to avoid divergence
    return min(step, 1.0)


def subgradient_descent(a, time_budget=300.0, top_k=5, use_momentum=False,
                        momentum=0.9, verbose=True, seed=42, 
                        anneal_k=False, k_min=1, k_max=15):
    """
    Subgradient descent with Polyak step size on the simplex.
    Optionally uses Nesterov momentum and adaptive top_k.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    obj_lb = 2.0 * n / (2.0 * n - 1.0)  # Theoretical lower bound
    
    start = time.monotonic()
    v = np.zeros(n)  # momentum
    it = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)
    
    # Adaptive k
    current_k = top_k
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Adaptive top_k: start small, increase over time
        if anneal_k:
            current_k = max(k_min, int(k_max - (k_max - k_min) * progress))
        else:
            current_k = top_k
        
        # Compute subgradient
        grad, conv, active_idx, max_conv = subgradient_topk(a, current_k)
        
        # Objective (we already have max_conv, so obj = 2*n*max_conv since sum=1)
        obj_val = 2.0 * n * max_conv
        
        # Polyak step
        grad_norm_sq = float(np.sum(grad * grad))
        step = polyak_step(obj_val, obj_lb, grad_norm_sq)
        
        # Step decay for stability
        step *= (1.0 - 0.5 * progress)
        
        if use_momentum:
            v = momentum * v + grad
            a_new = a - step * v
        else:
            a_new = a - step * grad
        
        a_new = project_simplex(a_new)
        
        # Evaluate
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Reset momentum and add perturbation
                v = np.zeros(n)
                pert = rng.normal(0, 0.005, n)
                a = project_simplex(best_a + pert)
                stall = 0
        
        a = a_new
        it += 1
        
        if verbose and it % 200 == 0:
            print(f"    SubGD it{it}: obj={best_obj:.8f}, k={current_k}, "
                  f"step={step:.6f}, t={elapsed:.1f}s")
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_norm, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, solver=None):
    """Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]
    if solver is not None:
        solvers = [solver] + [s for s in solvers if s != solver]
    
    for s in solvers:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s_val = float(np.sum(g_val))
                if s_val > 1e-10:
                    return g_val / s_val, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget=120.0, max_iters=40, verbose=False):
    """LP-based polishing."""
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(999)
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Line search for blend
        best_t = 0
        best_obj_t = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = project_simplex((1 - t) * a + t * g_lp)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t
        
        if best_t > 0 and best_obj_t < best_obj - 1e-10:
            a_new = project_simplex((1 - best_t) * a + best_t * g_lp)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        it += 1
        if verbose and it % 5 == 0:
            print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, "
                  f"elapsed={time.monotonic()-start:.1f}s")
        
        if stall >= 5:
            noise = rng.normal(0, 0.003 * stall, n)
            a = project_simplex(best_a + noise)
            stall = 0
    
    return best_a, best_obj


def pairwise_optimize(a, time_budget=60.0, verbose=False):
    """
    Pairwise coordinate optimization: for pairs (i,j), find best delta
    that transfers mass from one to the other.
    """
    n = len(a)
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(555)
    it = 0
    
    while time.monotonic() - start < time_budget:
        # Pick a random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        j = min(j, n - 1)
        if i == j:
            j = (j + 1) % n
        
        # Try several delta values
        best_delta = 0
        best_obj_delta = best_obj
        scale = max(a[i], a[j], 0.01)
        for d in [-0.5*scale, -0.3*scale, -0.1*scale, 
                   0.1*scale, 0.3*scale, 0.5*scale]:
            a_try = a.copy()
            a_try[i] += d
            a_try[j] -= d
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_delta:
                best_obj_delta = obj_try
                best_delta = d
        
        if best_delta != 0 and best_obj_delta < best_obj - 1e-10:
            a[i] += best_delta
            a[j] -= best_delta
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            a = a / s
            best_obj = best_obj_delta
            best_a = a.copy()
        
        it += 1
        if verbose and it % 1000 == 0:
            print(f"    Pairwise it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
    
    return best_a, best_obj


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    return project_simplex(a)


def arcsine_init(n):
    """Arcsine distribution initialization."""
    x = np.linspace(0.01, 0.99, n)
    a = 1.0 / np.sqrt(np.maximum(1 - (2 * x - 1) ** 2, 0.001))
    return project_simplex(a)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = (np.exp(-((idx - n // 4) ** 2) / (2 * (n // 6) ** 2)) +
             np.exp(-((idx - 3 * n // 4) ** 2) / (2 * (n // 6) ** 2)) + 0.01)
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(5 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (1.0 + 10.0 * (2 * x - 1) ** 2)
    elif mode == 6:
        a = 1.0 / np.cosh(4.0 * (2 * x - 1))
    elif mode == 7:
        a = np.ones(n)
        a[0] = 2.0
        a[-1] = 2.0
        a[1:3] = 1.5
        a[-3:-1] = 1.5
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s
