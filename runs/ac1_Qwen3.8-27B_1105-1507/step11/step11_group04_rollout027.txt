"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses alternating LP (block coordinate descent) with exact CD polishing,
Fourier perturbations for restart, and multi-length exploration.
New: bidirectional LP (optimize both a and g via LP), smarter CD with 
analytical candidate generation, and adaptive time allocation.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_fixed(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g).
    Here a is fixed, g is variable."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_for_variable(fixed, warm_start=None):
    """
    Solve LP: min M s.t. (fixed * x)[k] <= M for all k, sum(x)=1, x>=0.
    'fixed' is the known sequence, x is the variable.
    Returns (x_opt, M_opt).
    """
    n = len(fixed)
    fixed_norm = fixed / float(np.sum(fixed))
    A = build_conv_matrix_fixed(fixed_norm)
    
    x = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ x <= M, cp.sum(x) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if warm_start is not None:
        try:
            x.value = np.maximum(np.asarray(warm_start), 0)
        except Exception:
            pass
    
    for solver_name in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=getattr(cp, solver_name))
            if x.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                x_val = np.maximum(np.asarray(x.value), 0)
                s = float(np.sum(x_val))
                if s > 1e-10:
                    return x_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def alternating_lp(a, time_budget, seed=42, verbose=True):
    """
    Alternating LP: optimize g given a, then optimize a given g, repeat.
    This is a block coordinate descent on the bilinear problem.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_x = None
    iter_count = 0
    stall = 0
    
    while time.monotonic() - start < time_budget - 2:
        # Step 1: Optimize g given a (LP)
        g, M_g = solve_lp_for_variable(a, warm_start=prev_x)
        if g is None:
            break
        prev_x = g
        iter_count += 1
        
        # Step 2: Optimize a given g (LP)
        a_new, M_a = solve_lp_for_variable(g, warm_start=a)
        if a_new is None:
            a_new = 0.5 * a + 0.5 * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / float(np.sum(a_new))
        
        # Evaluate
        obj_new = compute_obj_norm(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            if verbose and iter_count % 10 == 0:
                elapsed = time.monotonic() - start
                print(f"    AltLP {iter_count}: obj={best_obj:.10f}, M_g={M_g:.6f}, M_a={M_a:.6f}, t={elapsed:.1f}s")
        else:
            # Try convex combination
            improved = False
            for t in [1.0, 0.7, 0.5, 0.3, 0.1]:
                a_try = (1 - t) * a + t * a_new
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj_norm(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    stall = 0
                    improved = True
                    break
            
            if not improved:
                stall += 1
                if stall >= 5:

                    pert = rng.standard_normal(n) * 0.01
