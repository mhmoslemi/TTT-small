def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    for j in range(n):

    return A

Uses improved LP alternating with Nesterov momentum and adaptive step size,

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = np.maximum(warm_start, 0).copy()
        except Exception:
            pass

    for solver in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def fine_cd(a, time_budget, seed=42, verbose=False):

def diverse_restart(a, rng, mode=0):
    """Generate diverse restarts from a base sequence."""
    n = len(a)
    x = np.arange(n)

    if mode == 0:
        # Smooth Gaussian bump
        center = rng.uniform(0.2, 0.8) * n
        width = rng.uniform(0.1, 0.4) * n
        bump = np.exp(-0.5 * ((x - center) / width) ** 2)
        a_new = a * 0.7 + bump * 0.3
    elif mode == 1:
        # Bilateral (peaked at edges)
        edge = np.abs(x - n / 2) / (n / 2)
        a_new = a * 0.6 + (1.0 - edge) * 0.2 + edge**2 * 0.2
    elif mode == 2:
        # Fourier perturbation
        pert = np.zeros(n)
        for k in range(1, 12):
            freq = 2.0 * np.pi * k / n
            amp = rng.uniform(0.5, 2.0) / k
            phase = rng.uniform(0, 2.0 * np.pi)
            pert += amp * np.cos(freq * x + phase)
        a_new = a * np.exp(0.05 * pert)
    elif mode == 3:
        # Random smooth
        base = rng.uniform(0.5, 1.5, n)
        # Smooth with a kernel
        kernel_size = max(3, n // 50)
        kernel = np.ones(kernel_size) / kernel_size
        base = np.convolve(base, kernel, mode='same')
        a_new = a * 0.5 + base / base.sum() 0.5
    elif mode == 4:
        # Concentrated middle
        center = n // 2
        half = n // 4
        mask = np.zeros(n)
        mask[center - half:center + half] = 1.0
        # Smooth edges
        edge_width = max(3, n // 40)
        for i in range(edge_width):
            frac = 1.0 - i / edge_width
            mask[center - half + i] *= frac
            mask[center + half - 1 - i] *= frac
        a_new = a * 0.5 + mask * 0.5
    else:
        # Multiplicative random
        a_new = a * rng.uniform(0.5, 1.5, n)

    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s
