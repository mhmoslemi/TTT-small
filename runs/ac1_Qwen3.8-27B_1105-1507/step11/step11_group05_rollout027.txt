Key innovation: exact subgradient is O(n) vs O(n^2) for LSE, enabling 100x more iterations.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    r = rho_idx[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n

        if lo <= hi:
            i_vals = np.arange(lo, hi + 1)
            grad[i_vals] += 2.0 * w * a[ki - i_vals]
    return grad, conv

def compute_conv(a):
    n = len(a)
    if n <= 1:
        return a.astype(np.float64).copy()
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    an = a / s
    c = compute_conv(an)
    return 2.0 * n * float(np.max(c))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    r = idx[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = float(proj.sum())
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n


def exact_subgrad(a, top_k):
    """Compute exact multi-peak subgradient. Returns (grad, conv, peak_indices)."""
    n = len(a)
    conv = compute_conv(a)
    if top_k >= len(conv):
        top_k = len(conv)
    order = np.argsort(conv)[::-1][:top_k]
    pv = conv[order]
    w = pv / (float(pv.sum()) + 1e-30)
    grad = np.zeros(n)
    for t in range(top_k):
        ki = int(order[t])
        wt = float(w[t])
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            iv = np.arange(lo, hi + 1)
            grad[iv] += 2.0 * wt * a[ki - iv]
    return grad, conv, order

def build_conv_matrix(a):
    """Build the convolution matrix A where (A*g)[k] = (a*g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def full_lp(a):
    """Solve full LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                s = float(np.sum(gv))
                if s > 1e-10:
                    return gv / s
        except Exception:
            continue
    return None


def lp_blend(a, g, prev_best):
