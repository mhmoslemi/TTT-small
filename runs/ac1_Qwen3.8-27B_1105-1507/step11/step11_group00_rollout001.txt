"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a hybrid approach: LP-driven direction finding with golden section line search,
coordinate descent polishing, spectral shaping, and multi-n exploration.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.ndimage import uniform_filter1d
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def gradient_from_p(a, p):
    """
    Compute 2 * sum_j a[j] * p[i+j] for i=0..n-1.
    p has length 2n-1 (distribution over convolution lags).
    """
    n = len(a)
    b = a[::-1]
    N_fft = 1
    while N_fft < 3 * n - 1:
        N_fft <<= 1
    c = irfft(rfft(b, N_fft) * rfft(p, N_fft), N_fft)
    return 2.0 * c[n - 1:2 * n - 1]


def smoothed_gradient(a, conv, tau):
    """Compute gradient of LSE(tau, conv(a,a)) with respect to a."""
    n = len(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    if shifted.max() > 500:
        shifted = shifted - shifted.max()
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(2 * n - 1)
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    return gradient_from_p(a, p)


def topk_subgradient(a, conv, k=3):
    """Subgradient using only the top-k peaks."""
    n = len(a)
    idx = np.argsort(conv)[-k:]
    p = np.zeros(2 * n - 1)
    p[idx] = 1.0 / k
    return gradient_from_p(a, p)


def lp_direction(a, active_k=None):
    """
    Solve LP: min M s.t. 2*conv(a,g)_k - conv(a,a)_k <= M for all k,
    sum(g) = 1, g >= 0.
    If active_k is provided, only enforce constraints on those lags tightly.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv_aa = compute_conv(a_norm)
    
    if active_k is not None:
        # Only enforce constraints on active lags
        # For other lags, use a loose bound
        constraints_A = []
        k_indices = list(active_k)
        A = np.zeros((len(k_indices), n))
        k_idx = np.array(k_indices)[:, None]
        j_idx = np.arange(n)[None, :]
        diff = k_idx - j_idx
        valid = (diff >= 0) & (diff < n)
        A[valid] = a_norm[diff[valid]]
        rhs = conv_aa[active_k].copy()
    else:
        A = np.zeros((2 * n - 1, n))
        k_idx = np.arange(2 * n - 1)[:, None]
        j_idx = np.arange(n)[None, :]
        diff = k_idx - j_idx
        valid = (diff >= 0) & (diff < n)
        A[valid] = a_norm[diff[valid]]
        rhs = conv_aa.copy()
    
    g = cp.Variable(len(A[1]), nonneg=True)
    M = cp.Variable()
    constraints = [2.0 * (A @ g) - M <= rhs, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def golden_section_search(a, g, f, a_lb=0.0, a_ub=1.0, tol=1e-6, max_iter=50):
    """
    Golden section search for min f(t) = obj((1-t)*a + t*g) on [a_lb, a_ub].
    f is the objective function.
    Returns (best_t, best_obj).
    """
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618
    
    c = a_ub - gr * (a_ub - a_lb)
    d = a_lb + gr * (a_ub - a_lb)
    
    # Evaluate f at c and d
    a_c = np.maximum((1 - c) * a + c * g, 0)
    s_c = float(np.sum(a_c))
    f_c = f(a_c / s_c) if s_c > 1e-10 else np.inf
    
    a_d = np.maximum((1 - d) * a + d * g, 0)
    s_d = float(np.sum(a_d))
    f_d = f(a_d / s_d) if s_d > 1e-10 else np.inf
    
    for _ in range(max_iter):
        if a_ub - a_lb < tol:
            break
        if f_c < f_d:
            a_ub = d
            d = c
            f_d = f_c
            c = a_ub - gr * (a_ub - a_lb)
            a_c = np.maximum((1 - c) * a + c * g, 0)
            s_c = float(np.sum(a_c))
            f_c = f(a_c / s_c) if s_c > 1e-10 else np.inf
        else:
            a_lb = c
            c = d
            f_c = f_d
            d = a_lb + gr * (a_ub - a_lb)
            a_d = np.maximum((1 - d) * a + d * g, 0)
            s_d = float(np.sum(a_d))
            f_d = f(a_d / s_d) if s_d > 1e-10 else np.inf
    
    # Evaluate at the best point
    t_best = (a_lb + a_ub) / 2
    a_best = np.maximum((1 - t_best) * a + t_best * g, 0)
    s_best = float(np.sum(a_best))
    f_best = f(a_best / s_best) if s_best > 1e-10 else np.inf
    
    return t_best, f_best


def line_search(a, g):
    """Combined line search: golden section + parabolic refinement."""
    # First, coarse grid to bracket the minimum
    ts = np.array([0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 0.99])
    objs = np.full(len(ts), np.inf)
    for i, t in enumerate(ts):
        a_try = np.maximum((1 - t) * a + t * g, 0)
        s = float(np.sum(a_try))
        if s > 1e-10:
            objs[i] = compute_obj(a_try / s)
    
    best_idx = int(np.argmin(objs))
    best_t = ts[best_idx]
    best_obj = objs[best_idx]
    
    # Golden section refinement around the best point
    if 0 < best_idx < len(ts) - 1:
        t_lo = ts[best_idx - 1]
        t_hi = ts[best_idx + 1]
    else:
        t_lo = max(0.0, best_t - 0.1)
        t_hi = min(1.0, best_t + 0.1)
    
    t_gs, obj_gs = golden_section_search(a, g, compute_obj, t_lo, t_hi, tol=1e-5, max_iter=40)
    if obj_gs < best_obj:
        best_t = t_gs
        best_obj = obj_gs
    
    return best_t, best_obj

def spectral_smoothing(a, sigma=1.0):
    """
    Smooth the spectrum of a to reduce peaks in autocorrelation.
    In frequency domain, apply a Gaussian filter to |A(omega)|^2,
    then reconstruct a (this is equivalent to circular smoothing).
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    # Compute spectrum
    size = 1
    while size < 4 * n:
        size <<= 1
    A = rfft(a, size)
    power = A.real**2 + A.imag**2
    
    # Apply Gaussian smoothing in frequency domain
    freqs = np.arange(size // 2 + 1)
    omega = 2 * np.pi * freqs / size
    gaussian = np.exp(-sigma**2 * omega**2 / 2)
    # Don't smooth DC too much
    gaussian[0] = 1.0
    power_smoothed = power * gaussian
    
    # Reconstruct: take inverse of sqrt of smoothed power (this gives a new a)
    # This is approximate - we take the real part of the inverse FFT
    # and clip to non-negative
    A_smoothed = np.sqrt(power_smoothed)
    a_new = irfft(A_smoothed, size)[:n]
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        return a
    return a_new / s_new

                        print(f"    LP: stopping after {it} iterations")
                    break

def mirror_descent_optimize(a, time_budget=300.0, tau_init=30.0, tau_end=1500.0,
                            verbose=True, blend_k=3, blend_alpha=0.2):
    """Mirror descent (multiplicative weights) with adaptive temperature."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    
    it = 0
    best_it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break
        
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        if progress < 0.7:
            tau = tau_init + (tau_end - tau_init) * progress / 0.7
        else:
            tau = tau_end
        
        conv = compute_conv(a)
        
        if it % 5 == 0:
            g = topk_subgradient(a, conv, k=blend_k)
        elif it % 5 == 3:
            g_smooth = smoothed_gradient(a, conv, tau)
            g_topk = topk_subgradient(a, conv, blend_k)
            g = blend_alpha * g_topk + (1.0 - blend_alpha) * g_smooth
        else:
            g = smoothed_gradient(a, conv, tau)
        
        g_max = float(np.max(np.abs(g)))
        if g_max < 1e-15:
            break
        eta = 2.0 / (g_max + 1e-10)
        eta *= (1.0 + 0.5 * (1.0 - progress))
        
        exponent = -eta * g
        exponent = np.clip(exponent, -50, 50)
        a_new = a * np.exp(exponent)
        s_new = float(np.sum(a_new))
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        if it % 30 == 20 and progress < 0.8:
            temp = 0.01 * (1.0 - progress)
            a_new = a_new + temp * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            a_half = a * np.exp(np.clip(-eta * 0.3 * g, -50, 50))
            s_half = float(np.sum(a_half))
            if s_half > 1e-300:
                a_half = a_half / s_half
            obj_half = compute_obj(a_half)
            if obj_half < best_obj - 1e-13:
                best_obj = obj_half
                best_a = a_half.copy()
                best_it = it
                a = a_half
                stall = 0
            elif stall >= 20:
                strength = 0.02 * (1.0 + progress)
                a_pert = a * np.exp(strength * rng.standard_normal(n))
                a_pert = np.maximum(a_pert, 0)
                s_pert = float(np.sum(a_pert))
                if s_pert > 1e-10:
                    a = a_pert / s_pert
                else:
                    a = best_a.copy()
                stall = 0
        
        a = a_new
        it += 1
        
        if verbose and (it % 100 == 0 or it == 1):
            print(f"    MD it{it}: obj={best_obj:.8f}, tau={tau:.0f}, "
                  f"stall={stall}, time={time.monotonic()-start:.0f}s")
    
    return best_a


def nesterov_gd_optimize(a, time_budget=150.0, tau_init=50.0, tau_end=1000.0,
                         verbose=False):
    """Nesterov accelerated gradient descent on smoothed max autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.85
    base_lr = 2.0 / np.sqrt(n)
    lr = base_lr
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-10))
        tau = tau_init + (tau_end - tau_init) * progress ** 1.5
        
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        conv_y = compute_conv(y)
        g = smoothed_gradient(y, conv_y, tau)
        g_norm = float(np.linalg.norm(g))
        
        if g_norm > 1e-15:
            adapt = 1.0 / (g_norm + 1e-10)
            adapt = min(max(adapt, 0.3 * base_lr), 2.0 * base_lr)
            lr = 0.8 * lr + 0.2 * adapt
        
        if prev_g is not None and it % 5 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb = ss / abs(sy)
                bb = min(max(bb, 1e-8), 4.0 / np.sqrt(n))
                lr = 0.7 * lr + 0.3 * bb
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.5
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)
        
        if it - best_it > 20000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr * 0.3
            best_it = it
    
    if verbose:
        print(f"    NesterovGD: obj={best_obj:.8f}, iters={it}")
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
