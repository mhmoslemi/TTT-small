Optimizes step function using aggressive KKT equalization, fast coordinate descent,
and broad n-exploration. Key innovation: direct KKT projection that drives the
subgradient to constant values across active coordinates, combined with a faster
CD using precomputed convolution updates.

import numpy as np
from scipy.fft import rfft, irfft
import time
import math


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def compute_subgradient(a):
    """Compute subgradient of max_k conv(a,a)[k]."""
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    # Find argmax set (with tolerance)
    tol = max_val * 1e-8 + 1e-15

        for j in range(n):
            kj = k - j
            jk = j - k
            if 0 <= kj < n:
                v[j] += a[kj]
            if 0 <= jk < n:
                v[j] += a[jk]
    v /= len(argmax)
    return v, max_val, argmax, conv

def kkt_equalize(a, time_budget=200.0, verbose=False):
    """
    Aggressive KKT equalization: directly project toward the KKT conditions.
    At optimum, subgradient should be constant across active coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(123)
    
    # We'll use a "mirroring" approach:
    # At KKT, v[i] should be constant for all i with a[i] > 0.
    # If v[i] > v[j] for active i, j, we should move mass from i to j.
    
    alpha = 0.1  # step size
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed / max(time_budget, 1) > 0.95:
            break
        
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-6 + 1e-15
        argmax = np.where(conv >= max_val - tol)[0]
        
        # Compute subgradient
        v = np.zeros(n)
        for k in argmax:
            for j in range(n):
                kj = k - j
                jk = j - k
                if 0 <= kj < n:
                    v[j] += a[kj]
                if 0 <= jk < n:
                    v[j] += a[jk]
        v /= max(len(argmax), 1)
        
        # Active set
        active = a > 1e-10
        
        if active.sum() < 2:
            break
        
        v_active = v[active]
        v_mean = float(np.mean(v_active))
        v_std = float(np.std(v_active))
        
        # The KKT violation is v_std. We want to reduce it.
        # Direction: decrease a where v is high, increase where v is low
        # d[i] = -(v[i] - v_mean) for active i, 0 for inactive
        d = np.zeros(n)
        d[active] = -(v[active] - v_mean) / (v_std + 1e-15)
        
        # Scale to maintain sum
        # We want a_new = a + alpha * d, then normalize
        # sum(a_new) = sum(a) + alpha * sum(d) = 1 + alpha * sum(d)
        # After normalization: a_new / sum(a_new)
        # For this to be a valid direction, we need sum(a + alpha*d) > 0
        sum_d = float(np.sum(d))
        if abs(sum_d) < 1e-15:
            # d already sums to ~0, good
            a_new = a + alpha * d
        else:
            # Adjust to keep sum positive
            a_new = a + alpha * (d - sum_d / n)
        
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            alpha *= 0.5
            continue
        a_new = a_new / s_new
        
        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Adaptive step: increase if improving fast
            alpha = min(alpha * 1.1, 0.5)
        else:
            stall += 1
            if stall % 5 == 0:
                alpha *= 0.7
            if stall > 100:
                # Add perturbation
                pert = rng.normal(0, 0.02, n)
                a_pert = a * np.exp(pert)
                a_pert = np.maximum(a_pert, 0)
                if a_pert.sum() > 0:
                    a_pert = a_pert / a_pert.sum()
                    obj_pert = 2.0 * n * float(np.max(compute_conv(a_pert)))
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert.copy()
                        a = a_pert
                alpha = 0.1
                stall = 0
        
        it += 1
        if verbose and it % 100 == 0:
            print(f"    KKT-EQ it{it}: obj={best_obj:.8f}, v_std={v_std:.6f}, t={elapsed:.1f}s")
    
    return best_a


def n_sweep(a_best, time_budget=120.0, verbose=False):
    """Try different sequence lengths by resampling."""
    n_best = len(a_best)
    best_a = a_best.copy()
    best_obj = 2.0 * n_best * float(np.max(compute_conv(a_best / a_best.sum())))
    
    start = time.monotonic()
    # Try various n values
    n_candidates = sorted(set([
        n_best - 200, n_best - 100, n_best - 50,
        n_best + 50, n_best + 100, n_best + 200, n_best + 400,
        400, 500, 600, 700, 900, 1000, 1100, 1200, 1400, 1600, 2000
    ]))
    n_candidates = [nc for nc in n_candidates if 200 <= nc <= 2500]
    
    for nc in n_candidates:
        if time.monotonic() - start >= time_budget:
            break
        if nc == n_best:
            continue
        
        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, nc)
        a_resampled = np.interp(x_new, x_old, a_best)
        a_resampled = np.maximum(a_resampled, 0)
        s = float(np.sum(a_resampled))
        if s < 1e-10:
            continue
        a_resampled = a_resampled / s
        
        # Quick CD
        remaining = time_budget - (time.monotonic() - start)
        cd_time = min(remaining * 0.3, 30.0)
        if cd_time > 5:
            a_resampled = coordinate_descent_fast(a_resampled, time_budget=cd_time, verbose=False)
        
        obj = 2.0 * nc * float(np.max(compute_conv(a_resampled / a_resampled.sum())))
        if obj < best_obj:
            best_obj = obj
            best_a = a_resampled.copy()
            if verbose:
                print(f"    n={nc}: obj={best_obj:.8f} **")
        elif verbose:
            print(f"    n={nc}: obj={obj:.8f}")
    
    return best_a, best_obj

Optimizes step function coefficients using fast coordinate descent, aggressive KKT
equalization, LP polishing with MOSEK, and broad n-exploration. Key innovations:
(1) Direct KKT projection driving subgradient to constant across active coords,
(2) LP polishing iterated with CD for joint refinement,

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft

import math


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming sum(a)=1. Returns 2n*max(conv)."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj(a):
    """Compute full objective for unnormalized a."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a))) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return None, None
    a_norm = a / s
    A = build_conv_matrix(a_norm)

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    g_sum = float(np.sum(g_val))
                    if g_sum > 1e-10:
                        g_val = g_val / g_sum
                        return g_val, float(np.max(A @ g_val))
            except Exception:
                continue
    except Exception:
        pass
    return None, None


def coordinate_descent(a, time_budget=100.0, verbose=False, seed=42):
    """
    Fast coordinate descent with incremental convolution updates and
    golden section 1D optimization per coordinate.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    a = a / S

    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        S_current = float(np.sum(a))
        improved = False

        for idx in range(n):
            if time.monotonic() - start >= time_budget:
                break
            i = order[idx]
            v = a[i]
            S_other = S_current - v
            if S_other < 1e-12:
                continue

            k_min = i
            k_max = min(2 * n - 2, i + n - 1)
            ks = np.arange(k_min, k_max + 1)
            a_ki = a[ks - i]
            c_k = conv[ks].copy()

            mask = (ks != 2 * i)
            ks_lin = ks[mask]
            a_ki_lin = a_ki[mask]
            c_k_lin = c_k[mask]

            k_quad = 2 * i
            c_2i = float(conv[k_quad]) if 0 <= k_quad <= 2 * n - 2 else 0.0

            def f_1d(x):
                if len(a_ki_lin) > 0:
                    lin_vals = c_k_lin + 2.0 * a_ki_lin * (x - v)
                    max_lin = float(np.max(lin_vals))
                else:
                    max_lin = 0.0
                quad_val = c_2i + x * x - v * v
                max_val = max(max_lin, quad_val)
                denom = (S_other + x) * (S_other + x)
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * max_val / denom

            # Trial candidates
            scale = max(v, S_current / n, 1e-6)
            candidates = [0.0, v * 0.25, v * 0.5, v * 0.75, v,
                          v * 1.25, v * 1.5, v * 2.0, S_current / (2 * n), S_current / n]
            best_x = v
            best_f = f_1d(v)
            for xc in candidates:
                xc = max(0.0, float(xc))
                fx = f_1d(xc)
                if fx < best_f:
                    best_f = fx
                    best_x = xc

            # Golden section refinement
            x_lo = max(0.0, best_x - 0.5 * scale)
            x_hi = best_x + scale
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = f_1d(c_pt)
            fd = f_1d(d_pt)

            for _ in range(35):
                if x_hi - x_lo < 1e-13 * max(1.0, abs(x_hi)):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = f_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = f_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-14:
                continue

            delta = x_opt - v
            a[i] = x_opt

            if 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * a_ki_lin

            S_current += delta
            improved = True

            S_new = float(np.sum(a))
            if S_new > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        # Periodic conv recompute to avoid drift
        if sweep % 8 == 0:
            conv = compute_conv(a)
            S_check = float(np.sum(a))
            if S_check > 1e-10:
                obj = 2.0 * n * float(np.max(conv)) / (S_check * S_check)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                a = a / S_check
                conv = compute_conv(a)

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a


def kkt_equalize(a, time_budget=200.0, verbose=False, seed=42):
    """
    Aggressive KKT equalization. At optimum, subgradient v[i] should be constant
    across all active coordinates. This method directly reduces the variance of v
    by moving mass from high-v to low-v coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    alpha = 0.05
    it = 0
    stall = 0
    best_v_std = np.inf

    while time.monotonic() - start < time_budget:
        if (time.monotonic() - start) / max(time_budget, 1.0) > 0.95:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-7 + 1e-15
        argmax = np.where(conv >= max_val - tol)[0]

        # Compute subgradient vectorially
        v = np.zeros(n)
        for k in argmax:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                v[lo:hi + 1] += a[k - hi:k - lo + 1]
            # Also the a[j-k] term
            lo2 = max(0, -k + 0)
            hi2 = min(n - 1, n - 1 - k + n - 1)
            # a[j-k] for j in [0,n-1]: j-k in [0,n-1] means j in [k, k+n-1]
            # intersect with [0,n-1]: j in [max(0,k), min(n-1, k+n-1)]
            j_lo = max(0, k)
            j_hi = min(n - 1, k + n - 1)
            if j_lo <= j_hi:
                v[j_lo:j_hi + 1] += a[j_lo - k:j_hi - k + 1]
        v /= max(len(argmax), 1)

        active = a > 1e-10
        n_active = int(active.sum())
        if n_active < 3:
            break

        v_act = v[active]
        v_mean = float(np.mean(v_act))
        v_std = float(np.std(v_act))

        if v_std < best_v_std:
            best_v_std = v_std

        # Direction: reduce v variance by opposing the deviation
        d = np.zeros(n)
        d[active] = -(v[active] - v_mean)
        d_norm = float(np.linalg.norm(d))
        if d_norm < 1e-15:
            break

        # Normalize direction to have unit L2
        d = d / d_norm

        # Step with adaptive alpha
        a_try = a + alpha * d
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            alpha *= 0.5
            continue
        a_try = a_try / s_try

        obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))

        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            stall = 0
            # Grow alpha if making good progress
            if v_std < 0.01:
                alpha = min(alpha * 1.05, 0.3)
            else:
                alpha = min(alpha * 1.02, 0.3)
        else:
            stall += 1
            if stall >= 10:
                alpha *= 0.6
            if stall >= 50:
                # Perturbation to escape
                pert = rng.normal(0, 0.03, n)
                a_pert = a * np.exp(pert)
                a_pert = np.maximum(a_pert, 0)
                if a_pert.sum() > 1e-10:
                    a_pert = a_pert / a_pert.sum()
                    obj_pert = 2.0 * n * float(np.max(compute_conv(a_pert)))
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert.copy()
                        a = a_pert
                alpha = 0.05
                stall = 0

        it += 1
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    KKT-EQ it{it}: obj={best_obj:.8f}, v_std={v_std:.6f}, alpha={alpha:.4f}, t={elapsed:.1f}s")

    return best_a


def lp_polish_iterate(a, time_budget=100.0, verbose=False, seed=42):
    """Iterative LP polishing: solve LP, blend, repeat."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        best_t = best_obj
        a_best_t = a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t:
                best_t = obj_try
                a_best_t = a_try.copy()

        if best_t < best_obj - 1e-13:
            best_obj = best_t
            best_a = a_best_t.copy()
            a = a_best_t
            stall = 0
        else:
            stall += 1
            if stall > 3:
                pert = rng.normal(0, 0.02, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0

        if verbose:
            elapsed = time.monotonic() - start
            print(f"    LP iter: obj={best_obj:.8f}, stall={stall}, t={elapsed:.1f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def n_sweep(a_best, time_budget=150.0, verbose=False):
    """Sweep over different sequence lengths."""
    n_best = len(a_best)
    best_a = a_best.copy()
    best_obj = 2.0 * n_best * float(np.max(compute_conv(a_best / a_best.sum())))
    start = time.monotonic()

    n_candidates = sorted(set([
        max(200, n_best - 300), max(200, n_best - 200), max(200, n_best - 100),
        max(200, n_best - 50),
        n_best + 50, n_best + 100, n_best + 200, n_best + 300, n_best + 500,
        400, 500, 600, 700, 900, 1000, 1200, 1500, 2000
    ]))
    n_candidates = [nc for nc in n_candidates if 250 <= nc <= 2500]
    # Remove duplicates that are too close
    filtered = []
    for nc in n_candidates:
        if all(abs(nc - f) > 40 for f in filtered):
            filtered.append(nc)
    n_candidates = filtered

    per_n_budget = time_budget / max(len(n_candidates), 1)

    for nc in n_candidates:
        if time.monotonic() - start >= time_budget:
            break
        if nc == n_best:
            continue

        a_resampled = resample_sequence(a_best, nc)
        s = float(np.sum(a_resampled))
        if s < 1e-10:
            continue
        a_resampled = a_resampled / s

        rem = time_budget - (time.monotonic() - start)
        cd_time = min(rem * 0.4, per_n_budget * 0.7, 40.0)
        if cd_time > 5:
            a_resampled = coordinate_descent(a_resampled, time_budget=cd_time, verbose=False, seed=nc)

        obj = 2.0 * nc * float(np.max(compute_conv(a_resampled / a_resampled.sum())))
        if obj < best_obj:
            best_obj = obj
            best_a = a_resampled.copy()
            if verbose:
                print(f"    n={nc}: obj={best_obj:.8f} **")
        elif verbose:
            print(f"    n={nc}: obj={obj:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines fast CD, KKT equalization, LP polishing, and
    broad n-exploration in an iterated framework.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    n_default = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.all(np.isfinite(parent_arr)):
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = 2.0 * parent.size * float(np.max(compute_conv(parent)))
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(n_default) / n_default
        best_score = 2.0
        print(f"Default init (n={n_default}): obj={best_score:.8f}")

    if best_seq is None or len(best_seq) == 0:
        best_seq = np.ones(800) / 800.0
        best_score = 2.0

    # === Phase 1: Initial Coordinate Descent ===
    print(f"\n=== Phase 1: Initial CD ===")
    remaining = deadline - time.monotonic()
    p1_time = min(remaining * 0.3, 400.0)
    if p1_time > 30:
        a_cd = coordinate_descent(best_seq, time_budget=p1_time, verbose=True, seed=seed)
        obj = 2.0 * a_cd.size * float(np.max(compute_conv(a_cd / a_cd.sum())))
        if obj < best_score:
            best_score = obj
            best_seq = a_cd.copy()
            print(f"  CD result: {best_score:.8f} **")
        else:
            print(f"  CD result: {obj:.8f}")

    # === Phase 2: KKT Equalization ===
    print(f"\n=== Phase 2: KKT Equalization ===")
    remaining = deadline - time.monotonic()
    p2_time = min(remaining * 0.15, 200.0)
    if p2_time > 30:
        a_kkt = kkt_equalize(best_seq, time_budget=p2_time, verbose=True, seed=seed + 1)
        obj = 2.0 * a_kkt.size * float(np.max(compute_conv(a_kkt / a_kkt.sum())))
        if obj < best_score:
            best_score = obj
            best_seq = a_kkt.copy()
            print(f"  KKT-EQ: {best_score:.8f} **")
        else:
            print(f"  KKT-EQ: {obj:.8f}")
            # CD from KKT result
            if deadline - time.monotonic() > 40:
                a_cd2 = coordinate_descent(a_kkt, time_budget=min(50, (deadline - time.monotonic()) * 0.15), seed=seed + 2)
                obj2 = 2.0 * a_cd2.size * float(np.max(compute_conv(a_cd2 / a_cd2.sum())))
                if obj2 < best_score:
                    best_score = obj2
                    best_seq = a_cd2.copy()
                    print(f"  CD post-KKT: {best_score:.8f} **")

    # === Phase 3: LP Polishing ===
    print(f"\n=== Phase 3: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p3_time = min(remaining * 0.2, 250.0)
    if p3_time > 20:
        a_lp = lp_polish_iterate(best_seq, time_budget=p3_time, verbose=True, seed=seed + 10)
        obj = 2.0 * a_lp.size * float(np.max(compute_conv(a_lp / a_lp.sum())))
        if obj < best_score:
            best_score = obj
            best_seq = a_lp.copy()
            print(f"  LP result: {best_score:.8f} **")
        else:
            print(f"  LP result: {obj:.8f}")
        # CD after LP
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_cd3 = coordinate_descent(best_seq, time_budget=min(40, remaining * 0.15), seed=seed + 20)
            obj3 = 2.0 * a_cd3.size * float(np.max(compute_conv(a_cd3 / a_cd3.sum())))
            if obj3 < best_score:
                best_score = obj3
                best_seq = a_cd3.copy()
                print(f"  CD post-LP: {best_score:.8f} **")

    # === Phase 4: n-Sweep ===
    print(f"\n=== Phase 4: n-Sweep ===")
    remaining = deadline - time.monotonic()
    p4_time = min(remaining * 0.2, 250.0)
    if p4_time > 40:
        best_seq, best_score = n_sweep(best_seq, time_budget=p4_time, verbose=True)
        print(f"  Best after n-sweep: n={len(best_seq)}, obj={best_score:.8f}")

    # === Phase 5: Second refinement cycle (CD + KKT + CD) ===
    print(f"\n=== Phase 5: Refinement Cycle ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        # CD
        cd_t = min(remaining * 0.3, 150.0)
        if cd_t > 20:
            a_r = coordinate_descent(best_seq, time_budget=cd_t, verbose=True, seed=seed + 100)
            obj = 2.0 * a_r.size * float(np.max(compute_conv(a_r / a_r.sum())))
            if obj < best_score:
                best_score = obj
                best_seq = a_r.copy()
                print(f"  CD: {best_score:.8f} **")

        # KKT
        remaining = deadline - time.monotonic()
        kkt_t = min(remaining * 0.25, 120.0)
        if kkt_t > 20:
            a_r2 = kkt_equalize(best_seq, time_budget=kkt_t, verbose=False, seed=seed + 200)
            obj = 2.0 * a_r2.size * float(np.max(compute_conv(a_r2 / a_r2.sum())))
            if obj < best_score:
                best_score = obj
                best_seq = a_r2.copy()
                print(f"  KKT-EQ: {best_score:.8f} **")

        # Final CD
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_r3 = coordinate_descent(best_seq, time_budget=min(remaining * 0.4, 80.0), seed=seed + 300)
            obj = 2.0 * a_r3.size * float(np.max(compute_conv(a_r3 / a_r3.sum())))
            if obj < best_score:
                best_score = obj
                best_seq = a_r3.copy()
                print(f"  Final CD: {best_score:.8f} **")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    restart_modes = [
        "gaussian_perturb", "exponential_perturb", "two_bump_blend", "edge_reduction"
    ]
    for r, mode in enumerate(restart_modes):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if mode == "gaussian_perturb":
            pert = rng.normal(0, 0.05, n_cur)
            a_r = best_seq * np.exp(pert)
        elif mode == "exponential_perturb":
            pert = rng.exponential(0.03, n_cur) - 0.03
            a_r = best_seq * np.exp(pert)
        elif mode == "two_bump_blend":
            x = np.linspace(0, 1, n_cur)
            bump1 = np.exp(-((x - 0.3) ** 2) / 0.04)
            bump2 = np.exp(-((x - 0.7) ** 2) / 0.04)
            blend = (bump1 + bump2) / (bump1 + bump2 + 1)
            a_r = (1 - 0.4 * blend) * best_seq
        else:  # edge_reduction
            a_r = best_seq.copy()
            edge_size = n_cur // 8
            a_r[:edge_size] *= 0.5
            a_r[-edge_size:] *= 0.5
            # Boost center
            center_size = n_cur // 4
            mid = n_cur // 2
            a_r[mid - center_size:mid + center_size] *= 1.2

        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        alloc = min(remaining * 0.25, 50.0)
        if alloc > 15:
            a_r = coordinate_descent(a_r, time_budget=alloc, verbose=False, seed=seed + 500 + r * 50)
            obj = 2.0 * a_r.size * float(np.max(compute_conv(a_r / a_r.sum())))
            if obj < best_score:
                best_score = obj
                best_seq = a_r.copy()
                print(f"  Restart {mode}: {best_score:.8f} **")
            else:
                print(f"  Restart {mode}: {obj:.8f}")

    # === Phase 7: Final n-sweep + polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        # Quick n-sweep
        ns_time = min(remaining * 0.3, 100.0)
        if ns_time > 30:
            best_seq, best_score = n_sweep(best_seq, time_budget=ns_time, verbose=True)
            print(f"  After final n-sweep: n={len(best_seq)}, obj={best_score:.8f}")

        # Final CD
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_fin = coordinate_descent(best_seq, time_budget=min(remaining * 0.7, 60.0), seed=seed + 999)
            obj = 2.0 * a_fin.size * float(np.max(compute_conv(a_fin / a_fin.sum())))
            if obj < best_score:
                best_score = obj
                best_seq = a_fin.copy()
                print(f"  Final final CD: {best_score:.8f} **")

    # Convert to list and validate
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    result = [float(x) for x in best_seq]

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
