"""
Multi-phase optimizer for step function height sequences.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Symmetric soft-max gradient descent (FFT-accelerated)
2. Exact subgradient with adaptive Polyak step size
3. Aggressive LP polishing with symmetry
4. 1D coordinate descent with exact line search
5. Length exploration
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
import math


def compute_conv(a):
    """Full linear convolution a * a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def symmetrize(a):
    """Enforce a[i] = a[n-1-i] by averaging with reverse."""
    return 0.5 * (a + a[::-1])


def compute_soft_gradient(a, p):
    """Compute gradient: 2 * cross_corr(a_rev, p)[n-1:n+n]."""
    n = len(a)
    p_len = len(p)
    a_rev = a[::-1]
    size = 1
    while size < n + p_len - 1:
        size <<= 1
    A = rfft(a_rev, size)
    P = rfft(p, size)
    full = irfft(A * P, size)
    q = full[n - 1:n - 1 + n]
    return 2.0 * q


def get_subgradient(a):
    """
    Compute a subgradient of max_k conv(a,a)_k w.r.t. a.
    Returns (grad, active_indices) where grad[i] = mean of a[k-i] over active k.
    """
    n = len(a)
    conv = compute_conv(a)
    c_max = float(np.max(conv))
    # Find active indices (within numerical tolerance)
    tol = 1e-10 * c_max
    active = np.where(conv >= c_max - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        # d(conv_k)/d(a_i) = a[k-i] if 0 <= k-i < n
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += a[j]
    grad /= len(active)
    return grad, active, c_max


def symm_smax_gd(a, time_budget=200.0, beta_start=10.0, beta_end=60.0,
                 lr_start=0.008, momentum=0.9, verbose=True, seed=42):
    """Symmetric soft-max gradient descent with momentum."""
    n = len(a)
    a = project_simplex(a.astype(np.float64))
    a = symmetrize(a)
    a = project_simplex(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    lr = lr_start
    it = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress
        lr = lr_start * (1.0 - 0.3 * progress)
        
        conv = compute_conv(a)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p = exp_vals / float(np.sum(exp_vals))
        
        grad = compute_soft_gradient(a, p)
        
        # Symmetrize gradient (for symmetric optimization)
        grad_sym = 0.5 * (grad + grad[::-1])
        
        v = momentum * v + grad_sym
        a_new = a - lr * v
        a_new = symmetrize(a_new)
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall > 50:
                v = np.zeros(n)
                stall = 0
                noise = rng.normal(0, 0.001, n)
                noise = symmetrize(noise)
                a = project_simplex(best_a + noise)
        
        a = a_new
        it += 1
        
        if verbose and it % 300 == 0:
            print(f"    SmaxGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, t={elapsed:.1f}s")
        
        if it - best_it > 3000 and it > 500:
            a = best_a.copy()
            v = np.zeros(n)
            it = best_it
    
    return best_a, best_obj


def subgradient_method(a, time_budget=100.0, verbose=True, seed=42):

    a = project_simplex(a.astype(np.float64))
    a = symmetrize(a)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a[:n]
        # Wait, conv(a, g)_k = sum_j a_j * g_{k-j}
        # So A[k, j] = a_j if 0 <= k-j < n, i.e., j <= k < j+n
        # A[k, j] = a[k-j] ... no
        # A[k, j] such that A @ g[k] = sum_j A[k,j] * g[j] = sum_j a[k-j] * g[j]
        # So A[k, j] = a[k-j] if 0 <= k-j < n
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[k, j] = a[k - j]
    return A


def build_conv_matrix_fast(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g) using toeplitz."""
    n = len(a)
    # conv(a, g)_k = sum_j a[k-j] * g[j] for valid j
    # This is a Toeplitz matrix
    from scipy.linalg import toeplitz
    # First column: a[0], a[1], ..., a[n-1], 0, ..., 0 (length 2n-1)
    # First row: a[0], 0, 0, ..., 0
    # Actually, let's just use the direct construction
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # a_j contributes to conv[k] for k = j, j+1, ..., j+n-1
        # conv[k] += a_j * g[k-j] ... wait no
        # conv(a,g)[k] = sum_i a[i] * g[k-i]
        # So for fixed j (index of g), g[j] appears in conv[k] for k = j to j+n-1
        # conv[k] = sum_i a[i] * g[k-i], so g[j] = g[k-i] means i = k-j
        # coeff of g[j] in conv[k] is a[k-j] (if 0 <= k-j < n)
        for k in range(j, min(j + n, 2 * n - 1)):
            A[k, j] = a[k - j]
    return A


def solve_lp_symmetric(a):
    """
    Solve LP for symmetric case:
    min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0, g[i]=g[n-1-i]
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    # Symmetric: g[i] = g[n-1-i]
    # Variables: g[0], ..., g[(n-1)//2]
    n_sym = (n + 1) // 2
    
    # Build conv matrix for symmetric case
    # conv(a, g_sym)_k where g[j] = g_sym[min(j, n-1-j)]
    A_full = build_conv_matrix_fast(a_norm)
    
    # Map: for column j of A_full, the variable is g_sym[min(j, n-1-j)]
    # So A_sym[k, s] = sum of A_full[k, j] for all j where min(j, n-1-j) = s
    A_sym = np.zeros((2 * n - 1, n_sym))
    for j in range(n):
        s = min(j, n - 1 - j)
        A_sym[:, s] += A_full[:, j]
    
    g_sym = cp.Variable(n_sym, nonneg=True)
    M = cp.Variable()
    
    # Sum constraint: sum(g) = sum over all j of g_sym[min(j, n-1-j)]
    # For j = 0: g_sym[0]
    # For j = n-1: g_sym[0] (same)
    # For j = 1: g_sym[1]
    # For j = n-2: g_sym[1] (same)
    # ...
    # For middle (if n odd): g_sym[n_sym-1] (only once)
    sum_coeff = np.zeros(n_sym)
    for j in range(n):
        s = min(j, n - 1 - j)
        sum_coeff[s] += 1
    
    constraints = [A_sym @ g_sym <= M, sum_coeff @ g_sym == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g_sym.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_sym_val = np.maximum(g_sym.value, 0)
                # Unfold to full length
                g_full = np.zeros(n)
                for j in range(n):
                    g_full[j] = g_sym_val[min(j, n - 1 - j)]
                s = float(np.sum(g_full))
                if s > 1e-10:
                    return g_full / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_full(a):
    """Solve LP without symmetry constraint."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix_fast(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_symmetric(a, g, max_t=1.5, time_budget=2.0):
    """Golden section + grid search for blend factor t in a_new = (1-t)*a + t*g."""
    def obj_at(t):
        if t < 0:
            t = 0
        a_try = (1 - t) * a + t * g
        a_try = symmetrize(a_try)
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)
    
    # Grid search first
    best_t = 0.0
    best_obj = obj_at(0.0)
    for t in np.arange(0.05, 1.5, 0.05):
        o = obj_at(t)
        if o < best_obj:
            best_obj = o
            best_t = t
    
    # Refine with golden section around best_t
    lo = max(0, best_t - 0.1)
    hi = best_t + 0.1
    if hi < 0:
        return 0.0, best_obj
    
    phi = (1 + np.sqrt(5)) / 2
    gr = 2 - phi
    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)
    
    start = time.monotonic()
    for _ in range(25):
        if time.monotonic() - start >= time_budget:
            break
        if hi - lo < 1e-6:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)
    
    t_best = c if fc < fd else d
    obj_best = min(fc, fd)
    if obj_best < best_obj:
        best_obj = obj_best
        best_t = t_best
    
    return best_t, best_obj

"""
Multi-phase optimizer for step function height sequences minimizing 2*n*max(conv(a,a))/(sum(a))^2.
New approaches vs. previous: (1) symmetry enforcement throughout, (2) mirror descent phase,
(3) faster LP matrix construction with more iterations, (4) pairwise transfer local search,
(5) adaptive time allocation based on observed progress.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
import math


def compute_conv(a):
    """Full linear convolution a * a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def symmetrize(a):
    """Enforce a[i] = a[n-1-i] by averaging with reverse."""
    return 0.5 * (a + a[::-1])


def compute_soft_gradient(a, p):
    """Compute gradient vector: 2 * cross-correlation of a_rev and p, extracted to length n."""
    n = len(a)
    p_len = len(p)
    a_rev = a[::-1]
    size = 1
    while size < n + p_len - 1:
        size <<= 1
    A = rfft(a_rev, size)
    P = rfft(p, size)
    full = irfft(A * P, size)
    q = full[n - 1:n - 1 + n]
    return 2.0 * q


def get_exact_subgradient(a):
    """
    Compute exact subgradient of max_k conv(a,a)_k w.r.t. a.
    Returns (grad_vector, list_of_active_indices, max_value).
    """
    n = len(a)
    conv = compute_conv(a)
    c_max = float(np.max(conv))
    tol = max(1e-12, 1e-9 * c_max)
    active = np.where(conv >= c_max - tol)[0]

    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    grad /= float(len(active))
    return grad, active.tolist(), c_max


def build_conv_matrix_fast(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    return A


def smax_gd_sym(a, time_budget=200.0, beta_start=10.0, beta_end=60.0,
                lr_start=0.008, momentum=0.9, verbose=True, seed=42):
    """Symmetric soft-max gradient descent with Nesterov momentum."""
    n = len(a)
    a = symmetrize(a.astype(np.float64))
    a = project_simplex(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    it = 0
    stall = 0
    best_it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress
        lr = lr_start * (1.0 - 0.4 * progress)

        conv = compute_conv(a)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p = exp_vals / float(np.sum(exp_vals))

        grad = compute_soft_gradient(a, p)
        grad_sym = 0.5 * (grad + grad[::-1])

        v = momentum * v + grad_sym
        a_new = a - lr * v
        a_new = symmetrize(a_new)
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall = 0
        else:
            stall += 1
            if stall > 80:
                v = np.zeros(n)
                stall = 0
                noise = rng.normal(0, 0.001, n)
                a = project_simplex(best_a + symmetrize(noise))

        a = a_new
        it += 1

        if verbose and it % 400 == 0:
            print(f"    SmaxGD it{it}: obj={best_obj:.8f}, beta={beta:.1f}, t={elapsed:.1f}s")

        if it - best_it > 4000 and it > 1000:
            a = best_a.copy()
            v = np.zeros(n)
            it = best_it

    return best_a, best_obj


def mirror_descent_sym(a, time_budget=100.0, lr_start=0.01, beta=30.0,
                       verbose=True, seed=42):
    """Mirror descent (exponential parameterization) with symmetry."""
    n = len(a)
    a = symmetrize(a.astype(np.float64))
    a = project_simplex(a)
    theta = np.log(a + 1e-15)
    theta_sym = 0.5 * (theta + theta[::-1])

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        lr = lr_start * (1.0 - 0.5 * progress)

        a = np.exp(theta_sym - theta_sym.max())
        a /= a.sum()

        conv = compute_conv(a)
        c_max = float(np.max(conv))
        exp_vals = np.exp(beta * (conv - c_max))
        p = exp_vals / float(np.sum(exp_vals))

        grad_a = compute_soft_gradient(a, p)
        grad_a_sym = 0.5 * (grad_a + grad_a[::-1])

        # Gradient w.r.t. theta for softmax parameterization
        grad_weighted = float(np.sum(a * grad_a_sym))
        grad_theta = a * (grad_a_sym - grad_weighted)

        theta_sym -= lr * grad_theta
        theta_sym = 0.5 * (theta_sym + theta_sym[::-1])

        a_new = np.exp(theta_sym - theta_sym.max())
        a_new /= a_new.sum()
        a_new = symmetrize(a_new)
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        theta_sym = np.log(a + 1e-15)
        theta_sym = 0.5 * (theta_sym + theta_sym[::-1])
        it += 1

        if verbose and it % 200 == 0:
            print(f"    Mirror it{it}: obj={best_obj:.8f}, lr={lr:.5f}, t={elapsed:.1f}s")

    return best_a, best_obj


def solve_lp_symmetric(a):
    """Solve LP with symmetry: min M s.t. conv(a,g)[k]<=M, sum(g)=1, g>=0, g symmetric."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A_full = build_conv_matrix_fast(a_norm)

    n_sym = (n + 1) // 2
    A_sym = np.zeros((2 * n - 1, n_sym))
    sum_coeff = np.zeros(n_sym)
    for j in range(n):
        s = min(j, n - 1 - j)
        A_sym[:, s] += A_full[:, j]
        sum_coeff[s] += 1.0

    g_sym = cp.Variable(n_sym, nonneg=True)
    M = cp.Variable()
    constraints = [A_sym @ g_sym <= M, sum_coeff @ g_sym == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g_sym.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_sym_val = np.maximum(g_sym.value, 0)
                g_full = np.zeros(n)
                for j in range(n):
                    g_full[j] = g_sym_val[min(j, n - 1 - j)]
                s = float(np.sum(g_full))
                if s > 1e-10:
                    return g_full / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_full(a):
    """Solve LP without symmetry constraint."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix_fast(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g, time_budget=2.0):
    """Find optimal t for a_new = (1-t)*a + t*g, with symmetrization."""
    def obj_at(t):
        if t < 0:
            t = 0.0
        a_try = (1.0 - t) * a + t * g
        a_try = symmetrize(a_try)
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            return np.inf
        return compute_obj(a_try / s)

    best_t = 0.0
    best_obj = obj_at(0.0)

    for t in np.arange(0.05, 1.55, 0.05):
        o = obj_at(t)
        if o < best_obj:
            best_obj = o
            best_t = t

    # Golden section refinement
    lo = max(0.0, best_t - 0.1)
    hi = best_t + 0.1
    if hi <= lo:
        lo, hi = 0.0, 0.5

    phi = (1.0 + np.sqrt(5.0)) / 2.0
    gr = 2.0 - phi
    c = lo + gr * (hi - lo)
    d = lo + (1.0 - gr) * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    start = time.monotonic()
    for _ in range(30):
        if time.monotonic() - start >= time_budget:
            break
        if hi - lo < 1e-6:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)

    t_best = c if fc < fd else d
    obj_best = min(fc, fd)
    if obj_best < best_obj:
        best_obj = obj_best
        best_t = t_best

    return best_t, best_obj


def lp_polish_sym(a, time_budget=180.0, max_iters=150, verbose=False):
    """Aggressive LP polishing with symmetry enforcement."""
    n = len(a)
    a = symmetrize(a.astype(np.float64))
    a = project_simplex(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Alternate between symmetric and full LP
        if it % 5 < 3:
            g_lp, M_lp = solve_lp_symmetric(a)
        else:
            g_lp, M_lp = solve_lp_full(a)

        if g_lp is None:
            stall += 1
            it += 1
            if stall > 10:
                break
            continue

        g_lp = symmetrize(g_lp)
        g_lp = project_simplex(g_lp)

        t_opt, obj_opt = line_search_blend(a, g_lp, time_budget=min(1.5, remaining * 0.03))

        if t_opt > 0.01 and obj_opt < best_obj - 1e-10:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp
            a_new = symmetrize(a_new)
            a_new = project_simplex(np.maximum(a_new, 0))
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP-Sym it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                  f"M={M_lp:.6f}, {elapsed:.1f}s/{time_budget:.0f}s")

        if stall >= 6:
            noise = rng.normal(0, 0.002 * stall, n)
            a = project_simplex(best_a + symmetrize(noise))
            stall = 0

    return best_a, best_obj


def pairwise_transfer(a, time_budget=60.0, verbose=False):
    """
    Pairwise mass transfer local search with symmetry.
    Try moving mass between pairs of symmetric positions.
    """
    n = len(a)
    a = symmetrize(a.astype(np.float64))
    a = project_simplex(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    n_half = (n + 1) // 2

    it = 0
    improving_any = True
    while improving_any and time.monotonic() - start < time_budget:
        improving_any = False
        it += 1

        for i in range(n_half):
            if time.monotonic() - start >= time_budget:
                break
            for j in range(max(0, i - 20), min(n_half, i + 21)):
                if j == i:
                    continue
                for delta in [0.001, 0.003, 0.005, 0.01, -0.001, -0.003, -0.005, -0.01]:
                    a_try = a.copy()
                    a_try[i] += delta
                    a_try[j] -= delta
                    if i != n - 1 - i:
                        a_try[n - 1 - i] += delta
                    if j != n - 1 - j:
                        a_try[n - 1 - j] -= delta

                    if np.any(a_try < -1e-10):
                        continue
                    a_try = np.maximum(a_try, 0)
                    s = float(np.sum(a_try))
                    if s < 0.01:
                        continue
                    a_try = a_try / s
                    obj_try = compute_obj(a_try)

                    if obj_try < best_obj - 1e-11:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improving_any = True

        if verbose:
            print(f"    Pairwise round{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

    return best_a, best_obj


def subgradient_sym(a, time_budget=80.0, verbose=True, seed=42):
    """Exact subgradient method with adaptive step size and symmetrization."""
    n = len(a)
    a = symmetrize(a.astype(np.float64))
    a = project_simplex(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Adaptive step size
    t = 0.05
    no_improve = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed > time_budget * 0.7:
            t *= 0.8

        grad, active, c_max = get_exact_subgradient(a)
        grad_sym = 0.5 * (grad + grad[::-1])

        g_norm_sq = float(np.sum(grad_sym ** 2))
        if g_norm_sq < 1e-20:
            break

        a_new = a - t * grad_sym
        a_new = symmetrize(a_new)
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            t = min(t * 1.1, 0.1)
        else:
            no_improve += 1
            t *= 0.7
            if no_improve > 30:
                noise = rng.normal(0, 0.002, n)
                a = project_simplex(best_a + symmetrize(noise))
                no_improve = 0
                t = 0.03

        if verbose and int(elapsed * 2) % 5 == 0 and int(elapsed * 2) > 0:
            print(f"    SubGrad: obj={best_obj:.8f}, t={t:.5f}, "
                  f"|active|={len(active)}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def arcsine_init_sym(n):

    a = 1.0 / np.sqrt(1.0 - (2.0 * x - 1.0) ** 2 + 1e-10)

def u_shaped_init_sym(n, sharp=3.0):
    """Symmetric U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharp * (2.0 * x - 1.0) ** 2)
    a = symmetrize(a)
    return project_simplex(a)


def diverse_sym_init(n, mode, rng):
    """Diverse symmetric initialization."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = (2 * x - 1) ** 4 + 0.05
    elif mode == 1:
        a = np.exp(3.0 * (2 * x - 1) ** 2)
    elif mode == 2:
        a = 1.0 / np.cosh(2.0 * (2 * x - 1))
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        a = np.convolve(a, np.ones(9) / 9.0, mode='same')
    elif mode == 4:
        a = 2.0 - np.cos(np.pi * (2 * x - 1)) ** 4
    elif mode == 5:
        a = np.exp(-4.0 * (x - 0.1) ** 2) + np.exp(-4.0 * (x - 0.9) ** 2) + 0.02
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    a = symmetrize(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase:
    1. Initialization exploration
    2. Symmetric soft-max GD
    3. Subgradient method
    4. Mirror descent
    5. Aggressive LP polishing
    6. Pairwise transfer local search
    7. Length exploration
    8. Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_current = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                parent = symmetrize(parent)
                parent = project_simplex(parent)
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_current = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = arcsine_init_sym(n_current)
        best_score = compute_obj(best_seq)
        print(f"Default arcsine init (n={n_current}): obj={best_score:.8f}")

    # Try alternative initializations
    for init_name, init_fn in [("U-shape", lambda: u_shaped_init_sym(n_current, 3.0)),
                               ("Arcsine", lambda: arcsine_init_sym(n_current))]:
        a_try = init_fn()
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"{init_name} init: obj={best_score:.8f} **")

    # === Phase 1: Soft-max GD (smooth phase) ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 1: SmaxGD sym (beta 5->25), budget={remaining:.0f}s ===")
        alloc = min(remaining * 0.18, 200.0)
        a1, obj1 = smax_gd_sym(best_seq, time_budget=alloc,
                               beta_start=5.0, beta_end=25.0,
                               lr_start=0.01, momentum=0.9,
                               verbose=True, seed=seed)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase1: obj={best_score:.8f} **")
        else:
            print(f"  Phase1: obj={obj1:.8f}")

    # === Phase 2: SmaxGD (sharper) ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 2: SmaxGD sym (beta 25->70), budget={remaining:.0f}s ===")
        alloc = min(remaining * 0.15, 180.0)
        a2, obj2 = smax_gd_sym(best_seq, time_budget=alloc,
                               beta_start=25.0, beta_end=70.0,
                               lr_start=0.005, momentum=0.92,
                               verbose=True, seed=seed + 1)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase2: obj={best_score:.8f} **")
        else:
            print(f"  Phase2: obj={obj2:.8f}")

    # === Phase 3: Subgradient method ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 3: Exact Subgradient, budget={remaining:.0f}s ===")
        alloc = min(remaining * 0.10, 100.0)
        a3, obj3 = subgradient_sym(best_seq, time_budget=alloc,
                                   verbose=True, seed=seed + 2)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase3: obj={best_score:.8f} **")
        else:
            print(f"  Phase3: obj={obj3:.8f}")

    # === Phase 4: Mirror descent ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: Mirror Descent, budget={remaining:.0f}s ===")
        alloc = min(remaining * 0.08, 80.0)
        a4, obj4 = mirror_descent_sym(best_seq, time_budget=alloc,
                                      lr_start=0.01, beta=40.0,
                                      verbose=True, seed=seed + 3)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase4: obj={best_score:.8f} **")
        else:
            print(f"  Phase4: obj={obj4:.8f}")

    # === Phase 5: Aggressive LP polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 5: LP Polish (aggressive), budget={remaining:.0f}s ===")
        alloc = min(remaining * 0.25, 300.0)
        a5, obj5 = lp_polish_sym(best_seq, time_budget=alloc,
                                 max_iters=200, verbose=True)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  Phase5: obj={best_score:.8f} **")
        else:
            print(f"  Phase5: obj={obj5:.8f}")

    # === Phase 6: Pairwise transfer ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 6: Pairwise Transfer, budget={remaining:.0f}s ===")
        alloc = min(remaining * 0.10, 80.0)
        a6, obj6 = pairwise_transfer(best_seq, time_budget=alloc, verbose=True)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Phase6: obj={best_score:.8f} **")
        else:
            print(f"  Phase6: obj={obj6:.8f}")

    # === Phase 7: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 7: Diverse Restarts, budget={remaining:.0f}s ===")
        n_cur = len(best_seq)
        for r in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            mode = r % 5
            a_r = diverse_sym_init(n_cur, mode, rng)
            blend = 0.5
            a_r = project_simplex((1 - blend) * a_r + blend * best_seq)

            alloc_gd = min(remaining * 0.3, 50.0)
            if alloc_gd > 10:
                a_r, obj_r = smax_gd_sym(a_r, time_budget=alloc_gd,
                                         beta_start=10.0, beta_end=50.0,
                                         lr_start=0.008, momentum=0.9,
                                         verbose=False, seed=seed + 100 + r)

            remaining = deadline - time.monotonic()
            if remaining > 30:
                alloc_lp = min(remaining * 0.3, 40.0)
                a_r, obj_r = lp_polish_sym(a_r, time_budget=alloc_lp,
                                           max_iters=30, verbose=False)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Length exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 8: Length Exploration, budget={remaining:.0f}s ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            n_best - 100, n_best - 50, n_best + 50, n_best + 100,
            n_best + 200, 600, 700, 900, 1000, 1200
        ]))
        n_options = [n for n in n_options if 400 <= n <= 1600 and abs(n - n_best) >= 30]
        n_options = n_options[:3]
        print(f"  Trying n values: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break

            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = symmetrize(a_try)
            a_try = project_simplex(np.maximum(a_try, 0))

            alloc_gd = min(remaining * 0.35, 40.0)
            if alloc_gd > 10:
                a_try, obj_try = smax_gd_sym(a_try, time_budget=alloc_gd,
                                             beta_start=10.0, beta_end=40.0,
                                             lr_start=0.008, momentum=0.9,
                                             verbose=False, seed=seed + 200)

            remaining = deadline - time.monotonic()
            if remaining > 25:
                alloc_lp = min(remaining * 0.4, 30.0)
                a_try, obj_try = lp_polish_sym(a_try, time_budget=alloc_lp,
                                               max_iters=25, verbose=False)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 9: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 9: Final Polish, budget={remaining:.0f}s ===")
        # One more LP round
        alloc_lp = min(remaining * 0.5, 150.0)
        if alloc_lp > 20:
            a_f, obj_f = lp_polish_sym(best_seq, time_budget=alloc_lp,
                                       max_iters=100, verbose=True)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        # Final pairwise
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc_pw = min(remaining * 0.4, 40.0)
            a_f2, obj_f2 = pairwise_transfer(best_seq, time_budget=alloc_pw,
                                             verbose=False)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final PW: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
