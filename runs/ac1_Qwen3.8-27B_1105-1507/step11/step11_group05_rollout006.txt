OK let me step back and think about this more carefully. The LP approach is the strongest. Let me focus on making it work well with full constraints.

The LP is:
- Variables: g_0, ..., g_{n-1}, M (n+1 variables)
- Minimize M
- Subject to:
  - For each k in 0, ..., 2n-2: sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[j] * g[k-j] <= M
  - sum(g) = 1
  - g >= 0

The constraint matrix A has shape (2n-1) x n, where:
A[k, j] = a[j] if 0 <= k-j < n (i.e., k-n+1 <= j <= k), else 0

Wait, let me be more careful. (a*g)[k] = sum_j a[j] * g[k-j] where we need 0 <= j < n and 0 <= k-j < n.
So j must satisfy: max(0, k-n+1) <= j <= min(n-1, k).
And the coefficient of g[m] (where m = k-j) in constraint k is a[k-m] (where 0 <= k-m < n).

So A[k, m] = a[k-m] if 0 <= k-m < n, else 0.
Equivalently: A[k, m] = a[k-m] if max(0, k-n+1) <= m <= min(n-1, k), else 0.

For n=800, this is a 1599 x 800 matrix. Very manageable.

Let me also think about the blending step more carefully. After solving the LP, we get g. We want to find t that minimizes:
f((1-t)*a + t*g) = 2n * max_k (((1-t)*a + t*g) * ((1-t)*a + t*g))[k]

where a and g are both normalized (sum = 1).

(((1-t)*a + t*g) * ((1-t)*a + t*g))[k]
= (1-t)^2 (a*a)[k] + 2t(1-t) (a*g)[k] + t^2 (g*g)[k]

We know (a*a) and (a*g) from the LP. We can also compute (g*g). So we can evaluate this for any t in O(n) (just compute the max over k).

The optimal t is the one that minimizes max_k [(1-t)^2 A_k + 2t(1-t) B_k + t^2 C_k]
where A_k = (a*a)[k], B_k = (a*g)[k], C_k = (g*g)[k].

This is a 1D optimization problem. We can solve it with golden section search or just evaluate at many t values.

Actually, since the LP already ensures B_k <= M for all k, and A_k <= M (since M was the max of a*a before), the best t might be close to 1 (i.e., just use g). But the C_k terms might be large, so t < 1 might be better.

"""
Full-constraint LP optimization for step function coefficients.
Uses all 2n-1 convolution constraints in the LP for maximum quality solutions.
Combined with FISTA refinement and systematic length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):
    """Cross convolution a*b via FFT."""
    n1, n2 = len(a), len(b)
    if n1 == 0 or n2 == 0:
        return np.array([])
    size = 1
    while size < n1 + n2 - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n1 + n2 - 1]

def compute_obj_conv(a_norm):
    """Compute 2n*max(conv) for normalized a (sum=1)."""
    n = len(a_norm)
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def compute_obj(a):
    """Compute objective for unnormalized a."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return compute_obj_conv(a / s)

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n

def build_full_lp_matrix(a_norm):
    """Build the full (2n-1) x n constraint matrix for the LP."""
    n = len(a_norm)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            m = k - j  # index in g
            A[k, m] = a_norm[j]
    return A

def solve_full_lp(a_norm, solver_list=None):
    """Solve the full LP: min M s.t. A@w <= M, sum(w)=1, w>=0."""
    n = len(a_norm)
    A = build_full_lp_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:
        solver_list = [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception as e:
            continue
    return None, np.inf

def find_best_blend(a, g, conv_aa, conv_ag, conv_gg, num_points=50):
    """Find optimal blend t minimizing max of the blended autocorrelation."""
    t_vals = np.linspace(0, 1.0, num_points)
    best_t = 0.0
    best_val = np.inf
    
    for t in t_vals:
        blended = (1-t)**2 * conv_aa + 2*t*(1-t) * conv_ag + t**2 * conv_gg
        val = np.max(blended)
        if val < best_val:
            best_val = val
            best_t = t
    
    # Refine with golden section
    if 0 < best_t < 1:
        a_lo = max(0, best_t - 0.02)
        a_hi = min(1, best_t + 0.02)
        gr = (np.sqrt(5) - 1) / 2
        x1 = a_hi - gr * (a_hi - a_lo)
        x2 = a_hi - (1-gr) * (a_hi - a_lo)
        f1 = np.max((1-x1)**2 * conv_aa + 2*x1*(1-x1) * conv_ag + x1**2 * conv_gg)
        f2 = np.max((1-x2)**2 * conv_aa + 2*x2*(1-x2) * conv_ag + x2**2 * conv_gg)
        for _ in range(30):
            if f1 < f2:
                a_hi = x2; x2 = x1; f2 = f1
                x1 = a_hi - gr * (a_hi - a_lo)
                f1 = np.max((1-x1)**2 * conv_aa + 2*x1*(1-x1) * conv_ag + x1**2 * conv_gg)
            else:
                a_lo = x1; x1 = x2; f1 = f2
                x2 = a_hi - (1-gr) * (a_hi - a_lo)
                f2 = np.max((1-x2)**2 * conv_aa + 2*x2*(1-x2) * conv_ag + x2**2 * conv_gg)
        best_t = (a_lo + a_hi) / 2
    
    return best_t

def lp_iteration(a, solver_list=None):
    """One LP iteration: solve LP, find best blend, return new a."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    
    g, M_val = solve_full_lp(a_norm, solver_list)
    if g is None:
        return a, 0.0
    
    # Compute convolutions for blending
    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g)
    conv_gg = compute_conv(g)
    
    t_opt = find_best_blend(a_norm, g, conv_aa, conv_ag, conv_gg)
    
    if t_opt <= 0:
        return a, 0.0
    
    a_new = (1 - t_opt) * a_norm + t_opt * g
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a, 0.0
    a_new = a_new / s
    
    return a_new, t_opt

def lp_polish(a, time_budget, max_iters=50, verbose=False):
    """Iterative LP polishing."""
    best_a = a.copy()
    best_obj = compute_obj_conv(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start >= time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        a_new, t_opt = lp_iteration(best_a)
        obj_new = compute_obj_conv(a_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            if verbose:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                      f"elapsed={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if verbose:
                print(f"    LP it{it}: obj={obj_new:.8f} (no improvement), stall={stall}")
            if stall >= 3:
                # Perturb
                rng = np.random.default_rng(it * 104729)
                noise = rng.normal(0, 0.002 * stall, len(best_a))
                a_pert = best_a + noise
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                best_a = a_pert
                best_obj = compute_obj_conv(best_a)
                stall = 0
    
    return best_a, best_obj

def lse_smoothed_objective(a, T):
    """LSE smoothed objective."""
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)
    exp_vals = np.exp(T * (conv - cmax))
    lse = cmax + np.log(np.sum(exp_vals)) / T
    return 2.0 * n * lse

def lse_grad_fft(a, T):
    """FFT-based gradient of LSE smoothed objective."""
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)
    exp_vals = np.exp(T * (conv - cmax))
    w = exp_vals / exp_vals.sum()
    
    # grad[i] = 2n * sum_j a[j] * (w[i+j] + w[i-j]) for valid indices
    # Use FFT-based cross-correlation
    size = 1
    while size < 3 * n:
        size <<= 1
    
    fa = rfft(a, size)
    fw = rfft(w, size)
    
    # cross_corr[a, w][i] = sum_j a[j] * w[j+i]
    # = ifft(fft(a) * conj(fft(w))) shifted
    # Actually for linear cross-correlation:
    # xcorr[i] = sum_j a[j] w[j+i]
    # This equals the (2n-2-i)th element of conv(a, reversed(w))... 
    # Let me just use np for simplicity since n is at most ~2000
    
    # Direct O(n^2) computation - fast enough for n <= 2000
    grad = np.zeros(n)
    for i in range(n):
        # First term: sum_j a[j] w[i+j] for j in [0, min(n, 2n-1-i))
        j_end = min(n, 2*n-1-i)
        if j_end > 0:
            grad[i] += np.dot(a[:j_end], w[i:i+j_end])
        # Second term: sum_j a[j] w[i-j] for j in [0, i+1)
        if i >= 0:
            grad[i] += np.dot(a[:i+1], w[i::-1])
    
    return 2.0 * n * grad

def fista_lse(a, time_budget, T_start=5.0, T_end=50.0, max_iters=2000, verbose=False):
    """FISTA with LSE smoothing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_conv(a)
    start = time.monotonic()
    
    y = a.copy()
    t_k = 1.0
    lr = 1.0 / (2.0 * n)  # Initial learning rate (Lipshitz constant estimate)
    
    it = 0
    for it in range(max_iters):
        if time.monotonic() - start >= time_budget:
            break
        
        # Anneal T
        frac = min(1.0, it / max(1, max_iters))
        T = T_start * (T_end / T_start) ** frac
        
        # Gradient
        grad = lse_grad_fft(y, T)
        
        # Step
        x_new = project_simplex(y - lr * grad)
        
        # FISTA update
        t_new = (1 + np.sqrt(1 + 4 * t_k**2)) / 2
        y = x_new + ((t_k - 1) / t_new) * (x_new - a)
        
        # Check objective
        if it % 10 == 0:
            obj = compute_obj_conv(x_new)
            if obj < best_obj:
                best_obj = obj
                best_a = x_new.copy()
        
        a = x_new
        t_k = t_new
        
        if verbose and it % 200 == 0:
            print(f"      FISTA it{it}: best_obj={best_obj:.8f}, T={T:.1f}, "
                  f"t={time.monotonic()-start:.1f}s")
    
    # Final check
    obj = compute_obj_conv(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()
    
    return best_a

def coordinate_descent(a, time_budget, max_sweeps=5, verbose=False):

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)

    if s < 1e-10:
        return a

    best_a = a.copy()

    start = time.monotonic()

            break
        sweep_improved = False
        
        for i in range(n):
            if time.monotonic() - start >= time_budget:
                break
            
            ai_old = a[i]
            
            # Try a few candidates
            candidates = [0.0, ai_old * 0.5, ai_old * 1.5, 2 * ai_old, ai_old * 0.1]
            best_val = ai_old
            best_local = best_obj
            
            for cv in candidates:
                a[i] = cv
                s_new = float(np.sum(a))
                if s_new < 0.01:
                    continue
                a_try = a / s_new
                obj_try = compute_obj_conv(a_try)
                if obj_try < best_local:
                    best_local = obj_try
                    best_val = cv
            
            if best_val != ai_old:
                a[i] = best_val
                s_new = float(np.sum(a))
                if s_new > 0.01:
                    a = a / s_new
                    sweep_improved = True
            else:
                a[i] = ai_old
        
        obj_now = compute_obj_conv(a)
        if obj_now < best_obj:
            best_obj = obj_now
            best_a = a.copy()
        
        if verbose:
            print(f"      CD sweep {sweep}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
        
        if not sweep_improved:
            break
    
    return best_a

def arcsine_init(n):
    """Arcsine distribution initialization."""
    x = np.linspace(1.0/n, 1.0 - 1.0/n, n)
    a = 1.0 / np.sqrt(np.maximum(1e-10, x * (1 - x)))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def u_shaped_init(n, sharpness):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.01 + sharpness * (2*x - 1)**2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s
