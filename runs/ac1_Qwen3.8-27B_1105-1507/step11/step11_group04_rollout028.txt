Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
1. Symmetric constraint (halves search space, valid due to autocorrelation symmetry)
2. Fast subgradient descent with top-K lag weighting and momentum

4. Multiple diverse initializations with annealing

import numpy as np

from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj_norm(a):
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def make_symmetric_from_half(half, n):
    """Construct full symmetric sequence from first half."""
    a = np.zeros(n)
    if n % 2 == 0:
        a[:n//2] = half
        a[n//2:] = half[::-1]
    else:
        a[:(n+1)//2] = half
        a[(n+1)//2:] = half[:-1][::-1]
    return a

def get_half(a, n):
    """Extract half of symmetric sequence."""
    if n % 2 == 0:
        return a[:n//2].copy()
    else:
        return a[:(n+1)//2].copy()

def grad_top_k(a, conv, K=8):
    """
    Compute gradient using top-K lags with exponential weighting.
    Assumes a is symmetric.
    """
    n = len(a)
    conv_max = float(np.max(conv))
    if conv_max < 1e-30:
        return np.zeros(n)
    
    # Get top K indices
    top_k_idx = np.argsort(conv)[-K:]
    
    # Exponential weights based on how close to max
    top_vals = conv[top_k_idx]
    scale = max(conv_max * 0.05, 1e-15)
    weights = np.exp((top_vals - conv_max) / scale)
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_k_idx):
        # d(conv[k])/d(a[i]) = 2*a[k-i] for valid i
        imin = max(0, k - (n - 1))
        imax = min(n - 1, k)
        if imin <= imax:
            indices = np.arange(imin, imax + 1)
            grad[indices] += w * 2.0 * a[k - indices]
    
    return grad

def symmetric_gradient_descent(a, time_budget, seed=42, verbose=True):
    """
    Gradient descent with symmetry constraint, momentum, and adaptive step size.
    Works on the half-sequence (free parameters).
    """
    n = len(a)

    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    # Enforce symmetry
    half = get_half(a, n)
    
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    lr = 0.1
    momentum = np.zeros_like(half)
    beta = 0.9
    stall = 0
    K = 8
    
    while time.monotonic() - start < time_budget:
        # Reconstruct full sequence
        a_full = make_symmetric_from_half(half, n)
        s = float(np.sum(a_full))
        if s < 1e-10:
            break
        a_full = a_full / s
        
        conv = compute_conv(a_full)
        obj = 2.0 * n * float(np.max(conv))
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_full.copy()
            stall = 0
        
        # Compute gradient (on full sequence, then project to half)
        grad_full = grad_top_k(a_full, conv, K)
        # Project gradient to half-space
        # For symmetric sequence, the effective gradient on half[i] is:
        # grad_half[i] = grad_full[i] + grad_full[n-1-i] (for i != n-1-i)
        # or grad_full[i] * 2 (for the center element if n is odd)
        half_len = len(half)
        grad_half = np.zeros(half_len)
        for i in range(half_len):
            j = n - 1 - i
            if i == j:
                grad_half[i] = grad_full[i]
            else:
                grad_half[i] = (grad_full[i] + grad_full[j]) / 2.0
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad_half
        half_new = half - lr * momentum
        half_new = np.maximum(half_new, 0)
        
        # Renormalize
        a_try = make_symmetric_from_half(half_new, n)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            lr *= 0.5
            continue
        a_try = a_try / s_try
        obj_try = compute_obj_norm(a_try)
        
        if obj_try < best_obj - 1e-14:
            half = half_new
            # Accept
            if verbose and stall == 0 and (int((time.monotonic()-start)*5) % 100 == 0):
                print(f"    SGD: obj={best_obj:.10f}, lr={lr:.4f}, t={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 20:
                lr *= 0.7
                if lr < 1e-6:
                    # Reset with perturbation
                    lr = rng.uniform(0.05, 0.3)
                    half = half + rng.standard_normal(half_len) * lr * 0.1
                    half = np.maximum(half, 0)
                    stall = 0
        
        # Adaptive: increase K over time (more lags in gradient = smoother)
        # K = min(20, 5 + int(2 * (time.monotonic() - start) / time_budget * 10))
    
    return best_a

"""

Uses a combination of:
1. Mirror descent (exponential weights) for the LP subproblem - avoids cvxpy overhead
2. Symmetric constraint exploitation (halves search space)

4. Discrete coordinate descent with in-place conv updates for local polishing

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Cross convolution a*b via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def make_symmetric(half, n):
    """Construct full symmetric sequence from first half."""
    a = np.zeros(n)
    if n % 2 == 0:
        a[:n // 2] = half
        a[n // 2:] = half[::-1]
    else:
        a[:(n + 1) // 2] = half
        a[(n + 1) // 2:] = half[:-1][::-1]
    return a


def get_half(a, n):
    """Extract first half of symmetric sequence."""
    if n % 2 == 0:
        return a[:n // 2].copy()
    else:
        return a[:(n + 1) // 2].copy()


def top_k_gradient(a, conv, K=10):
    """
    Compute gradient using top-K lags with exponential weighting.
    d(conv[k])/d(a[i]) = 2*a[k-i] for valid i (0<=k-i<n).
    """
    n = len(a)
    conv_max = float(np.max(conv))
    if conv_max < 1e-30:
        return np.zeros(n)

    top_k_idx = np.argsort(conv)[-K:]
    top_vals = conv[top_k_idx]
    scale = max(conv_max * 0.05, 1e-15)
    weights = np.exp((top_vals - conv_max) / scale)
    weights /= np.sum(weights)

    grad = np.zeros(n)
    for w, k in zip(weights, top_k_idx):
        imin = max(0, k - (n - 1))
        imax = min(n - 1, k)
        if imin <= imax:
            indices = np.arange(imin, imax + 1)
            grad[indices] += w * 2.0 * a[k - indices]

    return grad


def mirror_descent_lp(a, time_budget, K=5, verbose=False):
    """
    Solve LP subproblem: min_g max_k (a*g)_k s.t. g in simplex, g>=0
    using mirror descent (exponential weights / Hedge algorithm).
    Returns optimal g.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))

    # Initialize g
    g = np.ones(n) / n
    eta = 1.0 / np.sqrt(n)

    best_g = g.copy()
    best_max_val = float(np.max(compute_conv_cross(a_norm, g)))
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        # Compute cross-convolution
        cross = compute_conv_cross(a_norm, g)
        cross_max = float(np.max(cross))

        if cross_max < best_max_val - 1e-12:
            best_max_val = cross_max
            best_g = g.copy()
            stall = 0
        else:
            stall += 1
            if stall % 100 == 0:
                eta *= 0.8
            if stall > 2000:
                break

        # Subgradient: row k* of the convolution matrix
        k_star = int(np.argmax(cross))

        # The subgradient w.r.t. g is: (A @ e_{k_star})_j = a_norm[k_star - j] for valid j
        # i.e., grad[j] = a_norm[k_star - j] if 0 <= k_star - j < n else 0
        j_min = max(0, k_star - (n - 1))
        j_max = min(n - 1, k_star)
        grad = np.zeros(n)
        if j_min <= j_max:
            grad[j_min:j_max + 1] = a_norm[k_star - np.arange(j_min, j_max + 1)]

        # Mirror descent step (exponential weights)
        log_g = np.log(g + 1e-30)
        log_g -= eta * grad
        # Normalize (staying on simplex)
        log_g -= np.max(log_g)  # numerical stability
        g_new = np.exp(log_g)
        g_new /= np.sum(g_new)
        g = g_new

    return best_g


def lp_alternate_fast(a, time_budget, seed=42, verbose=True):
    """
    LP alternating minimization using mirror descent for the LP subproblem.
    Uses backtracking line search.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 20

    while time.monotonic() - start < time_budget - 5:
        remaining = time_budget - (time.monotonic() - start)
        lp_budget = min(remaining * 0.1, 30.0)
        if lp_budget < 1.0:
            break

        # Solve LP subproblem
        g_lp = mirror_descent_lp(a, time_budget=lp_budget, K=5)

        # Backtracking line search
        best_local_obj = best_obj
        best_local_a = a.copy()
        t = 1.0

        for _ in range(10):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                t *= 0.5
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_norm(a_try)

            if obj_try < best_obj - 1e-12:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                break
            t *= 0.6

        if best_local_obj < best_obj - 1e-12:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall = 0
            if verbose and int((time.monotonic() - start) // 30) != int((time.monotonic() - start - 1) // 30):
                print(f"    LP: obj={best_obj:.10f}, t={t:.3f}, elapsed={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall >= max_stall:
                # Perturb
                pert_strength = 0.01 + 0.005 * min(stall - max_stall, 5)
                pert = rng.standard_normal(n) * pert_strength
                a_pert = a * (1.0 + pert)
                a_pert = np.maximum(a_pert, 0)
                s_pert = float(np.sum(a_pert))
                if s_pert > 1e-10:
                    a = a_pert / s_pert
                stall = 0
                max_stall = int(max_stall * 1.3)

    if verbose:
        print(f"    LP done: best={best_obj:.10f}")
    return best_a


def gradient_descent_symmetric(a, time_budget, seed=42, verbose=True):
    """
    Mirror descent on normalized symmetric sequence.
    Uses top-K subgradient with momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    # Enforce symmetry
    half = get_half(a, n)
    half_len = len(half)

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    lr = 0.3
    momentum = np.zeros(half_len)
    beta = 0.85
    stall = 0
    K = 8

    while time.monotonic() - start < time_budget:
        a_full = make_symmetric(half, n)
        s = float(np.sum(a_full))
        if s < 1e-10:
            break
        a_full = a_full / s

        conv = compute_conv(a_full)
        obj = 2.0 * n * float(np.max(conv))

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_full.copy()
            stall = 0
            if verbose and (int((time.monotonic() - start) * 3) % 100 == 0):
                print(f"    GD: obj={best_obj:.10f}, lr={lr:.4f}, t={time.monotonic()-start:.1f}s")

        # Compute gradient
        grad_full = top_k_gradient(a_full, conv, K)

        # Project to half-space
        grad_half = np.zeros(half_len)
        for i in range(half_len):
            j = n - 1 - i
            if i == j:
                grad_half[i] = grad_full[i]
            else:
                grad_half[i] = 0.5 * (grad_full[i] + grad_full[j])

        # Mirror descent (multiplicative) step on half
        log_half = np.log(half + 1e-30) - lr * grad_half
        log_half -= np.max(log_half)
        half_new = np.exp(log_half)
        half_new = np.maximum(half_new, 1e-15)

        a_try = make_symmetric(half_new, n)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            lr *= 0.5
            continue
        a_try = a_try / s_try
        obj_try = compute_obj_norm(a_try)

        if obj_try < best_obj - 1e-13:
            half = half_new
        else:
            stall += 1
            if stall > 50:
                lr *= 0.8
            if lr < 1e-5:
                lr = rng.uniform(0.1, 0.5)
                half = half + rng.standard_normal(half_len) * 0.01
                half = np.maximum(half, 0)
                stall = 0

    return best_a


def coordinate_descent_polish(a, time_budget, seed=42, verbose=True):
    """
    Discrete coordinate descent with in-place convolution updates.
    Tries multiple candidate values per coordinate.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    mean_val = 1.0 / n
    no_improve = 0
    max_no_improve = 4

    while time.monotonic() - start < time_budget:
        sweep_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget:
                break

            a_j = a[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)

            # Unaffected max
            max_unaffected = 0.0
            if k_start > 0:
                max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
            if k_end < 2 * n - 2:
                max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))

            # Candidate values
            if a_j < 1e-12:
                candidates = [0.0, mean_val * 0.1, mean_val * 0.3, mean_val * 0.7, mean_val, mean_val * 1.5, mean_val * 3]
            else:
                candidates = [
                    a_j * 0.3, a_j * 0.5, a_j * 0.7, a_j * 1.0,
                    a_j * 1.3, a_j * 1.5, a_j * 2.0, a_j * 3.0,
                    mean_val * 0.3, mean_val * 0.7, mean_val * 1.3, mean_val * 2.0, mean_val * 4.0
                ]
                candidates = [c for c in candidates if 0 <= c <= 0.4]
                candidates = list(set(candidates))

            best_trial_obj = best_obj
            best_trial_val = a_j

            for val in candidates:
                delta = val - a_j
                if abs(delta) < 1e-15:
                    continue

                S_new = 1.0 + delta
                if S_new <= 1e-10:
                    continue

                # Compute max of affected conv entries
                indices = np.arange(k_start, k_end + 1)
                affected = conv[k_start:k_end + 1].copy()
                mask = (indices == 2 * j)
                if np.any(~mask):
                    affected[~mask] = conv[k_start:k_end + 1][~mask] + 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    affected[mask] = conv[2 * j] + (val * val - a_j * a_j)

                max_conv_val = max(max_unaffected, float(np.max(affected)))
                obj_trial = 2.0 * n * max_conv_val / (S_new * S_new)

                if obj_trial < best_trial_obj - 1e-13:
                    best_trial_obj = obj_trial
                    best_trial_val = val

            if best_trial_val != a_j:
                delta = best_trial_val - a_j
                a[j] = best_trial_val
                S_new = 1.0 + delta

                # Update conv in-place
                indices = np.arange(k_start, k_end + 1)
                mask = (indices == 2 * j)
                if np.any(~mask):
                    conv[k_start:k_end + 1][~mask] += 2.0 * delta * a[indices[~mask] - j]
                if np.any(mask):
                    conv[2 * j] += (best_trial_val * best_trial_val - a_j * a_j)

                inv_S = 1.0 / S_new
                a *= inv_S
                conv *= inv_S * inv_S

                new_obj = 2.0 * n * float(np.max(conv))
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved = True

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and no_improve == 0 and int((time.monotonic() - start)) % 30 == 0:
            print(f"    CD: obj={best_obj:.10f}, t={time.monotonic()-start:.1f}s")

    return best_a


def fourier_perturb(a, rng, strength=0.015):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 8):
        freq = k * 2.0 * np.pi / n
        amp = strength * rng.uniform(0.5, 1.5) / k
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

        a_pert = fourier_perturb(best_a, rng, strength=0.02)
        alloc2 = min(remaining * 0.3, 100.0)
        a2 = gradient_descent_symmetric(a_pert, time_budget=alloc2, seed=seed + 1000, verbose=False)
        obj2 = compute_obj_norm(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            print(f"  GD2: {best_obj:.10f} **")

    # === Phase 2: LP Alternating (35% time) ===
    print("\n--- Phase 2: LP Alternating ---")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.4, 500.0)
    if p2_budget > 30:
        a_lp = lp_alternate_fast(best_a, time_budget=p2_budget, seed=seed + 2000, verbose=True)
        obj_lp = compute_obj_norm(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            print(f"  LP: {best_obj:.10f} **")
        else:
            print(f"  LP: {obj_lp:.10f}")

    # === Phase 3: Coordinate Descent Polish (25% time) ===
    print("\n--- Phase 3: CD Polish ---")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.4, 350.0)
    if p3_budget > 20:
        a_cd = coordinate_descent_polish(best_a, time_budget=p3_budget, seed=seed + 3000, verbose=True)
        obj_cd = compute_obj_norm(a_cd)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()
            print(f"  CD: {best_obj:.10f} **")
        else:
            print(f"  CD: {obj_cd:.10f}")

    # === Phase 4: LP from perturbed state (10% time) ===
    print("\n--- Phase 4: LP Restart ---")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        a_pert2 = fourier_perturb(best_a, rng, strength=0.008)
        p4_budget = min(remaining * 0.35, 180.0)
        a_lp2 = lp_alternate_fast(a_pert2, time_budget=p4_budget, seed=seed + 4000, verbose=False)
        obj_lp2 = compute_obj_norm(a_lp2)
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2.copy()
            print(f"  LP2: {best_obj:.10f} **")
        else:
            print(f"  LP2: {obj_lp2:.10f}")

    # === Phase 5: Final CD Polish ===
    print("\n--- Phase 5: Final Polish ---")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        p5_budget = min(remaining * 0.7, 100.0)
        a_cd2 = coordinate_descent_polish(best_a, time_budget=p5_budget, seed=seed + 5000, verbose=False)
        obj_cd2 = compute_obj_norm(a_cd2)
        if obj_cd2 < best_obj:
            best_obj = obj_cd2
            best_a = a_cd2.copy()
            print(f"  Final CD: {best_obj:.10f} **")

    # === Phase 6: Try adjacent lengths ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n--- Phase 6: Length Exploration ---")
        n_best = len(best_a)
        n_options = [n_best + 20, n_best - 20, n_best + 40, n_best - 40, n_best + 60]
        n_options = sorted(set([n for n in n_options if 200 <= n <= 2000 and n != n_best]))

        for n_try in n_options[:4]:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_a)
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                continue
            a_try = a_try / s

            alloc_lp = min(remaining * 0.25, 80.0)
            alloc_cd = min(remaining * 0.15, 40.0)

            if alloc_lp > 15:
                a_try = lp_alternate_fast(a_try, time_budget=alloc_lp, seed=n_try * 7, verbose=False)
            remaining = deadline - time.monotonic()
            if remaining > 15 and alloc_cd > 10:
                a_try = coordinate_descent_polish(a_try, time_budget=alloc_cd, seed=n_try * 13, verbose=False)

            obj_try = compute_obj_norm(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                n_best = n_try
                print(f"  ** n={n_try}: {best_obj:.10f} **")
            else:
                print(f"  n={n_try}: {obj_try:.10f}")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
