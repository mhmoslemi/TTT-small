    while size < 2 * n - 1:

    """Compute linear convolution a * a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = get_fft_size(n)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a)) for normalized a."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_conv_and_grad_max(a):
    """Compute convolution and subgradient of max_k conv[k]."""
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    # Find active set: lags where conv[k] is within tolerance of max
    tol = 1e-8 * max_val + 1e-12
    active = np.where(conv >= max_val - tol)[0]
    # Compute subgradient: average of gradients at active lags

            if 0 <= k_minus_i < n:
                grad[i] += 2.0 * a[k_minus_i]
    grad /= len(active)
    return conv, max_val, grad


def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) - (cssv - 1) > 0)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    n = len(a)

    if s < 1e-10:
        return a, np.inf
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    z = a.copy()
    t = 1.0
    lr = 0.1  # Learning rate (step size)
    # For non-convex, use a fixed or decreasing step size
    
    it = 0
    best_it = 0
    patience = 500
    
    while time.monotonic() - start < time_budget:

        conv_y = compute_conv(y)
        max_y = float(np.max(conv_y))
        active = np.where(conv_y >= max_y - 1e-6 * max_y)[0]
        
        # Subgradient: average of active gradients
        grad = np.zeros(n)
        for k in active:
            valid = (k - np.arange(n))
            mask = (valid >= 0) & (valid < n)
            grad[mask] += 2.0 * y[valid[mask]]

            y = best_a.copy()
            t = 1.0
            lr = lr * 0.5
            patience = max(100, patience // 2)

    n_total = max(10, int(time_budget * 40))  # estimate total iterations

    while time.monotonic() - start < time_budget - 2:
        frac = min(1.0, it / max(1, n_total))
        T = T_start * (T_end / T_start) ** frac

        # Compute conv at y (FISTA iterate)
        conv_y = compute_conv(y)

        # Soft-max weights
        scaled = conv_y / T
        max_s = float(np.max(scaled))
        exps = np.exp(scaled - max_s)
        w = exps / np.sum(exps)

        # Gradient via FFT
        grad = compute_smooth_grad_fft(y, w)

        # FISTA update
        a_new = y - lr * grad
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        z = project_simplex(z)
        t_k = t_new
        a = a_new
        y = z

        # Track best
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            best_it = it

        it += 1

        # Adaptive restart
        if it - best_it > patience:
            a = best_a.copy()
            y = best_a.copy()
            z = best_a.copy()
            t_k = 1.0
            lr *= 0.6
            patience = max(50, patience // 2)
            best_it = it

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    {label} it{it}: T={T:.5f}, obj={best_obj:.8f}, lr={lr:.5f}, t={elapsed:.0f}s")

    # Final evaluation
    final_obj = compute_obj(best_a)
    if final_obj < best_obj:
        best_obj = final_obj
    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a, g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0).astype(np.float64)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def line_search(a, g, t_max=1.5, n_coarse=30, n_fine=20):
    """Exhaustive line search for blend parameter t."""
    best_t = 0.0
    best_obj = compute_obj(a)
    ts = np.linspace(0.0, t_max, n_coarse)
    for t in ts:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 0.01:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    if 0.0 < best_t < t_max:
        lo = max(0.0, best_t - 0.05)
        hi = min(t_max, best_t + 0.05)
        ts_fine = np.linspace(lo, hi, n_fine)
        for t in ts_fine:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 0.01:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t
    return best_t, best_obj


def lp_polish(a, time_budget=100.0, max_iters=100, verbose=False):
    """LP-based polishing: repeatedly find best direction and blend."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(42 + n)

    while time.monotonic() - start < time_budget - 5 and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        T0 = time.monotonic()
        g_lp, M_lp = solve_lp(a)
        lp_time = time.monotonic() - T0
        if g_lp is None:
            break

        t_opt, obj_opt = line_search(a, g_lp)
        if t_opt > 1e-6 and obj_opt < best_obj - 1e-14:
            a_new = (1.0 - t_opt) * a + t_opt * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 0.01:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            # Try direct t=1
            obj_direct = compute_obj(g_lp)
            if obj_direct < best_obj:
                best_obj = obj_direct
                best_a = g_lp.copy()
                a = g_lp.copy()
                stall = 0
            else:
                stall += 1

        it += 1
        if verbose and it % 3 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                  f"lp_time={lp_time:.1f}s, elapsed={elapsed:.0f}s")

        if stall >= 3:
            if stall <= 5:
                freq = rng.integers(1, max(2, n // 100))
                phase = rng.uniform(0, 2 * np.pi)
                idx = np.arange(n, dtype=np.float64)
                pert = 0.02 * stall * np.cos(2 * np.pi * freq * idx / n + phase)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 0.01:
                    a = a / s
                else:
                    a = best_a.copy()
            else:
                mode = it % 5
                a_div = diverse_init(n, rng, mode=mode)
                a = 0.3 * a_div + 0.7 * best_a
                a = np.maximum(a, 0)
                a = a / a.sum()
            stall = 0

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    idx = np.arange(n)
    if mode == 0:
        a = 1.0 / (0.05 + 3.0 * (2 * x - 1) ** 2)
    elif mode == 1:
        a = np.exp(-((idx - n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) + \
            np.exp(-((idx - 3 * n // 4) ** 2) / (2 * max(1, n // 6) ** 2)) + 0.01
    elif mode == 2:
        a = 4 * np.abs(x - 0.5) + 0.05
    elif mode == 3:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 4:
        a = np.exp(4 * (2 * x - 1) ** 2)
    elif mode == 5:
        a = 1.0 / (0.02 + 4.0 * (2 * x - 1) ** 4)
    elif mode == 6:
        # Two-hump with adjustable width
        a = np.exp(3.0 * (2 * x - 1) ** 2) + 0.1 * np.exp(-20 * (2 * x - 1) ** 2)
    else:
        a = 1.0 / (0.1 + 2.0 * np.abs(2 * x - 1))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(a_src, n_dst):
    """Interpolate sequence to different length."""
    n_src = len(a_src)
    if n_src == 1:
        return np.ones(n_dst) / n_dst
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = float(np.sum(a_dst))
    if s < 0.01:
        return None
    return a_dst / s


def u_shaped_init(n, sharpness=3.0):
    """U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def exact_subgradient_step(a, lr, n_active_max=5):
    """Single exact subgradient step on max_k conv[k]."""
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    # Get top lags
    tol = 1e-6 * max_val
    active = np.where(conv >= max_val - tol)[0]
    if len(active) > n_active_max:
        # Take top n_active_max
        top_idx = np.argsort(conv)[::-1][:n_active_max]
        active = top_idx

    grad = np.zeros(n)
    for k in active:
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        if i_min <= i_max:
            indices = np.arange(i_min, i_max + 1)
            grad[indices] += 2.0 * a[k - indices]
    grad /= len(active)

    a_new = a - lr * grad
    a_new = project_simplex(a_new)
    return a_new, conv, max_val


def fista_exact(a, time_budget=30.0, lr_init=0.02, verbose=False):
    """FISTA with exact subgradient of max."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    y = a.copy()
    z = a.copy()
    t_k = 1.0
    lr = lr_init
    it = 0
    best_it = 0
    patience = 200

    while time.monotonic() - start < time_budget - 2:
        # Subgradient at y
        conv_y = compute_conv(y)
        max_y = float(np.max(conv_y))
        tol = 1e-5 * max_y
        active = np.where(conv_y >= max_y - tol)[0]
        if len(active) > 5:
            active = np.argsort(conv_y)[::-1][:5]

        grad = np.zeros(n)
        for k in active:
            i_min = max(0, k - n + 1)
            i_max = min(k, n - 1)
            if i_min <= i_max:
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += 2.0 * y[k - indices]
        grad /= len(active)

        a_new = y - lr * grad
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        z = project_simplex(z)
        t_k = t_new
        a = a_new
        y = z

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            best_it = it

        it += 1
        if it - best_it > patience:
            a = best_a.copy()
            y = best_a.copy()
            z = best_a.copy()
            t_k = 1.0
            lr *= 0.5
            patience = max(50, patience // 2)
            best_it = it

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    FISTA-exact it{it}: obj={best_obj:.8f}, lr={lr:.5f}, t={elapsed:.0f}s")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """Main search function: finds the best sequence of non-negative heights."""
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 40.0  # safety margin
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        s = float(np.sum(parent_arr))
        if s > 0.01:
            best_seq = parent_arr / s
            best_score = compute_obj(best_seq)
            print(f"Parent loaded: n={parent_arr.size}, obj={best_score:.8f}")
        else:
            raise ValueError("Sum too small")
    except Exception:
        best_seq = u_shaped_init(800, sharpness=3.0)
        best_score = compute_obj(best_seq)
        print(f"Default init: n={len(best_seq)}, obj={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Quick diverse initialization at best n ===
    print(f"\n=== Phase 1: Diverse Inits (n={n_best}) ===")
    for mode in range(7):
        a_alt = diverse_init(n_best, rng, mode=mode)
        obj_alt = compute_obj(a_alt)
        if obj_alt < best_score:
            best_score = obj_alt
            best_seq = a_alt.copy()
            print(f"  Mode {mode}: obj={best_score:.8f} **")
        else:
            print(f"  Mode {mode}: obj={obj_alt:.8f}")

    # Quick anneal on best init
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_ann, obj_ann = annealed_fista(best_seq, time_budget=min(remaining * 0.15, 60.0),
                                        T_start=0.03, T_end=0.001, lr_init=0.05,
                                        verbose=False, label="P1")
        if obj_ann < best_score:
            best_score = obj_ann
            best_seq = a_ann.copy()
            print(f"  Quick anneal: obj={best_score:.8f} **")

    # === Phase 2: n Exploration ===
    print(f"\n=== Phase 2: n Exploration ===")
    remaining = deadline - time.monotonic()
    if remaining > 150:
        n_base = len(best_seq)
        n_candidates = sorted(set([
            n_base - 100, n_base - 50, n_base + 50, n_base + 100,
            n_base + 200, n_base + 300, n_base + 500,
            400, 500, 600, 700, 800, 900, 1000, 1100, 1200,
            1400, 1600, 1800, 2000
        ]))
        n_candidates = [n for n in n_candidates if 300 <= n <= 2500 and abs(n - n_base) >= 40]
        # Sort by distance to current best
        n_candidates.sort(key=lambda x: abs(x - n_base))
        n_candidates = n_candidates[:8]
        print(f"  Candidates: {n_candidates}")

        for n_try in n_candidates:
            remaining = deadline - time.monotonic()
            if remaining < 80:
                break
            a_try = interpolate_sequence(best_seq, n_try)
            if a_try is None:
                continue
            # Quick anneal
            alloc = min(remaining * 0.25, 50.0)
            a_try2, obj_try = annealed_fista(a_try, time_budget=alloc,
                                             T_start=0.03, T_end=0.001,
                                             lr_init=0.05, verbose=False, label=f"n{n_try}")
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try2.copy()
                n_best = n_try
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 3: Main annealed FISTA (long run) ===
    print(f"\n=== Phase 3: Main Annealed FISTA (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        alloc = min(remaining * 0.35, 400.0)
        print(f"  Budget: {alloc:.0f}s")
        a_main, obj_main = annealed_fista(best_seq, time_budget=alloc,
                                          T_start=0.02, T_end=0.0003,
                                          lr_init=0.04, verbose=True, label="Main")
        if obj_main < best_score:
            best_score = obj_main
            best_seq = a_main.copy()
            print(f"  Main FISTA: obj={best_score:.8f} **")
        else:
            print(f"  Main FISTA: obj={obj_main:.8f} (no improvement)")

    # === Phase 4: LP Polish ===
    print(f"\n=== Phase 4: LP Polish (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        alloc = min(remaining * 0.25, 200.0)
        print(f"  Budget: {alloc:.0f}s")
        a_lp = lp_polish(best_seq, time_budget=alloc, max_iters=80, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP polish: obj={best_score:.8f} **")
        else:
            print(f"  LP polish: obj={obj_lp:.8f} (no improvement)")

    # === Phase 5: Exact subgradient refinement ===
    print(f"\n=== Phase 5: Exact Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(remaining * 0.15, 60.0)
        a_sg, obj_sg = fista_exact(best_seq, time_budget=alloc, lr_init=0.01, verbose=False)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Exact SG: obj={best_score:.8f} **")
        else:
            print(f"  Exact SG: obj={obj_sg:.8f} (no improvement)")

    # === Phase 6: Combined refinement (anneal + LP alternating) ===
    print(f"\n=== Phase 6: Combined Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_cur = len(best_seq)
        for round_idx in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            # Short anneal
            alloc_ann = min(remaining * 0.4, 40.0)
            a_r, obj_r = annealed_fista(best_seq, time_budget=alloc_ann,
                                        T_start=0.005, T_end=0.0002,
                                        lr_init=0.02, verbose=False, label=f"R{round_idx}a")
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
            # Short LP
            remaining = deadline - time.monotonic()
            if remaining > 20:
                alloc_lp = min(remaining * 0.4, 30.0)
                a_lp2 = lp_polish(best_seq, time_budget=alloc_lp, max_iters=20, verbose=False)
                obj_lp2 = compute_obj(a_lp2)
                if obj_lp2 < best_score:
                    best_score = obj_lp2
                    best_seq = a_lp2.copy()
            print(f"  Round {round_idx}: obj={best_score:.8f}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining - 5, 50.0)
        if alloc > 10:
            a_final = lp_polish(best_seq, time_budget=alloc, max_iters=30, verbose=False)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final LP: obj={best_score:.8f} **")
            # Quick exact SG
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                a_fsg, obj_fsg = fista_exact(best_seq, time_budget=min(remaining2 - 3, 15.0),
                                             lr_init=0.005, verbose=False)
                if obj_fsg < best_score:
                    best_score = obj_fsg
                    best_seq = a_fsg.copy()
                    print(f"  Final SG: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
