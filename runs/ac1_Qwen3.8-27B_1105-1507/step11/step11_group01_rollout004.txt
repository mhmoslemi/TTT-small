from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):

    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 for non-empty a."""
    n = len(a)
    if n == 0:

    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

                # Reset momentum, add perturbation
                v_mom = np.zeros(n)
                pert = rng.normal(0, 0.02, n)
                a_pert = a * np.exp(pert)
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = a_pert / a_pert.sum()
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                    a = a_pert
                lr = 3.0 / np.sqrt(n)
                stall = 0

        lr *= lr_decay
        it += 1

    return best_a


def exact_cd(a, time_budget=60.0, seed=42, verbose=False):
    """
    Exact coordinate descent: for each coordinate, find the optimal value
    via golden section search on the exact 1D subproblem.
    """
    n = len(a)
    a = normalize(a.copy())
    conv = compute_conv(a)
    best_obj = float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    sweep = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        prev_best = best_obj
        order = rng.permutation(n)
        S = float(np.sum(a))

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            v = a[i]

            # Affected convolution entries: k ranges from i to i+n-1
            # b[k] = sum_j a[j]*a[k-j]
            # For j=i: contributes a[i]*a[k-i]
            # For k-j=i, i.e., j=k-i: contributes a[k-i]*a[i]
            # So b[k] changes by: if k != 2i: 2*a[k-i]*(new_a[i]-v); if k==2i: (new_a[i])^2 - v^2
            k_min = i
            k_max = i + n - 1
            if k_min > 2 * n - 2:
                continue
            k_max = min(k_max, 2 * n - 2)

            ks = np.arange(k_min, k_max + 1)
            indices = ks - i  # a[indices] are the other factors
            valid = (indices >= 0) & (indices < n)
            ks_v = ks[valid]
            idx_v = indices[valid]
            c_v = conv[ks_v].copy()

            k_quad = 2 * i
            mask_quad = (ks_v == k_quad)
            ks_lin = ks_v[~mask_quad]
            idx_lin = idx_v[~mask_quad]
            c_lin = c_v[~mask_quad].copy()

            c_2i = float(conv[k_quad]) if (0 <= k_quad <= 2 * n - 2) else 0.0
            has_quad = k_min <= k_quad <= k_max

            # Define 1D objective
            def obj_1d(x):
                if len(c_lin) > 0:
                    lin_max = float(np.max(c_lin + 2.0 * idx_lin * (x - v)))
                else:
                    lin_max = 0.0
                if has_quad:
                    q_val = c_2i + x * x - v * v
                else:
                    q_val = 0.0
                return max(lin_max, q_val)

            # Candidate points for initial bracketing
            scale = max(v, 1.0 / n, 1e-6)
            candidates = [0.0, v * 0.5, v, v * 2.0, v * 3.0, 2.0 / n, 5.0 / n, 10.0 / n]
            candidates = sorted(set(max(0.0, float(c)) for c in candidates))

            best_x = v
            best_f = obj_1d(v)
            for x in candidates:
                fx = obj_1d(x)
                if fx < best_f:
                    best_f = fx
                    best_x = x

            # Golden section
            x_lo = max(0.0, best_x - scale)
            x_hi = best_x + scale * 2.0
            if x_hi - x_lo < 1e-14:
                x_hi = x_lo + 1e-8

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = obj_1d(c_pt)
            fd = obj_1d(d_pt)

            for _ in range(45):
                if x_hi - x_lo < 1e-13 * max(1.0, x_hi):
                    break
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = obj_1d(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = obj_1d(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - v) < 1e-15:
                continue

            delta = x_opt - v
            a[i] = x_opt

            # Update convolution
            if len(ks_lin) > 0:
                conv[ks_lin] += 2.0 * delta * idx_lin
            if has_quad and 0 <= k_quad <= 2 * n - 2:
                conv[k_quad] += x_opt * x_opt - v * v

            # Update best
            new_max = float(np.max(conv))
            if new_max < best_obj:
                best_obj = new_max
                best_a = a.copy()

        # Track stall
        if best_obj < prev_best - 1e-13:
            stall = 0
        else:
            stall += 1

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"      CD sweep {sweep}: max_conv={best_obj:.10f}, t={elapsed:.1f}s")

    return best_a


def peak_push(a, time_budget=30.0, seed=42):
    """
    Targeted peak reduction: identify the lag with maximum convolution value,
    then adjust coordinates to reduce that specific peak while monitoring others.
    """
    n = len(a)
    a = normalize(a.copy())
    conv = compute_conv(a)
    best_obj = float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    step = 1.0 / (n * 10)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        if max_val < best_obj:
            best_obj = max_val
            best_a = a.copy()

        # Find active lags (within 1% of max)
        threshold = max_val * 0.995
        active_ks = np.where(conv >= threshold)[0]

        # Compute pressure: p[i] = sum_{k in active} a[k-i] (clamped to valid range)
        pressure = np.zeros(n)
        for k in active_ks:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                pressure[lo:hi + 1] += a[k - hi:k - lo + 1]

        # Normalize pressure to [0, 1]
        p_max = float(np.max(pressure))
        if p_max > 1e-10:
            pressure /= p_max

        # Update: move mass from high-pressure to low-pressure coordinates
        mean_p = float(np.mean(pressure))
        update = (mean_p - pressure) * step
        a_new = a + update
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = float(np.max(compute_conv(a_new)))
        if obj_new < max_val:
            a = a_new
            step = min(step * 1.1, 5.0 / n)
        else:
            # Rejected - try smaller step or perturbation
            step *= 0.8
            if step < 1e-8:
                pert = rng.normal(0, 0.03, n)
                a = a * np.exp(pert)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                step = 1.0 / (n * 10)

    return best_a


def structured_init(n, mode=0, rng=None):
    """Generate structured initializations for a given length n."""
    if rng is None:
        rng = np.random.default_rng(0)

    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        # Linearly decreasing
        a = np.linspace(1.0, 0.1, n)
    elif mode == 2:
        # Exponentially decreasing
        r = 0.95
        a = r ** np.arange(n)
    elif mode == 3:
        # Two-level: first half high, second half low
        a = np.concatenate([np.ones(n // 2) * 2.0, np.ones(n - n // 2) * 0.5])
    elif mode == 4:
        # Square root ramp up then down (asymmetric)
        x = np.linspace(0, 1, n)
        a = np.sqrt(x) * (1 - 0.5 * x)
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** (-0.5)
    elif mode == 6:
        # Smooth random (low-freq)
        a = np.cumsum(rng.normal(0, 1, n))
        a = a - np.min(a)
        a = a ** 0.5
    elif mode == 7:
        # Cosine window (asymmetric)
        x = np.linspace(0, 1, n)
        a = (1 + np.cos(np.pi * x * 0.7)) / 2.0
        a = np.maximum(a, 0.01)
    elif mode == 8:
        # Logistic ramp
        x = np.linspace(-5, 5, n)
        a = 1.0 / (1.0 + np.exp(-x))
    else:
        a = rng.uniform(0, 1, n)

    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample_seq(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def perturb_and_refine(a, refine_fn, time_budget=30.0, n_tries=3, seed=42, verbose=False):
    """Apply random perturbations and refine, keeping the best result."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    for t in range(n_tries):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        alloc = remaining / (n_tries - t)

        # Perturb
        noise_scale = 0.02 + 0.01 * t
        pert = rng.normal(0, noise_scale, len(a))
        a_try = a * np.exp(pert)
        a_try = np.maximum(a_try, 0.0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s

        # Refine
        a_try = refine_fn(a_try, time_budget=alloc * 0.8, seed=seed + t * 100)
        obj_try = compute_obj(a_try)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            if verbose:
                print(f"        Perturb {t}: obj={best_obj:.8f} **")
        elif verbose:
            print(f"        Perturb {t}: obj={obj_try:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Quick multi-n exploration with structured inits
    Phase 2: Deep optimization on top candidates (Nesterov GD + CD)
    Phase 3: Perturb-and-refine for local minimum escape
    Phase 4: Final tight CD polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                n_parent = len(best_seq)
                print(f"Parent (n={n_parent}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(n_parent) / n_parent
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_parent}): obj={best_score:.8f}")

    if best_seq is None:
        best_seq = np.ones(800)
        best_score = compute_obj(best_seq)

    # === Phase 1: Quick multi-n exploration ===
    print(f"\n=== Phase 1: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()

    n_values = []
    for nc in [400, 500, 600, 700, 800, 900, 1000, 1200, 1500, 2000, 2500, 3000, 4000]:
        if nc not in n_values:
            n_values.append(nc)

    # Sort by distance to current best n, prioritize nearby
    n_values.sort(key=lambda x: abs(x - n_parent))
    # But always include a spread
    n_values = sorted(set(n_values[:4] + [500, 800, 1200, 2000, 3000]))

    print(f"  Testing n values: {n_values}")
    candidates = []  # list of (obj, seq)

    # Always include parent
    candidates.append((best_score, best_seq.copy()))

    per_n_budget = min(remaining * 0.35 / len(n_values), 40.0)

    for n_try in n_values:
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        if len(n_try) if isinstance(n_try, list) else n_try == n_parent:
            continue

        alloc = min(per_n_budget, remaining * 0.3)
        if alloc < 5:
            break

        # Try 2 modes for this n
        for mode in [0, 1]:
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            a_init = structured_init(n_try, mode=mode, rng=rng)

            if mode == 0:
                # For uniform init, just run CD quickly
                a_opt = exact_cd(a_init, time_budget=alloc * 0.6, seed=n_try + mode)
            else:
                a_opt = exact_cd(a_init, time_budget=alloc * 0.6, seed=n_try + mode)

            obj = compute_obj(a_opt)

            # Quick Nesterov GD
            remaining = deadline - time.monotonic()
            gd_alloc = remaining * 0.15
            if gd_alloc > 8:
                a_gd = nesterov_smoothed_gd(a_opt, time_budget=gd_alloc,
                                            tau_start=15.0, tau_end=800.0,
                                            seed=n_try + mode * 100)
                obj_gd = compute_obj(a_gd)
                if obj_gd < obj:
                    a_opt = a_gd
                    obj = obj_gd

            candidates.append((obj, a_opt.copy()))
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}, mode={mode}: obj={best_score:.8f}")
            else:
                print(f"  n={n_try}, mode={mode}: obj={obj:.8f}")

    # Sort candidates by objective
    candidates.sort(key=lambda x: x[0])
    print(f"  Top candidate: obj={candidates[0][0]:.8f}, n={len(candidates[0][1])}")

    # === Phase 2: Deep optimization on top candidates ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    remaining = deadline - time.monotonic()
    top_k = min(3, len(candidates))
    per_cand_budget = min(remaining * 0.40 / top_k, 200.0)

    for ci in range(top_k):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(per_cand_budget, remaining * 0.4)
        if alloc < 20:
            break

        a_cand = candidates[ci][1].copy()
        obj_cand = candidates[ci][0]
        n_cand = len(a_cand)
        print(f"  Candidate {ci}: n={n_cand}, obj={obj_cand:.8f}")

        # Step 2a: Nesterov smoothed GD
        gd_alloc = alloc * 0.3
        if gd_alloc > 10:
            a_gd = nesterov_smoothed_gd(a_cand, time_budget=gd_alloc,
                                        tau_start=10.0, tau_end=1500.0,
                                        seed=seed + ci * 50)
            obj_gd = compute_obj(a_gd)
            if obj_gd < obj_cand:
                obj_cand = obj_gd
                a_cand = a_gd
                print(f"    After GD: obj={obj_cand:.8f}")

        # Step 2b: Exact CD
        cd_alloc = alloc * 0.4
        if cd_alloc > 10:
            a_cd = exact_cd(a_cand, time_budget=cd_alloc, seed=seed + 1000 + ci)
            obj_cd = compute_obj(a_cd)
            if obj_cd < obj_cand:
                obj_cand = obj_cd
                a_cand = a_cd
                print(f"    After CD: obj={obj_cand:.8f}")

        # Step 2c: Peak push
        pp_alloc = alloc * 0.15
        if pp_alloc > 5:
            a_pp = peak_push(a_cand, time_budget=pp_alloc, seed=seed + 2000 + ci)
            obj_pp = compute_obj(a_pp)
            if obj_pp < obj_cand:
                obj_cand = obj_pp
                a_cand = a_pp
                print(f"    After PeakPush: obj={obj_cand:.8f}")

        # Step 2d: Short CD polish
        cd2_alloc = alloc * 0.15
        if cd2_alloc > 5:
            a_cd2 = exact_cd(a_cand, time_budget=cd2_alloc, seed=seed + 3000 + ci)
            obj_cd2 = compute_obj(a_cd2)
            if obj_cd2 < obj_cand:
                obj_cand = obj_cd2
                a_cand = a_cd2
                print(f"    After CD2: obj={obj_cd2:.8f}")

        print(f"    Final candidate {ci}: obj={obj_cand:.8f}")

        if obj_cand < best_score:
            best_score = obj_cand
            best_seq = a_cand.copy()
            print(f"    ** New best: {best_score:.8f}")

    # === Phase 3: Perturb and refine ===
    print(f"\n=== Phase 3: Perturb & Refine ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_refine = min(4, int(remaining / 40))
        ppr_budget = remaining * 0.35 / max(n_refine, 1)

        for t in range(n_refine):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(ppr_budget, remaining * 0.3)
            if alloc < 15:
                break

            a_r = perturb_and_refine(
                best_seq,
                refine_fn=exact_cd,
                time_budget=alloc,
                n_tries=2,
                seed=seed + 5000 + t * 100,
                verbose=True
            )
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  PPR {t}: obj={best_score:.8f} **")

    # === Phase 4: Try a few nearby n values with the best shape ===
    print(f"\n=== Phase 4: Nearby n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_neighbors = []
    for delta in [-200, -100, 100, 200, 300, 500]:
        nn = n_best + delta
        if 200 <= nn <= 6000 and nn != n_best and nn not in n_neighbors:
            n_neighbors.append(nn)
    n_neighbors = n_neighbors[:4]

    for nn in n_neighbors:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.1, 50.0)
        if alloc < 15:
            break

        a_nn = resample_seq(best_seq, nn)
        a_nn = exact_cd(a_nn, time_budget=alloc, seed=nn)
        obj_nn = compute_obj(a_nn)

        if obj_nn < best_score:
            best_score = obj_nn
            best_seq = a_nn.copy()
            print(f"  ** n={nn}: obj={best_score:.8f}")
        else:
            print(f"  n={nn}: obj={obj_nn:.8f}")

    # === Phase 5: Final tight CD polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        # Multiple CD rounds with decreasing time
        for round_i in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.3, 60.0)
            if alloc < 8:
                break

            a_f = exact_cd(best_seq, time_budget=alloc, seed=seed + 9000 + round_i)
            obj_f = compute_obj(a_f)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final CD {round_i}: obj={best_score:.8f} **")
            else:
                print(f"  Final CD {round_i}: obj={obj_f:.8f}")

        # One more Nesterov GD
        remaining = deadline - time.monotonic()
        if remaining > 20:
            gd_alloc = min(remaining * 0.4, 60.0)
            a_gd = nesterov_smoothed_gd(best_seq, time_budget=gd_alloc,
                                        tau_start=50.0, tau_end=3000.0,
                                        seed=seed + 9999)
            obj_gd = compute_obj(a_gd)
            if obj_gd < best_score:
                best_score = obj_gd
                best_seq = a_gd.copy()
                print(f"  Final GD: obj={best_score:.8f} **")

            # One more CD
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_final = exact_cd(best_seq, time_budget=min(remaining * 0.5, 30.0),
                                   seed=seed + 99999)
                obj_final = compute_obj(a_final)
                if obj_final < best_score:
                    best_score = obj_final
                    best_seq = a_final.copy()
                    print(f"  Final CD after GD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = 2.0

    best_seq = np.maximum(best_seq, 0.0)
    result = [float(min(1000.0, x)) for x in best_seq]

    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
