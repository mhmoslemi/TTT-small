Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def lse_obj_and_grad(a, tau):
    """
    Compute log-sum-exp smoothed objective and gradient.
    Objective: (1/tau) * log(sum(exp(tau * c_k))) / (sum a)^2
    where c = conv(a, a).
    
    Returns (obj_value, gradient w.r.t. a).
    We fix sum(a) = 1 for the gradient computation.
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = tau * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))

        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
        lse_val = cmax
    else:
        p = exp_vals / sum_exp
        lse_val = cmax + np.log(sum_exp) / tau
    
    # Gradient of (1/tau)*log(sum(exp(tau*c_k))) w.r.t. a
    # = sum_k p_k * (1/tau) * (dc_k/da) * tau * ... 
    # Actually: d/d(a_i) [(1/tau) log sum_k exp(tau c_k)]
    # = (1/tau) * sum_k exp(tau c_k) * tau * (dc_k/da_i) / sum_k exp(tau c_k)
    # = sum_k p_k * (dc_k/da_i)

    # So grad_i = 2 * sum_k p_k * a_{k-i}
    
    # This is 2 * (p * a_reversed) but let's use the correlation formula
    N = 1

def lp_solve(a, solver_list=None):
    """Solve min M s.t. (a*g)_k <= M for all k, g >= 0, sum(g) = 1."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                sg = float(np.sum(g_val))
                if sg > 1e-10:
                    return g_val / sg, float(M_var.value)
        except Exception:
            continue
    return None, None


def geometric_blend(a, g, alpha):
    """Geometric (multiplicative) blend on the simplex.
    a_new = normalize(a^alpha * g^(1-alpha))
    """
    a_safe = np.maximum(a, 1e-300)
    g_safe = np.maximum(g, 1e-300)
    a_new = np.exp(alpha * np.log(a_safe) + (1 - alpha) * np.log(g_safe))
    s = float(np.sum(a_new))
    if s < 1e-300:
        return a.copy()
    return a_new / s


def lp_fixed_point_v2(a, time_budget, verbose=False, seed=42, max_iters=300):
    """
    LP fixed-point with geometric blending and adaptive step.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    stall_count = 0
    alpha_current = 0.3  # Start with moderate geometric step

    for it in range(max_iters):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        g_lp, M_lp = lp_solve(a)
        if g_lp is None:
            stall_count += 1
            if stall_count > 10:
                break
            continue
        stall_count = 0

        # Try geometric blending with various alpha values
        # alpha=1 means stay, alpha=0 means jump to g
        alpha_candidates = [
            0.1, 0.15, 0.2, 0.25, alpha_current, 
            0.35, 0.4, 0.5, 0.6, 0.7,
            max(0.05, alpha_current * 0.7),
            min(0.95, alpha_current * 1.3),
        ]
        alpha_candidates = sorted(set([max(0.05, min(0.95, x)) for x in alpha_candidates]))

        best_local_obj = best_obj
        best_local_a = best_a.copy()
        best_local_alpha = alpha_current

        for alpha in alpha_candidates:
            a_try = geometric_blend(a, g_lp, alpha)
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_alpha = alpha

        # Also try arithmetic blending for comparison
        for t in [0.1, 0.2, 0.3, 0.5]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj - 1e-14:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_local_alpha = alpha_current  # keep alpha for geometric

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a
            alpha_current = max(0.1, min(0.9, best_local_alpha * 0.9))
            if verbose and it % 20 == 0:
                print(f"    LP-v2 it{it:3d}: obj={best_obj:.10f}, alpha={best_local_alpha:.3f}")
        else:
            stall_count += 1
            # Perturbation to escape
            if stall_count % 5 == 3:
                a = a + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
            
            if stall_count > 50:
                a = best_a.copy()
                a = a + 0.03 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall_count = 0
                alpha_current = 0.3

    return best_a


def nesterov_mirror_descent(a, time_budget, verbose=False, seed=42, max_iters=100000):
    """
    Nesterov-accelerated mirror descent on the simplex.
    Uses log-sum-exp smoothing with adaptive tau.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Adaptive tau schedule: start smooth, get sharper
    # tau крупный = smooth approximation
    # We'll use a geometric schedule for tau
    tau_start = 10.0
    tau_end = 5000.0
    # Number of tau steps
    n_tau_steps = 12
    tau_levels = np.geomspace(tau_start, tau_end, n_tau_steps)

    beta = 0.9  # Nesterov momentum
    lr_base = 0.5 / np.sqrt(n)

    v = np.zeros(n)  # Nesterov velocity (in log-space)
    log_a = np.log(np.maximum(a, 1e-300))

    for tau in tau_levels:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        tau_iters = 0
        max_tau_iters = max(1000, int(remaining * 50 / max(tau, 1)))
        # Scale learning rate with tau (smaller tau -> larger steps)
        lr = lr_base / max(1.0, np.sqrt(tau))
        
        while tau_iters < max_tau_iters:
            remaining = time_budget - (time.monotonic() - start)
            if remaining < 1:
                break

            # Nesterov step: y = a + beta * v (in log space)
            log_y = log_a + beta * v
            # Convert back to simplex
            y = np.exp(log_y - np.log(np.sum(np.exp(log_y))))
            y = np.maximum(y, 1e-300)
            y = y / y.sum()

            # Compute gradient
            _, grad = lse_obj_and_grad(y, tau)
            
            # Project gradient to tangent space of simplex
            grad_proj = grad - np.mean(grad)
            
            # Update velocity
            v = beta * v - lr * grad_proj
            log_a_new = log_y + v
            a_new = np.exp(log_a_new - np.log(np.sum(np.exp(log_a_new - np.log(np.sum(np.exp(log_a_new)))))))
            # Simpler: just normalize
            a_new = np.exp(log_a_new)
            a_new = a_new / a_new.sum()
            a_new = np.maximum(a_new, 1e-300)
            a_new = a_new / a_new.sum()

            # Check objective
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()

            log_a = np.log(a_new)
            a = a_new
            tau_iters += 1
            lr = max(1e-8, lr * 0.9999)

        if verbose:
            print(f"    NMD tau={tau:8.1f}: obj={best_obj:.10f}, iters={tau_iters}")

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def make_init(n, mode):
    """Create structured initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.maximum(np.sinc(x), 0.001)
        return a / a.sum()
    elif mode == 2:
        a = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
        return a / a.sum()
    elif mode == 3:
        x = np.arange(n) / (n - 1)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        return a / a.sum()
    elif mode == 4:
        x = np.linspace(-3, 3, n)
        a = np.exp(-x ** 2)
        return a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
        return a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * np.abs(x - 0.5), 0.01)
        return a / a.sum()
    else:
        rng = np.random.default_rng(mode + 100)
        a = rng.uniform(0.1, 1.0, n)
        return a / a.sum()
