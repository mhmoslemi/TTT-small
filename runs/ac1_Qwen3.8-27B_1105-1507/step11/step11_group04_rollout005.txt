import numpy as np

from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
            M.value = float(np.max(A @ warm_start))
        except Exception:
            pass

    for solver in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses improved LP alternating minimization with analytic coordinate descent,
subgradient updates, adaptive restarts with diverse perturbations, and
length exploration. Key improvements over baseline: (1) analytic trial points
for 1D subproblems in CD, (2) subgradient-guided updates, (3) more aggressive
and diverse restart strategies, (4) adaptive time allocation.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Compute objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, warm_start=None):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if warm_start is not None:
        try:
            g.value = warm_start.copy()
            M.value = float(np.max(A @ warm_start))
        except Exception:
            pass

    for solver in ['CLARABEL', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(np.asarray(g.value), 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def cd_sweep_analytic(a, conv, n, order, max_sweeps=100, time_start=None, time_budget=None):
    """
    One pass of coordinate descent with analytic trial points.
    Returns (a, conv, improved) after in-place updates.
    """
    improved = False
    mean_val = 1.0 / n

    for j in order:
        if time_start is not None and time.monotonic() - time_start > time_budget:
            break

        a_j = a[j]
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)

        # Unaffected max
        max_unaffected = 0.0
        if k_start > 0:
            max_unaffected = max(max_unaffected, float(np.max(conv[:k_start])))
        if k_end < 2 * n - 2:
            max_unaffected = max(max_unaffected, float(np.max(conv[k_end + 1:])))

        current_max = float(np.max(conv))
        current_obj = 2.0 * n * current_max

        # Build candidate deltas: heuristic + analytic
        candidates = [-a_j, -a_j * 0.7, -a_j * 0.4, -a_j * 0.1, 0.0,
                      a_j * 0.1, a_j * 0.3, a_j * 0.7, a_j * 1.5, a_j * 3.0,
                      mean_val * 0.5, mean_val * 1.5, mean_val * 3.0]

        # Add analytic optima for affected entries near the max
        affected_range = conv[k_start:k_end + 1]
        indices = np.arange(k_start, k_end + 1)
        # Only check entries within 1.5x of current max (top ~30%)
        threshold = current_max * 0.7
        for idx in range(len(indices)):
            k = indices[idx]
            if affected_range[idx] < threshold and k_start < len(conv) - 1:
                continue
            if k == 2 * j:
                c_k = 2.0 * a_j
                d_k = 1.0
            else:
                c_k = 2.0 * a[k - j]
                d_k = 0.0

            if abs(c_k) > 1e-14:
                if d_k == 0:
                    delta_opt = 2.0 * conv[k] / c_k - 1.0
                else:
                    denom = 2.0 - c_k
                    if abs(denom) > 1e-14:
                        delta_opt = (2.0 * conv[k] - c_k) / denom
                    else:
                        delta_opt = 0.0
                if -a_j - 0.001 < delta_opt < a_j * 5.0 + 0.1:
                    candidates.append(delta_opt)

        # Also add delta that makes affected = max_unaffected
        if max_unaffected > 0:
            for idx in range(len(indices)):
                k = indices[idx]
                base = conv[k]
                if k == 2 * j:
                    # (base + 2*d*a_j + d^2) = max_unaffected
                    # d^2 + 2*a_j*d + (base - max_unaffected) = 0
                    disc = 4 * a_j * a_j - 4 * (base - max_unaffected)
                    if disc >= 0:
                        d1 = (-2 * a_j + np.sqrt(disc)) / 2
                        d2 = (-2 * a_j - np.sqrt(disc)) / 2
                        for d in [d1, d2]:
                            if -a_j - 0.001 < d < a_j * 5 + 0.1:
                                candidates.append(d)
                else:
                    c_k = 2.0 * a[k - j]
                    if abs(c_k) > 1e-14:
                        d_opt = (max_unaffected - base) / c_k
                        if -a_j - 0.001 < d_opt < a_j * 5 + 0.1:
                            candidates.append(d_opt)

        candidates = np.array(candidates)
        candidates = np.unique(candidates)

        best_delta = 0.0
        best_trial_obj = current_obj

        for delta in candidates:
            S_new = 1.0 + delta
            if S_new <= 1e-10:
                continue

            # Vectorized affected computation
            mask_2j = (indices == 2 * j)
            affected = affected_range.copy()
            if np.any(~mask_2j):
                affected[~mask_2j] = affected_range[~mask_2j] + 2.0 * delta * a[indices[~mask_2j] - j]
            if np.any(mask_2j):
                affected[mask_2j] = affected_range[mask_2j] + 2.0 * delta * a_j + delta * delta

            max_new = max(max_unaffected, float(np.max(affected)))
            obj_trial = 2.0 * n * max_new / (S_new * S_new)

            if obj_trial < best_trial_obj - 1e-15:
                best_trial_obj = obj_trial
                best_delta = delta

        if best_delta != 0.0:
            # Apply update with clamping
            new_val = a_j + best_delta
            if new_val < 0:
                new_val = 0.0
            effective_delta = new_val - a_j
            a[j] = new_val
            S_new = 1.0 + effective_delta

            if S_new <= 1e-10:
                continue

            # Update conv in-place
            if np.any(~mask_2j):
                conv[k_start:k_end + 1][~mask_2j] += 2.0 * effective_delta * a[indices[~mask_2j] - j]
            if np.any(mask_2j):
                conv[2 * j] += 2.0 * effective_delta * a_j + effective_delta * effective_delta

            # Normalize
            inv_S = 1.0 / S_new
            a *= inv_S
            conv *= inv_S * inv_S
            improved = True

    return a, conv, improved


def coordinate_descent_v2(a, time_budget, seed=42, verbose=True):
    """
    Improved coordinate descent with analytic trial points.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    sweep = 0
    no_improve = 0
    max_no_improve = 4

    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        a, conv, improved = cd_sweep_analytic(a, conv, n, order,
                                              time_start=start, time_budget=time_budget)
        if improved:
            no_improve = 0
            new_obj = 2.0 * n * float(np.max(conv))
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD v2 sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.1f}s")

    return best_a


def lp_alternate_v2(a, time_budget, seed=42, verbose=True):
    """
    LP alternating minimization with adaptive step size, subgradient mixing,
    and adaptive restarts.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    prev_g = None
    iter_count = 0
    stall = 0

    while time.monotonic() - start < time_budget - 5:
        t0 = time.monotonic()
        g_lp, M_lp = solve_lp(a, warm_start=prev_g)
        if g_lp is None:
            # Try without warm start
            g_lp, M_lp = solve_lp(a, warm_start=None)
            if g_lp is None:
                break
        prev_g = g_lp.copy()
        iter_count += 1

        # Compute subgradient of max(conv(a,a)) w.r.t. a
        conv_a = compute_conv(a)
        k_star = int(np.argmax(conv_a))
        # subgrad[j] = a[k_star - j] + a[j - (k_star - n + 1)] (with boundary)
        subgrad = np.zeros(n)
        for j in range(n):
            val = 0.0
            i1 = k_star - j
            if 0 <= i1 < n:
                val += a[i1]
            i2 = j
            i3 = k_star - j
            # conv[k] = sum_i a[i]*a[k-i], d/d a[j] = a[k-j] + a[j-(k-j)] = a[k-j] + a[2j-k]
            # Wait: d(conv[k])/da[j] = a[k-j] (from term i=j) + a[k-j] (from term i=k-j if k-j!=j)
            # Actually: conv[k] = sum_i a[i]*a[k-i]
            # d/d a[j]: appears when i=j: a[k-j]; when k-i=j i.e. i=k-j: a[k-j]
            # If j == k-j (i.e. k=2j): only one term, derivative is 2*a[j]
            # Otherwise: 2*a[k-j]
            # But we want subgrad of max_k conv[k], which is just the gradient of conv[k*]
            i1 = k_star - j
            if 0 <= i1 < n:
                val += a[i1]
            i2 = 2 * j - k_star
            if 0 <= i2 < n and i2 != i1:
                val += a[i2]
            subgrad[j] = val

        # Normalize subgradient
        sg_norm = float(np.linalg.norm(subgrad))
        if sg_norm > 1e-10:
            subgrad = subgrad / sg_norm

        # Adaptive line search with multiple t values, mixing LP and subgradient
        best_local_obj = best_obj
        best_local_a = a.copy()

        t_values = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]
        alpha_values = [0.0, 0.05, 0.1, 0.2]  # subgradient mixing

        for t in t_values[:6]:
            for alpha in alpha_values:
                a_try = (1.0 - t) * a + t * g_lp
                if alpha > 0 and sg_norm > 1e-10:
                    # Step size for subgradient
                    step = alpha * 0.01
                    a_try = a_try - step * subgrad
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj_norm(a_try)
                if obj_try < best_local_obj - 1e-14:
                    best_local_obj = obj_try
                    best_local_a = a_try.copy()

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_local_a.copy()
            stall = 0
            if verbose and iter_count % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP {iter_count}: obj={best_obj:.10f}, elapsed={elapsed:.0f}s")
        else:
            stall += 1
            if stall >= 5:
                # Progressive perturbation to escape
                if stall < 8:
                    strength = 0.005 * (stall - 4)
                    pert = rng.standard_normal(n) * strength
                    a = best_a * np.exp(pert)
                elif stall < 12:
                    # Fourier perturbation
                    pert = np.zeros(n)
                    for k in range(1, 11):
                        freq = 2.0 * np.pi * k / n
                        amp = 0.02 * rng.uniform(0.5, 1.5) / np.sqrt(k)
                        phase = rng.uniform(0, 2.0 * np.pi)
                        pert += amp * np.cos(freq * np.arange(n) + phase)
                    a = best_a * np.exp(pert)
                else:
                    # Aggressive restart
                    a = best_a.copy()
                    mask = rng.random(n) < 0.3
                    a[mask] *= rng.uniform(0.1, 3.0, mask.sum())
                    a = np.maximum(a, 0)

                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                prev_g = None
                stall = 0

        per_iter_time = time.monotonic() - t0
        if per_iter_time > time_budget * 0.1:
            # LP is getting slow, reduce problem or stop
            break

        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - start
            remaining = time_budget - elapsed
            print(f"    LP iter {iter_count}: best={best_obj:.10f}, M_bp={M_lp:.6f}, rem={remaining:.0f}s")

    if verbose:
        print(f"    LP done: {iter_count} iters, best={best_obj:.10f}")
    return best_a


def fourier_perturb(a, rng, strength=0.015):
    """Add structured Fourier perturbation."""
    n = len(a)
    pert = np.zeros(n)
    for k in range(1, 9):
        freq = 2.0 * np.pi * k / n
        amp = strength * rng.uniform(0.5, 2.0) / np.sqrt(k)
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.cos(freq * np.arange(n) + phase)
    a_new = a * np.exp(pert)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def random_perturb(a, rng, strength=0.05):
    """Add random multiplicative perturbation."""
    n = len(a)
    pert = 1.0 + strength * rng.standard_normal(n)
    a_new = a * pert
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def block_perturb(a, rng, block_size=3, strength=0.1):
    """Perturb blocks of adjacent coordinates."""
    n = len(a)
    a_new = a.copy()
    n_blocks = n // block_size
    for _ in range(n_blocks // 3):
        start = rng.integers(0, n - block_size)
        scale = rng.uniform(1.0 - strength, 1.0 + strength, block_size)
        a_new[start:start + block_size] *= scale
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

            break

        # Interpolate to new length
        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, n_try)
        a_try = np.interp(x_new, x_old, best_a)
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s

        alloc = min(remaining * 0.25, 100.0)
        if alloc > 25:
            # Quick LP
            a_try = lp_alternate_v2(a_try, time_budget=alloc * 0.5, seed=n_try * 7, verbose=False)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_try = coordinate_descent_v2(a_try, time_budget=min(remaining * 0.3, 40.0),
                                              seed=n_try * 13, verbose=False)

        obj_try = compute_obj_norm(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            print(f"  ** n={n_try}: {best_obj:.10f} **")
        else:
            print(f"  n={n_try}: {obj_try:.10f}")

    # === Phase 6: Final CD polish ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print("\n--- Phase 6: Final Polish ---")
        a_fin = coordinate_descent_v2(best_a, time_budget=min(remaining * 0.7, 50.0),
                                      seed=seed + 99999, verbose=False)
        obj_fin = compute_obj_norm(a_fin)
        if obj_fin < best_obj:
            best_obj = obj_fin
            best_a = a_fin.copy()
            print(f"  Final: {best_obj:.10f} **")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")
    return result
