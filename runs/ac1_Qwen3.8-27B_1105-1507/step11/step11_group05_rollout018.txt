"""

Key improvements over previous approach:

2. Full-constraint LP polishing (all lags, leveraging sparse Toeplitz structure)

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        return proj / s
    return np.ones(n) / n


def lse_obj_and_grad(a, T):
    """
    Compute LSE-smoothed objective and gradient.
    a must be normalized (sum=1).
    Returns (objective_value, gradient_array).
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    exp_vals = np.exp(T * (conv - cmax))
    w = exp_vals / exp_vals.sum()
    lse_val = cmax + np.log(np.sum(exp_vals)) / T

    # grad[i] = 2 * sum_{m=0}^{n-1} w[i+m] * a[m]
    # k = i+m, so k ranges from i to i+n-1
    grad = np.zeros(n)
    for i in range(n):
        lo = i
        hi = i + n  # exclusive
        grad[i] = 2.0 * np.dot(w[lo:hi], a)
    return 2.0 * n * lse_val, 2.0 * n * grad


def nesterov_lse_gd(a, time_budget, T_start=5.0, T_end=60.0, verbose=False):
    """
    Nesterov-accelerated LSE smoothed gradient descent.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = a.copy()
    y = a.copy()
    t_k = 1.0
    lr = 5.0 / (n * T_start)
    it = 0
    best_it = 0
    patience = 500

    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** frac
        lr_curr = 5.0 / (n * T)

        obj_y, grad_y = lse_obj_and_grad(y, T)

        # Nesterov momentum
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_next
        y_new = x + beta * (y - x)
        y_new = project_simplex(y_new)

        # Gradient step from y_new
        obj_y2, grad_y2 = lse_obj_and_grad(y_new, T)

        # Backtracking line search
        x_old = x.copy()
        t_step = lr_curr
        improved = False
        for _ in range(12):
            x_try = y_new - t_step * grad_y2
            x_try = project_simplex(x_try)
            obj_try, _ = lse_obj_and_grad(x_try, T)
            if obj_try < obj_y2 - 1e-14:
                x = x_try
                improved = True
                break
            t_step *= 0.5

        if not improved:
            # Try smaller fixed step
            x_try = x_old - 0.01 * lr_curr * grad_y2
            x_try = project_simplex(x_try)
            obj_try, _ = lse_obj_and_grad(x_try, T)
            if obj_try < obj_y2:
                x = x_try
            else:
                # Perturb
                x = x_old + np.random.randn(n) * 0.002 / np.sqrt(n)
                x = project_simplex(x)

        y = x.copy()
        t_k = t_next

        # Check true objective
        obj_true = compute_obj(x)
        if obj_true < best_obj:
            best_obj = obj_true
            best_a = x.copy()
            best_it = it

        it += 1
        if it - best_it > patience:
            x = best_a.copy()
            y = best_a.copy()
            t_k = 1.0
            lr = lr * 0.5
            patience = max(100, patience // 2)
            best_it = it

        if verbose and it % 200 == 0:
            print(f"      Nesterov-LSE it{it}: obj={best_obj:.8f}, T={T:.1f}, "
                  f"t_step={t_step:.6e}, elapsed={time.monotonic()-start:.1f}s")

    return best_a


def subgradient_at_peaks(a, top_k=10):
    """
    Compute weighted subgradient at top-k peaks of the convolution.
    Returns (gradient, peak_indices).
    """
    n = len(a)
    conv = compute_conv(a)
    peak_indices = np.argsort(conv)[::-1][:top_k]
    peak_vals = conv[peak_indices]
    # Exponential weighting for concentration
    weights = np.exp(20 * (peak_vals - peak_vals[0]))
    weights = weights / weights.sum()

    grad = np.zeros(n)
    for ki, w in zip(peak_indices, weights):
        ki = int(ki)
        # d conv[ki] / d a[i] = 2 * a[ki - i] if 0 <= ki - i < n
        # So grad[i] += w * 2 * a[ki - i] for valid i
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            grad[lo:hi + 1] += w * 2.0 * a[ki - hi:ki - lo + 1][::-1]

    return 2.0 * n * grad, peak_indices[0]


def full_lp_polish(a, time_budget, verbose=False):
    """
    LP polishing using ALL convolution constraints.
    min M s.t. (a_norm * g)[k] <= M for all k, sum(g)=1, g>=0.
    """
    n = len(a)
    a_norm = np.maximum(a, 0)
    s = float(np.sum(a_norm))
    if s < 1e-10:
        return a
    a_norm = a_norm / s

    conv_a = compute_conv(a_norm)
    best_obj = 2.0 * n * float(np.max(conv_a))
    best_g = a_norm.copy()
    start = time.monotonic()
    it = 0

    # Build the full convolution matrix (sparse-friendly)
    # A[k, j] = a_norm[k-j] if 0 <= k-j < n else 0
    # Shape: (2n-1, n)
    n_conv = 2 * n - 1
    A = np.zeros((n_conv, n))
    for k in range(n_conv):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[k, lo:hi + 1] = a_norm[k - hi:k - lo + 1][::-1]

    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)

            while time.monotonic() - start < time_budget and it < 20:
                prob.solve(solver=solver)
                if g.value is None or prob.status not in ('optimal', 'optimal_inaccurate'):
                    it += 5
                    break

                g_val = np.maximum(g.value, 0)
                s_g = float(np.sum(g_val))
                if s_g < 1e-10:
                    it += 5
                    break
                g_val = g_val / s_g

                # Evaluate
                conv_new = compute_conv(g_val)
                max_new = float(np.max(conv_new))
                obj_new = 2.0 * n * max_new

                if obj_new < best_obj - 1e-12:
                    # Blend
                    for blend in [0.3, 0.5, 0.7, 1.0]:
                        a_try = (1 - blend) * a_norm + blend * g_val
                        a_try = np.maximum(a_try, 0)
                        s_try = float(np.sum(a_try))
                        if s_try > 0.01:
                            a_try = a_try / s_try
                            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
                            if obj_try < best_obj:
                                best_obj = obj_try
                                best_g = a_try.copy()
                                a_norm = a_try.copy()
                                # Rebuild A
                                for k in range(n_conv):
                                    lo = max(0, k - n + 1)
                                    hi = min(n - 1, k)
                                    A[k, :] = 0
                                    if lo <= hi:
                                        A[k, lo:hi + 1] = a_norm[k - hi:k - lo + 1][::-1]
                                if verbose:
                                    print(f"      FullLP it{it}: obj={best_obj:.8f}, "
                                          f"blend={blend}, elapsed={time.monotonic()-start:.1f}s")
                                break
                    it += 1
                else:
                    # No improvement, try perturbation
                    noise = np.random.randn(n) * 0.003
                    a_pert = a_norm + noise
                    a_pert = np.maximum(a_pert, 0)
                    a_pert = a_pert / a_pert.sum()
                    a_norm = a_pert
                    for k in range(n_conv):
                        lo = max(0, k - n + 1)
                        hi = min(n - 1, k)
                        A[k, :] = 0
                        if lo <= hi:
                            A[k, lo:hi + 1] = a_norm[k - hi:k - lo + 1][::-1]
                    it += 2

            break  # Success with this solver
        except Exception as e:
            if verbose:
                print(f"      LP solver {solver} failed: {e}")
            continue

    return best_g

def spectral_equalize(a, n_iters=10, alpha=0.3):
    """
    Spectral equalization: try to flatten |FFT(a)|^2.
    Only effective for symmetric sequences where conv = autocorrelation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    # Make symmetric
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0)
    a_sym = a_sym / a_sym.sum()

    for _ in range(n_iters):
        fa = fft(a_sym, n=2 * n)
        mag = np.abs(fa)
        # Target: constant magnitude
        target_mag = np.ones_like(mag)
        # Blend
        new_mag = (1 - alpha) * mag + alpha * target_mag
        # Preserve phase
        phase = fa / (mag + 1e-30)
        new_fa = new_mag * phase
        # Enforce Hermitian symmetry for real signal
        new_fa = 0.5 * (new_fa + np.conj(new_fa[np.remainder(-np.arange(len(new_fa)), len(new_fa))]))
        a_try = np.real(ifft(new_fa, n=2 * n))[:n]
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try > 0.01:
            a_try = a_try / s_try
            a_sym = a_try

    return a_sym


def symmetric_init(n, mode=0, sharpness=3.0):
    """Generate symmetric initializations."""
    half = n // 2
    x = np.linspace(0, 1, half + 1)
    if mode == 0:  # U-shaped
        a_half = 1.0 / (0.05 + sharpness * (2 * x - 1) ** 2)
    elif mode == 1:  # Power law
        a_half = x ** (-1.5) + 0.1
    elif mode == 2:  # Arcsine-like
        a_half = 1.0 / np.sqrt(np.maximum(0.01, 1 - (2 * x - 1) ** 2))
    elif mode == 3:  # Cosine trend
        a_half = 0.5 + 0.5 * np.cos(np.pi * x)
    else:  # Linear ramp
        a_half = x

    # Make full symmetric sequence
    if n % 2 == 0:
        a = np.concatenate([a_half[:half], a_half[:half][::-1]])
    else:
        a = np.concatenate([a_half[:half], a_half[half:half+1], a_half[:half][::-1]])

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def coordinate_descent(a, time_budget, max_sweeps=5, verbose=False):
    """Coordinate descent with random restarts within sweep."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for sweep in range(max_sweeps):
        if time.monotonic() - start >= time_budget:
            break
        sweep_improved = False
        # Random order for diversity
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start >= time_budget:
                break

            ai_old = a[i]

            # Try multiple candidate values
            candidates = [0.0, ai_old * 0.5, ai_old * 1.5]
            # Also try values relative to neighbors
            if i > 0:
                candidates.append(a[i - 1] * 0.8)
            if i < n - 1:
                candidates.append(a[i + 1] * 0.8)

            best_val = ai_old
            best_local = best_obj
            conv_cache = compute_conv(a)

            for cv in candidates:
                a[i] = cv
                s_new = float(np.sum(a))
                if s_new < 0.01:
                    continue
                a_try = a / s_new
                obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
                if obj_try < best_local:
                    best_local = obj_try
                    best_val = cv

            if best_val != ai_old:
                a[i] = best_val
                s_new = float(np.sum(a))
                if s_new > 0.01:
                    a = a / s_new
                    sweep_improved = True
            else:
                a[i] = ai_old

        obj_now = compute_obj(a)
        if obj_now < best_obj:
            best_obj = obj_now
            best_a = a.copy()

        if verbose:
            print(f"      CD sweep {sweep}: obj={best_obj:.8f}, "
                  f"elapsed={time.monotonic()-start:.1f}s")

        if not sweep_improved:
            break

    return best_a


def multi_start_optimize(a_init, time_budget, verbose=False):
    """
    Run a full optimization pipeline from a given initialization.
    """
    n = len(a_init)
    best_a = a_init.copy()
    best_obj = compute_obj(best_a)

    # Phase 1: Nesterov LSE GD
    t1 = min(30.0, time_budget * 0.2)
    if t1 > 5:
        a1 = nesterov_lse_gd(best_a, t1, T_start=5.0, T_end=40.0)
        obj1 = compute_obj(a1)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1
            if verbose:
                print(f"        Nesterov-LSE: {best_obj:.8f} **")

    # Phase 2: Truncated LP
    t2 = min(40.0, time_budget * 0.3)
    if t2 > 5:
        a2 = truncated_lp_polish(best_a, top_m=20, time_budget=t2)
        obj2 = compute_obj(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2
            if verbose:
                print(f"        TruncLP: {best_obj:.8f} **")

    # Phase 3: Coordinate descent
    t3 = min(20.0, time_budget * 0.2)
    if t3 > 3:
        a3 = coordinate_descent(best_a, t3, max_sweeps=2)
        obj3 = compute_obj(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3
            if verbose:
                print(f"        CD: {best_obj:.8f} **")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: multi-method optimization with spectral insights.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # === Load parent or initialize ===
    n_default = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.all(np.isfinite(parent_arr)):
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Loaded parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        print("Could not load parent, using initialization")
        best_seq = symmetric_init(n_default, mode=0, sharpness=3.0)
        best_score = compute_obj(best_seq)
        print(f"Symmetric init (n={n_default}): obj={best_score:.8f}")

    # === Try diverse initializations ===
    print("\n=== Trying diverse initializations ===")
    n_cur = len(best_seq)
    init_candidates = []

    for mode in range(4):
        for sharp in [2.0, 4.0, 6.0]:
            a_cand = symmetric_init(n_cur, mode=mode, sharpness=sharp)
            obj_cand = compute_obj(a_cand)
            init_candidates.append((obj_cand, a_cand, f"sym_m{mode}_s{sharp}"))
            if obj_cand < best_score:
                best_score = obj_cand
                best_seq = a_cand.copy()
                print(f"  ** sym_m{mode}_s{sharp}: {obj_cand:.8f}")

    # Non-symmetric inits
    x = np.linspace(0, 1, n_cur)
    for alpha in [1.5, 2.0, 2.5, 3.0]:
        a_cand = np.maximum(x ** (-alpha) + 0.01, 0)
        a_cand = a_cand / a_cand.sum()
        obj_cand = compute_obj(a_cand)
        if obj_cand < best_score:
            best_score = obj_cand
            best_seq = a_cand.copy()
            print(f"  ** power_{alpha}: {obj_cand:.8f}")

    # Spectral equalization of parent
    a_spect = spectral_equalize(best_seq, n_iters=15, alpha=0.4)
    obj_spect = compute_obj(a_spect)
    if obj_spect < best_score:
        best_score = obj_spect
        best_seq = a_spect.copy()
        print(f"  ** spectral_eq: {best_score:.8f}")
    else:
        print(f"  spectral_eq: {obj_spect:.8f}")

    print(f"\nBest initial: n={len(best_seq)}, obj={best_score:.8f}")

    # === Phase 1: Nesterov LSE GD (main gradient phase) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 1: Nesterov LSE GD (budget={min(120, remaining*0.15):.0f}s) ===")
        alloc = min(120.0, remaining * 0.15)
        a_lse = nesterov_lse_gd(best_seq, alloc, T_start=5.0, T_end=50.0, verbose=True)
        obj_lse = compute_obj(a_lse)
        if obj_lse < best_score:
            best_score = obj_lse
            best_seq = a_lse.copy()
            print(f"  ** After Nesterov: obj={best_score:.8f}")
        else:
            print(f"  After Nesterov: obj={obj_lse:.8f}")

    # === Phase 2: Full LP polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 2: Full LP Polish (budget={min(200, remaining*0.25):.0f}s) ===")
        alloc = min(200.0, remaining * 0.25)
        a_lp = full_lp_polish(best_seq, alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** After FullLP: obj={best_score:.8f}")
        else:
            print(f"  After FullLP: obj={obj_lp:.8f}")

    # === Phase 3: Nesterov round 2 (higher T) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 3: Nesterov Round 2 ===")
        alloc = min(60.0, remaining * 0.1)
        a_lse2 = nesterov_lse_gd(best_seq, alloc, T_start=20.0, T_end=100.0)
        obj_lse2 = compute_obj(a_lse2)
        if obj_lse2 < best_score:
            best_score = obj_lse2
            best_seq = a_lse2.copy()
            print(f"  ** Nesterov2: obj={best_score:.8f}")

    # === Phase 4: Truncated LP (focused) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 4: Truncated LP (focused) ===")
        alloc = min(100.0, remaining * 0.15)
        a_tl = truncated_lp_polish(best_seq, top_m=12, time_budget=alloc, verbose=True)
        obj_tl = compute_obj(a_tl)
        if obj_tl < best_score:
            best_score = obj_tl
            best_seq = a_tl.copy()
            print(f"  ** TruncLP: obj={best_score:.8f}")

    # === Phase 5: Coordinate descent ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 5: Coordinate Descent ===")
        alloc = min(80.0, remaining * 0.15)
        a_cd = coordinate_descent(best_seq, alloc, max_sweeps=3, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  ** CD: obj={best_score:.8f}")

    # === Phase 6: Full LP round 2 ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 6: Full LP Round 2 ===")
        alloc = min(150.0, remaining * 0.2)
        a_lp2 = full_lp_polish(best_seq, alloc, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  ** FullLP2: obj={best_score:.8f}")

    # === Phase 7: n exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 7: n Exploration ===")
        n_best = len(best_seq)
        n_options = sorted(set([
            n_best - 100, n_best + 100, n_best + 200,
            n_best - 50, n_best + 50,
        ]))
        n_options = [n for n in n_options if 400 <= n <= 2000 and abs(n - n_best) >= 30]
        n_options = n_options[:3]
        print(f"  Trying n: {n_options}")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            # Interpolate
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_seq)
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 0.01:
                continue
            a_try = a_try / s_try

            # Quick optimization
            t_opt = min(30.0, remaining * 0.4)
            a_try, obj_try = multi_start_optimize(a_try, t_opt, verbose=False)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f}")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 8: Final Polish ===")
        # Full LP
        alloc = min(remaining * 0.3, 60.0)
        if alloc > 10:
            a_fl = full_lp_polish(best_seq, alloc)
            obj_fl = compute_obj(a_fl)
            if obj_fl < best_score:
                best_score = obj_fl
                best_seq = a_fl.copy()
                print(f"  ** FinalLP: obj={best_score:.8f}")

        # CD
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.3, 40.0)
        if alloc > 5:
            a_fc = coordinate_descent(best_seq, alloc, max_sweeps=2)
            obj_fc = compute_obj(a_fc)
            if obj_fc < best_score:
                best_score = obj_fc
                best_seq = a_fc.copy()
                print(f"  ** FinalCD: obj={best_score:.8f}")

        # Nesterov with high T
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.3, 30.0)
        if alloc > 5:
            a_fn = nesterov_lse_gd(best_seq, alloc, T_start=50.0, T_end=200.0)
            obj_fn = compute_obj(a_fn)
            if obj_fn < best_score:
                best_score = obj_fn
                best_seq = a_fn.copy()
                print(f"  ** FinalNest: obj={best_score:.8f}")

    # === Phase 9: Diverse restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 9: Diverse Restarts ===")
        n_cur = len(best_seq)
        for r in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            mode = r % 4
            a_r = symmetric_init(n_cur, mode=mode, sharpness=3.0 + r * 2)
            # Blend with current best
            blend = 0.3
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

            t_arg = min(20.0, remaining * 0.3)
            a_r, obj_r = multi_start_optimize(a_r, t_arg, verbose=False)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  ** Restart {r} (mode={mode}): obj={best_score:.8f}")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Final coordinate descent ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_final = coordinate_descent(best_seq, min(remaining - 5, 15.0), max_sweeps=1)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  ** Last CD: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
