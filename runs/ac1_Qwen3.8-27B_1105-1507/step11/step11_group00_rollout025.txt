"""

Uses a combination of: (1) two-level binary sequence initialization, (2) mirror descent

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def gradient_from_p(a, p):
    """
    Compute gradient of sum_k p[k]*conv(a,a)_k with respect to a.
    Uses FFT for efficiency.
    """
    n = len(a)
    b = a[::-1]
    N_fft = 1
    while N_fft < 3 * n - 1:
        N_fft <<= 1
    c = irfft(rfft(b, N_fft) * rfft(p, N_fft), N_fft)
    return 2.0 * c[n - 1:2 * n - 1]


def smoothed_gradient(a, conv, tau):
    """Compute gradient of smoothed max (LSE approximation) of conv."""
    n = len(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    if shifted.max() > 500:
        shifted = shifted - shifted.max()
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.zeros(2 * n - 1)
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_vals / sum_exp
    return gradient_from_p(a, p)


def topk_subgradient(a, conv, k=3):
    """Subgradient using only the top-k peaks."""
    n = len(a)
    idx = np.argsort(conv)[-k:]
    p = np.zeros(2 * n - 1)
    p[idx] = 1.0 / k
    return gradient_from_p(a, p)


def mirrored_topk_gradient(a, conv, k=5, tau=100.0):
    """Blend of top-k and smoothed gradient for better direction."""
    g_top = topk_subgradient(a, conv, k=k)
    g_smooth = smoothed_gradient(a, conv, tau)
    return 0.4 * g_top + 0.6 * g_smooth


def mirror_descent_step(a, grad, eta):
    """Multiplicative weights update with normalization."""
    exponent = -eta * grad
    exponent = np.clip(exponent, -60, 60)
    a_new = a * np.exp(exponent)
    s = float(np.sum(a_new))
    if s < 1e-300:
        return a.copy()
    return a_new / s


def lp_direction_reduced(a, top_k=30, max_time=2.0):
    """
    Solve reduced LP: min M s.t. 2*conv(a,g)_k - conv(a,a)_k <= M for top-k lags,
    sum(g) = 1, g >= 0.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv_aa = compute_conv(a_norm)
    
    # Select top-k lags
    k_idx = np.argsort(conv_aa)[-top_k:]
    
    # Build reduced A matrix
    A = np.zeros((top_k, n))
    for row, k in enumerate(k_idx):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[row, j] = a_norm[k - j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    rhs = conv_aa[k_idx]
    constraints = [2.0 * (A @ g) - M <= rhs, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation on [0,1]."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def make_msequence(m, rng=None):
    """Generate an m-sequence of length 2^m - 1 as a binary array (0/1)."""
    length = (1 << m) - 1
    # Use a known LFSR tap for m up to 16
    taps = {
        3: 0b011, 4: 0b0011, 5: 0b00101, 6: 0b01011,
        7: 0b01011, 8: 0b01011, 9: 0b01011, 10: 0b01011,
        11: 0b10011, 12: 0b010011, 13: 0b0010011,
        14: 0b0100101, 15: 0b1001011, 16: 0b1001011
    }
    if m not in taps:
        # Fallback: use m=10
        m = 10
    length = (1 << m) - 1
    tap = taps.get(m, 0b1001011)
    
    # Generate LFSR sequence
    reg = (1 << m) - 1  # Non-zero initial state
    seq = []
    for _ in range(length):
        seq.append(reg & 1)
        feedback = 0
        for bit in range(m):
            if tap & (1 << bit):
                feedback ^= (reg >> bit) & 1
        reg = (reg >> 1) | (feedback << (m - 1))
    
    return np.array(seq, dtype=np.float64)

    return b


def generate_initializations(n, rng):
    """Generate diverse initial candidates for a given n."""
    candidates = []
    
    # 1. Uniform
    candidates.append(np.ones(n) / n)

    j = np.arange(n)
    cos_win = np.cos(np.pi * j / (2 * n))
    cos_win = np.maximum(cos_win, 0.01)
    candidates.append(cos_win / cos_win.sum())
    
    # 3. Exponential window
    exp_win = np.exp(-0.5 * j / n)
    candidates.append(exp_win / exp_win.sum())
    
    # 4. Two-level from m-sequence (truncated/padded to n)
    m = max(4, int(np.log2(n)) + 1)
    m = min(m, 16)
    mseq = make_msequence(m, rng)
    if len(mseq) >= n:
        b = mseq[:n]
    else:

        b = np.tile(mseq, tiles)[:n]
    for ratio in [0.3, 0.5, 0.7]:
        a = np.ones(n) * (1 - ratio) + np.ones(n) * ratio * b
        a = np.maximum(a, 0.01)
        candidates.append(a / a.sum())
    
    # 5. Two-level from random binary with spectral shaping
    b = (rng.random(n) > 0.5).astype(np.float64)
    for ratio in [0.3, 0.5, 0.7]:

        a = np.maximum(a, 0.01)
        candidates.append(a / a.sum())
    
    # 6. "Anti-triangular" (high at edges)
    center = n / 2.0
    anti_tri = 1.0 - np.abs(j - center) / center
    anti_tri = 1.0 / np.maximum(anti_tri, 0.1)
    candidates.append(anti_tri / anti_tri.sum())
    
    # 7. Sinusoidal modulation

def pairwise_exchange_search(a, time_budget=100.0, verbose=False, seed=123):

        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    no_improve = 0

    while time.monotonic() - start < time_budget:
        if it % 500 == 0:

            it += 1
            continue

                a_try2[i] = max(a_try2[i], 0)
                obj_try2 = compute_obj(a_try2)
                if obj_try2 < best_trial_obj:
                    best_trial_obj = obj_try2
                    best_d = -d  # Negative means reverse direction

        if best_d != 0 and best_trial_obj < best_obj - 1e-14:
            a_new = a.copy()
            if best_d > 0:
                a_new[i] += best_d
                a_new[m] -= best_d
            else:
                a_new[i] -= abs(best_d)
                a_new[m] += abs(best_d)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                best_obj = best_trial_obj
                best_a = a_new.copy()
                a = a_new
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Periodic: try larger structured moves
        if it % 200 == 100:
            # Move mass between a random block of positions
            block_size = max(1, n // 20)
            i_start = rng.integers(0, n - block_size)
            m_start = rng.integers(0, n - block_size)
            d_block = 0.05 / n
            a_try = a.copy()
            a_try[i_start:i_start+block_size] += d_block
            a_try[m_start:m_start+block_size] -= d_block
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    no_improve = 0

        it += 1

    return best_a


def lp_driven_optimize(a, time_budget=200.0, verbose=False, top_k=30):
    """LP-driven optimization with reduced constraints and line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99)

    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        # Solve reduced LP
        t_lp_start = time.monotonic()
        g = lp_direction_reduced(a, top_k=top_k, max_time=min(remaining * 0.3, 3.0))
        lp_time = time.monotonic() - t_lp_start

        if g is None:
            # Fallback: use gradient
            conv = compute_conv(a)
            g = topk_subgradient(a, conv, k=5)
            g = np.maximum(g, 0)
            s_g = float(np.sum(g))
            if s_g < 1e-10:
                g = np.ones(n) / n
            else:
                g = g / s_g

        # Line search
        best_t = 0.1
        best_obj_t = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
            if time.monotonic() - start > time_budget - 1:
                break
            a_try = np.maximum((1 - t) * a + t * g, 0)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj_t:
                    best_obj_t = obj_try
                    best_t = t

        if best_obj_t < best_obj - 1e-12:
            a_new = np.maximum((1 - best_t) * a + best_t * g, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                best_obj = best_obj_t
                best_a = a_new.copy()
                a = a_new
                stall = 0
                if verbose and it % 10 == 0:
                    print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.3f}, "
                          f"top_k={top_k}, time={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall >= 5:
                # Perturb and continue
                a = a * np.exp(0.02 * rng.standard_normal(n))
                a = np.maximum(a, 0)
                s_p = float(np.sum(a))
                if s_p > 1e-10:
                    a = a / s_p
                else:
                    a = best_a.copy()
                stall = 0
                top_k = min(50, top_k + 5)  # Increase constraints

        it += 1

    return best_a


def coordinate_descent_refine(a, time_budget=60.0, verbose=False):
    """
    Focused coordinate descent: for each position i, find the optimal value
    by scanning a small range around the current value.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(55)

    # Try adjusting individual coordinates
    it = 0
    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 0.5:
            break

        i = it % n
        # Try a few values for a[i]
        current = a[i]
        base_delta = 0.01 / n

        best_local = current
        best_local_obj = best_obj

        for scale in [0.5, 1.0, 2.0, 4.0]:
            for direction in [-1, 1]:
                delta = direction * base_delta * scale
                a_try = a.copy()
                a_try[i] = max(0, current + delta)
                # Compensate: reduce a[0] (or distribute)
                diff = a_try[i] - current
                if abs(diff) > 1e-15:
                    # Spread the difference across all other positions
                    if i == 0:
                        a_try[1:] -= diff / (n - 1)
                    else:
                        a_try[0] -= diff
                        if a_try[0] < 0:
                            # Redistribute
                            a_try = a.copy()
                            a_try[i] = max(0, current + delta)
                            excess = a_try[i] - current
                            others = np.arange(n) != i
                            a_try[others] -= excess / (n - 1)
                            a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try > 1e-10:
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_local_obj:
                            best_local_obj = obj_try
                            best_local = a_try[i]

        if best_local_obj < best_obj - 1e-14:
            a[i] = best_local
            s_new = float(np.sum(a))
            if s_new > 1e-10:
                a = a / s_new
                best_obj = best_local_obj
                best_a = a.copy()

        it += 1

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines multiple optimization strategies:
    1. Diverse initialization (binary sequences, windows, random)
    2. Mirror descent with adaptive gradients
    3. LP-driven direction finding (reduced constraints)
    4. Pairwise exchange local search
    5. Multi-n exploration
    6. Final polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # === Load parent if available ===
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                n_parent = len(parent)
                print(f"Parent (n={n_parent}): obj={best_score:.8f}")
    except Exception:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    # === Phase 0: Try diverse initializations ===
    print(f"\n=== Phase 0: Diverse Initializations (n={n_parent}) ===")
    inits = generate_initializations(n_parent, rng)
    for i, init in enumerate(inits):
        obj_i = compute_obj(init)
        if obj_i < best_score:
            best_score = obj_i
            best_seq = init.copy()
            print(f"  Init {i}: obj={obj_i:.8f} **")
    
    # Also try a few different n values for initialization
    for n_try in [600, 1000, 1200]:
        if n_try == n_parent:
            continue
        init = np.ones(n_try) / n_try
        obj_i = compute_obj(init)
        # Only keep if potentially competitive
        if obj_i < best_score + 0.1:
            print(f"  n={n_try} uniform: obj={obj_i:.8f}")

    # === Phase 1: Mirror Descent (main optimization) ===
    print(f"\n=== Phase 1: Mirror Descent (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 30:
        a_md = mirror_descent_optimize(best_seq, time_budget=p1_budget,
                                       tau_init=50.0, tau_end=2000.0,
                                       verbose=True, seed=seed)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Phase1 MD: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 MD: obj={obj_md:.8f} (no improvement)")

    # === Phase 2: LP-Driven ===
    print(f"\n=== Phase 2: LP-Driven ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 200.0)
    if p2_budget > 20:
        a_lp = lp_driven_optimize(best_seq, time_budget=p2_budget,
                                  verbose=True, top_k=25)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Pairwise Exchange Local Search ===
    print(f"\n=== Phase 3: Pairwise Exchange Search ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    if p3_budget > 20:
        a_pw = pairwise_exchange_search(best_seq, time_budget=p3_budget,
                                        verbose=True, seed=seed + 1)
        obj_pw = compute_obj(a_pw)
        if obj_pw < best_score:
            best_score = obj_pw
            best_seq = a_pw.copy()
            print(f"  Phase3 PW: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 PW: obj={obj_pw:.8f} (no improvement)")

    # === Phase 4: Second Mirror Descent (higher tau) ===
    print(f"\n=== Phase 4: Second MD Round ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.15, 150.0)
    if p4_budget > 20:
        a_md2 = mirror_descent_optimize(best_seq, time_budget=p4_budget,
                                        tau_init=200.0, tau_end=5000.0,
                                        verbose=True, seed=seed + 2)
        obj_md2 = compute_obj(a_md2)
        if obj_md2 < best_score:
            best_score = obj_md2
            best_seq = a_md2.copy()
            print(f"  Phase4 MD2: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 MD2: obj={obj_md2:.8f} (no improvement)")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = []
    for delta in [-200, -100, 100, 200, 400, 600, 800]:
        n_try = n_best + delta
        if 400 <= n_try <= 2000 and n_try not in n_options:
            n_options.append(n_try)
    for n_abs in [500, 700, 900, 1100, 1400, 1600]:
        if n_abs not in n_options:
            n_options.append(n_abs)
    n_options = sorted(set(n_options))[:6]
    print(f"  Trying n values: {n_options}")

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Mirror descent
        alloc = min(remaining * 0.3, 60.0)
        if alloc > 10:
            a_try = mirror_descent_optimize(a_try, time_budget=alloc,
                                            tau_init=50.0, tau_end=800.0,
                                            verbose=False, seed=seed + n_try)

        # LP
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.2, 40.0)
        if alloc > 8:
            a_try = lp_driven_optimize(a_try, time_budget=alloc,
                                       verbose=False, top_k=20)

        # Pairwise
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.15, 20.0)
        if alloc > 5:
            a_try = pairwise_exchange_search(a_try, time_budget=alloc,
                                             verbose=False, seed=seed + n_try + 10)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_seq)

    restart_strategies = [
        "mirror_sym", "fourier_perturb", "sparse_bump", "random_mix"
    ]
    for r, name in enumerate(restart_strategies):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if name == "mirror_sym":
            a_r = (best_seq + best_seq[::-1]) / 2.0
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r
            else:
                a_r = best_seq.copy()
        elif name == "fourier_perturb":
            pert = np.zeros(n_cur)
            for k in range(1, 8):
                freq = k * 2.0 * np.pi / n_cur
                amp = 0.05 * rng.uniform(0.3, 1.0) / k
                phase = rng.uniform(0, 2.0 * np.pi)
                pert += amp * np.cos(freq * np.arange(n_cur) + phase)
            a_r = best_seq * np.exp(pert)
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r
        elif name == "sparse_bump":
            a_r = best_seq.copy()
            # Add a small bump at a random position
            pos = rng.integers(0, n_cur)
            width = max(3, n_cur // 50)
            bump = np.zeros(n_cur)
            bump[max(0, pos - width):min(n_cur, pos + width)] = 0.1
            a_r = a_r + bump
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r
        else:
            a_r = 0.85 * best_seq + 0.15 * (rng.uniform(0.5, 1.5, n_cur) / n_cur)
            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r

        # Optimize the restart
        alloc = min(remaining * 0.25, 50.0)
        if alloc > 10:
            a_r = mirror_descent_optimize(a_r, time_budget=alloc * 0.6,
                                          tau_init=50.0, tau_end=500.0,
                                          verbose=False, seed=seed + 100 + r)
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.15, 25.0)
        if alloc > 5:
            a_r = pairwise_exchange_search(a_r, time_budget=alloc,
                                           verbose=False, seed=seed + 200 + r)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart ({name}): obj={best_score:.8f} **")
        else:
            print(f"  Restart ({name}): obj={obj_r:.8f}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-temperature MD
    if remaining > 30:
        alloc = min(remaining * 0.35, 120.0)
        a_final = mirror_descent_optimize(best_seq, time_budget=alloc,
                                          tau_init=300.0, tau_end=8000.0,
                                          verbose=True, seed=seed + 300)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.3, 80.0)
        a_final2 = lp_driven_optimize(best_seq, time_budget=alloc,
                                      verbose=True, top_k=30)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final pairwise
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 60.0)
        a_final3 = pairwise_exchange_search(best_seq, time_budget=alloc,
                                            verbose=True, seed=seed + 400)
        obj_final3 = compute_obj(a_final3)
        if obj_final3 < best_score:
            best_score = obj_final3
            best_seq = a_final3.copy()
            print(f"  Final PW: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
