"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a novel bilinear game (von Neumann minimax) approach: alternates between
solving a convex QP (min a^T M_p a over the simplex, where M_p is a Hankel
matrix built from distribution p over lags) via FISTA, and updating p via
multiplicative weights. Exploits Hankel structure for O(n log n) mat-vec via
FFT. Combined with mirror descent polishing, adaptive restarts, and multi-n
exploration for escaping local minima.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def hansel_mv_fft(p, a):
    """
    Compute M_p @ a where M_p[i,j] = p[i+j] (Hankel matrix).
    Uses FFT: result[i] = conv(p[::-1], a)[2n-2-i].
    p has length 2n-1, a has length n.
    """
    n = len(a)
    p_rev = p[::-1]
    size = 1
    while size < len(p_rev) + n - 1:
        size <<= 1
    P = rfft(p_rev, size)
    A = rfft(a, size)
    conv = irfft(P * A, size)
    return conv[n - 1:2 * n - 1][::-1]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def fista_qp(p, a_init, n_inner=30, n=None):
    """
    Solve min a^T M_p a s.t. a in simplex using FISTA.
    M_p is the Hankel matrix with M_p[i,j] = p[i+j].
    """
    if n is None:
        n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    t_k = 1.0
    # Step size: L = 2 * lambda_max(M_p) <= 2 (conservative)
    # Use 1.0 as step size (safe since lambda_max <= 1)
    step = 1.0

    for _ in range(n_inner):
        # Gradient: 2 * M_p @ y
        grad = 2.0 * hansel_mv_fft(p, y)
        # Gradient step
        a_new = y - step * grad
        # Project onto simplex
        a_new = project_simplex(a_new)
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = project_simplex(y)
        a = a_new
        t_k = t_new

    return a


def multiplicative_weights_update(p, c, beta):
    """Update p via multiplicative weights: p_new ∝ p * exp(beta * c)."""
    c_shifted = c - np.max(c)
    weights = np.exp(beta * c_shifted)
    p_new = p * weights
    s = float(np.sum(p_new))
    if s < 1e-300:
        p_new = np.full(len(c), 1.0 / len(c))
    else:
        p_new = p_new / s
    return p_new


def bilinear_game_optimize(a, time_budget=600.0, verbose=True):

def mirror_descent_optimize(a, time_budget=150.0, tau_init=50.0, tau_end=2000.0,

        a = a_new

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
